     1→// =============================================================================

     3→// =============================================================================
     4→// This module replaces the old 12-node graph and heuristic probability system.
     5→// The decision graph IS the primary probability engine — all 9 scenario
     6→// probabilities (R1–R9) are derived from aggregating path probabilities through
     7→// this directed acyclic graph (DAG).

    10→// === Helper Functions =========================================================
    11→
    12→function clamp(v: number, min: number, max: number): number {
    13→  return Math.max(min, Math.min(max, v));
    14→}
    15→
    16→function normalize(arr: number[]): number[] {
    17→  const sum = arr.reduce((a, b) => a + b, 0);
    18→  if (sum === 0) return arr.map(() => 1 / arr.length);
    19→  return arr.map((v) => v / sum);
    20→}
    21→
    22→function sigmoid(x: number): number {
    23→  return 1 / (1 + Math.exp(-x));
    24→}

   988→// === Probability Context Builder ==============================================
   989→
   990→function buildContext(input: GraphInput): ProbContext {
   991→  const adxNorm = clamp(input.adx / 60, 0, 1);
   992→  const srProximity = clamp(
   993→    1 - Math.min(input.distToR1, input.distToS1) * 10,
   994→    0,
   995→    1
   996→  );
   997→  const oscillatorExtreme = clamp(
   998→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
   999→    0,
  1000→    1
  1001→  );
  1002→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1003→
  1004→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1005→  const overboughtRisk =
  1006→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1007→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1008→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1009→
  1010→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1011→  const oversoldBounce =
  1012→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1013→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1014→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1015→
  1016→  return {
  1017→    input,
  1018→    adxNorm,
  1019→    srProximity,
  1020→    oscillatorExtreme,
  1021→    divergenceProxy,
  1022→    overboughtRisk,
  1023→    oversoldBounce,
  1024→  };
  1025→}

  1027→// === Node Probability Computation =============================================
  1028→// For each non-terminal node, returns an array of probabilities for its
  1029→// outgoing edges. The order MUST match the order of edges in createEdges().
  1030→// =============================================================================
  1031→
  1032→function computeNodeProbabilities(
  1033→  nodeId: string,
  1034→  ctx: ProbContext,
  1035→  _numChildren: number
  1036→): number[] {
  1037→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1038→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1039→

  1042→    case 'ROOT': {

  1053→      const flat =
  1054→        (1 - Math.abs(input.bullConsensus - 0.5) * 2) * (1 - adxNorm * 0.7);

  1065→              : 0),
  1066→          0,
  1067→          1

  1291→    case 'N_R_B_SR': {

  1300→    // Mirror of N_R_BULL
  1301→    case 'N_R_BEAR': {

  1328→    // Resistance rejection — explicit small R5

  1338→    // ── N_R_NONE → [R5] ──────────────────────────────────────────────────

  1341→    }
  1342→
  1343→    // ── Terminal nodes (R1–R9) have no outgoing edges ─────────────────────
  1344→    default:
  1345→      return [];
  1346→  }
  1347→}

  1349→// === Graph Traversal ==========================================================
  1350→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1351→// =============================================================================
  1352→
  1353→function traverseGraph(
  1354→  nodeId: string,
  1355→  pathProb: number,
  1356→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1357→  ctx: ProbContext,
  1358→  adj: Map<string, number[]>,
  1359→  edges: GraphEdge[],
  1360→  edgeProbabilities: Record<number, number>,
  1361→  nodeAccum: Record<string, number>,
  1362→  scenarios: Record<string, number>,
  1363→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1364→): void {
  1365→  // Accumulate path probability at this node
  1366→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1367→
  1368→  const outEdgeIndices = adj.get(nodeId);
  1369→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1370→
  1371→  // Compute outgoing edge probabilities for this node
  1372→  const probs = computeNodeProbabilities(
  1373→    nodeId,
  1374→    ctx,
  1375→    outEdgeIndices.length
  1376→  );
  1377→
  1378→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1379→    const edgeIdx = outEdgeIndices[i];
  1380→    const edge = edges[edgeIdx];
  1381→    const prob = probs[i] || 0;
  1382→
  1383→    // Store edge probability
  1384→    edgeProbabilities[edgeIdx] = prob;
  1385→
  1386→    // Determine branch for child nodes
  1387→    let childBranch = currentBranch;
  1388→    if (nodeId === 'ROOT') {
  1389→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1390→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1391→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1392→    }
  1393→
  1394→    const childPathProb = pathProb * prob;
  1395→
  1396→    // Check if target is a terminal node
  1397→    const isTerminal =
  1398→      edge.to.length === 2 &&
  1399→      edge.to[0] === 'R' &&
  1400→      edge.to[1] >= '1' &&
  1401→      edge.to[1] <= '9';
  1402→
  1403→    if (isTerminal) {
  1404→      // Accumulate scenario probability
  1405→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1406→      // Accumulate branch contribution
  1407→      if (childBranch) {
  1408→        contributions[edge.to][childBranch] += childPathProb;
  1409→      }
  1410→    }
  1411→
  1412→    // Recurse into child
  1413→    traverseGraph(
  1414→      edge.to,
  1415→      childPathProb,
  1416→      childBranch,
  1417→      ctx,
  1418→      adj,
  1419→      edges,
  1420→      edgeProbabilities,
  1421→      nodeAccum,
  1422→      scenarios,
  1423→      contributions
  1424→    );
  1425→  }
  1426→}

  1428→// === Main Export: buildDecisionGraph ===========================================
  1429→
  1430→export function buildDecisionGraph(input: GraphInput): GraphData {
  1431→  // 1. Build static graph structure
  1432→  const nodes = createNodes();
  1433→  const edges = createEdges();
  1434→  const nodePositions = createPositions();
  1435→
  1436→  // 2. Build adjacency map: nodeId → array of edge indices
  1437→  const adj = new Map<string, number[]>();
  1438→  for (let i = 0; i < edges.length; i++) {
  1439→    const from = edges[i].from;
  1440→    if (!adj.has(from)) adj.set(from, []);
  1441→    adj.get(from)!.push(i);
  1442→  }
  1443→
  1444→  // 3. Build probability context
  1445→  const ctx = buildContext(input);
  1446→
  1447→  // 4. Initialize accumulation structures
  1448→  const edgeProbabilities: Record<number, number> = {};
  1449→  const nodeAccum: Record<string, number> = {};
  1450→  const scenarios: Record<string, number> = {};
  1451→  const contributions: Record<
  1452→    string,
  1453→    { trend: number; breakout: number; reversal: number }
  1454→  > = {};
  1455→
  1456→  for (let i = 1; i <= 9; i++) {
  1457→    const key = `R${i}`;
  1458→    scenarios[key] = 0;
  1459→    contributions[key] = { trend: 0, breakout: 0, reversal: 0 };
  1460→  }
  1461→
  1462→  // 5. Traverse graph from ROOT
  1463→  traverseGraph(
  1464→    'ROOT',
  1465→    1.0,
  1466→    null,
  1467→    ctx,
  1468→    adj,
  1469→    edges,
  1470→    edgeProbabilities,
  1471→    nodeAccum,
  1472→    scenarios,
  1473→    contributions
  1474→  );
  1475→
  1476→  // 6. Post-process scenario probabilities
  1477→  // Convert to percentage, clamp [2, 35], re-normalize to sum=100
  1478→  const rawPcts: Record<string, number> = {};
  1479→  for (let i = 1; i <= 9; i++) {
  1480→    const key = `R${i}`;
  1481→    rawPcts[key] = Math.round((scenarios[key] || 0) * 100);
  1482→  }
  1483→
  1484→  // Clamp each to [2, 35]
  1485→  const clamped = Object.entries(rawPcts).map(([k, v]) => ({
  1486→    key: k,
  1487→    value: clamp(v, 2, 35),
  1488→  }));
  1489→
  1490→  // Re-normalize to sum=100
  1491→  const clampedSum = clamped.reduce((s, c) => s + c.value, 0);
  1492→  const finalPcts: Record<string, number> = {};
  1493→  if (clampedSum > 0) {
  1494→    // Distribute proportionally and round, then fix rounding errors
  1495→    let runningTotal = 0;
  1496→    const entries = clamped
  1497→      .map((c) => {
  1498→        const raw = (c.value / clampedSum) * 100;
  1499→        return { key: c.key, raw };
  1500→      })
  1501→      .sort((a, b) => b.raw - a.raw);
  1502→
  1503→    for (let i = 0; i < entries.length; i++) {
  1504→      const isLast = i === entries.length - 1;
  1505→      if (isLast) {
  1506→        finalPcts[entries[i].key] = Math.round(100 - runningTotal);
  1507→      } else {
  1508→        const rounded = Math.round(entries[i].raw);
  1509→        finalPcts[entries[i].key] = rounded;
  1510→        runningTotal += rounded;
  1511→      }
  1512→    }
  1513→  }
  1514→
  1515→  // Ensure no negative values due to rounding
  1516→  for (const key of Object.keys(finalPcts)) {
  1517→    if (finalPcts[key] < 0) finalPcts[key] = 0;
  1518→  }
  1519→
  1520→  // 7. Compute branch probabilities (from ROOT outgoing edges)
  1521→  const rootEdgeIndices = adj.get('ROOT') || [];
  1522→  const rootProbs = computeNodeProbabilities('ROOT', ctx, rootEdgeIndices.length);
  1523→  const branchProbabilities = {
  1524→    trend: rootProbs[0] || 0,
  1525→    breakout: rootProbs[1] || 0,
  1526→    reversal: rootProbs[2] || 0,
  1527→  };
  1528→
  1529→  // 8. Build node display values (path probability as percentage)
  1530→  const nodeValues: Record<string, string> = {};
  1531→  for (const [nodeId, accum] of Object.entries(nodeAccum)) {
  1532→    nodeValues[nodeId] = `${Math.round(accum * 100)}%`;
  1533→  }
  1534→
  1535→  // Override terminal node values with final post-processed percentages
  1536→  for (const [key, pct] of Object.entries(finalPcts)) {
  1537→    nodeValues[key] = `${pct}%`;
  1538→  }
  1539→
  1540→  // 9. Build path contributions (as percentages)
  1541→  const pathContributions: Record<
  1542→    string,
  1543→    { trend: number; breakout: number; reversal: number }
  1544→  > = {};
  1545→  for (let i = 1; i <= 9; i++) {
  1546→    const key = `R${i}`;
  1547→    const c = contributions[key];
  1548→    const total = c.trend + c.breakout + c.reversal;
  1549→    if (total > 0) {
  1550→      pathContributions[key] = {
  1551→        trend: Math.round((c.trend / total) * 100),
  1552→        breakout: Math.round((c.breakout / total) * 100),
  1553→        reversal: Math.round((c.reversal / total) * 100),
  1554→      };
  1555→    } else {
  1556→      pathContributions[key] = { trend: 33, breakout: 34, reversal: 33 };
  1557→    }
  1558→  }
  1559→
  1560→  // 10. Return complete GraphData
  1561→  return {
  1562→    nodes,
  1563→    edges,
  1564→    nodePositions,
  1565→    edgeProbabilities,
  1566→    nodeValues,
  1567→    branchProbabilities,
  1568→    scenarioProbabilities: finalPcts,
  1569→    pathContributions,
  1570→  };
  1571→}
  1572→
  1573→// === 30-Day Probability Trend Calculation ======================================
  1574→
  1575→// Dual exponential decay for forecast weighting
  1576→// W_forecast_h = α * exp(-h/τ₁) + (1-α) * exp(-h/τ₂)
  1577→export function dualDecay(h: number, alpha = 0.7, tauShort = 5, tauLong = 20): number {
  1578→  return alpha * Math.exp(-h / tauShort) + (1 - alpha) * Math.exp(-h / tauLong);
  1579→}
  1580→
  1581→// Normalize decay factors so they sum to 1
  1582→export function normalizedDecayFactors(horizon: number): number[] {
  1583→  const factors: number[] = [];
  1584→  for (let h = 1; h <= horizon; h++) {
  1585→    factors.push(dualDecay(h));
  1586→  }
  1587→  const total = factors.reduce((a, b) => a + b, 0);
  1588→  return factors.map(f => f / total);
  1589→}

  1622→export function calculateProbabilityTrend(
  1623→  scenarioProbabilities: Array<{key: string; name: string; nameEn: string; probability: number; isBullish?: boolean; isBearish?: boolean; isNeutral?: boolean}>,
  1624→  horizon: number = 30
  1625→): ProbabilityTrendResult {
  1626→  const decay = normalizedDecayFactors(horizon);
  1627→  
  1628→  const scenarios: ScenarioTrend[] = scenarioProbabilities.map(sp => {
  1629→    const trend: DayProbability[] = [];
  1630→    let cumProb = 0;
  1631→    let peakDay = 1;
  1632→    let peakProb = 0;
  1633→    
  1634→    for (let h = 1; h <= horizon; h++) {
  1635→      const individualProb = sp.probability * decay[h - 1];
  1636→      cumProb += individualProb;
  1637→      trend.push({ day: h, individualProb: Math.round(individualProb * 10000) / 10000, cumulativeProb: Math.round(cumProb * 10000) / 10000 });
  1638→      if (individualProb > peakProb) { peakProb = individualProb; peakDay = h; }
  1639→    }
  1640→    
  1641→    // Determine trend direction
  1642→    const firstThird = trend.slice(0, Math.floor(horizon / 3));
  1643→    const lastThird = trend.slice(Math.floor(horizon * 2 / 3));
  1644→    const avgFirst = firstThird.reduce((a, d) => a + d.individualProb, 0) / firstThird.length;
  1645→    const avgLast = lastThird.reduce((a, d) => a + d.individualProb, 0) / lastThird.length;
  1646→    const direction: ScenarioTrend['trendDirection'] = avgLast > avgFirst * 1.05 ? 'صعودی' : avgLast < avgFirst * 0.95 ? 'نزولی' : 'پایدار';
  1647→    
  1648→    return { name: sp.name, nameEn: sp.nameEn, key: sp.key, currentProbability: sp.probability, trend, trendDirection: direction, peakDay, peakProbability: Math.round(peakProb * 10000) / 10000, interpretation: '' };
  1649→  });
  1650→  
  1651→  // Group trends (bullish, bearish, neutral)
  1652→  const groupTrends: CumulativeGroupTrend = { bullish: [], bearish: [], neutral: [] };
  1653→  for (let h = 0; h < horizon; h++) {
  1654→    const bullDay = { day: h + 1, individualProb: 0, cumulativeProb: 0 };
  1655→    const bearDay = { day: h + 1, individualProb: 0, cumulativeProb: 0 };
  1656→    const neutralDay = { day: h + 1, individualProb: 0, cumulativeProb: 0 };
  1657→    
  1658→    scenarioProbabilities.forEach(sp => {
  1659→      const dayData = scenarios.find(s => s.key === sp.key)?.trend[h];
  1660→      if (!dayData) return;
  1661→      if (sp.isBullish) { bullDay.individualProb += dayData.individualProb; }
  1662→      else if (sp.isBearish) { bearDay.individualProb += dayData.individualProb; }
  1663→      else { neutralDay.individualProb += dayData.individualProb; }
  1664→    });
  1665→    
  1666→    bullDay.cumulativeProb = Math.round(bullDay.individualProb * 10000) / 10000;
  1667→    bearDay.cumulativeProb = Math.round(bearDay.individualProb * 10000) / 10000;
  1668→    neutralDay.cumulativeProb = Math.round(neutralDay.individualProb * 10000) / 10000;
  1669→    
  1670→    groupTrends.bullish.push({ ...bullDay, individualProb: Math.round(bullDay.individualProb * 10000) / 10000 });
  1671→    groupTrends.bearish.push({ ...bearDay, individualProb: Math.round(bearDay.individualProb * 10000) / 10000 });
  1672→    groupTrends.neutral.push({ ...neutralDay, individualProb: Math.round(neutralDay.individualProb * 10000) / 10000 });
  1673→  }
  1674→  
  1675→  // Generate key observations and actionable insights
  1676→  const topScenario = [...scenarios].sort((a, b) => b.currentProbability - a.currentProbability)[0];
  1677→  const keyObservations = [
  1678→    `سناریوی غالب: ${topScenario?.name || 'N/A'} (${topScenario ? Math.round(topScenario.currentProbability * 100) : 0}%)`,
  1679→    `اوج احتمال ${topScenario?.name || ''} در روز ${topScenario?.peakDay || 'N/A'}`,
  1680→  ];
  1681→  
  1682→  const actionableInsights = [
  1683→    topScenario?.isBullish ? 'روند صعودی غالب - بررسی نقاط ورود' : topScenario?.isBearish ? 'روند نزولی غالب - احتیاط در معاملات' : 'بازار در حالت رنج - انتظار برای شکست',
  1684→  ];
  1685→  
  1686→  return { scenarios, groupTrends, keyObservations, actionableInsights };
  1687→}
