   347→/**
   348→ * Parse a number that might contain Persian digits, commas, or other formatting.
   349→ * Handles null/undefined/empty strings gracefully.
   350→ */
   351→function parsePersianNum(str: string | null | undefined | number): number {
   352→  if (str === null || str === undefined) return 0;
   353→  const s = String(str);
   354→  if (!s || s.trim() === '-' || s.trim() === '') return 0;
   355→  return Number(s.replace(/[,۰-۹]/g, (c: string) => {
   356→    const persianDigits = '۰۱۲۳۴۵۶۷۸۹';
   357→    const idx = persianDigits.indexOf(c);
   358→    return idx >= 0 ? String(idx) : c === ',' ? '' : c;
   359→  }));
   360→}

   362→/**
   363→ * Check if a value looks like a date string (contains / or - and is not a number).
   364→ */
   365→function looksLikeDate(val: string | null | undefined | number): boolean {
   366→  if (!val) return false;
   367→  const s = String(val).trim();
   368→  return /[\/\-]/.test(s) && isNaN(Number(s.replace(/[,۰-۹\/\-]/g, '')));
   369→}
   370→
   371→/**
   372→ * Check if a value looks like a percentage (ends with % or is small like 0.5 to 50).
   373→ */
   374→function looksLikePercentage(val: string | null | undefined | number): boolean {
   375→  if (!val) return false;
   376→  const s = String(val).trim();
   377→  if (s.endsWith('%')) return true;
   378→  const n = parsePersianNum(val);
   379→  return !isNaN(n) && Math.abs(n) < 100 && n !== 0;
   380→}

   382→/**
   383→ * Auto-detect column mapping for TGJU API response.
   384→ * Expected format: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   385→ * But the API may change column order.
   386→ */
   387→function detectColumnMap(row: (string | number)[]): {
   388→  openIdx: number; highIdx: number; lowIdx: number; closeIdx: number;
   389→  jalaliDateIdx: number; gregorianDateIdx: number;
   390→} | null {
   391→  if (!row || row.length < 6) return null;
   392→
   393→  let jalaliDateIdx = -1;
   394→  let gregorianDateIdx = -1;
   395→  const numericIndices: number[] = [];
   396→  const percentageIndices: number[] = [];
   397→
   398→  for (let i = 0; i < row.length; i++) {
   399→    const val = row[i];
   400→    if (looksLikeDate(val)) {
   401→      // Jalali dates have Persian digits or format like 1404/05/25
   402→      const s = String(val);
   403→      if (/[۰-۹]/.test(s) || /^14\d{2}/.test(s.replace(/[۰-۹]/g, (d: string) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d))))) {
   404→        if (jalaliDateIdx === -1) jalaliDateIdx = i;
   405→      } else {
   406→        if (gregorianDateIdx === -1) gregorianDateIdx = i;
   407→      }
   408→    } else if (looksLikePercentage(val)) {
   409→      percentageIndices.push(i);
   410→    } else {
   411→      const n = parsePersianNum(val);
   412→      if (isFinite(n) && n !== 0) {
   413→        numericIndices.push(i);
   414→      }
   415→    }
   416→  }
   417→
   418→  // If we couldn't auto-detect dates, fall back to assumed positions
   419→  if (jalaliDateIdx === -1 && row.length >= 8) {
   420→    // Default assumption: row[6] = gregorian, row[7] = jalali
   421→    if (looksLikeDate(row[7])) jalaliDateIdx = 7;
   422→    else if (looksLikeDate(row[6])) jalaliDateIdx = 6;
   423→    if (gregorianDateIdx === -1 && looksLikeDate(row[6]) && jalaliDateIdx !== 6) gregorianDateIdx = 6;
   424→    else if (gregorianDateIdx === -1 && looksLikeDate(row[7]) && jalaliDateIdx !== 7) gregorianDateIdx = 7;
   425→  }
   426→
   427→  // Numeric columns (excluding percentages) should be OHLC + change
   428→  // OHLC: typically 4 largest absolute values among non-percentage numerics
   429→  // change: the remaining one
   430→  const ohlcCandidates = numericIndices.filter(i => !percentageIndices.includes(i));
   431→
   432→  if (ohlcCandidates.length < 4) {
   433→    // Fallback: assume first 4 numeric columns are OHLC
   434→    if (numericIndices.length >= 4) {
   435→      return {
   436→        openIdx: numericIndices[0],
   437→        highIdx: numericIndices[2],
   438→        lowIdx: numericIndices[1],
   439→        closeIdx: numericIndices[3],
   440→        jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : 7,
   441→        gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : 6,
   442→      };
   443→    }
   444→    return null;
   445→  }
   446→
   447→  // Sort by absolute value descending to identify OHLC (largest) vs change (smallest)
   448→  const sorted = ohlcCandidates
   449→    .map(i => ({ idx: i, val: Math.abs(parsePersianNum(row[i])) }))
   450→    .sort((a, b) => b.val - a.val);
   451→
   452→  // The 4 largest should be OHLC. Among them, identify high (max) and low (min).
   453→  // open and close are the other two.
   454→  const top4 = sorted.slice(0, 4);
   455→  const maxIdx = top4.reduce((a, b) => a.val > b.val ? a : b).idx;
   456→  const minIdx = top4.reduce((a, b) => a.val < b.val ? a : b).idx;
   457→  const others = top4.filter(x => x.idx !== maxIdx && x.idx !== minIdx);
   458→
   459→  // Among the two remaining, the one closer to high is open, the other is close
   460→  // (heuristic: in an uptrend open < close, but we can't assume direction)
   461→  // Default: first = open, second = close
   462→  const openIdx = others[0]?.idx ?? ohlcCandidates[0];
   463→  const closeIdx = others[1]?.idx ?? ohlcCandidates[3];
   464→
   465→  return {
   466→    openIdx,
   467→    highIdx: maxIdx,
   468→    lowIdx: minIdx,
   469→    closeIdx,
   470→    jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : row.length - 1,
   471→    gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : row.length - 2,
   472→  };
   473→}

   475→/**
   476→ * Validate a candle's OHLC values for sanity.
   477→ * Returns true if the candle is valid.
   478→ */
   479→function isValidCandle(o: number, h: number, l: number, c: number): boolean {
   480→  if (!isFinite(o) || !isFinite(h) || !isFinite(l) || !isFinite(c)) return false;
   481→  if (o <= 0 || h <= 0 || l <= 0 || c <= 0) return false;
   482→  // High must be >= Low, Open, Close
   483→  if (h < l || h < o || h < c) return false;
   484→  // Low must be <= High, Open, Close
   485→  if (l > o || l > c) return false;
   486→  // No infinite or extreme values
   487→  if (o > 1e15 || h > 1e15 || l > 1e15 || c > 1e15) return false;
   488→  return true;
   489→}

   503→/* ─── Fetch live instruments (currency + gold from tgju-api) + static list ── */
   504→
   505→export async function fetchTgjuInstruments(): Promise<TgjuInstrument[]> {
   506→  const now = Date.now();
   507→  if (liveInstrumentsCache.data && now - liveInstrumentsCache.ts < INSTRUMENTS_TTL) {
   508→    return liveInstrumentsCache.data;
   509→  }
   510→
   511→  const controller = new AbortController();
   512→  const timeout = setTimeout(() => controller.abort(), 10000);
   513→
   514→  try {
   515→    const [currencyRes, goldRes] = await Promise.all([
   516→      fetch(`${TGJU_API_BASE}/api/price/currency`, { signal: controller.signal }),
   517→      fetch(`${TGJU_API_BASE}/api/price/gold`, { signal: controller.signal }),
   518→    ]);
   519→
   520→    clearTimeout(timeout);
   521→
   522→    const instruments: TgjuInstrument[] = [];
   523→
   524→    if (currencyRes.ok) {
   525→      const currencies: TgjuPriceItem[] = await currencyRes.json();
   526→      for (const c of currencies) {
   527→        const price = parsePersianNum(c.price);
   528→        const low = parsePersianNum(c.low_price);
   529→        const high = parsePersianNum(c.high_price);
   530→        instruments.push({
   531→          title: c.title,
   532→          key: c.key,
   533→          price,
   534→          highPrice: high,
   535→          lowPrice: low,
   536→          change: 0,
   537→          changePercent: 0,
   538→          category: 'currency',
   539→        });
   540→      }
   541→    }
   542→
   543→    if (goldRes.ok) {
   544→      const goldCategories: TgjuGoldCategory[] = await goldRes.json();
   545→      for (const cat of goldCategories) {
   546→        for (const item of cat.prices) {
   547→          const price = parsePersianNum(item.price);
   548→          const low = parsePersianNum(item.low_price);
   549→          const high = parsePersianNum(item.high_price);
   550→
   551→          let category: TgjuCategoryType = 'gold';
   552→          if (item.key.startsWith('ime_fund_')) category = 'gold_etf';
   553→          else if (item.key.startsWith('silver_')) category = 'silver';
   554→
   555→          instruments.push({
   556→            title: item.title,
   557→            key: item.key,
   558→            price,
   559→            highPrice: high,
   560→            lowPrice: low,
   561→            change: 0,
   562→            changePercent: 0,
   563→            category,
   564→            groupTitle: cat.title,
   565→          });
   566→        }
   567→      }
   568→    }
   569→
   570→    // Add static instruments (crypto, world indices, forex, energy, metals, commodities)
   571→    for (const s of STATIC_INSTRUMENTS) {
   572→      const cached = staticPriceCache.get(s.key);
   573→      const price = cached?.close ?? 0;
   574→      instruments.push({
   575→        title: s.title,
   576→        key: s.key,
   577→        price,
   578→        highPrice: cached?.high ?? 0,
   579→        lowPrice: cached?.low ?? 0,
   580→        change: cached ? (cached.close - cached.prevClose) : 0,
   581→        changePercent: cached && cached.prevClose > 0 ? ((cached.close - cached.prevClose) / cached.prevClose) * 100 : 0,
   582→        category: s.category,
   583→        groupTitle: s.groupTitle,
   584→      });
   585→    }
   586→
   587→    liveInstrumentsCache.data = instruments;
   588→    liveInstrumentsCache.ts = now;
   589→
   590→    // ── Background: fetch chart data for currency/gold to get accurate daily change ──
   591→    backgroundFetchCurrencyGoldChange(instruments).catch(() => {});
   592→
   593→    // ── Background: fetch Yahoo prices for static instruments with price=0 (TGJU API 403) ──
   594→    backgroundFetchYahooFallbackPrices(instruments).catch(() => {});
   595→
   596→    return instruments;
   597→  } catch {
   598→    clearTimeout(timeout);
   599→    return liveInstrumentsCache.data || [];
   600→  }
   601→}

   603→/**
   604→ * Background task: fetch last 2 candles for currency/gold items to compute accurate daily change.
   605→ * Uses direct fetch to avoid queue contention.
   606→ * Only updates change/changePercent — keeps the live price from tgju-api.
   607→ */
   608→async function backgroundFetchCurrencyGoldChange(instruments: TgjuInstrument[]) {
   609→  const currencyGoldItems = instruments.filter(
   610→    (i) => i.category === 'currency' || i.category === 'gold' || i.category === 'silver' || i.category === 'gold_etf'
   611→  );
   612→
   613→  // Limit to first 10 to avoid rate-limiting (most important: USD, EUR, gold coin, etc.)
   614→  const priorityKeys = currencyGoldItems.slice(0, 10).map((i) => i.key);
   615→
   616→  for (const key of priorityKeys) {
   617→    try {
   618→      const url = `${TGJU_CHART_API}/${key}?lang=fa&order_dir=desc&start=0&length=2`;
   619→      const res = await fetch(url, { signal: AbortSignal.timeout(10_000) });
   620→      if (!res.ok) continue;
   621→      const jsonStr = await res.text();
   622→      if (!jsonStr.startsWith('{')) continue;
   623→
   624→      const data = JSON.parse(jsonStr);
   625→      const rows: string[][] = data.data;
   626→      if (!rows || rows.length < 2) continue;
   627→
   628→      const newest = rows[0];
   629→      const prev = rows[1];
   630→      const latestClose = parsePersianNum(newest[3]);
   631→      const prevClose = parsePersianNum(prev[3]);
   632→
   633→      if (latestClose > 0 && prevClose > 0) {
   634→        const inst = liveInstrumentsCache.data?.find((i) => i.key === key);
   635→        if (inst) {
   636→          inst.change = latestClose - prevClose;
   637→          inst.changePercent = ((latestClose - prevClose) / prevClose) * 100;
   638→        }
   639→      }
   640→    } catch {
   641→      // skip individual errors
   642→    }
   643→  }
   644→}

   654→/**
   655→ * Background task: fetch Yahoo Finance prices for static instruments that have price=0
   656→ * (happens when api.tgju.org returns 403). Updates the live cache in-place.
   657→ */
   658→async function backgroundFetchYahooFallbackPrices(instruments: TgjuInstrument[]) {
   659→  // Collect static instruments with price=0 that have a Yahoo mapping
   660→  const zeroPriceItems = instruments.filter(
   661→    (i) => i.price === 0 && i.key in TGJU_TO_YAHOO_MAP
   662→  );
   663→
   664→  if (zeroPriceItems.length === 0) return;
   665→
   666→  console.log(`[TGJU] ${zeroPriceItems.length} static instruments have price=0, fetching from Yahoo Finance as fallback...`);
   667→
   668→  // Collect unique Yahoo symbols
   669→  const keyToSymbol = new Map<string, string>();
   670→  const symbolToKeys = new Map<string, string[]>();
   671→  for (const item of zeroPriceItems) {
   672→    const yahooSym = TGJU_TO_YAHOO_MAP[item.key];
   673→    if (!yahooSym) continue;
   674→    keyToSymbol.set(item.key, yahooSym);
   675→    if (!symbolToKeys.has(yahooSym)) symbolToKeys.set(yahooSym, []);
   676→    symbolToKeys.get(yahooSym)!.push(item.key);
   677→  }
   678→
   679→  const uniqueSymbols = [...symbolToKeys.keys()];
   680→
   681→  try {
   682→    // Dynamic import to avoid loading yahoo-finance2 for TSE-only users
   683→    const YahooFinance = (await import('yahoo-finance2')).default;
   684→    const yahooFinance = new YahooFinance({ suppressNotices: ['yahooSurvey'], validation: { logErrors: false } });
   685→
   686→    // Fetch in batches of 15 (smaller batches reduce Yahoo validation errors)
   687→    const BATCH_SIZE = 15;
   688→    for (let i = 0; i < uniqueSymbols.length; i += BATCH_SIZE) {
   689→      const batch = uniqueSymbols.slice(i, i + BATCH_SIZE);
   690→      try {
   691→        const results = await yahooFinance.quote(batch, {
   692→          fields: ['symbol', 'regularMarketPrice', 'regularMarketChange', 'regularMarketChangePercent', 'regularMarketPreviousClose', 'regularMarketDayHigh', 'regularMarketDayLow'],
   693→        });
   694→
   695→        for (const r of results) {
   696→          if (r.regularMarketPrice == null) continue;
   697→          const tgjuKeys = symbolToKeys.get(r.symbol);
   698→          if (!tgjuKeys) continue;
   699→
   700→          for (const tgjuKey of tgjuKeys) {
   701→            const inst = liveInstrumentsCache.data?.find((x) => x.key === tgjuKey);
   702→            if (inst && inst.price === 0) {
   703→              inst.price = r.regularMarketPrice;
   704→              inst.highPrice = r.regularMarketDayHigh ?? r.regularMarketPrice;
   705→              inst.lowPrice = r.regularMarketDayLow ?? r.regularMarketPrice;
   706→              inst.change = r.regularMarketChange ?? 0;
   707→              inst.changePercent = r.regularMarketChangePercent ?? 0;
   708→
   709→              // Also update static price cache
   710→              staticPriceCache.set(tgjuKey, {
   711→                close: r.regularMarketPrice,
   712→                high: r.regularMarketDayHigh ?? r.regularMarketPrice,
   713→                low: r.regularMarketDayLow ?? r.regularMarketPrice,
   714→                prevClose: r.regularMarketPreviousClose ?? r.regularMarketPrice,
   715→                ts: Date.now(),
   716→              });
   717→            }
   718→          }
   719→        }
   720→      } catch (err) {
   721→        const errMsg = err instanceof Error ? err.message : String(err);
   722→        if (errMsg.includes('validation') || errMsg.includes('FailedYahooValidationError')) {
   723→          // Batch failed due to schema validation — retry individually
   724→          console.warn(`[TGJU] Yahoo fallback batch ${Math.floor(i / BATCH_SIZE) + 1} validation error, retrying individually...`);
   725→          for (const symbol of batch) {
   726→            try {
   727→              const singleResults = await yahooFinance.quote([symbol], {
   728→                fields: ['symbol', 'regularMarketPrice', 'regularMarketChange', 'regularMarketChangePercent', 'regularMarketPreviousClose', 'regularMarketDayHigh', 'regularMarketDayLow'],
   729→              });
   730→              for (const r of singleResults) {
   731→                if (r.regularMarketPrice == null) continue;
   732→                const tgjuKeys = symbolToKeys.get(r.symbol);
   733→                if (!tgjuKeys) continue;
   734→                for (const tgjuKey of tgjuKeys) {
   735→                  const inst = liveInstrumentsCache.data?.find((x) => x.key === tgjuKey);
   736→                  if (inst && inst.price === 0) {
   737→                    inst.price = r.regularMarketPrice;
   738→                    inst.highPrice = r.regularMarketDayHigh ?? r.regularMarketPrice;
   739→                    inst.lowPrice = r.regularMarketDayLow ?? r.regularMarketPrice;
   740→                    inst.change = r.regularMarketChange ?? 0;
   741→                    inst.changePercent = r.regularMarketChangePercent ?? 0;
   742→                    staticPriceCache.set(tgjuKey, {
   743→                      close: r.regularMarketPrice,
   744→                      high: r.regularMarketDayHigh ?? r.regularMarketPrice,
   745→                      low: r.regularMarketDayLow ?? r.regularMarketPrice,
   746→                      prevClose: r.regularMarketPreviousClose ?? r.regularMarketPrice,
   747→                      ts: Date.now(),
   748→                    });
   749→                  }
   750→                }
   751→              }
   752→            } catch {
   753→              // skip individual symbol errors
   754→            }
   755→          }
   756→        } else {
   757→          console.warn(`[TGJU] Yahoo fallback batch ${Math.floor(i / BATCH_SIZE) + 1} failed:`, errMsg);
   758→        }
   759→      }
   760→    }
   761→
   762→    const filledCount = zeroPriceItems.filter((i) => i.price > 0).length;
   763→    console.log(`[TGJU] Yahoo fallback filled ${filledCount}/${zeroPriceItems.length} instrument prices`);
   764→  } catch (err) {
   765→    console.error(`[TGJU] Yahoo fallback fetch failed:`, err);
   766→  }
   767→}

   769→/* ─── Fetch historical OHLC data ── */
   770→
   771→/**
   772→ * Fetch historical OHLC data from TGJU chart API.
   773→ * Strategy:
   774→ * 1. Try direct fetch (fast, ~100ms) — works when Cloudflare is not blocking
   775→ * 2. If 403/blocked, fall back to z-ai page_reader (bypasses Cloudflare, ~3s)
   776→ * 3. If both fail, return cached data if available
   777→ *
   778→ * All results are cached for HISTORY_TTL (30 min) to minimize page_reader calls.
   779→ */
   780→export async function fetchTgjuHistory(tgjuKey: string): Promise<TgjuOHLC[]> {
   781→  const cached = historyCache.get(tgjuKey);
   782→  const now = Date.now();
   783→  if (cached && now - cached.ts < HISTORY_TTL) {
   784→    return cached.data;
   785→  }
   786→
   787→  // ── Step 1: Try direct fetch (fast path) ──
   788→  const directResult = await tryDirectFetch(tgjuKey);
   789→  if (directResult && directResult.length >= 30) {
   790→    cacheHistory(tgjuKey, directResult, now);
   791→    return directResult;
   792→  }
   793→
   794→  // ── Step 2: Fall back to page_reader (bypasses Cloudflare) ──
   795→  console.log(`[TGJU] Direct fetch returned ${directResult?.length ?? 0} candles for ${tgjuKey}, trying page_reader...`);
   796→  const pageReaderResult = await fetchTgjuHistoryViaPageReader(tgjuKey);
   797→  if (pageReaderResult && pageReaderResult.length >= 30) {
   798→    cacheHistory(tgjuKey, pageReaderResult, now);
   799→    return pageReaderResult;
   800→  }
   801→
   802→  // ── Step 3: Both failed — return stale cache or empty ──
   803→  console.warn(`[TGJU] All methods failed for ${tgjuKey}, returning ${cached?.data?.length ?? 0} cached candles`);
   804→  return cached?.data || [];
   805→}

   807→/**
   808→ * Try direct HTTP fetch to TGJU chart API.
   809→ * Returns parsed candles or null on failure.
   810→ */
   811→async function tryDirectFetch(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   812→  try {
   813→    const url = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=365`;
   814→    const controller = new AbortController();
   815→    const timer = setTimeout(() => controller.abort(), 12_000);
   816→
   817→    const res = await fetch(url, { signal: controller.signal, headers: { 'Accept': 'application/json' } });
   818→    clearTimeout(timer);
   819→
   820→    if (!res.ok) {
   821→      console.warn(`[TGJU] Direct fetch HTTP ${res.status} for ${tgjuKey}`);
   822→      return null;
   823→    }
   824→
   825→    const jsonStr = await res.text();
   826→    if (!jsonStr.startsWith('{')) {
   827→      console.warn(`[TGJU] Direct fetch non-JSON for ${tgjuKey}`);
   828→      return null;
   829→    }
   830→
   831→    return parseTgjuChartData(tgjuKey, jsonStr);
   832→  } catch (err) {
   833→    console.warn(`[TGJU] Direct fetch error for ${tgjuKey}:`, err instanceof Error ? err.message : err);
   834→    return null;
   835→  }
   836→}
   837→
   838→/**
   839→ * Fetch TGJU historical data via z-ai CLI page_reader (bypasses Cloudflare).
   840→ * Uses child_process to call `z-ai function -n page_reader` which runs
   841→ * a headless browser internally and can solve Cloudflare challenges.
   842→ * Slower (~5-15s) but reliable when direct fetch is blocked.
   843→ */
   844→async function fetchTgjuHistoryViaPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   845→  const { execFile } = await import('child_process');
   846→  const { readFile, unlink } = await import('fs/promises');
   847→  const path = await import('path');
   848→  const os = await import('os');
   849→
   850→  const tmpFile = path.join(os.tmpdir(), `tgju-${tgjuKey}-${Date.now()}.json`);
   851→  const apiUrl = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=365`;
   852→
   853→  try {
   854→    // Call z-ai CLI to invoke page_reader
   855→    await new Promise<void>((resolve, reject) => {
   856→      execFile(
   857→        'z-ai',
   858→        ['function', '-n', 'page_reader', '-a', JSON.stringify({ url: apiUrl }), '-o', tmpFile],
   859→        { timeout: 45_000, maxBuffer: 50 * 1024 * 1024 },
   860→        (error) => {
   861→          if (error) reject(error);
   862→          else resolve();
   863→        },
   864→      );
   865→    });
   866→
   867→    // Read the output file
   868→    const fileContent = await readFile(tmpFile, 'utf-8');
   869→    const pageData = JSON.parse(fileContent);
   870→
   871→    if (pageData.code !== 200 || !pageData.data?.html) {
   872→      console.warn(`[TGJU] page_reader CLI failed for ${tgjuKey}: code=${pageData.code}`);
   873→      return null;
   874→    }
   875→
   876→    // Extract JSON from <pre> tag in the HTML
   877→    const html = pageData.data.html;
   878→    const preMatch = html.match(/<pre[^>]*>([\s\S]*?)<\/pre>/);
   879→    if (!preMatch) {
   880→      console.warn(`[TGJU] page_reader: no <pre> block found for ${tgjuKey}`);
   881→      return null;
   882→    }
   883→
   884→    // Unescape HTML entities
   885→    const jsonStr = preMatch[1]
   886→      .replace(/&quot;/g, '"')
   887→      .replace(/&amp;/g, '&')
   888→      .replace(/&lt;/g, '<')
   889→      .replace(/&gt;/g, '>');
   890→
   891→    if (!jsonStr.startsWith('{')) {
   892→      console.warn(`[TGJU] page_reader: extracted content is not JSON for ${tgjuKey}`);
   893→      return null;
   894→    }
   895→
   896→    return parseTgjuChartData(tgjuKey, jsonStr);
   897→  } catch (err) {
   898→    console.error(`[TGJU] page_reader CLI error for ${tgjuKey}:`, err instanceof Error ? err.message : err);
   899→    return null;
   900→  } finally {
   901→    // Clean up temp file
   902→    try { await unlink(tmpFile); } catch {}
   903→  }
   904→}

   906→/**
   907→ * Parse the JSON data from TGJU chart API response.
   908→ * Used by both direct fetch and page_reader paths.
   909→ */
   910→function parseTgjuChartData(tgjuKey: string, jsonStr: string): TgjuOHLC[] | null {
   911→  try {
   912→    const parsed = JSON.parse(jsonStr);
   913→    const rows: (string | number)[][] = parsed.data;
   914→
   915→    if (!rows || rows.length === 0) {
   916→      console.warn(`[TGJU] Empty data array for ${tgjuKey}`);
   917→      return null;
   918→    }
   919→
   920→    // ── Auto-detect column mapping from first valid row ──
   921→    let colMap = detectColumnMap(rows[0]);
   922→    if (!colMap) {
   923→      for (let r = 1; r < Math.min(5, rows.length); r++) {
   924→        colMap = detectColumnMap(rows[r]);
   925→        if (colMap) break;
   926→      }
   927→    }
   928→    if (!colMap) {
   929→      console.warn(`[TGJU] Could not auto-detect columns for ${tgjuKey}, using default order`);
   930→      colMap = { openIdx: 0, highIdx: 2, lowIdx: 1, closeIdx: 3, jalaliDateIdx: 7, gregorianDateIdx: 6 };
   931→    } else {
   932→      console.log(`[TGJU] Column map for ${tgjuKey}: O=${colMap.openIdx} H=${colMap.highIdx} L=${colMap.lowIdx} C=${colMap.closeIdx} Date=${colMap.jalaliDateIdx}`);
   933→    }
   934→
   935→    const { openIdx, highIdx, lowIdx, closeIdx, jalaliDateIdx, gregorianDateIdx } = colMap;
   936→
   937→    // ── Parse rows ──
   938→    const candles: TgjuOHLC[] = [];
   939→    let invalidCount = 0;
   940→
   941→    for (const row of rows) {
   942→      const rawDate = String(row[jalaliDateIdx] || row[gregorianDateIdx] || '');
   943→      const date = rawDate.replace(/[\/]/g, '-');
   944→      const o = parsePersianNum(row[openIdx]);
   945→      const h = parsePersianNum(row[highIdx]);
   946→      const l = parsePersianNum(row[lowIdx]);
   947→      const c = parsePersianNum(row[closeIdx]);
   948→
   949→      if (!isValidCandle(o, h, l, c) || date.length === 0) {
   950→        invalidCount++;
   951→        continue;
   952→      }
   953→
   954→      candles.push({ date, open: o, high: h, low: l, close: c });
   955→    }
   956→
   957→    if (invalidCount > 0) {
   958→      console.warn(`[TGJU] ${tgjuKey}: ${invalidCount}/${rows.length} invalid candles skipped`);
   959→    }
   960→
   961→    if (candles.length === 0) {
   962→      return null;
   963→    }
   964→
   965→    // ── Detect and fix data drift ──
   966→    let driftCount = 0;
   967→    for (let i = 1; i < candles.length; i++) {
   968→      const prevClose = candles[i - 1].close;
   969→      const currClose = candles[i].close;
   970→      if (prevClose > 0) {
   971→        const pctChange = Math.abs(currClose - prevClose) / prevClose;
   972→        if (pctChange > 0.5) {
   973→          driftCount++;
   974→        }
   975→      }
   976→    }
   977→    if (driftCount > candles.length * 0.1) {
   978→      console.warn(`[TGJU] ${tgjuKey}: Possible data drift detected (${driftCount} suspicious jumps out of ${candles.length} candles)`);
   979→    }
   980→
   981→    // Reverse: API returns oldest-first (asc), we want newest-last for charting
   982→    const firstDate = candles[0].date;
   983→    const lastDate = candles[candles.length - 1].date;
   984→    if (firstDate > lastDate) {
   985→      candles.reverse();
   986→    }
   987→
   988→    console.log(`[TGJU] ${tgjuKey}: Parsed ${candles.length} candles (${candles[0]?.date} to ${candles[candles.length - 1]?.date})`);
   989→
   990→    return candles;
   991→  } catch (err) {
   992→    console.error(`[TGJU] Failed to parse chart data for ${tgjuKey}:`, err);
   993→    return null;
   994→  }
   995→}

   997→/**
   998→ * Cache parsed history data and update related caches.
   999→ */
  1000→function cacheHistory(tgjuKey: string, candles: TgjuOHLC[], now: number): void {
  1001→  historyCache.set(tgjuKey, { data: candles, ts: now });
  1002→
  1003→  // Also update static price cache for this instrument
  1004→  if (candles.length >= 2) {
  1005→    const last = candles[candles.length - 1];
  1006→    const prev = candles[candles.length - 2];
  1007→    staticPriceCache.set(tgjuKey, {
  1008→      close: last.close,
  1009→      high: last.high,
  1010→      low: last.low,
  1011→      prevClose: prev.close,
  1012→      ts: now,
  1013→    });
  1014→
  1015→    // Update live instruments cache if this is a static instrument
  1016→    if (liveInstrumentsCache.data) {
  1017→      const inst = liveInstrumentsCache.data.find((i) => i.key === tgjuKey);
  1018→      if (inst && inst.price === 0) {
  1019→        inst.price = last.close;
  1020→        inst.highPrice = last.high;
  1021→        inst.lowPrice = last.low;
  1022→        inst.change = last.close - prev.close;
  1023→        inst.changePercent = prev.close > 0 ? ((last.close - prev.close) / prev.close) * 100 : 0;
  1024→      }
  1025→    }
  1026→  }
  1027→}
