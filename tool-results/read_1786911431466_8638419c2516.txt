    84→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
    85→function sma(closes: number[], period: number): number {
    86→  if (closes.length < period) return 0;
    87→  const slice = closes.slice(-period);
    88→  return slice.reduce((a, b) => a + b, 0) / period;
    89→}

    91→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
    92→function emaCalc(closes: number[], period: number): number {
    93→  if (closes.length < period) return 0;
    94→  const k = 2 / (period + 1);
    95→  let emaVal = sma(closes.slice(0, period), period);
    96→  for (let i = period; i < closes.length; i++) {
    97→    emaVal = closes[i] * k + emaVal * (1 - k);
    98→  }
    99→  return emaVal;
   100→}

   102→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   103→function calcRSI(closes: number[], period: number = 14): number {
   104→  if (closes.length < period + 1) return 50;
   105→  let avgGain = 0;
   106→  let avgLoss = 0;
   107→  for (let i = 1; i <= period; i++) {
   108→    const diff = closes[i] - closes[i - 1];
   109→    if (diff > 0) avgGain += diff;
   110→    else avgLoss += Math.abs(diff);
   111→  }
   112→  avgGain /= period;
   113→  avgLoss /= period;
   114→  for (let i = period + 1; i < closes.length; i++) {
   115→    const diff = closes[i] - closes[i - 1];
   116→    const gain = diff > 0 ? diff : 0;
   117→    const loss = diff < 0 ? Math.abs(diff) : 0;
   118→    avgGain = (avgGain * (period - 1) + gain) / period;
   119→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   120→  }
   121→  if (avgLoss === 0) return 100;
   122→  const rs = avgGain / avgLoss;
   123→  return 100 - 100 / (1 + rs);
   124→}

   126→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   127→function calcMFI(data: OHLCV[], period: number = 14): number {
   128→  if (data.length < period + 1) return 50;
   129→  let posFlow = 0;
   130→  let negFlow = 0;
   131→  for (let i = data.length - period; i < data.length; i++) {
   132→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   133→    const mf = tp * data[i].volume;
   134→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   135→    if (tp > prevTp) posFlow += mf;
   136→    else negFlow += mf;
   137→  }
   138→  if (negFlow === 0) return 100;
   139→  const mfr = posFlow / negFlow;
   140→  return 100 - 100 / (1 + mfr);
   141→}

   143→// ─── Helper: CCI(20) ──────────────────────────────────────────────────────────
   144→function calcCCI(data: OHLCV[], period: number = 20): number {
   145→  if (data.length < period) return 0;
   146→  const tps: number[] = [];
   147→  for (let i = data.length - period; i < data.length; i++) {
   148→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   149→  }
   150→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   151→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   152→  if (meanDev === 0) return 0;
   153→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   154→}

   156→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   157→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   158→  if (data.length < kPeriod) return { k: 50, d: 50 };
   159→  const rawKs: number[] = [];
   160→  for (let i = kPeriod - 1; i < data.length; i++) {
   161→    let lowest = Infinity;
   162→    let highest = -Infinity;
   163→    for (let j = i - kPeriod + 1; j <= i; j++) {
   164→      if (data[j].low < lowest) lowest = data[j].low;
   165→      if (data[j].high > highest) highest = data[j].high;
   166→    }
   167→    const range = highest - lowest;
   168→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   169→  }
   170→  const smoothedK: number[] = [];
   171→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   172→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   173→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   174→  }
   175→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   176→  const dSlice = smoothedK.slice(-smoothD);
   177→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   178→  return { k, d };
   179→}

   181→// ─── Helper: Williams %R(14) ──────────────────────────────────────────────────
   182→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   183→  if (data.length < period) return -50;
   184→  const slice = data.slice(-period);
   185→  let highest = -Infinity;
   186→  let lowest = Infinity;
   187→  for (const c of slice) {
   188→    if (c.high > highest) highest = c.high;
   189→    if (c.low < lowest) lowest = c.low;
   190→  }
   191→  const range = highest - lowest;
   192→  if (range === 0) return -50;
   193→  return ((highest - data[data.length - 1].close) / range) * -100;
   194→}

   196→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   197→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   198→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   199→  const emaFast = emaCalc(closes, fast);
   200→  const emaSlow = emaCalc(closes, slow);
   201→  const macdLine = emaFast - emaSlow;
   202→  // Calculate signal as EMA of MACD values
   203→  const macdValues: number[] = [];
   204→  const kFast = 2 / (fast + 1);
   205→  const kSlow = 2 / (slow + 1);
   206→  let eFast = sma(closes.slice(0, fast), fast);
   207→  let eSlow = sma(closes.slice(0, slow), slow);
   208→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   209→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   210→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   211→    macdValues.push(eFast - eSlow);
   212→  }
   213→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   214→  const kSig = 2 / (sig + 1);
   215→  let signalVal = sma(macdValues.slice(0, sig), sig);
   216→  for (let i = sig; i < macdValues.length; i++) {
   217→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   218→  }
   219→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   220→}

   222→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   223→function calcATR(data: OHLCV[], period: number = 14): number {
   224→  if (data.length < 2) return 0;
   225→  const trs: number[] = [];
   226→  for (let i = 1; i < data.length; i++) {
   227→    const tr = Math.max(
   228→      data[i].high - data[i].low,
   229→      Math.abs(data[i].high - data[i - 1].close),
   230→      Math.abs(data[i].low - data[i - 1].close)
   231→    );
   232→    trs.push(tr);
   233→  }
   234→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   235→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   236→  for (let i = period; i < trs.length; i++) {
   237→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   238→  }
   239→  return atrVal;
   240→}

   242→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   243→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   244→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   245→  const plusDMs: number[] = [];
   246→  const minusDMs: number[] = [];
   247→  const trs: number[] = [];
   248→  for (let i = 1; i < data.length; i++) {
   249→    const upMove = data[i].high - data[i - 1].high;
   250→    const downMove = data[i - 1].low - data[i].low;
   251→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   252→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   253→    trs.push(Math.max(
   254→      data[i].high - data[i].low,
   255→      Math.abs(data[i].high - data[i - 1].close),
   256→      Math.abs(data[i].low - data[i - 1].close)
   257→    ));
   258→  }
   259→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   260→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   261→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   262→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   263→  const dxValues: number[] = [];
   264→  for (let i = period; i < trs.length; i++) {
   265→    smoothTR = smoothTR - smoothTR / period + trs[i];
   266→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   267→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   268→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   269→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   270→    const diSum = diPlus + diMinus;
   271→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   272→    dxValues.push(dx);
   273→  }
   274→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   275→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   276→  for (let i = period; i < dxValues.length; i++) {
   277→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   278→  }
   279→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   280→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   281→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   282→}

   284→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   285→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   286→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   287→  let isLong = true;
   288→  let af = step;
   289→  let ep = data[0].high;
   290→  let sar = data[0].low;
   291→  for (let i = 1; i < data.length; i++) {
   292→    const prevSar = sar;
   293→    sar = prevSar + af * (ep - prevSar);
   294→    // Ensure SAR is within bounds
   295→    if (isLong) {
   296→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   297→      if (data[i].low < sar) {
   298→        isLong = false;
   299→        sar = ep;
   300→        ep = data[i].low;
   301→        af = step;
   302→      } else {
   303→        if (data[i].high > ep) {
   304→          ep = data[i].high;
   305→          af = Math.min(af + step, max);
   306→        }
   307→      }
   308→    } else {
   309→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   310→      if (data[i].high > sar) {
   311→        isLong = true;
   312→        sar = ep;
   313→        ep = data[i].high;
   314→        af = step;
   315→      } else {
   316→        if (data[i].low < ep) {
   317→          ep = data[i].low;
   318→          af = Math.min(af + step, max);
   319→        }
   320→      }
   321→    }
   322→  }
   323→  return sar;
   324→}

   326→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   327→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   328→  if (closes.length < period) {
   329→    const c = closes[closes.length - 1] ?? 0;
   330→    return { upper: c, middle: c, lower: c };
   331→  }
   332→  const slice = closes.slice(-period);
   333→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   334→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   335→  const sd = Math.sqrt(variance);
   336→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   337→}

   339→// ─── Helper: OBV ──────────────────────────────────────────────────────────────
   340→function calcOBV(data: OHLCV[]): number {
   341→  let obv = 0;
   342→  for (let i = 1; i < data.length; i++) {
   343→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   344→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   345→  }
   346→  return obv;
   347→}

   349→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   350→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   351→  const n = values.length;
   352→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   353→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   354→  for (let i = 0; i < n; i++) {
   355→    sumX += i;
   356→    sumY += values[i];
   357→    sumXY += i * values[i];
   358→    sumX2 += i * i;
   359→    sumY2 += values[i] * values[i];
   360→  }
   361→  const denom = n * sumX2 - sumX * sumX;
   362→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   363→  const slope = (n * sumXY - sumX * sumY) / denom;
   364→  const intercept = (sumY - slope * sumX) / n;
   365→  const meanY = sumY / n;
   366→  let ssRes = 0, ssTot = 0;
   367→  for (let i = 0; i < n; i++) {
   368→    const predicted = slope * i + intercept;
   369→    ssRes += (values[i] - predicted) ** 2;
   370→    ssTot += (values[i] - meanY) ** 2;
   371→  }
   372→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   373→  return { slope, intercept, r2 };
   374→}

   376→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   377→function roundToNice(price: number): number {
   378→  const abs = Math.abs(price);
   379→  let step: number;
   380→  if (abs >= 100000) step = 100;
   381→  else if (abs >= 10000) step = 50;
   382→  else if (abs >= 1000) step = 10;
   383→  else if (abs >= 100) step = 5;
   384→  else step = 1;
   385→  return Math.round(price / step) * step;
   386→}

   388→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
   389→function findSwingLevels(data: OHLCV[]): { highs: number[]; lows: number[] } {
   390→  const highs: number[] = [];
   391→  const lows: number[] = [];
   392→  const lookback = 3;
   393→  for (let i = lookback; i < data.length - lookback; i++) {
   394→    let isHigh = true;
   395→    let isLow = true;
   396→    for (let j = i - lookback; j <= i + lookback; j++) {
   397→      if (j === i) continue;
   398→      if (data[j].high >= data[i].high) isHigh = false;
   399→      if (data[j].low <= data[i].low) isLow = false;
   400→    }
   401→    if (isHigh) highs.push(data[i].high);
   402→    if (isLow) lows.push(data[i].low);
   403→  }
   404→  return { highs, lows };
   405→}

   407→// ─── Support / Resistance (5 each, rounded, with strength) ──────────────────
   408→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   409→  resistances: number[];
   410→  supports: number[];
   411→  supportStrengths: LevelStrength[];
   412→  resistanceStrengths: LevelStrength[];
   413→  priceTargets: LevelStrength[];
   414→} {
   415→  const allRawLevels: number[] = [];
   416→
   417→  // 1. Pivot Points from previous day
   418→  const prev = data[data.length - 2] ?? data[data.length - 1];
   419→  if (prev) {
   420→    const pp = (prev.high + prev.low + prev.close) / 3;
   421→    const hl = prev.high - prev.low;
   422→    const pivotR = [
   423→      2 * pp - prev.low,
   424→      pp + hl,
   425→      prev.high + 2 * (pp - prev.low),
   426→      prev.high + 2 * (pp - prev.low) + hl,
   427→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   428→    ];
   429→    const pivotS = [
   430→      2 * pp - prev.high,
   431→      pp - hl,
   432→      prev.low - 2 * (prev.high - pp),
   433→      prev.low - 2 * (prev.high - pp) - hl,
   434→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   435→    ];
   436→    allRawLevels.push(...pivotR, ...pivotS);
   437→  }
   438→
   439→  // 2. Swing levels
   440→  const swings = findSwingLevels(data);
   441→  allRawLevels.push(...swings.highs, ...swings.lows);
   442→
   443→  // 3. Round all and filter invalid
   444→  const rounded = allRawLevels.map(roundToNice).filter(l => l > 0);
   445→
   446→  // 4. Remove duplicates (within 0.5%)
   447→  const unique: number[] = [];
   448→  for (const level of rounded) {
   449→    const isDup = unique.some(u => Math.abs(u - level) / u < 0.005);
   450→    if (!isDup) unique.push(level);
   451→  }
   452→
   453→  // 5. Calculate strength for each unique level
   454→  // Strength = count of other levels within ±5% range
   455→  // More levels nearby = stronger
   456→  const levelStrengths: LevelStrength[] = unique.map(level => {
   457→    const range5pct = level * 0.05;
   458→    const nearbyCount = unique.filter(other => other !== level && Math.abs(other - level) <= range5pct).length;
   459→    // Base strength: 1 for level itself, +1 for each nearby level (max 10)
   460→    const strength = Math.min(10, Math.max(1, 1 + nearbyCount * 2));
   461→    return { price: level, strength, isTarget: false };
   462→  });
   463→
   464→  // 6. Separate above/below and enforce 5% min gap
   465→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   466→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   467→
   468→  // Enforce 5% minimum gap between adjacent levels
   469→  function enforceGap(levels: LevelStrength[], minGapPct: number = 0.05): LevelStrength[] {
   470→    if (levels.length === 0) return [];
   471→    const result: LevelStrength[] = [levels[0]];
   472→    for (let i = 1; i < levels.length; i++) {
   473→      const prev = result[result.length - 1].price;
   474→      const curr = levels[i].price;
   475→      const gap = Math.abs(curr - prev) / prev;
   476→      if (gap >= minGapPct) {
   477→        result.push(levels[i]);
   478→      } else {
   479→        // Merge: keep the one with higher strength
   480→        if (levels[i].strength > result[result.length - 1].strength) {
   481→          result[result.length - 1] = { ...levels[i], strength: Math.min(10, result[result.length - 1].strength + 1) };
   482→        } else {
   483→          result[result.length - 1] = { ...result[result.length - 1], strength: Math.min(10, result[result.length - 1].strength + 1) };
   484→        }
   485→      }
   486→    }
   487→    return result;
   488→  }
   489→
   490→  const finalResistances = enforceGap(above, 0.05).slice(0, 5);
   491→  const finalSupports = enforceGap(below, 0.05).slice(0, 5);
   492→
   493→  // 7. Determine price targets: strength > 7 is a target
   494→  // If none > 7, pick top 2 strongest
   495→  const allLevels = [...finalResistances, ...finalSupports];
   496→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   497→  let targets: LevelStrength[];
   498→  if (sortedByStrength.some(l => l.strength > 7)) {
   499→    targets = sortedByStrength.filter(l => l.strength > 7);
   500→  } else {
   501→    targets = sortedByStrength.slice(0, 2);
   502→  }
   503→
   504→  // Mark targets
   505→  targets = targets.map(t => ({ ...t, isTarget: true }));
   506→
   507→  // Also mark in final arrays
   508→  finalResistances.forEach(r => {
   509→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   510→  });
   511→  finalSupports.forEach(s => {
   512→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   513→  });
   514→
   515→  return {
   516→    resistances: finalResistances.map(l => l.price),
   517→    supports: finalSupports.map(l => l.price),
   518→    supportStrengths: finalSupports,
   519→    resistanceStrengths: finalResistances,
   520→    priceTargets: targets,
   521→  };
   522→}

   524→// ─── Trend Analysis ───────────────────────────────────────────────────────────
   525→function calcTrend(closes: number[], period: number): TrendResult {
   526→  const slice = closes.slice(-Math.min(period, closes.length));
   527→  const { slope, r2 } = linearRegression(slice);
   528→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   529→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   530→  const priceThreshold = avgPrice * 0.001; // 0.1%
   531→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   532→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   533→}

   555→      bullScore: 0.5, bearScore: 0.5, overallSignal: 'neutral',
   556→    };
   557→  }
   558→

   561→

   570→  };

   574→  const mfi = calcMFI(data);
   575→  const cci = calcCCI(data);
   576→  const stoch = calcStochastic(data);
   577→  const williamsR = calcWilliamsR(data);
   578→  const macd = calcMACD(closes);
   579→
   580→  // ── Trend ────────────────────────────────────────────────────────────────
   581→  const adxResult = calcADX(data);
   582→  const sar = calcSAR(data);
   583→
   584→  // ── Volatility ───────────────────────────────────────────────────────────
   585→  const atr = calcATR(data);
   586→  const bb = calcBollingerBands(closes);
   587→

   589→  const obv = calcOBV(data);
   590→
   591→  // ── Support / Resistance ────────────────────────────────────────────────
   592→  const { resistances, supports, supportStrengths, resistanceStrengths, priceTargets } = calcSupportResistance(data, price);
   593→
   594→  // ── Trend Lines ─────────────────────────────────────────────────────────
   595→  const trend = {
   596→    short: calcTrend(closes, 21),
   597→    medium: calcTrend(closes, 50),
   598→    long: calcTrend(closes, Math.min(100, closes.length)),
   599→  };

   601→  // ═══════════════════════════════════════════════════════════════════════════
   602→  // PROBABILITY ENGINE — uses ALL indicators
   603→  // ═══════════════════════════════════════════════════════════════════════════
   604→
   605→  const R1_level = resistances[0] ?? price * 1.05;
   606→  const R2_level = resistances[1] ?? price * 1.10;
   607→  const R3_level = resistances[2] ?? price * 1.15;
   608→  const R4_level = resistances[3] ?? price * 1.20;
   609→  const R5_level = resistances[4] ?? price * 1.30;
   610→  const S1_level = supports[0] ?? price * 0.95;
   611→  const S2_level = supports[1] ?? price * 0.90;
   612→  const S3_level = supports[2] ?? price * 0.85;
   613→  const S4_level = supports[3] ?? price * 0.80;
   614→  const S5_level = supports[4] ?? price * 0.75;
   615→
   616→  // ── Bull Score: weighted average of all 20 signals ──────────────────────
   617→  const signals: number[] = [];
   618→
   619→  // 1. RSI
   620→  signals.push(rsi > 70 ? 1 : rsi > 50 ? 0.5 : 0);
   621→  // 2. MFI
   622→  signals.push(mfi > 80 ? 1 : mfi > 50 ? 0.5 : 0);
   623→  // 3. CCI
   624→  signals.push(cci > 100 ? 1 : cci > 0 ? 0.5 : 0);
   625→  // 4. Stochastic %K
   626→  signals.push(stoch.k > 80 ? 1 : stoch.k > 50 ? 0.5 : 0);
   627→  // 5. Williams %R
   628→  signals.push(williamsR > -20 ? 1 : williamsR > -50 ? 0.5 : 0);
   629→  // 6. MACD
   630→  signals.push(macd.line > macd.signal ? 1 : macd.line > 0 ? 0.5 : 0);
   631→  // 7. DI+ > DI-
   632→  signals.push(adxResult.diPlus > adxResult.diMinus ? 1 : 0);
   633→  // 8-13. Price vs MAs
   634→  signals.push(price > (smaResult.sma21 ?? 0) ? 1 : 0);
   635→  signals.push(price > (smaResult.sma50 ?? 0) ? 1 : 0);
   636→  signals.push(price > (smaResult.sma100 ?? 0) ? 1 : 0);
   637→  signals.push(price > (smaResult.sma200 ?? 0) ? 1 : 0);
   638→  signals.push(price > (emaResult.ema12 ?? 0) ? 1 : 0);
   639→  signals.push(price > (emaResult.ema26 ?? 0) ? 1 : 0);
   640→  // 14. Price vs SAR
   641→  signals.push(price > sar ? 1 : 0);
   642→  // 15. OBV trend
   643→  const obvSlice = Math.min(10, data.length - 1);
   644→  const obvTrend = obvSlice > 0 && data.length > obvSlice
   645→    ? (data.slice(-obvSlice).reduce((s, d, _, arr) => s + (d.close > arr[Math.max(0, arr.length - obvSlice - 1)]?.close ? 1 : 0), 0) / obvSlice) : 0.5;
   646→  signals.push(obvTrend > 0.6 ? 1 : obvTrend > 0.4 ? 0.5 : 0);
   647→  // 16. BB position
   648→  const bbRange = bb.upper - bb.lower;
   649→  const bbPos = bbRange === 0 ? 0.5 : Math.max(0, Math.min(1, (price - bb.lower) / bbRange));
   650→  signals.push(bbPos);
   651→  // 17-19. Trend directions weighted by R²
   652→  signals.push((trend.short.direction === 'up' ? 1 : trend.short.direction === 'down' ? 0 : 0.5) * trend.short.r2);
   653→  signals.push((trend.medium.direction === 'up' ? 1 : trend.medium.direction === 'down' ? 0 : 0.5) * trend.medium.r2);
   654→  signals.push((trend.long.direction === 'up' ? 1 : trend.long.direction === 'down' ? 0 : 0.5) * trend.long.r2);
   655→  // 20. ADX strength
   656→  signals.push(Math.min(adxResult.adx / 100, 1));
   657→

   659→  const bearScore = 1 - bullScore;

   661→  // ── Raw Probability Calculation ─────────────────────────────────────────
   662→  const dR1 = Math.exp(-5 * Math.abs(price - R1_level) / R1_level);
   663→  const dS1 = Math.exp(-5 * Math.abs(price - S1_level) / S1_level);

   665→  const distanceRatio = rangeWidth === 0 ? 0.5 : (price - S1_level) / rangeWidth;

   674→  const raw_R2 = 0.25 * bullScore * (1 - dR1) * distanceRatio * trendWeight;
   675→  const raw_R3 = 0.20 * (1 - Math.abs(bullScore - 0.5) * 1.6);

   677→  const raw_R5 = 0.20 * bearScore * dS1 * oversoldFactor * trendWeight;

   681→  const pR1 = sum === 0 ? 20 : Math.round(raw_R1 / sum * 100);
   682→  const pR2 = sum === 0 ? 20 : Math.round(raw_R2 / sum * 100);
   683→  const pR3 = sum === 0 ? 20 : Math.round(raw_R3 / sum * 100);
   684→  const pR4 = sum === 0 ? 20 : Math.round(raw_R4 / sum * 100);
   685→  const pR5 = 100 - (pR1 + pR2 + pR3 + pR4); // remainder ensures sum = 100
   686→
