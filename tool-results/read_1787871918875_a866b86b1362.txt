   202→   202→// Helper: resolve asset type
   203→   203→// ═══════════════════════════════════════════════════════════════
   204→   204→
   205→   205→function resolveAsset(raw: string): AssetType {
   206→   206→  const v = raw.toLowerCase();
   207→   207→  if (v.includes('crypto') || v.includes('btc') || v.includes('eth') || v.includes('usdt')) return 'cryptocurrency';
   208→   208→  if (v.includes('forex') || v.includes('eur') || v.includes('usd') || v.includes('jpy')) return 'forex';
   209→   209→  if (v.includes('gold') || v.includes('طلا') || v.includes('xau') || v.includes('oil') || v.includes('نفت')) return 'commodities';
   210→   210→  return 'stocks';
   211→   211→}
   212→   212→
   213→   213→function resolveTimeframe(raw: string): TimeframeType {
   214→   214→  const v = raw.toLowerCase();
   215→   215→  if (v.includes('h') || v.includes('hour') || v.includes('4h') || v.includes('1h')) return 'hourly';
   216→   216→  if (v.includes('w') || v.includes('week')) return 'weekly';
   217→   217→  if (v.includes('m') || v.includes('month')) return 'monthly';
   218→   218→  return 'daily';
   219→   219→}
   220→   220→
   221→   221→function resolveRegime(raw: string): RegimeType {
   222→   222→  if (raw.includes('Strong Bull') || raw.includes('strong_bull')) return 'Strong Bull';
   223→   223→  if (raw.includes('Weak Bull') || raw.includes('weak_bull')) return 'Weak Bull';
   224→   224→  if (raw.includes('Strong Bear') || raw.includes('strong_bear')) return 'Strong Bear';
   225→   225→  if (raw.includes('Weak Bear') || raw.includes('weak_bear')) return 'Weak Bear';
   226→   226→  return 'Range';
   227→   227→}
   228→   228→
   229→   229→function toPersianNum(n: number): string {
   230→   230→  if (!isFinite(n) || isNaN(n)) return '\u06f0';
   231→   231→  const fixed = Math.round(n * 100) / 100;
   232→   232→  return String(fixed).replace(/\d/g, d => String.fromCharCode('\u0660'.charCodeAt(0) + parseInt(d)));
   233→   233→}

   239→   239→function selectSchools(ctx: MSLV4Context) {
   240→   240→  const asset = resolveAsset(ctx.asset);
   241→   241→  const tf = resolveTimeframe(ctx.timeframe);
   242→   242→  const regime = resolveRegime(ctx.regime);
   243→   243→  const am = ASSET_MULT[asset] ?? ASSET_MULT.stocks;
   244→   244→  const tm = TF_MULT[tf] ?? TF_MULT.daily;
   245→   245→  const rm = REGIME_MULT[regime] ?? REGIME_MULT.Range;
   246→   246→
   247→   247→  const rawWeights: Record<SchoolId, number> = {} as Record<SchoolId, number>;
   248→   248→  for (const s of SCHOOLS) {
   249→   249→    let w = s.base_weight * am[s.id] * tm[s.id] * rm[s.id];
   250→   250→    // Adaptive: zero out volume school if no volume data
   251→   251→    if (s.id === 'volume' && ctx.hasVolume === false) w *= 0.2;
   252→   252→    rawWeights[s.id] = w;
   253→   253→  }
   254→   254→
   255→   255→  const total = Object.values(rawWeights).reduce((a, b) => a + b, 0);
   256→   256→  const normWeights: Record<string, number> = {};
   257→   257→  for (const id of SCHOOL_IDS) normWeights[id] = rawWeights[id] / total;
   258→   258→
   259→   259→  const sorted = SCHOOL_IDS.map(id => ({ id, w: normWeights[id] })).sort((a, b) => b.w - a.w);
   260→   260→  const primary = sorted[0];
   261→   261→  const secondary = sorted[1];
   262→   262→
   263→   263→  let status: string;
   264→   264→  if (primary.id === 'hybrid' && primary.w < 0.35) {
   265→   265→    status = 'hybrid';
   266→   266→  } else if (primary.w > 0.45) {
   267→   267→    status = 'confirmed';
   268→   268→  } else if (primary.w > 0.35) {
   269→   269→    status = 'tentative';
   270→   270→  } else {
   271→   271→    status = 'hybrid';
   272→   272→  }
   273→   273→
   274→   274→  const primaryDef = SCHOOLS.find(s => s.id === primary.id)!;
   275→   275→  const secondaryDef = SCHOOLS.find(s => s.id === secondary.id)!;
   276→   276→
   277→   277→  const reasoning = buildSchoolReasoning(primaryDef, normWeights, ctx, asset, tf, regime);
   278→   278→
   279→   279→  return {
   280→   280→    primary: {
   281→   281→      id: primary.id, name: primaryDef.name, nameEn: primaryDef.nameEn,
   282→   282→      weight: Math.round(primary.w * 100) / 100,
   283→   283→      confidence: ctx.confidence > 0.75 ? 'high' : ctx.confidence > 0.5 ? 'medium' : 'low',
   284→   284→      status, reasoning,
   285→   285→    },
   286→   286→    secondary: {
   287→   287→      id: secondary.id, name: secondaryDef.name,
   288→   288→      weight: Math.round(secondary.w * 100) / 100,
   289→   289→      reasoning: `مکتب ${secondaryDef.name} با وزن ${toPersianNum(secondary.w)} به عنوان مکتب مکمل انتخاب شد`,
   290→   290→    },
   291→   291→    all_schools: Object.fromEntries(SCHOOL_IDS.map(id => [id, Math.round(normWeights[id] * 100) / 100])),
   292→   292→  };
   293→   293→}
   294→   294→
   295→   295→function buildSchoolReasoning(s: SchoolDef, w: Record<string, number>, ctx: MSLV4Context, asset: string, tf: string, regime: string): string[] {
   296→   296→  const r: string[] = [];
   297→   297→  const pc = ctx.pattern_counts?.classical ?? 0;
   298→   298→  const phc = ctx.pattern_counts?.harmonic ?? 0;
   299→   299→  const ec = ctx.pattern_counts?.elliott ?? 0;
   300→   300→  if (s.id === 'classical' && pc >= 2) r.push(`تشخیص ${toPersianNum(pc)} الگوی کلاسیک`);
   301→   301→  if (s.id === 'oscillator' && ctx.divergence_present) r.push('واگرایی تشخیص داده شده');
   302→   302→  if (s.id === 'oscillator' && ctx.rsi !== undefined && (ctx.rsi > 70 || ctx.rsi < 30)) r.push('RSI در منطقه اشباع');
   303→   303→  if (s.id === 'harmonic' && phc >= 2) r.push(`تشخیص ${toPersianNum(phc)} الگوی هارمونیک`);
   304→   304→  if (s.id === 'elliott' && ec >= 3) r.push(`تشخیص ${toPersianNum(ec)} موج الیوت`);
   305→   305→  if (s.id === 'volume' && ctx.hasVolume) r.push('داده‌های حجم موجود و تأییدکننده');
   306→   306→  r.push(`رژیم بازار ${regime} تطابق دارد`);
   307→   307→  r.push(`نوع دارایی: ${asset}، تایم‌فریم: ${tf}`);
   308→   308→  return r;
   309→   309→}

   315→   315→function selectStyles(primarySchoolId: SchoolId, ctx: MSLV4Context) {
   316→   316→  const scores: Record<StyleId, number> = {} as Record<StyleId, number>;
   317→   317→  const audience = ctx.audience_level ?? 'pro';
   318→   318→
   319→   319→  for (const style of STYLES) {
   320→   320→    const compatibility = SS_MATRIX[primarySchoolId]?.[style.id] ?? 0.5;
   321→   321→    const audienceMatch = getAudienceMatch(style.id, audience);
   322→   322→    const marketMatch = getMarketContextMatch(style.id, ctx);
   323→   323→    const schoolMatch = SS_MATRIX[primarySchoolId]?.[style.id] ?? 0.5;
   324→   324→    scores[style.id] = compatibility * 0.4 + audienceMatch * 0.3 + marketMatch * 0.2 + schoolMatch * 0.1;
   325→   325→  }
   326→   326→
   327→   327→  const sorted = (Object.entries(scores) as [StyleId, number][]).sort((a, b) => b[1] - a[1]);
   328→   328→  const [pId, pScore] = sorted[0];
   329→   329→  const [sId, sScore] = sorted[1];
   330→   330→  const pDef = STYLES.find(s => s.id === pId)!;
   331→   331→  const sDef = STYLES.find(s => s.id === sId)!;
   332→   332→
   333→   333→  return {
   334→   334→    primary: {
   335→   335→      id: pId, name: pDef.name, nameEn: pDef.nameEn,
   336→   336→      score: Math.round(pScore * 100) / 100,
   337→   337→      detail_level: pDef.detailLevel, length: pDef.length, focus: pDef.focus,
   338→   338→      reasoning: buildStyleReasoning(pDef, primarySchoolId, ctx),
   339→   339→    },
   340→   340→    secondary: { id: sId, name: sDef.name, score: Math.round(sScore * 100) / 100 },
   341→   341→  };
   342→   342→}
   343→   343→
   344→   344→function getAudienceMatch(styleId: StyleId, audience: string): number {
   345→   345→  const map: Record<string, Record<StyleId, number>> = {
   346→   346→    pro:    { executive: 0.9, analytical: 1.0, forecasting: 0.8, trading: 0.9, educational: 0.3 },
   347→   347→    mid:    { executive: 0.7, analytical: 0.8, forecasting: 0.7, trading: 0.8, educational: 0.6 },
   348→   348→    begin:  { executive: 0.4, analytical: 0.5, forecasting: 0.4, trading: 0.5, educational: 1.0 },
   349→   349→  };
   350→   350→  const level = audience.includes('begin') || audience.includes('مبتدی') ? 'begin'
   351→   351→    : audience.includes('mid') || audience.includes('متوسط') ? 'mid' : 'pro';
   352→   352→  return map[level]?.[styleId] ?? 0.6;
   353→   353→}
   354→   354→
   355→   355→function getMarketContextMatch(styleId: StyleId, ctx: MSLV4Context): number {
   356→   356→  let score = 0.5;
   357→   357→  const isBull = ctx.regime.includes('Bull');
   358→   358→  const isBear = ctx.regime.includes('Bear');
   359→   359→  if (styleId === 'executive' && ctx.confidence > 0.7) score = 0.9;
   360→   360→  if (styleId === 'trading' && (isBull || isBear) && ctx.trend_strength > 0.5) score = 0.9;
   361→   361→  if (styleId === 'forecasting' && ctx.confidence > 0.5) score = 0.8;
   362→   362→  if (styleId === 'analytical' && ctx.confidence > 0.5) score = 0.85;
   363→   363→  if (styleId === 'educational' && ctx.confidence < 0.5) score = 0.85;
   364→   364→  return score;
   365→   365→}

   367→   367→function buildStyleReasoning(s: StyleDef, schoolId: SchoolId, ctx: MSLV4Context): string[] {
   368→   368→  const r: string[] = [];
   369→   369→  const schoolName = SCHOOLS.find(x => x.id === schoolId)?.name ?? '';
   370→   370→  r.push(`مکتب ${schoolName} نیاز به تحلیل ${s.focus === 'decision' ? 'تصمیم‌محور' : s.focus === 'analysis' ? 'دقیق' : s.focus === 'forecast' ? 'پیش‌بینی' : s.focus === 'action' ? 'عملیاتی' : 'آموزشی'} دارد`);
   371→   371→  if (ctx.confidence > 0.7) r.push(`اطمینان کلی ${toPersianNum(ctx.confidence)} (بالا)`);
   372→   372→  else if (ctx.confidence > 0.5) r.push(`اطمینان کلی ${toPersianNum(ctx.confidence)} (متوسط)`);
   373→   373→  return r;
   374→   374→}

   380→   380→function selectTones(schoolId: SchoolId, styleId: StyleId, ctx: MSLV4Context) {
   381→   381→  const scores: Record<ToneId, number> = {} as Record<ToneId, number>;
   382→   382→
   383→   383→  for (const tone of TONES) {
   384→   384→    const condMatch = checkToneConditions(tone.id, ctx);
   385→   385→    const schoolMatch = ST_MATRIX[schoolId]?.[tone.id] ?? 0.5;
   386→   386→    const styleMatch = STYLE_TONE_MATRIX[styleId]?.[tone.id] ?? 0.5;
   387→   387→    const intensity = 0.5 + ctx.confidence * 0.3 + ctx.volatility * 0.2;
   388→   388→    scores[tone.id] = condMatch * 0.5 + schoolMatch * 0.2 + styleMatch * 0.2 + Math.min(intensity, 1) * 0.1;
   389→   389→  }
   390→   390→
   391→   391→  const sorted = (Object.entries(scores) as [ToneId, number][]).sort((a, b) => b[1] - a[1]);
   392→   392→  const [pId, pScore] = sorted[0];
   393→   393→  const [sId, sScore] = sorted[1];
   394→   394→  const pDef = TONES.find(t => t.id === pId)!;
   395→   395→  const sDef = TONES.find(t => t.id === sId)!;
   396→   396→  const intensity = calcIntensity(pId, ctx);
   397→   397→
   398→   398→  return {
   399→   399→    primary: {
   400→   400→      id: pId, name: pDef.name, nameEn: pDef.nameEn,
   401→   401→      score: Math.round(pScore * 100) / 100,
   402→   402→      intensity, keywords: pDef.keywords.slice(0, 3),
   403→   403→      reasoning: buildToneReasoning(pDef, ctx),
   404→   404→    },
   405→   405→    secondary: { id: sId, name: sDef.name, score: Math.round(sScore * 100) / 100 },
   406→   406→  };
   407→   407→}
   408→   408→
   409→   409→function checkToneConditions(toneId: ToneId, ctx: MSLV4Context): number {
   410→   410→  const isBull = ctx.regime.includes('Bull');
   411→   411→  const isBear = ctx.regime.includes('Bear');
   412→   412→  switch (toneId) {
   413→   413→    case 'conservative': return ctx.confidence < 0.6 ? 0.9 : ctx.volatility > 0.03 ? 0.7 : 0.4;
   414→   414→    case 'aggressive': return ctx.confidence > 0.8 && isBull ? 0.9 : ctx.confidence > 0.7 ? 0.6 : 0.3;
   415→   415→    case 'balanced': return ctx.confidence >= 0.6 && ctx.confidence <= 0.8 ? 0.9 : 0.5;
   416→   416→    case 'warning': return ctx.divergence_present ? 0.9 : isBear && ctx.volatility > 0.03 ? 0.8 : 0.3;
   417→   417→    case 'optimistic': return isBull && ctx.confidence > 0.7 ? 0.9 : isBull ? 0.5 : 0.2;
   418→   418→    case 'realistic': return ctx.confidence >= 0.4 && ctx.confidence <= 0.7 ? 0.9 : 0.5;
   419→   419→    default: return 0.5;
   420→   420→  }
   421→   421→}
   422→   422→
   423→   423→function calcIntensity(toneId: ToneId, ctx: MSLV4Context): string {
   424→   424→  const base = ctx.confidence + ctx.volatility;
   425→   425→  if (toneId === 'warning') return base > 0.9 ? 'high' : base > 0.6 ? 'medium' : 'low';
   426→   426→  if (toneId === 'aggressive' || toneId === 'optimistic') return base > 0.85 ? 'high' : base > 0.55 ? 'medium' : 'low';
   427→   427→  if (toneId === 'conservative') return ctx.volatility > 0.03 ? 'high' : ctx.volatility > 0.015 ? 'medium' : 'low';
   428→   428→  return base > 0.8 ? 'medium-high' : base > 0.5 ? 'medium' : 'low';
   429→   429→}

   431→   431→function buildToneReasoning(t: ToneDef, ctx: MSLV4Context): string[] {
   432→   432→  const r: string[] = [];
   433→   433→  r.push(`اطمینان کلی = ${toPersianNum(ctx.confidence)}`);
   434→   434→  r.push(`رژیم بازار = ${ctx.regime}`);
   435→   435→  r.push(`نوسان = ${ctx.volatility > 0.03 ? 'بالا' : ctx.volatility > 0.01 ? 'متوسط' : 'پایین'} (ATR/Price = ${toPersianNum(ctx.volatility)})`);
   436→   436→  if (ctx.divergence_present) r.push('واگرایی تشخیص داده شده');
   437→   437→  return r;
   438→   438→}
