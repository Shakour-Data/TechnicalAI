   982→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   983→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   984→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   985→  grade?: string; overlapCount?: number; touchCount?: number;
   986→}) {
   987→  const isSupport = type === 'support';
   988→  const isTargetType = type === 'target';
   989→
   990→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   991→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   992→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   993→
   994→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   995→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   996→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   997→
   998→  return (
   999→    <div
  1000→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
  1001→      style={{
  1002→        background: bgAccent,
  1003→        border: `1px solid ${borderAccent}`,
  1004→        borderRight: `3px solid ${accentColor}`,
  1005→      }}
  1006→    >
  1007→      {/* Label + Price */}
  1008→      <div className="flex-1 min-w-0">
  1009→        <div className="flex items-center gap-2">
  1010→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
  1011→          {isTarget && <span className="text-[9px]">🎯</span>}
  1012→          {overlapCount !== undefined && overlapCount > 0 && (
  1013→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
  1014→              {toPersianDigits(overlapCount)} هم‌پوشانی
  1015→            </span>
  1016→          )}
  1017→        </div>
  1018→        <div className="flex items-baseline gap-1">
  1019→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
  1020→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
  1021→        </div>
  1022→      </div>
  1023→
  1024→      {/* Strength + Grade */}
  1025→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
  1026→        <div className="flex items-center gap-1.5">
  1027→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
  1028→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
  1029→          <span className="text-[9px] text-gray-500">/۱۰</span>
  1030→        </div>
  1031→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
  1032→          <div
  1033→            className="h-full rounded-full transition-all duration-500"
  1034→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
  1035→          />
  1036→        </div>
  1037→        {touchCount !== undefined && touchCount > 0 && (
  1038→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
  1039→        )}
  1040→      </div>
  1041→    </div>
  1042→  );
  1043→}
