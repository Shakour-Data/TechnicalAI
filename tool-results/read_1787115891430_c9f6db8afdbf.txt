   122→  { key: 'up', label: 'صعودی' },

   131→// Component
   132→// ═══════════════════════════════════════════════════════════════════════════════
   133→
   134→export default function VdssGraph(props: VdssGraphProps) {
   135→  const { symbolName, currentPrice, resistances, supports, ma100, scenarios, rsi, adx, bullScore, edgeWeights, currencyUnit, overallSignal } = props;
   136→
   137→  // ── Dynamic scenario labels (context-aware) ──────────────────
   138→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal || 'neutral'), [overallSignal]);
   139→
   140→  // ── Dynamic NODE_DEFS: R1-R5 titles/descriptions adapt to signal ──
   141→  const nodeDefs = useMemo(() => ({
   142→    ...NODE_DEFS,
   143→    R1: { ...NODE_DEFS.R1, title: dynamicMeta.R1.nodeTitle, desc: dynamicMeta.R1.nodeDesc },
   144→    R2: { ...NODE_DEFS.R2, title: dynamicMeta.R2.nodeTitle, desc: dynamicMeta.R2.nodeDesc },
   145→    R3: { ...NODE_DEFS.R3, title: dynamicMeta.R3.nodeTitle, desc: dynamicMeta.R3.nodeDesc },
   146→    R4: { ...NODE_DEFS.R4, title: dynamicMeta.R4.nodeTitle, desc: dynamicMeta.R4.nodeDesc },
   147→    R5: { ...NODE_DEFS.R5, title: dynamicMeta.R5.nodeTitle, desc: dynamicMeta.R5.nodeDesc },
   148→  }), [dynamicMeta]);
   149→  const unit = currencyUnit || 'ریال';
   150→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   151→  const S1_level = supports[0] ?? currentPrice * 0.95;
   152→
   153→  const [activeFilter, setActiveFilter] = useState('all');
   154→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   155→  const graphRef = useRef<HTMLDivElement>(null);
   156→  const svgRef = useRef<SVGSVGElement>(null);
   157→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   158→  const shellRef = useRef<HTMLDivElement>(null);
   159→
   160→  // ── Edge probabilities (using TA engine's bullScore) ──────────────
   161→  const edgeProbs = useMemo(() =>
   162→    calcEdgeProbabilities(edgeWeights),
   163→  [edgeWeights]
   164→  );
   165→
   166→  // ── All paths with raw probabilities ─────────────────────────────
   167→  const allPathsRaw = useMemo(() => findAllPaths(edgeProbs), [edgeProbs]);
   168→
   169→  // ── Calibrate path probabilities to match TA engine scenario probabilities ──
   170→  const allPaths = useMemo(() => {
   171→    // Compute raw sum per target
   172→    const rawSums: Record<string, number> = {};
   173→    for (const key of SCENARIO_KEYS) rawSums[key] = 0;
   174→    for (const p of allPathsRaw) rawSums[p.target] += p.prob;
   175→
   176→    // Scale each path so the sum per target matches scenario probability
   177→    return allPathsRaw.map(p => {
   178→      const rawSum = rawSums[p.target] || 0.0001;
   179→      const targetProb = scenarios[p.target as keyof typeof scenarios].probability / 100;
   180→      const scale = targetProb / rawSum;
   181→      return { ...p, prob: p.prob * scale };
   182→    });
   183→  }, [allPathsRaw, scenarios]);
   184→
   185→  // ── Filtered paths ────────────────────────────────────────────────
   186→  const filteredPaths = useMemo(() => {
   187→    if (activeFilter === 'all') return allPaths;
   188→    // Check if filter is a scenario key
   189→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   190→      return allPaths.filter(p => p.target === activeFilter);
   191→    }
   192→    // Type filter: paths must use only edges of this type
   193→    return allPaths.filter(p =>
   194→      p.edges.every(ei => EDGES[ei][3] === activeFilter)
   195→    );
   196→  }, [allPaths, activeFilter]);
   197→
   198→  // ── Visible edges/nodes based on filter ────────────────────────────
   199→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   200→    if (activeFilter === 'all') {
   201→      const allEdgeIdx = EDGES.map((_, i) => i);
   202→      const allNodes = new Set(Object.keys(NODE_POSITIONS));
   203→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   204→    }
   205→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   206→      // Show all edges/nodes on paths to this scenario
   207→      const edgeSet = new Set<number>();
   208→      const nodeSet = new Set<string>();
   209→      for (const p of filteredPaths) {
   210→        p.edges.forEach(ei => edgeSet.add(ei));
   211→        p.nodes.forEach(n => nodeSet.add(n));
   212→      }
   213→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   214→    }
   215→    // Type filter
   216→    const edgeSet = new Set<number>();
   217→    const nodeSet = new Set<string>();
   218→    EDGES.forEach((e, i) => {
   219→      if (e[3] === activeFilter) {
   220→        edgeSet.add(i);
   221→        nodeSet.add(e[0]);
   222→        nodeSet.add(e[1]);
   223→      }
   224→    });
   225→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   226→  }, [activeFilter, filteredPaths]);
   227→
   228→  // ── Node values ──────────────────────────────────────────────────
   229→  const nodeValues: Record<string, string> = {
   230→    A: toFa(currentPrice),
   231→    B: toFa(R1_level),
   232→    D: toFa(resistances[1] ?? currentPrice * 1.10),
   233→    E: `${toFa(Math.round(R1_level + (resistances[1] ?? R1_level * 1.05 - R1_level) * 0.5))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 0.8))}`,
   234→    F: `${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 1.2))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 2.0))}`,
   235→    G: toFa(S1_level),
   236→    H: toFa(supports[1] ?? currentPrice * 0.90),
   237→    I: toFa(supports[2] ?? currentPrice * 0.85),
   238→    J: toFa(supports[3] ?? currentPrice * 0.80),
   239→    K: toFa(ma100),
   240→    L: props.trendDirection === 'up' ? 'صعودی' : props.trendDirection === 'down' ? 'نزولی' : 'خنثی',
   241→  };
   242→  for (const key of SCENARIO_KEYS) {
   243→    const s = scenarios[key];
   244→    nodeValues[key] = `${toFa(s.targetMin)} — ${toFa(s.targetMax)} ${unit}`;
   245→  }
   246→
   247→  // ── Draw SVG edges ───────────────────────────────────────────────
   248→  const drawEdges = useCallback(() => {
   249→    const svg = svgRef.current;
   250→    const graph = graphRef.current;
   251→    if (!svg || !graph) return;
   252→
   253→    let markersSvg = '';
   254→    for (const [k, c] of Object.entries(EDGE_COLORS)) {
   255→      markersSvg += `<marker id="arrow-${k}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   256→    }
   257→
   258→    let pathsSvg = '';
   259→    const graphRect = graph.getBoundingClientRect();
   260→
   261→    for (let ei = 0; ei < EDGES.length; ei++) {
   262→      const [fromId, toId, label, type] = EDGES[ei];
   263→      const isVisible = visibleEdgeIndices.includes(ei);
   264→      const fromEl = nodeRefs.current[fromId];
   265→      const toEl = nodeRefs.current[toId];
   266→      if (!fromEl || !toEl) continue;
   267→
   268→      const fRect = fromEl.getBoundingClientRect();
   269→      const tRect = toEl.getBoundingClientRect();
   270→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   271→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   272→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   273→      const by = tRect.top - graphRect.top + tRect.height / 2;
   274→      const dx = bx - ax;
   275→      const dy = by - ay;
   276→      const dist = Math.sqrt(dx * dx + dy * dy);
   277→      if (dist === 0) continue;
   278→
   279→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   280→      const mx = (ax + bx) / 2;
   281→      const my = (ay + by) / 2;
   282→      const nx = -dy / dist;
   283→      const ny = dx / dist;
   284→      const cx = mx + nx * bend;
   285→      const cy = my + ny * bend;
   286→
   287→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   288→      const prob = edgeProbs[String(ei)] ?? 0;
   289→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   290→
   291→      pathsSvg += `<path d="${d}" stroke="${EDGE_COLORS[type]}" stroke-width="${isVisible ? 2 : 1}" opacity="${isVisible ? 0.72 : 0.06}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   292→
   293→      // Edge probability label (only for visible edges)
   294→      if (isVisible) {
   295→        pathsSvg += `<text x="${cx}" y="${cy - 6}" fill="#cde4ef" font-size="9" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3.5" stroke-linejoin="round" opacity="0.85" data-type="${type}" class="edge-label">${label}</text>`;
   296→        pathsSvg += `<text x="${cx}" y="${cy + 8}" fill="${EDGE_COLORS[type]}" font-size="10" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   297→      }
   298→    }
   299→
   300→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   301→  }, [visibleEdgeIndices, edgeProbs]);
   302→
   303→  useEffect(() => { drawEdges(); }, [drawEdges]);
   304→  useEffect(() => {
   305→    const el = shellRef.current;
   306→    if (!el) return;
   307→    const observer = new ResizeObserver(() => drawEdges());
   308→    observer.observe(el);
   309→    return () => observer.disconnect();
   310→  }, [drawEdges]);
   311→
   312→  // ── Aggregated path probabilities per target (calibrated = matches scenario prob) ──
   313→  const pathProbsByTarget = useMemo(() => {
   314→    const result: Record<string, number> = {};
   315→    for (const key of SCENARIO_KEYS) result[key] = 0;
   316→    for (const p of filteredPaths) {
   317→      result[p.target] = (result[p.target] ?? 0) + p.prob;
   318→    }
   319→    // After calibration, these should match scenario probabilities
   320→    return result;
   321→  }, [filteredPaths]);
   322→
   323→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   324→
   325→  // ── Detail panel content ─────────────────────────────────────────
   326→  const detailContent = useMemo(() => {
   327→    if (!selectedNode) return null;
   328→    const def = nodeDefs[selectedNode];
   329→    if (!def) return null;
   330→    const relations = EDGES.filter(e => e[0] === selectedNode || e[1] === selectedNode);
   331→    const inputs = relations.filter(e => e[1] === selectedNode);
   332→    const outputs = relations.filter(e => e[0] === selectedNode);
   333→
   334→    return (
   335→      <div className="space-y-3">
   336→        <h3 className="text-sm font-bold text-gray-100">{def.title}</h3>
   337→        <p className="text-xs text-gray-300"><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   338→        <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border border-white/10 bg-white/5 text-gray-300">{def.type}</span>
   339→        <p className="text-xs text-gray-400 leading-relaxed">{def.desc}</p>
   340→        {inputs.length > 0 && (
   341→          <div>
   342→            <p className="text-xs font-medium text-gray-300 mb-1">مسیرهای ورودی ({inputs.length}):</p>
   343→            <ul className="space-y-1">
   344→              {inputs.map((e, i) => {
   345→                const ei = EDGES.indexOf(e);
   346→                const ep = edgeProbs[String(ei)] ?? 0;
   347→                return (
   348→                  <li key={i} className="text-[11px] text-gray-500 leading-relaxed border-t border-dashed border-white/10 pt-1.5">
   349→                    <b className="text-gray-400">{e[0]} ← {e[1]}</b>
   350→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${EDGE_COLORS[e[3]]}20`, color: EDGE_COLORS[e[3]] }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   351→                    <br />{e[2]}
   352→                  </li>
   353→                );
   354→              })}
   355→            </ul>
   356→          </div>
   357→        )}
   358→        {outputs.length > 0 && (
   359→          <div>
   360→            <p className="text-xs font-medium text-gray-300 mb-1">مسیرهای خروجی ({outputs.length}):</p>
   361→            <ul className="space-y-1">
   362→              {outputs.map((e, i) => {
   363→                const ei = EDGES.indexOf(e);
   364→                const ep = edgeProbs[String(ei)] ?? 0;
   365→                return (
   366→                  <li key={i} className="text-[11px] text-gray-500 leading-relaxed border-t border-dashed border-white/10 pt-1.5">
   367→                    <b className="text-gray-400">{e[0]} → {e[1]}</b>
   368→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${EDGE_COLORS[e[3]]}20`, color: EDGE_COLORS[e[3]] }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   369→                    <br />{e[2]}
   370→                  </li>
   371→                );
   372→              })}
   373→            </ul>
   374→          </div>
   375→        )}
   376→      </div>
   377→    );
   378→  }, [selectedNode, nodeValues, edgeProbs]);
   379→
   380→  // ── All filter buttons ────────────────────────────────────────────
   381→  const allFilters = [
   382→    ...TYPE_FILTERS,
   383→    { key: 'sep1', label: '│', isSep: true as const },
