   106→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   107→function sma(closes: number[], period: number): number {
   108→  if (closes.length < period) return 0;
   109→  const slice = closes.slice(-period);
   110→  return slice.reduce((a, b) => a + b, 0) / period;
   111→}
   112→
   113→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   114→function emaCalc(closes: number[], period: number): number {
   115→  if (closes.length < period) return 0;
   116→  const k = 2 / (period + 1);
   117→  let emaVal = sma(closes.slice(0, period), period);
   118→  for (let i = period; i < closes.length; i++) {
   119→    emaVal = closes[i] * k + emaVal * (1 - k);
   120→  }
   121→  return emaVal;
   122→}

   124→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   125→function calcRSI(closes: number[], period: number = 14): number {
   126→  if (closes.length < period + 1) return 50;
   127→  let avgGain = 0;
   128→  let avgLoss = 0;
   129→  for (let i = 1; i <= period; i++) {
   130→    const diff = closes[i] - closes[i - 1];
   131→    if (diff > 0) avgGain += diff;
   132→    else avgLoss += Math.abs(diff);
   133→  }
   134→  avgGain /= period;
   135→  avgLoss /= period;
   136→  for (let i = period + 1; i < closes.length; i++) {
   137→    const diff = closes[i] - closes[i - 1];
   138→    const gain = diff > 0 ? diff : 0;
   139→    const loss = diff < 0 ? Math.abs(diff) : 0;
   140→    avgGain = (avgGain * (period - 1) + gain) / period;
   141→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   142→  }
   143→  if (avgLoss === 0) return 100;
   144→  const rs = avgGain / avgLoss;
   145→  return 100 - 100 / (1 + rs);
   146→}

   148→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   149→function calcMFI(data: OHLCV[], period: number = 14): number {
   150→  if (data.length < period + 1) return 50;
   151→  let posFlow = 0;
   152→  let negFlow = 0;
   153→  for (let i = data.length - period; i < data.length; i++) {
   154→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   155→    const mf = tp * data[i].volume;
   156→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   157→    if (tp > prevTp) posFlow += mf;
   158→    else negFlow += mf;
   159→  }
   160→  if (negFlow === 0) return 100;
   161→  const mfr = posFlow / negFlow;
   162→  return 100 - 100 / (1 + mfr);
   163→}

   165→// ─── Helper: CCI(20) ──────────────────────────────────────────────────────────
   166→function calcCCI(data: OHLCV[], period: number = 20): number {
   167→  if (data.length < period) return 0;
   168→  const tps: number[] = [];
   169→  for (let i = data.length - period; i < data.length; i++) {
   170→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   171→  }
   172→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   173→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   174→  if (meanDev === 0) return 0;
   175→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   176→}

   178→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   179→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   180→  if (data.length < kPeriod) return { k: 50, d: 50 };
   181→  const rawKs: number[] = [];
   182→  for (let i = kPeriod - 1; i < data.length; i++) {
   183→    let lowest = Infinity;
   184→    let highest = -Infinity;
   185→    for (let j = i - kPeriod + 1; j <= i; j++) {
   186→      if (data[j].low < lowest) lowest = data[j].low;
   187→      if (data[j].high > highest) highest = data[j].high;
   188→    }
   189→    const range = highest - lowest;
   190→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   191→  }
   192→  const smoothedK: number[] = [];
   193→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   194→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   195→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   196→  }
   197→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   198→  const dSlice = smoothedK.slice(-smoothD);
   199→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   200→  return { k, d };
   201→}

   203→// ─── Helper: Williams %R(14) ──────────────────────────────────────────────────
   204→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   205→  if (data.length < period) return -50;
   206→  const slice = data.slice(-period);
   207→  let highest = -Infinity;
   208→  let lowest = Infinity;
   209→  for (const c of slice) {
   210→    if (c.high > highest) highest = c.high;
   211→    if (c.low < lowest) lowest = c.low;
   212→  }
   213→  const range = highest - lowest;
   214→  if (range === 0) return -50;
   215→  return ((highest - data[data.length - 1].close) / range) * -100;
   216→}

   218→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   219→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   220→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   221→  const emaFast = emaCalc(closes, fast);
   222→  const emaSlow = emaCalc(closes, slow);
   223→  const macdLine = emaFast - emaSlow;
   224→  // Calculate signal as EMA of MACD values
   225→  const macdValues: number[] = [];
   226→  const kFast = 2 / (fast + 1);
   227→  const kSlow = 2 / (slow + 1);
   228→  let eFast = sma(closes.slice(0, fast), fast);
   229→  let eSlow = sma(closes.slice(0, slow), slow);
   230→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   231→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   232→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   233→    macdValues.push(eFast - eSlow);
   234→  }
   235→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   236→  const kSig = 2 / (sig + 1);
   237→  let signalVal = sma(macdValues.slice(0, sig), sig);
   238→  for (let i = sig; i < macdValues.length; i++) {
   239→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   240→  }
   241→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   242→}

   244→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   245→function calcATR(data: OHLCV[], period: number = 14): number {
   246→  if (data.length < 2) return 0;
   247→  const trs: number[] = [];
   248→  for (let i = 1; i < data.length; i++) {
   249→    const tr = Math.max(
   250→      data[i].high - data[i].low,
   251→      Math.abs(data[i].high - data[i - 1].close),
   252→      Math.abs(data[i].low - data[i - 1].close)
   253→    );
   254→    trs.push(tr);
   255→  }
   256→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   257→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   258→  for (let i = period; i < trs.length; i++) {
   259→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   260→  }
   261→  return atrVal;
   262→}

   264→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   265→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   266→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   267→  const plusDMs: number[] = [];
   268→  const minusDMs: number[] = [];
   269→  const trs: number[] = [];
   270→  for (let i = 1; i < data.length; i++) {
   271→    const upMove = data[i].high - data[i - 1].high;
   272→    const downMove = data[i - 1].low - data[i].low;
   273→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   274→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   275→    trs.push(Math.max(
   276→      data[i].high - data[i].low,
   277→      Math.abs(data[i].high - data[i - 1].close),
   278→      Math.abs(data[i].low - data[i - 1].close)
   279→    ));
   280→  }
   281→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   282→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   283→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   284→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   285→  const dxValues: number[] = [];
   286→  for (let i = period; i < trs.length; i++) {
   287→    smoothTR = smoothTR - smoothTR / period + trs[i];
   288→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   289→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   290→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   291→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   292→    const diSum = diPlus + diMinus;
   293→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   294→    dxValues.push(dx);
   295→  }
   296→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   297→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   298→  for (let i = period; i < dxValues.length; i++) {
   299→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   300→  }
   301→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   302→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   303→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   304→}

   306→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   307→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   308→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   309→  let isLong = true;
   310→  let af = step;
   311→  let ep = data[0].high;
   312→  let sar = data[0].low;
   313→  for (let i = 1; i < data.length; i++) {
   314→    const prevSar = sar;
   315→    sar = prevSar + af * (ep - prevSar);
   316→    // Ensure SAR is within bounds
   317→    if (isLong) {
   318→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   319→      if (data[i].low < sar) {
   320→        isLong = false;
   321→        sar = ep;
   322→        ep = data[i].low;
   323→        af = step;
   324→      } else {
   325→        if (data[i].high > ep) {
   326→          ep = data[i].high;
   327→          af = Math.min(af + step, max);
   328→        }
   329→      }
   330→    } else {
   331→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   332→      if (data[i].high > sar) {
   333→        isLong = true;
   334→        sar = ep;
   335→        ep = data[i].high;
   336→        af = step;
   337→      } else {
   338→        if (data[i].low < ep) {
   339→          ep = data[i].low;
   340→          af = Math.min(af + step, max);
   341→        }
   342→      }
   343→    }
   344→  }
   345→  return sar;
   346→}

   348→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   349→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   350→  if (closes.length < period) {
   351→    const c = closes[closes.length - 1] ?? 0;
   352→    return { upper: c, middle: c, lower: c };
   353→  }
   354→  const slice = closes.slice(-period);
   355→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   356→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   357→  const sd = Math.sqrt(variance);
   358→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   359→}

   362→function calcOBV(data: OHLCV[]): number {

   371→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   372→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   373→  const n = values.length;
   374→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   375→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   376→  for (let i = 0; i < n; i++) {
   377→    sumX += i;
   378→    sumY += values[i];
   379→    sumXY += i * values[i];
   380→    sumX2 += i * i;
   381→    sumY2 += values[i] * values[i];
   382→  }
   383→  const denom = n * sumX2 - sumX * sumX;
   384→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   385→  const slope = (n * sumXY - sumX * sumY) / denom;
   386→  const intercept = (sumY - slope * sumX) / n;
   387→  const meanY = sumY / n;
   388→  let ssRes = 0, ssTot = 0;
   389→  for (let i = 0; i < n; i++) {
   390→    const predicted = slope * i + intercept;
   391→    ssRes += (values[i] - predicted) ** 2;
   392→    ssTot += (values[i] - meanY) ** 2;
   393→  }
   394→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   395→  return { slope, intercept, r2 };
   396→}

   398→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   399→function roundToNice(price: number): number {
   400→  const abs = Math.abs(price);
   401→  let step: number;
   402→  if (abs >= 100000) step = 100;
   403→  else if (abs >= 10000) step = 50;
   404→  else if (abs >= 1000) step = 10;
   405→  else if (abs >= 100) step = 5;
   406→  else step = 1;
   407→  return Math.round(price / step) * step;
   408→}

   411→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: { price: number; idx: number }[]; lows: { price: number; idx: number }[] } {
   412→  const highs: { price: number; idx: number }[] = [];
   413→  const lows: { price: number; idx: number }[] = [];
   414→  for (let i = lookback; i < data.length - lookback; i++) {
   415→    let isHigh = true;
   416→    let isLow = true;
   417→    for (let j = i - lookback; j <= i + lookback; j++) {
   418→      if (j === i) continue;
   419→      if (data[j].high >= data[i].high) isHigh = false;
   420→      if (data[j].low <= data[i].low) isLow = false;
   421→    }
   422→    if (isHigh) highs.push({ price: data[i].high, idx: i });
   423→    if (isLow) lows.push({ price: data[i].low, idx: i });
   424→  }
   425→  return { highs, lows };
   426→}

   472→  let bestSwing: SwingPair | null = null;
   473→
   474→  for (const lookback of [3, 5, 7, 10, 15, 20]) {
   475→    const swings = findSwingLevels(data, lookback);
   476→    if (swings.highs.length === 0 || swings.lows.length === 0) continue;
   477→
   478→    // Try all combinations of a high followed by a low, or low followed by a high
   479→    for (const h of swings.highs) {
   480→      for (const l of swings.lows) {
   481→        if (Math.abs(h.idx - l.idx) < 3) continue; // Too close together
   482→        const range = Math.abs(h.price - l.price);
   483→        const relRange = range / currentPrice;
   484→        if (relRange < 0.03) continue; // Ignore tiny swings (< 3% of price)
   485→
   486→        if (!bestSwing || range > bestSwing.range) {
   487→          bestSwing = {
   488→            high: Math.max(h.price, l.price),
   489→            highIdx: h.price > l.price ? h.idx : l.idx,
   490→            low: Math.min(h.price, l.price),
   491→            lowIdx: h.price < l.price ? h.idx : l.idx,
   492→            range,
   493→          };
   494→        }
   495→      }
   496→    }
   497→  }
   498→
   499→  // ── Step 2: Also find secondary swings for confluence ──
   500→  const secondarySwings: SwingPair[] = [];
   501→  for (const lookback of [3, 5, 7, 10, 15]) {
   502→    const swings = findSwingLevels(data, lookback);
   503→    for (const h of swings.highs) {
   504→      for (const l of swings.lows) {
   505→        if (Math.abs(h.idx - l.idx) < 3) continue;
   506→        const range = Math.abs(h.price - l.price);
   507→        const relRange = range / currentPrice;
   508→        if (relRange < 0.03) continue;
   509→        if (bestSwing && Math.abs(range - bestSwing.range) / bestSwing.range < 0.1) continue; // Skip duplicates of best
   510→        if (range > currentPrice * 0.05) { // Only significant secondary swings
   511→          secondarySwings.push({
   512→            high: Math.max(h.price, l.price),
   513→            highIdx: h.price > l.price ? h.idx : l.idx,
   514→            low: Math.min(h.price, l.price),
   515→            lowIdx: h.price < l.price ? l.idx : h.idx,
   516→            range,
   517→          });
   518→        }
   519→      }
   520→    }
   521→  }
   522→  // Keep top 3 secondary swings by range
   523→  secondarySwings.sort((a, b) => b.range - a.range);
   524→  const topSecondary = secondarySwings.slice(0, 3);
   525→
   526→  // ── Step 3: Generate ALL Fibonacci levels from the best swing ──
   527→  interface FibLevel {
   528→    price: number;
   529→    ratio: number;
   530→    ratioStr: string;
   531→    label: string;
   532→    source: 'retracement' | 'extension' | 'retracement-inv';
   533→    confluenceCount: number;
   534→  }
   535→
   536→  const allLevels: FibLevel[] = [];
   537→
   538→  if (bestSwing) {
   539→    const H = bestSwing.high;
   540→    const L = bestSwing.low;
   541→    const range = H - L;
   542→
   543→    // Fibonacci Retracement levels (measured from high downward)
   544→    // 0% = H, 100% = L
   545→    for (const ratio of FIB_RETRACEMENTS) {
   546→      const price = H - ratio * range;
   547→      if (price > 0) {
   548→        allLevels.push({
   549→          price: roundToNice(price),
   550→          ratio,
   551→          ratioStr: String(ratio),
   552→          label: FIB_LABELS[String(ratio)] || `${(ratio * 100).toFixed(1)}٪`,
   553→          source: 'retracement',
   554→          confluenceCount: 1,
   555→        });
   556→      }
   557→    }
   558→
   559→    // Fibonacci Extension levels (projected above H)
   560→    for (const extRatio of FIB_EXTENSIONS) {
   561→      const price = H + (extRatio - 1) * range;
   562→      if (price > 0) {
   563→        allLevels.push({
   564→          price: roundToNice(price),
   565→          ratio: extRatio,
   566→          ratioStr: String(extRatio),
   567→          label: FIB_LABELS[String(extRatio)] || `${(extRatio * 100).toFixed(1)}٪`,
   568→          source: 'extension',
   569→          confluenceCount: 1,
   570→        });
   571→      }
   572→    }
   573→
   574→    // Fibonacci Extension levels (projected below L — inverse extensions)
   575→    for (const extRatio of FIB_EXTENSIONS) {
   576→      const price = L - (extRatio - 1) * range;
   577→      if (price > 0) {
   578→        allLevels.push({
   579→          price: roundToNice(price),
   580→          ratio: -(extRatio),
   581→          ratioStr: String(-extRatio),
   582→          label: `-${FIB_LABELS[String(extRatio)] || `${(extRatio * 100).toFixed(1)}٪`}`,
   583→          source: 'retracement-inv',
   584→          confluenceCount: 1,
   585→        });
   586→      }
   587→    }
   588→  }
   589→
   590→  // ── Step 4: Generate levels from secondary swings and check confluence ──
   591→  const confluenceThreshold = currentPrice * 0.005; // 0.5%
   592→
   593→  for (const sec of topSecondary) {
   594→    const H = sec.high;
   595→    const L = sec.low;
   596→    const range = H - L;
   597→
   598→    for (const ratio of FIB_RETRACEMENTS) {
   599→      const price = roundToNice(H - ratio * range);
   600→      if (price <= 0) continue;
   601→      // Check if this level is close to an existing one
   602→      for (const existing of allLevels) {
   603→        if (Math.abs(existing.price - price) < confluenceThreshold) {
   604→          existing.confluenceCount++;
   605→        }
   606→      }
   607→    }
   608→
   609→    for (const extRatio of FIB_EXTENSIONS) {
   610→      const price = roundToNice(H + (extRatio - 1) * range);
   611→      if (price <= 0) continue;
   612→      for (const existing of allLevels) {
   613→        if (Math.abs(existing.price - price) < confluenceThreshold) {
   614→          existing.confluenceCount++;
   615→        }
   616→      }
   617→    }
   618→  }
   619→
   620→  // ── Step 5: Score each level ──
   621→  const avgVolume = data.reduce((s, d) => s + d.volume, 0) / data.length;
   622→  const dataHasVolume = data.some(d => d.volume > 0);
   623→
   624→  const scoredLevels: LevelStrength[] = allLevels.map(lvl => {
   625→    // Factor 1: Confluence (multiple swings agree) — 0-3 pts
   626→    const confluenceScore = Math.min(3, lvl.confluenceCount * 1.0);
   627→
   628→    // Factor 2: Proximity to a KEY Fibonacci ratio — 0-2 pts
   629→    // Levels at key ratios (0.382, 0.5, 0.618, 1.618, 2.618) are stronger
   630→    let keyRatioScore = 0;
   631→    const absRatio = Math.abs(lvl.ratio);
   632→    for (const keyR of KEY_FIB_RATIOS) {
   633→      if (Math.abs(absRatio - keyR) < 0.01) {
   634→        keyRatioScore = 2.0;
   635→        break;
   636→      }
   637→    }
   638→    // 0% and 100% are the swing extremes — also strong
   639→    if (absRatio < 0.01 || Math.abs(absRatio - 1.0) < 0.01) keyRatioScore = 1.5;
   640→
   641→    // Factor 3: Proximity to current price — 0-2 pts
   642→    const distRatio = Math.abs(lvl.price - currentPrice) / currentPrice;
   643→    let proximityScore = 0;
   644→    if (distRatio < 0.01) proximityScore = 2.0;
   645→    else if (distRatio < 0.02) proximityScore = 1.7;
   646→    else if (distRatio < 0.04) proximityScore = 1.3;
   647→    else if (distRatio < 0.07) proximityScore = 1.0;
   648→    else if (distRatio < 0.10) proximityScore = 0.6;
   649→    else if (distRatio < 0.15) proximityScore = 0.3;
   650→    else proximityScore = 0.1;
   651→
   652→    // Factor 4: Touch/rejection count at this level — 0-2 pts
   653→    let touchScore = 0;
   654→    const touchThreshold = currentPrice * 0.002;
   655→    let touchCount = 0;
   656→    let totalVolAtTouches = 0;
   657→    for (let i = 0; i < data.length; i++) {
   658→      const bar = data[i];
   659→      const isAbove = lvl.price > currentPrice;
   660→      if (isAbove) {
   661→        if (bar.high >= lvl.price - touchThreshold && bar.close < lvl.price) {
   662→          touchCount++;
   663→          totalVolAtTouches += bar.volume;
   664→        }
   665→      } else {
   666→        if (bar.low <= lvl.price + touchThreshold && bar.close > lvl.price) {
   667→          touchCount++;
   668→          totalVolAtTouches += bar.volume;
   669→        }
   670→      }
   671→    }
   672→    if (touchCount >= 5) touchScore = 2.0;
   673→    else if (touchCount >= 3) touchScore = 1.5;
   674→    else if (touchCount >= 2) touchScore = 1.0;
   675→    else if (touchCount === 1) touchScore = 0.5;
   676→
   677→    // Factor 5: Volume at touches — 0-1 pt
   678→    let volumeTouchScore = 0;
   679→    if (dataHasVolume && touchCount > 0) {
   680→      const avgVolAtTouch = totalVolAtTouches / touchCount;
   681→      const volRatio = avgVolume > 0 ? avgVolAtTouch / avgVolume : 1;
   682→      if (volRatio > 2.0) volumeTouchScore = 1.0;
   683→      else if (volRatio > 1.5) volumeTouchScore = 0.7;
   684→      else if (volRatio > 1.0) volumeTouchScore = 0.4;
   685→    }
   686→
   687→    // Factor 6: Swing recency bonus — 0-1 pt
   688→    let recencyScore = 0;
   689→    if (bestSwing) {
   690→      const barsSinceSwing = data.length - 1 - Math.max(bestSwing.highIdx, bestSwing.lowIdx);
   691→      if (barsSinceSwing <= 5) recencyScore = 1.0;
   692→      else if (barsSinceSwing <= 15) recencyScore = 0.7;
   693→      else if (barsSinceSwing <= 30) recencyScore = 0.4;
   694→      else if (barsSinceSwing <= 60) recencyScore = 0.2;
   695→    }
   696→
   697→    const total = confluenceScore + keyRatioScore + proximityScore + touchScore + volumeTouchScore + recencyScore;
   698→    const strength = Math.min(10, Math.max(1, Math.round(total)));
   699→
   700→    return {
   701→      price: lvl.price,
   702→      strength,
   703→      isTarget: false,
   704→      fibRatio: lvl.ratioStr,
   705→      fibLabel: lvl.label,
   706→    };
   707→  });
   708→
   709→  // ── Step 6: Separate into above/below current price ──
   710→  const above = scoredLevels.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   711→  const below = scoredLevels.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   712→
   713→  // ── Step 7: Pick top 6 resistances and 6 supports ──
   714→  // Prefer levels with higher strength, but ensure we pick nearest levels too
   715→  // Use a composite score: strength * 2 + proximity bonus
   716→  function pickTop6(levels: LevelStrength[]): LevelStrength[] {
   717→    if (levels.length <= 6) return levels;
   718→
   719→    // Score each level: combine strength with proximity to current price
   720→    const scored = levels.map((l) => {
   721→      const dist = Math.abs(l.price - currentPrice) / currentPrice;
   722→      // Proximity bonus: closer = higher bonus (up to 3 pts)
   723→      const proxBonus = Math.max(0, 3 - dist * 20);
   724→      const compositeScore = l.strength * 2 + proxBonus;
   725→      return { level: l, compositeScore };
   726→    }); 
   727→
   728→    // Sort by composite score descending, take top 6
   729→    scored.sort((a, b) => b.compositeScore - a.compositeScore);
   730→    const picked = scored.slice(0, 6).map(s => s.level);
   731→
   732→    // Re-sort by price (nearest first)
   733→    const isAbove = picked[0]?.price > currentPrice;
   734→    if (isAbove) {
   735→      picked.sort((a, b) => a.price - b.price);
   736→    } else {
   737→      picked.sort((a, b) => b.price - a.price);
   738→    }
   739→
   740→    return picked;
   741→  }
   742→
   743→  const finalResistances = pickTop6(above);
   744→  const finalSupports = pickTop6(below);
   745→
   746→  // ── Step 8: Mark price targets (top 2 strongest across all levels) ──
   747→  const combinedLevels = [...finalResistances, ...finalSupports];
   748→  const sortedByStrength = [...combinedLevels].sort((a, b) => b.strength - a.strength);
   749→  const targets = sortedByStrength.slice(0, 2).map(t => ({ ...t, isTarget: true }));
   750→
   751→  // Mark targets in final arrays
   752→  finalResistances.forEach(r => {
   753→    if (targets.some(t => Math.abs(t.price - r.price) < 1)) r.isTarget = true;
   754→  });
   755→  finalSupports.forEach(s => {
   756→    if (targets.some(t => Math.abs(t.price - s.price) < 1)) s.isTarget = true;
   757→  });
   758→
   759→  return {
   760→    resistances: finalResistances.map(l => l.price),
   761→    supports: finalSupports.map(l => l.price),
   762→    supportStrengths: finalSupports,
   763→    resistanceStrengths: finalResistances,
   764→    priceTargets: targets,
   765→  };
   766→}

   768→// ─── Trend Analysis ───────────────────────────────────────────────────────────
   769→function calcTrend(closes: number[], period: number): TrendResult {
   770→  const slice = closes.slice(-Math.min(period, closes.length));
   771→  const { slope, r2 } = linearRegression(slice);
   772→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   773→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   774→  const priceThreshold = avgPrice * 0.001; // 0.1%
   775→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   776→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   777→}

   784→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
   785→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
   786→  const slice = data.slice(0, barIdx + 1);
   787→  const price = slice[slice.length - 1].close;
   788→  let nearestR = price * 1.05;
   789→  let nearestS = price * 0.95;
   790→
   791→  // Collect all swing highs (resistance) and swing lows (support)
   792→  const resistances: number[] = [];
   793→  const supports: number[] = [];
   794→  for (const lb of [3, 5, 7]) {
   795→    const swings = findSwingLevels(slice, lb);
   796→    resistances.push(...swings.highs);
   797→    supports.push(...swings.lows);
   798→  }
   799→
   800→  // Find nearest resistance above price
   801→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
   802→  if (abovePrice.length > 0) nearestR = abovePrice[0];
   803→
   804→  // Find nearest support below price
   805→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
   806→  if (belowPrice.length > 0) nearestS = belowPrice[0];
   807→
   808→  return { R1: nearestR, S1: nearestS };
   809→}

   811→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
   812→// Returns true if bullish divergence detected, false if bearish, null if none
   813→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
   814→  const lookback = 20;
   815→  const start = Math.max(0, barIdx - lookback);
   816→  const slice = data.slice(start, barIdx + 1);
   817→  if (slice.length < 10) return null;
   818→
   819→  const mid = Math.floor(slice.length / 2);
   820→  const firstHalf = slice.slice(0, mid);
   821→  const secondHalf = slice.slice(mid);
   822→
   823→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
   824→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
   825→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
   826→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
   827→
   828→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
   829→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
   830→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
   831→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
   832→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
   833→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
   834→
   835→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
   836→    return false; // bearish divergence
   837→  }
   838→
   839→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
   840→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
   841→
   842→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
   843→    return true; // bullish divergence
   844→  }
   845→
   846→  return null;
   847→}

   853→/**
   854→ * Computes all 16 ML features at a specific historical bar.
   855→ * Uses data[0..barIdx] (inclusive) for all indicator calculations.
   856→ * Returns an object keyed by feature name matching FEATURES_ORDER.
   857→ */
   858→function computeFeaturesAtBar(data: OHLCV[], barIdx: number): Record<string, number> {
   859→  // Need at least 30 bars for meaningful indicators
   860→  if (barIdx < 30) {
   861→    const empty: Record<string, number> = {};
   862→    for (const key of FEATURES_ORDER) empty[key] = 0.5;
   863→    return empty;
   864→  }
   865→
   866→  const slice = data.slice(0, barIdx + 1);
   867→  const closes = slice.map(d => d.close);
   868→  const price = closes[closes.length - 1];
   869→  const eps = 1e-10;
   870→
   871→  // ── Compute all base indicators ───────────────────────────────────────────
   872→  const rsiVal = calcRSI(closes);
   873→  const sliceHasVolume = slice.some(d => d.volume > 0);
   874→  const mfiVal = sliceHasVolume ? calcMFI(slice) : 50; // neutral when no volume
   875→  const cciVal = calcCCI(slice);
   876→  const stoch = calcStochastic(slice);
   877→  const macd = calcMACD(closes);
   878→  const adxResult = calcADX(slice);
   879→  const atrVal = calcATR(slice);
   880→  const bb = calcBollingerBands(closes);
   881→  const ma21 = sma(closes, 21);
   882→  const ma100 = sma(closes, 100);
   883→  const ema12 = emaCalc(closes, 12);
   884→  const ema26 = emaCalc(closes, 26);
   885→  const trendResult = calcTrend(closes, 21);
   886→  const sr = simplifiedSR(data, barIdx);
   887→
   888→  // ── Layer 1: Raw Scores (s_*) ────────────────────────────────────────────
   889→
   890→  // s_rsi — RSI mapped to 0-1
   891→  const s_rsi = rsiVal > 80 ? 0.95
   892→    : rsiVal > 70 ? 0.85
   893→    : rsiVal > 60 ? 0.70
   894→    : rsiVal > 50 ? 0.55
   895→    : rsiVal > 40 ? 0.45
   896→    : rsiVal > 30 ? 0.30
   897→    : rsiVal > 20 ? 0.15
   898→    : 0.05;
   899→
   900→  // s_mfi — MFI mapped to 0-1 (same formula as RSI)
   901→  const s_mfi = mfiVal > 80 ? 0.95
   902→    : mfiVal > 70 ? 0.85
   903→    : mfiVal > 60 ? 0.70
   904→    : mfiVal > 50 ? 0.55
   905→    : mfiVal > 40 ? 0.45
   906→    : mfiVal > 30 ? 0.30
   907→    : mfiVal > 20 ? 0.15
   908→    : 0.05;
   909→
   910→  // s_cci — CCI mapped to 0-1
   911→  const s_cci = cciVal > 200 ? 0.90
   912→    : cciVal > 100 ? 0.75
   913→    : cciVal > 0 ? 0.60
   914→    : cciVal > -100 ? 0.40
   915→    : cciVal > -200 ? 0.25
   916→    : 0.10;
   917→
   918→  // s_adx — Directional bias
   919→  const s_adx = (adxResult.diPlus - adxResult.diMinus + 100) / 200;
   920→
   921→  // s_macd — MACD histogram + signal combined
   922→  const absMacdLine = Math.abs(macd.line) + 1;
   923→  const s_macd = macd.line > macd.signal
   924→    ? 0.5 + Math.min(0.5, macd.histogram / absMacdLine)
   925→    : 0.5 - Math.min(0.5, Math.abs(macd.histogram) / absMacdLine);
   926→
   927→  // s_stoch — Stochastic %K mapped to 0-1
   928→  const s_stoch = stoch.k / 100;
   929→
   930→  // s_bb — Bollinger Band position
   931→  const bbRange = bb.upper - bb.lower;
   932→  const s_bb = bbRange > eps
   933→    ? clamp((price - bb.lower) / bbRange, 0, 1)
   934→    : 0.5;
   935→
   936→  // s_ma21 — Price relative to 21-period SMA
   937→  const s_ma21 = ma21 > eps
   938→    ? clamp(0.5 + (price - ma21) / ma21, 0, 1)
   939→    : 0.5;
   940→
   941→  // s_ma100 — Price relative to 100-period SMA
   942→  const s_ma100 = ma100 > eps
   943→    ? clamp(0.5 + (price - ma100) / ma100, 0, 1)
   944→    : 0.5;
   945→
   946→  // s_ema — EMA12 vs EMA26 spread
   947→  const s_ema = ema26 > eps
   948→    ? clamp(0.5 + (ema12 - ema26) / ema26 * 5, 0, 1)
   949→    : 0.5;
   950→
   951→  // s_atr — Inverse volatility (lower ATR = higher score)
   952→  const s_atr = price > eps ? 1 - Math.min(atrVal / price * 10, 1) : 0.5;
   953→
   954→  // s_trend — R² + angle combination
   955→  const s_trend = (trendResult.r2 * 0.7) +
   956→    (Math.max(0, Math.min(1, (trendResult.angle + 45) / 90)) * 0.3);
   957→
   958→  // s_sr — Support/Resistance proximity
   959→  const dR1 = (sr.R1 - price) / (price > eps ? price : 1);
   960→  const dS1 = (price - sr.S1) / (price > eps ? price : 1);
   961→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
   962→
   963→  // ── Layer 2: Corrections (f_*) ───────────────────────────────────────────
   964→
   965→  // f_rsi — RSI with momentum correction
   966→  const rsiCurrent = rsiVal;
   967→  const rsi5ago = barIdx >= 5
   968→    ? calcRSI(closes.slice(0, barIdx - 5 + 1))
   969→    : rsiCurrent;
   970→  const f_rsi = clamp(s_rsi + (rsiCurrent - rsi5ago) / 100 * 0.08, 0, 1);
   971→
   972→  // f_mfi — MFI with momentum correction (skip when no volume)
   973→  const mfiCurrent = mfiVal;
   974→  const mfi5ago = (barIdx >= 5 && sliceHasVolume)
   975→    ? calcMFI(data.slice(0, barIdx - 5 + 1))
   976→    : mfiCurrent;
   977→  const f_mfi = sliceHasVolume
   978→    ? clamp(s_mfi + (mfiCurrent - mfi5ago) / 100 * 0.08, 0, 1)
   979→    : 0.5; // neutral when no volume
   980→
   981→  // f_cci — CCI with momentum correction (scale: /200 instead of /100)
   982→  const cciCurrent = cciVal;
   983→  const cci5ago = barIdx >= 5
   984→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
   985→    : cciCurrent;
   986→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
   987→
   988→  // f_macd — Same as s_macd (no additional correction)
   989→  const f_macd = s_macd;
   990→
   991→  // f_stoch — Same as s_stoch
   992→  const f_stoch = s_stoch;
   993→
   994→  // f_stochCross — Stochastic K/D crossover detection
   995→  let f_stochCross = 0.1;
   996→  if (barIdx >= 1) {
   997→    const prevSlice = data.slice(0, barIdx);
   998→    const prevStoch = calcStochastic(prevSlice);
   999→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
  1000→      f_stochCross = 0.9; // K crossed above D (bullish)
  1001→    }
  1002→  }
  1003→
  1004→  // f_macdCross — MACD/Signal crossover detection
  1005→  let f_macdCross = 0.1;
  1006→  if (barIdx >= 1) {
  1007→    const prevCloses = closes.slice(0, -1);
  1008→    if (prevCloses.length >= 35) {
  1009→      const prevMacd = calcMACD(prevCloses);
  1010→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
  1011→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
  1012→      }
  1013→    }
  1014→  }
  1015→
  1016→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
  1017→  const divResult = detectDivergenceSimple(data, barIdx);
  1018→  const f_div = divResult === true ? 0.85 : 0.15;
  1019→
  1020→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
  1021→  return {
  1022→    rsi: f_rsi,
  1023→    mfi: f_mfi,
  1024→    cci: f_cci,
  1025→    adx: s_adx,
  1026→    macd: f_macd,
  1027→    stoch: f_stoch,
  1028→    bb: s_bb,
  1029→    ma21: s_ma21,
  1030→    ma100: s_ma100,
  1031→    ema: s_ema,
  1032→    atr: s_atr,
  1033→    trend: s_trend,
  1034→    sr: s_sr,
  1035→    stochCross: f_stochCross,
  1036→    macdCross: f_macdCross,
  1037→    div: f_div,
  1038→  };
  1039→}
  1040→
  1041→// ═══════════════════════════════════════════════════════════════════════════════
  1042→// TRAINING DATA CREATION (Layer 0 — Data Preparation)
  1043→// ═══════════════════════════════════════════════════════════════════════════════
  1044→
  1045→/**
  1046→ * Creates supervised training data from historical OHLCV data.
  1047→ * For each bar, computes features and labels based on future price movement.
  1048→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
  1049→ * Samples every `sampleStep` bars to reduce computation.
  1050→ */
  1051→function createTrainingData(
  1052→  data: OHLCV[],
  1053→  horizon: number = 5,
  1054→  sampleStep: number = 5
  1055→): { X: number[][]; y: number[] } {
  1056→  const X: number[][] = [];
  1057→  const y: number[] = [];
  1058→
  1059→  // Start from horizon+30 to ensure enough data for indicators
  1060→  // End at data.length - horizon - 1 to have future bars for labeling
  1061→  // Sample every sampleStep bars for performance
  1062→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
  1063→    const features = computeFeaturesAtBar(data, i);
  1064→
  1065→    // Convert to array following FEATURES_ORDER
  1066→    const featureVec: number[] = [];
  1067→    let valid = true;
  1068→    for (const key of FEATURES_ORDER) {
  1069→      const val = features[key];
  1070→      if (val === undefined || val === null || Number.isNaN(val)) {
  1071→        valid = false;
  1072→        break;
  1073→      }
  1074→      featureVec.push(val);
  1075→    }
  1076→    if (!valid) continue;
  1077→
  1078→    // Label: 1 if future close > 1% above current close
  1079→    const currentClose = data[i].close;
  1080→    const futureClose = data[i + horizon].close;
  1081→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
  1082→
  1083→    X.push(featureVec);
  1084→    y.push(label);
  1085→  }
  1086→
  1087→  return { X, y };
  1088→}
  1089→
  1090→// ═══════════════════════════════════════════════════════════════════════════════
  1091→// VDss GRAPH STRUCTURE (Layer 5 — State Machine)
  1092→// ═══════════════════════════════════════════════════════════════════════════════
  1093→
  1094→/**
  1095→ * VDss Graph: Each node represents a market state.
  1096→ * Edges represent transitions with types: 'up', 'down', 'pullback', 'risk', 'terminal'.
  1097→ * Terminal nodes map to scenario results (R1=strong bullish, R5=deep correction).
  1098→ */
  1099→const VDSS_GRAPH = {
  1100→  // outgoing edges from each node: [target, type]
  1101→  outgoing: {
  1102→    A: [['B', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1103→    B: [['C', 'up'], ['G', 'pullback'], ['A', 'risk']] as [string, string][],
  1104→    C: [['D', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1105→    D: [['E', 'up'], ['B', 'pullback'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1106→    E: [['F', 'up'], ['D', 'pullback'], ['G', 'pullback'], ['H', 'down'], ['R1', 'terminal']] as [string, string][],
  1107→    F: [['R1', 'terminal'], ['E', 'pullback'], ['D', 'down'], ['G', 'risk']] as [string, string][],
  1108→    G: [['B', 'up'], ['A', 'pullback'], ['H', 'down'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1109→    H: [['G', 'pullback'], ['I', 'down'], ['B', 'up'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1110→    I: [['H', 'pullback'], ['J', 'down'], ['G', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1111→    J: [['I', 'pullback'], ['K', 'down'], ['H', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1112→    K: [['J', 'pullback'], ['I', 'up'], ['L', 'risk'], ['R4', 'terminal'], ['R5', 'terminal']] as [string, string][],
  1113→    L: [['C', 'up'], ['H', 'pullback'], ['J', 'down'], ['K', 'risk'], ['R5', 'terminal']] as [string, string][],
  1114→  },
  1115→  // which result scenario each terminal node maps to
  1116→  terminalMap: { R1: 'R1', R2: 'R2', R3: 'R3', R4: 'R4', R5: 'R5' } as Record<string, string>,
  1117→  startNode: 'A' as string,
  1118→};
  1119→
  1120→// ═══════════════════════════════════════════════════════════════════════════════
  1121→// LAYER 5 — Edge Weight Calculation
  1122→// ═══════════════════════════════════════════════════════════════════════════════
  1123→
  1124→/**
  1125→ * Calculates transition edge weights for the VDss graph.
  1126→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
  1127→ * to determine the relative probability of up/down/pullback/risk transitions.
  1128→ */
  1129→function calculateEdgeWeights(
  1130→  bullConsensus: number,
  1131→  adx: number,
  1132→  mlModel: AdaptiveWeightModel
  1133→): { up: number; down: number; pullback: number; risk: number } {
  1134→  // Extract ML coefficient insights if model is trained
  1135→  let trend_coef = 1 / 3;
  1136→  let momentum_coef = 1 / 3;
  1137→  let volatility_coef = 1 / 3;
  1138→
  1139→  if (mlModel.isTrained && mlModel.weights) {
  1140→    const coefs = mlModel.getCoefficients();
  1141→    // Map feature indices to coefficient categories
  1142→    // FEATURES_ORDER: ['rsi','mfi','cci','adx','macd','stoch','bb','ma21','ma100','ema','atr','trend','sr','stochCross','macdCross','div']
  1143→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER
  1144→    const rsiIdx = 0;     // 'rsi' in FEATURES_ORDER
  1145→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
  1146→
  1147→    const absTrend = Math.abs(coefs[trendIdx] ?? 0);
  1148→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
  1149→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
  1150→
  1151→    const coefSum = absTrend + absRsi + absAtr;
  1152→    if (coefSum > 1e-10) {
  1153→      trend_coef = absTrend / coefSum;
  1154→      momentum_coef = absRsi / coefSum;
  1155→      volatility_coef = absAtr / coefSum;
  1156→    }
  1157→  }
  1158→
  1159→  // Base weights driven by bull consensus
  1160→  const up_base = bullConsensus * 0.7 + 0.15;
  1161→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
  1162→  const pullback_base = 0.25;
  1163→  const risk_base = 0.12 * (1.2 - adx / 100);
  1164→
  1165→  // Apply ML coefficient adjustments
  1166→  const up = clamp(
  1167→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
  1168→    0.05, 0.95
  1169→  );
  1170→  const down = clamp(
  1171→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
  1172→    0.05, 0.95
  1173→  );
  1174→  const pullback = clamp(
  1175→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
  1176→    0.05, 0.50
  1177→  );
  1178→  const risk = clamp(
  1179→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
  1180→    0.02, 0.30
  1181→  );
  1182→
  1183→  return { up, down, pullback, risk };
  1184→}
  1185→
  1186→// ═══════════════════════════════════════════════════════════════════════════════
  1187→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)
  1188→// ═══════════════════════════════════════════════════════════════════════════════
  1189→
  1190→/**
  1191→ * Performs iterative DFS from the start node through the VDss graph,
  1192→ * computing raw path probabilities, then calibrating them against
  1193→ * Layer 4 scenario probabilities.
  1194→ *
  1195→ * Returns:
  1196→ * - calibratedPaths: Map<pathString, probability>
  1197→ * - calibrationFactors: per-scenario multiplier used
  1198→ * - scenarioSums: raw probability sums per scenario before calibration
  1199→ */
  1200→function calculatePathProbabilities(
  1201→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
  1202→  scenarioProbs: Record<string, number>
  1203→): {
  1204→  calibratedPaths: Map<string, number>;
  1205→  calibrationFactors: Record<string, number>;
  1206→  scenarioSums: Record<string, number>;
  1207→} {
  1208→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
  1209→  const MAX_PATHS = 200;
  1210→
  1211→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
  1212→  type StackEntry = [string, string, number];
  1213→
  1214→  // Collect all terminal paths with raw probabilities
  1215→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
  1216→
  1217→  // Iterative DFS
  1218→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
  1219→
  1220→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
  1221→    const [node, pathStr, pathProb] = stack.pop()!;
  1222→
  1223→    const edges = outgoing[node as keyof typeof outgoing];
  1224→    if (!edges) continue;
  1225→
  1226→    // Normalize outgoing edge weights to probabilities for this node
  1227→    // Map edge types to their weights
  1228→    let totalWeight = 0;
  1229→    const edgeWeightsList: number[] = [];
  1230→    for (const [, edgeType] of edges) {
  1231→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
  1232→      edgeWeightsList.push(w);
  1233→      totalWeight += w;
  1234→    }
  1235→
  1236→    if (totalWeight < 1e-10) continue;
  1237→
  1238→    // Expand edges
  1239→    for (let i = 0; i < edges.length; i++) {
  1240→      const [target, edgeType] = edges[i];
  1241→      const edgeProb = edgeWeightsList[i] / totalWeight;
  1242→      const newProb = pathProb * edgeProb;
  1243→      const newPath = pathStr + '→' + target;
  1244→
  1245→      if (edgeType === 'terminal') {
  1246→        // Reached a terminal node
  1247→        const scenario = terminalMap[target] ?? 'R3';
  1248→        rawPaths.push({ path: newPath, scenario, prob: newProb });
  1249→      } else {
  1250→        // Continue DFS
  1251→        stack.push([target, newPath, newProb]);
  1252→      }
  1253→    }
  1254→  }
  1255→
  1256→  // Group raw probabilities by scenario
  1257→  const rawSums: Record<string, number> = { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0 };
  1258→  for (const rp of rawPaths) {
  1259→    rawSums[rp.scenario] = (rawSums[rp.scenario] ?? 0) + rp.prob;
  1260→  }
  1261→
  1262→  // Compute calibration factors: scenarioProbs[scenario] / rawSum[scenario]
  1263→  const calibrationFactors: Record<string, number> = {};
  1264→  for (const scenario of ['R1', 'R2', 'R3', 'R4', 'R5']) {
  1265→    const rawSum = rawSums[scenario] ?? 0;
  1266→    const targetProb = (scenarioProbs[scenario] ?? 20) / 100; // convert from percentage
  1267→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
  1268→  }
  1269→
  1270→  // Apply calibration to each path
  1271→  const calibratedPaths = new Map<string, number>();
  1272→  for (const rp of rawPaths) {
  1273→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
  1274→    calibratedPaths.set(rp.path, rp.prob * factor);
  1275→  }
  1276→
  1277→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
  1278→}
  1279→
  1280→// ═══════════════════════════════════════════════════════════════════════════════
  1281→// LAYER 4 — Scenario Probability Calculation (Adaptive with ML)
  1282→// ═══════════════════════════════════════════════════════════════════════════════
  1283→
  1284→/**
  1285→ * Calculates the 5 VDss scenario probabilities (R1-R5) that sum to 100.
  1286→ * Uses bull/bear consensus, key indicator values, S/R levels, and
  1287→ * ML-derived adaptive factors for fine-tuning.
  1288→ */
  1289→function calculateScenarioProbabilities(
  1290→  bullConsensus: number,
  1291→  price: number,
  1292→  R1: number,
  1293→  S1: number,
  1294→  MA100: number,
  1295→  rsi: number,
  1296→  mfi: number,
  1297→  stochK: number,
  1298→  mlModel: AdaptiveWeightModel
  1299→): { R1: number; R2: number; R3: number; R4: number; R5: number; factors: { momentum: number; volatility: number; trend: number } } {
  1300→  // Distance metrics to key levels (spec uses exp(-3 * ...))
  1301→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0;
  1302→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0;
  1303→
  1304→  // Binary risk flags (per spec) — MFI only used when volume is available
  1305→  const overboughtRisk = (rsi > 70 || (mfi > 80 && mfi !== 50) || stochK > 80) ? 1 : 0;
  1306→  const oversoldBounce = (rsi < 30 || (mfi < 20 && mfi !== 50) || stochK < 20) ? 1 : 0;
  1307→
  1308→  // Extract adaptive factors from ML model if trained
  1309→  let momentum = 0.7;
  1310→  let volatility = 0.5;
  1311→  let trend = 0.6;
  1312→
  1313→  if (mlModel.isTrained && mlModel.weights) {
  1314→    const coefs = mlModel.getCoefficients();
  1315→    momentum = sigmoid(coefs[0] ?? 0);   // rsi coefficient → momentum factor
  1316→    volatility = sigmoid(coefs[10] ?? 0); // atr coefficient → volatility factor
  1317→    trend = sigmoid(coefs[11] ?? 0);     // trend coefficient → trend factor
  1318→  }
  1319→
  1320→  // Below MA100 flag
  1321→  const belowMA100 = price < MA100 ? 1 : 0;
  1322→
  1323→  // ── Raw scenario probabilities (matching user spec) ─────────────────────
  1324→
  1325→  // R1: Strong Bullish — aggressive breakout
  1326→  const raw_R1 =
  1327→    bullConsensus * distR1 * (1 - overboughtRisk * momentum * 0.7) +
  1328→    oversoldBounce * 0.3 * distR1 * momentum;
  1329→
  1330→  // R2: Gradual Uptrend — steady rise
  1331→  const raw_R2 =
  1332→    ((1 - Math.abs(bullConsensus - 0.65)) * 0.8 +
  1333→    (bullConsensus > 0.4 ? 0.2 : 0)) * trend;
  1334→
  1335→  // R3: Range-bound
  1336→  const raw_R3 =
  1337→    (1 - Math.abs(bullConsensus - 0.5)) * 0.6 * (1 + (1 - volatility) * 0.2);
  1338→
  1339→  // R4: Moderate Correction
  1340→  const raw_R4 =
  1341→    ((1 - bullConsensus) * distS1 * 0.7 +
  1342→    overboughtRisk * 0.5 * momentum) *
  1343→    (0.8 + 0.2 * (1 - trend));
  1344→
  1345→  // R5: Deep Correction
  1346→  const raw_R5 =
  1347→    ((1 - bullConsensus) ** 2 * 0.5 + belowMA100 * 0.2) *
  1348→    (1 + (1 - trend) * 0.3);
  1349→
  1350→  // ── Normalize to sum = 100 ───────────────────────────────────────────
  1351→  const sum = raw_R1 + raw_R2 + raw_R3 + raw_R4 + raw_R5;
  1352→  const pR1 = sum > 0 ? Math.round(raw_R1 / sum * 100) : 20;
  1353→  const pR2 = sum > 0 ? Math.round(raw_R2 / sum * 100) : 20;
  1354→  const pR3 = sum > 0 ? Math.round(raw_R3 / sum * 100) : 20;
  1355→  const pR4 = sum > 0 ? Math.round(raw_R4 / sum * 100) : 20;
  1356→  const pR5 = 100 - (pR1 + pR2 + pR3 + pR4);
  1357→
  1358→  return {
  1359→    R1: pR1, R2: pR2, R3: pR3, R4: pR4, R5: pR5,
  1360→    factors: { momentum, volatility, trend },
  1361→  };
  1362→}

  1364→// ═══════════════════════════════════════════════════════════════════════════════
  1365→// MAIN ANALYSIS FUNCTION
  1366→// ═══════════════════════════════════════════════════════════════════════════════
  1367→export function analyze(data: OHLCV[]): TAResult {
  1368→  if (!data || data.length < 2) {
  1369→    const empty = () => 0;
  1370→    const emptyScenario = (): ScenarioResult => ({ name: '', nameEn: '', probability: 20, targetMin: 0, targetMax: 0, description: '' });
  1371→    return {
  1372→      sma: {}, ema: {}, rsi: 50, mfi: 50, cci: 0, stochK: 50, stochD: 50,
  1373→      williamsR: -50, macd: { line: 0, signal: 0, histogram: 0 },
  1374→      adx: 0, diPlus: 0, diMinus: 0, sar: 0, atr: 0,
  1375→      bollingerBands: { upper: 0, middle: 0, lower: 0 }, obv: 0,
  1376→      resistances: [], supports: [],
  1377→      supportStrengths: [], resistanceStrengths: [], priceTargets: [],
  1378→      trend: {
  1379→        short: { direction: 'flat', slope: 0, angle: 0, r2: 0 },
  1380→        medium: { direction: 'flat', slope: 0, angle: 0, r2: 0 },
  1381→        long: { direction: 'flat', slope: 0, angle: 0, r2: 0 },
  1382→      },
  1383→      scenarios: { R1: emptyScenario(), R2: emptyScenario(), R3: emptyScenario(), R4: emptyScenario(), R5: emptyScenario() },
  1384→      bullScore: 0.5, bearScore: 0.5, overallSignal: 'neutral',
  1385→      bullConsensus: 0.5, isMLTrained: false, mlAccuracy: 0.5, mlWeights: null,
  1386→      edgeWeights: { up: 0.3, down: 0.3, pullback: 0.25, risk: 0.15 },
  1387→      calibrationFactors: { R1: 1, R2: 1, R3: 1, R4: 1, R5: 1 },
  1388→      scenarioSums: { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0 },
  1389→      adaptiveFactors: { momentum: 0.7, volatility: 0.5, trend: 0.6 },
  1390→      hasVolume: false,

  1392→  }
  1393→
  1394→  const closes = data.map(d => d.close);
  1395→  const price = closes[closes.length - 1];
  1396→
  1397→  // ── Detect volume availability ──────────────────────────────────────────
  1398→  const hasVolume = data.some(d => d.volume > 0);
  1399→
  1400→  // ── Moving Averages ──────────────────────────────────────────────────────
