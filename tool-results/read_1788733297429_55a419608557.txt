    44→function findSwingPoints(data: OHLCV[], order: number = 3): SwingPoint[] {
    45→  const swings: SwingPoint[] = [];
    46→  const len = data.length;
    47→  for (let i = order; i < len - order; i++) {
    48→    let isHigh = true;
    49→    let isLow = true;
    50→    for (let j = 1; j <= order; j++) {
    51→      if (data[i].high <= data[i - j].high || data[i].high <= data[i + j].high) isHigh = false;
    52→      if (data[i].low >= data[i - j].low || data[i].low >= data[i + j].low) isLow = false;
    53→    }
    54→    if (isHigh) swings.push({ index: i, price: data[i].high, type: 'high' });
    55→    if (isLow) swings.push({ index: i, price: data[i].low, type: 'low' });
    56→  }
    57→  return swings;
    58→}

    60→function avg(arr: number[]): number {
    61→  if (arr.length === 0) return 0;
    62→  return arr.reduce((s, v) => s + v, 0) / arr.length;

    98→function sma(values: number[], period: number): number {
    99→  if (values.length < period) return 0;
   100→  const slice = values.slice(-period);
   101→  return slice.reduce((a, b) => a + b, 0) / period;
   102→}

   104→function ema(values: number[], period: number): number {
   105→  if (values.length < period) return 0;
   106→  const k = 2 / (period + 1);
   107→  let result = values.slice(0, period).reduce((a, b) => a + b, 0) / period;
   108→  for (let i = period; i < values.length; i++) {
   109→    result = values[i] * k + result * (1 - k);
   110→  }
   111→  return result;
   112→}

   129→function detectHeadAndShoulders(data: OHLCV[]): PatternResult[] {
   130→  const results: PatternResult[] = [];
   131→  if (data.length < 30) return results;
   132→  const swings = findSwingPoints(data, 3);
   133→  const highs = swings.filter(s => s.type === 'high');
   134→  const lows = swings.filter(s => s.type === 'low');
   135→
   136→  // Bearish Head & Shoulders: left shoulder < head > right shoulder, neckline support
   137→  for (let i = 2; i < highs.length; i++) {
   138→    const ls = highs[i - 2];
   139→    const hd = highs[i - 1];
   140→    const rs = highs[i];
   141→    if (hd.price > ls.price && hd.price > rs.price) {
   142→      const sim = 1 - Math.abs(ls.price - rs.price) / hd.price;
   143→      if (sim > 0.85) {
   144→        const neckline = (ls.price + rs.price) / 2;
   145→        const last = data[data.length - 1];
   146→        const dist = (last.close - neckline) / neckline;
   147→        const status = dist > 0.01 ? 'forming' : dist < -0.005 ? 'completed' : 'forming';
   148→        results.push({
   149→          name: 'سر و شانه', nameEn: 'Head & Shoulders', category: 'classic',
   150→          direction: 'bearish', strength: sim * 0.8 + (status === 'completed' ? 0.2 : 0),
   151→          status, priceLevel: neckline,
   152→          description: `الگوی سر و شانه نزولی با خط گردن ${neckline.toFixed(0)}`
   153→        });
   154→      }
   155→    }
   156→  }
   157→
   158→  // Bullish Inverse Head & Shoulders
   159→  for (let i = 2; i < lows.length; i++) {
   160→    const ls = lows[i - 2];
   161→    const hd = lows[i - 1];
   162→    const rs = lows[i];
   163→    if (hd.price < ls.price && hd.price < rs.price) {
   164→      const sim = 1 - Math.abs(ls.price - rs.price) / hd.price;
   165→      if (sim > 0.85) {
   166→        const neckline = (ls.price + rs.price) / 2;
   167→        const last = data[data.length - 1];
   168→        const dist = (last.close - neckline) / neckline;
   169→        const status = dist < -0.01 ? 'forming' : dist > 0.005 ? 'completed' : 'forming';
   170→        results.push({
   171→          name: 'سر و شانه معکوس', nameEn: 'Inverse Head & Shoulders', category: 'classic',
   172→          direction: 'bullish', strength: sim * 0.8 + (status === 'completed' ? 0.2 : 0),
   173→          status, priceLevel: neckline,
   174→          description: `الگوی سر و شانه معکوس صعودی با خط گردن ${neckline.toFixed(0)}`
   175→        });
   176→      }
   177→    }
   178→  }
   179→
   180→  return results;
   181→}

   183→function detectDoubleTop(data: OHLCV[]): PatternResult {
   184→  if (data.length < 20) return null!;
   185→  const swings = findSwingPoints(data, 3);
   186→  const highs = swings.filter(s => s.type === 'high');

   188→
   189→  const len = highs.length;
   190→  const h1 = highs[len - 2];
   191→  const h2 = highs[len - 1];
   192→  const sim = 1 - Math.abs(h1.price - h2.price) / h1.price;
   193→  if (sim < 0.96) return null!;

   195→  const trough = Math.min(...data.slice(h1.index, h2.index + 1).map(d => d.low));

   197→  const broken = last.close < trough;
   198→
   199→  return {
   200→    name: 'دو سقف', nameEn: 'Double Top', category: 'classic',
   201→    direction: 'bearish', strength: Math.min(sim, 1),
   202→    status: broken ? 'completed' : 'forming',
   203→    priceLevel: trough,
   204→    description: `الگوی دو سقف نزولی با خط گردن ${trough.toFixed(0)}`

   233→function detectTripleTop(data: OHLCV[]): PatternResult {
   234→  if (data.length < 30) return null!;
   235→  const swings = findSwingPoints(data, 3);
   236→  const highs = swings.filter(s => s.type === 'high');
   237→  if (highs.length < 3) return null!;
   238→
   239→  const h = [highs[highs.length - 3], highs[highs.length - 2], highs[highs.length - 1]];
   240→  const avgPrice = (h[0].price + h[1].price + h[2].price) / 3;
   241→  const variance = h.reduce((s, p) => s + Math.abs(p.price - avgPrice), 0) / avgPrice;
   242→  if (variance > 0.04) return null!;
   243→
   244→  const neckline = Math.min(
   245→    Math.min(...data.slice(h[0].index, h[1].index + 1).map(d => d.low)),
   246→    Math.min(...data.slice(h[1].index, h[2].index + 1).map(d => d.low))
   247→  );
   248→  const last = data[data.length - 1];
   249→  const broken = last.close < neckline;
   250→
   251→  return {
   252→    name: 'سه سقف', nameEn: 'Triple Top', category: 'classic',
   253→    direction: 'bearish', strength: 1 - variance * 10,
   254→    status: broken ? 'completed' : 'forming',
   255→    priceLevel: neckline,
   256→    description: `الگوی سه سقف نزولی با خط گردن ${neckline.toFixed(0)}`

   287→function detectAscendingTriangle(data: OHLCV[]): PatternResult {
   288→  if (data.length < 20) return null!;
   289→  const swings = findSwingPoints(data, 3);
   290→  const highs = swings.filter(s => s.type === 'high');
   291→  const lows = swings.filter(s => s.type === 'low');
   292→  if (highs.length < 2 || lows.length < 2) return null!;
   293→
   294→  const recentHighs = highs.slice(-3);
   295→  const recentLows = lows.slice(-3);
   296→  const hAvg = avg(recentHighs.map(h => h.price));
   297→  const hVar = recentHighs.reduce((s, h) => s + Math.abs(h.price - hAvg), 0) / hAvg;
   298→  const lRising = recentLows.length >= 2 && recentLows[recentLows.length - 1].price > recentLows[0].price;
   299→
   300→  if (hVar > 0.02 || !lRising) return null!;
   301→
   302→  const last = data[data.length - 1];
   303→  const broken = last.close > hAvg;
   304→  return {
   305→    name: 'مثلث صعودی', nameEn: 'Ascending Triangle', category: 'classic',
   306→    direction: 'bullish', strength: 0.6 + (broken ? 0.4 : lRising ? 0.2 : 0),
   307→    status: broken ? 'completed' : 'forming',
   308→    priceLevel: hAvg,
   309→    description: `الگوی مثلث صعودی با مقاومت ${hAvg.toFixed(0)}`

   339→function detectSymmetricTriangle(data: OHLCV[]): PatternResult {
   340→  if (data.length < 20) return null!;
   341→  const swings = findSwingPoints(data, 3);
   342→  const highs = swings.filter(s => s.type === 'high');
   343→  const lows = swings.filter(s => s.type === 'low');
   344→  if (highs.length < 3 || lows.length < 3) return null!;
   345→
   346→  const rh = highs.slice(-3);
   347→  const rl = lows.slice(-3);
   348→  const hConverging = rh[rh.length - 1].price < rh[0].price;
   349→  const lConverging = rl[rl.length - 1].price > rl[0].price;
   350→  const hSlope = (rh[rh.length - 1].price - rh[0].price) / (rh[rh.length - 1].index - rh[0].index);
   351→  const lSlope = (rl[rl.length - 1].price - rl[0].price) / (rl[rl.length - 1].index - rl[0].index);
   352→  const slopeSim = 1 - Math.abs(Math.abs(hSlope) - Math.abs(lSlope)) / (Math.abs(hSlope) + Math.abs(lSlope) + 0.001);
   353→
   354→  if (!hConverging || !lConverging || slopeSim < 0.5) return null!;
   355→
   356→  const apex = (rh[rh.length - 1].price + rl[rl.length - 1].price) / 2;
   357→  const last = data[data.length - 1];
   358→  const dir = last.close > apex ? 'bullish' : 'bearish';
   359→
   360→  return {
   361→    name: 'مثلث متقارن', nameEn: 'Symmetric Triangle', category: 'classic',
   362→    direction: dir as 'bullish' | 'bearish',
   363→    strength: slopeSim * 0.7 + 0.3,
   364→    status: 'forming',
   365→    priceLevel: apex,
   366→    description: `الگوی مثلث متقارن با رأس تقریبی ${apex.toFixed(0)}`
   367→  };
   368→}

   427→function detectBullFlag(data: OHLCV[]): PatternResult {
   428→  if (data.length < 20) return null!;
   429→  const poleLen = Math.min(10, Math.floor(data.length * 0.3));
   430→  const flagLen = Math.min(10, Math.floor(data.length * 0.2));
   431→  const flagStart = data.length - flagLen;
   432→  const poleEnd = flagStart;
   433→  const poleStart = Math.max(0, poleEnd - poleLen);
   434→
   435→  if (poleStart >= poleEnd) return null!;
   436→  const poleRise = data[poleEnd - 1].close - data[poleStart].close;
   437→  if (poleRise <= 0) return null!;
   438→
   439→  const flagHighs = data.slice(flagStart).map(d => d.high);
   440→  const flagLows = data.slice(flagStart).map(d => d.low);
   441→  const flagTrend = sma(flagHighs, flagLen) - sma(flagLows, flagLen);
   442→  const flagSlope = (flagHighs[flagHighs.length - 1] - flagHighs[0]) / (flagLen || 1);
   443→
   444→  if (flagSlope > 0 || flagTrend > poleRise * 0.3) return null!;
   445→
   446→  const last = data[data.length - 1];
   447→  return {
   448→    name: 'پرچم صعودی', nameEn: 'Bull Flag', category: 'classic',
   449→    direction: 'bullish', strength: 0.5 + Math.min(poleRise / (last.close || 1), 0.3) + (flagSlope < 0 ? 0.2 : 0),
   450→    status: 'forming',
   451→    priceLevel: Math.max(...flagHighs),
   452→    description: `الگوی پرچم صعودی — ادامه روند صعودی مورد انتظار`
   453→  };
   454→}

   461→  const poleEnd = flagStart;
   462→  const poleStart = Math.max(0, poleEnd - poleLen);
   463→
   464→  if (poleStart >= poleEnd) return null!;
   465→  const poleDrop = data[poleStart].close - data[poleEnd - 1].close;
   466→  if (poleDrop <= 0) return null!;
   467→
   468→  const flagHighs = data.slice(flagStart).map(d => d.high);
   469→  const flagLows = data.slice(flagStart).map(d => d.low);
   470→  const flagSlope = (flagLows[flagLows.length - 1] - flagLows[0]) / (flagLen || 1);
   471→
   472→  if (flagSlope < 0) return null!;
   473→
   474→  const last = data[data.length - 1];
   475→  return {
   476→    name: 'پرچم نزولی', nameEn: 'Bear Flag', category: 'classic',
   477→    direction: 'bearish', strength: 0.5 + Math.min(poleDrop / (last.close || 1), 0.3) + (flagSlope > 0 ? 0.2 : 0),
   478→    status: 'forming',
   479→    priceLevel: Math.min(...flagLows),
   480→    description: `الگوی پرچم نزولی — ادامه روند نزولی مورد انتظار`
   481→  };

   484→function detectPennant(data: OHLCV[]): PatternResult[] {
   485→  const results: PatternResult[] = [];
   486→  if (data.length < 20) return results;
   487→
   488→  const poleLen = Math.min(10, Math.floor(data.length * 0.25));
   489→  const flagLen = Math.min(10, Math.floor(data.length * 0.15));
   490→  const flagStart = data.length - flagLen;
   491→  const poleEnd = flagStart;
   492→  const poleStart = Math.max(0, poleEnd - poleLen);
   493→
   494→  if (poleStart >= poleEnd || flagLen < 3) return results;
   495→
   496→  const poleMove = data[poleEnd - 1].close - data[poleStart].close;
   497→  const absPoleMove = Math.abs(poleMove);
   498→  if (absPoleMove === 0) return results;
   499→
   500→  const highs = data.slice(flagStart).map(d => d.high);
   501→  const lows = data.slice(flagStart).map(d => d.low);
   502→  const hConverge = highs[highs.length - 1] < highs[0];
   503→  const lConverge = lows[lows.length - 1] > lows[0];
   504→  if (!(hConverge && lConverge)) return results;
   505→
   506→  const isBull = poleMove > 0;
   507→  results.push({
   508→    name: isBull ? 'پرچم‌نما صعودی' : 'پرچم‌نما نزولی',
   509→    nameEn: isBull ? 'Bull Pennant' : 'Bear Pennant',

   511→    direction: isBull ? 'bullish' : 'bearish',
   512→    strength: 0.7,
   513→    status: 'forming',
   514→    priceLevel: avg(highs),
   515→    description: isBull
   516→      ? 'الگوی پرچم‌نما صعودی — ادامۀ روند صعودی'
   517→      : 'الگوی پرچم‌نما نزولی — ادامۀ روند نزولی'
   518→  });
   519→  return results;

   522→function detectCupAndHandle(data: OHLCV[]): PatternResult {
   523→  if (data.length < 40) return null!;
   524→  const swings = findSwingPoints(data, 3);
   525→  const lows = swings.filter(s => s.type === 'low');
   526→  if (lows.length < 3) return null!;
   527→
   528→  // Look for U-shape: high, low, high with a small dip (handle)
   529→  const recentLows = lows.slice(-5);
   530→  if (recentLows.length < 3) return null!;
   531→
   532→  // Cup: first and last swing lows should be near same level, middle ones lower
   533→  const cupL = recentLows.slice(0, Math.ceil(recentLows.length * 0.7));
   534→  const cupBottom = Math.min(...cupL.map(l => l.price));
   535→  const cupRim = avg([recentLows[0].price, cupL[cupL.length - 1].price]);
   536→  const depth = (cupRim - cupBottom) / cupRim;
   537→  if (depth < 0.05 || depth > 0.5) return null!;
   538→
   539→  const handleLows = recentLows.slice(Math.ceil(recentLows.length * 0.7));
   540→  if (handleLows.length === 0) return null!;
   541→  const handleDepth = (cupRim - avg(handleLows.map(l => l.price))) / cupRim;
   542→  if (handleDepth > 0.3 * depth) return null!;
   543→
   544→  const last = data[data.length - 1];
   545→  const broken = last.close > cupRim;
   546→  return {
   547→    name: 'فنجان و دسته', nameEn: 'Cup & Handle', category: 'classic',
   548→    direction: 'bullish', strength: 0.5 + (broken ? 0.3 : 0) + Math.min(depth, 0.2),
   549→    status: broken ? 'completed' : 'forming',
   550→    priceLevel: cupRim,
   551→    description: `الگوی فنجان و دسته صعودی با سطح مقاومت ${cupRim.toFixed(0)}`
   552→  };
   553→}

   555→function detectRectangle(data: OHLCV[]): PatternResult {
   556→  if (data.length < 20) return null!;
   557→  const swings = findSwingPoints(data, 3);
   558→  const highs = swings.filter(s => s.type === 'high');
   559→  const lows = swings.filter(s => s.type === 'low');
   560→  if (highs.length < 2 || lows.length < 2) return null!;
   561→
   562→  const rh = highs.slice(-3);
   563→  const rl = lows.slice(-3);
   564→  const hAvg = avg(rh.map(h => h.price));
   565→  const lAvg = avg(rl.map(l => l.price));
   566→  const hVar = rh.reduce((s, h) => s + Math.abs(h.price - hAvg), 0) / hAvg;
   567→  const lVar = rl.reduce((s, l) => s + Math.abs(l.price - lAvg), 0) / lAvg;
   568→
   569→  if (hVar > 0.03 || lVar > 0.03) return null!;
   570→  if ((hAvg - lAvg) / lAvg < 0.02) return null!;
   571→
   572→  const last = data[data.length - 1];
   573→  const mid = (hAvg + lAvg) / 2;
   574→  const dir = last.close > mid ? 'bullish' : 'bearish';
   575→
   576→  return {
   577→    name: 'مستطیل / کانال', nameEn: 'Rectangle / Channel', category: 'classic',
   578→    direction: dir as 'bullish' | 'bearish',
   579→    strength: (1 - hVar * 20) * (1 - lVar * 20),
   580→    status: 'forming',
   581→    priceLevel: mid,
   582→    description: `الگوی مستطیل با محدوده ${lAvg.toFixed(0)} تا ${hAvg.toFixed(0)}`
   583→  };
   584→}

   738→function detectHarmonicFromPoints(
   739→  X: number, A: number, B: number, C: number, D: number,
   740→  spec: HarmonicSpec,
   741→  direction: 'bullish' | 'bearish'
   742→): PatternResult | null {
   743→  const xLeg = Math.abs(A - X);
   744→  const aLeg = Math.abs(B - A);
   745→  const bLeg = Math.abs(C - B);
   746→  const cLeg = Math.abs(D - C);
   747→
   748→  if (xLeg === 0 || aLeg === 0 || bLeg === 0) return null;
   749→
   750→  const abR = aLeg / xLeg;
   751→  const bcR = bLeg / aLeg;
   752→  const cdR = cLeg / bLeg;
   753→
   754→  const abMatch = nearRatio(abR, spec.abRatio, FIB_TOL);
   755→  const bcMatch = nearRatio(bcR, spec.bcRatio, FIB_TOL);
   756→  const cdMatch = nearRatio(cdR, spec.cdRatio, FIB_TOL);
   757→
   758→  let conditions = 0;
   759→  if (abMatch) conditions++;
   760→  if (bcMatch) conditions++;
   761→  if (cdMatch) conditions++;
   762→
   763→  if (conditions < 2) return null;
   764→
   765→  const strength = conditions === 3 ? 0.9 : 0.6;
   766→  const prz = (B + C) / 2; // Potential Reversal Zone
   767→
   768→  return {
   769→    name: spec.name,
   770→    nameEn: spec.nameEn,
   771→    category: 'harmonic',
   772→    direction,
   773→    strength,
   774→    status: 'forming',
   775→    priceLevel: prz,
   776→    description: `الگوی هارمونیک ${spec.nameEn} در ناحیه بازگشتی ${prz.toFixed(0)}`
   777→  };
   778→}

   804→      // Bearish: X is low, A is high, B is low, C is high, D is low

   828→function detectHammer(data: OHLCV[]): PatternResult | null {
   829→  if (data.length < 1) return null;
   830→  const c = data[data.length - 1];
   831→  const range = totalRange(c);
   832→  if (range === 0) return null;
   833→  const body = bodySize(c);
   834→  const ls = lowerShadow(c);
   835→  const us = upperShadow(c);
   836→  if (ls < body * 2) return null;
   837→  if (us > body * 0.5) return null;
   838→  // Must be in downtrend
   839→  if (data.length < 5) return null;
   840→  const prevTrend = data.slice(-6, -1).filter(d => isBearish(d)).length >= 3;
   841→  if (!prevTrend) return null;
   842→  return {
   843→    name: 'چکش', nameEn: 'Hammer', category: 'candlestick',
   844→    direction: 'bullish', strength: 0.6 + Math.min(ls / range, 0.3),
   845→    status: 'completed', priceLevel: c.low,
   846→    description: 'الگوی چکش در انتهای روند نزولی — سیگنال صعودی'
   847→  };
   848→}

   871→function detectBullishEngulfing(data: OHLCV[]): PatternResult | null {
   872→  if (data.length < 2) return null;
   873→  const prev = data[data.length - 2];
   874→  const curr = data[data.length - 1];
   875→  if (!isBearish(prev) || !isBullish(curr)) return null;
   876→  if (curr.open <= prev.close) return null;
   877→  if (curr.close >= prev.open) return null;
   878→  const engulfs = bodySize(curr) > bodySize(prev);
   879→  if (!engulfs) return null;
   880→  return {
   881→    name: 'پوشای صعودی', nameEn: 'Bullish Engulfing', category: 'candlestick',

  1576→function detectImpulseWave(data: OHLCV[]): PatternResult | null {
  1577→  if (data.length < 30) return null;
  1578→  const swings = findSwingPoints(data, 3);
  1579→  if (swings.length < 7) return null;
  1580→
  1581→  // Look for 5-wave structure in recent swings
  1582→  const recent = swings.slice(-7);
  1583→  // Impulse: wave 1 > wave 2 < wave 3 > wave 4 < wave 5
  1584→  // with wave 3 not shortest, wave 2 doesn't retrace past wave 1 start
  1585→  const prices = recent.map(s => s.price);
  1586→
  1587→  // Try to find 5-wave pattern: 1(up) 2(down) 3(up) 4(down) 5(up)
  1588→  for (let i = 0; i <= prices.length - 5; i++) {
  1589→    const w = prices.slice(i, i + 5);
  1590→    const isUp = (a: number, b: number) => b > a;
  1591→    const isDown = (a: number, b: number) => b < a;
  1592→
  1593→    if (isUp(w[0], w[1]) && isDown(w[1], w[2]) && isUp(w[2], w[3]) && isDown(w[3], w[4])) {
  1594→      // Wave 3 should be the longest (or at least not shortest) impulse
  1595→      const impulse1 = Math.abs(w[1] - w[0]);
  1596→      const impulse3 = Math.abs(w[3] - w[2]);
  1597→      const impulse5 = Math.abs(w[4] - w[3]);
  1598→      if (impulse3 >= impulse1 && impulse3 >= impulse5 * 0.8) {
  1599→        // Wave 2 should not retrace beyond wave 1 start
  1600→        const retrace2 = Math.abs(w[2] - w[1]) / impulse1;
  1601→        if (retrace2 < 1 && retrace2 > 0.2) {
  1602→          return {
  1603→            name: 'موج تکانه‌ای', nameEn: 'Impulse Wave', category: 'elliott',
  1604→            direction: 'bullish', strength: 0.7,
  1605→            status: 'forming',
  1606→            priceLevel: w[3],
  1607→            description: 'الگوی موج تکانه‌ای ۵ موجی الیوت — روند صعودی'
  1608→          };
  1609→        }
  1610→      }
  1611→    }
  1612→  }
  1613→  return null;
  1614→}

  1713→  if (data.length < 15) return null;
  1714→  const swings = findSwingPoints(data, 3);
  1715→  if (swings.length < 4) return null;
  1716→
  1717→  const recent = swings.slice(-4);
  1718→  const prices = recent.map(s => s.price);
  1719→  // Expanded flat: B extends beyond start of A, C extends beyond end of A
  1720→  if (recent.length < 4) return null;
  1721→  const start = prices[0];
  1722→  const aEnd = prices[1];
  1723→  const bEnd = prices[2];
  1724→  const cEnd = prices[3];
  1725→
  1726→  const bBeyond = (recent[2].type === 'high' && bEnd > start) || (recent[2].type === 'low' && bEnd < start);
  1727→  const cBeyond = (recent[3].type === 'low' && cEnd < aEnd) || (recent[3].type === 'high' && cEnd > aEnd);
  1728→
  1729→  if (!bBeyond || !cBeyond) return null;
  1730→
  1731→  return {
  1732→    name: 'تخت گسترده', nameEn: 'Expanded Flat', category: 'elliott',
  1733→    direction: 'neutral', strength: 0.55,
  1734→    status: 'forming',

  1739→function detectRunningFlat(data: OHLCV[]): PatternResult | null {
  1740→  if (data.length < 15) return null;
  1741→  const swings = findSwingPoints(data, 3);
  1742→  if (swings.length < 3) return null;
  1743→
  1744→  const recent = swings.slice(-3);
  1745→  const prices = recent.map(s => s.price);
  1746→  // Running flat: B ends beyond start of impulse, C doesn't reach end of A
  1747→  const aMove = Math.abs(prices[1] - prices[0]);
  1748→  const cMove = Math.abs(prices[2] - prices[1]);
  1749→  if (cMove > aMove) return null;
  1750→  if (cMove < aMove * 0.5) return null;

  1760→function detectElliottTriangle(data: OHLCV[]): PatternResult | null {
  1761→  if (data.length < 20) return null;
  1762→  const swings = findSwingPoints(data, 3);
  1763→  const highs = swings.filter(s => s.type === 'high');
  1764→  const lows = swings.filter(s => s.type === 'low');
  1765→  if (highs.length < 3 || lows.length < 3) return null;
  1766→
  1767→  const rh = highs.slice(-3);
  1768→  const rl = lows.slice(-3);
  1769→  const hConverge = rh[rh.length - 1].price < rh[0].price;
  1770→  const lConverge = rl[rl.length - 1].price > rl[0].price;
  1771→  if (!hConverge || !lConverge) return null;

  1773→  // Each sub-wave should have 3 waves (corrective)

  1782→function detectComplexCorrection(data: OHLCV[]): PatternResult | null {
  1783→  if (data.length < 35) return null;
  1784→  const swings = findSwingPoints(data, 3);
  1785→  if (swings.length < 7) return null;
  1786→
  1787→  const recent = swings.slice(-7);
  1788→  const prices = recent.map(s => s.price);
  1789→  // WXY: three corrective patterns linked — look for alternating wave directions
  1790→  let alternating = true;
  1791→  for (let i = 1; i < recent.length; i++) {
  1792→    if (recent[i].type === recent[i - 1].type) { alternating = false; break; }
  1793→  }
  1794→  if (!alternating) return null;
  1795→
  1796→  // Check for at least 2 distinct corrective sections
  1797→  const wEnd = Math.abs(prices[2] - prices[0]);
  1798→  const yEnd = Math.abs(prices[prices.length - 1] - prices[prices.length - 3]);
  1799→  if (wEnd === 0 || yEnd === 0) return null;
  1800→
  1801→  const ratio = yEnd / wEnd;
  1802→  if (ratio < 0.5 || ratio > 2) return null;
  1803→
  1804→  return {
  1805→    name: 'تصحیح پیچیده', nameEn: 'Complex Correction (WXY)', category: 'elliott',
  1806→    direction: 'neutral', strength: 0.5,
  1807→    status: 'forming',
  1808→    description: 'الگوی تصحیح پیچیده WXY — ترکیب چند الگوی تصحیحی'
  1809→  };
  1810→}

  1812→function detectDoubleThree(data: OHLCV[]): PatternResult | null {
  1813→  if (data.length < 30) return null;
  1814→  const swings = findSwingPoints(data, 3);
  1815→  if (swings.length < 6) return null;
  1816→
  1817→  const recent = swings.slice(-6);
  1818→  // Double three: two corrective patterns joined by an X wave
  1819→  // Look for roughly 6 alternating swings with similar amplitude ranges
  1820→  const prices = recent.map(s => s.price);
  1821→  const ranges: number[] = [];
  1822→  for (let i = 1; i < prices.length; i++) {
  1823→    ranges.push(Math.abs(prices[i] - prices[i - 1]));
  1824→  }
  1825→  const avgRange = avg(ranges);
  1826→  if (avgRange === 0) return null;
  1827→  const variance = ranges.reduce((s, r) => s + Math.abs(r - avgRange), 0) / avgRange;
  1828→  if (variance > 1.5) return null; // too irregular
  1829→
  1830→  return {
  1831→    name: 'دو سه‌تایی', nameEn: 'Double Three', category: 'elliott',
  1832→    direction: 'neutral', strength: 0.5,
  1833→    status: 'forming',
  1834→    description: 'الگوی دو سه‌تایی — ترکیب دو الگوی تصحیحی ساده'
  1835→  };
  1836→}

  1858→function computeSchoolScores(classic: PatternResult[], harmonic: PatternResult[], candlestick: PatternResult[], elliott: PatternResult[]): DetectedPatterns['schoolScores'] {
  1859→  const classicScore = classic.length > 0
  1860→    ? Math.min(1, avg(classic.map(p => p.strength)) * (1 + classic.length * 0.1))
  1861→    : 0;
  1862→
  1863→  const harmonicScore = harmonic.length > 0
  1864→    ? Math.min(1, avg(harmonic.map(p => p.strength)) * (1 + harmonic.length * 0.2))
  1865→    : 0;
  1866→
  1867→  const candleOscScore = candlestick.filter(p =>
  1868→    p.nameEn.includes('Doji') || p.nameEn.includes('Star') ||
  1869→    p.nameEn.includes('Harami') || p.nameEn.includes('Engulfing')
  1870→  );
  1871→  const oscillatorScore = candleOscScore.length > 0
  1872→    ? Math.min(1, avg(candleOscScore.map(p => p.strength)))
  1873→    : 0;
  1874→
  1875→  const candleVolScore = candlestick.filter(p =>
  1876→    p.nameEn.includes('Marubozu') || p.nameEn.includes('Kicker') ||
  1877→    p.nameEn.includes('Three Methods')
  1878→  );
  1879→  const volumeScore = candleVolScore.length > 0
  1880→    ? Math.min(1, avg(candleVolScore.map(p => p.strength)))
  1881→    : 0;
  1882→
  1883→  const elliottScore = elliott.length > 0
  1884→    ? Math.min(1, avg(elliott.map(p => p.strength)) * (1 + elliott.length * 0.15))
  1885→    : 0;
  1886→
  1887→  const allPatterns = [...classic, ...harmonic, ...candlestick, ...elliott];
  1888→  const hybridScore = allPatterns.length > 0
  1889→    ? Math.min(1, avg(allPatterns.map(p => p.strength)) * Math.min(allPatterns.length * 0.05, 1.5))
  1890→    : 0;
  1891→
  1892→  return {
  1893→    classical: Math.round(classicScore * 1000) / 1000,
  1894→    oscillator: Math.round(oscillatorScore * 1000) / 1000,
  1895→    volume: Math.round(volumeScore * 1000) / 1000,
  1896→    harmonic: Math.round(harmonicScore * 1000) / 1000,
  1897→    elliott: Math.round(elliottScore * 1000) / 1000,
  1898→    hybrid: Math.round(hybridScore * 1000) / 1000,
  1899→  };
  1900→}

  1904→export function detectAllPatterns(data: OHLCV[]): DetectedPatterns {
  1905→  const classic = detectAllClassic(data);

  1915→
  1916→  return {
