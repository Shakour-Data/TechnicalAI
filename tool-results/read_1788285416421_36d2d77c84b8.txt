   263→  // Store last fetch params for auto-refresh
   264→  const lastFetchRef = useRef<{ symbol: string; category?: string; insCode?: string; tgjuKey?: string; finpySector?: string; finpyIndex?: string; webId?: number; yahooSymbol?: string } | null>(null);
   265→  const refreshTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   266→  const fetchControllerRef = useRef<AbortController | null>(null);
   267→
   268→  // Chart container ref for image export
   269→  const chartWrapperRef = useRef<HTMLDivElement>(null);
   270→
   271→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number, yahooSymbol?: string) => {
   272→    // Abort any previous in-flight fetch
   273→    if (fetchControllerRef.current) {
   274→      fetchControllerRef.current.abort();
   275→    }
   276→    const controller = new AbortController();
   277→    fetchControllerRef.current = controller;
   278→
   279→    // Yahoo Finance instrument: fetch historical data via Yahoo Finance API
   280→    // This handles ALL Yahoo instruments (stocks, indices, energy, metals, forex, crypto, ETFs, etc.)
   281→    if (yahooSymbol) {
   282→      setLoading(true);
   283→      setLoadingMessage('در حال دریافت داده‌های تاریخی از یاهو فایننس ...');
   284→      setError(null);
   285→      setData(null);
   286→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId, yahooSymbol };
   287→      try {
   288→        const res = await fetch('/api/yahoo-analysis?symbol=' + encodeURIComponent(yahooSymbol), { signal: controller.signal });
   289→        if (controller.signal.aborted) return;
   290→        if (!res.ok) {
   291→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   292→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   293→          throw new Error(errMsg);
   294→        }
   295→        const json = await res.json();
   296→        if (controller.signal.aborted) return;
   297→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   298→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   299→        }
   300→        setData({ ...json, isYahoo: true });
   301→        setActivePanel('visual');
   302→      } catch (err) {
   303→        if (controller.signal.aborted) return;
   304→        setError(String(err));
   305→      } finally {
   306→        if (!controller.signal.aborted) {
   307→          setLoading(false);
   308→          setLoadingMessage(null);
   309→        }
   310→      }
   311→      return;
   312→    }
   313→
   314→    // TGJU instrument: fetch historical data via tgju.org chart API
   315→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   316→      setLoading(true);
   317→      setLoadingMessage('در حال دریافت داده‌های تاریخی ...');
   318→      setError(null);
   319→      setData(null);
