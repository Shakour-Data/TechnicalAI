    40→function hexToRgba(hex: string, alpha: number): string {
    41→  const r = parseInt(hex.slice(1, 3), 16);
    42→  const g = parseInt(hex.slice(3, 5), 16);
    43→  const b = parseInt(hex.slice(5, 7), 16);
    44→  return `rgba(${r},${g},${b},${alpha})`;
    45→}
    46→
    47→const toFa = (n: number) => Math.round(n).toLocaleString('fa-IR');
    48→const toFa1 = (n: number) => toPersianDigits(n.toFixed(1)).replace(/\./g, '٫').replace(/-/g, '−');
    49→const toFa2 = (n: number) => toPersianDigits(n.toFixed(2)).replace(/\./g, '٫').replace(/-/g, '−');
    50→
    51→function rsiSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    52→  if (v > 70) return 'bearish';
    53→  if (v < 30) return 'bullish';
    54→  return 'neutral';
    55→}
    56→
    57→function macdSignal(h: number): 'bullish' | 'bearish' | 'neutral' {
    58→  if (h > 0) return 'bullish';
    59→  if (h < 0) return 'bearish';
    60→  return 'neutral';
    61→}
    62→
    63→function adxSignal(v: number): 'strong' | 'neutral' {
    64→  if (v > 25) return 'strong';
    65→  return 'neutral';
    66→}
    67→
    68→function stochSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    69→  if (v > 80) return 'bearish';
    70→  if (v < 20) return 'bullish';
    71→  return 'neutral';
    72→}
    73→
    74→function signalDotBg(signal: 'bullish' | 'bearish' | 'neutral' | 'strong', bullColor: string, bearColor: string, neutralColor: string, cardSubFg: string): string {
    75→  switch (signal) {
    76→    case 'bullish': return bullColor;
    77→    case 'bearish': return bearColor;
    78→    case 'strong': return neutralColor;
    79→    default: return cardSubFg;
    80→  }
    81→}
    82→
    83→function gradeLabel(grade: string): string {
    84→  switch (grade) {
    85→    case 'Very Strong': return 'بسیار قوی';
    86→    case 'Strong': return 'قوی';
    87→    case 'Moderate': return 'متوسط';
    88→    default: return 'ضعیف';
    89→  }
    90→}
    91→
    92→function gradeStyle(grade: string, bearColor: string, neutralColor: string, primary: string, cardBg: string, cardSubFg: string): React.CSSProperties {
    93→  switch (grade) {
    94→    case 'Very Strong': return { background: hexToRgba(bearColor, 0.15), color: bearColor };
    95→    case 'Strong': return { background: hexToRgba(neutralColor, 0.15), color: neutralColor };
    96→    case 'Moderate': return { background: hexToRgba(primary, 0.15), color: primary };
    97→    default: return { background: hexToRgba(cardSubFg, 0.1), color: cardSubFg };
    98→  }
    99→}
   100→
   101→function formatNumber(n: number): string {
   102→  if (Math.abs(n) >= 1e9) return toPersianDigits((n / 1e9).toFixed(1)) + 'B';
   103→  if (Math.abs(n) >= 1e6) return toPersianDigits((n / 1e6).toFixed(1)) + 'M';
   104→  if (Math.abs(n) >= 1e3) return toPersianDigits((n / 1e3).toFixed(1)) + 'K';
   105→  return toFa(n);
   106→}
   107→
   108→function pctDistance(from: number, to: number): number {
   109→  if (from === 0) return 0;
   110→  return ((to - from) / from) * 100;
   111→}
   112→
   113→function trendDirectionLabel(dir: string, bullColor: string, bearColor: string, neutralColor: string): { label: string; color: string } {
   114→  const d = dir?.toLowerCase() ?? '';
   115→  if (d.includes('up') || d === 'صعودی') return { label: 'صعودی', color: bullColor };
   116→  if (d.includes('down') || d === 'نزولی') return { label: 'نزولی', color: bearColor };
   117→  return { label: 'خنثی', color: neutralColor };
   118→}

   131→function GlassCard({ children, className = '', style }: { children: React.ReactNode; className?: string; style?: React.CSSProperties }) {

   137→    >
   138→      {children}
   139→    </div>

   143→function ProgressBar({ value, max, color, height = 4 }: { value: number; max: number; color: string; height?: number }) {
   144→  const { colors: C } = useTheme();
   145→  const pct = Math.min(100, Math.max(0, (value / max) * 100));
   146→  return (
   147→    <div className="w-full rounded-full overflow-hidden" style={{ height, background: C.cardBorder }}>
   148→      <div className="h-full rounded-full transition-all duration-500" style={{ width: `${pct}%`, background: color }} />
   149→    </div>
   150→  );
   151→}
