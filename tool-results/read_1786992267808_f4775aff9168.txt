   104→   104→   104→function calcRSI(closes: number[], period: number = 14): number {
   105→   105→   105→  if (closes.length < period + 1) return 50;
   106→   106→   106→  let avgGain = 0;
   107→   107→   107→  let avgLoss = 0;
   108→   108→   108→  for (let i = 1; i <= period; i++) {
   109→   109→   109→    const diff = closes[i] - closes[i - 1];
   110→   110→   110→    if (diff > 0) avgGain += diff;
   111→   111→   111→    else avgLoss += Math.abs(diff);
   112→   112→   112→  }
   113→   113→   113→  avgGain /= period;
   114→   114→   114→  avgLoss /= period;
   115→   115→   115→  for (let i = period + 1; i < closes.length; i++) {
   116→   116→   116→    const diff = closes[i] - closes[i - 1];
   117→   117→   117→    const gain = diff > 0 ? diff : 0;
   118→   118→   118→    const loss = diff < 0 ? Math.abs(diff) : 0;
   119→   119→   119→    avgGain = (avgGain * (period - 1) + gain) / period;
   120→   120→   120→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   121→   121→   121→  }
   122→   122→   122→  if (avgLoss === 0) return 100;
   123→   123→   123→  const rs = avgGain / avgLoss;
   124→   124→   124→  return 100 - 100 / (1 + rs);
   125→   125→   125→}

   133→   133→   133→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   134→   134→   134→    const mf = tp * data[i].volume;
   135→   135→   135→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   136→   136→   136→    if (tp > prevTp) posFlow += mf;
   137→   137→   137→    else negFlow += mf;
   138→   138→   138→  }
   139→   139→   139→  if (negFlow === 0) return 100;
   140→   140→   140→  const mfr = posFlow / negFlow;
   141→   141→   141→  return 100 - 100 / (1 + mfr);
   142→   142→   142→}

   158→   158→   158→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   159→   159→   159→  if (data.length < kPeriod) return { k: 50, d: 50 };
   160→   160→   160→  const rawKs: number[] = [];
   161→   161→   161→  for (let i = kPeriod - 1; i < data.length; i++) {
   162→   162→   162→    let lowest = Infinity;
   163→   163→   163→    let highest = -Infinity;
   164→   164→   164→    for (let j = i - kPeriod + 1; j <= i; j++) {
   165→   165→   165→      if (data[j].low < lowest) lowest = data[j].low;
   166→   166→   166→      if (data[j].high > highest) highest = data[j].high;
   167→   167→   167→    }
   168→   168→   168→    const range = highest - lowest;
   169→   169→   169→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   170→   170→   170→  }
   171→   171→   171→  const smoothedK: number[] = [];
   172→   172→   172→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   173→   173→   173→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   174→   174→   174→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   175→   175→   175→  }
   176→   176→   176→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   177→   177→   177→  const dSlice = smoothedK.slice(-smoothD);
   178→   178→   178→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   179→   179→   179→  return { k, d };
   180→   180→   180→}

   198→   198→   198→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   199→   199→   199→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   200→   200→   200→  const emaFast = emaCalc(closes, fast);
   201→   201→   201→  const emaSlow = emaCalc(closes, slow);
   202→   202→   202→  const macdLine = emaFast - emaSlow;
   203→   203→   203→  // Calculate signal as EMA of MACD values
   204→   204→   204→  const macdValues: number[] = [];
   205→   205→   205→  const kFast = 2 / (fast + 1);
   206→   206→   206→  const kSlow = 2 / (slow + 1);
   207→   207→   207→  let eFast = sma(closes.slice(0, fast), fast);
   208→   208→   208→  let eSlow = sma(closes.slice(0, slow), slow);
   209→   209→   209→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   210→   210→   210→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   211→   211→   211→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   212→   212→   212→    macdValues.push(eFast - eSlow);
   213→   213→   213→  }
   214→   214→   214→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   215→   215→   215→  const kSig = 2 / (sig + 1);
   216→   216→   216→  let signalVal = sma(macdValues.slice(0, sig), sig);
   217→   217→   217→  for (let i = sig; i < macdValues.length; i++) {
   218→   218→   218→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   219→   219→   219→  }
   220→   220→   220→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   221→   221→   221→}

   224→   224→   224→function calcATR(data: OHLCV[], period: number = 14): number {

   230→   230→   230→      Math.abs(data[i].high - data[i - 1].close),
   231→   231→   231→      Math.abs(data[i].low - data[i - 1].close)
   232→   232→   232→    );
   233→   233→   233→    trs.push(tr);
   234→   234→   234→  }
   235→   235→   235→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   236→   236→   236→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   237→   237→   237→  for (let i = period; i < trs.length; i++) {
   238→   238→   238→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   239→   239→   239→  }
   240→   240→   240→  return atrVal;

   244→   244→   244→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   245→   245→   245→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   246→   246→   246→  const plusDMs: number[] = [];
   247→   247→   247→  const minusDMs: number[] = [];
   248→   248→   248→  const trs: number[] = [];
   249→   249→   249→  for (let i = 1; i < data.length; i++) {
   250→   250→   250→    const upMove = data[i].high - data[i - 1].high;
   251→   251→   251→    const downMove = data[i - 1].low - data[i].low;
   252→   252→   252→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   253→   253→   253→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   254→   254→   254→    trs.push(Math.max(
   255→   255→   255→      data[i].high - data[i].low,
   256→   256→   256→      Math.abs(data[i].high - data[i - 1].close),
   257→   257→   257→      Math.abs(data[i].low - data[i - 1].close)
   258→   258→   258→    ));
   259→   259→   259→  }
   260→   260→   260→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   261→   261→   261→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   262→   262→   262→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   263→   263→   263→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   264→   264→   264→  const dxValues: number[] = [];
   265→   265→   265→  for (let i = period; i < trs.length; i++) {
   266→   266→   266→    smoothTR = smoothTR - smoothTR / period + trs[i];
   267→   267→   267→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   268→   268→   268→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   269→   269→   269→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   270→   270→   270→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   271→   271→   271→    const diSum = diPlus + diMinus;
   272→   272→   272→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   273→   273→   273→    dxValues.push(dx);
   274→   274→   274→  }
   275→   275→   275→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   276→   276→   276→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   277→   277→   277→  for (let i = period; i < dxValues.length; i++) {
   278→   278→   278→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   279→   279→   279→  }
   280→   280→   280→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   281→   281→   281→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   282→   282→   282→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   283→   283→   283→}

   286→   286→   286→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   287→   287→   287→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   288→   288→   288→  let isLong = true;
   289→   289→   289→  let af = step;
   290→   290→   290→  let ep = data[0].high;
   291→   291→   291→  let sar = data[0].low;
   292→   292→   292→  for (let i = 1; i < data.length; i++) {
   293→   293→   293→    const prevSar = sar;
   294→   294→   294→    sar = prevSar + af * (ep - prevSar);
   295→   295→   295→    // Ensure SAR is within bounds
   296→   296→   296→    if (isLong) {
   297→   297→   297→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   298→   298→   298→      if (data[i].low < sar) {
   299→   299→   299→        isLong = false;
   300→   300→   300→        sar = ep;
   301→   301→   301→        ep = data[i].low;
   302→   302→   302→        af = step;
   303→   303→   303→      } else {
   304→   304→   304→        if (data[i].high > ep) {
   305→   305→   305→          ep = data[i].high;
   306→   306→   306→          af = Math.min(af + step, max);
   307→   307→   307→        }
   308→   308→   308→      }
   309→   309→   309→    } else {
   310→   310→   310→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   311→   311→   311→      if (data[i].high > sar) {
   312→   312→   312→        isLong = true;
   313→   313→   313→        sar = ep;
   314→   314→   314→        ep = data[i].high;
   315→   315→   315→        af = step;
   316→   316→   316→      } else {
   317→   317→   317→        if (data[i].low < ep) {
   318→   318→   318→          ep = data[i].low;
   319→   319→   319→          af = Math.min(af + step, max);
   320→   320→   320→        }
   321→   321→   321→      }
   322→   322→   322→    }
   323→   323→   323→  }
   324→   324→   324→  return sar;
   325→   325→   325→}

   351→   351→   351→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   352→   352→   352→  const n = values.length;
   353→   353→   353→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   354→   354→   354→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   355→   355→   355→  for (let i = 0; i < n; i++) {
   356→   356→   356→    sumX += i;
   357→   357→   357→    sumY += values[i];
   358→   358→   358→    sumXY += i * values[i];
   359→   359→   359→    sumX2 += i * i;
   360→   360→   360→    sumY2 += values[i] * values[i];
   361→   361→   361→  }
   362→   362→   362→  const denom = n * sumX2 - sumX * sumX;
   363→   363→   363→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   364→   364→   364→  const slope = (n * sumXY - sumX * sumY) / denom;
   365→   365→   365→  const intercept = (sumY - slope * sumX) / n;
   366→   366→   366→  const meanY = sumY / n;
   367→   367→   367→  let ssRes = 0, ssTot = 0;
   368→   368→   368→  for (let i = 0; i < n; i++) {
   369→   369→   369→    const predicted = slope * i + intercept;
   370→   370→   370→    ssRes += (values[i] - predicted) ** 2;
   371→   371→   371→    ssTot += (values[i] - meanY) ** 2;
   372→   372→   372→  }
   373→   373→   373→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   374→   374→   374→  return { slope, intercept, r2 };
   375→   375→   375→}

   413→   413→   413→  priceTargets: LevelStrength[];

   415→   415→   415→  const allRawLevels: number[] = [];
   416→   416→   416→  const closes = data.map(d => d.close);
   417→   417→   417→
   418→   418→   418→  // 1. Pivot Points from previous day (5R + 5S)
   419→   419→   419→  const prev = data[data.length - 2] ?? data[data.length - 1];
   420→   420→   420→  if (prev) {
   421→   421→   421→    const pp = (prev.high + prev.low + prev.close) / 3;
   422→   422→   422→    const hl = prev.high - prev.low;
   423→   423→   423→    const pivotR = [
   424→   424→   424→      2 * pp - prev.low,
   425→   425→   425→      pp + hl,
   426→   426→   426→      prev.high + 2 * (pp - prev.low),
   427→   427→   427→      prev.high + 2 * (pp - prev.low) + hl,
   428→   428→   428→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   429→   429→   429→    ];
   430→   430→   430→    const pivotS = [
   431→   431→   431→      2 * pp - prev.high,
   432→   432→   432→      pp - hl,
   433→   433→   433→      prev.low - 2 * (prev.high - pp),
   434→   434→   434→      prev.low - 2 * (prev.high - pp) - hl,
   435→   435→   435→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   436→   436→   436→    ];
   437→   437→   437→    allRawLevels.push(...pivotR, ...pivotS);
   438→   438→   438→  }
   439→   439→   439→
   440→   440→   440→  // 2. Swing levels (multiple lookback periods for better coverage)
   441→   441→   441→  for (const lb of [3, 5, 7, 10]) {
   442→   442→   442→    const swings = findSwingLevels(data, lb);
   443→   443→   443→    allRawLevels.push(...swings.highs, ...swings.lows);
   444→   444→   444→  }
   445→   445→   445→
   446→   446→   446→  // 3. Moving Average levels (potential dynamic S/R)
   447→   447→   447→  for (const period of [5, 10, 21, 50, 100, 200]) {
   448→   448→   448→    const v = sma(closes, period);
   449→   449→   449→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01) allRawLevels.push(v);
   450→   450→   450→  }
   451→   451→   451→  const ema12 = emaCalc(closes, 12);
   452→   452→   452→  const ema26 = emaCalc(closes, 26);
   453→   453→   453→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01) allRawLevels.push(ema12);
   454→   454→   454→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01) allRawLevels.push(ema26);
   455→   455→   455→
   456→   456→   456→  // 4. Bollinger Band levels
   457→   457→   457→  const bbLevels = calcBollingerBands(closes);
   458→   458→   458→  if (bbLevels.upper > 0) allRawLevels.push(bbLevels.upper);
   459→   459→   459→  if (bbLevels.lower > 0) allRawLevels.push(bbLevels.lower);
   460→   460→   460→
   461→   461→   461→  // 5. Round number / psychological levels
   462→   462→   462→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
   463→   463→   463→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
   464→   464→   464→    allRawLevels.push(Math.round(currentPrice * mult / magnitude) * magnitude);
   465→   465→   465→  }
   466→   466→   466→
   467→   467→   467→  // 6. Recent high/low zones (5-day, 22-day, 66-day)
   468→   468→   468→  for (const window of [5, 22, 66]) {
   469→   469→   469→    const slice = data.slice(-Math.min(window, data.length));
   470→   470→   470→    if (slice.length > 0) {
   471→   471→   471→      allRawLevels.push(Math.max(...slice.map(d => d.high)));
   472→   472→   472→      allRawLevels.push(Math.min(...slice.map(d => d.low)));
   473→   473→   473→    }
   474→   474→   474→  }
   475→   475→   475→
   476→   476→   476→  // 7. Round all and filter invalid
   477→   477→   477→  const rounded = allRawLevels.map(roundToNice).filter(l => l > 0);
   478→   478→   478→
   479→   479→   479→  // 8. Remove duplicates (within 0.5%)
   480→   480→   480→  const unique: number[] = [];
   481→   481→   481→  for (const level of rounded) {
   482→   482→   482→    const isDup = unique.some(u => Math.abs(u - level) / Math.max(u, 1) < 0.005);
   483→   483→   483→    if (!isDup) unique.push(level);
   484→   484→   484→  }
   485→   485→   485→
   486→   486→   486→  // 9. Calculate strength for each unique level — 10% window (5% above + 5% below)
   487→   487→   487→  const levelStrengths: LevelStrength[] = unique.map(level => {
   488→   488→   488→    const range10pct = level * 0.10;
   489→   489→   489→    const nearbyCount = unique.filter(other =>
   490→   490→   490→      other !== level && Math.abs(other - level) <= range10pct
   491→   491→   491→    ).length;
   492→   492→   492→    // Normalize: 0 nearby = 1, each nearby adds ~1.5, max 10
   493→   493→   493→    const strength = Math.min(10, Math.max(1, Math.round(1 + nearbyCount * 1.5)));
   494→   494→   494→    return { price: level, strength, isTarget: false };
   495→   495→   495→  });
   496→   496→   496→
   497→   497→   497→  // 10. Separate above/below current price
   498→   498→   498→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   499→   499→   499→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   500→   500→   500→
   501→   501→   501→  // 11. Enforce 2%-5% gap, always return exactly targetCount levels
   502→   502→   502→  //    Step 1: Merge levels closer than 2%
   503→   503→   503→  //    Step 2: Walk from nearest level, keep only chain within 2-5% gap
   504→   504→   504→  //    Step 3: If fewer than targetCount, extend chain by computing ~3% steps outward
   505→   505→   505→  //    Step 4: If more than targetCount, trim by keeping highest strength
   506→   506→   506→  function enforceGap(
   507→   507→   507→    levels: LevelStrength[],
   508→   508→   508→    targetCount: number,
   509→   509→   509→    _allUnique: LevelStrength[],
   510→   510→   510→  ): LevelStrength[] {
   511→   511→   511→    if (levels.length === 0) return [];
   512→   512→   512→
   513→   513→   513→    const MIN_GAP = 0.02;
   514→   514→   514→    const MAX_GAP = 0.05;
   515→   515→   515→    const EXTEND_GAP = 0.035; // 3.5% — middle of 2-5% range
   516→   516→   516→    const step = currentPrice >= 100000 ? 100 : currentPrice >= 10000 ? 50 : currentPrice >= 1000 ? 10 : currentPrice >= 100 ? 5 : 1;
   517→   517→   517→    const isDescending = levels.length > 0 && levels[0].price > (levels[levels.length - 1]?.price ?? 0);
   518→   518→   518→
   519→   519→   519→    // Step 1: Merge levels closer than MIN_GAP
   520→   520→   520→    const merged: LevelStrength[] = [levels[0]];
   521→   521→   521→    for (let i = 1; i < levels.length; i++) {
   522→   522→   522→      const prevPrice = merged[merged.length - 1].price;
   523→   523→   523→      const currPrice = levels[i].price;
   524→   524→   524→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   525→   525→   525→      if (gap < MIN_GAP) {
   526→   526→   526→        const mergedStrength = Math.min(10, Math.max(merged[merged.length - 1].strength, levels[i].strength) + 1);
   527→   527→   527→        merged[merged.length - 1] = {
   528→   528→   528→          ...(levels[i].strength > merged[merged.length - 1].strength ? levels[i] : merged[merged.length - 1]),
   529→   529→   529→          strength: mergedStrength,
   530→   530→   530→        };
   531→   531→   531→      } else {
   532→   532→   532→        merged.push(levels[i]);
   533→   533→   533→      }
   534→   534→   534→    }
   535→   535→   535→
   536→   536→   536→    // Step 2: Walk from first level (nearest to price), keep only if gap is <= MAX_GAP
   537→   537→   537→    const chain: LevelStrength[] = [merged[0]];
   538→   538→   538→    for (let i = 1; i < merged.length; i++) {
   539→   539→   539→      const prevPrice = chain[chain.length - 1].price;
   540→   540→   540→      const currPrice = merged[i].price;
   541→   541→   541→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   542→   542→   542→      if (gap <= MAX_GAP) {
   543→   543→   543→        chain.push(merged[i]);
   544→   544→   544→      }
   545→   545→   545→    }
   546→   546→   546→
   547→   547→   547→    // Step 3: If fewer than targetCount, extend the chain outward at ~3.5% intervals
   548→   548→   548→    while (chain.length < targetCount) {
   549→   549→   549→      const lastPrice = chain[chain.length - 1].price;
   550→   550→   550→      const nextRaw = isDescending
   551→   551→   551→        ? lastPrice * (1 - EXTEND_GAP)  // supports go down
   552→   552→   552→        : lastPrice * (1 + EXTEND_GAP); // resistances go up
   553→   553→   553→      const nextPrice = Math.round(nextRaw / step) * step;
   554→   554→   554→      if (nextPrice <= 0) break;
   555→   555→   555→      if (nextPrice === lastPrice) break; // rounding made no change (very low price)
   556→   556→   556→      chain.push({ price: nextPrice, strength: 2, isTarget: false });
   557→   557→   557→    }
   558→   558→   558→
   559→   559→   559→    // Step 4: If more than targetCount, trim by keeping highest strength
   560→   560→   560→    if (chain.length > targetCount) {
   561→   561→   561→      chain.sort((a, b) => b.strength - a.strength);
   562→   562→   562→      const kept = chain.slice(0, targetCount);
   563→   563→   563→      kept.sort((a, b) => isDescending ? b.price - a.price : a.price - b.price);
   564→   564→   564→      return kept;
   565→   565→   565→    }
   566→   566→   566→
   567→   567→   567→    return chain;
   568→   568→   568→  }
   569→   569→   569→
   570→   570→   570→  const finalResistances = enforceGap(above, 6, levelStrengths);
   571→   571→   571→  const finalSupports = enforceGap(below, 6, levelStrengths);
   572→   572→   572→
   573→   573→   573→  // 12. Determine price targets: strength > 7 is a target
   574→   574→   574→  const allLevels = [...finalResistances, ...finalSupports];
   575→   575→   575→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   576→   576→   576→  let targets: LevelStrength[];
   577→   577→   577→  if (sortedByStrength.some(l => l.strength > 7)) {
   578→   578→   578→    targets = sortedByStrength.filter(l => l.strength > 7);
   579→   579→   579→  } else {
   580→   580→   580→    targets = sortedByStrength.slice(0, 2);
   581→   581→   581→  }
   582→   582→   582→
   583→   583→   583→  // Mark targets
   584→   584→   584→  targets = targets.map(t => ({ ...t, isTarget: true }));
   585→   585→   585→
   586→   586→   586→  // Also mark in final arrays
   587→   587→   587→  finalResistances.forEach(r => {
   588→   588→   588→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   589→   589→   589→  });
   590→   590→   590→  finalSupports.forEach(s => {
   591→   591→   591→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   592→   592→   592→  });
   593→   593→   593→
   594→   594→   594→  return {
   595→   595→   595→    resistances: finalResistances.map(l => l.price),
   596→   596→   596→    supports: finalSupports.map(l => l.price),
   597→   597→   597→    supportStrengths: finalSupports,
   598→   598→   598→    resistanceStrengths: finalResistances,
   599→   599→   599→    priceTargets: targets,
   600→   600→   600→  };
   601→   601→   601→}

   604→   604→   604→function calcTrend(closes: number[], period: number): TrendResult {
   605→   605→   605→  const slice = closes.slice(-Math.min(period, closes.length));
   606→   606→   606→  const { slope, r2 } = linearRegression(slice);
   607→   607→   607→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   608→   608→   608→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   609→   609→   609→  const priceThreshold = avgPrice * 0.001; // 0.1%
   610→   610→   610→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   611→   611→   611→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   612→   612→   612→}
