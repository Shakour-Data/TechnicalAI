    62→    62→function hexToRgba(hex: string, alpha: number): string {
    63→    63→  const r = parseInt(hex.slice(1, 3), 16);
    64→    64→  const g = parseInt(hex.slice(3, 5), 16);
    65→    65→  const b = parseInt(hex.slice(5, 7), 16);
    66→    66→  return `rgba(${r},${g},${b},${alpha})`;
    67→    67→}

    71→    71→function rsiSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    72→    72→  if (v > 70) return 'bearish';
    73→    73→  if (v < 30) return 'bullish';
    74→    74→  return 'neutral';
    75→    75→}

    77→    77→function mfiSignal(v: number): 'bullish' | 'bearish' | 'neutral' {

    83→    83→function stochSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    84→    84→  if (v > 80) return 'bearish';
    85→    85→  if (v < 20) return 'bullish';
    86→    86→  return 'neutral';

    89→    89→function macdSignal(v: { line: number; signal: number; histogram: number }): 'bullish' | 'bearish' | 'neutral' {
    90→    90→  if (v.histogram > 0 && v.line > v.signal) return 'bullish';
    91→    91→  if (v.histogram < 0 && v.line < v.signal) return 'bearish';
    92→    92→  return 'neutral';
    93→    93→}

    96→    96→  if (adx < 20) return 'neutral';
