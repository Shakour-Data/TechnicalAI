    87→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
    88→function sma(closes: number[], period: number): number {
    89→  if (closes.length < period) return 0;
    90→  const slice = closes.slice(-period);
    91→  return slice.reduce((a, b) => a + b, 0) / period;
    92→}

    94→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
    95→function emaCalc(closes: number[], period: number): number {
    96→  if (closes.length < period) return 0;
    97→  const k = 2 / (period + 1);
    98→  let emaVal = sma(closes.slice(0, period), period);
    99→  for (let i = period; i < closes.length; i++) {
   100→    emaVal = closes[i] * k + emaVal * (1 - k);
   101→  }
   102→  return emaVal;
   103→}

   105→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   106→function calcRSI(closes: number[], period: number = 14): number {
   107→  if (closes.length < period + 1) return 50;
   108→  let avgGain = 0;
   109→  let avgLoss = 0;
   110→  for (let i = 1; i <= period; i++) {
   111→    const diff = closes[i] - closes[i - 1];
   112→    if (diff > 0) avgGain += diff;
   113→    else avgLoss += Math.abs(diff);
   114→  }
   115→  avgGain /= period;
   116→  avgLoss /= period;
   117→  for (let i = period + 1; i < closes.length; i++) {
   118→    const diff = closes[i] - closes[i - 1];
   119→    const gain = diff > 0 ? diff : 0;
   120→    const loss = diff < 0 ? Math.abs(diff) : 0;
   121→    avgGain = (avgGain * (period - 1) + gain) / period;
   122→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   123→  }
   124→  if (avgLoss === 0) return 100;
   125→  const rs = avgGain / avgLoss;
   126→  return 100 - 100 / (1 + rs);
   127→}

   129→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   130→function calcMFI(data: OHLCV[], period: number = 14): number {
   131→  if (data.length < period + 1) return 50;
   132→  let posFlow = 0;
   133→  let negFlow = 0;
   134→  for (let i = data.length - period; i < data.length; i++) {
   135→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   136→    const mf = tp * data[i].volume;
   137→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   138→    if (tp > prevTp) posFlow += mf;
   139→    else negFlow += mf;
   140→  }
   141→  if (negFlow === 0) return 100;
   142→  const mfr = posFlow / negFlow;
   143→  return 100 - 100 / (1 + mfr);
   144→}

   146→// ─── Helper: CCI(20) ──────────────────────────────────────────────────────────
   147→function calcCCI(data: OHLCV[], period: number = 20): number {
   148→  if (data.length < period) return 0;
   149→  const tps: number[] = [];
   150→  for (let i = data.length - period; i < data.length; i++) {
   151→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   152→  }
   153→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   154→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   155→  if (meanDev === 0) return 0;
   156→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   157→}

   159→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   160→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   161→  if (data.length < kPeriod) return { k: 50, d: 50 };
   162→  const rawKs: number[] = [];
   163→  for (let i = kPeriod - 1; i < data.length; i++) {
   164→    let lowest = Infinity;
   165→    let highest = -Infinity;
   166→    for (let j = i - kPeriod + 1; j <= i; j++) {
   167→      if (data[j].low < lowest) lowest = data[j].low;
   168→      if (data[j].high > highest) highest = data[j].high;
   169→    }
   170→    const range = highest - lowest;
   171→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   172→  }
   173→  const smoothedK: number[] = [];
   174→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   175→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   176→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   177→  }
   178→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   179→  const dSlice = smoothedK.slice(-smoothD);
   180→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   181→  return { k, d };
   182→}

   184→// ─── Helper: Williams %R(14) ──────────────────────────────────────────────────
   185→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   186→  if (data.length < period) return -50;
   187→  const slice = data.slice(-period);
   188→  let highest = -Infinity;
   189→  let lowest = Infinity;
   190→  for (const c of slice) {
   191→    if (c.high > highest) highest = c.high;
   192→    if (c.low < lowest) lowest = c.low;
   193→  }
   194→  const range = highest - lowest;
   195→  if (range === 0) return -50;
   196→  return ((highest - data[data.length - 1].close) / range) * -100;
   197→}

   199→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   200→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   201→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   202→  const emaFast = emaCalc(closes, fast);
   203→  const emaSlow = emaCalc(closes, slow);
   204→  const macdLine = emaFast - emaSlow;
   205→  // Calculate signal as EMA of MACD values
   206→  const macdValues: number[] = [];
   207→  const kFast = 2 / (fast + 1);
   208→  const kSlow = 2 / (slow + 1);
   209→  let eFast = sma(closes.slice(0, fast), fast);
   210→  let eSlow = sma(closes.slice(0, slow), slow);
   211→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   212→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   213→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   214→    macdValues.push(eFast - eSlow);
   215→  }
   216→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   217→  const kSig = 2 / (sig + 1);
   218→  let signalVal = sma(macdValues.slice(0, sig), sig);
   219→  for (let i = sig; i < macdValues.length; i++) {
   220→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   221→  }
   222→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   223→}

   225→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   226→function calcATR(data: OHLCV[], period: number = 14): number {
   227→  if (data.length < 2) return 0;
   228→  const trs: number[] = [];
   229→  for (let i = 1; i < data.length; i++) {
   230→    const tr = Math.max(
   231→      data[i].high - data[i].low,
   232→      Math.abs(data[i].high - data[i - 1].close),
   233→      Math.abs(data[i].low - data[i - 1].close)
   234→    );
   235→    trs.push(tr);
   236→  }
   237→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   238→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   239→  for (let i = period; i < trs.length; i++) {
   240→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   241→  }
   242→  return atrVal;
   243→}

   245→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   246→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   247→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   248→  const plusDMs: number[] = [];
   249→  const minusDMs: number[] = [];
   250→  const trs: number[] = [];
   251→  for (let i = 1; i < data.length; i++) {
   252→    const upMove = data[i].high - data[i - 1].high;
   253→    const downMove = data[i - 1].low - data[i].low;
   254→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   255→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   256→    trs.push(Math.max(
   257→      data[i].high - data[i].low,
   258→      Math.abs(data[i].high - data[i - 1].close),
   259→      Math.abs(data[i].low - data[i - 1].close)
   260→    ));
   261→  }
   262→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   263→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   264→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   265→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   266→  const dxValues: number[] = [];
   267→  for (let i = period; i < trs.length; i++) {
   268→    smoothTR = smoothTR - smoothTR / period + trs[i];
   269→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   270→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   271→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   272→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   273→    const diSum = diPlus + diMinus;
   274→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   275→    dxValues.push(dx);
   276→  }
   277→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   278→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   279→  for (let i = period; i < dxValues.length; i++) {
   280→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   281→  }
   282→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   283→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   284→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   285→}

   287→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   288→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   289→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   290→  let isLong = true;
   291→  let af = step;
   292→  let ep = data[0].high;
   293→  let sar = data[0].low;
   294→  for (let i = 1; i < data.length; i++) {
   295→    const prevSar = sar;
   296→    sar = prevSar + af * (ep - prevSar);
   297→    // Ensure SAR is within bounds
   298→    if (isLong) {
   299→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   300→      if (data[i].low < sar) {
   301→        isLong = false;
   302→        sar = ep;
   303→        ep = data[i].low;
   304→        af = step;
   305→      } else {
   306→        if (data[i].high > ep) {
   307→          ep = data[i].high;
   308→          af = Math.min(af + step, max);
   309→        }
   310→      }
   311→    } else {
   312→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   313→      if (data[i].high > sar) {
   314→        isLong = true;
   315→        sar = ep;
   316→        ep = data[i].high;
   317→        af = step;
   318→      } else {
   319→        if (data[i].low < ep) {
   320→          ep = data[i].low;
   321→          af = Math.min(af + step, max);
   322→        }
   323→      }
   324→    }
   325→  }
   326→  return sar;
   327→}

   329→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   330→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   331→  if (closes.length < period) {
   332→    const c = closes[closes.length - 1] ?? 0;
   333→    return { upper: c, middle: c, lower: c };
   334→  }
   335→  const slice = closes.slice(-period);
   336→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   337→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   338→  const sd = Math.sqrt(variance);
   339→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   340→}

   342→// ─── Helper: OBV ──────────────────────────────────────────────────────────────
   343→function calcOBV(data: OHLCV[]): number {
   344→  let obv = 0;
   345→  for (let i = 1; i < data.length; i++) {
   346→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   347→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   348→  }
   349→  return obv;

   352→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   353→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   354→  const n = values.length;
   355→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   356→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   357→  for (let i = 0; i < n; i++) {
   358→    sumX += i;
   359→    sumY += values[i];
   360→    sumXY += i * values[i];
   361→    sumX2 += i * i;
   362→    sumY2 += values[i] * values[i];
   363→  }
   364→  const denom = n * sumX2 - sumX * sumX;
   365→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   366→  const slope = (n * sumXY - sumX * sumY) / denom;
   367→  const intercept = (sumY - slope * sumX) / n;
   368→  const meanY = sumY / n;
   369→  let ssRes = 0, ssTot = 0;
   370→  for (let i = 0; i < n; i++) {
   371→    const predicted = slope * i + intercept;
   372→    ssRes += (values[i] - predicted) ** 2;
   373→    ssTot += (values[i] - meanY) ** 2;
   374→  }
   375→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   376→  return { slope, intercept, r2 };
   377→}

   379→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   380→function roundToNice(price: number): number {
   381→  const abs = Math.abs(price);
   382→  let step: number;
   383→  if (abs >= 100000) step = 100;
   384→  else if (abs >= 10000) step = 50;
   385→  else if (abs >= 1000) step = 10;
   386→  else if (abs >= 100) step = 5;
   387→  else step = 1;
   388→  return Math.round(price / step) * step;
   389→}

   391→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
   392→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: number[]; lows: number[] } {
   393→  const highs: number[] = [];
   394→  const lows: number[] = [];
   395→  for (let i = lookback; i < data.length - lookback; i++) {
   396→    let isHigh = true;
   397→    let isLow = true;
   398→    for (let j = i - lookback; j <= i + lookback; j++) {
   399→      if (j === i) continue;
   400→      if (data[j].high >= data[i].high) isHigh = false;
   401→      if (data[j].low <= data[i].low) isLow = false;
   402→    }
   403→    if (isHigh) highs.push(data[i].high);
   404→    if (isLow) lows.push(data[i].low);
   405→  }
   406→  return { highs, lows };
   407→}

   409→// ─── Support / Resistance (6 each, adaptive gap, multi-source, 10% window strength) ─
   410→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   411→  resistances: number[];
   412→  supports: number[];
   413→  supportStrengths: LevelStrength[];
   414→  resistanceStrengths: LevelStrength[];
   415→  priceTargets: LevelStrength[];
   416→} {
   417→  const allRawLevels: number[] = [];
   418→  const closes = data.map(d => d.close);
   419→
   420→  // 1. Pivot Points from previous day (5R + 5S)
   421→  const prev = data[data.length - 2] ?? data[data.length - 1];
   422→  if (prev) {
   423→    const pp = (prev.high + prev.low + prev.close) / 3;
   424→    const hl = prev.high - prev.low;
   425→    const pivotR = [
   426→      2 * pp - prev.low,
   427→      pp + hl,
   428→      prev.high + 2 * (pp - prev.low),
   429→      prev.high + 2 * (pp - prev.low) + hl,
   430→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   431→    ];
   432→    const pivotS = [
   433→      2 * pp - prev.high,
   434→      pp - hl,
   435→      prev.low - 2 * (prev.high - pp),
   436→      prev.low - 2 * (prev.high - pp) - hl,
   437→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   438→    ];
   439→    allRawLevels.push(...pivotR, ...pivotS);
   440→  }
   441→
   442→  // 2. Swing levels (multiple lookback periods for better coverage)
   443→  for (const lb of [3, 5, 7, 10]) {
   444→    const swings = findSwingLevels(data, lb);
   445→    allRawLevels.push(...swings.highs, ...swings.lows);
   446→  }
   447→
   448→  // 3. Moving Average levels (potential dynamic S/R)
   449→  for (const period of [5, 10, 21, 50, 100, 200]) {
   450→    const v = sma(closes, period);
   451→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01) allRawLevels.push(v);
   452→  }
   453→  const ema12 = emaCalc(closes, 12);
   454→  const ema26 = emaCalc(closes, 26);
   455→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01) allRawLevels.push(ema12);
   456→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01) allRawLevels.push(ema26);
   457→
   458→  // 4. Bollinger Band levels
   459→  const bbLevels = calcBollingerBands(closes);
   460→  if (bbLevels.upper > 0) allRawLevels.push(bbLevels.upper);
   461→  if (bbLevels.lower > 0) allRawLevels.push(bbLevels.lower);
   462→
   463→  // 5. Round number / psychological levels
   464→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
   465→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
   466→    allRawLevels.push(Math.round(currentPrice * mult / magnitude) * magnitude);
   467→  }
   468→
   469→  // 6. Recent high/low zones (5-day, 22-day, 66-day)
   470→  for (const window of [5, 22, 66]) {
   471→    const slice = data.slice(-Math.min(window, data.length));
   472→    if (slice.length > 0) {
   473→      allRawLevels.push(Math.max(...slice.map(d => d.high)));
   474→      allRawLevels.push(Math.min(...slice.map(d => d.low)));
   475→    }
   476→  }
   477→
   478→  // 7. Round all and filter invalid
   479→  const rounded = allRawLevels.map(roundToNice).filter(l => l > 0);
   480→
   481→  // 8. Remove duplicates (within 0.5%)
   482→  const unique: number[] = [];
   483→  for (const level of rounded) {
   484→    const isDup = unique.some(u => Math.abs(u - level) / Math.max(u, 1) < 0.005);
   485→    if (!isDup) unique.push(level);
   486→  }
   487→
   488→  // 9. Calculate strength for each unique level — 10% window (5% above + 5% below)
   489→  const levelStrengths: LevelStrength[] = unique.map(level => {
   490→    const range10pct = level * 0.10;
   491→    const nearbyCount = unique.filter(other =>
   492→      other !== level && Math.abs(other - level) <= range10pct
   493→    ).length;
   494→    // Normalize: 0 nearby = 1, each nearby adds ~1.5, max 10
   495→    const strength = Math.min(10, Math.max(1, Math.round(1 + nearbyCount * 1.5)));
   496→    return { price: level, strength, isTarget: false };
   497→  });
   498→
   499→  // 10. Separate above/below current price
   500→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   501→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   502→
   503→  // 11. Enforce 2%-5% gap, always return exactly targetCount levels
   504→  //    Step 1: Merge levels closer than 2%
   505→  //    Step 2: Walk from nearest level, keep only chain within 2-5% gap
   506→  //    Step 3: If fewer than targetCount, extend chain by computing ~3% steps outward
   507→  //    Step 4: If more than targetCount, trim by keeping highest strength
   508→  function enforceGap(
   509→    levels: LevelStrength[],
   510→    targetCount: number,
   511→    _allUnique: LevelStrength[],
   512→  ): LevelStrength[] {
   513→    if (levels.length === 0) return [];
   514→
   515→    const MIN_GAP = 0.02;
   516→    const MAX_GAP = 0.05;
   517→    const EXTEND_GAP = 0.035; // 3.5% — middle of 2-5% range
   518→    const step = currentPrice >= 100000 ? 100 : currentPrice >= 10000 ? 50 : currentPrice >= 1000 ? 10 : currentPrice >= 100 ? 5 : 1;
   519→    const isDescending = levels.length > 0 && levels[0].price > (levels[levels.length - 1]?.price ?? 0);
   520→
   521→    // Step 1: Merge levels closer than MIN_GAP
   522→    const merged: LevelStrength[] = [levels[0]];
   523→    for (let i = 1; i < levels.length; i++) {
   524→      const prevPrice = merged[merged.length - 1].price;
   525→      const currPrice = levels[i].price;
   526→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   527→      if (gap < MIN_GAP) {
   528→        const mergedStrength = Math.min(10, Math.max(merged[merged.length - 1].strength, levels[i].strength) + 1);
   529→        merged[merged.length - 1] = {
   530→          ...(levels[i].strength > merged[merged.length - 1].strength ? levels[i] : merged[merged.length - 1]),
   531→          strength: mergedStrength,
   532→        };
   533→      } else {
   534→        merged.push(levels[i]);
   535→      }
   536→    }
   537→
   538→    // Step 2: Walk from first level (nearest to price), keep only if gap is <= MAX_GAP
   539→    const chain: LevelStrength[] = [merged[0]];
   540→    for (let i = 1; i < merged.length; i++) {
   541→      const prevPrice = chain[chain.length - 1].price;
   542→      const currPrice = merged[i].price;
   543→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   544→      if (gap <= MAX_GAP) {
   545→        chain.push(merged[i]);
   546→      }
   547→    }
   548→
   549→    // Step 3: If fewer than targetCount, extend the chain outward at ~3.5% intervals
   550→    while (chain.length < targetCount) {
   551→      const lastPrice = chain[chain.length - 1].price;
   552→      const nextRaw = isDescending
   553→        ? lastPrice * (1 - EXTEND_GAP)  // supports go down
   554→        : lastPrice * (1 + EXTEND_GAP); // resistances go up
   555→      const nextPrice = Math.round(nextRaw / step) * step;
   556→      if (nextPrice <= 0) break;
   557→      if (nextPrice === lastPrice) break; // rounding made no change (very low price)
   558→      chain.push({ price: nextPrice, strength: 2, isTarget: false });
   559→    }
   560→
   561→    // Step 4: If more than targetCount, trim by keeping highest strength
   562→    if (chain.length > targetCount) {
   563→      chain.sort((a, b) => b.strength - a.strength);
   564→      const kept = chain.slice(0, targetCount);
   565→      kept.sort((a, b) => isDescending ? b.price - a.price : a.price - b.price);
   566→      return kept;
   567→    }
   568→
   569→    return chain;
   570→  }
   571→
   572→  const finalResistances = enforceGap(above, 6, levelStrengths);
   573→  const finalSupports = enforceGap(below, 6, levelStrengths);
   574→
   575→  // 12. Determine price targets: strength > 7 is a target
   576→  const allLevels = [...finalResistances, ...finalSupports];
   577→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   578→  let targets: LevelStrength[];
   579→  if (sortedByStrength.some(l => l.strength > 7)) {
   580→    targets = sortedByStrength.filter(l => l.strength > 7);
   581→  } else {
   582→    targets = sortedByStrength.slice(0, 2);
   583→  }
   584→
   585→  // Mark targets
   586→  targets = targets.map(t => ({ ...t, isTarget: true }));
   587→
   588→  // Also mark in final arrays
   589→  finalResistances.forEach(r => {
   590→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   591→  });
   592→  finalSupports.forEach(s => {
   593→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   594→  });
   595→
   596→  return {
   597→    resistances: finalResistances.map(l => l.price),
   598→    supports: finalSupports.map(l => l.price),
   599→    supportStrengths: finalSupports,
   600→    resistanceStrengths: finalResistances,
   601→    priceTargets: targets,
   602→  };
   603→}

   605→// ─── Trend Analysis ───────────────────────────────────────────────────────────
   606→function calcTrend(closes: number[], period: number): TrendResult {
   607→  const slice = closes.slice(-Math.min(period, closes.length));
   608→  const { slope, r2 } = linearRegression(slice);
   609→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   610→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   611→  const priceThreshold = avgPrice * 0.001; // 0.1%
   612→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   613→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   614→}

   631→        short: { direction: 'flat', slope: 0, angle: 0, r2: 0 },

   633→        long: { direction: 'flat', slope: 0, angle: 0, r2: 0 },
   634→      },
   635→      scenarios: { R1: emptyScenario(), R2: emptyScenario(), R3: emptyScenario(), R4: emptyScenario(), R5: emptyScenario() },
   636→      bullScore: 0.5, bearScore: 0.5, overallSignal: 'neutral',
   637→      edgeWeights: { up: 0.5, down: 0.5, pullback: 0.25, risk: 0.12 },
   638→    };
   639→  }
   640→
   641→  const closes = data.map(d => d.close);
   642→  const price = closes[closes.length - 1];

   644→  // ── Volume detection ───────────────────────────────────────────────────

   649→  const smaResult: Record<string, number> = {};
   650→  for (const p of smaPeriods) smaResult[`sma${p}`] = sma(closes, p);
   651→
   652→  const emaResult: Record<string, number> = {
   653→    ema12: emaCalc(closes, 12),
   654→    ema26: emaCalc(closes, 26),
   655→  };
   656→
   657→  // ── Oscillators ──────────────────────────────────────────────────────────
   658→  const rsi = calcRSI(closes);
   659→  const mfi = hasVolume ? calcMFI(data) : 50; // neutral when no volume
   660→  const cci = calcCCI(data);
   661→  const stoch = calcStochastic(data);
   662→  const williamsR = calcWilliamsR(data);
   663→  const macd = calcMACD(closes);
   664→
   665→  // ── Trend ────────────────────────────────────────────────────────────────
   666→  const adxResult = calcADX(data);
   667→  const sar = calcSAR(data);
   668→
   669→  // ── Volatility ───────────────────────────────────────────────────────────
   670→  const atr = calcATR(data);
   671→  const bb = calcBollingerBands(closes);
   672→
   673→  // ── Volume (only when data has volume) ─────────────────────────────────
   674→  const obv = hasVolume ? calcOBV(data) : 0;
   675→
   676→  // ── Support / Resistance ────────────────────────────────────────────────
   677→  const { resistances, supports, supportStrengths, resistanceStrengths, priceTargets } = calcSupportResistance(data, price);
   678→
   679→  // ── Trend Lines ─────────────────────────────────────────────────────────
   680→  const trend = {
   681→    short: calcTrend(closes, 21),
   682→    medium: calcTrend(closes, 50),
   683→    long: calcTrend(closes, Math.min(100, closes.length)),
   684→  };
   685→
   686→  // ── S/R Level References ──────────────────────────────────────────────
   687→  const R1_level = resistances[0] ?? price * 1.05;
   688→  const R2_level = resistances[1] ?? price * 1.10;
   689→  const R3_level = resistances[2] ?? price * 1.15;
   690→  const R4_level = resistances[3] ?? price * 1.20;
   691→  const R5_level = resistances[4] ?? price * 1.30;
   692→  const R6_level = resistances[5] ?? price * 1.40;
   693→  const S1_level = supports[0] ?? price * 0.95;
   694→  const S2_level = supports[1] ?? price * 0.90;
   695→  const S3_level = supports[2] ?? price * 0.85;
   696→  const S4_level = supports[3] ?? price * 0.80;
   697→  const S5_level = supports[4] ?? price * 0.75;
   698→  const S6_level = supports[5] ?? price * 0.70;
   699→
   700→  // ═══════════════════════════════════════════════════════════════════════════
   701→  // ENHANCED PROBABILITY ENGINE — 6 Layers
   702→  // L0: Indicator trends | L1: Weighted scores | L2: Trend-adjusted scores
   703→  // L3: Bull consensus | L4: 5-scenario distribution | L5: Graph edge weights
   704→  // ═══════════════════════════════════════════════════════════════════════════
   705→
   706→  const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));
   707→
   708→  // ── LAYER 0: Indicator Trends (current vs 5 bars ago) ──────────────────
   709→  const TB = 5;
   710→  const pastData = data.length > TB + 15 ? data.slice(0, -TB) : data;
   711→  const pastCloses = pastData.map(d => d.close);
   712→  const rsiPast = calcRSI(pastCloses);
   713→  const mfiPast = hasVolume ? calcMFI(pastData) : 50;
   714→  const cciPast = calcCCI(pastData);
   715→  const macdPast = calcMACD(pastCloses);
   716→  const stochPast = calcStochastic(pastData);
   717→  const bbPast = calcBollingerBands(pastCloses);
   718→  const ema12Past = emaCalc(pastCloses, 12);
   719→  const ema26Past = emaCalc(pastCloses, 26);
   720→
   721→  const IT = {
   722→    rsi: clamp((rsi - rsiPast) / 100, -1, 1),
   723→    mfi: clamp((mfi - mfiPast) / 100, -1, 1),
   724→    cci: clamp((cci - cciPast) / 300, -1, 1),
   725→    macd: clamp((macd.histogram - macdPast.histogram) / (Math.abs(macd.histogram) + Math.abs(macdPast.histogram) + 1) * 3, -1, 1),
   726→    stoch: clamp((stoch.k - stochPast.k) / 50, -1, 1),
   727→    bb: (() => {
   728→      const pN = bb.upper > bb.lower ? (price - bb.lower) / (bb.upper - bb.lower) : 0.5;
   729→      const pP = bbPast.upper > bbPast.lower ? (pastCloses[pastCloses.length - 1] - bbPast.lower) / (bbPast.upper - bbPast.lower) : 0.5;
   730→      return clamp((pN - pP) * 5, -1, 1);
   731→    })(),
   732→    ema: (() => {
   733→      const gN = (emaResult.ema12 ?? 0) - (emaResult.ema26 ?? 0);
   734→      const gP = ema12Past - ema26Past;
   735→      return clamp((gN - gP) / (Math.abs(gN) + Math.abs(gP) + 1) * 3, -1, 1);
   736→    })(),
   737→  };
   738→
   739→  // ── LAYER 1: Weighted Indicator Scores (0-1) ──────────────────────────
   740→  // Each indicator gets a nuanced step-function score (0=completely bearish, 1=completely bullish)
   741→
   742→  // RSI (weight: 15) — zones: >80→0.95, >70→0.85, >60→0.7, >50→0.55, ... <20→0.05
   743→  const s_rsi = rsi > 80 ? 0.95 : rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   744→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   745→  // MFI (weight: 12) — volume dependent
   746→  const s_mfi = mfi > 80 ? 0.95 : mfi > 70 ? 0.85 : mfi > 60 ? 0.7 : mfi > 50 ? 0.55
   747→    : mfi > 40 ? 0.45 : mfi > 30 ? 0.3 : mfi > 20 ? 0.15 : 0.05;
   748→  // CCI (weight: 10) — >200→0.9, >100→0.75, >0→0.6, >-100→0.4, >-200→0.25, else→0.1
   749→  const s_cci = cci > 200 ? 0.9 : cci > 100 ? 0.75 : cci > 0 ? 0.6
   750→    : cci > -100 ? 0.4 : cci > -200 ? 0.25 : 0.1;
   751→  // ADX direction (weight: 8) — DI+ vs DI- with relative strength
   752→  const s_adx = adxResult.diPlus > adxResult.diMinus
   753→    ? clamp(0.5 + (adxResult.diPlus - adxResult.diMinus) / 100, 0.5, 1)
   754→    : clamp(0.5 - (adxResult.diMinus - adxResult.diPlus) / 100, 0, 0.5);
   755→  // MACD (weight: 14) — 3-factor: histogram sign, line vs signal
   756→  const s_macd = macd.histogram > 0 && macd.line > macd.signal ? 0.9
   757→    : macd.histogram > 0 ? 0.7 : macd.line > macd.signal ? 0.55
   758→    : macd.histogram < 0 && macd.line < macd.signal ? 0.1 : 0.3;
   759→  // Stochastic (weight: 10) — %K zone + K/D momentum
   760→  const s_stoch = clamp(
   761→    (stoch.k > 80 ? 0.9 : stoch.k > 70 ? 0.8 : stoch.k > 50 ? 0.6
   762→      : stoch.k > 30 ? 0.4 : stoch.k > 20 ? 0.2 : 0.1)
   763→    + (stoch.k > stoch.d ? 0.1 : -0.1), 0, 1);
   764→  // Bollinger position (weight: 8) — price position within bands
   765→  const bbRange = bb.upper - bb.lower;
   766→  const s_bb = bbRange === 0 ? 0.5 : clamp((price - bb.lower) / bbRange, 0, 1);
   767→  // MA21 (weight: 10) — above/below + distance bonus
   768→  const ma21 = smaResult.sma21 ?? 0;
   769→  const s_ma21 = clamp(price > ma21 ? 0.7 + ((price - ma21) / ma21) * 0.5 : 0.3 + ((price - ma21) / ma21) * 0.5, 0, 1);
   770→  // MA100 (weight: 8) — above/below + smaller distance bonus
   771→  const ma100v = smaResult.sma100 ?? 0;
   772→  const s_ma100 = clamp(price > ma100v ? 0.65 + ((price - ma100v) / ma100v) * 0.3 : 0.35 + ((price - ma100v) / ma100v) * 0.3, 0, 1);
   773→  // EMA12/26 crossover (weight: 10) — gap between EMAs normalized
   774→  const e12v = emaResult.ema12 ?? 0;
   775→  const e26v = emaResult.ema26 ?? 0;
   776→  const emaGap = e12v > 0 && e26v > 0 ? (e12v - e26v) / e26v : 0;
   777→  const s_ema = clamp(0.5 + emaGap * 5, 0, 1);
   778→  // ATR (weight: 5) — low volatility = stability = slightly bullish
   779→  const s_atr = clamp(1 - (price > 0 ? atr / price : 0) * 10, 0.2, 1);
   780→  // Medium-term trend (weight: 10) — direction × R² × angle strength
   781→  const tStr = trend.medium.r2 * Math.abs(trend.medium.angle) / 45;
   782→  const s_trend = trend.medium.direction === 'up' ? clamp(0.5 + tStr / 2, 0.5, 1)
   783→    : trend.medium.direction === 'down' ? clamp(0.5 - tStr / 2, 0, 0.5) : 0.5;
   784→  // S/R proximity (weight: 5) — near resistance=penalty, near support=bonus
   785→  const dR1 = R1_level > 0 ? Math.abs(price - R1_level) / R1_level : 1;
   786→  const dS1 = S1_level > 0 ? Math.abs(price - S1_level) / S1_level : 1;
   787→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
   788→
   789→  // ── LAYER 2: Trend-Adjusted Scores ────────────────────────────────────
   790→  // Each score adjusted by its indicator trend (L0) with calibrated multiplier
   791→  const f_rsi = clamp(s_rsi + IT.rsi * 0.08, 0, 1);
   792→  const f_mfi = clamp(s_mfi + IT.mfi * 0.08, 0, 1);
   793→  const f_cci = clamp(s_cci + IT.cci * 0.06, 0, 1);
   794→  const f_macd = clamp(s_macd + IT.macd * 0.08, 0, 1);
   795→  const f_stoch = clamp(s_stoch + IT.stoch * 0.06, 0, 1);
   796→  const f_bb = clamp(s_bb + IT.bb * 0.04, 0, 1);
   797→  const f_ema = clamp(s_ema + IT.ema * 0.08, 0, 1);
   798→
   799→  // Cross signals (bonus factors)
   800→  // Stochastic crossover: was K<D, now K>D → bullish (+1), inverse → bearish (-1)
   801→  const stochCross = stochPast.k < stochPast.d && stoch.k > stoch.d ? 1
   802→    : stochPast.k > stochPast.d && stoch.k < stoch.d ? -1 : 0;
   803→  const f_stochCross = 0.5 + stochCross * 0.4;
   804→  // MACD crossover: was line<signal, now line>signal → bullish
   805→  const macdCross = macdPast.line < macdPast.signal && macd.line > macd.signal ? 1
   806→    : macdPast.line > macdPast.signal && macd.line < macd.signal ? -1 : 0;
   807→  const f_macdCross = 0.5 + macdCross * 0.4;
   808→  // Price/RSI divergence (20-bar lookback)
   809→  let divergence = 0;
   810→  if (closes.length > 40) {
   811→    const p20 = closes[closes.length - 21];
   812→    const rsi20 = calcRSI(closes.slice(0, -20));
   813→    const pChange = price - p20;
   814→    const rsiChange = rsi - rsi20;
   815→    if (pChange !== 0 && rsiChange !== 0) {
   816→      divergence = (pChange > 0 && rsiChange < 0) ? -1  // bearish divergence
   817→        : (pChange < 0 && rsiChange > 0) ? 1   // bullish divergence
   818→        : 0;
   819→    }
   820→  }
   821→  const f_div = 0.5 + divergence * 0.35;
   822→
   823→  // ── LAYER 3: Weighted Bull Consensus ───────────────────────────────────
   824→  // Σ(weight × finalScore) / Σ(weight) → 0-1
   825→  const weighted: [number, number][] = [
   826→    [15, f_rsi], [10, f_cci], [8, s_adx], [14, f_macd], [10, f_stoch],
   827→    [8, f_bb], [10, s_ma21], [8, s_ma100], [10, f_ema], [5, s_atr],
   828→    [10, s_trend], [5, s_sr], [6, f_stochCross], [6, f_macdCross], [8, f_div],
   829→  ];
   830→  if (hasVolume) weighted.splice(1, 0, [12, f_mfi]); // insert MFI after RSI (w:12)
   831→  const totalW = weighted.reduce((s, [w]) => s + w, 0);
   832→  const bullConsensus = totalW > 0 ? weighted.reduce((s, [w, v]) => s + w * v, 0) / totalW : 0.5;
   833→  const bullScore = bullConsensus;
   834→  const bearScore = 1 - bullConsensus;
   835→
   836→  // ── LAYER 4: 5-Scenario Probability Distribution ──────────────────────
   837→  const distR1 = R1_level > 0 ? Math.exp(-3 * Math.abs(price - R1_level) / R1_level) : 0.5;
   838→  const distS1 = S1_level > 0 ? Math.exp(-3 * Math.abs(price - S1_level) / S1_level) : 0.5;
   839→  const overboughtRisk = (rsi > 70 || (hasVolume && mfi > 80) || stoch.k > 80) ? 1 : 0;
   840→  const oversoldBounce = (rsi < 30 || (hasVolume && mfi < 20) || stoch.k < 20) ? 1 : 0;
   841→
   842→  const raw_R1 = bullConsensus * distR1 * (1 - overboughtRisk * 0.7) + oversoldBounce * 0.3 * distR1;
   843→  const raw_R2 = (1 - Math.abs(bullConsensus - 0.65)) * 0.8 + (bullConsensus > 0.4 ? 0.2 : 0);
