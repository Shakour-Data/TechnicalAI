   276→  const refreshTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);

   282→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number, yahooSymbol?: string) => {
   283→    // Abort any previous in-flight fetch
   284→    if (fetchControllerRef.current) {
   285→      fetchControllerRef.current.abort();
   286→    }
   287→    const controller = new AbortController();
   288→    fetchControllerRef.current = controller;
   289→
   290→    // Yahoo Finance instrument: fetch historical data via Yahoo Finance API
   291→    // This handles ALL Yahoo instruments (stocks, indices, energy, metals, forex, crypto, ETFs, etc.)
   292→    if (yahooSymbol) {
   293→      setLoading(true);
   294→      setLoadingMessage('در حال دریافت داده‌های تاریخی از یاهو فایننس ...');
   295→      setError(null);
   296→      setData(null);
   297→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId, yahooSymbol };
   298→      try {
   299→        const res = await fetch('/api/yahoo-analysis?symbol=' + encodeURIComponent(yahooSymbol), { signal: controller.signal });
   300→        if (controller.signal.aborted) return;
   301→        if (!res.ok) {
   302→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   303→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   304→          throw new Error(errMsg);
   305→        }
   306→        const json = await res.json();
   307→        if (controller.signal.aborted) return;
   308→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   309→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   310→        }
   311→        setData({ ...json, isYahoo: true });
   312→        setActivePanel('visual');
   313→      } catch (err) {
   314→        if (controller.signal.aborted) return;
   315→        setError(String(err));
   316→      } finally {
   317→        if (!controller.signal.aborted) {
   318→          setLoading(false);
   319→          setLoadingMessage(null);
