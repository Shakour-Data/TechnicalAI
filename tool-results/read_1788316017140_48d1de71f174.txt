   353→/**
   354→ * Parse a number that might contain Persian digits, commas, or other formatting.
   355→ * Handles null/undefined/empty strings gracefully.
   356→ */
   357→function parsePersianNum(str: string | null | undefined | number): number {
   358→  if (str === null || str === undefined) return 0;
   359→  let s = String(str).trim();
   360→  if (!s || s === '-' || s === '') return 0;
   361→  // Strip trailing % sign (e.g. "0.05%" → "0.05")
   362→  if (s.endsWith('%')) s = s.slice(0, -1).trim();
   363→  // Replace commas and Persian digits
   364→  const cleaned = s.replace(/[,۰-۹]/g, (c: string) => {
   365→    const persianDigits = '۰۱۲۳۴۵۶۷۸۹';
   366→    const idx = persianDigits.indexOf(c);
   367→    return idx >= 0 ? String(idx) : c === ',' ? '' : c;
   368→  });
   369→  const n = Number(cleaned);
   370→  return isFinite(n) ? n : 0;
   371→}

   373→/**
   374→ * Check if a value looks like a date string (contains / or - and is not a number).
   375→ */
   376→function looksLikeDate(val: string | null | undefined | number): boolean {
   377→  if (!val) return false;
   378→  const s = String(val).trim();
   379→  return /[\/\-]/.test(s) && isNaN(Number(s.replace(/[,۰-۹\/\-]/g, '')));
   380→}
   381→
   382→/**
   383→ * Check if a value looks like a percentage (ends with % or is small like 0.5 to 50).
   384→ */
   385→function looksLikePercentage(val: string | null | undefined | number): boolean {
   386→  if (!val) return false;
   387→  const s = String(val).trim();
   388→  if (s.endsWith('%')) return true;
   389→  const n = parsePersianNum(val);
   390→  return !isNaN(n) && Math.abs(n) < 100 && n !== 0;
   391→}

   393→/**
   394→ * Auto-detect column mapping for TGJU API response.
   395→ * Expected format: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   396→ * But the API may change column order.
   397→ */
   398→function detectColumnMap(row: (string | number)[]): {
   399→  openIdx: number; highIdx: number; lowIdx: number; closeIdx: number;
   400→  jalaliDateIdx: number; gregorianDateIdx: number;
   401→} | null {
   402→  if (!row || row.length < 6) return null;
   403→
   404→  let jalaliDateIdx = -1;
   405→  let gregorianDateIdx = -1;
   406→  const numericIndices: number[] = [];
   407→  const percentageIndices: number[] = [];
   408→
   409→  for (let i = 0; i < row.length; i++) {
   410→    const val = row[i];
   411→    if (looksLikeDate(val)) {
   412→      // Jalali dates have Persian digits or format like 1404/05/25
   413→      const s = String(val);
   414→      if (/[۰-۹]/.test(s) || /^14\d{2}/.test(s.replace(/[۰-۹]/g, (d: string) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d))))) {
   415→        if (jalaliDateIdx === -1) jalaliDateIdx = i;
   416→      } else {
   417→        if (gregorianDateIdx === -1) gregorianDateIdx = i;
   418→      }
   419→    } else if (looksLikePercentage(val)) {
   420→      percentageIndices.push(i);
   421→    } else {
   422→      const n = parsePersianNum(val);
   423→      if (isFinite(n) && n !== 0) {
   424→        numericIndices.push(i);
   425→      }
   426→    }
   427→  }
   428→
   429→  // If we couldn't auto-detect dates, fall back to assumed positions
   430→  if (jalaliDateIdx === -1 && row.length >= 8) {
   431→    // Default assumption: row[6] = gregorian, row[7] = jalali
   432→    if (looksLikeDate(row[7])) jalaliDateIdx = 7;
   433→    else if (looksLikeDate(row[6])) jalaliDateIdx = 6;
   434→    if (gregorianDateIdx === -1 && looksLikeDate(row[6]) && jalaliDateIdx !== 6) gregorianDateIdx = 6;
   435→    else if (gregorianDateIdx === -1 && looksLikeDate(row[7]) && jalaliDateIdx !== 7) gregorianDateIdx = 7;
   436→  }
   437→
   438→  // Numeric columns (excluding percentages) should be OHLC + change
   439→  // OHLC: typically 4 largest absolute values among non-percentage numerics
   440→  // change: the remaining one
   441→  const ohlcCandidates = numericIndices.filter(i => !percentageIndices.includes(i));
   442→
   443→  if (ohlcCandidates.length < 4) {
   444→    // Fallback: assume first 4 numeric columns are OHLC
   445→    if (numericIndices.length >= 4) {
   446→      return {
   447→        openIdx: numericIndices[0],
   448→        highIdx: numericIndices[2],
   449→        lowIdx: numericIndices[1],
   450→        closeIdx: numericIndices[3],
   451→        jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : 7,
   452→        gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : 6,
   453→      };
   454→    }
   455→    return null;
   456→  }
   457→
   458→  // Sort by absolute value descending to identify OHLC (largest) vs change (smallest)
   459→  const sorted = ohlcCandidates
   460→    .map(i => ({ idx: i, val: Math.abs(parsePersianNum(row[i])) }))
   461→    .sort((a, b) => b.val - a.val);
   462→
   463→  // The 4 largest should be OHLC. Among them, identify high (max) and low (min).
   464→  // open and close are the other two.
   465→  const top4 = sorted.slice(0, 4);
   466→  const maxIdx = top4.reduce((a, b) => a.val > b.val ? a : b).idx;
   467→  const minIdx = top4.reduce((a, b) => a.val < b.val ? a : b).idx;
   468→  const others = top4.filter(x => x.idx !== maxIdx && x.idx !== minIdx);
   469→
   470→  // Among the two remaining, the one closer to high is open, the other is close
   471→  // (heuristic: in an uptrend open < close, but we can't assume direction)
   472→  // Default: first = open, second = close
   473→  const openIdx = others[0]?.idx ?? ohlcCandidates[0];
   474→  const closeIdx = others[1]?.idx ?? ohlcCandidates[3];
   475→
   476→  return {
   477→    openIdx,
   478→    highIdx: maxIdx,
   479→    lowIdx: minIdx,
   480→    closeIdx,
   481→    jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : row.length - 1,
   482→    gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : row.length - 2,
   483→  };
   484→}

   486→/**
   487→ * Validate a candle's OHLC values for sanity.
   488→ * Returns true if the candle is valid.
   489→ */
   490→function isValidCandle(o: number, h: number, l: number, c: number): boolean {
   491→  if (!isFinite(o) || !isFinite(h) || !isFinite(l) || !isFinite(c)) return false;
   492→  if (o <= 0 || h <= 0 || l <= 0 || c <= 0) return false;
   493→  // High must be >= Low, Open, Close
   494→  if (h < l || h < o || h < c) return false;
   495→  // Low must be <= High, Open, Close
   496→  if (l > o || l > c) return false;
   497→  // No infinite or extreme values
   498→  if (o > 1e15 || h > 1e15 || l > 1e15 || c > 1e15) return false;
   499→  return true;
   500→}

   502→/**
   503→ * Check if parsed candle data is reasonably fresh.
   504→ * Rejects data where the most recent candle is older than 2 years.
   505→ * This prevents serving data from e.g. 1392 when current year is 1404.
   506→ */
   507→function isDataFresh(candles: TgjuOHLC[]): boolean {
   508→  if (candles.length === 0) return false;
   509→  const lastDateStr = candles[candles.length - 1].date;
   510→  const yearMatch = lastDateStr.match(/(\d{4})/);
   511→  if (!yearMatch) return true; // Can't determine year, assume fresh
   512→
   513→  const year = parseInt(yearMatch[1], 10);
   514→  const now = new Date();
   515→  const gregorianYear = now.getFullYear();
   516→  // Approximate current Jalali year (Gregorian - 621 or 622)
   517→  const approxJalaliYear = gregorianYear - 621;
   518→
   519→  // Jalali years: 1390-1500 range
   520→  if (year >= 1390 && year <= 1500) {
   521→    return year >= approxJalaliYear - 2; // Allow up to 2 years old
   522→  }
   523→
   524→  // Gregorian years
   525→  if (year >= 2020 && year <= 2035) {
   526→    return year >= gregorianYear - 2;
   527→  }
   528→
   529→  // Unknown format, don't reject
   530→  return true;
   531→}

   580→/* ─── Fetch live instruments (currency + gold from tgju-api) + static list ── */

   582→export async function fetchTgjuInstruments(): Promise<TgjuInstrument[]> {
   583→  const now = Date.now();
   584→  if (liveInstrumentsCache.data && now - liveInstrumentsCache.ts < INSTRUMENTS_TTL) {
   585→    return liveInstrumentsCache.data;
   586→  }
   587→
   588→  const controller = new AbortController();
   589→  const timeout = setTimeout(() => controller.abort(), 10000);
   590→
   591→  try {
   592→    const [currencyRes, goldRes] = await Promise.all([
   593→      fetch(`${TGJU_API_BASE}/api/price/currency`, { signal: controller.signal }),
   594→      fetch(`${TGJU_API_BASE}/api/price/gold`, { signal: controller.signal }),
   595→    ]);
   596→
   597→    clearTimeout(timeout);
   598→
   599→    const instruments: TgjuInstrument[] = [];
   600→
   601→    if (currencyRes.ok) {
   602→      const currencies: TgjuPriceItem[] = await currencyRes.json();
   603→      for (const c of currencies) {
   604→        const price = parsePersianNum(c.price);
   605→        const low = parsePersianNum(c.low_price);
   606→        const high = parsePersianNum(c.high_price);
   607→        instruments.push({
   608→          title: c.title,
   609→          key: c.key,
   610→          price,
   611→          highPrice: high,
   612→          lowPrice: low,
   613→          change: 0,
   614→          changePercent: 0,
   615→          category: 'currency',
   616→        });
   617→      }
   618→    }
   619→
   620→    if (goldRes.ok) {
   621→      const goldCategories: TgjuGoldCategory[] = await goldRes.json();
   622→      for (const cat of goldCategories) {
   623→        for (const item of cat.prices) {
   624→          const price = parsePersianNum(item.price);
   625→          const low = parsePersianNum(item.low_price);
   626→          const high = parsePersianNum(item.high_price);
   627→
   628→          let category: TgjuCategoryType = 'gold';
   629→          if (item.key.startsWith('ime_fund_')) category = 'gold_etf';
   630→          else if (item.key.startsWith('silver_')) category = 'silver';
   631→
   632→          instruments.push({
   633→            title: item.title,
   634→            key: item.key,
   635→            price,
   636→            highPrice: high,
   637→            lowPrice: low,
   638→            change: 0,
   639→            changePercent: 0,
   640→            category,
   641→            groupTitle: cat.title,
   642→          });
   643→        }
   644→      }
   645→    }
   646→
   647→    // Add static instruments (crypto, world indices, forex, energy, metals, commodities)
   648→    for (const s of STATIC_INSTRUMENTS) {
   649→      const cached = staticPriceCache.get(s.key);
   650→      const price = cached?.close ?? 0;
   651→      instruments.push({
   652→        title: s.title,
   653→        key: s.key,
   654→        price,
   655→        highPrice: cached?.high ?? 0,
   656→        lowPrice: cached?.low ?? 0,
   657→        change: cached ? (cached.close - cached.prevClose) : 0,
   658→        changePercent: cached && cached.prevClose > 0 ? ((cached.close - cached.prevClose) / cached.prevClose) * 100 : 0,
   659→        category: s.category,
   660→        groupTitle: s.groupTitle,
   661→      });
   662→    }
   663→
   664→    liveInstrumentsCache.data = instruments;
   665→    liveInstrumentsCache.ts = now;
   666→
   667→    // ── Background: fetch chart data for currency/gold to get accurate daily change ──
   668→    backgroundFetchCurrencyGoldChange(instruments).catch(() => {});
   669→
   670→    // ── Background: fetch Yahoo prices for static instruments with price=0 (TGJU API 403) ──
   671→    backgroundFetchYahooFallbackPrices(instruments).catch(() => {});
   672→
   673→    return instruments;
   674→  } catch {
   675→    clearTimeout(timeout);
   676→    return liveInstrumentsCache.data || [];
   677→  }
   678→}

   680→/**
   681→ * Background task: fetch last 2 candles for currency/gold items to compute accurate daily change.
   682→ * Uses direct fetch to avoid queue contention.
   683→ * Only updates change/changePercent — keeps the live price from tgju-api.
   684→ */
   685→async function backgroundFetchCurrencyGoldChange(instruments: TgjuInstrument[]) {
   686→  const currencyGoldItems = instruments.filter(
   687→    (i) => i.category === 'currency' || i.category === 'gold' || i.category === 'silver' || i.category === 'gold_etf'
   688→  );
   689→
   690→  // Limit to first 10 to avoid rate-limiting (most important: USD, EUR, gold coin, etc.)
   691→  const priorityKeys = currencyGoldItems.slice(0, 10).map((i) => i.key);
   692→
   693→  for (const key of priorityKeys) {
   694→    try {
   695→      const url = `${TGJU_CHART_API}/${key}?lang=fa&order_dir=desc&start=0&length=2`;
   696→      const res = await fetch(url, { signal: AbortSignal.timeout(10_000) });
   697→      if (!res.ok) continue;
   698→      const jsonStr = await res.text();
   699→      if (!jsonStr.startsWith('{')) continue;
   700→
   701→      const data = JSON.parse(jsonStr);
   702→      const rows: string[][] = data.data;
   703→      if (!rows || rows.length < 2) continue;
   704→
   705→      const newest = rows[0];
   706→      const prev = rows[1];
   707→      const latestClose = parsePersianNum(newest[3]);
   708→      const prevClose = parsePersianNum(prev[3]);
   709→
   710→      if (latestClose > 0 && prevClose > 0) {
   711→        const inst = liveInstrumentsCache.data?.find((i) => i.key === key);
   712→        if (inst) {
   713→          inst.change = latestClose - prevClose;
   714→          inst.changePercent = ((latestClose - prevClose) / prevClose) * 100;
   715→        }
   716→      }
   717→    } catch {
   718→      // skip individual errors
   719→    }
   720→  }
   721→}

   732→ * Background task: fetch Yahoo Finance prices for static instruments that have price=0
   733→ * (happens when api.tgju.org returns 403). Updates the live cache in-place.
   734→ */
   735→async function backgroundFetchYahooFallbackPrices(instruments: TgjuInstrument[]) {
   736→  // Collect static instruments with price=0 that have a Yahoo mapping
   737→  const zeroPriceItems = instruments.filter(
   738→    (i) => i.price === 0 && i.key in TGJU_TO_YAHOO_MAP
   739→  );
   740→
   741→  if (zeroPriceItems.length === 0) return;
   742→
   743→  console.log(`[TGJU] ${zeroPriceItems.length} static instruments have price=0, fetching from Yahoo Finance as fallback...`);
   744→
   745→  // Collect unique Yahoo symbols
   746→  const keyToSymbol = new Map<string, string>();
   747→  const symbolToKeys = new Map<string, string[]>();
   748→  for (const item of zeroPriceItems) {
   749→    const yahooSym = TGJU_TO_YAHOO_MAP[item.key];
   750→    if (!yahooSym) continue;
   751→    keyToSymbol.set(item.key, yahooSym);
   752→    if (!symbolToKeys.has(yahooSym)) symbolToKeys.set(yahooSym, []);
   753→    symbolToKeys.get(yahooSym)!.push(item.key);
   754→  }
   755→
   756→  const uniqueSymbols = [...symbolToKeys.keys()];
   757→

   852→let pageReaderQueue: Promise<void> = Promise.resolve();
   853→let lastPageReaderCall = 0;
   854→const PAGE_READER_MIN_INTERVAL = 10_000; // 10 seconds between calls
   855→const PAGE_READER_MAX_RETRIES = 3;
   856→const PAGE_READER_RETRY_BASE_DELAY = 3_000; // 3 seconds base retry delay

   858→/**
   859→ * Fetch historical OHLC data for a TGJU instrument.
   860→ *
   861→ * Strategy (permanent fix):
   862→ * 1. Check in-memory cache (4h TTL)
   863→ * 2. For global instruments with Yahoo mapping → use Yahoo Finance as PRIMARY
   864→ * 3. For Iranian instruments → try direct fetch, then queued page_reader with retry
   865→ * 4. If all fail → return stale cache (up to 24h) or file-based cache
   866→ */
   867→export async function fetchTgjuHistory(tgjuKey: string): Promise<TgjuOHLC[]> {
   868→  const now = Date.now();
   869→
   870→  // ── Step 0: Check in-memory cache (with freshness validation) ──
   871→  const cached = historyCache.get(tgjuKey);
   872→  if (cached && now - cached.ts < HISTORY_TTL) {
   873→    // Validate cached data is not stale (e.g. 1392-era data from a previous bug)
   874→    if (isDataFresh(cached.data)) {
   875→      return cached.data;
   876→    } else {
   877→      console.warn(`[TGJU] In-memory cache for ${tgjuKey} is STALE (latest: ${cached.data[cached.data.length - 1]?.date}), invalidating`);
   878→      historyCache.delete(tgjuKey);
   879→    }
   880→  }
   881→
   882→  // ── Step 1: For global instruments, use Yahoo Finance as PRIMARY source ──
   883→  const yahooSymbol = TGJU_TO_YAHOO_MAP[tgjuKey];
   884→  if (yahooSymbol) {
   885→    try {
   886→      const YahooFinance = (await import('yahoo-finance2')).default;
   887→      const yf = new YahooFinance({ suppressNotices: ['yahooSurvey'], validation: { logErrors: false } });
   888→      const result = await yf.chart(yahooSymbol, {
   889→        period1: new Date(Date.now() - 400 * 24 * 60 * 60 * 1000),
   890→        period2: new Date(),
   891→        interval: '1d',
   892→      });
   893→      if (result.quotes && result.quotes.length >= 30) {
   894→        const candles: TgjuOHLC[] = result.quotes
   895→          .filter((q: any) => q.close != null && q.open != null)
   896→          .map((q: any) => ({
   897→            date: q.date?.toISOString().split('T')[0] || '',
   898→            open: q.open!,
   899→            high: q.high || q.close!,
   900→            low: q.low || q.close!,
   901→            close: q.close!,
   902→          }));
   903→        if (candles.length >= 30) {
   904→          console.log(`[TGJU] ${tgjuKey}: Got ${candles.length} candles from Yahoo (${yahooSymbol})`);
   905→          yahooFetchedKeys.add(tgjuKey);
   906→          cacheHistory(tgjuKey, candles, now);
   907→          return candles;
   908→        }
   909→      }
   910→    } catch (err) {
   911→      console.warn(`[TGJU] Yahoo historical fetch failed for ${tgjuKey} (${yahooSymbol}):`, err instanceof Error ? err.message : err);
   912→    }
   913→  }
   914→
   915→  // ── Step 2: For Iranian instruments (or Yahoo fallback failed), try TGJU sources ──
   916→  // 2a: Try direct fetch (fast path, works when Cloudflare not blocking)
   917→  const directResult = await tryDirectFetch(tgjuKey);
   918→  if (directResult && directResult.length >= 30) {
   919→    cacheHistory(tgjuKey, directResult, now);
   920→    return directResult;
   921→  }
   922→
   923→  // 2b: Queued page_reader with retry + rate limiting
   924→  console.log(`[TGJU] Direct fetch returned ${directResult?.length ?? 0} candles for ${tgjuKey}, using queued page_reader...`);
   925→  const pageReaderResult = await fetchViaQueuedPageReader(tgjuKey);
   926→  if (pageReaderResult && pageReaderResult.length >= 30) {
   927→    cacheHistory(tgjuKey, pageReaderResult, now);
   928→    return pageReaderResult;
   929→  }
   930→
   931→  // ── Step 3: All methods failed — return stale cache (only if fresh) ──
   932→  if (cached && cached.data.length >= 30 && isDataFresh(cached.data)) {
   933→    const age = ((now - cached.ts) / 60_000).toFixed(0);
   934→    console.warn(`[TGJU] All methods failed for ${tgjuKey}, returning stale cache (${age}min old, ${cached.data.length} candles)`);
   935→    return cached.data;
   936→  }
   937→
   938→  // ── Step 4: Try file-based cache ──
   939→  const fileCached = await readFromFileCache(tgjuKey);
   940→  if (fileCached && fileCached.length >= 30) {
   941→    console.warn(`[TGJU] Returning file-cached data for ${tgjuKey} (${fileCached.length} candles)`);
   942→    historyCache.set(tgjuKey, { data: fileCached, ts: Date.now() - HISTORY_TTL + 60_000 }); // Mark as almost-expired
   943→    return fileCached;
   944→  }
   945→
   946→  console.error(`[TGJU] All methods and caches failed for ${tgjuKey}`);
   947→  return [];
   948→}

   950→/**
   951→ * Try direct HTTP fetch to TGJU chart API.
   952→ * Returns parsed candles or null on failure.
   953→ */
   954→async function tryDirectFetch(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   955→  try {
   956→    // CRITICAL: order_dir=desc to get the MOST RECENT candles, not the oldest.
   957→    // With asc+start=0, we'd get data from 1392 for long-history instruments!
   958→    const url = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=desc&start=0&length=365`;
   959→    const controller = new AbortController();
   960→    const timer = setTimeout(() => controller.abort(), 12_000);
   961→
   962→    const res = await fetch(url, { signal: controller.signal, headers: { 'Accept': 'application/json' } });
   963→    clearTimeout(timer);
   964→
   965→    if (!res.ok) {
   966→      console.warn(`[TGJU] Direct fetch HTTP ${res.status} for ${tgjuKey}`);
   967→      return null;
   968→    }
   969→
   970→    const jsonStr = await res.text();
   971→    if (!jsonStr.startsWith('{')) {
   972→      console.warn(`[TGJU] Direct fetch non-JSON for ${tgjuKey}`);
   973→      return null;
   974→    }
   975→
   976→    return parseTgjuChartData(tgjuKey, jsonStr);
   977→  } catch (err) {
   978→    console.warn(`[TGJU] Direct fetch error for ${tgjuKey}:`, err instanceof Error ? err.message : err);
   979→    return null;
   980→  }
   981→}

   983→/**
   984→ * Fetch TGJU historical data via queued page_reader with retry and rate limiting.
   985→ * This prevents 429 errors by serializing requests and adding delays between calls.
   986→ */
   987→async function fetchViaQueuedPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   988→  // Enqueue this request — ensures only one page_reader call at a time
   989→  let result: TgjuOHLC[] | null = null;
   990→  const taskPromise = (async () => {
   991→    // Rate limit: wait if last call was too recent
   992→    const timeSinceLastCall = Date.now() - lastPageReaderCall;
   993→    if (timeSinceLastCall < PAGE_READER_MIN_INTERVAL) {
   994→      const waitTime = PAGE_READER_MIN_INTERVAL - timeSinceLastCall;
   995→      console.log(`[TGJU] Rate limiting page_reader: waiting ${waitTime}ms...`);
   996→      await new Promise(r => setTimeout(r, waitTime));
   997→    }
   998→
   999→    // Retry with exponential backoff
  1000→    for (let attempt = 1; attempt <= PAGE_READER_MAX_RETRIES; attempt++) {
  1001→      try {
  1002→        lastPageReaderCall = Date.now();
  1003→        const fetchResult = await fetchTgjuHistoryViaPageReader(tgjuKey);
  1004→        if (fetchResult && fetchResult.length >= 30) {
  1005→          result = fetchResult;
  1006→          return; // success
  1007→        }
  1008→        // Got data but not enough candles — don't retry
  1009→        if (fetchResult && fetchResult.length > 0) {
  1010→          console.warn(`[TGJU] page_reader returned only ${fetchResult.length} candles for ${tgjuKey}`);
  1011→          return;
  1012→        }
  1013→      } catch (err) {
  1014→        const errMsg = err instanceof Error ? err.message : String(err);
  1015→        const isRateLimit = errMsg.includes('429') || errMsg.includes('Too many requests');
  1016→        if (isRateLimit && attempt < PAGE_READER_MAX_RETRIES) {
  1017→          const delay = PAGE_READER_RETRY_BASE_DELAY * Math.pow(3, attempt - 1); // 3s, 9s, 27s
  1018→          console.warn(`[TGJU] page_reader 429 for ${tgjuKey}, retry ${attempt}/${PAGE_READER_MAX_RETRIES} in ${delay}ms...`);
  1019→          await new Promise(r => setTimeout(r, delay));
  1020→          continue;
  1021→        }
  1022→        console.error(`[TGJU] page_reader error for ${tgjuKey} (attempt ${attempt}):`, errMsg);
  1023→        return;
  1024→      }
  1025→    }
  1026→  })();
  1027→
  1028→  // Chain onto the queue
  1029→  pageReaderQueue = pageReaderQueue.then(() => taskPromise, () => taskPromise);
  1030→  await pageReaderQueue;
  1031→  return result;
  1032→}

  1034→/**
  1035→ * Single page_reader call to fetch TGJU historical data.
  1036→ * Uses z-ai SDK to bypass Cloudflare challenges.
  1037→ * Throws on failure so the caller (fetchViaQueuedPageReader) can retry.
  1038→ */
  1039→async function fetchTgjuHistoryViaPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
  1040→  // CRITICAL: order_dir=desc to get the MOST RECENT candles, not the oldest.
  1041→  const apiUrl = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=desc&start=0&length=365`;
  1042→
  1043→  const ZAI = (await import('z-ai-web-dev-sdk')).default;
  1044→  const zai = await ZAI.create();
  1045→  console.log(`[TGJU] Using z-ai SDK page_reader for ${tgjuKey}...`);
  1046→  const invokeResult = await zai.functions.invoke('page_reader', { url: apiUrl });
  1047→
  1048→  if (invokeResult.code !== 200 || !invokeResult.data?.html) {
  1049→    const errMsg = invokeResult.error || `code=${invokeResult.code}`;
  1050→    throw new Error(errMsg);
  1051→  }
  1052→
  1053→  // Extract JSON from <pre> tag in the HTML
  1054→  const html = invokeResult.data.html;
  1055→  const preMatch = html.match(/<pre[^>]*>([\s\S]*?)<\/pre>/);
  1056→  if (!preMatch) {
  1057→    throw new Error('no <pre> block found in page_reader response');
  1058→  }
  1059→
  1060→  // Unescape HTML entities and strip HTML tags from values
  1061→  const jsonStr = preMatch[1]
  1062→    .replace(/<[^>]*>/g, '')        // Strip ALL HTML tags first
  1063→    .replace(/&quot;/g, '"')
  1064→    .replace(/&amp;/g, '&')
  1065→    .replace(/&lt;/g, '<')
  1066→    .replace(/&gt;/g, '>')
  1067→    .replace(/\s+/g, ' ')           // Normalize whitespace
  1068→    .trim();
  1069→
  1070→  if (!jsonStr.startsWith('{')) {
  1071→    throw new Error('extracted content is not JSON');
  1072→  }
  1073→
  1074→  return parseTgjuChartData(tgjuKey, jsonStr);
  1075→}

  1077→/**
  1078→ * Parse the JSON data from TGJU chart API response.
  1079→ * Used by both direct fetch and page_reader paths.
  1080→ */
  1081→function parseTgjuChartData(tgjuKey: string, jsonStr: string): TgjuOHLC[] | null {
  1082→  try {
  1083→    const parsed = JSON.parse(jsonStr);
  1084→    const rawRows: (string | number)[][] = parsed.data;
  1085→
  1086→    if (!rawRows || rawRows.length === 0) {
  1087→      console.warn(`[TGJU] Empty data array for ${tgjuKey}`);
  1088→      return null;
  1089→    }
  1090→
  1091→    // Sanitize row values: strip any residual HTML tags and trim
  1092→    const rows = rawRows.map(row =>
  1093→      row.map(cell => {
  1094→        const s = String(cell).trim();
  1095→        // Strip HTML tags if any remain (safety for direct fetch path)
  1096→        return s.replace(/<[^>]*>/g, '').replace(/\s+/g, ' ');
  1097→      })
  1098→    );
  1099→
  1100→    // ── Auto-detect column mapping from first valid row ──
  1101→    let colMap = detectColumnMap(rows[0]);
  1102→    if (!colMap) {
  1103→      for (let r = 1; r < Math.min(5, rows.length); r++) {
  1104→        colMap = detectColumnMap(rows[r]);
  1105→        if (colMap) break;
  1106→      }
  1107→    }
  1108→    if (!colMap) {
  1109→      console.warn(`[TGJU] Could not auto-detect columns for ${tgjuKey}, using default order`);
  1110→      colMap = { openIdx: 0, highIdx: 2, lowIdx: 1, closeIdx: 3, jalaliDateIdx: 7, gregorianDateIdx: 6 };
  1111→    } else {
  1112→      console.log(`[TGJU] Column map for ${tgjuKey}: O=${colMap.openIdx} H=${colMap.highIdx} L=${colMap.lowIdx} C=${colMap.closeIdx} Date=${colMap.jalaliDateIdx}`);
  1113→    }
  1114→
  1115→    const { openIdx, highIdx, lowIdx, closeIdx, jalaliDateIdx, gregorianDateIdx } = colMap;
  1116→
  1117→    // ── Parse rows ──
  1118→    const candles: TgjuOHLC[] = [];
  1119→    let invalidCount = 0;
  1120→
  1121→    for (const row of rows) {
  1122→      const rawDate = String(row[jalaliDateIdx] || row[gregorianDateIdx] || '');
  1123→      const date = rawDate.replace(/[\/]/g, '-');
  1124→      const o = parsePersianNum(row[openIdx]);
  1125→      const h = parsePersianNum(row[highIdx]);
  1126→      const l = parsePersianNum(row[lowIdx]);
  1127→      const c = parsePersianNum(row[closeIdx]);
  1128→
  1129→      if (!isValidCandle(o, h, l, c) || date.length === 0) {
  1130→        invalidCount++;
  1131→        continue;
  1132→      }
  1133→
  1134→      candles.push({ date, open: o, high: h, low: l, close: c });
  1135→    }
  1136→
  1137→    if (invalidCount > 0) {
  1138→      console.warn(`[TGJU] ${tgjuKey}: ${invalidCount}/${rows.length} invalid candles skipped`);
  1139→    }
  1140→
  1141→    if (candles.length === 0) {
  1142→      return null;
  1143→    }
  1144→
  1145→    // ── Detect and fix data drift ──
  1146→    let driftCount = 0;
  1147→    for (let i = 1; i < candles.length; i++) {
  1148→      const prevClose = candles[i - 1].close;
  1149→      const currClose = candles[i].close;
  1150→      if (prevClose > 0) {
  1151→        const pctChange = Math.abs(currClose - prevClose) / prevClose;
  1152→        if (pctChange > 0.5) {
  1153→          driftCount++;
  1154→        }
  1155→      }
  1156→    }
  1157→    if (driftCount > candles.length * 0.1) {
  1158→      console.warn(`[TGJU] ${tgjuKey}: Possible data drift detected (${driftCount} suspicious jumps out of ${candles.length} candles)`);
  1159→    }
  1160→
  1161→    // Reverse: API returns oldest-first (asc), we want newest-last for charting
  1162→    const firstDate = candles[0].date;
  1163→    const lastDate = candles[candles.length - 1].date;
  1164→    if (firstDate > lastDate) {
  1165→      candles.reverse();
  1166→    }
  1167→
  1168→    // ── Freshness validation: reject data that is too old (e.g. from 1392) ──
  1169→    if (!isDataFresh(candles)) {
  1170→      console.warn(`[TGJU] ${tgjuKey}: Data is STALE (latest candle: ${candles[candles.length - 1]?.date}), REJECTING`);
  1171→      return null;
  1172→    }
  1173→
  1174→    console.log(`[TGJU] ${tgjuKey}: Parsed ${candles.length} candles (${candles[0]?.date} to ${candles[candles.length - 1]?.date})`);
  1175→
  1176→    return candles;
  1177→  } catch (err) {
  1178→    console.error(`[TGJU] Failed to parse chart data for ${tgjuKey}:`, err);
  1179→    return null;
  1180→  }
  1181→}

  1183→/**
  1184→ * Cache parsed history data and update related caches.
  1185→ */
  1186→function cacheHistory(tgjuKey: string, candles: TgjuOHLC[], now: number): void {
  1187→  historyCache.set(tgjuKey, { data: candles, ts: now });
  1188→  // Also persist to file for server restart resilience
  1189→  writeToFileCache(tgjuKey, candles).catch(() => {});
  1190→
  1191→  // Also update static price cache for this instrument
  1192→  if (candles.length >= 2) {
  1193→    const last = candles[candles.length - 1];
  1194→    const prev = candles[candles.length - 2];
  1195→    staticPriceCache.set(tgjuKey, {
  1196→      close: last.close,
  1197→      high: last.high,
  1198→      low: last.low,
  1199→      prevClose: prev.close,
  1200→      ts: now,
  1201→    });
  1202→
  1203→    // Update live instruments cache if this is a static instrument
  1204→    if (liveInstrumentsCache.data) {
  1205→      const inst = liveInstrumentsCache.data.find((i) => i.key === tgjuKey);
  1206→      if (inst && inst.price === 0) {
  1207→        inst.price = last.close;
  1208→        inst.highPrice = last.high;
  1209→        inst.lowPrice = last.low;
  1210→        inst.change = last.close - prev.close;
  1211→        inst.changePercent = prev.close > 0 ? ((last.close - prev.close) / prev.close) * 100 : 0;
  1212→      }
  1213→    }
  1214→  }
  1215→}
