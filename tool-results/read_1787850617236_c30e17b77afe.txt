   108→  { key: 'all', label: 'همه مسیرها' },
   109→  { key: 'up', label: 'صعودی' },
   110→  { key: 'pullback', label: 'رنج و خنثی' },

   116→// Component
   117→// ═══════════════════════════════════════════════════════════════════════════════
   118→
   119→export default function VdssGraph(props: VdssGraphProps) {
   120→  const { colors: C } = useTheme();
   121→  const {
   122→    symbolName,
   123→    currentPrice,
   124→    scenarios,
   125→    decisionGraph,
   126→  } = props;
   127→
   128→  const [activeFilter, setActiveFilter] = useState('all');
   129→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   130→  const graphRef = useRef<HTMLDivElement>(null);
   131→  const svgRef = useRef<SVGSVGElement>(null);
   132→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   133→  const shellRef = useRef<HTMLDivElement>(null);
   134→
   135→  // ── Derived data from decisionGraph ──────────────────────────────
   136→  const edges = decisionGraph?.edges ?? [];
   137→  const nodes = decisionGraph?.nodes ?? [];
   138→  const nodePositions = decisionGraph?.nodePositions ?? {};
   139→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   140→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   141→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   142→  const pathContributions = decisionGraph?.pathContributions ?? {};
   143→  const nodeValues = decisionGraph?.nodeValues ?? {};
   144→
   145→  // Build node lookup map
   146→  const nodeMap = useMemo(() => {
   147→    const map: Record<string, (typeof nodes)[number]> = {};
   148→    for (const n of nodes) map[n.id] = n;
   149→    return map;
   150→  }, [nodes]);
   151→
   152→  // ── Visible edges/nodes based on filter ──────────────────────────
   153→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   154→    if (activeFilter === 'all') {
   155→      const allEdgeIdx = edges.map((_, i) => i);
   156→      const allNodes = new Set(nodes.map(n => n.id));
   157→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   158→    }
   159→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   160→      // Show edges going TO this scenario node, plus their source nodes
   161→      const edgeSet = new Set<number>();
   162→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   163→      edges.forEach((e, i) => {
   164→        if (e.to === activeFilter) {
   165→          edgeSet.add(i);
   166→          nodeSet.add(e.from);
   167→        }
   168→      });
   169→      // Also show intermediate nodes connected to those sources
   170→      edges.forEach((e, i) => {
   171→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   172→          edgeSet.add(i);
   173→          nodeSet.add(e.from);
   174→        }
   175→      });
   176→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   177→    }
   178→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   179→      const branchType = `branch-${activeFilter}` as const;
   180→      const edgeSet = new Set<number>();
   181→      const nodeSet = new Set<string>(['ROOT']);
   182→      edges.forEach((e, i) => {
   183→        if (e.type === branchType || (e.type !== `branch-trend` && e.type !== `branch-breakout` && e.type !== `branch-reversal` && nodeSet.has(e.from))) {
   184→          // Include branch-type edges
   185→          if (e.type === branchType) {
   186→            edgeSet.add(i);
   187→            nodeSet.add(e.from);
   188→            nodeSet.add(e.to);
   189→          }
   190→        }
   191→      });
   192→      // Now add all edges whose source is already in nodeSet
   193→      const expanded = new Set<number>();
   194→      const expandedNodes = new Set<string>(nodeSet);
   195→      edges.forEach((e, i) => {
   196→        if (expandedNodes.has(e.from)) {
   197→          expanded.add(i);
   198→          expandedNodes.add(e.to);
   199→        }
   200→      });
   201→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   202→    }
   203→    // Type filter (up/pullback/down/risk)
   204→    const edgeSet = new Set<number>();
   205→    const nodeSet = new Set<string>(['ROOT']);
   206→    edges.forEach((e, i) => {
   207→      if (e.type === activeFilter) {
   208→        edgeSet.add(i);
   209→        nodeSet.add(e.from);
   210→        nodeSet.add(e.to);
   211→      }
   212→    });
   213→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   214→  }, [activeFilter, edges, nodes]);
   215→
   216→  // ── Draw SVG edges ───────────────────────────────────────────────
   217→  const drawEdges = useCallback(() => {
   218→    const svg = svgRef.current;
   219→    const graph = graphRef.current;
   220→    if (!svg || !graph || edges.length === 0) return;
   221→
   222→    let markersSvg = '';
   223→    const seenColors = new Set<string>();
   224→    for (const e of edges) {
   225→      const c = EDGE_COLORS[e.type];
   226→      if (c && !seenColors.has(e.type)) {
   227→        seenColors.add(e.type);
   228→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="8" markerHeight="8" refX="7" refY="4" orient="auto"><path d="M0,0 L8,4 L0,8 z" fill="${c}"/></marker>`;
   229→      }
   230→    }
   231→
   232→    let pathsSvg = '';
   233→    const graphRect = graph.getBoundingClientRect();
   234→
   235→    for (let ei = 0; ei < edges.length; ei++) {
   236→      const { from: fromId, to: toId, label, type } = edges[ei];
   237→      const isVisible = visibleEdgeIndices.includes(ei);
   238→      const fromEl = nodeRefs.current[fromId];
   239→      const toEl = nodeRefs.current[toId];
   240→      if (!fromEl || !toEl) continue;
   241→
   242→      const fRect = fromEl.getBoundingClientRect();
   243→      const tRect = toEl.getBoundingClientRect();
   244→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   245→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   246→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   247→      const by = tRect.top - graphRect.top + tRect.height / 2;
   248→      const dx = bx - ax;
   249→      const dy = by - ay;
   250→      const dist = Math.sqrt(dx * dx + dy * dy);
   251→      if (dist === 0) continue;
   252→
   253→      // Curved edges with bend proportional to distance
   254→      const bend = Math.min(40, Math.max(12, dist * 0.08));
   255→      const mx = (ax + bx) / 2;
   256→      const my = (ay + by) / 2;
   257→      const nx = -dy / dist;
   258→      const ny = dx / dist;
   259→      const cx = mx + nx * bend;
   260→      const cy = my + ny * bend;
   261→
   262→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   263→      const prob = edgeProbabilities[ei] ?? 0;
   264→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   265→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   266→
   267→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.7 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   268→
   269→      // Edge labels — skip deterministic root→branch and branch→event edges (index < 6)
   270→      if (isVisible && ei >= 6) {
   271→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#374151" font-size="8" text-anchor="middle" paint-order="stroke" stroke="#ffffff" stroke-width="3" stroke-linejoin="round" opacity="0.8" data-type="${type}" class="edge-label">${label}</text>`;
   272→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#ffffff" stroke-width="2.5" stroke-linejoin="round" opacity="0.85" class="edge-prob">${probLabel}</text>`;
   273→      }
   274→    }
   275→
   276→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   277→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   278→
   279→  useEffect(() => { drawEdges(); }, [drawEdges]);
   280→
   281→  useEffect(() => {
   282→    const el = shellRef.current;
   283→    if (!el) return;
   284→    const observer = new ResizeObserver(() => drawEdges());
   285→    observer.observe(el);
   286→    return () => observer.disconnect();
   287→  }, [drawEdges]);
   288→
   289→
   290→  // ── Detail panel content ─────────────────────────────────────────
   291→  const detailContent = useMemo(() => {
   292→    if (!selectedNode) return null;
   293→    const node = nodeMap[selectedNode];
   294→    if (!node) return null;
   295→
   296→    // For terminal scenario nodes, show branch breakdown
   297→    if (node.isTerminal && SCENARIO_KEYS.includes(selectedNode as typeof SCENARIO_KEYS[number])) {
   298→      const contrib = pathContributions[selectedNode] ?? { trend: 0, breakout: 0, reversal: 0 };
   299→      const meta = SCENARIO_META[selectedNode];
   300→      const s = scenarios[selectedNode as keyof typeof scenarios];
   301→      return (
   302→        <div className="space-y-3">
   303→          <h3 className="text-sm font-bold" style={{ color: meta.color }}>{node.title}</h3>
   304→          <p className="text-xs" style={{ color: C.cardSubFg }}>{node.desc}</p>
   305→          <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border bg-gray-50 text-gray-500">{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   306→          <div className="text-xs" style={{ color: C.cardSubFg }} dir="ltr">{nodeValues[selectedNode] ?? '--'}</div>
   307→
   308→          <div className="mt-3 pt-3" style={{ borderTop: `1px solid ${C.cardBorder}` }}>
   309→            <p className="text-xs font-bold mb-2" style={{ color: C.cardFg }}>سهم هر استراتژی:</p>
   310→            {Object.entries(BRANCH_META).map(([bKey, bMeta]) => {
   311→              const val = contrib[bKey as 'trend' | 'breakout' | 'reversal'];
   312→              const pct = (val * 100).toFixed(1);
   313→              return (
   314→                <div key={bKey} className="flex items-center justify-between py-1.5" style={{ borderBottom: `1px dashed ${C.cardBorder}` }}>
   315→                  <div className="flex items-center gap-1.5">
   316→                    <span className="inline-block w-2 h-2 rounded-full" style={{ background: bMeta.color }} />
   317→                    <span className="text-[11px]" style={{ color: C.cardFg }}>{bMeta.label}</span>
   318→                  </div>
   319→                  <span className="text-[11px] font-bold" style={{ color: bMeta.color }}>{toPersianDigits(pct)}٪</span>
   320→                </div>
   321→              );
   322→            })}
   323→            <div className="flex items-center justify-between py-1.5">
   324→              <span className="text-[11px] font-bold" style={{ color: C.cardFg }}>مجموع</span>
   325→              <span className="text-[11px] font-black" style={{ color: meta.color }}>{toFa(s?.probability ?? 0)}٪</span>
   326→            </div>
   327→          </div>
   328→        </div>
   329→      );
   330→    }
   331→
   332→    // For non-terminal nodes, show incoming/outgoing edges
   333→    const inputs = edges.filter(e => e.to === selectedNode);
   334→    const outputs = edges.filter(e => e.from === selectedNode);
   335→
   336→    return (
   337→      <div className="space-y-3">
   338→        <h3 className="text-sm font-bold" style={{ color: C.cardFg }}>{node.title}</h3>
   339→        <p className="text-xs" style={{ color: C.cardSubFg }}><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   340→        <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border" style={{ borderColor: C.cardBorder, background: C.cardBorder, color: C.cardSubFg }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   341→        <p className="text-xs leading-relaxed" style={{ color: C.cardSubFg }}>{node.desc}</p>
   342→        {inputs.length > 0 && (
   343→          <div>
   344→            <p className="text-xs font-medium mb-1" style={{ color: C.cardFg }}>مسیرهای ورودی ({toFa(inputs.length)}):</p>
   345→            <ul className="space-y-1">
   346→              {inputs.map((e, i) => {
   347→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   348→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   349→                return (
   350→                  <li key={i} className="text-[11px] leading-relaxed pt-1.5" style={{ color: C.cardSubFg, borderTop: `1px dashed ${C.cardBorder}` }}>
   351→                    <b style={{ color: C.cardFg }}>{SCENARIO_DISPLAY[e.from] || e.from} ← {SCENARIO_DISPLAY[e.to] || e.to}</b>
   352→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${edgeColor}20`, color: edgeColor }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   353→                    <br />{e.label}
   354→                  </li>
   355→                );
   356→              })}
   357→            </ul>
   358→          </div>
   359→        )}
   360→        {outputs.length > 0 && (
   361→          <div>
   362→            <p className="text-xs font-medium mb-1" style={{ color: C.cardFg }}>مسیرهای خروجی ({toFa(outputs.length)}):</p>
   363→            <ul className="space-y-1">
   364→              {outputs.map((e, i) => {
   365→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   366→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   367→                return (
   368→                  <li key={i} className="text-[11px] leading-relaxed pt-1.5" style={{ color: C.cardSubFg, borderTop: `1px dashed ${C.cardBorder}` }}>
   369→                    <b className="text-[#374151]">{SCENARIO_DISPLAY[e.from] || e.from} → {SCENARIO_DISPLAY[e.to] || e.to}</b>
   370→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${edgeColor}20`, color: edgeColor }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   371→                    <br />{e.label}
   372→                  </li>
   373→                );
   374→              })}
   375→            </ul>
   376→          </div>
   377→        )}
   378→      </div>
   379→    );
   380→  }, [selectedNode, nodeMap, nodeValues, edges, edgeProbabilities, pathContributions, scenarios]);
   381→
   382→  // ── All filter buttons ────────────────────────────────────────────
   383→  const allFilters = [
   384→    ...TYPE_FILTERS,
   385→    { key: 'sep1', label: '│', isSep: true as const },
   386→    ...Object.entries(BRANCH_META).map(([k, v]) => ({ key: k, label: v.label, branchKey: k, branchColor: v.color })),
   387→    { key: 'sep2', label: '│', isSep: true as const },
   388→    ...SCENARIO_KEYS.map(k => ({ key: k, label: SCENARIO_META[k].label, scenarioKey: k })),
   389→  ];
   390→
   391→  // ═══ Design dimensions ═══
   392→  const DESIGN_W = 1500;
   393→  const DESIGN_H = 820;
   394→  const DISPLAY_W = 1200;
   395→  const DISPLAY_H = DESIGN_H;
   396→  const scaleX = DISPLAY_W / DESIGN_W;
   397→  const scaleY = DISPLAY_H / DESIGN_H;
   398→
   399→  // ── Loading state ────────────────────────────────────────────────
   400→  if (!decisionGraph) {
   401→    return <VdssGraphSkeleton />;
   402→  }
   403→
   404→  return (
   405→    <div className="space-y-3" dir="rtl">
   406→      {/* ═══ HEADER ═══ */}
   407→      <div className="flex items-center justify-between gap-4 px-4 py-3 rounded-2xl" style={{ border: `1px solid ${C.cardBorder}`, background: C.cardBg, boxShadow: '0 4px 16px rgba(0,0,0,.06)' }}>
   408→        <div className="flex items-center gap-3">
   409→          <div className="w-10 h-10 rounded-xl border border-amber-200 flex items-center justify-center text-amber-800 text-xl font-bold"
   410→            style={{ boxShadow: 'inset 0 0 22px rgba(146,64,14,.06), 0 0 22px rgba(146,64,14,.04)' }}>◈</div>
   411→          <div>
   412→            <h2 className="text-base font-bold" style={{ color: C.cardFg }}>گراف تصمیم {symbolName}</h2>
   413→            <p className="text-[11px]" style={{ color: C.cardSubFg }}>مدل ۳‌شاخه‌ای | پیروی از روند، شکست، بازگشت | ۲۷ مسیر به ۹ سناریو</p>
   414→          </div>
   415→        </div>
   416→        <div className="text-left text-xs leading-relaxed pr-4" style={{ color: C.cardSubFg, borderRight: `1px solid ${C.cardBorder}` }}>
   417→          نقطه مرجع: <b className="text-cyan-700">{toFa(currentPrice)}</b><br />
   418→          افق برآورد: ۱۰ تا ۲۵ جلسه معاملاتی
   419→        </div>
   420→      </div>

   814→                {TREND_DAYS.map(day => (
   815→                  <TableHead key={day} colSpan={2} className="text-center text-[#374151] font-bold px-1.5 py-2 border-r border-[#e5e7eb]">
   816→                    روز {toPersianDigits(day.toString())}
   817→                  </TableHead>
   818→                ))}
   819→                <TableHead className="text-center text-[#374151] font-bold px-2 py-2 border-r border-[#e5e7eb]">اوج</TableHead>
   820→              </TableRow>
   821→              <TableRow className="border-b border-[#e5e7eb] bg-[#f9fafb]">
   822→                <TableHead className="px-2 py-1" />
   823→                {TREND_DAYS.map(day => (

   828→                ))}
   829→                <TableHead className="text-center text-[10px] text-[#6b7280] font-medium px-2 py-1">روز</TableHead>
   830→              </TableRow>
   831→            </TableHeader>
   832→            <TableBody>
   833→              {/* Scenario rows */}
   834→              {['R1','R2','R3','R4','R5','R6','R7','R8','R9'].map(key => {
   835→                const sc = scenarioMap[key];
   836→                if (!sc) return null;
   837→                const gc = TREND_GROUP_COLOR[sc.group];
   838→                return (
   839→                  <TableRow key={key} className="border-b border-[#e5e7eb]/60 hover:bg-[#f9fafb]">

   843→                    </TableCell>
   844→                    {TREND_DAYS.map(day => {

   846→                      const isPeak = sc.peakDay === day;
   847→                      return (
   848→                        <React.Fragment key={day}>
   849→                          <TableCell
   850→                            className={`text-center px-1 py-1 tabular-nums ${isPeak ? 'font-black' : 'font-medium'}`}
   851→                            style={isPeak ? { background: gc.badge, color: gc.text, borderRadius: 4 } : { color: '#374151' }}
   852→                          >
   853→                            {dp ? fmtPct(dp.individualProb) : '—'}
   854→                          </TableCell>
   855→                          <TableCell
   856→                            className={`text-center px-1 py-1 tabular-nums ${isPeak ? 'font-black' : 'font-medium'}`}

   867→                  </TableRow>
   868→                );

   882→                  const buCum = getDay(bullG?.trend ?? [], day)?.cumulativeProb ?? 0;

   891→                        </div>
   892→                        <span className="text-[10px] font-bold tabular-nums" style={{ color: barColor }}>
   893→                          {fmtPct(bullPct)}
   894→                        </span>
   895→                      </div>
   896→                    </TableCell>
   897→                  );
   898→                })}

   913→                    {TREND_DAYS.map(day => {
   914→                      const dp = getDay(gData.trend, day);
   915→                      return (
   916→                        <TableCell key={day} colSpan={2} className="text-center px-1 py-1 tabular-nums font-medium" style={{ color: gc.text }}>
   917→                          {dp ? fmtPct(dp.cumulativeProb) : '—'}
   918→                        </TableCell>
   919→                      );
   920→                    })}
   921→                    <TableCell className="text-center px-2 py-1 font-bold tabular-nums text-[10px]" style={{ color: gc.text }}>
   922→                      {lastDay ? fmtPct(lastDay.cumulativeProb) : '—'}
   923→                    </TableCell>
   924→                  </TableRow>
   925→                );
   926→              })}
   927→            </TableBody>
   928→          </Table>
   929→
   930→          <div className="mt-3 space-y-2">
   931→            <div className="px-4 py-2.5 rounded-lg border-r-3 border-emerald-700/60 bg-emerald-50 text-[11px] text-[#374151] leading-relaxed">
   932→              <b>توضیح:</b> احتمال اختصاصی = احتمال وقوع سناریو در آن روز خاص.
   933→              احتمال تجمعی (CDF) = مجموع احتمال از ضعیف‌ترین سناریو تا این سناریو در همان روز.
   934→              صعودی: از شوک صعودی (R9) تجمعی تا صعودی خفیف (R6) — هرچه قوی‌تر، احتمال تجمعی بیشتر.
   935→              نزولی: از شوک نزولی (R1) تجمعی تا نزولی خفیف (R4) — هرچه قوی‌تر، احتمال تجمعی بیشتر.
   936→              سلول‌های برجسته نشان‌دهنده روز اوج احتمال هر سناریو هستند. نوار روند گروهی نسبت تجمعی گاوی به کل را نشان می‌دهد.
   937→            </div>
   938→            {/* Scenario interpretations */}
   939→            {data.scenarios.filter(s => s.interpretation).map(s => (
   940→              <div key={s.scenarioKey} className="px-4 py-1.5 text-[10px] text-[#6b7280]">
   941→                <b>{s.label}:</b> {s.interpretation}
   942→              </div>
   943→            ))}
   944→          </div>
   945→        </div>
   946→      </CollapsibleContent>
   947→    </Collapsible>
   948→  );
   949→}
