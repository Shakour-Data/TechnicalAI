   928→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   929→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   930→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   931→  grade?: string; overlapCount?: number; touchCount?: number;
   932→}) {
   933→  const isSupport = type === 'support';
   934→  const isTargetType = type === 'target';
   935→
   936→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   937→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   938→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   939→
   940→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   941→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   942→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   943→
   944→  return (
   945→    <div
   946→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   947→      style={{
   948→        background: bgAccent,
   949→        border: `1px solid ${borderAccent}`,
   950→        borderRight: `3px solid ${accentColor}`,
   951→      }}
   952→    >
   953→      {/* Label + Price */}
   954→      <div className="flex-1 min-w-0">
   955→        <div className="flex items-center gap-2">
   956→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   957→          {isTarget && <span className="text-[9px]">🎯</span>}
   958→          {overlapCount !== undefined && overlapCount > 0 && (
   959→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   960→              {toPersianDigits(overlapCount)} هم‌پوشانی
   961→            </span>
   962→          )}
   963→        </div>
   964→        <div className="flex items-baseline gap-1">
   965→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   966→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   967→        </div>
   968→      </div>
