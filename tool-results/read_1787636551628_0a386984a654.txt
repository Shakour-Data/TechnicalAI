     1→   997→function buildContext(input: GraphInput): ProbContext {
     2→   998→  const adxNorm = clamp(input.adx / 60, 0, 1);
     3→   999→  const srProximity = clamp(
     4→  1000→    1 - Math.min(input.distToR1, input.distToS1) * 10,
     5→  1001→    0,
     6→  1002→    1
     7→  1003→  );
     8→  1004→  const oscillatorExtreme = clamp(
     9→  1005→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
    10→  1006→    0,
    11→  1007→    1
    12→  1008→  );
    13→  1009→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
    14→  1010→
    15→  1011→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
    16→  1012→  const overboughtRisk =
    17→  1013→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
    18→  1014→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
    19→  1015→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
    20→  1016→
    21→  1017→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
    22→  1018→  const oversoldBounce =
    23→  1019→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
    24→  1020→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
    25→  1021→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
    26→  1022→
    27→  1023→  return {
    28→  1024→    input,
    29→  1025→    adxNorm,
    30→  1026→    srProximity,
    31→  1027→    oscillatorExtreme,
    32→  1028→    divergenceProxy,
    33→  1029→    overboughtRisk,
    34→  1030→    oversoldBounce,
    35→  1031→  };
    36→  1032→}

    38→  1034→// === Node Probability Computation =============================================
    39→  1035→// For each non-terminal node, returns an array of probabilities for its
    40→  1036→// outgoing edges. The order MUST match the order of edges in createEdges().

    44→  1040→  nodeId: string,

    50→  1046→
    51→  1047→  switch (nodeId) {
    52→  1048→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
    53→  1049→    case 'ROOT': {
    54→  1050→      const trendW = adxNorm * 0.45 + input.mlTrend * 0.15 + 0.12;
    55→  1051→      const breakW = srProximity * 0.35 + (1 - adxNorm) * 0.1 + 0.1;

    58→  1054→    }

    65→  1061→        (1 - Math.abs(input.bullConsensus - 0.5) * 2) * (1 - adxNorm * 0.7);

    74→  1070→            (input.macdHist > 0
    75→  1071→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
    76→  1072→              : 0),
    77→  1073→          0,

    97→  1093→      const pR1 = 0.5 * (1 - input.confidenceIndex) + 0.15;

   354→  1350→    // ── Terminal nodes (R1–R9) have no outgoing edges ─────────────────────
   355→  1351→    default:
   356→  1352→      return [];
   357→  1353→  }
   358→  1354→}
   359→  1355→
   360→  1356→// === Graph Traversal ==========================================================
   361→  1357→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
   362→  1358→// =============================================================================
   363→  1359→
   364→  1360→function traverseGraph(
   365→  1361→  nodeId: string,
   366→  1362→  pathProb: number,
   367→  1363→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
   368→  1364→  ctx: ProbContext,
   369→  1365→  adj: Map<string, number[]>,
   370→  1366→  edges: GraphEdge[],
   371→  1367→  edgeProbabilities: Record<number, number>,
   372→  1368→  nodeAccum: Record<string, number>,
   373→  1369→  scenarios: Record<string, number>,
   374→  1370→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
   375→  1371→): void {
   376→  1372→  // Accumulate path probability at this node
   377→  1373→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
   378→  1374→
   379→  1375→  const outEdgeIndices = adj.get(nodeId);
   380→  1376→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
   381→  1377→
   382→  1378→  // Compute outgoing edge probabilities for this node
   383→  1379→  const probs = computeNodeProbabilities(
   384→  1380→    nodeId,
   385→  1381→    ctx,
   386→  1382→    outEdgeIndices.length
   387→  1383→  );
   388→  1384→
   389→  1385→  for (let i = 0; i < outEdgeIndices.length; i++) {
   390→  1386→    const edgeIdx = outEdgeIndices[i];
   391→  1387→    const edge = edges[edgeIdx];
   392→  1388→    const prob = probs[i] || 0;
   393→  1389→
   394→  1390→    // Store edge probability
   395→  1391→    edgeProbabilities[edgeIdx] = prob;
   396→  1392→
   397→  1393→    // Determine branch for child nodes
   398→  1394→    let childBranch = currentBranch;
   399→  1395→    if (nodeId === 'ROOT') {
   400→  1396→      if (edge.to === 'N_TREND') childBranch = 'trend';
   401→  1397→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
   402→  1398→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
   403→  1399→    }
   404→  1400→
   405→  1401→    const childPathProb = pathProb * prob;
   406→  1402→
   407→  1403→    // Check if target is a terminal node
   408→  1404→    const isTerminal =
   409→  1405→      edge.to.length === 2 &&
   410→  1406→      edge.to[0] === 'R' &&
   411→  1407→      edge.to[1] >= '1' &&
   412→  1408→      edge.to[1] <= '9';
   413→  1409→
   414→  1410→    if (isTerminal) {
   415→  1411→      // Accumulate scenario probability
   416→  1412→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
   417→  1413→      // Accumulate branch contribution
   418→  1414→      if (childBranch) {
   419→  1415→        contributions[edge.to][childBranch] += childPathProb;
   420→  1416→      }
   421→  1417→    }
   422→  1418→
   423→  1419→    // Recurse into child
   424→  1420→    traverseGraph(
   425→  1421→      edge.to,
   426→  1422→      childPathProb,
   427→  1423→      childBranch,
   428→  1424→      ctx,
   429→  1425→      adj,
   430→  1426→      edges,
   431→  1427→      edgeProbabilities,
   432→  1428→      nodeAccum,
   433→  1429→      scenarios,
   434→  1430→      contributions
   435→  1431→    );
   436→  1432→  }
   437→  1433→}

   439→  1435→// === Main Export: buildDecisionGraph ===========================================
   440→  1436→
   441→  1437→export function buildDecisionGraph(input: GraphInput): GraphData {
   442→  1438→  // 1. Build static graph structure
   443→  1439→  const nodes = createNodes();
   444→  1440→  const edges = createEdges();
   445→  1441→  const nodePositions = createPositions();
   446→  1442→
   447→  1443→  // 2. Build adjacency map: nodeId → array of edge indices
   448→  1444→  const adj = new Map<string, number[]>();
   449→  1445→  for (let i = 0; i < edges.length; i++) {
   450→  1446→    const from = edges[i].from;
   451→  1447→    if (!adj.has(from)) adj.set(from, []);
   452→  1448→    adj.get(from)!.push(i);
   453→  1449→  }
   454→  1450→
   455→  1451→  // 3. Build probability context
   456→  1452→  const ctx = buildContext(input);
   457→  1453→
   458→  1454→  // 4. Initialize accumulation structures
   459→  1455→  const edgeProbabilities: Record<number, number> = {};
   460→  1456→  const nodeAccum: Record<string, number> = {};
   461→  1457→  const scenarios: Record<string, number> = {};
   462→  1458→  const contributions: Record<
   463→  1459→    string,
   464→  1460→    { trend: number; breakout: number; reversal: number }
   465→  1461→  > = {};
   466→  1462→
   467→  1463→  for (let i = 1; i <= 9; i++) {
   468→  1464→    const key = `R${i}`;
   469→  1465→    scenarios[key] = 0;
   470→  1466→    contributions[key] = { trend: 0, breakout: 0, reversal: 0 };
   471→  1467→  }
   472→  1468→
   473→  1469→  // 5. Traverse graph from ROOT
   474→  1470→  traverseGraph(
   475→  1471→    'ROOT',
   476→  1472→    1.0,
   477→  1473→    null,
   478→  1474→    ctx,
   479→  1475→    adj,
   480→  1476→    edges,
   481→  1477→    edgeProbabilities,
   482→  1478→    nodeAccum,
   483→  1479→    scenarios,
   484→  1480→    contributions
   485→  1481→  );
   486→  1482→
   487→  1483→  // 6. Post-process scenario probabilities
   488→  1484→  // Convert to percentage, clamp [2, 35], re-normalize to sum=100
   489→  1485→  const rawPcts: Record<string, number> = {};
   490→  1486→  for (let i = 1; i <= 9; i++) {
   491→  1487→    const key = `R${i}`;
   492→  1488→    rawPcts[key] = Math.round((scenarios[key] || 0) * 100);
   493→  1489→  }
   494→  1490→
   495→  1491→  // Clamp each to [2, 35]
   496→  1492→  const clamped = Object.entries(rawPcts).map(([k, v]) => ({
   497→  1493→    key: k,
   498→  1494→    value: clamp(v, 2, 35),
   499→  1495→  }));
   500→  1496→
   501→  1497→  // Re-normalize to sum=100
   502→  1498→  const clampedSum = clamped.reduce((s, c) => s + c.value, 0);
   503→  1499→  const finalPcts: Record<string, number> = {};
   504→  1500→  if (clampedSum > 0) {
   505→  1501→    // Distribute proportionally and round, then fix rounding errors
   506→  1502→    let runningTotal = 0;
   507→  1503→    const entries = clamped
   508→  1504→      .map((c) => {
   509→  1505→        const raw = (c.value / clampedSum) * 100;
   510→  1506→        return { key: c.key, raw };
   511→  1507→      })
   512→  1508→      .sort((a, b) => b.raw - a.raw);
   513→  1509→
   514→  1510→    for (let i = 0; i < entries.length; i++) {
   515→  1511→      const isLast = i === entries.length - 1;
   516→  1512→      if (isLast) {
   517→  1513→        finalPcts[entries[i].key] = Math.round(100 - runningTotal);
   518→  1514→      } else {
   519→  1515→        const rounded = Math.round(entries[i].raw);
   520→  1516→        finalPcts[entries[i].key] = rounded;
   521→  1517→        runningTotal += rounded;
   522→  1518→      }
   523→  1519→    }
   524→  1520→  }
   525→  1521→
   526→  1522→  // Ensure no negative values due to rounding
   527→  1523→  for (const key of Object.keys(finalPcts)) {
   528→  1524→    if (finalPcts[key] < 0) finalPcts[key] = 0;
   529→  1525→  }
   530→  1526→
   531→  1527→  // 7. Compute branch probabilities (from ROOT outgoing edges)
   532→  1528→  const rootEdgeIndices = adj.get('ROOT') || [];
   533→  1529→  const rootProbs = computeNodeProbabilities('ROOT', ctx, rootEdgeIndices.length);
   534→  1530→  const branchProbabilities = {
   535→  1531→    trend: rootProbs[0] || 0,
   536→  1532→    breakout: rootProbs[1] || 0,
   537→  1533→    reversal: rootProbs[2] || 0,
   538→  1534→  };
   539→  1535→
   540→  1536→  // 8. Build node display values (path probability as percentage)
   541→  1537→  const nodeValues: Record<string, string> = {};
   542→  1538→  for (const [nodeId, accum] of Object.entries(nodeAccum)) {
   543→  1539→    nodeValues[nodeId] = `${Math.round(accum * 100)}%`;
   544→  1540→  }
   545→  1541→
   546→  1542→  // Override terminal node values with final post-processed percentages
   547→  1543→  for (const [key, pct] of Object.entries(finalPcts)) {
   548→  1544→    nodeValues[key] = `${pct}%`;
   549→  1545→  }
   550→  1546→
   551→  1547→  // 9. Build path contributions (as percentages)
   552→  1548→  const pathContributions: Record<
   553→  1549→    string,
   554→  1550→    { trend: number; breakout: number; reversal: number }
   555→  1551→  > = {};
   556→  1552→  for (let i = 1; i <= 9; i++) {
   557→  1553→    const key = `R${i}`;
   558→  1554→    const c = contributions[key];
   559→  1555→    const total = c.trend + c.breakout + c.reversal;
   560→  1556→    if (total > 0) {
   561→  1557→      pathContributions[key] = {
   562→  1558→        trend: Math.round((c.trend / total) * 100),
   563→  1559→        breakout: Math.round((c.breakout / total) * 100),
   564→  1560→        reversal: Math.round((c.reversal / total) * 100),
   565→  1561→      };
   566→  1562→    } else {
   567→  1563→      pathContributions[key] = { trend: 33, breakout: 34, reversal: 33 };
   568→  1564→    }
   569→  1565→  }
   570→  1566→
   571→  1567→  // 10. Compute 30-day probability trend
   572→  1568→  const probFractions: Record<string, number> = {};
   573→  1569→  for (const [k, v] of Object.entries(finalPcts)) {
   574→  1570→    probFractions[k] = v / 100;
   575→  1571→  }
   576→  1572→  const probabilityTrend = calcTrendFromProbs(probFractions, 30);
   577→  1573→
   578→  1574→  // 11. Return complete GraphData
   579→  1575→  return {
   580→  1576→    nodes,
   581→  1577→    edges,
   582→  1578→    nodePositions,
   583→  1579→    edgeProbabilities,
   584→  1580→    nodeValues,
   585→  1581→    branchProbabilities,
   586→  1582→    scenarioProbabilities: finalPcts,
   587→  1583→    pathContributions,
   588→  1584→    probabilityTrend,
   589→  1585→  };
   590→  1586→}
   591→  1587→
   592→  1588→// === 30-Day Probability Trend Calculation ======================================
   593→  1589→
   594→  1590→// Dual exponential decay for forecast weighting
   595→  1591→// W_forecast_h = α * exp(-h/τ₁) + (1-α) * exp(-h/τ₂)
   596→  1592→export function dualDecay(h: number, alpha = 0.7, tauShort = 5, tauLong = 20): number {
   597→  1593→  return alpha * Math.exp(-h / tauShort) + (1 - alpha) * Math.exp(-h / tauLong);
   598→  1594→}
   599→  1595→
   600→  1596→// Normalize decay factors so they sum to 1
   601→  1597→export function normalizedDecayFactors(horizon: number): number[] {
   602→  1598→  const factors: number[] = [];
   603→  1599→  for (let h = 1; h <= horizon; h++) {
   604→  1600→    factors.push(dualDecay(h));
   605→  1601→  }
   606→  1602→  const total = factors.reduce((a, b) => a + b, 0);
   607→  1603→  return factors.map(f => f / total);
   608→  1604→}

   641→  1637→export function calculateProbabilityTrend(
   642→  1638→  scenarioProbabilities: Array<{key: string; name: string; nameEn: string; probability: number; isBullish?: boolean; isBearish?: boolean; isNeutral?: boolean}>,
   643→  1639→  horizon: number = 30
   644→  1640→): ProbabilityTrendResult {
   645→  1641→  const decay = normalizedDecayFactors(horizon);
   646→  1642→  
   647→  1643→  const scenarios: ScenarioTrend[] = scenarioProbabilities.map(sp => {
   648→  1644→    const trend: DayProbability[] = [];
   649→  1645→    let cumProb = 0;
   650→  1646→    let peakDay = 1;
   651→  1647→    let peakProb = 0;
   652→  1648→    
   653→  1649→    for (let h = 1; h <= horizon; h++) {
   654→  1650→      const individualProb = sp.probability * decay[h - 1];
   655→  1651→      cumProb += individualProb;
   656→  1652→      trend.push({ day: h, individualProb: Math.round(individualProb * 10000) / 10000, cumulativeProb: Math.round(cumProb * 10000) / 10000 });
   657→  1653→      if (individualProb > peakProb) { peakProb = individualProb; peakDay = h; }
   658→  1654→    }
   659→  1655→    
   660→  1656→    // Determine trend direction
   661→  1657→    const firstThird = trend.slice(0, Math.floor(horizon / 3));
   662→  1658→    const lastThird = trend.slice(Math.floor(horizon * 2 / 3));
   663→  1659→    const avgFirst = firstThird.reduce((a, d) => a + d.individualProb, 0) / firstThird.length;
   664→  1660→    const avgLast = lastThird.reduce((a, d) => a + d.individualProb, 0) / lastThird.length;
   665→  1661→    const direction: ScenarioTrend['trendDirection'] = avgLast > avgFirst * 1.05 ? 'صعودی' : avgLast < avgFirst * 0.95 ? 'نزولی' : 'پایدار';
   666→  1662→    
   667→  1663→    return { name: sp.name, nameEn: sp.nameEn, key: sp.key, currentProbability: sp.probability, trend, trendDirection: direction, peakDay, peakProbability: Math.round(peakProb * 10000) / 10000, interpretation: '' };
   668→  1664→  });
   669→  1665→  
   670→  1666→  // Group trends (bullish, bearish, neutral)
   671→  1667→  const groupTrends: CumulativeGroupTrend = { bullish: [], bearish: [], neutral: [] };
   672→  1668→  for (let h = 0; h < horizon; h++) {
   673→  1669→    const bullDay = { day: h + 1, individualProb: 0, cumulativeProb: 0 };
   674→  1670→    const bearDay = { day: h + 1, individualProb: 0, cumulativeProb: 0 };
   675→  1671→    const neutralDay = { day: h + 1, individualProb: 0, cumulativeProb: 0 };
   676→  1672→    
   677→  1673→    scenarioProbabilities.forEach(sp => {
   678→  1674→      const dayData = scenarios.find(s => s.key === sp.key)?.trend[h];
   679→  1675→      if (!dayData) return;
   680→  1676→      if (sp.isBullish) { bullDay.individualProb += dayData.individualProb; }
   681→  1677→      else if (sp.isBearish) { bearDay.individualProb += dayData.individualProb; }
   682→  1678→      else { neutralDay.individualProb += dayData.individualProb; }
   683→  1679→    });
   684→  1680→    
   685→  1681→    bullDay.cumulativeProb = Math.round(bullDay.individualProb * 10000) / 10000;
   686→  1682→    bearDay.cumulativeProb = Math.round(bearDay.individualProb * 10000) / 10000;
   687→  1683→    neutralDay.cumulativeProb = Math.round(neutralDay.individualProb * 10000) / 10000;
   688→  1684→    
   689→  1685→    groupTrends.bullish.push({ ...bullDay, individualProb: Math.round(bullDay.individualProb * 10000) / 10000 });
   690→  1686→    groupTrends.bearish.push({ ...bearDay, individualProb: Math.round(bearDay.individualProb * 10000) / 10000 });
   691→  1687→    groupTrends.neutral.push({ ...neutralDay, individualProb: Math.round(neutralDay.individualProb * 10000) / 10000 });
   692→  1688→  }
   693→  1689→  
   694→  1690→  // Generate key observations and actionable insights
   695→  1691→  const topScenario = [...scenarios].sort((a, b) => b.currentProbability - a.currentProbability)[0];
   696→  1692→  const keyObservations = [
   697→  1693→    `سناریوی غالب: ${topScenario?.name || 'N/A'} (${topScenario ? Math.round(topScenario.currentProbability * 100) : 0}%)`,
   698→  1694→    `اوج احتمال ${topScenario?.name || ''} در روز ${topScenario?.peakDay || 'N/A'}`,
   699→  1695→  ];
   700→  1696→  
   701→  1697→  const actionableInsights = [
   702→  1698→    topScenario?.isBullish ? 'روند صعودی غالب - بررسی نقاط ورود' : topScenario?.isBearish ? 'روند نزولی غالب - احتیاط در معاملات' : 'بازار در حالت رنج - انتظار برای شکست',
   703→  1699→  ];
   704→  1700→  
   705→  1701→  return { scenarios, groupTrends, keyObservations, actionableInsights };
   706→  1702→}
