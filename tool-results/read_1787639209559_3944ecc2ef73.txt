     5→## 🎯 معرفی کلی
     6→
     7→سیستم **مکتب، سبک و لحن تحلیل** یک چارچوب هوشمند و خودکار برای تعیین **روش‌شناسی، ساختار ارائه و حالت احساسی** تحلیل‌های تولیدشده توسط سیستم گراف تصمیم است. این سیستم با استفاده از یادگیری ماشین، داده‌های بازار و بازخورد عملکرد، به‌صورت پویا مناسب‌ترین ترکیب را برای هر شرایط بازار و هر مخاطب خاص انتخاب می‌کند.

    25→## ۱. مکتب تحلیل (School of Analysis)
    26→
    27→### ۱.۱. تعریف و مفهوم
    28→
    29→**مکتب تحلیل** به چارچوب فکری، روش‌شناسی و فلسفه تحلیلی گفته می‌شود که بر اساس آن، داده‌های بازار تفسیر و تحلیل می‌شوند. هر مکتب دارای اصول، ابزارها و قواعد خاص خود است که نحوه نگاه به بازار و استخراج سیگنال‌های معاملاتی را تعیین می‌کند.
    30→

    34→
    35→```yaml
    36→school_classical:
    37→  id: "classical"
    38→  name: "مکتب کلاسیک"
    39→  name_en: "Classical School"
    40→  
    41→  # اصول بنیادین
    42→  core_principles:
    43→    - "قیمت همه‌چیز را تعدیل می‌کند"
    44→    - "تاریخ خود را تکرار می‌کند"
    45→    - "روندها تا زمانی که شکسته نشوند، معتبر هستند"
    46→    
    47→  # ابزارهای اصلی
    48→  primary_tools:
    49→    - "سطوح حمایت و مقاومت"
    50→    - "خطوط روند و کانال‌ها"
    51→    - "الگوهای کلاسیک (سر و شانه، مثلث، پرچم، گوه)"
    52→    - "میانگین‌های متحرک ساده"
    53→    
    54→  # نشانه‌های تشخیص
    55→  detection_signals:
    56→    - "تعداد الگوهای کلاسیک تشخیص‌داده‌شده ≥ ۲"
    57→    - "اعتبار الگوهای کلاسیک > ۰.۶"
    58→    - "قدرت سطوح حمایت/مقاومت > ۳"
    59→    - "نسبت الگوهای کلاسیک به کل الگوها > ۰.۴"
    60→    
    61→  # نقاط قوت
    62→  strengths:
    63→    - "سادگی و فهم آسان"
    64→    - "کاربرد در تمام بازارها"
    65→    - "تأیید بصری قوی"
    66→    - "مناسب برای تایم‌فریم‌های بلندمدت"
    67→    
    68→  # نقاط ضعف
    69→  weaknesses:
    70→    - "تشخیص الگوها ذهنی است"
    71→    - "در بازارهای با نوسان بالا کمتر مؤثر است"
    72→    - "زمان‌بر بودن تشخیص"
    73→    
    74→  # وزن پایه

    76→  
    77→  # ضریب تطبیق با شرایط
    78→  adaptive_factors:
    79→    trend_strength: 0.2
    80→    pattern_quality: 0.4
    81→    timeframe_weight: 0.2
    82→    volatility_adjustment: 0.2
    83→```

   447→## ۲. سبک تحلیل (Analysis Style)
   448→
   449→### ۲.۱. تعریف و مفهوم
   450→
   451→**سبک تحلیل** به نحوه ارائه، ساختاردهی و جزئیات محتوای تحلیلی گفته می‌شود. سبک تعیین می‌کند که تحلیل چقدر مفصل باشد، بر چه جنبه‌هایی تأکید شود و برای چه مخاطبی مناسب است.

  1271→def update_weights_daily(performance_data):
  1272→    """
  1273→    به‌روزرسانی روزانه وزن‌ها بر اساس عملکرد واقعی
  1274→    """

  1278→        sharpe = calculate_sharpe(school['id'], performance_data)
  1279→        drawdown = calculate_drawdown(school['id'], performance_data)
  1280→        
  1281→        # محاسبه ضریب تنظیم
  1282→        adjustment_factor = 1.0
  1283→        adjustment_factor *= (1 + (win_rate - 0.55) * 0.5)  # افزایش/کاهش بر اساس نرخ برد
  1284→        adjustment_factor *= (1 + (sharpe - 1.5) * 0.2)     # افزایش/کاهش بر اساس شارپ
  1285→        adjustment_factor *= (1 - (drawdown - 0.15) * 0.3)  # کاهش در صورت افزایش ضرر
  1286→        
  1287→        # محدود کردن ضریب
  1288→        adjustment_factor = max(0.8, min(1.2, adjustment_factor))
  1289→        
  1290→        # به‌روزرسانی وزن
  1291→        school['base_weight'] *= adjustment_factor
  1292→        
  1293→    # نرمال‌سازی نهایی
  1294→    normalize_weights(SCHOOLS)
