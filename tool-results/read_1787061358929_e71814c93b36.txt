     1→   427→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
     2→   428→  resistances: number[];
     3→   429→  supports: number[];
     4→   430→  supportStrengths: LevelStrength[];
     5→   431→  resistanceStrengths: LevelStrength[];
     6→   432→  priceTargets: LevelStrength[];
     7→   433→} {
     8→   434→  const closes = data.map(d => d.close);
     9→   435→  const avgVolume = data.reduce((s, d) => s + d.volume, 0) / data.length;
    10→   436→
    11→   437→  // Tag each raw level with its source type for confluence scoring
    12→   438→  interface TaggedLevel { price: number; source: string };
    13→   439→  const allRaw: TaggedLevel[] = [];
    14→   440→
    15→   441→  // 1. Pivot Points (source: 'pivot')
    16→   442→  const prev = data[data.length - 2] ?? data[data.length - 1];
    17→   443→  if (prev) {
    18→   444→    const pp = (prev.high + prev.low + prev.close) / 3;
    19→   445→    const hl = prev.high - prev.low;
    20→   446→    const pivotR = [
    21→   447→      2 * pp - prev.low, pp + hl,
    22→   448→      prev.high + 2 * (pp - prev.low),
    23→   449→      prev.high + 2 * (pp - prev.low) + hl,
    24→   450→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
    25→   451→    ];
    26→   452→    const pivotS = [
    27→   453→      2 * pp - prev.high, pp - hl,
    28→   454→      prev.low - 2 * (prev.high - pp),
    29→   455→      prev.low - 2 * (prev.high - pp) - hl,
    30→   456→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
    31→   457→    ];
    32→   458→    for (const p of pivotR) allRaw.push({ price: p, source: 'pivot' });
    33→   459→    for (const p of pivotS) allRaw.push({ price: p, source: 'pivot' });
    34→   460→  }
    35→   461→
    36→   462→  // 2. Swing levels (source: 'swing')
    37→   463→  for (const lb of [3, 5, 7, 10]) {
    38→   464→    const swings = findSwingLevels(data, lb);
    39→   465→    for (const h of swings.highs) allRaw.push({ price: h, source: 'swing' });
    40→   466→    for (const l of swings.lows) allRaw.push({ price: l, source: 'swing' });
    41→   467→  }
    42→   468→
    43→   469→  // 3. Moving Average levels (source: 'ma')
    44→   470→  for (const period of [5, 10, 21, 50, 100, 200]) {
    45→   471→    const v = sma(closes, period);
    46→   472→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01)
    47→   473→      allRaw.push({ price: v, source: 'ma' });
    48→   474→  }
    49→   475→  const ema12 = emaCalc(closes, 12);
    50→   476→  const ema26 = emaCalc(closes, 26);
    51→   477→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01)
    52→   478→    allRaw.push({ price: ema12, source: 'ma' });
    53→   479→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01)
    54→   480→    allRaw.push({ price: ema26, source: 'ma' });
    55→   481→
    56→   482→  // 4. Bollinger Band levels (source: 'bb')
    57→   483→  const bbLevels = calcBollingerBands(closes);
    58→   484→  if (bbLevels.upper > 0) allRaw.push({ price: bbLevels.upper, source: 'bb' });
    59→   485→  if (bbLevels.lower > 0) allRaw.push({ price: bbLevels.lower, source: 'bb' });
    60→   486→
    61→   487→  // 5. Round number / psychological levels (source: 'round')
    62→   488→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
    63→   489→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
    64→   490→    const p = Math.round(currentPrice * mult / magnitude) * magnitude;
    65→   491→    allRaw.push({ price: p, source: 'round' });
    66→   492→  }
    67→   493→
    68→   494→  // 6. Recent high/low zones (source: 'hilo')
    69→   495→  for (const w of [5, 22, 66]) {
    70→   496→    const slice = data.slice(-Math.min(w, data.length));
    71→   497→    if (slice.length > 0) {
    72→   498→      allRaw.push({ price: Math.max(...slice.map(d => d.high)), source: 'hilo' });
    73→   499→      allRaw.push({ price: Math.min(...slice.map(d => d.low)), source: 'hilo' });
    74→   500→    }
    75→   501→  }
    76→   502→
    77→   503→  // 7. Round all and filter invalid
    78→   504→  const rounded: TaggedLevel[] = allRaw
    79→   505→    .map(l => ({ price: roundToNice(l.price), source: l.source }))
    80→   506→    .filter(l => l.price > 0);
    81→   507→
    82→   508→  // 8. Cluster nearby levels (within 0.5%) — merge sources, keep most common price
    83→   509→  interface Cluster {
    84→   510→    price: number;
    85→   511→    sources: Set<string>;
    86→   512→    touchCount: number;
    87→   513→    totalVolumeAtTouches: number;
    88→   514→    mostRecentTouchBar: number;
    89→   515→  }
    90→   516→  const clusters: Cluster[] = [];
    91→   517→  for (const lvl of rounded) {
    92→   518→    let merged = false;
    93→   519→    for (const c of clusters) {
    94→   520→      if (Math.abs(c.price - lvl.price) / Math.max(c.price, 1) < 0.005) {
    95→   521→        c.sources.add(lvl.source);
    96→   522→        merged = true;
    97→   523→        break;
    98→   524→      }
    99→   525→    }
   100→   526→    if (!merged) {
   101→   527→      clusters.push({ price: lvl.price, sources: new Set([lvl.source]), touchCount: 0, totalVolumeAtTouches: 0, mostRecentTouchBar: -1 });
   102→   528→    }
   103→   529→  }
   104→   530→
   105→   531→  // 9. Count actual price TOUCHES (rejections) and volume at each cluster level
   106→   532→  // A real touch = bar wick reached the level AND bar body closed away from it
   107→   533→  const touchThreshold = currentPrice * 0.002; // 0.2% tight proximity
   108→   534→  for (let i = 0; i < data.length; i++) {
   109→   535→    const bar = data[i];
   110→   536→    for (const c of clusters) {
   111→   537→      const isResistance = c.price > currentPrice;
   112→   538→      if (isResistance) {
   113→   539→        // Resistance: bar high reached near the level AND closed below it (rejection)
   114→   540→        if (bar.high >= c.price - touchThreshold && bar.close < c.price) {
   115→   541→          c.touchCount++;
   116→   542→          c.totalVolumeAtTouches += bar.volume;
   117→   543→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   118→   544→        }
   119→   545→      } else {
   120→   546→        // Support: bar low reached near the level AND closed above it (rejection)
   121→   547→        if (bar.low <= c.price + touchThreshold && bar.close > c.price) {
   122→   548→          c.touchCount++;
   123→   549→          c.totalVolumeAtTouches += bar.volume;
   124→   550→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   125→   551→        }
   126→   552→      }
   127→   553→    }
   128→   554→  }
   129→   555→
   130→   556→  // 10. Calculate multi-factor strength (0-10)
   131→   557→  const levelStrengths: LevelStrength[] = clusters.map(c => {
   132→   558→    // Factor 1: Source Confluence (0-2.0 pts)
   133→   559→    const sourceScore = Math.min(2.0, c.sources.size * 0.4);
   134→   560→    const hasSwingAndPivot = c.sources.has('swing') && c.sources.has('pivot') ? 0.4 : 0;
   135→   561→    const maConfluence = c.sources.has('ma') ? 0.4 : 0;
   136→   562→    const confluenceScore = Math.min(2.0, sourceScore + hasSwingAndPivot + maConfluence);
   137→   563→
   138→   564→    // Factor 2: Touch Count — CAPPED by recency (0-1.5 pts)
   139→   565→    // Only count touches in the most recent 50% of data to avoid ancient history inflating
   140→   566→    const recentBarStart = Math.floor(data.length * 0.5);
   141→   567→    const recentTouches = c.mostRecentTouchBar >= recentBarStart ? c.touchCount : Math.max(0, c.touchCount - 1);
   142→   568→    let touchScore: number;
   143→   569→    if (recentTouches === 0) touchScore = 0;
   144→   570→    else if (recentTouches === 1) touchScore = 0.3;
   145→   571→    else if (recentTouches <= 2) touchScore = 0.6;
   146→   572→    else if (recentTouches <= 4) touchScore = 0.9;
   147→   573→    else if (recentTouches <= 7) touchScore = 1.2;
   148→   574→    else touchScore = 1.5;
   149→   575→
   150→   576→    // Factor 3: Volume Confirmation (0-1.0 pts) — skip when no volume available
   151→   577→    let volumeScore = 0;
   152→   578→    const dataHasVolume = data.some(d => d.volume > 0);
   153→   579→    if (dataHasVolume && c.touchCount > 0) {
   154→   580→      const avgVolAtTouch = c.totalVolumeAtTouches / c.touchCount;
   155→   581→      const volRatio = avgVolume > 0 ? avgVolAtTouch / avgVolume : 1;
   156→   582→      if (volRatio > 2.0) volumeScore = 1.0;
   157→   583→      else if (volRatio > 1.5) volumeScore = 0.8;
   158→   584→      else if (volRatio > 1.0) volumeScore = 0.5;
   159→   585→      else if (volRatio > 0.7) volumeScore = 0.25;
   160→   586→      else volumeScore = 0.1;
   161→   587→    }
   162→   588→
   163→   589→    // Factor 4: Freshness / Recency (0-1.0 pts)
   164→   590→    let freshnessScore = 0;
   165→   591→    if (c.touchCount > 0 && data.length > 0) {
   166→   592→      const barsAgo = data.length - 1 - c.mostRecentTouchBar;
   167→   593→      if (barsAgo <= 3) freshnessScore = 1.0;
   168→   594→      else if (barsAgo <= 10) freshnessScore = 0.8;
   169→   595→      else if (barsAgo <= 25) freshnessScore = 0.5;
   170→   596→      else if (barsAgo <= 50) freshnessScore = 0.25;
   171→   597→      else freshnessScore = 0.05;
   172→   598→    }
   173→   599→
   174→   600→    // Factor 5: Proximity to current price (0-0.5 pts)
   175→   601→    const distRatio = Math.abs(c.price - currentPrice) / currentPrice;
   176→   602→    let proximityScore = 0;
   177→   603→    if (distRatio < 0.01) proximityScore = 0.5;
   178→   604→    else if (distRatio < 0.02) proximityScore = 0.35;
   179→   605→    else if (distRatio < 0.04) proximityScore = 0.2;
   180→   606→    else if (distRatio < 0.07) proximityScore = 0.1;
   181→   607→
   182→   608→    const total = confluenceScore + touchScore + volumeScore + freshnessScore + proximityScore;
   183→   609→    const strength = Math.min(10, Math.max(1, Math.round(total * 10 / 6.0)));
   184→   610→
   185→   611→    return { price: c.price, strength, isTarget: false };
   186→   612→  });
   187→   613→
   188→   614→  // 11. Separate above/below current price
   189→   615→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   190→   616→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   191→   617→
   192→   618→  // 12. Adaptive gap enforcement — keep highest strength in cluster
   193→   619→  // v2: enforce 5-10% gap between consecutive S/R lines
   194→   620→  function enforceGap(levels: LevelStrength[], targetCount: number): LevelStrength[] {
   195→   621→    const gapSteps = [0.10, 0.09, 0.08, 0.07, 0.06, 0.05];
   196→   622→    for (const gapPct of gapSteps) {
   197→   623→      if (levels.length === 0) return [];
   198→   624→      const result: LevelStrength[] = [levels[0]];
   199→   625→      for (let i = 1; i < levels.length; i++) {
   200→   626→        if (result.length >= targetCount) break;
   201→   627→        const prevPrice = result[result.length - 1].price;
   202→   628→        const currPrice = levels[i].price;
   203→   629→        const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   204→   630→        if (gap >= gapPct) {
   205→   631→          result.push(levels[i]);
   206→   632→        } else {
   207→   633→          // Merge: keep stronger level's strength (no inflation)
   208→   634→          const keep = levels[i].strength >= result[result.length - 1].strength ? levels[i] : result[result.length - 1];
   209→   635→          result[result.length - 1] = { ...keep, strength: keep.strength };
   210→   636→        }
   211→   637→      }
   212→   638→      if (result.length >= targetCount) return result;
   213→   639→    }
   214→   640→    return levels.slice(0, targetCount);
   215→   641→  }
   216→   642→
   217→   643→  const finalResistances = enforceGap(above, 6);
   218→   644→  const finalSupports = enforceGap(below, 6);
   219→   645→
   220→   646→  // 13. Determine price targets: top 2 by strength, or any above 7
   221→   647→  const allLevels = [...finalResistances, ...finalSupports];
   222→   648→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   223→   649→  let targets: LevelStrength[];
   224→   650→  if (sortedByStrength.some(l => l.strength >= 7)) {
   225→   651→    targets = sortedByStrength.filter(l => l.strength >= 7).slice(0, 3);
   226→   652→  } else {
   227→   653→    targets = sortedByStrength.slice(0, 2);
   228→   654→  }
   229→   655→
   230→   656→  // Mark targets
   231→   657→  targets = targets.map(t => ({ ...t, isTarget: true }));
   232→   658→
   233→   659→  // Also mark in final arrays
   234→   660→  finalResistances.forEach(r => {
   235→   661→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   236→   662→  });
   237→   663→  finalSupports.forEach(s => {
   238→   664→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   239→   665→  });
   240→   666→
   241→   667→  return {
   242→   668→    resistances: finalResistances.map(l => l.price),
   243→   669→    supports: finalSupports.map(l => l.price),
   244→   670→    supportStrengths: finalSupports,
   245→   671→    resistanceStrengths: finalResistances,
   246→   672→    priceTargets: targets,
   247→   673→  };
   248→   674→}

   251→   677→function calcTrend(closes: number[], period: number): TrendResult {
   252→   678→  const slice = closes.slice(-Math.min(period, closes.length));
   253→   679→  const { slope, r2 } = linearRegression(slice);
   254→   680→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   255→   681→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   256→   682→  const priceThreshold = avgPrice * 0.001; // 0.1%
   257→   683→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   258→   684→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   259→   685→}

   266→   692→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
   267→   693→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
   268→   694→  const slice = data.slice(0, barIdx + 1);
   269→   695→  const price = slice[slice.length - 1].close;
   270→   696→  let nearestR = price * 1.05;
   271→   697→  let nearestS = price * 0.95;
   272→   698→
   273→   699→  // Collect all swing highs (resistance) and swing lows (support)
   274→   700→  const resistances: number[] = [];
   275→   701→  const supports: number[] = [];
   276→   702→  for (const lb of [3, 5, 7]) {
   277→   703→    const swings = findSwingLevels(slice, lb);
   278→   704→    resistances.push(...swings.highs);
   279→   705→    supports.push(...swings.lows);
   280→   706→  }
   281→   707→
   282→   708→  // Find nearest resistance above price
   283→   709→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
   284→   710→  if (abovePrice.length > 0) nearestR = abovePrice[0];
   285→   711→
   286→   712→  // Find nearest support below price
   287→   713→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
   288→   714→  if (belowPrice.length > 0) nearestS = belowPrice[0];
   289→   715→
   290→   716→  return { R1: nearestR, S1: nearestS };
   291→   717→}

   293→   719→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
   294→   720→// Returns true if bullish divergence detected, false if bearish, null if none
   295→   721→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
   296→   722→  const lookback = 20;
   297→   723→  const start = Math.max(0, barIdx - lookback);
   298→   724→  const slice = data.slice(start, barIdx + 1);
   299→   725→  if (slice.length < 10) return null;
   300→   726→
   301→   727→  const mid = Math.floor(slice.length / 2);
   302→   728→  const firstHalf = slice.slice(0, mid);
   303→   729→  const secondHalf = slice.slice(mid);
   304→   730→
   305→   731→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
   306→   732→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
   307→   733→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
   308→   734→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
   309→   735→
   310→   736→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
   311→   737→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
   312→   738→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
   313→   739→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
   314→   740→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
   315→   741→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
   316→   742→
   317→   743→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
   318→   744→    return false; // bearish divergence
   319→   745→  }
   320→   746→
   321→   747→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
   322→   748→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
   323→   749→
   324→   750→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
   325→   751→    return true; // bullish divergence
   326→   752→  }
   327→   753→
   328→   754→  return null;
   329→   755→}

   388→   814→    : mfiVal > 30 ? 0.30
   389→   815→    : mfiVal > 20 ? 0.15
   390→   816→    : 0.05;
   391→   817→
   392→   818→  // s_cci — CCI mapped to 0-1
   393→   819→  const s_cci = cciVal > 200 ? 0.90
   394→   820→    : cciVal > 100 ? 0.75
   395→   821→    : cciVal > 0 ? 0.60
   396→   822→    : cciVal > -100 ? 0.40
   397→   823→    : cciVal > -200 ? 0.25
   398→   824→    : 0.10;
   399→   825→
   400→   826→  // s_adx — Directional bias
   401→   827→  const s_adx = (adxResult.diPlus - adxResult.diMinus + 100) / 200;
   402→   828→
   403→   829→  // s_macd — MACD histogram + signal combined
   404→   830→  const absMacdLine = Math.abs(macd.line) + 1;
   405→   831→  const s_macd = macd.line > macd.signal
   406→   832→    ? 0.5 + Math.min(0.5, macd.histogram / absMacdLine)
   407→   833→    : 0.5 - Math.min(0.5, Math.abs(macd.histogram) / absMacdLine);
   408→   834→
   409→   835→  // s_stoch — Stochastic %K mapped to 0-1
   410→   836→  const s_stoch = stoch.k / 100;
   411→   837→
   412→   838→  // s_bb — Bollinger Band position
   413→   839→  const bbRange = bb.upper - bb.lower;
   414→   840→  const s_bb = bbRange > eps
   415→   841→    ? clamp((price - bb.lower) / bbRange, 0, 1)
   416→   842→    : 0.5;
   417→   843→
   418→   844→  // s_ma21 — Price relative to 21-period SMA
   419→   845→  const s_ma21 = ma21 > eps
   420→   846→    ? clamp(0.5 + (price - ma21) / ma21, 0, 1)
   421→   847→    : 0.5;
   422→   848→
   423→   849→  // s_ma100 — Price relative to 100-period SMA
   424→   850→  const s_ma100 = ma100 > eps
   425→   851→    ? clamp(0.5 + (price - ma100) / ma100, 0, 1)
   426→   852→    : 0.5;
   427→   853→
   428→   854→  // s_ema — EMA12 vs EMA26 spread
   429→   855→  const s_ema = ema26 > eps
   430→   856→    ? clamp(0.5 + (ema12 - ema26) / ema26 * 5, 0, 1)
   431→   857→    : 0.5;
   432→   858→
   433→   859→  // s_atr — Inverse volatility (lower ATR = higher score)
   434→   860→  const s_atr = price > eps ? 1 - Math.min(atrVal / price * 10, 1) : 0.5;
   435→   861→
   436→   862→  // s_trend — R² + angle combination
   437→   863→  const s_trend = (trendResult.r2 * 0.7) +
   438→   864→    (Math.max(0, Math.min(1, (trendResult.angle + 45) / 90)) * 0.3);
   439→   865→
   440→   866→  // s_sr — Support/Resistance proximity
   441→   867→  const dR1 = (sr.R1 - price) / (price > eps ? price : 1);
   442→   868→  const dS1 = (price - sr.S1) / (price > eps ? price : 1);
   443→   869→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
   444→   870→
   445→   871→  // ── Layer 2: Corrections (f_*) ───────────────────────────────────────────
   446→   872→
   447→   873→  // f_rsi — RSI with momentum correction
   448→   874→  const rsiCurrent = rsiVal;
   449→   875→  const rsi5ago = barIdx >= 5
   450→   876→    ? calcRSI(closes.slice(0, barIdx - 5 + 1))
   451→   877→    : rsiCurrent;
   452→   878→  const f_rsi = clamp(s_rsi + (rsiCurrent - rsi5ago) / 100 * 0.08, 0, 1);
   453→   879→
   454→   880→  // f_mfi — MFI with momentum correction (skip when no volume)
   455→   881→  const mfiCurrent = mfiVal;
   456→   882→  const mfi5ago = (barIdx >= 5 && sliceHasVolume)
   457→   883→    ? calcMFI(data.slice(0, barIdx - 5 + 1))
   458→   884→    : mfiCurrent;
   459→   885→  const f_mfi = sliceHasVolume
   460→   886→    ? clamp(s_mfi + (mfiCurrent - mfi5ago) / 100 * 0.08, 0, 1)
   461→   887→    : 0.5; // neutral when no volume
   462→   888→
   463→   889→  // f_cci — CCI with momentum correction (scale: /200 instead of /100)
   464→   890→  const cciCurrent = cciVal;
   465→   891→  const cci5ago = barIdx >= 5
   466→   892→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
   467→   893→    : cciCurrent;
   468→   894→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
   469→   895→
   470→   896→  // f_macd — Same as s_macd (no additional correction)
   471→   897→  const f_macd = s_macd;
   472→   898→
   473→   899→  // f_stoch — Same as s_stoch
   474→   900→  const f_stoch = s_stoch;
   475→   901→
   476→   902→  // f_stochCross — Stochastic K/D crossover detection
   477→   903→  let f_stochCross = 0.1;
   478→   904→  if (barIdx >= 1) {
   479→   905→    const prevSlice = data.slice(0, barIdx);
   480→   906→    const prevStoch = calcStochastic(prevSlice);
   481→   907→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
   482→   908→      f_stochCross = 0.9; // K crossed above D (bullish)
   483→   909→    }
   484→   910→  }
   485→   911→
   486→   912→  // f_macdCross — MACD/Signal crossover detection
   487→   913→  let f_macdCross = 0.1;
   488→   914→  if (barIdx >= 1) {
   489→   915→    const prevCloses = closes.slice(0, -1);
   490→   916→    if (prevCloses.length >= 35) {
   491→   917→      const prevMacd = calcMACD(prevCloses);
   492→   918→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
   493→   919→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
   494→   920→      }
   495→   921→    }
   496→   922→  }
   497→   923→
   498→   924→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
   499→   925→  const divResult = detectDivergenceSimple(data, barIdx);
   500→   926→  const f_div = divResult === true ? 0.85 : 0.15;
   501→   927→
   502→   928→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
   503→   929→  return {
   504→   930→    rsi: f_rsi,
   505→   931→    mfi: f_mfi,
   506→   932→    cci: f_cci,
   507→   933→    adx: s_adx,
   508→   934→    macd: f_macd,
   509→   935→    stoch: f_stoch,
   510→   936→    bb: s_bb,
   511→   937→    ma21: s_ma21,
   512→   938→    ma100: s_ma100,
   513→   939→    ema: s_ema,
   514→   940→    atr: s_atr,
   515→   941→    trend: s_trend,
   516→   942→    sr: s_sr,
   517→   943→    stochCross: f_stochCross,
   518→   944→    macdCross: f_macdCross,
   519→   945→    div: f_div,
   520→   946→  };
   521→   947→}
   522→   948→
   523→   949→// ═══════════════════════════════════════════════════════════════════════════════
   524→   950→// TRAINING DATA CREATION (Layer 0 — Data Preparation)
   525→   951→// ═══════════════════════════════════════════════════════════════════════════════
   526→   952→
   527→   953→/**
   528→   954→ * Creates supervised training data from historical OHLCV data.
   529→   955→ * For each bar, computes features and labels based on future price movement.
   530→   956→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
   531→   957→ * Samples every `sampleStep` bars to reduce computation.
   532→   958→ */
   533→   959→function createTrainingData(
   534→   960→  data: OHLCV[],
   535→   961→  horizon: number = 5,
   536→   962→  sampleStep: number = 5
   537→   963→): { X: number[][]; y: number[] } {
   538→   964→  const X: number[][] = [];
   539→   965→  const y: number[] = [];
   540→   966→
   541→   967→  // Start from horizon+30 to ensure enough data for indicators
   542→   968→  // End at data.length - horizon - 1 to have future bars for labeling
   543→   969→  // Sample every sampleStep bars for performance
   544→   970→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
   545→   971→    const features = computeFeaturesAtBar(data, i);
   546→   972→
   547→   973→    // Convert to array following FEATURES_ORDER
   548→   974→    const featureVec: number[] = [];
   549→   975→    let valid = true;
   550→   976→    for (const key of FEATURES_ORDER) {
   551→   977→      const val = features[key];
   552→   978→      if (val === undefined || val === null || Number.isNaN(val)) {
   553→   979→        valid = false;
   554→   980→        break;
   555→   981→      }
   556→   982→      featureVec.push(val);
   557→   983→    }
   558→   984→    if (!valid) continue;
   559→   985→
   560→   986→    // Label: 1 if future close > 1% above current close
   561→   987→    const currentClose = data[i].close;
   562→   988→    const futureClose = data[i + horizon].close;
   563→   989→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
   564→   990→
   565→   991→    X.push(featureVec);
   566→   992→    y.push(label);
   567→   993→  }
   568→   994→
   569→   995→  return { X, y };
   570→   996→}
   571→   997→
   572→   998→// ═══════════════════════════════════════════════════════════════════════════════
   573→   999→// VDss GRAPH STRUCTURE (Layer 5 — State Machine)
   574→  1000→// ═══════════════════════════════════════════════════════════════════════════════
   575→  1001→
   576→  1002→/**
   577→  1003→ * VDss Graph: Each node represents a market state.
   578→  1004→ * Edges represent transitions with types: 'up', 'down', 'pullback', 'risk', 'terminal'.
   579→  1005→ * Terminal nodes map to scenario results (R1=strong bullish, R5=deep correction).
   580→  1006→ */
   581→  1007→const VDSS_GRAPH = {
   582→  1008→  // outgoing edges from each node: [target, type]
   583→  1009→  outgoing: {
   584→  1010→    A: [['B', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
   585→  1011→    B: [['C', 'up'], ['G', 'pullback'], ['A', 'risk']] as [string, string][],
   586→  1012→    C: [['D', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
   587→  1013→    D: [['E', 'up'], ['B', 'pullback'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
   588→  1014→    E: [['F', 'up'], ['D', 'pullback'], ['G', 'pullback'], ['H', 'down'], ['R1', 'terminal']] as [string, string][],
   589→  1015→    F: [['R1', 'terminal'], ['E', 'pullback'], ['D', 'down'], ['G', 'risk']] as [string, string][],
   590→  1016→    G: [['B', 'up'], ['A', 'pullback'], ['H', 'down'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
   591→  1017→    H: [['G', 'pullback'], ['I', 'down'], ['B', 'up'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
   592→  1018→    I: [['H', 'pullback'], ['J', 'down'], ['G', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
   593→  1019→    J: [['I', 'pullback'], ['K', 'down'], ['H', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
   594→  1020→    K: [['J', 'pullback'], ['I', 'up'], ['L', 'risk'], ['R4', 'terminal'], ['R5', 'terminal']] as [string, string][],
   595→  1021→    L: [['C', 'up'], ['H', 'pullback'], ['J', 'down'], ['K', 'risk'], ['R5', 'terminal']] as [string, string][],
   596→  1022→  },
   597→  1023→  // which result scenario each terminal node maps to
   598→  1024→  terminalMap: { R1: 'R1', R2: 'R2', R3: 'R3', R4: 'R4', R5: 'R5' } as Record<string, string>,
   599→  1025→  startNode: 'A' as string,
   600→  1026→};
   601→  1027→
   602→  1028→// ═══════════════════════════════════════════════════════════════════════════════
   603→  1029→// LAYER 5 — Edge Weight Calculation
   604→  1030→// ═══════════════════════════════════════════════════════════════════════════════
   605→  1031→
   606→  1032→/**
   607→  1033→ * Calculates transition edge weights for the VDss graph.
   608→  1034→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
   609→  1035→ * to determine the relative probability of up/down/pullback/risk transitions.
   610→  1036→ */
   611→  1037→function calculateEdgeWeights(
   612→  1038→  bullConsensus: number,
   613→  1039→  adx: number,
   614→  1040→  mlModel: AdaptiveWeightModel
   615→  1041→): { up: number; down: number; pullback: number; risk: number } {
   616→  1042→  // Extract ML coefficient insights if model is trained
   617→  1043→  let trend_coef = 1 / 3;
   618→  1044→  let momentum_coef = 1 / 3;
   619→  1045→  let volatility_coef = 1 / 3;
   620→  1046→
   621→  1047→  if (mlModel.isTrained && mlModel.weights) {
   622→  1048→    const coefs = mlModel.getCoefficients();
   623→  1049→    // Map feature indices to coefficient categories
   624→  1050→    // FEATURES_ORDER: ['rsi','mfi','cci','adx','macd','stoch','bb','ma21','ma100','ema','atr','trend','sr','stochCross','macdCross','div']
   625→  1051→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER
   626→  1052→    const rsiIdx = 0;     // 'rsi' in FEATURES_ORDER
   627→  1053→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
   628→  1054→
   629→  1055→    const absTrend = Math.abs(coefs[trendIdx] ?? 0);
   630→  1056→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
   631→  1057→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
   632→  1058→
   633→  1059→    const coefSum = absTrend + absRsi + absAtr;
   634→  1060→    if (coefSum > 1e-10) {
   635→  1061→      trend_coef = absTrend / coefSum;
   636→  1062→      momentum_coef = absRsi / coefSum;
   637→  1063→      volatility_coef = absAtr / coefSum;
   638→  1064→    }
   639→  1065→  }
   640→  1066→
   641→  1067→  // Base weights driven by bull consensus
   642→  1068→  const up_base = bullConsensus * 0.7 + 0.15;
   643→  1069→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
   644→  1070→  const pullback_base = 0.25;
   645→  1071→  const risk_base = 0.12 * (1.2 - adx / 100);
   646→  1072→
   647→  1073→  // Apply ML coefficient adjustments
   648→  1074→  const up = clamp(
   649→  1075→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
   650→  1076→    0.05, 0.95
   651→  1077→  );
   652→  1078→  const down = clamp(
   653→  1079→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
   654→  1080→    0.05, 0.95
   655→  1081→  );
   656→  1082→  const pullback = clamp(
   657→  1083→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
   658→  1084→    0.05, 0.50
   659→  1085→  );
   660→  1086→  const risk = clamp(
   661→  1087→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
   662→  1088→    0.02, 0.30
   663→  1089→  );
   664→  1090→
   665→  1091→  return { up, down, pullback, risk };
   666→  1092→}
   667→  1093→
   668→  1094→// ═══════════════════════════════════════════════════════════════════════════════
   669→  1095→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)
   670→  1096→// ═══════════════════════════════════════════════════════════════════════════════
   671→  1097→
   672→  1098→/**
   673→  1099→ * Performs iterative DFS from the start node through the VDss graph,
   674→  1100→ * computing raw path probabilities, then calibrating them against
   675→  1101→ * Layer 4 scenario probabilities.
   676→  1102→ *
   677→  1103→ * Returns:
   678→  1104→ * - calibratedPaths: Map<pathString, probability>
   679→  1105→ * - calibrationFactors: per-scenario multiplier used
   680→  1106→ * - scenarioSums: raw probability sums per scenario before calibration
   681→  1107→ */
   682→  1108→function calculatePathProbabilities(
   683→  1109→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
   684→  1110→  scenarioProbs: Record<string, number>
   685→  1111→): {
   686→  1112→  calibratedPaths: Map<string, number>;
   687→  1113→  calibrationFactors: Record<string, number>;
   688→  1114→  scenarioSums: Record<string, number>;
   689→  1115→} {
   690→  1116→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
   691→  1117→  const MAX_PATHS = 200;
   692→  1118→
   693→  1119→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
   694→  1120→  type StackEntry = [string, string, number];
   695→  1121→
   696→  1122→  // Collect all terminal paths with raw probabilities
   697→  1123→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
   698→  1124→
   699→  1125→  // Iterative DFS
   700→  1126→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
   701→  1127→
   702→  1128→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
   703→  1129→    const [node, pathStr, pathProb] = stack.pop()!;
   704→  1130→
   705→  1131→    const edges = outgoing[node as keyof typeof outgoing];
   706→  1132→    if (!edges) continue;
   707→  1133→
   708→  1134→    // Normalize outgoing edge weights to probabilities for this node
   709→  1135→    // Map edge types to their weights
   710→  1136→    let totalWeight = 0;
   711→  1137→    const edgeWeightsList: number[] = [];
   712→  1138→    for (const [, edgeType] of edges) {
   713→  1139→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
   714→  1140→      edgeWeightsList.push(w);
   715→  1141→      totalWeight += w;
   716→  1142→    }
   717→  1143→
   718→  1144→    if (totalWeight < 1e-10) continue;
   719→  1145→
   720→  1146→    // Expand edges
   721→  1147→    for (let i = 0; i < edges.length; i++) {
   722→  1148→      const [target, edgeType] = edges[i];
   723→  1149→      const edgeProb = edgeWeightsList[i] / totalWeight;
   724→  1150→      const newProb = pathProb * edgeProb;
   725→  1151→      const newPath = pathStr + '→' + target;
   726→  1152→
   727→  1153→      if (edgeType === 'terminal') {
   728→  1154→        // Reached a terminal node
   729→  1155→        const scenario = terminalMap[target] ?? 'R3';
   730→  1156→        rawPaths.push({ path: newPath, scenario, prob: newProb });
   731→  1157→      } else {
   732→  1158→        // Continue DFS
   733→  1159→        stack.push([target, newPath, newProb]);
   734→  1160→      }
   735→  1161→    }
   736→  1162→  }
   737→  1163→
   738→  1164→  // Group raw probabilities by scenario
   739→  1165→  const rawSums: Record<string, number> = { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0 };
   740→  1166→  for (const rp of rawPaths) {
   741→  1167→    rawSums[rp.scenario] = (rawSums[rp.scenario] ?? 0) + rp.prob;
   742→  1168→  }
   743→  1169→
   744→  1170→  // Compute calibration factors: scenarioProbs[scenario] / rawSum[scenario]
   745→  1171→  const calibrationFactors: Record<string, number> = {};
   746→  1172→  for (const scenario of ['R1', 'R2', 'R3', 'R4', 'R5']) {
   747→  1173→    const rawSum = rawSums[scenario] ?? 0;
   748→  1174→    const targetProb = (scenarioProbs[scenario] ?? 20) / 100; // convert from percentage
   749→  1175→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
   750→  1176→  }
   751→  1177→
   752→  1178→  // Apply calibration to each path
   753→  1179→  const calibratedPaths = new Map<string, number>();
   754→  1180→  for (const rp of rawPaths) {
   755→  1181→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
   756→  1182→    calibratedPaths.set(rp.path, rp.prob * factor);
   757→  1183→  }
   758→  1184→
   759→  1185→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
   760→  1186→}
   761→  1187→
   762→  1188→// ═══════════════════════════════════════════════════════════════════════════════
   763→  1189→// LAYER 4 — Scenario Probability Calculation (Adaptive with ML)
   764→  1190→// ═══════════════════════════════════════════════════════════════════════════════
   765→  1191→
   766→  1192→/**
   767→  1193→ * Calculates the 5 VDss scenario probabilities (R1-R5) that sum to 100.
   768→  1194→ * Uses bull/bear consensus, key indicator values, S/R levels, and
   769→  1195→ * ML-derived adaptive factors for fine-tuning.
   770→  1196→ */
   771→  1197→function calculateScenarioProbabilities(
   772→  1198→  bullConsensus: number,
   773→  1199→  price: number,
   774→  1200→  R1: number,
   775→  1201→  S1: number,
   776→  1202→  MA100: number,
   777→  1203→  rsi: number,
   778→  1204→  mfi: number,
   779→  1205→  stochK: number,
   780→  1206→  mlModel: AdaptiveWeightModel
   781→  1207→): { R1: number; R2: number; R3: number; R4: number; R5: number; factors: { momentum: number; volatility: number; trend: number } } {
   782→  1208→  // Distance metrics to key levels (spec uses exp(-3 * ...))
   783→  1209→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0;
   784→  1210→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0;
   785→  1211→
   786→  1212→  // Binary risk flags (per spec) — MFI only used when volume is available
   787→  1213→  const overboughtRisk = (rsi > 70 || (mfi > 80 && mfi !== 50) || stochK > 80) ? 1 : 0;
   788→  1214→  const oversoldBounce = (rsi < 30 || (mfi < 20 && mfi !== 50) || stochK < 20) ? 1 : 0;

   790→  1216→  // Extract adaptive factors from ML model if trained
   791→  1217→  let momentum = 0.7;
   792→  1218→  let volatility = 0.5;
   793→  1219→  let trend = 0.6;
   794→  1220→
   795→  1221→  if (mlModel.isTrained && mlModel.weights) {
   796→  1222→    const coefs = mlModel.getCoefficients();
   797→  1223→    momentum = sigmoid(coefs[0] ?? 0);   // rsi coefficient → momentum factor
   798→  1224→    volatility = sigmoid(coefs[10] ?? 0); // atr coefficient → volatility factor
   799→  1225→    trend = sigmoid(coefs[11] ?? 0);     // trend coefficient → trend factor
   800→  1226→  }
   801→  1227→
   802→  1228→  // Below MA100 flag
   803→  1229→  const belowMA100 = price < MA100 ? 1 : 0;
   804→  1230→
   805→  1231→  // ── Raw scenario probabilities (matching user spec) ─────────────────────
   806→  1232→
   807→  1233→  // R1: Strong Bullish — aggressive breakout
   808→  1234→  const raw_R1 =
   809→  1235→    bullConsensus * distR1 * (1 - overboughtRisk * momentum * 0.7) +
   810→  1236→    oversoldBounce * 0.3 * distR1 * momentum;
   811→  1237→
   812→  1238→  // R2: Gradual Uptrend — steady rise
   813→  1239→  const raw_R2 =
   814→  1240→    ((1 - Math.abs(bullConsensus - 0.65)) * 0.8 +
   815→  1241→    (bullConsensus > 0.4 ? 0.2 : 0)) * trend;
   816→  1242→
   817→  1243→  // R3: Range-bound
   818→  1244→  const raw_R3 =
   819→  1245→    (1 - Math.abs(bullConsensus - 0.5)) * 0.6 * (1 + (1 - volatility) * 0.2);
   820→  1246→
   821→  1247→  // R4: Moderate Correction
   822→  1248→  const raw_R4 =
   823→  1249→    ((1 - bullConsensus) * distS1 * 0.7 +
   824→  1250→    overboughtRisk * 0.5 * momentum) *
   825→  1251→    (0.8 + 0.2 * (1 - trend));
   826→  1252→
   827→  1253→  // R5: Deep Correction
   828→  1254→  const raw_R5 =
   829→  1255→    ((1 - bullConsensus) ** 2 * 0.5 + belowMA100 * 0.2) *
   830→  1256→    (1 + (1 - trend) * 0.3);
   831→  1257→
   832→  1258→  // ── Normalize to sum = 100 ───────────────────────────────────────────
   833→  1259→  const sum = raw_R1 + raw_R2 + raw_R3 + raw_R4 + raw_R5;
   834→  1260→  const pR1 = sum > 0 ? Math.round(raw_R1 / sum * 100) : 20;
   835→  1261→  const pR2 = sum > 0 ? Math.round(raw_R2 / sum * 100) : 20;
   836→  1262→  const pR3 = sum > 0 ? Math.round(raw_R3 / sum * 100) : 20;
   837→  1263→  const pR4 = sum > 0 ? Math.round(raw_R4 / sum * 100) : 20;
   838→  1264→  const pR5 = 100 - (pR1 + pR2 + pR3 + pR4);

   841→  1267→    R1: pR1, R2: pR2, R3: pR3, R4: pR4, R5: pR5,
