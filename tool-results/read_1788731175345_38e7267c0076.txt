    22→    22→function toPersianNum(n: number, decimals: number = 0): string {
    23→    23→  if (!isFinite(n) || isNaN(n)) return '\u06f0';
    24→    24→  return n.toLocaleString('fa-IR', {
    25→    25→    maximumFractionDigits: decimals,
    26→    26→    minimumFractionDigits: 0,
    27→    27→  });
    28→    28→}
    29→    29→
    30→    30→function toFaPrice(n: number, decimals: number = 0): string {
    31→    31→  if (!isFinite(n) || isNaN(n)) return '\u06f0';
    32→    32→  return n.toLocaleString('fa-IR', {
    33→    33→    maximumFractionDigits: decimals,
    34→    34→    minimumFractionDigits: decimals > 0 ? Math.min(decimals, 2) : 0,
    35→    35→  });
    36→    36→}
    37→    37→function srGrade(strength: number): string {
    38→    38→  if (strength >= 8.5) return '\u0628\u0633\u06cc\u0627\u0631 \u0642\u0648\u06cc';
    39→    39→  if (strength >= 7) return '\u0642\u0648\u06cc';
    40→    40→  if (strength >= 5) return '\u0645\u062a\u0648\u0633\u0637';
    41→    41→  if (strength >= 3) return '\u0636\u0639\u06cc\u0641';
    42→    42→  return '\u0628\u0633\u06cc\u0627\u0631 \u0636\u0639\u06cc\u0641';
    43→    43→}

    56→    56→function computePriceHash(price: number): string {
    57→    57→  if (!price || price <= 0 || !isFinite(price)) return '0';
    58→    58→  // Round to nearest 0.5% bucket
    59→    59→  const bucket = Math.round(price * 2) / 2; // Round to nearest 0.5%
    60→    60→  return bucket.toString();
    61→    61→}

    66→    66→function buildMLInput(body: Record<string, unknown>): MLSelectorInput {
    67→    67→  const price = (body.currentPrice as number) || 0;
    68→    68→  const trendDir = (body.trendDirection as string) || 'range';
    69→    69→  const adx = (body.adx as number) || 0;
    70→    70→  const rsi = (body.rsi as number) || 50;
    71→    71→  const stochK = (body.stochK as number) || 50;
    72→    72→  const macdHist = (body.macdHist as number) || 0;
    73→    73→  const obv = (body.obv as number) || 0;
    74→    74→  const bollingerUpper = (body.bollingerUpper as number) || 0;
    75→    75→  const bollingerLower = (body.bollingerLower as number) || 0;
    76→    76→  const resistance = (body.resistanceStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
    77→    77→  const support = (body.supportStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
    78→    78→  const atr = (body.atr as number) || 0;
    79→    79→  const bbRange = bollingerUpper - bollingerLower;
    80→    80→  const bbPosition = bbRange > 0 ? Math.min(100, Math.max(0, Math.round((price - bollingerLower) / bbRange * 100))) : 50;
    81→    81→
    82→    82→  // Find dominant scenario across ALL SC1-SC9
    83→    83→  const scenarios = body.scenarios as Record<string, { probability: number }> | undefined;
    84→    84→  let dominantKey = 'SC5';
    85→    85→  let dominantProb = 0;
    86→    86→  for (const k of ['SC1', 'SC2', 'SC3', 'SC4', 'SC5', 'SC6', 'SC7', 'SC8', 'SC9'] as const) {
    87→    87→    const p = scenarios?.[k]?.probability ?? 0;
    88→    88→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
    89→    89→  }
    90→    90→
    91→    91→  return {
    92→    92→    price,
    93→    93→    trend: trendDir === 'up' ? 'up' : trendDir === 'down' ? 'down' : 'range',
    94→    94→    adx, diPlus: (body.diPlus as number) || 0, diMinus: (body.diMinus as number) || 0,
    95→    95→    rsi, stochK, macdHist, obv, bbPosition, resistance, support, atr,
    96→    96→    scenarioDominant: dominantKey,
    97→    97→    hasVolume: (body.hasVolume as boolean) ?? false,
    98→    98→  };
    99→    99→}

   102→   102→function buildMSLV4Context(body: Record<string, unknown>): MSLV4Context {
   103→   103→  const price = (body.currentPrice as number) || 0;
   104→   104→  const atr = (body.atr as number) || 0;
   105→   105→  const atrRatio = price > 0 ? atr / price : 0;
   106→   106→
   107→   107→  const v11Probs = body.v11Probabilities as {
   108→   108→    scenarios: Array<{ key: string; name: string; rawProbability: number }>;
   109→   109→  } | undefined;
   110→   110→  let dominantScenario = '';
   111→   111→  if (v11Probs?.scenarios) {
   112→   112→    const sorted = [...v11Probs.scenarios].sort((a, b) => b.rawProbability - a.rawProbability);
   113→   113→    dominantScenario = sorted[0]?.name ?? '';
   114→   114→  } else {
   115→   115→    const scenarios = body.scenarios as Record<string, { name?: string; probability: number }> | undefined;
   116→   116→    let maxProb = 0;
   117→   117→    for (const k of ['SC1', 'SC2', 'SC3', 'SC4', 'SC5', 'SC6', 'SC7', 'SC8', 'SC9']) {
   118→   118→      const p = scenarios?.[k]?.probability ?? 0;
   119→   119→      if (p > maxProb) { maxProb = p; dominantScenario = scenarios?.[k]?.name ?? k; }
   120→   120→    }
   121→   121→  }
   122→   122→
   123→   123→  const trendDir = (body.trendDirection as string) || 'range';
   124→   124→  const isBull = trendDir === 'up';
   125→   125→  const isBear = trendDir === 'down';
   126→   126→  const adx = (body.adx as number) || 0;
   127→   127→  const trendStrength = adx > 40 ? 0.9 : adx > 25 ? 0.7 : adx > 15 ? 0.4 : 0.15;
   128→   128→
   129→   129→  // Use regime engine result if available from ta-engine, otherwise fallback to heuristic
   130→   130→  const regimeResultFromEngine = body.regimeResult as { regime: RegimeState; confidence: number } | undefined;
   131→   131→  let regime: MSLV4Context['regime'] = 'Range';
   132→   132→  if (regimeResultFromEngine?.regime) {
   133→   133→    regime = toMSLRegime(regimeResultFromEngine.regime, regimeResultFromEngine.confidence);
   134→   134→  } else {
   135→   135→    // Legacy fallback: basic heuristic from ADX + direction
   136→   136→    if (isBull && trendStrength > 0.6) regime = 'Strong Bull';
   137→   137→    else if (isBull) regime = 'Weak Bull';
   138→   138→    else if (isBear && trendStrength > 0.6) regime = 'Strong Bear';
   139→   139→    else if (isBear) regime = 'Weak Bear';
   140→   140→  }
   141→   141→
   142→   142→  const instrumentType = (body.instrumentType as string) || (body.symbolName as string) || 'stock';
   143→   143→  const ta = body.technicalAnalysis as {
   144→   144→    classicPatternCount?: number; harmonicPatternCount?: number;
   145→   145→    elliottWaveCount?: number; divergenceCount?: number;
   146→   146→  } | undefined;
   147→   147→  const overallConfidence = (body.overallConfidence as number) ?? (body.confidence as number) ?? 0.6;
   148→   148→
   149→   149→  // Extract probability trend signals for MSL selection
   150→   150→  const probTrend = extractProbTrendSignals(body);
   151→   151→
   152→   152→  return {
   153→   153→    asset: instrumentType, timeframe: (body.timeframe as string) || 'daily',
   154→   154→    regime, confidence: Math.max(0, Math.min(1, overallConfidence)),
   155→   155→    volatility: atrRatio, trend_strength: trendStrength, dominantScenario,
   156→   156→    pattern_counts: { classical: ta?.classicPatternCount ?? 0, harmonic: ta?.harmonicPatternCount ?? 0, elliott: ta?.elliottWaveCount ?? 0 },
   157→   157→    divergence_present: (ta?.divergenceCount ?? 0) > 0,
   158→   158→    hasVolume: (body.hasVolume as boolean) ?? false,
   159→   159→    rsi: (body.rsi as number) || 50, adx,
   160→   160→    audience_level: (body.audienceLevel as string) || 'pro',
   161→   161→    probTrend,
   162→   162→  };
   163→   163→}

   165→   165→// ─── Extract probability trend signals from body ──────────────────
   166→   166→function extractProbTrendSignals(body: Record<string, unknown>): MSLV4Context['probTrend'] {
   167→   167→  const pt = body.probabilityTrend as {
   168→   168→    scenarios?: { scenarioKey: string; group: string; trendDirection: string; trend: { individualProb: number; cumulativeProb: number }[] }[];
   169→   169→    groups?: { group: string; trendDirection: string; trend: { cumulativeProb: number }[] }[];
   170→   170→  } | undefined;
   171→   171→  if (!pt?.groups || pt.groups.length === 0) return undefined;
   172→   172→
   173→   173→  const findGroup = (g: string) => pt.groups!.find(x => x.group === g);
   174→   174→  const bullG = findGroup('bullish');
   175→   175→  const bearG = findGroup('bearish');
   176→   176→  const neutralG = findGroup('neutral');
   177→   177→
   178→   178→  // Find dominant scenario trend
   179→   179→  const scenarios = body.scenarios as Record<string, { probability: number }> | undefined;
   180→   180→  let domKey = 'SC5'; let domProb = 0;
   181→   181→  for (const k of ['SC1','SC2','SC3','SC4','SC5','SC6','SC7','SC8','SC9']) {
   182→   182→    const p = scenarios?.[k]?.probability ?? 0;
   183→   183→    if (p > domProb) { domProb = p; domKey = k; }
   184→   184→  }
   185→   185→  const domScenario = pt.scenarios?.find(s => s.scenarioKey === domKey);
   186→   186→
   187→   187→  // 7-day change (day 1 = today, day 7 = 7 days ago)
   188→   188→  const get7dChange = (trend: { cumulativeProb: number }[]) => {
   189→   189→    if (!trend || trend.length < 7) return 0;
   190→   190→    return trend[0].cumulativeProb - trend[6].cumulativeProb;
   191→   191→  };
   192→   192→
   193→   193→  return {
   194→   194→    bullGroupDir: bullG?.trendDirection as NonNullable<MSLV4Context['probTrend']>['bullGroupDir'],
   195→   195→    bearGroupDir: bearG?.trendDirection as NonNullable<MSLV4Context['probTrend']>['bearGroupDir'],
   196→   196→    neutralGroupDir: neutralG?.trendDirection as NonNullable<MSLV4Context['probTrend']>['neutralGroupDir'],
   197→   197→    bullGroupCum: bullG?.trend?.[0]?.cumulativeProb,
   198→   198→    bearGroupCum: bearG?.trend?.[0]?.cumulativeProb,
   199→   199→    neutralGroupCum: neutralG?.trend?.[0]?.cumulativeProb,
   200→   200→    dominantCumDir: domScenario?.trendDirection as NonNullable<MSLV4Context['probTrend']>['dominantCumDir'],
   201→   201→    dominantIndivDir: domScenario?.trendDirection as NonNullable<MSLV4Context['probTrend']>['dominantIndivDir'],
   202→   202→    bullGroupChange7d: get7dChange(bullG?.trend ?? []),
   203→   203→    bearGroupChange7d: get7dChange(bearG?.trend ?? []),
   204→   204→    neutralGroupChange7d: get7dChange(neutralG?.trend ?? []),
   205→   205→  };
   206→   206→}

   451→   451→24. **دقت عددی مطلق — تطابق با گراف تصمیم:** تحلیل شما باید کاملاً بر اساس داده‌های گراف تصمیم باشد. هر عددی که در متن می‌آورید (قیمت، درصد، سطح حمایت/مقاومت) باید **دقیقاً** با همان عددی که در داده‌های پایه ارائه شده مطابقت داشته باشد. **ممنوعیت مطلق:** هرگز عددی را از خودتان نسازید، تخمین نزنید، یا گرد نکنید. مثلاً اگر قیمت فعلی ۲,۰۷۶,۹۵۰ ریال است، نباید بنویسید ۲,۰۰۰,۰۰۰ یا ۲,۱۰۰,۰۰۰. اگر سطح مقاومت ۲,۱۸۰,۰۰۰ است، همان را بنویسید. اگر سناریوی صعودی ۳۵ درصد احتمال دارد، همان ۳۵ درصد را بنویسید.
   452→   452→25. **جهت اعداد:** تمام اعداد در متن فارسی باید از چپ به راست نوشته شوند. هرگز اعداد را برعکس یا از راست به چپ ننویسید. مثلاً «۱,۲۳۴» نه «۴,۳۲۱».

   454→   454→27. **منع تناقض:** هیچ جمله‌ای در تحلیل نباید با داده‌های ارائه‌شده تناقض داشته باشد. اگر داده‌ها نشان‌دهنده روند صعودی هستند، تحلیل نباید نزولی باشد و بالعکس. اگر احتمال صعودی بیشتر است، تحلیل باید بازتاب‌دهنده این اولویت باشد.
   455→   455→28. **حفظ مقیاس قیمت — ممنوعیت تغییر مقیاس:** تمام قیمت‌ها (میانگین متحرک، باند بولینگر، اهداف سناریو، حد ضرر، حمایت، مقاومت) باید در همان مقیاس قیمت فعلی باشند. اگر قیمت فعلی ۲,۱۴۰,۰۰۰ ریال است، میانگین متحرک باید حدود ۱,۸۰۰,۰۰۰ تا ۲,۱۰۰,۰۰۰ ریال باشد (نه ۳۰,۰۰۰ یا ۲۴,۰۰۰). هرگز صفرهای قیمت را حذف نکنید. هرگز مقیاس قیمت را تغییر ندهید.

   458→   458→// ─── POST Handler ────────────────────────────────────────────────
   459→   459→export async function POST(req: NextRequest) {
   460→   460→  const startTime = Date.now();
   461→   461→  try {
   462→   462→    const body = await req.json();
   463→   463→
   464→   464→    // ─── Input validation ─────────────────────────────────────────
   465→   465→    if (!body.currentPrice || typeof body.currentPrice !== 'number' || body.currentPrice <= 0) {
   466→   466→      return NextResponse.json({ error: 'currentPrice باید یک عدد مثبت باشد.' }, { status: 400 });
   467→   467→    }
   468→   468→    if (!body.symbolName || typeof body.symbolName !== 'string' || body.symbolName.trim().length === 0) {
   469→   469→      return NextResponse.json({ error: 'symbolName باید یک رشته غیرخالی باشد.' }, { status: 400 });
   470→   470→    }
   471→   471→    // Validate at least some key indicator values are present
   472→   472→    const indicatorKeys = ['rsi', 'adx', 'macdHist', 'stochK', 'bollingerUpper', 'bollingerLower'];
   473→   473→    const presentIndicators = indicatorKeys.filter(k => body[k] !== undefined && body[k] !== null);
   474→   474→    if (presentIndicators.length === 0) {
   475→   475→      return NextResponse.json({ error: 'حداقل چند اندیکاتور فنی باید ارائه شود.' }, { status: 400 });
   476→   476→    }
   477→   477→
   478→   478→    const symbolName = String(body.symbolName || 'unknown');
   479→   479→    const today = getTodayDateStr();
   480→   480→    const currentPrice = Number(body.currentPrice) || 0;
   481→   481→    const priceHash = computePriceHash(currentPrice);
   482→   482→    const forceRefresh = Boolean(body.forceRefresh);
   483→   483→
   484→   484→    // 1. Check time-based cache (Prisma/SQLite)
   485→   485→    // Cache key is (symbol, priceHash) — allows multiple analyses per day
   486→   486→    // when price moves significantly. Cache expires after CACHE_TTL_MS (5 min).
   487→   487→    if (!forceRefresh) {
   488→   488→      try {
   489→   489→        const cached = await db.aiAnalysisCache.findUnique({
   490→   490→          where: { symbol_priceHash: { symbol: symbolName, priceHash } },
   491→   491→        });
   492→   492→        if (cached && cached.text) {
   493→   493→          const ageMs = Date.now() - cached.updatedAt.getTime();
   494→   494→          const isExpired = ageMs > CACHE_TTL_MS;
   495→   495→          if (!isExpired) {
   496→   496→            console.log(`[AI] Cache HIT for ${symbolName} (priceHash=${priceHash}, age=${Math.round(ageMs / 1000)}s) [${Date.now() - startTime}ms]`);
   497→   497→            return NextResponse.json({
   498→   498→              text: cached.text,
   499→   499→              ml: cached.ml ? JSON.parse(cached.ml) : undefined,
   500→   500→              cached: true,
   501→   501→              cachedAge: Math.round(ageMs / 1000),
   502→   502→              priceAtGeneration: cached.price,
   503→   503→            });
   504→   504→          } else {
   505→   505→            console.log(`[AI] Cache EXPIRED for ${symbolName} (priceHash=${priceHash}, age=${Math.round(ageMs / 1000)}s > ${CACHE_TTL_MS / 1000}s), re-generating`);
   506→   506→          }
   507→   507→        }
   508→   508→      } catch (dbErr) {
   509→   509→        console.warn('[AI] DB cache read failed, continuing:', dbErr instanceof Error ? dbErr.message : dbErr);
   510→   510→      }
   511→   511→    } else {
   512→   512→      console.log(`[AI] Force refresh requested for ${symbolName}, skipping cache`);
   513→   513→    }
   514→   514→
   515→   515→    // 2. ML selection
   516→   516→    const mlInput = buildMLInput(body);
   517→   517→    const mlSelection = selectMLCombination(mlInput);

   519→   519→

   524→   524→    mlSelection.style = mslResult.analysis_style.primary.id as typeof mlSelection.style;

   526→   526→    mlSelection.reasoning = `MSLv4: ${mslResult.school_of_analysis.primary.id}/${mslResult.analysis_style.primary.id}/${mslResult.analysis_tone.primary.id}`;

   529→   529→    const mslSystemPrompt = buildMSLV4PromptSection(mslResult);
   530→   530→    const userMessage = buildPrompt(body, mlSelection, methods);
   531→   531→    const dynamicSystemPrompt = `${SYSTEM_PROMPT}\n\n${mslSystemPrompt}`;
   532→   532→
   533→   533→    console.log(`[AI] Generating for ${symbolName} (cache miss) [${Date.now() - startTime}ms]`);
   534→   534→

   538→   538→    const content = await dedicatedAIChatCompletion(

   541→   541→        { role: 'user', content: userMessage },
   542→   542→      ],
   543→   543→      {

   545→   545→        maxRetries: 2,
   546→   546→      }
   547→   547→    );

   552→   552→

   554→   554→    if (!priceValid) {

   556→   556→    }
   557→   557→    const mlData = { school: mlSelection.school, style: mlSelection.style, tone: mlSelection.tone, reasoning: mlSelection.reasoning, methods };
   558→   558→
   559→   559→    // 7. Save to time-based cache — ONLY if prices are valid
   560→   560→    // Cache key is (symbol, priceHash) — allows multiple entries per day
   561→   561→    if (priceValid) {
   562→   562→      try {

   564→   564→          where: { symbol_priceHash: { symbol: symbolName, priceHash } },

   570→   570→            price: currentPrice,

   575→   575→            text: cleaned,
   576→   576→            ml: JSON.stringify(mlData),
   577→   577→            price: currentPrice,
   578→   578→          },
   579→   579→        });

   581→   581→      } catch (dbErr) {

   583→   583→      }
   584→   584→    } else {
   585→   585→      console.log(`[AI] Skipped cache save for ${symbolName} due to price hallucination`);
   586→   586→    }
   587→   587→
   588→   588→    // 8. Clean up old cache entries for this symbol (keep only last 3 per symbol)
   589→   589→    try {
   590→   590→      const allEntries = await db.aiAnalysisCache.findMany({
   591→   591→        where: { symbol: symbolName },
   592→   592→        orderBy: { updatedAt: 'desc' },
   593→   593→        select: { id: true },
   594→   594→      });
   595→   595→      if (allEntries.length > 3) {
   596→   596→        const idsToDelete = allEntries.slice(3).map(e => e.id);
   597→   597→        await db.aiAnalysisCache.deleteMany({ where: { id: { in: idsToDelete } } });
   598→   598→      }
   599→   599→    } catch { /* ignore cleanup failure */ }
   600→   600→
   601→   601→    console.log(`[AI] Complete for ${symbolName} [${Date.now() - startTime}ms]`);
   602→   602→    return NextResponse.json({ text: cleaned, ml: mlData });
   603→   603→
   604→   604→  } catch (err) {

   606→   606→    const errorType = err instanceof Error ? err.constructor.name : 'Unknown';

   608→   608→    if (err instanceof Error && err.stack) {
   609→   609→      console.error('[AI] Stack trace:', err.stack);
   610→   610→    }
   611→   611→    return NextResponse.json({ error: userFriendlyError(err), text: '' }, { status: 500 });
   612→   612→  }

   615→   615→function userFriendlyError(err: unknown): string {

   617→   617→  if (msg.includes('Rate limited') || msg.includes('429') || msg.includes('ZAI_RATE_LIMITED')) {

   619→   619→  }
   620→   620→  if (msg.includes('Timed out') || msg.includes('timeout') || msg.includes('Timeout')) {
   621→   621→    return 'زمان پاسخدهی هوشمند به پایان رسید. این مشکل معمولاً موقتی است — لطفاً دوباره تلاش کنید.';
   622→   622→  }
