    87→const toFa = (n: number) => {
    88→  const rounded = Math.round(n);
    89→  if (!isFinite(rounded) || isNaN(rounded)) return '۰';
    90→  return rounded.toLocaleString('fa-IR');
    91→};
    92→const toFaD = (n: number, d: number = 1) => toPersianDigits(n.toFixed(d));
    93→// Grouped Persian digits with controlled decimal places
    94→const toFaG = (n: number | null | undefined, d: number = 0) => n != null ? toFaGrouped(Number(n.toFixed(d))) : '—';
    95→

   100→  R1: { border: '#59e39b', badgeBg: 'rgba(89,227,155,0.2)', badgeColor: '#59e39b' },
   101→  R2: { border: '#ffb25f', badgeBg: 'rgba(255,178,95,0.2)', badgeColor: '#ffb25f' },

   108→// Component
   109→// ═══════════════════════════════════════════════════════════════════════════════
   110→
   111→export default function VdesAnalysis(props: VdesAnalysisProps) {
   112→  const {
   113→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
   114→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
   115→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
   116→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
   117→    supportStrengths, resistanceStrengths, priceTargets,
   118→    isIndex, isTgju, currencyUnit,
   119→    edgeWeights,
   120→  } = props;
   121→
   122→  const unit = currencyUnit || 'ریال';
   123→
   124→  // ── Instrument Context (for intelligent naming) ─────────────────
   125→  const instrumentLabel = useMemo(() => {
   126→    if (isIndex) return { name: `شاخص ${symbolName}`, type: 'index' as const, verb: 'قرار دارد' };
   127→    if (isTgju) return { name: symbolName, type: 'tgju' as const, verb: 'قرار دارد' };
   128→    return { name: `سهم ${symbolName}`, type: 'stock' as const, verb: 'قرار دارد' };
   129→  }, [symbolName, isIndex, isTgju]);
   130→
   131→  const IL = instrumentLabel;
   132→
   133→  const containerRef = useRef<HTMLDivElement>(null);
   134→  const textAnalysisRef = useRef<HTMLDivElement>(null);
   135→  const [exporting, setExporting] = useState(false);
   136→
   137→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   138→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   139→  const S1_level = supports[0] ?? currentPrice * 0.95;
   140→  const S2_level = supports[1] ?? currentPrice * 0.90;
   141→  const S3_level = supports[2] ?? currentPrice * 0.85;
   142→
   143→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   144→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   145→
   146→  // ── Dominant scenario ──────────────────────────────────────────
   147→  let highestKey = 'R3';
   148→  let highestProb = 0;
   149→  for (const key of SCENARIO_KEYS) {
   150→    if (scenarios[key].probability > highestProb) {
   151→      highestProb = scenarios[key].probability;
   152→      highestKey = key;
   153→    }
   154→  }
   155→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   156→
   157→  // ── Dynamic scenario labels (context-aware based on overallSignal) ─
   158→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal), [overallSignal]);
   159→  const strategy = dynamicMeta[highestKey];
   160→
   161→  // ── Text variety: hash-based selector for varied language ──────────
   162→  const varietySeed = useMemo(() => {
   163→    const raw = `${rsi}${macdHist}${cci}${adx}${stochK}${trendAngle}${Math.round(currentPrice)}`;
   164→    let hash = 0;
   165→    for (let i = 0; i < raw.length; i++) hash = ((hash << 5) - hash + raw.charCodeAt(i)) | 0;
   166→    return Math.abs(hash);
   167→  }, [rsi, macdHist, cci, adx, stochK, trendAngle, currentPrice]);
   168→
   169→  // ── RSI signal ─────────────────────────────────────────────────
   170→  const rsiZone = rsi > 70 ? 'overbought' : rsi > 60 ? 'weak_ob' : rsi > 40 ? 'neutral' : rsi > 30 ? 'weak_os' : 'oversold';
   171→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   172→  const rsiColor = rsi > 70 ? 'text-red-700' : rsi > 60 ? 'text-amber-800' : rsi > 40 ? 'text-gray-600' : rsi > 30 ? 'text-amber-800' : 'text-emerald-700';
   173→
   174→  // ── Stochastic signal ──────────────────────────────────────────
   175→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   176→
   177→  // ── Trend text ─────────────────────────────────────────────────
   178→  const isUp = trendDirection === 'up';
   179→  const isDown = trendDirection === 'down';
   180→  const trendLabel = isUp ? 'صعودی' : isDown ? 'نزولی' : 'خنثی';
   181→  const trendColorCls = isUp ? 'text-emerald-700' : isDown ? 'text-red-700' : 'text-amber-800';
   182→  const adxStrength = adx > 40 ? 'بسیار قوی' : adx > 25 ? 'قوی' : adx > 15 ? 'متوسط' : 'ضعیف';
   183→  const adxColor = adx > 40 ? 'text-emerald-700' : adx > 25 ? 'text-amber-800' : 'text-gray-500';
   184→  const r2Pct = (trendR2 * 100).toFixed(1);
   185→
   186→  // ── Bollinger position ─────────────────────────────────────────
   187→  const bbRange = bollingerUpper - bollingerLower;
   188→  const bbPos = bbRange > 0 ? toPersianDigits(((currentPrice - bollingerLower) / bbRange * 100).toFixed(0)) : '۵۰';
   189→  const bbAbove = currentPrice > bollingerUpper;
   190→  const bbBelow = currentPrice < bollingerLower;
   191→  const bbSignal = bbAbove ? 'بالای باند بالایی' : bbBelow ? 'زیر باند پایینی' : `داخل باندها (${bbPos}٪)`;
   192→
   193→  // ── Distance to MA ─────────────────────────────────────────────
   194→  const distMa21 = ((currentPrice - ma21) / ma21 * 100).toFixed(1);
   195→  const distMa100 = ((currentPrice - ma100) / ma100 * 100).toFixed(1);
   196→
   197→  // ── TradingView scenario colors ────────────────────────────────
   198→  const chartCandles = useMemo(() => {
   199→    if (candles.length <= 2000) return candles;
   200→    return candles.slice(-2000);
   201→  }, [candles]);
   202→
   203→  const tvScenarios = useMemo(() => {
   204→    const colors: Record<string, string> = { R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62' };
   205→    return Object.fromEntries(
   206→      SCENARIO_KEYS.map(k => [k, { targetMin: scenarios[k].targetMin, targetMax: scenarios[k].targetMax, name: scenarios[k].name, probability: scenarios[k].probability, color: colors[k] }])
   207→    );
   208→  }, [scenarios]);
   209→
   210→  // ── VDss path probabilities (for result nodes section) ──────────
   211→  const VDSS_COLORS: Record<string, string> = {
   212→    R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62',
   213→  };
   214→
   215→  const vdssPathsInfo = useMemo(() => {
   216→    if (!edgeWeights) return null;
   217→
   218→    const edgeProbs = calcEdgeProbabilities(edgeWeights);
   219→    const allPathsRaw = findAllPaths(edgeProbs);
   220→
   221→    // Calibrate path probabilities to match TA engine scenario probabilities
   222→    const rawSums: Record<string, number> = {};
   223→    for (const key of SCENARIO_KEYS) rawSums[key] = 0;
   224→    for (const p of allPathsRaw) rawSums[p.target] += p.prob;
   225→
   226→    const allPaths: PathInfo[] = allPathsRaw.map(p => {
   227→      const rawSum = rawSums[p.target] || 0.0001;
   228→      const targetProb = scenarios[p.target as keyof typeof scenarios].probability / 100;
   229→      const scale = targetProb / rawSum;
   230→      return { ...p, prob: p.prob * scale };
   231→    });
   232→
   233→    // Find top path per scenario
   234→    const topPaths: Record<string, PathInfo | null> = {};
   235→    for (const key of SCENARIO_KEYS) {
   236→      const best = allPaths.filter(p => p.target === key).sort((a, b) => b.prob - a.prob)[0] ?? null;
   237→      topPaths[key] = best;
   238→    }
   239→
   240→    return { topPaths };
   241→  }, [edgeWeights, scenarios]);
   242→
   243→  // ── Strategy recommendation text (intelligent) ────────────────────
   244→  const mfiWarning = hasVolume ? ` (مقدار ${toFa(mfi)})` : '';
   245→  const strategyText = useMemo(() => {
   246→    const v = varietySeed % 3; // 3 variants per signal type
   247→    if (rsi > 70 || (hasVolume && mfi > 80)) {
   248→      const variants = [
   249→        `هشدار اشباع خرید فعال است (RSI: ${toFaG(rsi, 1)}${mfiWarning}). فاصله تا مقاومت اول ${toFaG(R1_level)} ${unit} است. بهترین استراتژی در این شرایط، صبر برای اصلاح قیمت تا محدوده حمایت ${toFaG(S1_level)} تا ${toFaG(S2_level)} و ورود در نقطه بهینه‌تر می‌باشد.`,
   250→        `با توجه به ورود به منطقه اشباع خرید (RSI=${toFaG(rsi, 1)}${mfiWarning}) و نزدیکی به مقاومت ${toFaG(R1_level)}، ریسک اصلاح کوتاه‌مدت بالا ارزیابی می‌شود. پیشنهاد: انتظار برای بازگشت قیمت به ${toFaG(S1_level)}-${toFaG(S2_level)} و سپس بررسی مجدد سیگنال‌ها.`,
   251→        `${IL.name} در محدوده اشباع خرید (RSI: ${toFaG(rsi, 1)}${mfiWarning}) ${IL.verb}. با در نظر گرفتن مقاومت ${toFaG(R1_level)} در بالا و حمایت ${toFaG(S1_level)} در پایین، ورود جدید در این نقطه توصیه نمی‌شود. بهتر است منتظر عقبنشینی به ${toFaG(S2_level)} بمانید.`,
   252→      ];
   253→      return variants[v];
   254→    }
   255→    if (rsi < 30 || (hasVolume && mfi < 20)) {
   256→      const variants = [
   257→        `اشباع فروش قابل توجه (RSI: ${toFaG(rsi, 1)}${mfiWarning}) در محدوده حمایت ${toFaG(S1_level)} شناسایی شد. این شرایط می‌تواند فرصت مناسبی برای ورود باشد، با حد ضرر ${toFaG(S2_level)}. هدف اول ${toFaG(R1_level)} و هدف دوم ${toFaG(R2_level)} ${unit}.`,
   258→        `${IL.name} به منطقه اشباع فروش رسیده (RSI=${toFaG(rsi, 1)}${mfiWarning}). فاصله تا حمایت کلیدی ${toFaG(S1_level)} کم است. اگر واکنش مثبت به این حمایت مشاهده شود، بازگشت به ${toFaG(R1_level)} محتمل خواهد بود.`,
   259→        `شاخص قدرت نسبی در محدوده اشباع فروش (مقدار ${toFaG(rsi, 1)}${mfiWarning}) ${IL.verb}. نزدیکی به حمایت ${toFaG(S1_level)} و سیگنال فروش بیش از حد، احتمال بازگشت کوتاه‌مدت را تقویت می‌کند. حد ضرر پیشنهادی: ${toFaG(S2_level)}.`,
   260→      ];
   261→      return variants[v];
   262→    }
   263→    const variants = [
   264→      `سیگنال‌ها در محدوده خنثی قرار دارند (RSI: ${toFaG(rsi, 1)}، قدرت روند: ${adxStrength}). محدوده رنج فعلی بین ${toFaG(S1_level)} و ${toFaG(R1_level)} است. تا خروج قیمت از این کانال، بهترین اقدام نظاره و عدم ورود است.`,
   265→      `وضعیت فعلی ${IL.name} نشان‌دهنده عدم قطعیت (RSI=${toFaG(rsi, 1)}، ADX=${toFaG(adx, 1)}). ${IL.name} در بازه ${toFaG(S1_level)} تا ${toFaG(R1_level)} در نوسان است. شکست معتبر یکی از این سطوح، مسیر آینده را مشخص خواهد کرد.`,
   266→      `شاخص‌ها هم‌جهت نیستند (RSI: ${toFaG(rsi, 1)}) و قدرت روند ${adxStrength} ارزیابی می‌شود. تا زمانی که ${IL.name} از محدوده ${toFaG(S1_level)}-${toFaG(R1_level)} خارج نشود، ورود با ریسک بالا همراه خواهد بود.`,
   267→    ];
   268→    return variants[v];

   270→

   274→
   275→  // ── Helper: capture container as high-res image (hides toolbar) ──

   277→    const el = containerRef.current;
   278→    if (!el) return null;

   280→    const toolbar = el.querySelector('[data-export-toolbar]') as HTMLElement | null;
   281→    if (toolbar) toolbar.style.visibility = 'hidden';

   283→      const { toPng } = await import('html-to-image');
   284→      return await toPng(el, { backgroundColor: '#ffffff', pixelRatio: 2, quality: 0.95 });

   286→      if (toolbar) toolbar.style.visibility = '';

   289→
   290→  // ── Export PDF (full page) ─────────────────────────────────────
   291→  const handleExportPDF = useCallback(async () => {
   292→    setExporting(true);
   293→    try {
   294→      const dataUrl = await captureContainer();
   295→      if (!dataUrl) return;
   296→      const { jsPDF } = await import('jspdf');
   297→      const img = new Image();
   298→      img.src = dataUrl;
   299→      await new Promise((resolve) => { img.onload = resolve; });

   301→      const pdfHeight = (img.height * pdfWidth) / img.width;
   302→      const pdf = new jsPDF({ orientation: pdfHeight > pdfWidth ? 'portrait' : 'landscape', unit: 'pt', format: [pdfWidth, Math.min(pdfHeight, 841.89)] });

   312→    try {

   314→      if (!dataUrl) return;
   315→      const now = new Date();

   317→      const base64 = dataUrl.replace(/^data:image\/png;base64,/, '');
   318→      const html = `<!DOCTYPE html>
   319→<html lang="fa" dir="rtl">
   320→<head>
   321→<meta charset="UTF-8">
   322→<meta name="viewport" content="width=device-width, initial-scale=1.0">
   323→<title>تحلیل تکنیکال ${IL.name}</title>
   324→<style>
   325→  * { margin: 0; padding: 0; box-sizing: border-box; }
   326→  body {
   327→    background: #ffffff;
   328→    display: flex;
   329→    justify-content: center;

   331→    min-height: 100vh;
   332→  }
   333→  .page-wrapper {
   334→    width: 100%;
   335→    max-width: 960px;
   336→  }
   337→  .page-wrapper img {
   338→    width: 100%;
   339→    height: auto;
   340→    display: block;
   341→    border-radius: 16px;
   342→  }
   343→  .footer {
   344→    text-align: center;
   345→    font-family: Tahoma, Arial, sans-serif;
   346→    font-size: 11px;
   347→    color: #6b7280;
   348→    margin-top: 16px;
   349→    padding-top: 12px;
   350→    border-top: 1px solid rgba(0,0,0,0.1);
   351→  }
   352→  @media print {
   353→    body { padding: 0; background: white; }
   354→    .footer { color: #999; }
   355→  }
   356→</style>
   357→</head>
   358→<body>
   359→<div class="page-wrapper">
   360→  <img src="data:image/png;base64,${base64}" alt="تحلیل تکنیکال ${IL.name}" />

   362→</div>
   363→</body>
   364→</html>`;
   365→      const blob = new Blob([html], { type: 'text/html;charset=utf-8' });
   366→      const url = URL.createObjectURL(blob);
   367→      const a = document.createElement('a'); a.href = url; a.download = `${IL.name}-تحلیل-تکنیکال.html`; a.click();

   369→    } catch (err) { console.error('HTML export failed:', err); }

   871→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   872→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   873→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   874→  grade?: string; overlapCount?: number; touchCount?: number;
   875→}) {
   876→  const isSupport = type === 'support';
   877→  const isTargetType = type === 'target';
   878→
   879→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   880→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   881→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   882→
   883→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   884→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   885→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   886→
   887→  return (
   888→    <div
   889→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   890→      style={{
   891→        background: bgAccent,
   892→        border: `1px solid ${borderAccent}`,
   893→        borderRight: `3px solid ${accentColor}`,
   894→      }}
   895→    >
   896→      {/* Label + Price */}
   897→      <div className="flex-1 min-w-0">
   898→        <div className="flex items-center gap-2">
   899→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   900→          {isTarget && <span className="text-[9px]">🎯</span>}
   901→          {overlapCount !== undefined && overlapCount > 0 && (
   902→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   903→              {toPersianDigits(overlapCount)} هم‌پوشانی
   904→            </span>
   905→          )}
   906→        </div>
   907→        <div className="flex items-baseline gap-1">
   908→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   909→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   910→        </div>
   911→      </div>
   912→
   913→      {/* Strength + Grade */}
   914→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   915→        <div className="flex items-center gap-1.5">
   916→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   917→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   918→          <span className="text-[9px] text-gray-500">/۱۰</span>
   919→        </div>
   920→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   921→          <div
   922→            className="h-full rounded-full transition-all duration-500"
   923→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   924→          />
   925→        </div>
   926→        {touchCount !== undefined && touchCount > 0 && (
   927→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   928→        )}
   929→      </div>
   930→    </div>
   931→  );
   932→}
