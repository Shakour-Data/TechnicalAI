   260→  const [refreshing, setRefreshing] = useState(false);
   261→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   262→  const [currentPage, setCurrentPage] = useState<PageId>('analysis');
   263→
   264→  // Store last fetch params for auto-refresh
   265→  const lastFetchRef = useRef<{ symbol: string; category?: string; insCode?: string; tgjuKey?: string; finpySector?: string; finpyIndex?: string; webId?: number; yahooSymbol?: string } | null>(null);
   266→  const refreshTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   267→  const fetchControllerRef = useRef<AbortController | null>(null);
   268→
   269→  // Chart container ref for image export
   270→  const chartWrapperRef = useRef<HTMLDivElement>(null);
   271→
   272→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number, yahooSymbol?: string) => {
   273→    // Abort any previous in-flight fetch
   274→    if (fetchControllerRef.current) {
   275→      fetchControllerRef.current.abort();
   276→    }
   277→    const controller = new AbortController();
   278→    fetchControllerRef.current = controller;
   279→
   280→    // Yahoo Finance instrument: fetch historical data via Yahoo Finance API
   281→    // This handles ALL Yahoo instruments (stocks, indices, energy, metals, forex, crypto, ETFs, etc.)
   282→    if (yahooSymbol) {
   283→      setLoading(true);
   284→      setLoadingMessage('در حال دریافت داده‌های تاریخی از یاهو فایننس ...');
   285→      setError(null);
   286→      setData(null);
   287→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId, yahooSymbol };
   288→      try {
   289→        const res = await fetch('/api/yahoo-analysis?symbol=' + encodeURIComponent(yahooSymbol), { signal: controller.signal });
   290→        if (controller.signal.aborted) return;
   291→        if (!res.ok) {
   292→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   293→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   294→          throw new Error(errMsg);
   295→        }
   296→        const json = await res.json();
   297→        if (controller.signal.aborted) return;
   298→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   299→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   300→        }
   301→        setData({ ...json, isYahoo: true });
   302→        setActivePanel('visual');
   303→      } catch (err) {
   304→        if (controller.signal.aborted) return;
   305→        setError(String(err));
   306→      } finally {
   307→        if (!controller.signal.aborted) {
   308→          setLoading(false);
   309→          setLoadingMessage(null);
   310→        }
   311→      }
   312→      return;
   313→    }
   314→
   315→    // TGJU instrument: fetch historical data via tgju.org chart API
   316→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   317→      setLoading(true);
   318→      setLoadingMessage('در حال دریافت داده‌های تاریخی ...');
