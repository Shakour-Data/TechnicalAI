   104→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   105→function sma(closes: number[], period: number): number {
   106→  if (closes.length < period) return 0;
   107→  const slice = closes.slice(-period);
   108→  return slice.reduce((a, b) => a + b, 0) / period;
   109→}

   111→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   112→function emaCalc(closes: number[], period: number): number {
   113→  if (closes.length < period) return 0;
   114→  const k = 2 / (period + 1);
   115→  let emaVal = sma(closes.slice(0, period), period);
   116→  for (let i = period; i < closes.length; i++) {
   117→    emaVal = closes[i] * k + emaVal * (1 - k);
   118→  }
   119→  return emaVal;
   120→}

   122→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   123→function calcRSI(closes: number[], period: number = 14): number {
   124→  if (closes.length < period + 1) return 50;
   125→  let avgGain = 0;
   126→  let avgLoss = 0;
   127→  for (let i = 1; i <= period; i++) {
   128→    const diff = closes[i] - closes[i - 1];
   129→    if (diff > 0) avgGain += diff;
   130→    else avgLoss += Math.abs(diff);
   131→  }
   132→  avgGain /= period;
   133→  avgLoss /= period;
   134→  for (let i = period + 1; i < closes.length; i++) {
   135→    const diff = closes[i] - closes[i - 1];
   136→    const gain = diff > 0 ? diff : 0;
   137→    const loss = diff < 0 ? Math.abs(diff) : 0;
   138→    avgGain = (avgGain * (period - 1) + gain) / period;
   139→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   140→  }
   141→  if (avgLoss === 0) return 100;
   142→  const rs = avgGain / avgLoss;
   143→  return 100 - 100 / (1 + rs);
   144→}

   146→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   147→function calcMFI(data: OHLCV[], period: number = 14): number {
   148→  if (data.length < period + 1) return 50;
   149→  let posFlow = 0;
   150→  let negFlow = 0;
   151→  for (let i = data.length - period; i < data.length; i++) {
   152→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   153→    const mf = tp * data[i].volume;
   154→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   155→    if (tp > prevTp) posFlow += mf;
   156→    else negFlow += mf;
   157→  }
   158→  if (negFlow === 0) return 100;
   159→  const mfr = posFlow / negFlow;
   160→  return 100 - 100 / (1 + mfr);
   161→}

   163→// ─── Helper: CCI(20) ──────────────────────────────────────────────────────────
   164→function calcCCI(data: OHLCV[], period: number = 20): number {
   165→  if (data.length < period) return 0;
   166→  const tps: number[] = [];
   167→  for (let i = data.length - period; i < data.length; i++) {
   168→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   169→  }
   170→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   171→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   172→  if (meanDev === 0) return 0;
   173→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   174→}

   176→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   177→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   178→  if (data.length < kPeriod) return { k: 50, d: 50 };
   179→  const rawKs: number[] = [];
   180→  for (let i = kPeriod - 1; i < data.length; i++) {
   181→    let lowest = Infinity;
   182→    let highest = -Infinity;
   183→    for (let j = i - kPeriod + 1; j <= i; j++) {
   184→      if (data[j].low < lowest) lowest = data[j].low;
   185→      if (data[j].high > highest) highest = data[j].high;
   186→    }
   187→    const range = highest - lowest;
   188→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   189→  }
   190→  const smoothedK: number[] = [];
   191→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   192→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   193→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   194→  }
   195→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   196→  const dSlice = smoothedK.slice(-smoothD);
   197→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   198→  return { k, d };
   199→}

   201→// ─── Helper: Williams %R(14) ──────────────────────────────────────────────────
   202→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   203→  if (data.length < period) return -50;
   204→  const slice = data.slice(-period);
   205→  let highest = -Infinity;
   206→  let lowest = Infinity;
   207→  for (const c of slice) {
   208→    if (c.high > highest) highest = c.high;
   209→    if (c.low < lowest) lowest = c.low;
   210→  }
   211→  const range = highest - lowest;
   212→  if (range === 0) return -50;
   213→  return ((highest - data[data.length - 1].close) / range) * -100;
   214→}

   216→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   217→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   218→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   219→  const emaFast = emaCalc(closes, fast);
   220→  const emaSlow = emaCalc(closes, slow);
   221→  const macdLine = emaFast - emaSlow;
   222→  // Calculate signal as EMA of MACD values
   223→  const macdValues: number[] = [];
   224→  const kFast = 2 / (fast + 1);
   225→  const kSlow = 2 / (slow + 1);
   226→  let eFast = sma(closes.slice(0, fast), fast);
   227→  let eSlow = sma(closes.slice(0, slow), slow);
   228→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   229→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   230→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   231→    macdValues.push(eFast - eSlow);
   232→  }
   233→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   234→  const kSig = 2 / (sig + 1);
   235→  let signalVal = sma(macdValues.slice(0, sig), sig);
   236→  for (let i = sig; i < macdValues.length; i++) {
   237→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   238→  }
   239→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   240→}

   242→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   243→function calcATR(data: OHLCV[], period: number = 14): number {
   244→  if (data.length < 2) return 0;
   245→  const trs: number[] = [];
   246→  for (let i = 1; i < data.length; i++) {
   247→    const tr = Math.max(
   248→      data[i].high - data[i].low,
   249→      Math.abs(data[i].high - data[i - 1].close),
   250→      Math.abs(data[i].low - data[i - 1].close)
   251→    );
   252→    trs.push(tr);
   253→  }
   254→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   255→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   256→  for (let i = period; i < trs.length; i++) {
   257→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   258→  }
   259→  return atrVal;
   260→}

   262→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   263→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   264→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   265→  const plusDMs: number[] = [];
   266→  const minusDMs: number[] = [];
   267→  const trs: number[] = [];
   268→  for (let i = 1; i < data.length; i++) {
   269→    const upMove = data[i].high - data[i - 1].high;
   270→    const downMove = data[i - 1].low - data[i].low;
   271→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   272→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   273→    trs.push(Math.max(
   274→      data[i].high - data[i].low,
   275→      Math.abs(data[i].high - data[i - 1].close),
   276→      Math.abs(data[i].low - data[i - 1].close)
   277→    ));
   278→  }
   279→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   280→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   281→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   282→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   283→  const dxValues: number[] = [];
   284→  for (let i = period; i < trs.length; i++) {
   285→    smoothTR = smoothTR - smoothTR / period + trs[i];
   286→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   287→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   288→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   289→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   290→    const diSum = diPlus + diMinus;
   291→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   292→    dxValues.push(dx);
   293→  }
   294→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   295→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   296→  for (let i = period; i < dxValues.length; i++) {
   297→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   298→  }
   299→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   300→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   301→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   302→}

   304→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   305→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   306→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   307→  let isLong = true;
   308→  let af = step;
   309→  let ep = data[0].high;
   310→  let sar = data[0].low;
   311→  for (let i = 1; i < data.length; i++) {
   312→    const prevSar = sar;
   313→    sar = prevSar + af * (ep - prevSar);
   314→    // Ensure SAR is within bounds
   315→    if (isLong) {
   316→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   317→      if (data[i].low < sar) {
   318→        isLong = false;
   319→        sar = ep;
   320→        ep = data[i].low;
   321→        af = step;
   322→      } else {
   323→        if (data[i].high > ep) {
   324→          ep = data[i].high;
   325→          af = Math.min(af + step, max);
   326→        }
   327→      }
   328→    } else {
   329→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   330→      if (data[i].high > sar) {
   331→        isLong = true;
   332→        sar = ep;
   333→        ep = data[i].high;
   334→        af = step;
   335→      } else {
   336→        if (data[i].low < ep) {
   337→          ep = data[i].low;
   338→          af = Math.min(af + step, max);
   339→        }
   340→      }
   341→    }
   342→  }
   343→  return sar;
   344→}

   346→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   347→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   348→  if (closes.length < period) {
   349→    const c = closes[closes.length - 1] ?? 0;
   350→    return { upper: c, middle: c, lower: c };
   351→  }
   352→  const slice = closes.slice(-period);
   353→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   354→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   355→  const sd = Math.sqrt(variance);
   356→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   357→}

   360→function calcOBV(data: OHLCV[]): number {
   361→  let obv = 0;
   362→  for (let i = 1; i < data.length; i++) {
   363→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   364→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   365→  }
   366→  return obv;
   367→}

   369→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   370→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   371→  const n = values.length;
   372→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   373→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   374→  for (let i = 0; i < n; i++) {
   375→    sumX += i;
   376→    sumY += values[i];
   377→    sumXY += i * values[i];
   378→    sumX2 += i * i;
   379→    sumY2 += values[i] * values[i];
   380→  }
   381→  const denom = n * sumX2 - sumX * sumX;
   382→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   383→  const slope = (n * sumXY - sumX * sumY) / denom;
   384→  const intercept = (sumY - slope * sumX) / n;
   385→  const meanY = sumY / n;
   386→  let ssRes = 0, ssTot = 0;
   387→  for (let i = 0; i < n; i++) {
   388→    const predicted = slope * i + intercept;
   389→    ssRes += (values[i] - predicted) ** 2;
   390→    ssTot += (values[i] - meanY) ** 2;
   391→  }
   392→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   393→  return { slope, intercept, r2 };
   394→}

   396→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   397→function roundToNice(price: number): number {
   398→  const abs = Math.abs(price);
   399→  let step: number;
   400→  if (abs >= 100000) step = 100;
   401→  else if (abs >= 10000) step = 50;
   402→  else if (abs >= 1000) step = 10;
   403→  else if (abs >= 100) step = 5;
   404→  else step = 1;
   405→  return Math.round(price / step) * step;
   406→}

   408→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
   409→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: number[]; lows: number[] } {
   410→  const highs: number[] = [];
   411→  const lows: number[] = [];
   412→  for (let i = lookback; i < data.length - lookback; i++) {
   413→    let isHigh = true;
   414→    let isLow = true;
   415→    for (let j = i - lookback; j <= i + lookback; j++) {
   416→      if (j === i) continue;
   417→      if (data[j].high >= data[i].high) isHigh = false;
   418→      if (data[j].low <= data[i].low) isLow = false;
   419→    }
   420→    if (isHigh) highs.push(data[i].high);
   421→    if (isLow) lows.push(data[i].low);
   422→  }
   423→  return { highs, lows };
   424→}

   426→// ─── Support / Resistance (6 each, adaptive gap, multi-factor strength) ───
   427→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   428→  resistances: number[];
   429→  supports: number[];
   430→  supportStrengths: LevelStrength[];
   431→  resistanceStrengths: LevelStrength[];
   432→  priceTargets: LevelStrength[];
   433→} {
   434→  const closes = data.map(d => d.close);
   435→  const avgVolume = data.reduce((s, d) => s + d.volume, 0) / data.length;
   436→
   437→  // Tag each raw level with its source type for confluence scoring
   438→  interface TaggedLevel { price: number; source: string };
   439→  const allRaw: TaggedLevel[] = [];
   440→
   441→  // 1. Pivot Points (source: 'pivot')
   442→  const prev = data[data.length - 2] ?? data[data.length - 1];
   443→  if (prev) {
   444→    const pp = (prev.high + prev.low + prev.close) / 3;
   445→    const hl = prev.high - prev.low;
   446→    const pivotR = [
   447→      2 * pp - prev.low, pp + hl,
   448→      prev.high + 2 * (pp - prev.low),
   449→      prev.high + 2 * (pp - prev.low) + hl,
   450→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   451→    ];
   452→    const pivotS = [
   453→      2 * pp - prev.high, pp - hl,
   454→      prev.low - 2 * (prev.high - pp),
   455→      prev.low - 2 * (prev.high - pp) - hl,
   456→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   457→    ];
   458→    for (const p of pivotR) allRaw.push({ price: p, source: 'pivot' });
   459→    for (const p of pivotS) allRaw.push({ price: p, source: 'pivot' });
   460→  }
   461→
   462→  // 2. Swing levels (source: 'swing')
   463→  for (const lb of [3, 5, 7, 10]) {
   464→    const swings = findSwingLevels(data, lb);
   465→    for (const h of swings.highs) allRaw.push({ price: h, source: 'swing' });
   466→    for (const l of swings.lows) allRaw.push({ price: l, source: 'swing' });
   467→  }
   468→
   469→  // 3. Moving Average levels (source: 'ma')
   470→  for (const period of [5, 10, 21, 50, 100, 200]) {
   471→    const v = sma(closes, period);
   472→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01)
   473→      allRaw.push({ price: v, source: 'ma' });
   474→  }
   475→  const ema12 = emaCalc(closes, 12);
   476→  const ema26 = emaCalc(closes, 26);
   477→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01)
   478→    allRaw.push({ price: ema12, source: 'ma' });
   479→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01)
   480→    allRaw.push({ price: ema26, source: 'ma' });
   481→
   482→  // 4. Bollinger Band levels (source: 'bb')
   483→  const bbLevels = calcBollingerBands(closes);
   484→  if (bbLevels.upper > 0) allRaw.push({ price: bbLevels.upper, source: 'bb' });
   485→  if (bbLevels.lower > 0) allRaw.push({ price: bbLevels.lower, source: 'bb' });
   486→
   487→  // 5. Round number / psychological levels (source: 'round')
   488→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
   489→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
   490→    const p = Math.round(currentPrice * mult / magnitude) * magnitude;
   491→    allRaw.push({ price: p, source: 'round' });
   492→  }
   493→
   494→  // 6. Recent high/low zones (source: 'hilo')
   495→  for (const w of [5, 22, 66]) {
   496→    const slice = data.slice(-Math.min(w, data.length));
   497→    if (slice.length > 0) {
   498→      allRaw.push({ price: Math.max(...slice.map(d => d.high)), source: 'hilo' });
   499→      allRaw.push({ price: Math.min(...slice.map(d => d.low)), source: 'hilo' });
   500→    }
   501→  }
   502→
   503→  // 7. Round all and filter invalid
   504→  const rounded: TaggedLevel[] = allRaw
   505→    .map(l => ({ price: roundToNice(l.price), source: l.source }))
   506→    .filter(l => l.price > 0);
   507→
   508→  // 8. Cluster nearby levels (within 0.5%) — merge sources, keep most common price
   509→  interface Cluster {
   510→    price: number;
   511→    sources: Set<string>;
   512→    touchCount: number;
   513→    totalVolumeAtTouches: number;
   514→    mostRecentTouchBar: number;
   515→  }
   516→  const clusters: Cluster[] = [];
   517→  for (const lvl of rounded) {
   518→    let merged = false;
   519→    for (const c of clusters) {
   520→      if (Math.abs(c.price - lvl.price) / Math.max(c.price, 1) < 0.005) {
   521→        c.sources.add(lvl.source);
   522→        merged = true;
   523→        break;
   524→      }
   525→    }
   526→    if (!merged) {
   527→      clusters.push({ price: lvl.price, sources: new Set([lvl.source]), touchCount: 0, totalVolumeAtTouches: 0, mostRecentTouchBar: -1 });
   528→    }
   529→  }
   530→
   531→  // 9. Count actual price TOUCHES (rejections) and volume at each cluster level
   532→  // A real touch = bar wick reached the level AND bar body closed away from it
   533→  const touchThreshold = currentPrice * 0.002; // 0.2% tight proximity
   534→  for (let i = 0; i < data.length; i++) {
   535→    const bar = data[i];
   536→    for (const c of clusters) {
   537→      const isResistance = c.price > currentPrice;
   538→      if (isResistance) {
   539→        // Resistance: bar high reached near the level AND closed below it (rejection)
   540→        if (bar.high >= c.price - touchThreshold && bar.close < c.price) {
   541→          c.touchCount++;
   542→          c.totalVolumeAtTouches += bar.volume;
   543→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   544→        }
   545→      } else {
   546→        // Support: bar low reached near the level AND closed above it (rejection)
   547→        if (bar.low <= c.price + touchThreshold && bar.close > c.price) {
   548→          c.touchCount++;
   549→          c.totalVolumeAtTouches += bar.volume;
   550→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   551→        }
   552→      }
   553→    }
   554→  }
   555→
   556→  // 10. Calculate multi-factor strength (0-10)
   557→  const levelStrengths: LevelStrength[] = clusters.map(c => {
   558→    // Factor 1: Source Confluence (0-2.0 pts)
   559→    const sourceScore = Math.min(2.0, c.sources.size * 0.4);
   560→    const hasSwingAndPivot = c.sources.has('swing') && c.sources.has('pivot') ? 0.4 : 0;
   561→    const maConfluence = c.sources.has('ma') ? 0.4 : 0;
   562→    const confluenceScore = Math.min(2.0, sourceScore + hasSwingAndPivot + maConfluence);
   563→
   564→    // Factor 2: Touch Count — CAPPED by recency (0-1.5 pts)
   565→    // Only count touches in the most recent 50% of data to avoid ancient history inflating
   566→    const recentBarStart = Math.floor(data.length * 0.5);
   567→    const recentTouches = c.mostRecentTouchBar >= recentBarStart ? c.touchCount : Math.max(0, c.touchCount - 1);
   568→    let touchScore: number;
   569→    if (recentTouches === 0) touchScore = 0;
   570→    else if (recentTouches === 1) touchScore = 0.3;
   571→    else if (recentTouches <= 2) touchScore = 0.6;
   572→    else if (recentTouches <= 4) touchScore = 0.9;
   573→    else if (recentTouches <= 7) touchScore = 1.2;
   574→    else touchScore = 1.5;
   575→
   576→    // Factor 3: Volume Confirmation (0-1.0 pts) — skip when no volume available
   577→    let volumeScore = 0;
   578→    const dataHasVolume = data.some(d => d.volume > 0);
   579→    if (dataHasVolume && c.touchCount > 0) {
   580→      const avgVolAtTouch = c.totalVolumeAtTouches / c.touchCount;
   581→      const volRatio = avgVolume > 0 ? avgVolAtTouch / avgVolume : 1;
   582→      if (volRatio > 2.0) volumeScore = 1.0;
   583→      else if (volRatio > 1.5) volumeScore = 0.8;
   584→      else if (volRatio > 1.0) volumeScore = 0.5;
   585→      else if (volRatio > 0.7) volumeScore = 0.25;
   586→      else volumeScore = 0.1;
   587→    }
   588→
   589→    // Factor 4: Freshness / Recency (0-1.0 pts)
   590→    let freshnessScore = 0;
   591→    if (c.touchCount > 0 && data.length > 0) {
   592→      const barsAgo = data.length - 1 - c.mostRecentTouchBar;
   593→      if (barsAgo <= 3) freshnessScore = 1.0;
   594→      else if (barsAgo <= 10) freshnessScore = 0.8;
   595→      else if (barsAgo <= 25) freshnessScore = 0.5;
   596→      else if (barsAgo <= 50) freshnessScore = 0.25;
   597→      else freshnessScore = 0.05;
   598→    }
   599→
   600→    // Factor 5: Proximity to current price (0-0.5 pts)
   601→    const distRatio = Math.abs(c.price - currentPrice) / currentPrice;
   602→    let proximityScore = 0;
   603→    if (distRatio < 0.01) proximityScore = 0.5;
   604→    else if (distRatio < 0.02) proximityScore = 0.35;
   605→    else if (distRatio < 0.04) proximityScore = 0.2;
   606→    else if (distRatio < 0.07) proximityScore = 0.1;
   607→
   608→    const total = confluenceScore + touchScore + volumeScore + freshnessScore + proximityScore;
   609→    const strength = Math.min(10, Math.max(1, Math.round(total * 10 / 6.0)));
   610→
   611→    return { price: c.price, strength, isTarget: false };
   612→  });
   613→
   614→  // 11. Separate above/below current price
   615→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   616→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   617→
   618→  // 12. Adaptive gap enforcement — keep highest strength in cluster
   619→  // v2: enforce 5-10% gap between consecutive S/R lines
   620→  function enforceGap(levels: LevelStrength[], targetCount: number): LevelStrength[] {
   621→    const gapSteps = [0.10, 0.09, 0.08, 0.07, 0.06, 0.05];
   622→    for (const gapPct of gapSteps) {
   623→      if (levels.length === 0) return [];
   624→      const result: LevelStrength[] = [levels[0]];
   625→      for (let i = 1; i < levels.length; i++) {
   626→        if (result.length >= targetCount) break;
   627→        const prevPrice = result[result.length - 1].price;
   628→        const currPrice = levels[i].price;
   629→        const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   630→        if (gap >= gapPct) {
   631→          result.push(levels[i]);
   632→        } else {
   633→          // Merge: keep stronger level's strength (no inflation)
   634→          const keep = levels[i].strength >= result[result.length - 1].strength ? levels[i] : result[result.length - 1];
   635→          result[result.length - 1] = { ...keep, strength: keep.strength };
   636→        }
   637→      }
   638→      if (result.length >= targetCount) return result;
   639→    }
   640→    // Fallback: still enforce minimum 5% gap, returning as many as possible
   641→    if (levels.length === 0) return [];
   642→    const fallback: LevelStrength[] = [levels[0]];
   643→    for (let i = 1; i < levels.length; i++) {
   644→      const prevPrice = fallback[fallback.length - 1].price;
   645→      const currPrice = levels[i].price;
   646→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   647→      if (gap >= 0.05) {
   648→        fallback.push(levels[i]);
   649→      } else {
   650→        const keep = levels[i].strength >= fallback[fallback.length - 1].strength ? levels[i] : fallback[fallback.length - 1];
   651→        fallback[fallback.length - 1] = { ...keep, strength: keep.strength };
   652→      }
   653→    }
   654→    return fallback;
   655→  }
   656→
   657→  const finalResistances = enforceGap(above, 6);
   658→  const finalSupports = enforceGap(below, 6);
   659→
   660→  // 13. Determine price targets: top 2 by strength, or any above 7
   661→  const allLevels = [...finalResistances, ...finalSupports];
   662→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   663→  let targets: LevelStrength[];
   664→  if (sortedByStrength.some(l => l.strength >= 7)) {
   665→    targets = sortedByStrength.filter(l => l.strength >= 7).slice(0, 3);
   666→  } else {
   667→    targets = sortedByStrength.slice(0, 2);
   668→  }
   669→
   670→  // Mark targets
   671→  targets = targets.map(t => ({ ...t, isTarget: true }));
   672→
   673→  // Also mark in final arrays
   674→  finalResistances.forEach(r => {
   675→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   676→  });
   677→  finalSupports.forEach(s => {
   678→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   679→  });
   680→
   681→  return {
   682→    resistances: finalResistances.map(l => l.price),
   683→    supports: finalSupports.map(l => l.price),
   684→    supportStrengths: finalSupports,
   685→    resistanceStrengths: finalResistances,
   686→    priceTargets: targets,
   687→  };
   688→}

   690→// ─── Trend Analysis ───────────────────────────────────────────────────────────
   691→function calcTrend(closes: number[], period: number): TrendResult {
   692→  const slice = closes.slice(-Math.min(period, closes.length));
   693→  const { slope, r2 } = linearRegression(slice);
   694→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   695→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   696→  const priceThreshold = avgPrice * 0.001; // 0.1%
   697→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   698→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   699→}

   706→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
   707→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
   708→  const slice = data.slice(0, barIdx + 1);
   709→  const price = slice[slice.length - 1].close;
   710→  let nearestR = price * 1.05;
   711→  let nearestS = price * 0.95;
   712→
   713→  // Collect all swing highs (resistance) and swing lows (support)
   714→  const resistances: number[] = [];
   715→  const supports: number[] = [];
   716→  for (const lb of [3, 5, 7]) {
   717→    const swings = findSwingLevels(slice, lb);
   718→    resistances.push(...swings.highs);
   719→    supports.push(...swings.lows);
   720→  }
   721→
   722→  // Find nearest resistance above price
   723→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
   724→  if (abovePrice.length > 0) nearestR = abovePrice[0];
   725→
   726→  // Find nearest support below price
   727→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
   728→  if (belowPrice.length > 0) nearestS = belowPrice[0];
   729→
   730→  return { R1: nearestR, S1: nearestS };
   731→}

   733→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
   734→// Returns true if bullish divergence detected, false if bearish, null if none
   735→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
   736→  const lookback = 20;
   737→  const start = Math.max(0, barIdx - lookback);
   738→  const slice = data.slice(start, barIdx + 1);
   739→  if (slice.length < 10) return null;
   740→
   741→  const mid = Math.floor(slice.length / 2);
   742→  const firstHalf = slice.slice(0, mid);
   743→  const secondHalf = slice.slice(mid);
   744→
   745→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
   746→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
   747→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
   748→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
   749→
   750→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
   751→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
   752→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
   753→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
   754→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
   755→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
   756→
   757→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
   758→    return false; // bearish divergence
   759→  }
   760→
   761→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
   762→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
   763→
   764→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
   765→    return true; // bullish divergence
   766→  }
   767→
   768→  return null;
   769→}

   772→// LAYER 1 & 2 — Feature Computation at a Historical Bar

   775→/**
   776→ * Computes all 16 ML features at a specific historical bar.
   777→ * Uses data[0..barIdx] (inclusive) for all indicator calculations.
   778→ * Returns an object keyed by feature name matching FEATURES_ORDER.
   779→ */
   780→function computeFeaturesAtBar(data: OHLCV[], barIdx: number): Record<string, number> {
   781→  // Need at least 30 bars for meaningful indicators
   782→  if (barIdx < 30) {
   783→    const empty: Record<string, number> = {};
   784→    for (const key of FEATURES_ORDER) empty[key] = 0.5;
   785→    return empty;
   786→  }
   787→
   788→  const slice = data.slice(0, barIdx + 1);
   789→  const closes = slice.map(d => d.close);
   790→  const price = closes[closes.length - 1];
   791→  const eps = 1e-10;
   792→
   793→  // ── Compute all base indicators ───────────────────────────────────────────
   794→  const rsiVal = calcRSI(closes);
   795→  const sliceHasVolume = slice.some(d => d.volume > 0);
   796→  const mfiVal = sliceHasVolume ? calcMFI(slice) : 50; // neutral when no volume
   797→  const cciVal = calcCCI(slice);
   798→  const stoch = calcStochastic(slice);
   799→  const macd = calcMACD(closes);
   800→  const adxResult = calcADX(slice);
   801→  const atrVal = calcATR(slice);
   802→  const bb = calcBollingerBands(closes);
   803→  const ma21 = sma(closes, 21);
   804→  const ma100 = sma(closes, 100);
   805→  const ema12 = emaCalc(closes, 12);
   806→  const ema26 = emaCalc(closes, 26);
   807→  const trendResult = calcTrend(closes, 21);
   808→  const sr = simplifiedSR(data, barIdx);
   809→
   810→  // ── Layer 1: Raw Scores (s_*) ────────────────────────────────────────────
   811→
   812→  // s_rsi — RSI mapped to 0-1
   813→  const s_rsi = rsiVal > 80 ? 0.95
   814→    : rsiVal > 70 ? 0.85
   815→    : rsiVal > 60 ? 0.70
   816→    : rsiVal > 50 ? 0.55
   817→    : rsiVal > 40 ? 0.45
   818→    : rsiVal > 30 ? 0.30
   819→    : rsiVal > 20 ? 0.15
   820→    : 0.05;
   821→
   822→  // s_mfi — MFI mapped to 0-1 (same formula as RSI)
   823→  const s_mfi = mfiVal > 80 ? 0.95
   824→    : mfiVal > 70 ? 0.85
   825→    : mfiVal > 60 ? 0.70
   826→    : mfiVal > 50 ? 0.55
   827→    : mfiVal > 40 ? 0.45
   828→    : mfiVal > 30 ? 0.30
   829→    : mfiVal > 20 ? 0.15
   830→    : 0.05;
   831→
   832→  // s_cci — CCI mapped to 0-1
   833→  const s_cci = cciVal > 200 ? 0.90
   834→    : cciVal > 100 ? 0.75
   835→    : cciVal > 0 ? 0.60
   836→    : cciVal > -100 ? 0.40
   837→    : cciVal > -200 ? 0.25
   838→    : 0.10;
   839→
   840→  // s_adx — Directional bias
   841→  const s_adx = (adxResult.diPlus - adxResult.diMinus + 100) / 200;
   842→
   843→  // s_macd — MACD histogram + signal combined
   844→  const absMacdLine = Math.abs(macd.line) + 1;
   845→  const s_macd = macd.line > macd.signal
   846→    ? 0.5 + Math.min(0.5, macd.histogram / absMacdLine)
   847→    : 0.5 - Math.min(0.5, Math.abs(macd.histogram) / absMacdLine);
   848→
   849→  // s_stoch — Stochastic %K mapped to 0-1
   850→  const s_stoch = stoch.k / 100;
   851→
   852→  // s_bb — Bollinger Band position
   853→  const bbRange = bb.upper - bb.lower;
   854→  const s_bb = bbRange > eps
   855→    ? clamp((price - bb.lower) / bbRange, 0, 1)
   856→    : 0.5;
   857→
   858→  // s_ma21 — Price relative to 21-period SMA
   859→  const s_ma21 = ma21 > eps
   860→    ? clamp(0.5 + (price - ma21) / ma21, 0, 1)
   861→    : 0.5;
   862→
   863→  // s_ma100 — Price relative to 100-period SMA
   864→  const s_ma100 = ma100 > eps
   865→    ? clamp(0.5 + (price - ma100) / ma100, 0, 1)
   866→    : 0.5;
   867→
   868→  // s_ema — EMA12 vs EMA26 spread
   869→  const s_ema = ema26 > eps
   870→    ? clamp(0.5 + (ema12 - ema26) / ema26 * 5, 0, 1)
   871→    : 0.5;
   872→
   873→  // s_atr — Inverse volatility (lower ATR = higher score)
   874→  const s_atr = price > eps ? 1 - Math.min(atrVal / price * 10, 1) : 0.5;
   875→
   876→  // s_trend — R² + angle combination
   877→  const s_trend = (trendResult.r2 * 0.7) +
   878→    (Math.max(0, Math.min(1, (trendResult.angle + 45) / 90)) * 0.3);
   879→
   880→  // s_sr — Support/Resistance proximity
   881→  const dR1 = (sr.R1 - price) / (price > eps ? price : 1);
   882→  const dS1 = (price - sr.S1) / (price > eps ? price : 1);
   883→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
   884→
   885→  // ── Layer 2: Corrections (f_*) ───────────────────────────────────────────
   886→
   887→  // f_rsi — RSI with momentum correction
   888→  const rsiCurrent = rsiVal;
   889→  const rsi5ago = barIdx >= 5
   890→    ? calcRSI(closes.slice(0, barIdx - 5 + 1))
   891→    : rsiCurrent;
   892→  const f_rsi = clamp(s_rsi + (rsiCurrent - rsi5ago) / 100 * 0.08, 0, 1);
   893→
   894→  // f_mfi — MFI with momentum correction (skip when no volume)
   895→  const mfiCurrent = mfiVal;
   896→  const mfi5ago = (barIdx >= 5 && sliceHasVolume)
   897→    ? calcMFI(data.slice(0, barIdx - 5 + 1))
   898→    : mfiCurrent;
   899→  const f_mfi = sliceHasVolume
   900→    ? clamp(s_mfi + (mfiCurrent - mfi5ago) / 100 * 0.08, 0, 1)
   901→    : 0.5; // neutral when no volume
   902→
   903→  // f_cci — CCI with momentum correction (scale: /200 instead of /100)
   904→  const cciCurrent = cciVal;
   905→  const cci5ago = barIdx >= 5
   906→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
   907→    : cciCurrent;
   908→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
   909→
   910→  // f_macd — Same as s_macd (no additional correction)
   911→  const f_macd = s_macd;
   912→
   913→  // f_stoch — Same as s_stoch
   914→  const f_stoch = s_stoch;
   915→
   916→  // f_stochCross — Stochastic K/D crossover detection
   917→  let f_stochCross = 0.1;
   918→  if (barIdx >= 1) {
   919→    const prevSlice = data.slice(0, barIdx);
   920→    const prevStoch = calcStochastic(prevSlice);
   921→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
   922→      f_stochCross = 0.9; // K crossed above D (bullish)
   923→    }
   924→  }
   925→
   926→  // f_macdCross — MACD/Signal crossover detection
   927→  let f_macdCross = 0.1;
   928→  if (barIdx >= 1) {
   929→    const prevCloses = closes.slice(0, -1);
   930→    if (prevCloses.length >= 35) {
   931→      const prevMacd = calcMACD(prevCloses);
   932→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
   933→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
   934→      }
   935→    }
   936→  }
   937→
   938→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
   939→  const divResult = detectDivergenceSimple(data, barIdx);
   940→  const f_div = divResult === true ? 0.85 : 0.15;
   941→
   942→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
   943→  return {
   944→    rsi: f_rsi,
   945→    mfi: f_mfi,
   946→    cci: f_cci,
   947→    adx: s_adx,
   948→    macd: f_macd,
   949→    stoch: f_stoch,
   950→    bb: s_bb,
   951→    ma21: s_ma21,
   952→    ma100: s_ma100,
   953→    ema: s_ema,
   954→    atr: s_atr,
   955→    trend: s_trend,
   956→    sr: s_sr,
   957→    stochCross: f_stochCross,
   958→    macdCross: f_macdCross,
   959→    div: f_div,
   960→  };
   961→}
   962→
   963→// ═══════════════════════════════════════════════════════════════════════════════
   964→// TRAINING DATA CREATION (Layer 0 — Data Preparation)
   965→// ═══════════════════════════════════════════════════════════════════════════════
   966→
   967→/**
   968→ * Creates supervised training data from historical OHLCV data.
   969→ * For each bar, computes features and labels based on future price movement.
   970→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
   971→ * Samples every `sampleStep` bars to reduce computation.
   972→ */
   973→function createTrainingData(
   974→  data: OHLCV[],
   975→  horizon: number = 5,
   976→  sampleStep: number = 5
   977→): { X: number[][]; y: number[] } {
   978→  const X: number[][] = [];
   979→  const y: number[] = [];
   980→
   981→  // Start from horizon+30 to ensure enough data for indicators
   982→  // End at data.length - horizon - 1 to have future bars for labeling
   983→  // Sample every sampleStep bars for performance
   984→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
   985→    const features = computeFeaturesAtBar(data, i);
   986→
   987→    // Convert to array following FEATURES_ORDER
   988→    const featureVec: number[] = [];
   989→    let valid = true;
   990→    for (const key of FEATURES_ORDER) {
   991→      const val = features[key];
   992→      if (val === undefined || val === null || Number.isNaN(val)) {
   993→        valid = false;
   994→        break;
   995→      }
   996→      featureVec.push(val);
   997→    }
   998→    if (!valid) continue;
   999→
  1000→    // Label: 1 if future close > 1% above current close
  1001→    const currentClose = data[i].close;
  1002→    const futureClose = data[i + horizon].close;
  1003→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
  1004→
  1005→    X.push(featureVec);
  1006→    y.push(label);
  1007→  }
  1008→
  1009→  return { X, y };
  1010→}
  1011→
  1012→// ═══════════════════════════════════════════════════════════════════════════════
  1013→// VDss GRAPH STRUCTURE (Layer 5 — State Machine)
  1014→// ═══════════════════════════════════════════════════════════════════════════════
  1015→
  1016→/**
  1017→ * VDss Graph: Each node represents a market state.
  1018→ * Edges represent transitions with types: 'up', 'down', 'pullback', 'risk', 'terminal'.
  1019→ * Terminal nodes map to scenario results (R1=strong bullish, R5=deep correction).
  1020→ */
  1021→const VDSS_GRAPH = {
  1022→  // outgoing edges from each node: [target, type]
  1023→  outgoing: {
  1024→    A: [['B', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1025→    B: [['C', 'up'], ['G', 'pullback'], ['A', 'risk']] as [string, string][],
  1026→    C: [['D', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1027→    D: [['E', 'up'], ['B', 'pullback'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1028→    E: [['F', 'up'], ['D', 'pullback'], ['G', 'pullback'], ['H', 'down'], ['R1', 'terminal']] as [string, string][],
  1029→    F: [['R1', 'terminal'], ['E', 'pullback'], ['D', 'down'], ['G', 'risk']] as [string, string][],
  1030→    G: [['B', 'up'], ['A', 'pullback'], ['H', 'down'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1031→    H: [['G', 'pullback'], ['I', 'down'], ['B', 'up'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1032→    I: [['H', 'pullback'], ['J', 'down'], ['G', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1033→    J: [['I', 'pullback'], ['K', 'down'], ['H', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1034→    K: [['J', 'pullback'], ['I', 'up'], ['L', 'risk'], ['R4', 'terminal'], ['R5', 'terminal']] as [string, string][],
  1035→    L: [['C', 'up'], ['H', 'pullback'], ['J', 'down'], ['K', 'risk'], ['R5', 'terminal']] as [string, string][],
  1036→  },
  1037→  // which result scenario each terminal node maps to
  1038→  terminalMap: { R1: 'R1', R2: 'R2', R3: 'R3', R4: 'R4', R5: 'R5' } as Record<string, string>,
  1039→  startNode: 'A' as string,
  1040→};
  1041→
  1042→// ═══════════════════════════════════════════════════════════════════════════════
  1043→// LAYER 5 — Edge Weight Calculation
  1044→// ═══════════════════════════════════════════════════════════════════════════════
  1045→
  1046→/**
  1047→ * Calculates transition edge weights for the VDss graph.
  1048→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
  1049→ * to determine the relative probability of up/down/pullback/risk transitions.
  1050→ */
  1051→function calculateEdgeWeights(
  1052→  bullConsensus: number,
  1053→  adx: number,
  1054→  mlModel: AdaptiveWeightModel
  1055→): { up: number; down: number; pullback: number; risk: number } {
  1056→  // Extract ML coefficient insights if model is trained
  1057→  let trend_coef = 1 / 3;
  1058→  let momentum_coef = 1 / 3;
  1059→  let volatility_coef = 1 / 3;
  1060→
  1061→  if (mlModel.isTrained && mlModel.weights) {
  1062→    const coefs = mlModel.getCoefficients();
  1063→    // Map feature indices to coefficient categories
  1064→    // FEATURES_ORDER: ['rsi','mfi','cci','adx','macd','stoch','bb','ma21','ma100','ema','atr','trend','sr','stochCross','macdCross','div']
  1065→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER
  1066→    const rsiIdx = 0;     // 'rsi' in FEATURES_ORDER
  1067→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
  1068→
  1069→    const absTrend = Math.abs(coefs[trendIdx] ?? 0);
  1070→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
  1071→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
  1072→
  1073→    const coefSum = absTrend + absRsi + absAtr;
  1074→    if (coefSum > 1e-10) {
  1075→      trend_coef = absTrend / coefSum;
  1076→      momentum_coef = absRsi / coefSum;
  1077→      volatility_coef = absAtr / coefSum;
  1078→    }
  1079→  }
  1080→
  1081→  // Base weights driven by bull consensus
  1082→  const up_base = bullConsensus * 0.7 + 0.15;
  1083→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
  1084→  const pullback_base = 0.25;
  1085→  const risk_base = 0.12 * (1.2 - adx / 100);
  1086→
  1087→  // Apply ML coefficient adjustments
  1088→  const up = clamp(
  1089→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
  1090→    0.05, 0.95
  1091→  );
  1092→  const down = clamp(
  1093→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
  1094→    0.05, 0.95
  1095→  );
  1096→  const pullback = clamp(
  1097→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
  1098→    0.05, 0.50
  1099→  );
  1100→  const risk = clamp(
  1101→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
  1102→    0.02, 0.30
  1103→  );
  1104→
  1105→  return { up, down, pullback, risk };
  1106→}
  1107→
  1108→// ═══════════════════════════════════════════════════════════════════════════════
  1109→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)
  1110→// ═══════════════════════════════════════════════════════════════════════════════
  1111→
  1112→/**
  1113→ * Performs iterative DFS from the start node through the VDss graph,
  1114→ * computing raw path probabilities, then calibrating them against
  1115→ * Layer 4 scenario probabilities.
  1116→ *
  1117→ * Returns:
  1118→ * - calibratedPaths: Map<pathString, probability>
  1119→ * - calibrationFactors: per-scenario multiplier used
  1120→ * - scenarioSums: raw probability sums per scenario before calibration
  1121→ */
  1122→function calculatePathProbabilities(
  1123→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
  1124→  scenarioProbs: Record<string, number>
  1125→): {
  1126→  calibratedPaths: Map<string, number>;
  1127→  calibrationFactors: Record<string, number>;
  1128→  scenarioSums: Record<string, number>;
  1129→} {
  1130→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
  1131→  const MAX_PATHS = 200;
  1132→
  1133→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
  1134→  type StackEntry = [string, string, number];
  1135→
  1136→  // Collect all terminal paths with raw probabilities
  1137→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
  1138→
  1139→  // Iterative DFS
  1140→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
  1141→
  1142→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
  1143→    const [node, pathStr, pathProb] = stack.pop()!;
  1144→
  1145→    const edges = outgoing[node as keyof typeof outgoing];
  1146→    if (!edges) continue;
  1147→
  1148→    // Normalize outgoing edge weights to probabilities for this node
  1149→    // Map edge types to their weights
  1150→    let totalWeight = 0;
  1151→    const edgeWeightsList: number[] = [];
  1152→    for (const [, edgeType] of edges) {
  1153→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
  1154→      edgeWeightsList.push(w);
  1155→      totalWeight += w;
  1156→    }
  1157→
  1158→    if (totalWeight < 1e-10) continue;
  1159→
  1160→    // Expand edges
  1161→    for (let i = 0; i < edges.length; i++) {
  1162→      const [target, edgeType] = edges[i];
  1163→      const edgeProb = edgeWeightsList[i] / totalWeight;
  1164→      const newProb = pathProb * edgeProb;
  1165→      const newPath = pathStr + '→' + target;
  1166→
  1167→      if (edgeType === 'terminal') {
  1168→        // Reached a terminal node
  1169→        const scenario = terminalMap[target] ?? 'R3';
  1170→        rawPaths.push({ path: newPath, scenario, prob: newProb });
  1171→      } else {
  1172→        // Continue DFS
  1173→        stack.push([target, newPath, newProb]);
  1174→      }
  1175→    }
  1176→  }
  1177→
  1178→  // Group raw probabilities by scenario
  1179→  const rawSums: Record<string, number> = { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0 };
  1180→  for (const rp of rawPaths) {
  1181→    rawSums[rp.scenario] = (rawSums[rp.scenario] ?? 0) + rp.prob;
  1182→  }
  1183→
  1184→  // Compute calibration factors: scenarioProbs[scenario] / rawSum[scenario]
  1185→  const calibrationFactors: Record<string, number> = {};
  1186→  for (const scenario of ['R1', 'R2', 'R3', 'R4', 'R5']) {
  1187→    const rawSum = rawSums[scenario] ?? 0;
  1188→    const targetProb = (scenarioProbs[scenario] ?? 20) / 100; // convert from percentage
  1189→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
  1190→  }
  1191→
  1192→  // Apply calibration to each path
  1193→  const calibratedPaths = new Map<string, number>();
  1194→  for (const rp of rawPaths) {
  1195→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
  1196→    calibratedPaths.set(rp.path, rp.prob * factor);
  1197→  }
  1198→
  1199→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
  1200→}
  1201→
  1202→// ═══════════════════════════════════════════════════════════════════════════════
  1203→// LAYER 4 — Scenario Probability Calculation (Adaptive with ML)
  1204→// ═══════════════════════════════════════════════════════════════════════════════
  1205→
  1206→/**
  1207→ * Calculates the 5 VDss scenario probabilities (R1-R5) that sum to 100.
  1208→ * Uses bull/bear consensus, key indicator values, S/R levels, and
  1209→ * ML-derived adaptive factors for fine-tuning.
  1210→ */
  1211→function calculateScenarioProbabilities(
  1212→  bullConsensus: number,
  1213→  price: number,
  1214→  R1: number,
  1215→  S1: number,
  1216→  MA100: number,
  1217→  rsi: number,
  1218→  mfi: number,
  1219→  stochK: number,
  1220→  mlModel: AdaptiveWeightModel
  1221→): { R1: number; R2: number; R3: number; R4: number; R5: number; factors: { momentum: number; volatility: number; trend: number } } {
  1222→  // Distance metrics to key levels (spec uses exp(-3 * ...))
  1223→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0;
  1224→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0;
  1225→
  1226→  // Binary risk flags (per spec) — MFI only used when volume is available
  1227→  const overboughtRisk = (rsi > 70 || (mfi > 80 && mfi !== 50) || stochK > 80) ? 1 : 0;
  1228→  const oversoldBounce = (rsi < 30 || (mfi < 20 && mfi !== 50) || stochK < 20) ? 1 : 0;
  1229→
  1230→  // Extract adaptive factors from ML model if trained
  1231→  let momentum = 0.7;
  1232→  let volatility = 0.5;
  1233→  let trend = 0.6;
  1234→
  1235→  if (mlModel.isTrained && mlModel.weights) {
  1236→    const coefs = mlModel.getCoefficients();
  1237→    momentum = sigmoid(coefs[0] ?? 0);   // rsi coefficient → momentum factor
  1238→    volatility = sigmoid(coefs[10] ?? 0); // atr coefficient → volatility factor
  1239→    trend = sigmoid(coefs[11] ?? 0);     // trend coefficient → trend factor
  1240→  }
  1241→
  1242→  // Below MA100 flag
  1243→  const belowMA100 = price < MA100 ? 1 : 0;
  1244→
  1245→  // ── Raw scenario probabilities (matching user spec) ─────────────────────
  1246→
  1247→  // R1: Strong Bullish — aggressive breakout
  1248→  const raw_R1 =
  1249→    bullConsensus * distR1 * (1 - overboughtRisk * momentum * 0.7) +
  1250→    oversoldBounce * 0.3 * distR1 * momentum;
  1251→
  1252→  // R2: Gradual Uptrend — steady rise
  1253→  const raw_R2 =
  1254→    ((1 - Math.abs(bullConsensus - 0.65)) * 0.8 +
  1255→    (bullConsensus > 0.4 ? 0.2 : 0)) * trend;
  1256→
  1257→  // R3: Range-bound
  1258→  const raw_R3 =
  1259→    (1 - Math.abs(bullConsensus - 0.5)) * 0.6 * (1 + (1 - volatility) * 0.2);
  1260→
  1261→  // R4: Moderate Correction
  1262→  const raw_R4 =
  1263→    ((1 - bullConsensus) * distS1 * 0.7 +
  1264→    overboughtRisk * 0.5 * momentum) *
  1265→    (0.8 + 0.2 * (1 - trend));
  1266→
  1267→  // R5: Deep Correction
  1268→  const raw_R5 =
  1269→    ((1 - bullConsensus) ** 2 * 0.5 + belowMA100 * 0.2) *
  1270→    (1 + (1 - trend) * 0.3);
  1271→
  1272→  // ── Normalize to sum = 100 ───────────────────────────────────────────
  1273→  const sum = raw_R1 + raw_R2 + raw_R3 + raw_R4 + raw_R5;
  1274→  const pR1 = sum > 0 ? Math.round(raw_R1 / sum * 100) : 20;
  1275→  const pR2 = sum > 0 ? Math.round(raw_R2 / sum * 100) : 20;
  1276→  const pR3 = sum > 0 ? Math.round(raw_R3 / sum * 100) : 20;
  1277→  const pR4 = sum > 0 ? Math.round(raw_R4 / sum * 100) : 20;
  1278→  const pR5 = 100 - (pR1 + pR2 + pR3 + pR4);
  1279→
  1280→  return {
  1281→    R1: pR1, R2: pR2, R3: pR3, R4: pR4, R5: pR5,
  1282→    factors: { momentum, volatility, trend },
  1283→  };
  1284→}
  1285→
  1286→// ═══════════════════════════════════════════════════════════════════════════════
  1287→// MAIN ANALYSIS FUNCTION

  1293→    return {

  1299→      supportStrengths: [], resistanceStrengths: [], priceTargets: [],
  1300→      trend: {

  1335→  const mfi = hasVolume ? calcMFI(data) : 50;
  1336→  const cci = calcCCI(data);
  1337→  const stoch = calcStochastic(data);
  1338→  const williamsR = calcWilliamsR(data);
  1339→  const macd = calcMACD(closes);
