    15→// === Helper Functions =========================================================
    16→
    17→function safeNum(v: number, fallback: number = 0): number {
    18→  return (typeof v === 'number' && isFinite(v)) ? v : fallback;
    19→}
    20→
    21→function clamp(v: number, min: number, max: number): number {
    22→  return Math.max(min, Math.min(max, safeNum(v, (min + max) / 2)));
    23→}
    24→
    25→function normalize(arr: number[]): number[] {
    26→  const safe = arr.map(v => safeNum(v, 0));
    27→  const sum = safe.reduce((a, b) => a + b, 0);
    28→  if (sum <= 0) return arr.map(() => 1 / arr.length);
    29→  return safe.map((v) => v / sum);
    30→}
    31→
    32→function sigmoid(x: number): number {
    33→  const sx = safeNum(x, 0);
    34→  if (Math.abs(sx) > 500) return sx > 0 ? 1 : 0; // prevent Infinity
    35→  return 1 / (1 + Math.exp(-sx));
    36→}

  1024→function buildContext(input: GraphInput): ProbContext {
  1025→  const adxNorm = clamp(input.adx / 60, 0, 1);
  1026→  const srProximity = clamp(
  1027→    1 - Math.min(input.distToR1, input.distToS1) * 10,
  1028→    0,
  1029→    1
  1030→  );
  1031→  const oscillatorExtreme = clamp(
  1032→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
  1033→    0,
  1034→    1
  1035→  );
  1036→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1037→
  1038→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1039→  const overboughtRisk =
  1040→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1041→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1042→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1043→
  1044→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1045→  const oversoldBounce =
  1046→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1047→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1048→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1049→
  1050→  return {
  1051→    input,
  1052→    adxNorm,
  1053→    srProximity,
  1054→    oscillatorExtreme,
  1055→    divergenceProxy,
  1056→    overboughtRisk,
  1057→    oversoldBounce,
  1058→  };
  1059→}
  1060→
  1061→// === Node Probability Computation =============================================
  1062→// For each non-terminal node, returns an array of probabilities for its
  1063→// outgoing edges. The order MUST match the order of edges in createEdges().
  1064→// =============================================================================
  1065→
  1066→function computeNodeProbabilities(
  1067→  nodeId: string,
  1068→  ctx: ProbContext,
  1069→  _numChildren: number
  1070→): number[] {
  1071→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1072→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1073→
  1074→  switch (nodeId) {
  1075→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1076→    case 'ROOT': {
  1077→      const trendW = adxNorm * 0.5 + input.mlTrend * 0.18 + 0.1;
  1078→      const breakW = srProximity * 0.3 + (1 - adxNorm) * 0.08 + 0.08;
  1079→      const revW = ctx.divergenceProxy * 0.35 + (1 - adxNorm) * 0.12 + 0.06;
  1080→      return normalize([trendW, breakW, revW]);
  1081→    }
  1082→
  1083→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1084→    case 'N_TREND': {
  1085→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1086→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1087→      const flatUncertainty = (1 - Math.abs(input.bullConsensus - 0.5) * 2);
  1088→      const flat = flatUncertainty * clamp(1 - adxNorm * 1.5, 0, 1) * 0.45;
  1089→      return normalize([bull, bear, flat]);
  1090→    }
  1091→
  1092→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1093→    case 'N_T_BULL': {
  1094→      const momHigh =
  1095→        clamp(
  1096→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1097→            (input.macdHist > 0
  1098→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1099→              : 0),
  1100→          0,
  1101→          1
  1102→        ) * input.mlMomentum;
  1103→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1104→      return normalize([momHigh, momMod, overboughtRisk]);
  1105→    }
  1106→
  1107→    // ── N_T_B_MOM_HIGH → [SC8, SC9, SC7] ────────────────────────────────────
  1108→    // Strong bullish momentum — all three outcomes are bullish (no range)
  1109→    case 'N_T_B_MOM_HIGH': {
  1110→      const pSC8 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1111→      const pSC9 = 0.15 + input.strengthIndex * 0.2;
  1112→      const pSC7 = 0.35 * input.confidenceIndex + 0.15;
  1113→      return normalize([pSC8, pSC9, pSC7]);
  1114→    }
  1115→
  1116→    // ── N_T_B_MOM_MOD → [SC7, SC6, SC5] ─────────────────────────────────────
  1117→    // Moderate bullish — small explicit SC5 based on uncertainty
  1118→    case 'N_T_B_MOM_MOD': {
  1119→      const pSC7 = 0.55 * input.confidenceIndex + 0.2;
  1120→      const pSC6 = 0.55 * (1 - input.confidenceIndex) + 0.15;
  1121→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.02, 0.15);
  1122→      return normalize([pSC7, pSC6, pSC5]);
  1123→    }
  1124→
  1125→    // ── N_T_B_OVERBOUGHT → [SC4, SC1, SC9] ──────────────────────────────────
  1126→    // Overbought in uptrend — correction or last-gasp surge (no range)
  1127→    case 'N_T_B_OVERBOUGHT': {
  1128→      const pSC4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1129→      const pSC1 = 0.15 + input.strengthIndex * 0.15;
  1130→      const pSC9 = 0.1 + input.mlMomentum * 0.2;
  1131→      return normalize([pSC4, pSC1, pSC9]);
  1132→    }
  1133→
  1134→    // ── N_T_BEAR → [N_T_BE_MOM_LOW, N_T_BE_MOM_MOD, N_T_BE_OVERSOLD] ────
  1135→    case 'N_T_BEAR': {
  1136→      const momLow =
  1137→        clamp(
  1138→          (input.momentum < 0 ? Math.abs(input.momentum) / 50 : 0) +
  1139→            (input.macdHist < 0
  1140→              ? (Math.abs(input.macdHist) / (atrSafe * 0.5)) * 0.3
  1141→              : 0),
  1142→          0,
  1143→          1
  1144→        ) * input.mlMomentum;
  1145→      const momMod = (1 - momLow) * 0.6 * input.mlMomentum;
  1146→      return normalize([momLow, momMod, oversoldBounce]);
  1147→    }
  1148→
  1149→    // ── N_T_BE_MOM_LOW → [SC2, SC1, SC3] ────────────────────────────────────
  1150→    // Strong bearish momentum — all three outcomes are bearish (no range)
  1151→    case 'N_T_BE_MOM_LOW': {
  1152→      const pSC2 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1153→      const pSC1 = 0.15 + input.strengthIndex * 0.2;
  1154→      const pSC3 = 0.35 * input.confidenceIndex + 0.15;
  1155→      return normalize([pSC2, pSC1, pSC3]);
  1156→    }
  1157→
  1158→    // ── N_T_BE_MOM_MOD → [SC3, SC4, SC5] ────────────────────────────────────
  1159→    // Moderate bearish — small explicit SC5 based on uncertainty
  1160→    case 'N_T_BE_MOM_MOD': {
  1161→      const pSC3 = 0.55 * input.confidenceIndex + 0.2;
  1162→      const pSC4 = 0.55 * (1 - input.confidenceIndex) + 0.15;
  1163→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.02, 0.15);
  1164→      return normalize([pSC3, pSC4, pSC5]);
  1165→    }
  1166→
  1167→    // ── N_T_BE_OVERSOLD → [SC6, SC9, SC4] ───────────────────────────────────
  1168→    // Oversold in downtrend — bounce or mild continuation (no range)
  1169→    case 'N_T_BE_OVERSOLD': {
  1170→      const pSC6 = 0.55 + (1 - input.mlMomentum) * 0.2;
  1171→      const pSC9 = 0.15 + input.strengthIndex * 0.15;
  1172→      const pSC4 = 0.1 + input.mlMomentum * 0.2;
  1173→      return normalize([pSC6, pSC9, pSC4]);
  1174→    }
  1175→
  1176→    // ── N_T_FLAT → [N_T_F_VOL_LOW, N_T_F_VOL_HIGH, N_T_F_VOL_MOD] ────────
  1177→    case 'N_T_FLAT': {
  1178→      const volLow =
  1179→        clamp(1 - (input.atr / input.price) * 20, 0, 1) *
  1180→        (1 - input.mlVolatility) *
  1181→        0.7;
  1182→      const volHigh =
  1183→        clamp((input.atr / input.price) * 20, 0, 1) * input.mlVolatility * 0.7;
  1184→      const volMod = (1 - volLow - volHigh) * 0.5 + 0.1;
  1185→      return normalize([volLow, volHigh, volMod]);
  1186→    }
  1187→
  1188→    // ── N_T_F_VOL_LOW → [SC5, SC6, SC4] ─────────────────────────────────────
  1189→    case 'N_T_F_VOL_LOW': {
  1190→      const pSC5 = 0.55 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1191→      const pSC6 = 0.2 * input.bullConsensus + 0.08;
  1192→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.08;
  1193→      return normalize([pSC5, pSC6, pSC4]);
  1194→    }
  1195→
  1196→    // ── N_T_F_VOL_HIGH → [SC9, SC1, SC5] ────────────────────────────────────
  1197→    // High volatility shock — directional shock much more likely than range
  1198→    case 'N_T_F_VOL_HIGH': {
  1199→      const pSC9 = 0.4 * input.bullConsensus + 0.15;
  1200→      const pSC1 = 0.4 * (1 - input.bullConsensus) + 0.15;
  1201→      const pSC5 = clamp((1 - input.mlVolatility) * 0.12, 0.02, 0.12);
  1202→      return normalize([pSC9, pSC1, pSC5]);
  1203→    }
  1204→
  1205→    // ── N_T_F_VOL_MOD → [SC6, SC4, SC5] ─────────────────────────────────────
  1206→    // Moderate volatility — directional more likely than range
  1207→    case 'N_T_F_VOL_MOD': {
  1208→      const pSC6 = 0.38 * input.bullConsensus + 0.15;
  1209→      const pSC4 = 0.38 * (1 - input.bullConsensus) + 0.15;
  1210→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.03, 0.15);
  1211→      return normalize([pSC6, pSC4, pSC5]);
  1212→    }
  1213→
  1214→    // ── N_BREAK → [N_B_UP, N_B_DOWN, N_B_NONE] ────────────────────────────
  1215→    case 'N_BREAK': {
  1216→      const upProb = input.bullConsensus * srProximity * 0.75 + 0.08;
  1217→      const downProb = (1 - input.bullConsensus) * srProximity * 0.75 + 0.08;
  1218→      const noneProb = (1 - srProximity) * 0.35 + 0.05;
  1219→      return normalize([upProb, downProb, noneProb]);
  1220→    }
  1221→
  1222→    // ── N_B_UP → [N_B_U_VOL_C, N_B_U_VOL_W] ─────────────────────────────
  1223→    case 'N_B_UP': {
  1224→      let volConfirm: number;
  1225→      let volWeak: number;
  1226→      if (input.hasVolume) {
  1227→        volConfirm = clamp(input.mfi / 60, 0, 1) * 0.6 + 0.2;
  1228→        volWeak = 1 - volConfirm;
  1229→      } else {
  1230→        volConfirm = 0.4;
  1231→        volWeak = 0.6;
  1232→      }
  1233→      return normalize([volConfirm, volWeak]);
  1234→    }
  1235→
  1236→    // ── N_B_U_VOL_C → [SC7, SC8, SC9] ──────────────────────────────────────
  1237→    // Confirmed bullish breakout — all bullish, no range
  1238→    case 'N_B_U_VOL_C': {
  1239→      const pSC7 = 0.45 + input.confidenceIndex * 0.15;
  1240→      const pSC8 = 0.2 + input.strengthIndex * 0.15;
  1241→      const pSC9 = 0.15 + input.strengthIndex * 0.1;
  1242→      return normalize([pSC7, pSC8, pSC9]);
  1243→    }
  1244→
  1245→    // ── N_B_U_VOL_W → [SC6, SC4, SC5] ──────────────────────────────────────
  1246→    // Weak volume breakout — fake breakout risk, capped SC5
  1247→    case 'N_B_U_VOL_W': {
  1248→      const pSC6 = 0.4 + input.bullConsensus * 0.15;
  1249→      const pSC4 = 0.3 + (1 - input.bullConsensus) * 0.15;
  1250→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.12, 0.02, 0.12);
  1251→      return normalize([pSC6, pSC4, pSC5]);
  1252→    }
  1253→
  1254→    // ── N_B_DOWN → [N_B_D_VOL_C, N_B_D_VOL_W] ────────────────────────────
  1255→    case 'N_B_DOWN': {
  1256→      let volConfirm: number;
  1257→      if (input.hasVolume) {
  1258→        volConfirm = clamp((100 - input.mfi) / 60, 0, 1) * 0.6 + 0.2;
  1259→      } else {
  1260→        volConfirm = 0.4;
  1261→      }
  1262→      const volWeak = 1 - volConfirm;
  1263→      return normalize([volConfirm, volWeak]);
  1264→    }
  1265→
  1266→    // ── N_B_D_VOL_C → [SC3, SC2, SC1] ──────────────────────────────────────
  1267→    // Confirmed bearish breakout — all bearish, no range
  1268→    case 'N_B_D_VOL_C': {
  1269→      const pSC3 = 0.45 + input.confidenceIndex * 0.15;
  1270→      const pSC2 = 0.2 + input.strengthIndex * 0.15;
  1271→      const pSC1 = 0.15 + input.strengthIndex * 0.1;
  1272→      return normalize([pSC3, pSC2, pSC1]);
  1273→    }
  1274→
  1275→    // ── N_B_D_VOL_W → [SC4, SC6, SC5] ──────────────────────────────────────
  1276→    // Weak volume bearish breakout — fake breakout risk, capped SC5
  1277→    case 'N_B_D_VOL_W': {
  1278→      const pSC4 = 0.4 + (1 - input.bullConsensus) * 0.15;
  1279→      const pSC6 = 0.3 + input.bullConsensus * 0.15;
  1280→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.12, 0.02, 0.12);
  1281→      return normalize([pSC4, pSC6, pSC5]);
  1282→    }
  1283→
  1284→    // ── N_B_NONE → [SC4, SC5, SC6] ──────────────────────────────────────────
  1285→    case 'N_B_NONE': {
  1286→      const pSC5 = 0.45 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1287→      const pSC6 = 0.2 * input.bullConsensus + 0.08;
  1288→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.08;
  1289→      return normalize([pSC4, pSC5, pSC6]);
  1290→    }
  1291→
  1292→    // ── N_REVERSAL → [N_R_BULL, N_R_BEAR, N_R_NONE] ──────────────────────
  1293→    case 'N_REVERSAL': {
  1294→      const bullRev = overboughtRisk;
  1295→      const bearRev = oversoldBounce;
  1296→      const maxRevSignal = Math.max(overboughtRisk, oversoldBounce);
  1297→      const noneRev = 1 - maxRevSignal * 0.85;
  1298→      return normalize([bullRev, bearRev, noneRev]);
  1299→    }
  1300→
  1301→    // ── N_R_BULL → [N_R_B_DIV, N_R_B_CANDLE, N_R_B_SR] ──────────────────
  1302→    case 'N_R_BULL': {
  1303→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1304→      const candleProb =
  1305→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1306→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1307→      return normalize([divProb, candleProb, srProb]);
  1308→    }
  1309→
  1310→    // ── N_R_B_DIV → [SC6, SC8, SC5] ────────────────────────────────────────
  1311→    // Bullish divergence — explicit small SC5
  1312→    case 'N_R_B_DIV': {
  1313→      const pSC6 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1314→      const pSC8 = 0.2 + input.mlMomentum * 0.15;
  1315→      const pSC5 = clamp((1 - adxNorm) * 0.1, 0.02, 0.1);
  1316→      return normalize([pSC6, pSC8, pSC5]);
  1317→    }
  1318→
  1319→    // ── N_R_B_CANDLE → [SC7, SC6, SC5] ─────────────────────────────────────
  1320→    // Bullish candle pattern — explicit small SC5
  1321→    case 'N_R_B_CANDLE': {
  1322→      const pSC7 = 0.5 * input.confidenceIndex + 0.2;
  1323→      const pSC6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1324→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.1, 0.02, 0.1);
  1325→      return normalize([pSC7, pSC6, pSC5]);
  1326→    }
  1327→
  1328→    // ── N_R_B_SR → [SC6, SC5, SC4] ─────────────────────────────────────────
  1329→    // Support bounce — explicit small SC5
  1330→    // Edge order: SC6, SC5, SC4 — probabilities computed then reordered
  1331→    case 'N_R_B_SR': {
  1332→      const pSC6 = 0.5 + input.srAvgStrength * 0.2;
  1333→      const pSC4 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1334→      const pSC5 = clamp((1 - input.srAvgStrength) * 0.1, 0.02, 0.1);
  1335→      // Return in edge order: [SC6, SC5, SC4]
  1336→      return normalize([pSC6, pSC5, pSC4]);
  1337→    }
  1338→
  1339→    // ── N_R_BEAR → [N_R_BE_DIV, N_R_BE_CANDLE, N_R_BE_SR] ───────────────
  1340→    // Mirror of N_R_BULL
  1341→    case 'N_R_BEAR': {
  1342→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1343→      const candleProb =
  1344→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1345→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1346→      return normalize([divProb, candleProb, srProb]);
  1347→    }
  1348→
  1349→    // ── N_R_BE_DIV → [SC4, SC2, SC5] ───────────────────────────────────────
  1350→    // Bearish divergence — explicit small SC5
  1351→    case 'N_R_BE_DIV': {
  1352→      const pSC4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1353→      const pSC2 = 0.2 + input.mlMomentum * 0.15;
  1354→      const pSC5 = clamp((1 - adxNorm) * 0.1, 0.02, 0.1);
  1355→      return normalize([pSC4, pSC2, pSC5]);
  1356→    }
  1357→
  1358→    // ── N_R_BE_CANDLE → [SC3, SC4, SC5] ────────────────────────────────────
  1359→    // Bearish candle pattern — explicit small SC5
  1360→    case 'N_R_BE_CANDLE': {
  1361→      const pSC3 = 0.5 * input.confidenceIndex + 0.2;
  1362→      const pSC4 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1363→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.1, 0.02, 0.1);
  1364→      return normalize([pSC3, pSC4, pSC5]);
  1365→    }
  1366→
  1367→    // ── N_R_BE_SR → [SC4, SC5, SC6] ─────────────────────────────────────────
  1368→    // Resistance rejection — explicit small SC5
  1369→    // Edge order: SC4, SC5, SC6 — probabilities computed then reordered
  1370→    case 'N_R_BE_SR': {
  1371→      const pSC4 = 0.5 + input.srAvgStrength * 0.2;
  1372→      const pSC6 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1373→      const pSC5 = clamp((1 - input.srAvgStrength) * 0.1, 0.02, 0.1);
  1374→      // Return in edge order: [SC4, SC5, SC6]
  1375→      return normalize([pSC4, pSC5, pSC6]);
  1376→    }
  1377→
  1378→    // ── N_R_NONE → [SC4, SC5, SC6] ──────────────────────────────────────────
  1379→    case 'N_R_NONE': {
  1380→      const pSC5 = 0.4 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1381→      const pSC6 = 0.2 * input.bullConsensus + 0.1;
  1382→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.1;
  1383→      return normalize([pSC4, pSC5, pSC6]);
  1384→    }
  1385→
  1386→    // ── Terminal nodes (SC1–SC9) have no outgoing edges ─────────────────────
  1387→    default:
  1388→      return [];
  1389→  }
  1390→}
  1391→
  1392→// === Graph Traversal ==========================================================
  1393→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1394→// =============================================================================
  1395→
  1396→function traverseGraph(
  1397→  nodeId: string,
  1398→  pathProb: number,
  1399→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1400→  ctx: ProbContext,
  1401→  adj: Map<string, number[]>,
  1402→  edges: GraphEdge[],
  1403→  edgeProbabilities: Record<number, number>,
  1404→  nodeAccum: Record<string, number>,
  1405→  scenarios: Record<string, number>,
  1406→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1407→): void {
  1408→  // Accumulate path probability at this node
  1409→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1410→
  1411→  const outEdgeIndices = adj.get(nodeId);
  1412→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1413→
  1414→  // Compute outgoing edge probabilities for this node
  1415→  const probs = computeNodeProbabilities(
  1416→    nodeId,
  1417→    ctx,
  1418→    outEdgeIndices.length
  1419→  );
  1420→
  1421→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1422→    const edgeIdx = outEdgeIndices[i];
  1423→    const edge = edges[edgeIdx];
  1424→    const prob = probs[i] || 0;
  1425→
  1426→    // Store edge probability
  1427→    edgeProbabilities[edgeIdx] = prob;
  1428→
  1429→    // Determine branch for child nodes
  1430→    let childBranch = currentBranch;
  1431→    if (nodeId === 'ROOT') {
  1432→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1433→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1434→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1435→    }
  1436→
  1437→    const childPathProb = pathProb * prob;
  1438→
  1439→    // Check if target is a terminal node (SC1–SC9)
  1440→    const isTerminal =
  1441→      edge.to.length === 3 &&
  1442→      edge.to[0] === 'S' &&
  1443→      edge.to[1] === 'C' &&
  1444→      edge.to[2] >= '1' &&
  1445→      edge.to[2] <= '9';
  1446→
  1447→    if (isTerminal) {
  1448→      // Accumulate scenario probability
  1449→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1450→      // Accumulate branch contribution
  1451→      if (childBranch) {
  1452→        contributions[edge.to][childBranch] += childPathProb;
  1453→      }
  1454→    }
  1455→
  1456→    // Recurse into child
  1457→    traverseGraph(
  1458→      edge.to,
  1459→      childPathProb,
  1460→      childBranch,
  1461→      ctx,
  1462→      adj,
  1463→      edges,
  1464→      edgeProbabilities,
  1465→      nodeAccum,
  1466→      scenarios,
  1467→      contributions
  1468→    );
  1469→  }
  1470→}

  1515→    maAlignment, momentum, awesomeOsc, fisherTransform,

  1569→  for (let i = 1; i <= 9; i++) {

  1572→  }
  1573→

  1576→    key: k,

  1579→
  1580→  // Re-normalize to sum=100
  1581→  const clampedSum = clamped.reduce((s, c) => s + c.value, 0);
  1582→  const finalPcts: Record<string, number> = {};

  1584→    // Distribute proportionally and round, then fix rounding errors

  1586→    const entries = clamped
  1587→      .map((c) => {
  1588→        const raw = (c.value / clampedSum) * 100;
  1589→        return { key: c.key, raw };

  1591→      .sort((a, b) => b.raw - a.raw);
  1592→
  1593→    for (let i = 0; i < entries.length; i++) {
  1594→      const isLast = i === entries.length - 1;

  1597→      } else {
