  1110→/**
  1111→ * Deterministic seeded shuffle using Fisher-Yates.
  1112→ */
  1113→function seededShuffle<T>(array: T[], seed: number): T[] {
  1114→  const result = [...array];
  1115→  let s = seed;
  1116→  for (let i = result.length - 1; i > 0; i--) {
  1117→    s = (s * 1664525 + 1013904223) & 0xffffffff;
  1118→    const j = s % (i + 1);
  1119→    [result[i], result[j]] = [result[j], result[i]];
  1120→  }
  1121→  return result;
  1122→}

  1124→/**
  1125→ * Get `count` phrases from a section/subcategory, deterministically shuffled by seed.
  1126→ * Returns fewer phrases if the subcategory doesn't have enough.
  1127→ */
  1128→export function getPhrases(section: string, subcategory: string, seed: number, count: number): string[] {
  1129→  const sec = PHRASE_BANK[section];
  1130→  if (!sec) return [];
  1131→  const phrases = sec.phrases[subcategory];
  1132→  if (!phrases || phrases.length === 0) return [];
  1133→  const shuffled = seededShuffle(phrases, seed);
  1134→  return shuffled.slice(0, Math.min(count, shuffled.length));
  1135→}
