   137→/* --─ Helpers -------------------------------------------- */
   138→
   139→function formatNum(num: number): string {
   140→  if (num == null) return '—';
   141→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 0 }).format(num);
   142→}
   143→
   144→function formatIdx(num: number): string {
   145→  if (num == null) return '—';
   146→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 2 }).format(num);
   147→}

   154→    </svg>
   155→  );
   156→}

   158→/* --─ Component ------------------------------------------ */
   159→
   160→export default function SymbolSearch({
   161→  onSelect,
   162→  placeholder = 'جستجوی ابزار مالی ...',
   163→  className,
   164→}: SymbolSearchProps) {
   165→  const [query, setQuery] = React.useState('');
   166→  const [open, setOpen] = React.useState(false);
   167→  const [results, setResults] = React.useState<InstrumentItem[]>([]);
   168→  const [loading, setLoading] = React.useState(false);
   169→  const [activeIndex, setActiveIndex] = React.useState(-1);
   170→  const [totalMatched, setTotalMatched] = React.useState(0);
   171→  const [activeCategory, setActiveCategory] = React.useState<CategoryKey>('all');
   172→  const [activeIndustry, setActiveIndustry] = React.useState<string | null>(null);
   173→  const [showIndustryPicker, setShowIndustryPicker] = React.useState(false);
   174→
   175→  const cacheRef = React.useRef<InstrumentsData | null>(null);
   176→  const tgjuCacheRef = React.useRef<TgjuData | null>(null);
   177→  const fetchRef = React.useRef<Promise<InstrumentsData> | null>(null);
   178→  const tgjuFetchRef = React.useRef<Promise<TgjuData | null> | null>(null);
   179→  const inputRef = React.useRef<HTMLInputElement>(null);
   180→  const dropdownRef = React.useRef<HTMLDivElement>(null);
   181→  const itemRefs = React.useRef<(HTMLDivElement | null)[]>([]);
   182→  const industryScrollRef = React.useRef<HTMLDivElement>(null);
   183→
   184→  /* -- Data fetching -- */
   185→  const fetchData = React.useCallback(async (): Promise<InstrumentsData> => {
   186→    if (cacheRef.current) return cacheRef.current;
   187→    if (fetchRef.current) return fetchRef.current;
   188→    const p = async () => {
   189→      try {
   190→        const r = await fetch('/api/instruments');
   191→        if (!r.ok) throw new Error(`${r.status}`);
   192→        const d: InstrumentsData = await r.json();
   193→        cacheRef.current = d;
   194→        return d;
   195→      } finally {
   196→        fetchRef.current = null;
   197→      }
   198→    };
   199→    fetchRef.current = p();
   200→    return fetchRef.current;
   201→  }, []);
   202→
   203→  const fetchTgjuData = React.useCallback(async (): Promise<TgjuData | null> => {
   204→    if (tgjuCacheRef.current) return tgjuCacheRef.current;
   205→    if (tgjuFetchRef.current) return tgjuFetchRef.current;
   206→    const p = async () => {
   207→      try {
   208→        const r = await fetch('/api/tgju-instruments');
   209→        if (!r.ok) return null;
   210→        const d: TgjuData = await r.json();
   211→        tgjuCacheRef.current = d;
   212→        return d;
   213→      } catch {
   214→        return null;
   215→      } finally {
   216→        tgjuFetchRef.current = null;
   217→      }
   218→    };
   219→    tgjuFetchRef.current = p();
   220→    return p();
   221→  }, []);
   222→
   223→  /* -- Check if category is TGJU-based -- */
   224→  const isTgjuCategory = (cat: CategoryKey) => TGJU_CATEGORIES.has(cat);
   225→
   226→  /* -- Get all items for current category -- */
   227→  const getItemsForCategory = React.useCallback(async (cat: CategoryKey): Promise<InstrumentItem[]> => {
   228→    if (isTgjuCategory(cat)) {
   229→      const tgju = tgjuCacheRef.current || await fetchTgjuData();
   230→      if (!tgju) return [];
   231→      switch (cat) {
   232→        case 'currency': return tgju.currencies;
   233→        case 'gold': return [...tgju.gold, ...tgju.silver];
   234→        case 'gold_etf': return tgju.goldEtfs;
   235→        case 'silver': return tgju.silver;
   236→        case 'crypto': return tgju.crypto;
   237→        case 'world_index': return tgju.worldIndices;
   238→        case 'forex': return tgju.forex;
   239→        case 'energy': return tgju.energy;
   240→        case 'metal': return tgju.metals;
   241→        case 'commodity': return tgju.commodities;
   242→        default: return [];
   243→      }
   244→    }
   245→
   246→    // TSE categories
   247→    const data = cacheRef.current || await fetchData();
   248→    switch (cat) {
   249→      case 'indices': return data.indices;
   250→      case 'stocks':
   251→        return activeIndustry
   252→          ? data.stocks.filter((s) => s.cs === activeIndustry)
   253→          : data.stocks;
   254→      case 'etf': return data.etfs;
   255→      case 'bond': return [...data.bonds, ...data.mortgage];
   256→      case 'derivative': return [...data.futures, ...data.salaf];
   257→      case 'all':
   258→      default: {
   259→        const tgju = tgjuCacheRef.current;
   260→        return [
   261→          ...data.indices,
   262→          ...data.stocks,
   263→          ...data.etfs,
   264→          ...(tgju?.currencies || []),
   265→          ...(tgju?.forex || []),
   266→          ...(tgju?.crypto || []),
   267→          ...(tgju?.gold || []),
   268→          ...(tgju?.silver || []),
   269→          ...(tgju?.goldEtfs || []),
   270→          ...(tgju?.worldIndices || []),
   271→          ...(tgju?.energy || []),
   272→          ...(tgju?.metals || []),
   273→          ...(tgju?.commodities || []),
   274→          ...data.bonds,
   275→          ...data.futures,
   276→          ...data.salaf,
   277→          ...data.mortgage,
   278→        ];
   279→      }
   280→    }
   281→  }, [activeIndustry, fetchData, fetchTgjuData]);
   282→
   283→  /* -- Popular items -- */
   284→  const getPopular = React.useCallback(async () => {
   285→    const data = cacheRef.current || await fetchData();
   286→    const tgju = tgjuCacheRef.current;
   287→    const all = [
   288→      ...(data.indices.map((i) => ({ ...i, tval: i.index || 0 }))),
   289→      ...data.stocks,
   290→      ...data.etfs,
   291→      ...(tgju?.currencies || []),
   292→      ...(tgju?.crypto?.slice(0, 5) || []),
   293→      ...(tgju?.gold?.slice(0, 3) || []),
   294→    ];
   295→    const sorted = [...all].sort((a, b) => (b.tval || 0) - (a.tval || 0)).slice(0, 15);
   296→    setResults(sorted);
   297→    setTotalMatched(all.length);
   298→  }, [fetchData]);
   299→
   300→  /* -- Filter by search query -- */
   301→  const doFilter = React.useCallback((items: InstrumentItem[], q: string) => {
   302→    const n = q.trim().toLowerCase();
   303→    const f = items.filter(
   304→      (s) =>
   305→        s.l18?.toLowerCase().includes(n) ||
   306→        s.l30?.toLowerCase().includes(n) ||
   307→        s.cs?.toLowerCase().includes(n),
   308→    );
   309→    setTotalMatched(f.length);
   310→    setResults(f.slice(0, MAX_RESULTS));
   311→    setActiveIndex(-1);
   312→  }, []);
   313→
   314→  /* -- Show dropdown with data -- */
   315→  const showDropdown = React.useCallback(async (q: string) => {
   316→    setOpen(true);
   317→    await Promise.all([fetchData(), fetchTgjuData()]);
   318→    const items = await getItemsForCategory(activeCategory);
   319→    if (q.trim().length === 0) {
   320→      if (activeCategory === 'all') {
   321→        await getPopular();
   322→      } else {
   323→        setResults(items.slice(0, MAX_RESULTS));
   324→        setTotalMatched(items.length);
   325→      }
   326→    } else {
   327→      doFilter(items, q);
   328→    }
   329→    setLoading(false);
   330→  }, [fetchData, fetchTgjuData, getItemsForCategory, activeCategory, getPopular, doFilter]);
   331→
   332→  /* -- Category change -- */
   333→  const handleCategoryChange = React.useCallback(async (cat: CategoryKey) => {
   334→    setActiveCategory(cat);
   335→    if (cat !== 'stocks') setActiveIndustry(null);
   336→    setShowIndustryPicker(false);
   337→
   338→    const items = await getItemsForCategory(cat);
   339→    if (query.trim().length > 0) {
   340→      doFilter(items, query);
   341→    } else if (cat === 'all') {
   342→      await getPopular();
   343→    } else {
   344→      setResults(items.slice(0, MAX_RESULTS));
   345→      setTotalMatched(items.length);
   346→    }
   347→    setActiveIndex(-1);
   348→  }, [getItemsForCategory, query, doFilter, getPopular]);
   349→
   350→  /* -- Industry change -- */
   351→  const handleIndustryChange = React.useCallback((industry: string | null) => {
   352→    setActiveIndustry(industry);
   353→    setShowIndustryPicker(false);
   354→    if (cacheRef.current) {
   355→      const items = industry
   356→        ? cacheRef.current.stocks.filter((s) => s.cs === industry)
   357→        : cacheRef.current.stocks;
   358→      if (query.trim().length > 0) {
   359→        doFilter(items, query);
   360→      } else {
   361→        setResults(items.slice(0, MAX_RESULTS));
   362→        setTotalMatched(items.length);
   363→      }
   364→      setActiveIndex(-1);
   365→    }
   366→  }, [query, doFilter]);
   367→
   368→  const handleChange = React.useCallback((value: string) => {
   369→    setQuery(value);
   370→    getItemsForCategory(activeCategory).then((items) => {
   371→      if (value.trim().length === 0) {
   372→        if (activeCategory === 'all') getPopular();
   373→        else {
   374→          setResults(items.slice(0, MAX_RESULTS));
   375→          setTotalMatched(items.length);
   376→        }
   377→      } else {
   378→        doFilter(items, value);
   379→      }
   380→    });
   381→  }, [activeCategory, getItemsForCategory, getPopular, doFilter]);
   382→
   383→  const handleFocus = React.useCallback(() => {
   384→    setLoading(true);
   385→    showDropdown(query);
   386→  }, [query, showDropdown]);
   387→
   388→  const selectSymbol = React.useCallback((s: InstrumentItem) => {
   389→    setQuery(s.l18);
   390→    setOpen(false);
   391→    onSelect?.(s.l18, s.category, s.insCode, s.tgjuKey);
   392→    inputRef.current?.blur();
   393→  }, [onSelect]);
   394→
   395→  const handleKeyDown = React.useCallback((e: React.KeyboardEvent) => {
   396→    if (!open || results.length === 0) {
   397→      if (e.key === 'ArrowDown' || e.key === 'Enter') { e.preventDefault(); handleFocus(); }
   398→      return;
   399→    }
   400→    if (e.key === 'ArrowDown') { e.preventDefault(); setActiveIndex((p) => p < results.length - 1 ? p + 1 : 0); }
   401→    else if (e.key === 'ArrowUp') { e.preventDefault(); setActiveIndex((p) => p > 0 ? p - 1 : results.length - 1); }
   402→    else if (e.key === 'Enter') { e.preventDefault(); if (activeIndex >= 0) selectSymbol(results[activeIndex]); }
   403→    else if (e.key === 'Escape') { e.preventDefault(); setOpen(false); inputRef.current?.blur(); }
   404→  }, [open, results, activeIndex, selectSymbol, handleFocus]);
   405→
   406→  React.useEffect(() => {
   407→    if (activeIndex >= 0) itemRefs.current[activeIndex]?.scrollIntoView({ block: 'nearest' });
   408→  }, [activeIndex]);
   409→
   410→  React.useEffect(() => {
   411→    if (!open) return;
   412→    const handler = (e: MouseEvent) => {
   413→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node) &&
   414→          inputRef.current && !inputRef.current.contains(e.target as Node)) {
   415→        setOpen(false);
   416→        setShowIndustryPicker(false);
   417→      }
   418→    };
   419→    document.addEventListener('mousedown', handler);
   420→    return () => document.removeEventListener('mousedown', handler);
   421→  }, [open]);
   422→
   423→  const clear = () => { setQuery(''); setActiveIndustry(null); setActiveCategory('all'); setResults([]); inputRef.current?.focus(); };
   424→  const isPopular = query.trim().length === 0 && activeCategory === 'all';
   425→  const isSearch = query.trim().length > 0;
   426→
   427→  /* -- Render instrument price -- */
   428→  const renderPrice = (item: InstrumentItem) => {
   429→    const isTgju = isTgjuCategory(item.category as CategoryKey);
   430→    if (item.category === 'index') {
   431→      return (
   432→        <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   433→          <span className='text-xs font-semibold text-[#2D2A26]'>{formatIdx(item.pl)}</span>
   434→          <span className={cn(
   435→            'text-[11px] font-bold px-1.5 py-0.5 rounded',
   436→            item.pcp > 0 ? 'bg-emerald-50 text-emerald-700' :
   437→            item.pcp < 0 ? 'bg-red-50 text-red-700' : 'text-[#8A837B]',
   438→          )}>
   439→            {item.pcp > 0 ? '+' : ''}{item.pcp?.toFixed(2)}%
   440→          </span>
   441→        </div>
   442→      );
   443→    }
   444→    // For static TGJU items with no live price, show dash
   445→    if (isTgju && item.pl === 0) {
   446→      return (
   447→        <div className='flex shrink-0 flex-col items-end gap-0.5'>
   448→          <span className='text-xs text-[#8A837B]'>—</span>
   449→        </div>
   450→      );
   451→    }
   452→    const isUp = item.pcp > 0;
   453→    const isDown = item.pcp < 0;
   454→    return (
   455→      <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   456→        <span className='text-xs font-semibold text-[#2D2A26]'>{formatNum(item.pl)}</span>
   457→        <span className={cn(
   458→          'text-[11px] font-bold px-1.5 py-0.5 rounded',
   459→          isUp ? 'bg-emerald-50 text-emerald-700' :
   460→          isDown ? 'bg-red-50 text-red-700' : 'text-[#8A837B]',
   461→        )}>
   462→          {isUp ? '+' : ''}{item.pcp?.toFixed(2)}%
   463→        </span>
   464→      </div>
   465→    );
   466→  };
   467→
   468→  const industries = cacheRef.current?.industries || [];
   469→
   470→  return (
   471→    <div dir='rtl' className={cn('relative', className)}>
   472→
   473→      <div className='relative'>
   474→        <Search className='absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-[#8A837B] pointer-events-none' />
   475→        <Input
   476→          ref={inputRef}
   477→          type='text'
   478→          value={query}
   479→          onChange={(e) => handleChange(e.target.value)}
   480→          onFocus={handleFocus}
   481→          onKeyDown={handleKeyDown}
   482→          placeholder={placeholder}
   483→          suppressHydrationWarning
   484→          className={cn(
   485→            'h-10 w-full rounded-lg border border-[#E5DFD6] bg-white',
   486→            'text-sm text-[#2D2A26] placeholder:text-[#8A837B]',
   487→            'transition-all',
   488→            'focus-visible:border-amber-500/60 focus-visible:ring-amber-500/20 focus-visible:ring-[3px]',
   489→            'pr-9',
   490→          )}
   491→          autoComplete='off' spellCheck={false}
   492→        />
   493→        {query && (
   494→          <button onClick={clear} className='absolute left-3 top-1/2 -translate-y-1/2 text-[#8A837B] hover:text-[#2D2A26] transition-colors' type='button'>
   495→            <X className='w-4 h-4' />
   496→          </button>
   497→        )}
   498→      </div>
   499→
   500→
   501→      {open && (
   502→        <div
   503→          ref={dropdownRef}
   504→          className='absolute top-full left-0 right-0 mt-1.5 z-50 rounded-xl border border-[#E5DFD6] bg-white shadow-xl shadow-black/10 overflow-hidden'
   505→          style={{ width: 'min(660px, 94vw)' }}
