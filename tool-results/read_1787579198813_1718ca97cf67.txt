    45→const toFa = (n: number) => {
    46→  const rounded = Math.round(n);
    47→  if (!isFinite(rounded) || isNaN(rounded)) return '۰';
    48→  return rounded.toLocaleString('fa-IR');
    49→};

    73→export default function Home() {
    74→  const [currentPage, setCurrentPage] = useState<'home' | 'analysis' | 'help' | 'docs' | 'indices'>('home');
    75→  const [data, setData] = useState<AnalysisData | null>(null);
    76→  const [loading, setLoading] = useState(false);
    77→  const [loadingMessage, setLoadingMessage] = useState<string | null>(null);
    78→  const [error, setError] = useState<string | null>(null);
    79→  const [activePanel, setActivePanel] = useState<PanelId>('visual');
    80→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
    81→  const [refreshing, setRefreshing] = useState(false);
    82→
    83→  const lastFetchRef = useRef<FetchParams | null>(null);
    84→  const refreshingRef = useRef(false);
    85→  // Track in-flight data fetches so they can be cancelled on re-fetch/unmount
    86→  const fetchAbortRef = useRef<AbortController | null>(null);
    87→  const refreshAbortRef = useRef<AbortController | null>(null);
    88→
    89→  // ── doRefresh callback ──
    90→  const doRefresh = useCallback(async () => {
    91→    if (refreshingRef.current || !lastFetchRef.current) return;
    92→    const params = lastFetchRef.current;
    93→    refreshingRef.current = true;
    94→    setRefreshing(true);
    95→    // Cancel any previous refresh fetch
    96→    refreshAbortRef.current?.abort('cancelled');
    97→    const controller = new AbortController();
    98→    refreshAbortRef.current = controller;
    99→    const { signal } = controller;
   100→    try {
   101→      // TGJU
   102→      if (params.category && TGJU_CATEGORIES.has(params.category) && params.tgjuKey) {
   103→        const res = await fetch(`/api/tgju-analysis?key=${encodeURIComponent(params.tgjuKey)}`, { signal });
   104→        if (res.ok) {
   105→          const json = await res.json();
   106→          if (json.candles && json.candles.length > 0 && json.ta) {
   107→            setData({ ...json, isTgju: true, category: params.category });
   108→          }
   109→        }
   110→        return;
   111→      }
   112→      // TSE Index (finpy-tse)
   113→      if (params.category === 'tse_index' && params.insCode) {
   114→        const res = await fetch(`/api/index-analysis?webId=${encodeURIComponent(params.insCode)}`, { signal });
   115→        if (res.ok) {
   116→          const json = await res.json();
   117→          if (json.candles && json.candles.length > 0 && json.ta) {
   118→            setData({ ...json, isIndex: true, category: params.category });
   119→          }
   120→        }
   121→        return;
   122→      }
   123→
   124→      // Regular TSE
   125→      const res = await fetch(`/api/analysis?symbol=${encodeURIComponent(params.symbol)}`, { signal });
   126→      if (res.ok) {
   127→        const json = await res.json();
   128→        setData(json);
   129→      }
   130→    } catch {
   131→      // Silently ignore ALL refresh errors (network reset, abort, timeout — best-effort)
   132→    } finally {
   133→      refreshingRef.current = false;
   134→      setRefreshing(false);
   135→    }
   136→  }, []);
   137→
   138→  // ── Auto-refresh every 15 min ──
   139→  useEffect(() => {
   140→    if (!lastFetchRef.current) return;
   141→    const id = setInterval(() => { doRefresh(); }, 15 * 60_000);
   142→    return () => { clearInterval(id); refreshAbortRef.current?.abort('cancelled'); };
   143→  }, [doRefresh]);
   144→
   145→  // ── Refresh on tab focus ──
   146→  useEffect(() => {
   147→    const onFocus = () => { doRefresh().catch(() => {}); };
   148→    window.addEventListener('focus', onFocus);
   149→    return () => { window.removeEventListener('focus', onFocus); fetchAbortRef.current?.abort('cancelled'); };
   150→  }, [doRefresh]);
   151→
   152→  // ── Cleanup all in-flight fetches on unmount ──
   153→  useEffect(() => {
   154→    return () => {
   155→      fetchAbortRef.current?.abort('cancelled');
   156→      refreshAbortRef.current?.abort('cancelled');
   157→    };
   158→  }, []);
   159→
   160→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string) => {
   161→    const fetchParams: FetchParams = { symbol, category, insCode, tgjuKey };
   162→    lastFetchRef.current = fetchParams;
   163→
   164→    // Cancel any previous in-flight fetch
   165→    fetchAbortRef.current?.abort('cancelled');
   166→    const controller = new AbortController();
   167→    fetchAbortRef.current = controller;
   168→    const { signal } = controller;
   169→
   170→    // TGJU instrument
   171→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   172→      setLoading(true);
   173→      setLoadingMessage('در حال دریافت داده‌های تاریخی ... (حدود ۱۵ ثانیه)');
   174→      setError(null);
   175→      setData(null);
   176→      try {
   177→        const res = await fetch(`/api/tgju-analysis?key=${encodeURIComponent(tgjuKey)}&name=${encodeURIComponent(symbol)}`, { signal });
   178→        if (!res.ok) {
   179→          const errData = await res.json().catch(() => ({}));
   180→          throw new Error(errData.error || 'خطا در دریافت داده‌های تاریخی');
   181→        }
   182→        const json = await res.json();
   183→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   184→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   185→        }
   186→        setData({ ...json, isTgju: true, category });
   187→        setActivePanel('visual'); setCurrentPage('analysis');
   188→      } catch (err) {
   189→        // Silently ignore cancelled/reset fetches (user switched instrument)
   190→        if (signal.aborted) return;
   191→        setError(err instanceof Error ? err.message : String(err));
   192→      } finally {
   193→        if (fetchAbortRef.current === controller) {
   194→          setLoading(false);
   195→          setLoadingMessage(null);
   196→        }
   197→      }
   198→      return;
   199→    }
   200→
   201→    // TSE Index (finpy-tse)
   202→    if (category === 'tse_index' && insCode) {
   203→      setLoading(true);
   204→      setLoadingMessage('در حال دریافت تاریخچه شاخص از TSETMC ...');
   205→      setError(null);
   206→      setData(null);
   207→      try {
   208→        const res = await fetch(`/api/index-analysis?webId=${encodeURIComponent(insCode)}`, { signal });
   209→        if (!res.ok) {
   210→          const errData = await res.json().catch(() => ({}));
   211→          throw new Error(errData.error || 'خطا در دریافت داده‌های شاخص');
   212→        }
   213→        const json = await res.json();
   214→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   215→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   216→        }
   217→        setData({ ...json, isIndex: true, category });
   218→        setActivePanel('visual'); setCurrentPage('analysis');
   219→      } catch (err) {
   220→        if (signal.aborted) return;
   221→        setError(err instanceof Error ? err.message : String(err));
   222→      } finally {
   223→        if (fetchAbortRef.current === controller) {
   224→          setLoading(false);
   225→          setLoadingMessage(null);
   226→        }
   227→      }
   228→      return;
   229→    }
   230→
   231→    // Regular TSE instrument
   232→    setLoading(true);
   233→    setError(null);
   234→    setData(null);
   235→    try {
   236→      const res = await fetch(`/api/analysis?symbol=${encodeURIComponent(symbol)}`, { signal });
   237→      if (!res.ok) {
   238→        const errData = await res.json().catch(() => ({}));
   239→        throw new Error(errData.error || 'خطا در دریافت داده‌ها');
   240→      }
   241→      const json = await res.json();
   242→      setData(json);
   243→      setActivePanel('visual'); setCurrentPage('analysis');
   244→    } catch (err) {
   245→      if (signal.aborted) return;
   246→      setError(err instanceof Error ? err.message : String(err));
   247→    } finally {
   248→      if (fetchAbortRef.current === controller) {
   249→        setLoading(false);
   250→      }
   251→    }
   252→  }, []);
   253→
   254→  const lastPrice = data?.ta ? data.candles[data.candles.length - 1]?.close ?? 0 : 0;
   255→  const signalColor = data?.ta?.overallSignal === 'bullish' ? 'text-emerald-600' : data?.ta?.overallSignal === 'bearish' ? 'text-red-600' : 'text-amber-600';
   256→  const signalBg = data?.ta?.overallSignal === 'bullish' ? 'bg-emerald-50 border-emerald-200' : data?.ta?.overallSignal === 'bearish' ? 'bg-red-50 border-red-200' : 'bg-amber-50 border-amber-200';
   257→  const SignalIcon = data?.ta?.overallSignal === 'bullish' ? TrendingUp : data?.ta?.overallSignal === 'bearish' ? TrendingDown : Activity;
   258→
   259→  const isTgjuData = data?.isTgju;
   260→  const isIndexData = data?.isIndex;
   261→
   262→  return (
   263→    <div dir="rtl" className="min-h-screen bg-white text-gray-900 flex flex-col">
   264→      {/* ── HEADER ────────────────────────────────────────────── */}
   265→      <header className="sticky top-0 z-50 border-b border-gray-200 bg-white/95 backdrop-blur-xl">
   266→        <div className="max-w-[1600px] mx-auto px-4 py-3 flex items-center gap-4 flex-wrap">
   267→          <div className="flex items-center gap-3 shrink-0">
   268→            <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-amber-400/20 to-amber-600/20 border border-amber-300/50 flex items-center justify-center">
   269→              <BarChart3 className="w-5 h-5 text-amber-600" />
   270→            </div>
   271→            <div>
   272→              <h1 className="text-base font-bold text-amber-700 leading-tight">تحلیل تکنیکال بازار</h1>
   273→              <p className="text-[10px] text-gray-400">بورس + ارز + طلا + کریپتو + فارکس + بورس جهانی (TSETMC &amp; TGJU)</p>
   274→            </div>
   275→          </div>
   276→
   277→          <div className="flex-1 min-w-[280px] max-w-2xl">
   278→            <SymbolSearch onSelect={handleSelect} placeholder='جستجوی نماد، ارز، طلا، کریپتو، شاخص ...' />
   279→          </div>
   280→
   281→          <div className="flex items-center gap-2 shrink-0">
   282→            <button
   283→              onClick={() => setCurrentPage('home')}
   284→              className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-medium transition-colors ${currentPage === 'home' ? 'bg-amber-50 text-amber-700 border border-amber-200' : 'text-gray-500 hover:text-gray-700 hover:bg-gray-50'}`}
   285→            ><HomeIcon className="w-3.5 h-3.5" /><span className="hidden sm:inline">صفحه اصلی</span></button>
   286→            <button
   287→              onClick={() => setCurrentPage('help')}
   288→              className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-medium transition-colors ${currentPage === 'help' ? 'bg-sky-50 text-sky-700 border border-sky-200' : 'text-gray-500 hover:text-gray-700 hover:bg-gray-50'}`}
   289→            ><HelpCircle className="w-3.5 h-3.5" /><span className="hidden sm:inline">راهنما</span></button>
   290→            <button
   291→              onClick={() => setCurrentPage('docs')}
   292→              className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-medium transition-colors ${currentPage === 'docs' ? 'bg-violet-50 text-violet-700 border border-violet-200' : 'text-gray-500 hover:text-gray-700 hover:bg-gray-50'}`}
   293→            ><BookOpen className="w-3.5 h-3.5" /><span className="hidden sm:inline">مستندات</span></button>
   294→            <button
   295→              onClick={() => setCurrentPage('indices')}
   296→              className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-medium transition-colors ${currentPage === 'indices' ? 'bg-rose-50 text-rose-700 border border-rose-200' : 'text-gray-500 hover:text-gray-700 hover:bg-gray-50'}`}
   297→            ><Database className="w-3.5 h-3.5" /><span className="hidden sm:inline">شاخص‌ها</span></button>
   298→          </div>
   299→
   300→          {data?.info && (
   301→            <div className="flex items-center gap-4 text-sm shrink-0">
   302→              <div className="text-left">
   303→                <div className="text-gray-500 text-xs">{data.info.name}</div>
   304→                <div className="font-bold text-lg">
   305→                  {toFa(data.info.lastPrice)}
   306→                  <span className={`text-xs mr-2 ${data.info.change >= 0 ? 'text-emerald-600' : 'text-red-600'}`}>
   307→                    {data.info.change >= 0 ? '▲' : '▼'} {toPersianDigits(Math.abs(data.info.change).toFixed(2))}%
   308→                  </span>
   309→                </div>
   310→              </div>
   311→              {isTgjuData && (
   312→                <span className="px-2 py-1 rounded-lg bg-teal-50 border border-teal-200 text-teal-700 text-[10px] font-bold flex items-center gap-1">
   313→                  <Coins className="w-3 h-3" /> TGJU
   314→                </span>
   315→              )}
   316→              {isIndexData && (
   317→                <span className="px-2 py-1 rounded-lg bg-rose-50 border border-rose-200 text-rose-700 text-[10px] font-bold flex items-center gap-1">
   318→                  <BarChart3 className="w-3 h-3" /> شاخص (finpy-tse)
   319→                </span>
   320→              )}
   321→              <div className={`px-3 py-1.5 rounded-lg border ${signalBg}`}>
   322→                <div className={`flex items-center gap-1.5 text-xs font-bold ${signalColor}`}>
   323→                  <SignalIcon className="w-3.5 h-3.5" />
   324→                  {data.ta.overallSignal === 'bullish' ? 'صعودی' : data.ta.overallSignal === 'bearish' ? 'نزولی' : 'خنثی'}
   325→                </div>
   326→              </div>
   327→              <button
   328→                onClick={doRefresh}
   329→                disabled={refreshing}
   330→                className="p-2 rounded-lg border border-gray-200 hover:bg-gray-50 text-gray-500 hover:text-gray-700 transition-colors disabled:opacity-50"
   331→                title="بازخوانی"
   332→              >
   333→                <RefreshCw className={`w-4 h-4 ${refreshing ? 'animate-spin' : ''}`} />
   334→              </button>
   335→            </div>
   336→          )}
   337→        </div>
   338→      </header>
   339→
   340→      {/* ── MAIN CONTENT ──────────────────────────────────────── */}
   341→      <main className="flex-1 max-w-[1600px] w-full mx-auto px-4 py-4">
   342→        {/* ── Help Page ── */}
   343→        {currentPage === 'help' && <HelpPage onGoHome={() => setCurrentPage('home')} onSearch={() => {
   344→          setCurrentPage('analysis');
   345→          setTimeout(() => {
   346→            const searchInput = document.querySelector('input[placeholder*="جستجو"]') as HTMLInputElement;
   347→            if (searchInput) searchInput.focus();
   348→          }, 100);
   349→        }} />}
   350→
   351→        {/* ── Documentation Page ── */}
   352→        {currentPage === 'docs' && <DocsPage />}
   353→
   354→        {/* ── Index Manager Page ── */}
   355→        {currentPage === 'indices' && <IndexManager onBack={() => setCurrentPage('home')} />}
   356→
   357→        {/* ── Home Page ── */}
   358→        {currentPage === 'home' && !data && !loading && !error && (
   359→          <Homepage onSelectCategory={(cat) => {
   360→            if (cat === 'indices') {
   361→              setCurrentPage('indices');
   362→              return;
   363→            }
   364→            const searchInput = document.querySelector('input[placeholder*="جستجو"]') as HTMLInputElement;
   365→            if (searchInput) {
   366→              searchInput.focus();
   367→            }
   368→          }} />
   369→        )}
   370→
   371→        {/* ── Analysis View ── */}
   372→        {(currentPage === 'analysis' || (currentPage === 'home' && data)) && loading && (
   373→          <div className="space-y-4 py-8">
   374→            {loadingMessage && (
   375→              <div className="flex items-center justify-center gap-3 py-6">
   376→                <div className="w-5 h-5 border-2 border-teal-300 border-t-teal-600 rounded-full animate-spin" />
   377→                <span className="text-sm text-teal-600">{loadingMessage}</span>
   378→              </div>
   379→            )}
   380→            <Skeleton className="h-8 w-48 bg-gray-200" />
   381→            <Skeleton className="h-[500px] w-full bg-gray-200 rounded-xl" />
   382→            <div className="grid grid-cols-3 gap-4">
   383→              <Skeleton className="h-32 bg-gray-200 rounded-xl" />
   384→              <Skeleton className="h-32 bg-gray-200 rounded-xl" />
   385→              <Skeleton className="h-32 bg-gray-200 rounded-xl" />
   386→            </div>
   387→          </div>
   388→        )}
   389→
   390→        {error && currentPage !== 'help' && currentPage !== 'docs' && (
   391→          <div className="flex flex-col items-center justify-center py-20 text-center">
   392→            <div className="w-16 h-16 rounded-full bg-red-50 flex items-center justify-center mb-4">
   393→              <TrendingDown className="w-8 h-8 text-red-500" />
   394→            </div>
   395→            <h3 className="text-lg font-bold text-red-600 mb-2">خطا</h3>
   396→            <p className="text-gray-500 text-sm max-w-md">{error}</p>
   397→          </div>
   398→        )}
   399→
   400→        {data && !loading && (currentPage === 'analysis' || currentPage === 'home') && (
   401→          <div className="flex gap-4">
   402→            {/* ── RIGHT SIDEBAR (Desktop) — first in DOM so RTL places it on the right ── */}
   403→            {!sidebarCollapsed ? (
   404→              <aside className="hidden lg:flex flex-col w-60 shrink-0 sticky top-[76px] self-start">
   405→                <div className="bg-gray-50/80 border border-gray-200 rounded-xl p-3">
   406→                  <div className="flex items-center justify-between mb-3">
   407→                    <span className="text-xs font-bold text-gray-500 tracking-wide">پنل‌های تحلیلی</span>
   408→                    <button
   409→                      onClick={() => setSidebarCollapsed(true)}
   410→                      className="p-1 rounded-md hover:bg-white text-gray-400 hover:text-gray-600 transition-colors border border-transparent hover:border-gray-200"
   411→                      title="بستن سایدبار"
   412→                    >
   413→                      <ChevronRight className="w-4 h-4" />
   414→                    </button>
   415→                  </div>
   416→                  <nav className="space-y-1.5">
   417→                    {SIDEBAR_ITEMS.map((item) => {
   418→                      const Icon = item.icon;
   419→                      const isActive = activePanel === item.id;
   420→                      return (
   421→                        <button
   422→                          key={item.id}
   423→                          onClick={() => setActivePanel(item.id)}
   424→                          className={`w-full flex items-center gap-3 px-3.5 py-3 rounded-xl text-sm font-medium transition-all duration-200 ${
   425→                            isActive
   426→                              ? `${item.activeBg} border shadow-sm scale-[1.02]`
