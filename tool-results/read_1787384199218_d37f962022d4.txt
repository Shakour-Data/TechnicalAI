   133→export default function VdesAnalysis(props: VdesAnalysisProps) {
   134→  const {
   135→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
   136→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
   137→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
   138→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
   139→    supportStrengths, resistanceStrengths, priceTargets,
   140→    isIndex, isTgju, currencyUnit,
   141→    edgeWeights,
   142→  } = props;
   143→
   144→  const unit = currencyUnit || 'ریال';
   145→
   146→  // ── Instrument Context (for intelligent naming) ─────────────────
   147→  const instrumentLabel = useMemo(() => {
   148→    if (isIndex) return { name: `شاخص ${symbolName}`, type: 'index' as const, verb: 'قرار دارد' };
   149→    if (isTgju) return { name: symbolName, type: 'tgju' as const, verb: 'قرار دارد' };
   150→    return { name: `سهم ${symbolName}`, type: 'stock' as const, verb: 'قرار دارد' };
   151→  }, [symbolName, isIndex, isTgju]);
   152→
   153→  const IL = instrumentLabel;
   154→
   155→  const containerRef = useRef<HTMLDivElement>(null);
   156→  const textAnalysisRef = useRef<HTMLDivElement>(null);
   157→  const [exporting, setExporting] = useState(false);
   158→
   159→  // ── AI Analysis state ───────────────────────────────────────────
   160→  const [aiAnalysis, setAiAnalysis] = useState<string | null>(null);
   161→  const [aiLoading, setAiLoading] = useState(false);
   162→  const [aiError, setAiError] = useState<string | null>(null);
   163→  const aiFetchIdRef = useRef(0);
   164→
   165→  // ── Analysis version selector (V7 locked / V8 / V9 locked / V10 active) ─────────
   166→  const [analysisVersion, setAnalysisVersion] = useState<7 | 8 | 9 | 10>(10);
   167→  // ── V10 Debug data ──────────────────────────────────────────────
   168→  const [v10Debug, setV10Debug] = useState<any>(null);
   169→
   170→  // ── V10 Scenarios (built from TA data + S/R levels) ───────────────
   171→  const v10Result = useMemo(() => {
   172→    if (analysisVersion !== 10) return null;
   173→    return buildV10Scenarios({
   174→      currentPrice,
   175→      atr,
   176→      overallSignal: overallSignal as 'bullish' | 'bearish' | 'neutral',
   177→      v9Scenarios: scenarios,
   178→      supports,
   179→      resistances,
   180→      bollingerUpper,
   181→      bollingerLower,
   182→      bollingerMiddle,
   183→      ma21,
   184→      ma100,
   185→    });
   186→  }, [analysisVersion, currentPrice, atr, overallSignal, scenarios, supports, resistances,
   187→      bollingerUpper, bollingerLower, bollingerMiddle, ma21, ma100]);
   188→
   189→  // ── Fetch AI analysis ───────────────────────────────────────────
   190→  const fetchAiAnalysis = useCallback(async () => {
   191→    const fetchId = ++aiFetchIdRef.current;
   192→    setAiLoading(true);
   193→    setAiError(null);
   194→    try {
   195→      const apiUrl = analysisVersion === 10 ? '/api/v10-analysis' : analysisVersion === 9 ? '/api/v9-analysis' : analysisVersion === 8 ? '/api/v8-analysis' : '/api/vdes-analysis';
   196→      // V10: use V10 scenarios (S1-S9), otherwise use V9 scenarios (R1-R9)
   197→      const activeScenarios = analysisVersion === 10 && v10Result
   198→        ? v10Result.scenarios
   199→        : scenarios;
   200→      const requestBody: any = {
   201→          symbolName, currentPrice, ma21, ma100,
   202→          rsi, mfi, cci, adx, stochK, stochD,
   203→          macdLine, macdSignal, macdHist,
   204→          diPlus, diMinus, sar, atr, obv, hasVolume,
   205→          bollingerUpper, bollingerMiddle, bollingerLower,
   206→          trendDirection, trendAngle, trendR2, overallSignal,
   207→          scenarios: activeScenarios, resistances, supports,
   208→          resistanceStrengths, supportStrengths,
   209→          isIndex, isTgju, currencyUnit,
   210→        };
   211→      // V9/V10: include OHLCV history for ML training/prediction
   212→      if ((analysisVersion === 9 || analysisVersion === 10) && candles.length > 0) {
   213→        requestBody.ohlcvHistory = candles.map(c => ({
   214→          date: c.date, open: c.open, high: c.high,
   215→          low: c.low, close: c.close, volume: c.volume,
   216→        }));
   217→      }
   218→
   219→      // AbortController for 90s client-side timeout
   220→      const controller = new AbortController();
   221→      const timeoutId = setTimeout(() => controller.abort(), 90000);
   222→
   223→      const res = await fetch(apiUrl, {
   224→        method: 'POST',
   225→        headers: { 'Content-Type': 'application/json' },
   226→        body: JSON.stringify(requestBody),
   227→        signal: controller.signal,
   228→      });
   229→      clearTimeout(timeoutId);
   230→
   231→      if (!res.ok) {
   232→        let errorMsg = `خطای سرور (${res.status})`;
   233→        try {
   234→          const errData = await res.json();
   235→          errorMsg = errData.error || errorMsg;
   236→        } catch {
   237→          // Response body is not JSON (e.g. 502 from Caddy)
   238→          if (res.status === 502) errorMsg = 'سرور در حال بارگذاری مجدد است. لطفاً چند ثانیه صبر کنید و دوباره تلاش کنید.';
   239→        }
   240→        throw new Error(errorMsg);
   241→      }
   242→      const data = await res.json();
   243→      if (fetchId === aiFetchIdRef.current) {
   244→        setAiAnalysis(data.analysis);
   245→        if (data._debug?.v10) setV10Debug({ ...data._debug.v10, v10Persona: data._debug.v10Persona, v10Template: data._debug.v10Template });
   246→        else setV10Debug(null);
   247→      }
   248→    } catch (err) {
   249→      if (fetchId === aiFetchIdRef.current) {
   250→        if (err instanceof Error && err.name === 'AbortError') {
   251→          setAiError('زمان پاسخگویی به پایان رسید. لطفاً دوباره تلاش کنید.');
   252→        } else {
   253→          setAiError(err instanceof Error ? err.message : 'خطای ناشناخته');
   254→        }
   255→      }
   256→    } finally {
   257→      if (fetchId === aiFetchIdRef.current) {
   258→        setAiLoading(false);
   259→      }
   260→    }
   261→  }, [symbolName, currentPrice, ma21, ma100, rsi, mfi, cci, adx, stochK, stochD,
   262→    macdLine, macdSignal, macdHist, diPlus, diMinus, sar, atr, obv, hasVolume,
   263→    bollingerUpper, bollingerMiddle, bollingerLower, trendDirection, trendAngle, trendR2,
   264→    overallSignal, scenarios, resistances, supports, resistanceStrengths, supportStrengths,
   265→    isIndex, isTgju, currencyUnit, analysisVersion, v10Result]);
   266→
   267→  useEffect(() => { fetchAiAnalysis().catch(() => {}); }, [fetchAiAnalysis]);
   268→
   269→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   270→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   271→  const S1_level = supports[0] ?? currentPrice * 0.95;
   272→  const S2_level = supports[1] ?? currentPrice * 0.90;
   273→  const S3_level = supports[2] ?? currentPrice * 0.85;
   274→
   275→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   276→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   277→
   278→  // ── Dominant scenario (V9) ──────────────────────────────────────
   279→  let highestKey = 'R3';
   280→  let highestProb = 0;
   281→  for (const key of SCENARIO_KEYS) {
   282→    if (scenarios[key].probability > highestProb) {
   283→      highestProb = scenarios[key].probability;
   284→      highestKey = key;
   285→    }
   286→  }
   287→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   288→
   289→  // ── V10 dominant + total ────────────────────────────────────────
   290→  const v10DominantKey = v10Result?.dominantKey || 'S5';
   291→  const v10TotalProb = v10Result
   292→    ? V10_KEYS.reduce((sum, k) => sum + (v10Result.scenarios[k]?.probability || 0), 0)
   293→    : 0;
   294→
   295→  // ── Sorted scenarios by target price (ascending) ─────────────
   296→  const sortedScenarios = useMemo(() => {
   297→    return [...SCENARIO_KEYS]
   298→      .map(k => ({ key: k, ...scenarios[k] }))
   299→      .sort((a, b) => a.targetMin - b.targetMin);
   300→  }, [scenarios]);
   301→
   302→  // ── Dynamic scenario labels (context-aware based on overallSignal) ─
   303→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal), [overallSignal]);
   304→  const strategy = dynamicMeta[highestKey];
   305→
   306→  // ── Text variety: hash-based selector for varied language ──────────
   307→  const varietySeed = useMemo(() => {

   309→    let hash = 0;
   310→    for (let i = 0; i < raw.length; i++) hash = ((hash << 5) - hash + raw.charCodeAt(i)) | 0;

   312→  }, [rsi, macdHist, cci, adx, stochK, trendAngle, currentPrice]);
   313→
   314→  // ── RSI signal ─────────────────────────────────────────────────
   315→  const rsiZone = rsi > 70 ? 'overbought' : rsi > 60 ? 'weak_ob' : rsi > 40 ? 'neutral' : rsi > 30 ? 'weak_os' : 'oversold';

   317→  const rsiColor = rsi > 70 ? 'text-red-700' : rsi > 60 ? 'text-amber-800' : rsi > 40 ? 'text-gray-600' : rsi > 30 ? 'text-amber-800' : 'text-emerald-700';
   318→
   319→  // ── Stochastic signal ──────────────────────────────────────────
   320→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';

   322→  // ── Trend text ─────────────────────────────────────────────────
   323→  const isUp = trendDirection === 'up';
   324→  const isDown = trendDirection === 'down';
   325→  const trendLabel = isUp ? 'صعودی' : isDown ? 'نزولی' : 'خنثی';
   326→  const trendColorCls = isUp ? 'text-emerald-700' : isDown ? 'text-red-700' : 'text-amber-800';

   330→
   331→  // ── Bollinger position ─────────────────────────────────────────
   332→  const bbRange = bollingerUpper - bollingerLower;
   333→  const bbPos = bbRange > 0 ? toPersianDigits(((currentPrice - bollingerLower) / bbRange * 100).toFixed(0)) : '۵۰';

   335→  const bbBelow = currentPrice < bollingerLower;
   336→  const bbSignal = bbAbove ? 'بالای باند بالایی' : bbBelow ? 'زیر باند پایینی' : `داخل باندها (${bbPos}٪)`;

   338→  // ── Distance to MA ─────────────────────────────────────────────
   339→  const distMa21 = ((currentPrice - ma21) / ma21 * 100).toFixed(1);
   340→  const distMa100 = ((currentPrice - ma100) / ma100 * 100).toFixed(1);
   341→
   342→  // ── TradingView scenario colors ────────────────────────────────
   343→  const chartCandles = useMemo(() => {
   344→    if (candles.length <= 2000) return candles;
   345→    return candles.slice(-2000);
   346→  }, [candles]);
   347→
   348→  const tvScenarios = useMemo(() => {
   349→    const colors: Record<string, string> = {
   350→      R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62',
   351→      R6: '#fca5a5', R7: '#fbbf24', R8: '#fb923c', R9: '#a855f7',
   352→    };
   353→    return Object.fromEntries(
   354→      SCENARIO_KEYS.map(k => [k, { targetMin: scenarios[k].targetMin, targetMax: scenarios[k].targetMax, name: scenarios[k].name, probability: scenarios[k].probability, color: colors[k] }])

   356→  }, [scenarios]);

   360→  const strategyText = useMemo(() => {
   361→    const v = varietySeed % 3; // 3 variants per signal type

   364→        `هشدار اشباع خرید فعال است (RSI: ${toFaG(rsi, 1)}${mfiWarning}). فاصله تا مقاومت اول ${toFaG(R1_level)} ${unit} است. بهترین استراتژی در این شرایط، صبر برای اصلاح قیمت تا محدوده حمایت ${toFaG(S1_level)} تا ${toFaG(S2_level)} و ورود در نقطه بهینه‌تر می‌باشد.`,

   913→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   914→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   915→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   916→  grade?: string; overlapCount?: number; touchCount?: number;
   917→}) {
   918→  const isSupport = type === 'support';
   919→  const isTargetType = type === 'target';
   920→
   921→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   922→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   923→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   924→
   925→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   926→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   927→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   928→
   929→  return (
   930→    <div
   931→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   932→      style={{
   933→        background: bgAccent,
   934→        border: `1px solid ${borderAccent}`,
   935→        borderRight: `3px solid ${accentColor}`,
   936→      }}
   937→    >
   938→      {/* Label + Price */}
   939→      <div className="flex-1 min-w-0">
   940→        <div className="flex items-center gap-2">
   941→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   942→          {isTarget && <span className="text-[9px]">🎯</span>}
   943→          {overlapCount !== undefined && overlapCount > 0 && (
   944→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   945→              {toPersianDigits(overlapCount)} هم‌پوشانی
   946→            </span>
   947→          )}
   948→        </div>
   949→        <div className="flex items-baseline gap-1">
   950→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   951→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   952→        </div>
   953→      </div>
   954→
   955→      {/* Strength + Grade */}
   956→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   957→        <div className="flex items-center gap-1.5">
   958→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   959→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   960→          <span className="text-[9px] text-gray-500">/۱۰</span>
   961→        </div>
   962→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   963→          <div
   964→            className="h-full rounded-full transition-all duration-500"
   965→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   966→          />
   967→        </div>
   968→        {touchCount !== undefined && touchCount > 0 && (
   969→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   970→        )}
   971→      </div>
   972→    </div>
   973→  );
   974→}
