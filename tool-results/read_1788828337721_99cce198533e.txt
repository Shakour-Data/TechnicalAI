   161→  };
   162→  for (const d of UML_STRUCT_DIAGRAMS) {

   164→    all.push({
   165→      id: d.id,
   166→      title: d.title,
   167→      description: d.description,
   168→      level: d.level,
   169→      code: d.code,
   170→      format: 'mermaid',
   171→      category: 'uml-struct',
   172→      subcategory: structTypeMap[typeKey] || typeKey,
   173→    });

   175→
   176→  // UML Behavioral diagrams

   178→    'UC': 'uml-usecase',

   180→    'SM': 'uml-statemachine',
   181→    'SEQ': 'uml-sequence',
   182→    'COMM': 'uml-communication',
   183→    'IO': 'uml-interaction-overview',
   184→    'TIM': 'uml-timing',

   186→  for (const d of UML_BEHAV_DIAGRAMS) {

   206→// Section ↔ Diagram Mapping
   207→// ═══════════════════════════════════════════════════════════════
   208→
   209→function getDiagramsForSection(sectionId: string): UnifiedDiagram[] {
   210→  // Level-based leaf sections
   211→  if (sectionId === 'dfd-l0') return ALL_DIAGRAMS.filter(d => d.category === 'dfd' && d.level === 0);
   212→  if (sectionId === 'dfd-l1') return ALL_DIAGRAMS.filter(d => d.category === 'dfd' && d.level === 1);
   213→  if (sectionId === 'dfd-l2') return ALL_DIAGRAMS.filter(d => d.category === 'dfd' && d.level === 2);
   214→  if (sectionId === 'dfd-l3') return ALL_DIAGRAMS.filter(d => d.category === 'dfd' && d.level === 3);
   215→
   216→  if (sectionId === 'bpmn-l1') return ALL_DIAGRAMS.filter(d => d.category === 'bpmn' && d.level === 1);
   217→  if (sectionId === 'bpmn-l2') return ALL_DIAGRAMS.filter(d => d.category === 'bpmn' && d.level === 2);
   218→  if (sectionId === 'bpmn-l3') return ALL_DIAGRAMS.filter(d => d.category === 'bpmn' && d.level === 3);
   219→
   220→  // UML structural type-level sections
   221→  if (sectionId.startsWith('uml-') && !sectionId.includes('-l')) {
   222→    // e.g., 'uml-class' — show all levels for this type
   223→    if (sectionId === 'uml-class') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-class');
   224→    if (sectionId === 'uml-object') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-object');
   225→    if (sectionId === 'uml-component') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-component');
   226→    if (sectionId === 'uml-deployment') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-deployment');
   227→    if (sectionId === 'uml-package') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-package');
   228→    if (sectionId === 'uml-composite') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-composite');
   229→    if (sectionId === 'uml-profile') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-profile');
   230→  }
   231→
   232→  // UML structural level sections (e.g., 'uml-class-l1')
   233→  const structLevelMatch = sectionId.match(/^uml-(class|object|component|deployment|package|composite|profile)-l(\d)$/);
   234→  if (structLevelMatch) {
   235→    const sub = `uml-${structLevelMatch[1]}`;
   236→    const lv = parseInt(structLevelMatch[2]);
   237→    return ALL_DIAGRAMS.filter(d => d.subcategory === sub && d.level === lv);
   238→  }
   239→
   240→  // UML behavioral type-level sections
   241→  if (sectionId === 'uml-usecase') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-usecase');
   242→  if (sectionId === 'uml-activity') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-activity');
   243→  if (sectionId === 'uml-statemachine') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-statemachine');
   244→  if (sectionId === 'uml-sequence') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-sequence');
   245→  if (sectionId === 'uml-communication') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-communication');
   246→  if (sectionId === 'uml-interaction-overview') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-interaction-overview');
   247→  if (sectionId === 'uml-timing') return ALL_DIAGRAMS.filter(d => d.subcategory === 'uml-timing');
   248→
   249→  // Parent sections — show all child diagrams
   250→  if (sectionId === 'dfd') return ALL_DIAGRAMS.filter(d => d.category === 'dfd');
   251→  if (sectionId === 'bpmn') return ALL_DIAGRAMS.filter(d => d.category === 'bpmn');
   252→  if (sectionId === 'uml-struct') return ALL_DIAGRAMS.filter(d => d.category === 'uml-struct');
   253→  if (sectionId === 'uml-behav') return ALL_DIAGRAMS.filter(d => d.category === 'uml-behav');
   254→
   255→  return [];
   256→}
