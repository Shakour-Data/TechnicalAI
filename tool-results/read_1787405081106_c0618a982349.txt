   927→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   928→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   929→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   930→  grade?: string; overlapCount?: number; touchCount?: number;
   931→}) {
   932→  const isSupport = type === 'support';
   933→  const isTargetType = type === 'target';
   934→
   935→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   936→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   937→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   938→
   939→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   940→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   941→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   942→
   943→  return (
   944→    <div
   945→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   946→      style={{
   947→        background: bgAccent,
   948→        border: `1px solid ${borderAccent}`,
   949→        borderRight: `3px solid ${accentColor}`,
   950→      }}
   951→    >
   952→      {/* Label + Price */}
   953→      <div className="flex-1 min-w-0">
   954→        <div className="flex items-center gap-2">
   955→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   956→          {isTarget && <span className="text-[9px]">🎯</span>}
   957→          {overlapCount !== undefined && overlapCount > 0 && (
   958→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   959→              {toPersianDigits(overlapCount)} هم‌پوشانی
   960→            </span>
   961→          )}
   962→        </div>
   963→        <div className="flex items-baseline gap-1">
   964→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   965→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   966→        </div>
   967→      </div>
   968→
   969→      {/* Strength + Grade */}
   970→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   971→        <div className="flex items-center gap-1.5">
   972→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   973→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   974→          <span className="text-[9px] text-gray-500">/۱۰</span>
   975→        </div>
   976→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   977→          <div
   978→            className="h-full rounded-full transition-all duration-500"
   979→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   980→          />
   981→        </div>
   982→        {touchCount !== undefined && touchCount > 0 && (
   983→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   984→        )}
   985→      </div>
   986→    </div>
   987→  );
   988→}
