   269→/* Advanced query parser — supports operators like country:US, exchange:NASD, sector:Tech */
   270→interface ParsedQuery {
   271→  country?: string;
   272→  exchange?: string;
   273→  sector?: string;
   274→  terms: string[];
   275→}
   276→
   277→function parseQuery(q: string): ParsedQuery {
   278→  const country: string[] = [];
   279→  const exchange: string[] = [];
   280→  const sector: string[] = [];
   281→  const terms: string[] = [];
   282→
   283→  for (const part of q.split(/\s+/)) {
   284→    if (!part) continue;
   285→    const lower = part.toLowerCase();
   286→    if (lower.startsWith('country:') || lower.startsWith('کشور:')) {
   287→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   288→      if (val) country.push(val.toLowerCase());
   289→    } else if (lower.startsWith('exchange:') || lower.startsWith('صرافی:')) {
   290→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   291→      if (val) exchange.push(val.toLowerCase());
   292→    } else if (lower.startsWith('sector:') || lower.startsWith('صنعت:')) {
   293→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   294→      if (val) sector.push(val.toLowerCase());
   295→    } else {
   296→      terms.push(lower);
   297→    }
   298→  }
   299→
   300→  return {
   301→    country: country.length > 0 ? country.join(' ') : undefined,
   302→    exchange: exchange.length > 0 ? exchange.join(' ') : undefined,
   303→    sector: sector.length > 0 ? sector.join(' ') : undefined,
   304→    terms,
   305→  };
   306→}

   312→const RECENT_KEY = 'fin-search-recent';
   313→const MAX_RECENT = 8;
   314→
   315→function getRecentSearches(): string[] {
   316→  if (typeof window === 'undefined') return [];
   317→  try { const raw = localStorage.getItem(RECENT_KEY); return raw ? JSON.parse(raw) : []; }
   318→  catch { return []; }
   319→}
   320→
   321→function addRecentSearch(name: string) {
   322→  if (typeof window === 'undefined' || !name) return;
   323→  try {
   324→    let list = getRecentSearches();
   325→    list = list.filter((r) => r !== name);
   326→    list.unshift(name);
   327→    if (list.length > MAX_RECENT) list = list.slice(0, MAX_RECENT);
   328→    localStorage.setItem(RECENT_KEY, JSON.stringify(list));
   329→  } catch { /* ignore */ }
   330→}

   531→    const scoreAndFilter = (item: InstrumentItem) => {
   532→      /* Apply query operators */
   533→      if (opCountry) {
   534→        const match = (item.country?.toLowerCase().includes(opCountry)) ||
   535→          (item.countryEn?.toLowerCase().includes(opCountry));
   536→        if (!match) return null;
   537→      }
   538→      if (opExchange) {
   539→        const match = item.exchange?.toLowerCase().includes(opExchange);
   540→        if (!match) return null;
   541→      }
   542→      if (opSector) {
   543→        const match = item.sector?.toLowerCase().includes(opSector) ||
   544→          item.cs?.toLowerCase().includes(opSector);
   545→        if (!match) return null;
   546→      }
   547→
   548→      const fields = [item.l18, item.l30, item.cs, item.yahooSymbol || '', item.nameEn || '', item.exchange || '', item.country || '', item.countryEn || '', item.sector || ''];
   549→      const fullText = fields.join(' ').toLowerCase();
   550→      let score = 0;
   551→
   552→      for (const term of terms) {
   553→        if (!term) continue;
   554→        if (item.l18.toLowerCase().includes(term)) score += 10;
   555→        if (item.nameEn?.toLowerCase().includes(term)) score += 8;
   556→        if (item.yahooSymbol?.toLowerCase() === term) score += 20;
   557→        else if (item.yahooSymbol?.toLowerCase().includes(term)) score += 15;
   558→        if (item.l30.toLowerCase().includes(term)) score += 7;
   559→        if (item.exchange?.toLowerCase().includes(term)) score += 5;
   560→        if (item.country?.toLowerCase().includes(term)) score += 6;
   561→        if (item.countryEn?.toLowerCase() === term) score += 15;
   562→        else if (item.countryEn?.toLowerCase().includes(term)) score += 4;
   563→        if (item.sector?.toLowerCase().includes(term)) score += 5;
   564→        if (item.cs?.toLowerCase().includes(term)) score += 6;
   565→
   566→        if (!item.l18.toLowerCase().includes(term) && !item.l30.toLowerCase().includes(term) &&
   567→            !item.cs?.toLowerCase().includes(term) && !(item.yahooSymbol?.toLowerCase().includes(term)) &&
   568→            !(item.nameEn?.toLowerCase().includes(term)) && !(item.exchange?.toLowerCase().includes(term)) &&
   569→            !(item.country?.toLowerCase().includes(term)) && !(item.countryEn?.toLowerCase().includes(term)) &&
   570→            !(item.sector?.toLowerCase().includes(term))) {
   571→          return null;
   572→        }
   573→      }
   574→      /* If only operators (no terms), accept all filtered items */
   575→      if (terms.length === 0) score = 1;
   576→      return score;
   577→    };
   578→
   579→    const scored = items.map((item) => ({ item, score: scoreAndFilter(item) }))
   580→      .filter((r): r is { item: InstrumentItem; score: number } => r.score !== null && r.score > 0)
   581→      .sort((a, b) => b.score - a.score);
   582→
   583→    setTotalMatched(scored.length);
   584→    setResults(scored.slice(0, MAX_RESULTS).map((s) => s.item));

   586→  }, []);
