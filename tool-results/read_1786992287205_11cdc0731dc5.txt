   127→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   128→function calcMFI(data: OHLCV[], period: number = 14): number {
   129→  if (data.length < period + 1) return 50;
   130→  let posFlow = 0;
   131→  let negFlow = 0;
   132→  for (let i = data.length - period; i < data.length; i++) {
   133→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   134→    const mf = tp * data[i].volume;
   135→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   136→    if (tp > prevTp) posFlow += mf;
   137→    else negFlow += mf;
   138→  }
   139→  if (negFlow === 0) return 100;
   140→  const mfr = posFlow / negFlow;
   141→  return 100 - 100 / (1 + mfr);
   142→}

   145→function calcCCI(data: OHLCV[], period: number = 20): number {
   146→  if (data.length < period) return 0;
   147→  const tps: number[] = [];
   148→  for (let i = data.length - period; i < data.length; i++) {
   149→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   150→  }
   151→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   152→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   153→  if (meanDev === 0) return 0;
   154→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   155→}

   157→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   158→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   159→  if (data.length < kPeriod) return { k: 50, d: 50 };
   160→  const rawKs: number[] = [];
   161→  for (let i = kPeriod - 1; i < data.length; i++) {
   162→    let lowest = Infinity;
   163→    let highest = -Infinity;
   164→    for (let j = i - kPeriod + 1; j <= i; j++) {
   165→      if (data[j].low < lowest) lowest = data[j].low;
   166→      if (data[j].high > highest) highest = data[j].high;
   167→    }
   168→    const range = highest - lowest;
   169→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   170→  }
   171→  const smoothedK: number[] = [];
   172→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   173→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   174→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   175→  }
   176→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   177→  const dSlice = smoothedK.slice(-smoothD);
   178→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   179→  return { k, d };
   180→}

   183→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   184→  if (data.length < period) return -50;
   185→  const slice = data.slice(-period);
   186→  let highest = -Infinity;
   187→  let lowest = Infinity;
   188→  for (const c of slice) {
   189→    if (c.high > highest) highest = c.high;
   190→    if (c.low < lowest) lowest = c.low;
   191→  }
   192→  const range = highest - lowest;
   193→  if (range === 0) return -50;
   194→  return ((highest - data[data.length - 1].close) / range) * -100;
   195→}

   197→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   198→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   199→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   200→  const emaFast = emaCalc(closes, fast);
   201→  const emaSlow = emaCalc(closes, slow);
   202→  const macdLine = emaFast - emaSlow;
   203→  // Calculate signal as EMA of MACD values
   204→  const macdValues: number[] = [];
   205→  const kFast = 2 / (fast + 1);
   206→  const kSlow = 2 / (slow + 1);
   207→  let eFast = sma(closes.slice(0, fast), fast);
   208→  let eSlow = sma(closes.slice(0, slow), slow);
   209→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   210→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   211→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   212→    macdValues.push(eFast - eSlow);
   213→  }
   214→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   215→  const kSig = 2 / (sig + 1);
   216→  let signalVal = sma(macdValues.slice(0, sig), sig);
   217→  for (let i = sig; i < macdValues.length; i++) {
   218→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   219→  }
   220→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   221→}

   223→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   224→function calcATR(data: OHLCV[], period: number = 14): number {
   225→  if (data.length < 2) return 0;
   226→  const trs: number[] = [];
   227→  for (let i = 1; i < data.length; i++) {
   228→    const tr = Math.max(
   229→      data[i].high - data[i].low,
   230→      Math.abs(data[i].high - data[i - 1].close),
   231→      Math.abs(data[i].low - data[i - 1].close)
   232→    );
   233→    trs.push(tr);
   234→  }
   235→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   236→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   237→  for (let i = period; i < trs.length; i++) {
   238→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   239→  }
   240→  return atrVal;
   241→}

   243→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   244→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   245→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   246→  const plusDMs: number[] = [];
   247→  const minusDMs: number[] = [];
   248→  const trs: number[] = [];
   249→  for (let i = 1; i < data.length; i++) {
   250→    const upMove = data[i].high - data[i - 1].high;
   251→    const downMove = data[i - 1].low - data[i].low;
   252→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   253→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   254→    trs.push(Math.max(
   255→      data[i].high - data[i].low,
   256→      Math.abs(data[i].high - data[i - 1].close),
   257→      Math.abs(data[i].low - data[i - 1].close)
   258→    ));
   259→  }
   260→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   261→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   262→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   263→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   264→  const dxValues: number[] = [];
   265→  for (let i = period; i < trs.length; i++) {
   266→    smoothTR = smoothTR - smoothTR / period + trs[i];
   267→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   268→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   269→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   270→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   271→    const diSum = diPlus + diMinus;
   272→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   273→    dxValues.push(dx);
   274→  }
   275→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   276→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   277→  for (let i = period; i < dxValues.length; i++) {
   278→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   279→  }
   280→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   281→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   282→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   283→}

   285→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   286→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   287→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   288→  let isLong = true;
   289→  let af = step;
   290→  let ep = data[0].high;
   291→  let sar = data[0].low;
   292→  for (let i = 1; i < data.length; i++) {
   293→    const prevSar = sar;
   294→    sar = prevSar + af * (ep - prevSar);
   295→    // Ensure SAR is within bounds
   296→    if (isLong) {
   297→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   298→      if (data[i].low < sar) {
   299→        isLong = false;
   300→        sar = ep;
   301→        ep = data[i].low;
   302→        af = step;
   303→      } else {
   304→        if (data[i].high > ep) {
   305→          ep = data[i].high;
   306→          af = Math.min(af + step, max);
   307→        }
   308→      }
   309→    } else {
   310→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   311→      if (data[i].high > sar) {
   312→        isLong = true;
   313→        sar = ep;
   314→        ep = data[i].high;
   315→        af = step;
   316→      } else {
   317→        if (data[i].low < ep) {
   318→          ep = data[i].low;
   319→          af = Math.min(af + step, max);
   320→        }
   321→      }
   322→    }
   323→  }
   324→  return sar;
   325→}

   328→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   329→  if (closes.length < period) {
   330→    const c = closes[closes.length - 1] ?? 0;
   331→    return { upper: c, middle: c, lower: c };
   332→  }
   333→  const slice = closes.slice(-period);
   334→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   335→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   336→  const sd = Math.sqrt(variance);
   337→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   338→}

   350→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   351→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   352→  const n = values.length;
   353→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   354→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   355→  for (let i = 0; i < n; i++) {
   356→    sumX += i;
   357→    sumY += values[i];
   358→    sumXY += i * values[i];
   359→    sumX2 += i * i;
   360→    sumY2 += values[i] * values[i];
   361→  }
   362→  const denom = n * sumX2 - sumX * sumX;
   363→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   364→  const slope = (n * sumXY - sumX * sumY) / denom;
   365→  const intercept = (sumY - slope * sumX) / n;
   366→  const meanY = sumY / n;
   367→  let ssRes = 0, ssTot = 0;
   368→  for (let i = 0; i < n; i++) {
   369→    const predicted = slope * i + intercept;
   370→    ssRes += (values[i] - predicted) ** 2;
   371→    ssTot += (values[i] - meanY) ** 2;
   372→  }
   373→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   374→  return { slope, intercept, r2 };
   375→}

   377→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   378→function roundToNice(price: number): number {
   379→  const abs = Math.abs(price);
   380→  let step: number;
   381→  if (abs >= 100000) step = 100;
   382→  else if (abs >= 10000) step = 50;
   383→  else if (abs >= 1000) step = 10;
   384→  else if (abs >= 100) step = 5;
   385→  else step = 1;
   386→  return Math.round(price / step) * step;
   387→}

   391→  const highs: number[] = [];

   398→      if (data[j].high >= data[i].high) isHigh = false;

   407→// ─── Support / Resistance (6 each, adaptive gap, multi-source, 10% window strength) ─
   408→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   409→  resistances: number[];
   410→  supports: number[];
   411→  supportStrengths: LevelStrength[];
   412→  resistanceStrengths: LevelStrength[];
   413→  priceTargets: LevelStrength[];
   414→} {
   415→  const allRawLevels: number[] = [];
   416→  const closes = data.map(d => d.close);
   417→
   418→  // 1. Pivot Points from previous day (5R + 5S)
   419→  const prev = data[data.length - 2] ?? data[data.length - 1];
   420→  if (prev) {
   421→    const pp = (prev.high + prev.low + prev.close) / 3;
   422→    const hl = prev.high - prev.low;
   423→    const pivotR = [
   424→      2 * pp - prev.low,
   425→      pp + hl,
   426→      prev.high + 2 * (pp - prev.low),
   427→      prev.high + 2 * (pp - prev.low) + hl,
   428→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   429→    ];
   430→    const pivotS = [
   431→      2 * pp - prev.high,
   432→      pp - hl,
   433→      prev.low - 2 * (prev.high - pp),
   434→      prev.low - 2 * (prev.high - pp) - hl,
   435→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   436→    ];
   437→    allRawLevels.push(...pivotR, ...pivotS);
   438→  }
   439→
   440→  // 2. Swing levels (multiple lookback periods for better coverage)
   441→  for (const lb of [3, 5, 7, 10]) {
   442→    const swings = findSwingLevels(data, lb);
   443→    allRawLevels.push(...swings.highs, ...swings.lows);
   444→  }
   445→
   446→  // 3. Moving Average levels (potential dynamic S/R)
   447→  for (const period of [5, 10, 21, 50, 100, 200]) {
   448→    const v = sma(closes, period);
   449→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01) allRawLevels.push(v);
   450→  }
   451→  const ema12 = emaCalc(closes, 12);
   452→  const ema26 = emaCalc(closes, 26);
   453→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01) allRawLevels.push(ema12);
   454→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01) allRawLevels.push(ema26);
   455→
   456→  // 4. Bollinger Band levels
   457→  const bbLevels = calcBollingerBands(closes);
   458→  if (bbLevels.upper > 0) allRawLevels.push(bbLevels.upper);
   459→  if (bbLevels.lower > 0) allRawLevels.push(bbLevels.lower);
   460→
   461→  // 5. Round number / psychological levels
   462→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
   463→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
   464→    allRawLevels.push(Math.round(currentPrice * mult / magnitude) * magnitude);
   465→  }
   466→
   467→  // 6. Recent high/low zones (5-day, 22-day, 66-day)
   468→  for (const window of [5, 22, 66]) {
   469→    const slice = data.slice(-Math.min(window, data.length));
   470→    if (slice.length > 0) {
   471→      allRawLevels.push(Math.max(...slice.map(d => d.high)));
   472→      allRawLevels.push(Math.min(...slice.map(d => d.low)));
   473→    }
   474→  }
   475→
   476→  // 7. Round all and filter invalid
   477→  const rounded = allRawLevels.map(roundToNice).filter(l => l > 0);
   478→
   479→  // 8. Remove duplicates (within 0.5%)
   480→  const unique: number[] = [];
   481→  for (const level of rounded) {
   482→    const isDup = unique.some(u => Math.abs(u - level) / Math.max(u, 1) < 0.005);
   483→    if (!isDup) unique.push(level);
   484→  }
   485→
   486→  // 9. Calculate strength for each unique level — 10% window (5% above + 5% below)
   487→  const levelStrengths: LevelStrength[] = unique.map(level => {
   488→    const range10pct = level * 0.10;
   489→    const nearbyCount = unique.filter(other =>
   490→      other !== level && Math.abs(other - level) <= range10pct
   491→    ).length;
   492→    // Normalize: 0 nearby = 1, each nearby adds ~1.5, max 10
   493→    const strength = Math.min(10, Math.max(1, Math.round(1 + nearbyCount * 1.5)));
   494→    return { price: level, strength, isTarget: false };
   495→  });
   496→
   497→  // 10. Separate above/below current price
   498→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   499→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   500→
   501→  // 11. Enforce 2%-5% gap, always return exactly targetCount levels
   502→  //    Step 1: Merge levels closer than 2%
   503→  //    Step 2: Walk from nearest level, keep only chain within 2-5% gap
   504→  //    Step 3: If fewer than targetCount, extend chain by computing ~3% steps outward
   505→  //    Step 4: If more than targetCount, trim by keeping highest strength
   506→  function enforceGap(
   507→    levels: LevelStrength[],
   508→    targetCount: number,
   509→    _allUnique: LevelStrength[],
   510→  ): LevelStrength[] {
   511→    if (levels.length === 0) return [];
   512→
   513→    const MIN_GAP = 0.02;
   514→    const MAX_GAP = 0.05;
   515→    const EXTEND_GAP = 0.035; // 3.5% — middle of 2-5% range
   516→    const step = currentPrice >= 100000 ? 100 : currentPrice >= 10000 ? 50 : currentPrice >= 1000 ? 10 : currentPrice >= 100 ? 5 : 1;
   517→    const isDescending = levels.length > 0 && levels[0].price > (levels[levels.length - 1]?.price ?? 0);
   518→
   519→    // Step 1: Merge levels closer than MIN_GAP
   520→    const merged: LevelStrength[] = [levels[0]];
   521→    for (let i = 1; i < levels.length; i++) {
   522→      const prevPrice = merged[merged.length - 1].price;
   523→      const currPrice = levels[i].price;
   524→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   525→      if (gap < MIN_GAP) {
   526→        const mergedStrength = Math.min(10, Math.max(merged[merged.length - 1].strength, levels[i].strength) + 1);
   527→        merged[merged.length - 1] = {
   528→          ...(levels[i].strength > merged[merged.length - 1].strength ? levels[i] : merged[merged.length - 1]),
   529→          strength: mergedStrength,
   530→        };
   531→      } else {
   532→        merged.push(levels[i]);
   533→      }
   534→    }
   535→
   536→    // Step 2: Walk from first level (nearest to price), keep only if gap is <= MAX_GAP
   537→    const chain: LevelStrength[] = [merged[0]];
   538→    for (let i = 1; i < merged.length; i++) {
   539→      const prevPrice = chain[chain.length - 1].price;
   540→      const currPrice = merged[i].price;
   541→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   542→      if (gap <= MAX_GAP) {
   543→        chain.push(merged[i]);
   544→      }
   545→    }
   546→
   547→    // Step 3: If fewer than targetCount, extend the chain outward at ~3.5% intervals
   548→    while (chain.length < targetCount) {
   549→      const lastPrice = chain[chain.length - 1].price;
   550→      const nextRaw = isDescending
   551→        ? lastPrice * (1 - EXTEND_GAP)  // supports go down
   552→        : lastPrice * (1 + EXTEND_GAP); // resistances go up
   553→      const nextPrice = Math.round(nextRaw / step) * step;
   554→      if (nextPrice <= 0) break;
   555→      if (nextPrice === lastPrice) break; // rounding made no change (very low price)
   556→      chain.push({ price: nextPrice, strength: 2, isTarget: false });
   557→    }
   558→
   559→    // Step 4: If more than targetCount, trim by keeping highest strength
   560→    if (chain.length > targetCount) {
   561→      chain.sort((a, b) => b.strength - a.strength);
   562→      const kept = chain.slice(0, targetCount);
   563→      kept.sort((a, b) => isDescending ? b.price - a.price : a.price - b.price);
   564→      return kept;
   565→    }
   566→
   567→    return chain;
   568→  }
   569→
   570→  const finalResistances = enforceGap(above, 6, levelStrengths);
   571→  const finalSupports = enforceGap(below, 6, levelStrengths);
   572→
   573→  // 12. Determine price targets: strength > 7 is a target
   574→  const allLevels = [...finalResistances, ...finalSupports];
   575→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   576→  let targets: LevelStrength[];
   577→  if (sortedByStrength.some(l => l.strength > 7)) {
   578→    targets = sortedByStrength.filter(l => l.strength > 7);
   579→  } else {
   580→    targets = sortedByStrength.slice(0, 2);
   581→  }
   582→
   583→  // Mark targets
   584→  targets = targets.map(t => ({ ...t, isTarget: true }));
   585→
   586→  // Also mark in final arrays
   587→  finalResistances.forEach(r => {
   588→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   589→  });
   590→  finalSupports.forEach(s => {
   591→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   592→  });
   593→
   594→  return {
   595→    resistances: finalResistances.map(l => l.price),
   596→    supports: finalSupports.map(l => l.price),
   597→    supportStrengths: finalSupports,
   598→    resistanceStrengths: finalResistances,
   599→    priceTargets: targets,
   600→  };
   601→}

   603→// ─── Trend Analysis ───────────────────────────────────────────────────────────
   604→function calcTrend(closes: number[], period: number): TrendResult {
   605→  const slice = closes.slice(-Math.min(period, closes.length));
   606→  const { slope, r2 } = linearRegression(slice);
   607→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   608→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   609→  const priceThreshold = avgPrice * 0.001; // 0.1%
   610→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   611→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   612→}

   689→  const R3_level = resistances[2] ?? price * 1.15;
   690→  const R4_level = resistances[3] ?? price * 1.20;
   691→  const R5_level = resistances[4] ?? price * 1.30;
   692→  const R6_level = resistances[5] ?? price * 1.40;
   693→  const S1_level = supports[0] ?? price * 0.95;
   694→  const S2_level = supports[1] ?? price * 0.90;
   695→  const S3_level = supports[2] ?? price * 0.85;
   696→  const S4_level = supports[3] ?? price * 0.80;
   697→  const S5_level = supports[4] ?? price * 0.75;
   698→  const S6_level = supports[5] ?? price * 0.70;
   699→
   700→  // ── Bull Score: weighted average of signals (excl. volume when absent) ─────
   701→  const signals: number[] = [];
   702→

   704→  signals.push(rsi > 70 ? 1 : rsi > 50 ? 0.5 : 0);
   705→  // 2. MFI (volume-dependent)

   710→  signals.push(stoch.k > 80 ? 1 : stoch.k > 50 ? 0.5 : 0);

   712→  signals.push(williamsR > -20 ? 1 : williamsR > -50 ? 0.5 : 0);

   716→  signals.push(adxResult.diPlus > adxResult.diMinus ? 1 : 0);
   717→  // 8-13. Price vs MAs
   718→  signals.push(price > (smaResult.sma21 ?? 0) ? 1 : 0);
   719→  signals.push(price > (smaResult.sma50 ?? 0) ? 1 : 0);
   720→  signals.push(price > (smaResult.sma100 ?? 0) ? 1 : 0);

   722→  signals.push(price > (emaResult.ema12 ?? 0) ? 1 : 0);
   723→  signals.push(price > (emaResult.ema26 ?? 0) ? 1 : 0);
   724→  // 14. Price vs SAR
   725→  signals.push(price > sar ? 1 : 0);
   726→  // 15. OBV trend (volume-dependent)
   727→  if (hasVolume) {
   728→    const obvSlice = Math.min(10, data.length - 1);
   729→    const obvTrend = obvSlice > 0 && data.length > obvSlice
   730→      ? (data.slice(-obvSlice).reduce((s, d, _, arr) => s + (d.close > arr[Math.max(0, arr.length - obvSlice - 1)]?.close ? 1 : 0), 0) / obvSlice) : 0.5;

   733→  // 16. BB position
   734→  const bbRange = bb.upper - bb.lower;
   735→  const bbPos = bbRange === 0 ? 0.5 : Math.max(0, Math.min(1, (price - bb.lower) / bbRange));
