  1123→// === Probability Context Builder ==============================================

  1125→function buildContext(input: GraphInput): ProbContext {
  1126→  const adxNorm = clamp(input.adx / 60, 0, 1);
  1127→  const srProximity = clamp(
  1128→    1 - Math.min(input.distToR1, input.distToS1) * 10,
  1129→    0,
  1130→    1
  1131→  );
  1132→  const oscillatorExtreme = clamp(
  1133→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
  1134→    0,
  1135→    1
  1136→  );
  1137→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1138→
  1139→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1140→  const overboughtRisk =
  1141→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1142→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1143→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1144→
  1145→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1146→  const oversoldBounce =
  1147→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1148→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1149→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1150→
  1151→  return {
  1152→    input,
  1153→    adxNorm,
  1154→    srProximity,
  1155→    oscillatorExtreme,
  1156→    divergenceProxy,
  1157→    overboughtRisk,
  1158→    oversoldBounce,
  1159→  };
  1160→}
  1161→
  1162→// === Node Probability Computation =============================================
  1163→// For each non-terminal node, returns an array of probabilities for its
  1164→// outgoing edges. The order MUST match the order of edges in createEdges().
  1165→// =============================================================================
  1166→
  1167→function computeNodeProbabilities(
  1168→  nodeId: string,
  1169→  ctx: ProbContext,
  1170→  _numChildren: number
  1171→): number[] {
  1172→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1173→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1174→
  1175→  switch (nodeId) {
  1176→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1177→    case 'ROOT': {
  1178→      const trendW = adxNorm * 0.5 + input.mlTrend * 0.18 + 0.1;
  1179→      const breakW = srProximity * 0.3 + (1 - adxNorm) * 0.08 + 0.08;
  1180→      const revW = ctx.divergenceProxy * 0.35 + (1 - adxNorm) * 0.12 + 0.06;
  1181→      return normalize([trendW, breakW, revW]);
  1182→    }
  1183→
  1184→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1185→    case 'N_TREND': {
  1186→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1187→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1188→      const flatUncertainty = (1 - Math.abs(input.bullConsensus - 0.5) * 2);
  1189→      const flat = flatUncertainty * clamp(1 - adxNorm * 1.5, 0, 1) * 0.45;
  1190→      return normalize([bull, bear, flat]);
  1191→    }
  1192→
  1193→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1194→    case 'N_T_BULL': {
  1195→      const momHigh =
  1196→        clamp(
  1197→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1198→            (input.macdHist > 0
  1199→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1200→              : 0),
  1201→          0,
  1202→          1
  1203→        ) * input.mlMomentum;
  1204→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1205→      return normalize([momHigh, momMod, overboughtRisk]);
  1206→    }
  1207→
  1208→    // ── N_T_B_MOM_HIGH → [SC8, SC9, SC7] ────────────────────────────────────
  1209→    // Strong bullish momentum — all three outcomes are bullish (no range)
  1210→    case 'N_T_B_MOM_HIGH': {
  1211→      const pSC8 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1212→      const pSC9 = 0.15 + input.strengthIndex * 0.2;
  1213→      const pSC7 = 0.35 * input.confidenceIndex + 0.15;
  1214→      return normalize([pSC8, pSC9, pSC7]);
  1215→    }
  1216→
  1217→    // ── N_T_B_MOM_MOD → [SC7, SC6, SC5] ─────────────────────────────────────
  1218→    // Moderate bullish — small explicit SC5 based on uncertainty
  1219→    case 'N_T_B_MOM_MOD': {
  1220→      const pSC7 = 0.55 * input.confidenceIndex + 0.2;
  1221→      const pSC6 = 0.55 * (1 - input.confidenceIndex) + 0.15;
  1222→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.02, 0.15);
  1223→      return normalize([pSC7, pSC6, pSC5]);
  1224→    }
  1225→
  1226→    // ── N_T_B_OVERBOUGHT → [SC4, SC1, SC9] ──────────────────────────────────
  1227→    // Overbought in uptrend — correction or last-gasp surge (no range)
  1228→    case 'N_T_B_OVERBOUGHT': {
  1229→      const pSC4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1230→      const pSC1 = 0.15 + input.strengthIndex * 0.15;
  1231→      const pSC9 = 0.1 + input.mlMomentum * 0.2;
  1232→      return normalize([pSC4, pSC1, pSC9]);
  1233→    }
  1234→
  1235→    // ── N_T_BEAR → [N_T_BE_MOM_LOW, N_T_BE_MOM_MOD, N_T_BE_OVERSOLD] ────
  1236→    case 'N_T_BEAR': {
  1237→      const momLow =
  1238→        clamp(
  1239→          (input.momentum < 0 ? Math.abs(input.momentum) / 50 : 0) +
  1240→            (input.macdHist < 0
  1241→              ? (Math.abs(input.macdHist) / (atrSafe * 0.5)) * 0.3
  1242→              : 0),
  1243→          0,
  1244→          1
  1245→        ) * input.mlMomentum;
  1246→      const momMod = (1 - momLow) * 0.6 * input.mlMomentum;
  1247→      return normalize([momLow, momMod, oversoldBounce]);
  1248→    }
  1249→
  1250→    // ── N_T_BE_MOM_LOW → [SC2, SC1, SC3] ────────────────────────────────────
  1251→    // Strong bearish momentum — all three outcomes are bearish (no range)

  1493→// === Graph Traversal ==========================================================
  1494→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1495→// =============================================================================
  1496→
  1497→function traverseGraph(
  1498→  nodeId: string,
  1499→  pathProb: number,
  1500→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1501→  ctx: ProbContext,
  1502→  adj: Map<string, number[]>,
  1503→  edges: GraphEdge[],
  1504→  edgeProbabilities: Record<number, number>,
  1505→  nodeAccum: Record<string, number>,
  1506→  scenarios: Record<string, number>,
  1507→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1508→): void {
  1509→  // Accumulate path probability at this node
  1510→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1511→
  1512→  const outEdgeIndices = adj.get(nodeId);
  1513→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1514→
  1515→  // Compute outgoing edge probabilities for this node
  1516→  const probs = computeNodeProbabilities(
  1517→    nodeId,
  1518→    ctx,
  1519→    outEdgeIndices.length
  1520→  );
  1521→
  1522→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1523→    const edgeIdx = outEdgeIndices[i];
  1524→    const edge = edges[edgeIdx];
  1525→    const prob = probs[i] || 0;
  1526→
  1527→    // Store edge probability
  1528→    edgeProbabilities[edgeIdx] = prob;
  1529→
  1530→    // Determine branch for child nodes
  1531→    let childBranch = currentBranch;
  1532→    if (nodeId === 'ROOT') {
  1533→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1534→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1535→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1536→    }
  1537→
  1538→    const childPathProb = pathProb * prob;
  1539→
  1540→    // Check if target is a terminal node (SC1–SC9)
  1541→    const isTerminal =
  1542→      edge.to.length === 3 &&
  1543→      edge.to[0] === 'S' &&
  1544→      edge.to[1] === 'C' &&
  1545→      edge.to[2] >= '1' &&
  1546→      edge.to[2] <= '9';
  1547→
  1548→    if (isTerminal) {
  1549→      // Accumulate scenario probability
  1550→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1551→      // Accumulate branch contribution
  1552→      if (childBranch) {
  1553→        contributions[edge.to][childBranch] += childPathProb;
  1554→      }
  1555→    }
  1556→
  1557→    // Recurse into child
  1558→    traverseGraph(
  1559→      edge.to,
  1560→      childPathProb,
  1561→      childBranch,
  1562→      ctx,
  1563→      adj,
  1564→      edges,
  1565→      edgeProbabilities,
  1566→      nodeAccum,
  1567→      scenarios,
  1568→      contributions
  1569→    );
  1570→  }
  1571→}

  1669→  // guaranteed sum=100 with each value in [2, 35].
  1670→  const rawFloats: number[] = [];
  1671→  for (let i = 1; i <= 9; i++) {
  1672→    const key = `SC${i}`;
  1673→    rawFloats.push(safeNum(scenarios[key], 0) * 100);
  1674→  }
  1675→
  1676→  const roundedInts = enforceSumTo100(rawFloats, 2, 35);
  1677→
  1678→  const finalPcts: Record<string, number> = {};
  1679→  for (let i = 1; i <= 9; i++) {
  1680→    const key = `SC${i}`;
  1681→    finalPcts[key] = roundedInts[i - 1];
  1682→  }
  1683→
  1684→  // 7. Compute branch probabilities (from ROOT outgoing edges)
  1685→  const rootEdgeIndices = adj.get('ROOT') || [];
  1686→  const rootProbs = computeNodeProbabilities('ROOT', ctx, rootEdgeIndices.length);
  1687→  const branchProbabilities = {
  1688→    trend: rootProbs[0] || 0,
  1689→    breakout: rootProbs[1] || 0,
  1690→    reversal: rootProbs[2] || 0,
  1691→  };
  1692→
  1693→  // 8. Build node display values (path probability as percentage)
  1694→  const nodeValues: Record<string, string> = {};
  1695→  for (const [nodeId, accum] of Object.entries(nodeAccum)) {
  1696→    nodeValues[nodeId] = `${Math.round(accum * 100)}%`;
  1697→  }
  1698→
  1699→  // Override terminal node values with final post-processed percentages
  1700→  for (const [key, pct] of Object.entries(finalPcts)) {
  1701→    nodeValues[key] = `${pct}%`;
  1702→  }
  1703→
  1704→  // 9. Build path contributions (as percentages)
  1705→  const pathContributions: Record<
  1706→    string,
  1707→    { trend: number; breakout: number; reversal: number }
  1708→  > = {};
  1709→  for (let i = 1; i <= 9; i++) {
  1710→    const key = `SC${i}`;
  1711→    const c = contributions[key];
  1712→    const total = c.trend + c.breakout + c.reversal;
  1713→    if (total > 0) {
  1714→      pathContributions[key] = {
  1715→        trend: Math.round((c.trend / total) * 100),
  1716→        breakout: Math.round((c.breakout / total) * 100),
  1717→        reversal: Math.round((c.reversal / total) * 100),
  1718→      };
  1719→    } else {
  1720→      pathContributions[key] = { trend: 33, breakout: 34, reversal: 33 };
  1721→    }
  1722→  }
  1723→
  1724→  // 10. Compute 30-day probability trend
  1725→  const probFractions: Record<string, number> = {};
  1726→  for (const [k, v] of Object.entries(finalPcts)) {
  1727→    probFractions[k] = v / 100;
  1728→  }
  1729→  const probabilityTrend = calcTrendFromProbs(probFractions, 30);
  1730→
  1731→  // 11. Return complete GraphData
  1732→  return {
  1733→    nodes,
  1734→    edges,
  1735→    nodePositions,
  1736→    edgeProbabilities,
  1737→    nodeValues,
  1738→    branchProbabilities,
  1739→    scenarioProbabilities: finalPcts,
  1740→    pathContributions,
  1741→    probabilityTrend,
  1742→  };
  1743→}
