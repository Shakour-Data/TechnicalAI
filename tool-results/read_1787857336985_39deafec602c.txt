    15→// === Helper Functions =========================================================
    16→
    17→function clamp(v: number, min: number, max: number): number {
    18→  return Math.max(min, Math.min(max, v));
    19→}
    20→
    21→function normalize(arr: number[]): number[] {
    22→  const sum = arr.reduce((a, b) => a + b, 0);
    23→  if (sum === 0) return arr.map(() => 1 / arr.length);
    24→  return arr.map((v) => v / sum);
    25→}
    26→
    27→function sigmoid(x: number): number {
    28→  return 1 / (1 + Math.exp(-x));
    29→}

  1001→function buildContext(input: GraphInput): ProbContext {
  1002→  const adxNorm = clamp(input.adx / 60, 0, 1);
  1003→  const srProximity = clamp(
  1004→    1 - Math.min(input.distToR1, input.distToS1) * 10,
  1005→    0,
  1006→    1
  1007→  );
  1008→  const oscillatorExtreme = clamp(
  1009→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
  1010→    0,
  1011→    1
  1012→  );
  1013→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1014→
  1015→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1016→  const overboughtRisk =
  1017→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1018→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1019→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1020→
  1021→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1022→  const oversoldBounce =
  1023→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1024→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1025→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1026→
  1027→  return {
  1028→    input,
  1029→    adxNorm,
  1030→    srProximity,
  1031→    oscillatorExtreme,
  1032→    divergenceProxy,
  1033→    overboughtRisk,
  1034→    oversoldBounce,
  1035→  };
  1036→}
  1037→
  1038→// === Node Probability Computation =============================================
  1039→// For each non-terminal node, returns an array of probabilities for its
  1040→// outgoing edges. The order MUST match the order of edges in createEdges().
  1041→// =============================================================================
  1042→
  1043→function computeNodeProbabilities(
  1044→  nodeId: string,
  1045→  ctx: ProbContext,
  1046→  _numChildren: number
  1047→): number[] {
  1048→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1049→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1050→
  1051→  switch (nodeId) {
  1052→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1053→    case 'ROOT': {
  1054→      const trendW = adxNorm * 0.45 + input.mlTrend * 0.15 + 0.12;
  1055→      const breakW = srProximity * 0.35 + (1 - adxNorm) * 0.1 + 0.1;
  1056→      const revW = ctx.divergenceProxy * 0.3 + (1 - adxNorm) * 0.15 + 0.08;
  1057→      return normalize([trendW, breakW, revW]);
  1058→    }
  1059→
  1060→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1061→    case 'N_TREND': {
  1062→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1063→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1064→      const flat =
  1065→        (1 - Math.abs(input.bullConsensus - 0.5) * 2) * (1 - adxNorm * 0.7);
  1066→      return normalize([bull, bear, flat]);
  1067→    }
  1068→
  1069→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1070→    case 'N_T_BULL': {
  1071→      const momHigh =
  1072→        clamp(
  1073→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1074→            (input.macdHist > 0
  1075→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1076→              : 0),
  1077→          0,
  1078→          1
  1079→        ) * input.mlMomentum;
  1080→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1081→      return normalize([momHigh, momMod, overboughtRisk]);
  1082→    }
  1083→
  1084→    // ── N_T_B_MOM_HIGH → [R8, R9, R7] ────────────────────────────────────
  1085→    // Strong bullish momentum — all three outcomes are bullish (no range)
  1086→    case 'N_T_B_MOM_HIGH': {
  1087→      const pR8 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1088→      const pR9 = 0.15 + input.strengthIndex * 0.2;
  1089→      const pR7 = 0.35 * input.confidenceIndex + 0.15;
  1090→      return normalize([pR8, pR9, pR7]);
  1091→    }
  1092→
  1093→    // ── N_T_B_MOM_MOD → [R7, R6, R5] ─────────────────────────────────────
  1094→    // Moderate bullish — small explicit R5 based on uncertainty
  1095→    case 'N_T_B_MOM_MOD': {
  1096→      const pR7 = 0.5 * input.confidenceIndex + 0.2;
  1097→      const pR6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1098→      const pR5 = clamp((1 - adxNorm) * 0.35, 0.05, 0.25);
  1099→      return normalize([pR7, pR6, pR5]);
  1100→    }
  1101→
  1102→    // ── N_T_B_OVERBOUGHT → [R4, R1, R9] ──────────────────────────────────
  1103→    // Overbought in uptrend — correction or last-gasp surge (no range)
  1104→    case 'N_T_B_OVERBOUGHT': {
  1105→      const pR4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1106→      const pR1 = 0.15 + input.strengthIndex * 0.15;
  1107→      const pR9 = 0.1 + input.mlMomentum * 0.2;
  1108→      return normalize([pR4, pR1, pR9]);
  1109→    }
  1110→
  1111→    // ── N_T_BEAR → [N_T_BE_MOM_LOW, N_T_BE_MOM_MOD, N_T_BE_OVERSOLD] ────
  1112→    case 'N_T_BEAR': {
  1113→      const momLow =
  1114→        clamp(
  1115→          (input.momentum < 0 ? Math.abs(input.momentum) / 50 : 0) +
  1116→            (input.macdHist < 0
  1117→              ? (Math.abs(input.macdHist) / (atrSafe * 0.5)) * 0.3
  1118→              : 0),
  1119→          0,
  1120→          1
  1121→        ) * input.mlMomentum;
  1122→      const momMod = (1 - momLow) * 0.6 * input.mlMomentum;
  1123→      return normalize([momLow, momMod, oversoldBounce]);
  1124→    }
  1125→
  1126→    // ── N_T_BE_MOM_LOW → [R2, R1, R3] ────────────────────────────────────
  1127→    // Strong bearish momentum — all three outcomes are bearish (no range)
  1128→    case 'N_T_BE_MOM_LOW': {
  1129→      const pR2 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1130→      const pR1 = 0.15 + input.strengthIndex * 0.2;
  1131→      const pR3 = 0.35 * input.confidenceIndex + 0.15;
  1132→      return normalize([pR2, pR1, pR3]);
  1133→    }
  1134→
  1135→    // ── N_T_BE_MOM_MOD → [R3, R4, R5] ────────────────────────────────────
  1136→    // Moderate bearish — small explicit R5 based on uncertainty
  1137→    case 'N_T_BE_MOM_MOD': {
  1138→      const pR3 = 0.5 * input.confidenceIndex + 0.2;
  1139→      const pR4 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1140→      const pR5 = clamp((1 - adxNorm) * 0.35, 0.05, 0.25);
  1141→      return normalize([pR3, pR4, pR5]);
  1142→    }
  1143→
  1144→    // ── N_T_BE_OVERSOLD → [R6, R9, R4] ───────────────────────────────────
  1145→    // Oversold in downtrend — bounce or mild continuation (no range)
  1146→    case 'N_T_BE_OVERSOLD': {
  1147→      const pR6 = 0.55 + (1 - input.mlMomentum) * 0.2;
  1148→      const pR9 = 0.15 + input.strengthIndex * 0.15;
  1149→      const pR4 = 0.1 + input.mlMomentum * 0.2;
  1150→      return normalize([pR6, pR9, pR4]);
  1151→    }
  1152→
  1153→    // ── N_T_FLAT → [N_T_F_VOL_LOW, N_T_F_VOL_HIGH, N_T_F_VOL_MOD] ────────
  1154→    case 'N_T_FLAT': {
  1155→      const volLow =
  1156→        clamp(1 - (input.atr / input.price) * 20, 0, 1) *
  1157→        (1 - input.mlVolatility) *
  1158→        0.7;
  1159→      const volHigh =
  1160→        clamp((input.atr / input.price) * 20, 0, 1) * input.mlVolatility * 0.7;
  1161→      const volMod = (1 - volLow - volHigh) * 0.5 + 0.1;
  1162→      return normalize([volLow, volHigh, volMod]);
  1163→    }
  1164→
  1165→    // ── N_T_F_VOL_LOW → [R5] ─────────────────────────────────────────────
  1166→    case 'N_T_F_VOL_LOW': {
  1167→      return [1.0];
  1168→    }
  1169→
  1170→    // ── N_T_F_VOL_HIGH → [R9, R1, R5] ────────────────────────────────────
  1171→    // High volatility shock — directional shock more likely than range
  1172→    case 'N_T_F_VOL_HIGH': {
  1173→      const pR9 = 0.35 * input.bullConsensus + 0.15;
  1174→      const pR1 = 0.35 * (1 - input.bullConsensus) + 0.15;
  1175→      const pR5 = clamp((1 - input.mlVolatility) * 0.2, 0.05, 0.2);
  1176→      return normalize([pR9, pR1, pR5]);
  1177→    }
  1178→
  1179→    // ── N_T_F_VOL_MOD → [R6, R4, R5] ─────────────────────────────────────
  1180→    // Moderate volatility — range is plausible but capped
  1181→    case 'N_T_F_VOL_MOD': {
  1182→      const pR6 = 0.35 * input.bullConsensus + 0.12;
  1183→      const pR4 = 0.35 * (1 - input.bullConsensus) + 0.12;
  1184→      const pR5 = clamp((1 - adxNorm) * 0.3, 0.08, 0.25);
  1185→      return normalize([pR6, pR4, pR5]);
  1186→    }
  1187→
  1188→    // ── N_BREAK → [N_B_UP, N_B_DOWN, N_B_NONE] ────────────────────────────
  1189→    case 'N_BREAK': {
  1190→      const upProb = input.bullConsensus * srProximity * 0.7 + 0.05;
  1191→      const downProb = (1 - input.bullConsensus) * srProximity * 0.7 + 0.05;
  1192→      const noneProb = (1 - srProximity) * 0.5 + 0.1;
  1193→      return normalize([upProb, downProb, noneProb]);
  1194→    }
  1195→
  1196→    // ── N_B_UP → [N_B_U_VOL_C, N_B_U_VOL_W] ─────────────────────────────
  1197→    case 'N_B_UP': {
  1198→      let volConfirm: number;
  1199→      let volWeak: number;
  1200→      if (input.hasVolume) {
  1201→        volConfirm = clamp(input.mfi / 60, 0, 1) * 0.6 + 0.2;
  1202→        volWeak = 1 - volConfirm;
  1203→      } else {
  1204→        volConfirm = 0.4;
  1205→        volWeak = 0.6;
  1206→      }
  1207→      return normalize([volConfirm, volWeak]);
  1208→    }
  1209→
  1210→    // ── N_B_U_VOL_C → [R7, R8, R9] ──────────────────────────────────────
  1211→    // Confirmed bullish breakout — all bullish, no range
  1212→    case 'N_B_U_VOL_C': {
  1213→      const pR7 = 0.45 + input.confidenceIndex * 0.15;
  1214→      const pR8 = 0.2 + input.strengthIndex * 0.15;
  1215→      const pR9 = 0.15 + input.strengthIndex * 0.1;
  1216→      return normalize([pR7, pR8, pR9]);
  1217→    }
  1218→
  1219→    // ── N_B_U_VOL_W → [R6, R4, R5] ──────────────────────────────────────
  1220→    // Weak volume breakout — fake breakout risk, capped R5
  1221→    case 'N_B_U_VOL_W': {
  1222→      const pR6 = 0.4 + input.bullConsensus * 0.15;
  1223→      const pR4 = 0.3 + (1 - input.bullConsensus) * 0.15;
  1224→      const pR5 = clamp((1 - input.confidenceIndex) * 0.25, 0.05, 0.2);
  1225→      return normalize([pR6, pR4, pR5]);
  1226→    }
  1227→
  1228→    // ── N_B_DOWN → [N_B_D_VOL_C, N_B_D_VOL_W] ────────────────────────────
  1229→    case 'N_B_DOWN': {
  1230→      let volConfirm: number;
  1231→      if (input.hasVolume) {
  1232→        volConfirm = clamp((100 - input.mfi) / 60, 0, 1) * 0.6 + 0.2;
  1233→      } else {
  1234→        volConfirm = 0.4;
  1235→      }
  1236→      const volWeak = 1 - volConfirm;
  1237→      return normalize([volConfirm, volWeak]);
  1238→    }
  1239→
  1240→    // ── N_B_D_VOL_C → [R3, R2, R1] ──────────────────────────────────────
  1241→    // Confirmed bearish breakout — all bearish, no range
  1242→    case 'N_B_D_VOL_C': {
  1243→      const pR3 = 0.45 + input.confidenceIndex * 0.15;
  1244→      const pR2 = 0.2 + input.strengthIndex * 0.15;
  1245→      const pR1 = 0.15 + input.strengthIndex * 0.1;
  1246→      return normalize([pR3, pR2, pR1]);
  1247→    }
  1248→
  1249→    // ── N_B_D_VOL_W → [R4, R6, R5] ──────────────────────────────────────
  1250→    // Weak volume bearish breakout — fake breakout risk, capped R5
  1251→    case 'N_B_D_VOL_W': {
  1252→      const pR4 = 0.4 + (1 - input.bullConsensus) * 0.15;
  1253→      const pR6 = 0.3 + input.bullConsensus * 0.15;
  1254→      const pR5 = clamp((1 - input.confidenceIndex) * 0.25, 0.05, 0.2);
  1255→      return normalize([pR4, pR6, pR5]);
  1256→    }
  1257→
  1258→    // ── N_B_NONE → [R5] ──────────────────────────────────────────────────
  1259→    case 'N_B_NONE': {
  1260→      return [1.0];
  1261→    }
  1262→
  1263→    // ── N_REVERSAL → [N_R_BULL, N_R_BEAR, N_R_NONE] ──────────────────────
  1264→    case 'N_REVERSAL': {
  1265→      const bullRev = overboughtRisk;
  1266→      const bearRev = oversoldBounce;
  1267→      const noneRev =
  1268→        1 - Math.max(overboughtRisk, oversoldBounce) * 0.7;
  1269→      return normalize([bullRev, bearRev, noneRev]);
  1270→    }
  1271→
  1272→    // ── N_R_BULL → [N_R_B_DIV, N_R_B_CANDLE, N_R_B_SR] ──────────────────
  1273→    case 'N_R_BULL': {
  1274→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1275→      const candleProb =
  1276→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1277→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1278→      return normalize([divProb, candleProb, srProb]);
  1279→    }
  1280→
  1281→    // ── N_R_B_DIV → [R6, R8, R5] ────────────────────────────────────────
  1282→    // Bullish divergence — explicit small R5
  1283→    case 'N_R_B_DIV': {
  1284→      const pR6 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1285→      const pR8 = 0.2 + input.mlMomentum * 0.15;
  1286→      const pR5 = clamp((1 - adxNorm) * 0.25, 0.05, 0.2);
  1287→      return normalize([pR6, pR8, pR5]);
  1288→    }
  1289→
  1290→    // ── N_R_B_CANDLE → [R7, R6, R5] ─────────────────────────────────────
  1291→    // Bullish candle pattern — explicit small R5
  1292→    case 'N_R_B_CANDLE': {
  1293→      const pR7 = 0.5 * input.confidenceIndex + 0.2;
  1294→      const pR6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1295→      const pR5 = clamp((1 - input.confidenceIndex) * 0.2, 0.05, 0.18);
  1296→      return normalize([pR7, pR6, pR5]);
  1297→    }
  1298→
  1299→    // ── N_R_B_SR → [R6, R5, R4] ─────────────────────────────────────────
  1300→    // Support bounce — explicit small R5
  1301→    // Edge order: R6, R5, R4 — probabilities computed then reordered
  1302→    case 'N_R_B_SR': {
  1303→      const pR6 = 0.5 + input.srAvgStrength * 0.2;
  1304→      const pR4 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1305→      const pR5 = clamp((1 - input.srAvgStrength) * 0.2, 0.05, 0.18);
  1306→      // Return in edge order: [R6, R5, R4]
  1307→      return normalize([pR6, pR5, pR4]);
  1308→    }
  1309→
  1310→    // ── N_R_BEAR → [N_R_BE_DIV, N_R_BE_CANDLE, N_R_BE_SR] ───────────────
  1311→    // Mirror of N_R_BULL
  1312→    case 'N_R_BEAR': {
  1313→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1314→      const candleProb =
  1315→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1316→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1317→      return normalize([divProb, candleProb, srProb]);
  1318→    }
  1319→
  1320→    // ── N_R_BE_DIV → [R4, R2, R5] ───────────────────────────────────────
  1321→    // Bearish divergence — explicit small R5
  1322→    case 'N_R_BE_DIV': {
  1323→      const pR4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1324→      const pR2 = 0.2 + input.mlMomentum * 0.15;
  1325→      const pR5 = clamp((1 - adxNorm) * 0.25, 0.05, 0.2);
  1326→      return normalize([pR4, pR2, pR5]);
  1327→    }
  1328→
  1329→    // ── N_R_BE_CANDLE → [R3, R4, R5] ────────────────────────────────────
  1330→    // Bearish candle pattern — explicit small R5
  1331→    case 'N_R_BE_CANDLE': {
  1332→      const pR3 = 0.5 * input.confidenceIndex + 0.2;
  1333→      const pR4 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1334→      const pR5 = clamp((1 - input.confidenceIndex) * 0.2, 0.05, 0.18);
  1335→      return normalize([pR3, pR4, pR5]);
  1336→    }
  1337→
  1338→    // ── N_R_BE_SR → [R4, R5, R6] ─────────────────────────────────────────
  1339→    // Resistance rejection — explicit small R5
  1340→    // Edge order: R4, R5, R6 — probabilities computed then reordered
  1341→    case 'N_R_BE_SR': {
  1342→      const pR4 = 0.5 + input.srAvgStrength * 0.2;
  1343→      const pR6 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1344→      const pR5 = clamp((1 - input.srAvgStrength) * 0.2, 0.05, 0.18);
  1345→      // Return in edge order: [R4, R5, R6]
  1346→      return normalize([pR4, pR5, pR6]);
  1347→    }
  1348→
  1349→    // ── N_R_NONE → [R5] ──────────────────────────────────────────────────
  1350→    case 'N_R_NONE': {
  1351→      return [1.0];
  1352→    }
  1353→
  1354→    // ── Terminal nodes (R1–R9) have no outgoing edges ─────────────────────
  1355→    default:
  1356→      return [];
  1357→  }
  1358→}
  1359→
  1360→// === Graph Traversal ==========================================================
  1361→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1362→// =============================================================================
  1363→
  1364→function traverseGraph(
  1365→  nodeId: string,
  1366→  pathProb: number,
  1367→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1368→  ctx: ProbContext,
  1369→  adj: Map<string, number[]>,
  1370→  edges: GraphEdge[],
  1371→  edgeProbabilities: Record<number, number>,
  1372→  nodeAccum: Record<string, number>,
  1373→  scenarios: Record<string, number>,
  1374→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1375→): void {
  1376→  // Accumulate path probability at this node
  1377→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1378→
  1379→  const outEdgeIndices = adj.get(nodeId);
  1380→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1381→
  1382→  // Compute outgoing edge probabilities for this node
  1383→  const probs = computeNodeProbabilities(
  1384→    nodeId,
  1385→    ctx,
  1386→    outEdgeIndices.length
  1387→  );
  1388→
  1389→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1390→    const edgeIdx = outEdgeIndices[i];
  1391→    const edge = edges[edgeIdx];
  1392→    const prob = probs[i] || 0;
  1393→
  1394→    // Store edge probability
  1395→    edgeProbabilities[edgeIdx] = prob;
  1396→
  1397→    // Determine branch for child nodes
  1398→    let childBranch = currentBranch;
  1399→    if (nodeId === 'ROOT') {
  1400→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1401→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1402→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1403→    }
  1404→
  1405→    const childPathProb = pathProb * prob;
  1406→
  1407→    // Check if target is a terminal node
  1408→    const isTerminal =
  1409→      edge.to.length === 2 &&
  1410→      edge.to[0] === 'R' &&
  1411→      edge.to[1] >= '1' &&
  1412→      edge.to[1] <= '9';
  1413→
  1414→    if (isTerminal) {
  1415→      // Accumulate scenario probability
  1416→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1417→      // Accumulate branch contribution
  1418→      if (childBranch) {
  1419→        contributions[edge.to][childBranch] += childPathProb;
  1420→      }
  1421→    }
  1422→
  1423→    // Recurse into child
  1424→    traverseGraph(
  1425→      edge.to,
  1426→      childPathProb,
  1427→      childBranch,
  1428→      ctx,
  1429→      adj,
  1430→      edges,
  1431→      edgeProbabilities,
  1432→      nodeAccum,
  1433→      scenarios,
  1434→      contributions
  1435→    );
  1436→  }
  1437→}

  1439→// === Main Export: buildDecisionGraph ===========================================
  1440→
  1441→export function buildDecisionGraph(input: GraphInput): GraphData {
  1442→  // 1. Build static graph structure
  1443→  const nodes = createNodes();
  1444→  const edges = createEdges();
  1445→  const nodePositions = createPositions();
  1446→
  1447→  // 2. Build adjacency map: nodeId → array of edge indices
  1448→  const adj = new Map<string, number[]>();
  1449→  for (let i = 0; i < edges.length; i++) {
  1450→    const from = edges[i].from;
  1451→    if (!adj.has(from)) adj.set(from, []);
  1452→    adj.get(from)!.push(i);
  1453→  }
  1454→
  1455→  // 3. Build probability context
  1456→  const ctx = buildContext(input);
  1457→
  1458→  // 4. Initialize accumulation structures
  1459→  const edgeProbabilities: Record<number, number> = {};
  1460→  const nodeAccum: Record<string, number> = {};
  1461→  const scenarios: Record<string, number> = {};
  1462→  const contributions: Record<
  1463→    string,
  1464→    { trend: number; breakout: number; reversal: number }
  1465→  > = {};
  1466→
  1467→  for (let i = 1; i <= 9; i++) {
  1468→    const key = `R${i}`;
  1469→    scenarios[key] = 0;
  1470→    contributions[key] = { trend: 0, breakout: 0, reversal: 0 };
  1471→  }
  1472→
  1473→  // 5. Traverse graph from ROOT
  1474→  traverseGraph(
  1475→    'ROOT',
  1476→    1.0,
  1477→    null,
  1478→    ctx,
  1479→    adj,
  1480→    edges,
  1481→    edgeProbabilities,
  1482→    nodeAccum,
  1483→    scenarios,
  1484→    contributions
  1485→  );
  1486→
  1487→  // 6. Post-process scenario probabilities
  1488→  // Convert to percentage, clamp [2, 35], re-normalize to sum=100
  1489→  const rawPcts: Record<string, number> = {};
  1490→  for (let i = 1; i <= 9; i++) {
  1491→    const key = `R${i}`;
  1492→    rawPcts[key] = Math.round((scenarios[key] || 0) * 100);
  1493→  }
  1494→
  1495→  // Clamp each to [2, 35]
  1496→  const clamped = Object.entries(rawPcts).map(([k, v]) => ({
  1497→    key: k,
  1498→    value: clamp(v, 2, 35),
  1499→  }));
  1500→
  1501→  // Re-normalize to sum=100
  1502→  const clampedSum = clamped.reduce((s, c) => s + c.value, 0);
  1503→  const finalPcts: Record<string, number> = {};
  1504→  if (clampedSum > 0) {
  1505→    // Distribute proportionally and round, then fix rounding errors
  1506→    let runningTotal = 0;
  1507→    const entries = clamped
  1508→      .map((c) => {
  1509→        const raw = (c.value / clampedSum) * 100;
  1510→        return { key: c.key, raw };
  1511→      })
  1512→      .sort((a, b) => b.raw - a.raw);
  1513→
  1514→    for (let i = 0; i < entries.length; i++) {
  1515→      const isLast = i === entries.length - 1;
  1516→      if (isLast) {
  1517→        finalPcts[entries[i].key] = Math.round(100 - runningTotal);
  1518→      } else {
  1519→        const rounded = Math.round(entries[i].raw);
  1520→        finalPcts[entries[i].key] = rounded;
  1521→        runningTotal += rounded;
  1522→      }
  1523→    }
  1524→  }
  1525→
  1526→  // Ensure no negative values due to rounding
  1527→  for (const key of Object.keys(finalPcts)) {
  1528→    if (finalPcts[key] < 0) finalPcts[key] = 0;
  1529→  }
  1530→
  1531→  // 7. Compute branch probabilities (from ROOT outgoing edges)
  1532→  const rootEdgeIndices = adj.get('ROOT') || [];
  1533→  const rootProbs = computeNodeProbabilities('ROOT', ctx, rootEdgeIndices.length);
  1534→  const branchProbabilities = {
  1535→    trend: rootProbs[0] || 0,
  1536→    breakout: rootProbs[1] || 0,
  1537→    reversal: rootProbs[2] || 0,
  1538→  };
  1539→
  1540→  // 8. Build node display values (path probability as percentage)
  1541→  const nodeValues: Record<string, string> = {};
  1542→  for (const [nodeId, accum] of Object.entries(nodeAccum)) {
  1543→    nodeValues[nodeId] = `${Math.round(accum * 100)}%`;
  1544→  }
  1545→
  1546→  // Override terminal node values with final post-processed percentages
  1547→  for (const [key, pct] of Object.entries(finalPcts)) {
  1548→    nodeValues[key] = `${pct}%`;
  1549→  }
  1550→
  1551→  // 9. Build path contributions (as percentages)
  1552→  const pathContributions: Record<
  1553→    string,
  1554→    { trend: number; breakout: number; reversal: number }
  1555→  > = {};
  1556→  for (let i = 1; i <= 9; i++) {
  1557→    const key = `R${i}`;
  1558→    const c = contributions[key];
  1559→    const total = c.trend + c.breakout + c.reversal;
  1560→    if (total > 0) {
  1561→      pathContributions[key] = {
  1562→        trend: Math.round((c.trend / total) * 100),
  1563→        breakout: Math.round((c.breakout / total) * 100),
  1564→        reversal: Math.round((c.reversal / total) * 100),
  1565→      };
  1566→    } else {
  1567→      pathContributions[key] = { trend: 33, breakout: 34, reversal: 33 };
  1568→    }
  1569→  }
  1570→
  1571→  // 10. Compute 30-day probability trend
  1572→  const probFractions: Record<string, number> = {};
  1573→  for (const [k, v] of Object.entries(finalPcts)) {
  1574→    probFractions[k] = v / 100;
  1575→  }
  1576→  const probabilityTrend = calcTrendFromProbs(probFractions, 30);
  1577→
  1578→  // 11. Return complete GraphData
  1579→  return {
  1580→    nodes,
  1581→    edges,
  1582→    nodePositions,
  1583→    edgeProbabilities,
  1584→    nodeValues,
  1585→    branchProbabilities,
  1586→    scenarioProbabilities: finalPcts,
  1587→    pathContributions,
  1588→    probabilityTrend,
  1589→  };
  1590→}
