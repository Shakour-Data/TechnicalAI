   149→// Graph Algorithms
   150→// ═══════════════════════════════════════════════════════════════════════════════
   151→
   152→interface EdgeInfo {
   153→  from: string;
   154→  to: string;
   155→  label: string;
   156→  type: string;
   157→  prob: number;
   158→}
   159→
   160→interface PathInfo {
   161→  nodes: string[];
   162→  edges: number[];
   163→  prob: number;
   164→  target: string;
   165→}
   166→
   167→// Calculate edge probabilities based on market conditions
   168→function calcEdgeProbabilities(
   169→  bullScore: number, adxVal: number
   170→): Record<string, number> {
   171→  const adxW = Math.min(adxVal / 100, 1);
   172→  const typeWeights: Record<string, number> = {
   173→    up: bullScore * 0.6 + adxW * 0.4,
   174→    pullback: 0.3,
   175→    down: (1 - bullScore) * 0.6 + adxW * 0.4,
   176→    risk: 0.15 * (1.2 - adxW),
   177→  };
   178→
   179→  // Build adjacency: for each node, group outgoing edges by type
   180→  const adj: Record<string, { idx: number; type: string }[]> = {};
   181→  EDGES.forEach((e, i) => {
   182→    if (!adj[e[0]]) adj[e[0]] = [];
   183→    adj[e[0]].push({ idx: i, type: e[3] });
   184→  });
   185→
   186→  const edgeProbs: Record<string, number> = {};
   187→
   188→  for (const [node, outEdges] of Object.entries(adj)) {
   189→    // Sum weights of outgoing edges
   190→    const totalWeight = outEdges.reduce((sum, e) => sum + (typeWeights[e.type] ?? 0.1), 0);
   191→    if (totalWeight === 0) {
   192→      outEdges.forEach(e => { edgeProbs[String(e.idx)] = 1 / outEdges.length; });
   193→      continue;
   194→    }
   195→    outEdges.forEach(e => {
   196→      edgeProbs[String(e.idx)] = (typeWeights[e.type] ?? 0.1) / totalWeight;
   197→    });
   198→  }
   199→
   200→  return edgeProbs;
   201→}
   202→
   203→// Find all paths from 'A' to terminal nodes using DFS
   204→function findAllPaths(edgeProbs: Record<string, number>): PathInfo[] {
   205→  const adj: Record<string, { to: string; edgeIdx: number }[]> = {};
   206→  EDGES.forEach((e, i) => {
   207→    if (!adj[e[0]]) adj[e[0]] = [];
   208→    adj[e[0]].push({ to: e[1], edgeIdx: i });
   209→  });
   210→
   211→  const paths: PathInfo[] = [];
   212→  const MAX_PATHS = 200;
   213→
   214→  function dfs(node: string, visited: Set<string>, currentPath: string[], edgeIndices: number[], currentProb: number) {
   215→    if (SCENARIO_KEYS.includes(node as any)) {
   216→      paths.push({ nodes: [...currentPath, node], edges: [...edgeIndices], prob: currentProb, target: node });
   217→      return;
   218→    }
   219→    if (paths.length >= MAX_PATHS) return;
   220→    if (visited.has(node)) return;
   221→    visited.add(node);
   222→
   223→    const outs = adj[node] ?? [];
   224→    for (const { to, edgeIdx } of outs) {
   225→      const ep = edgeProbs[String(edgeIdx)] ?? 0.1;
   226→      dfs(to, visited, [...currentPath, node], [...edgeIndices, edgeIdx], currentProb * ep);
   227→    }
   228→    visited.delete(node);
   229→  }
   230→
   231→  dfs('A', new Set(), [], [], 1);
   232→  return paths.sort((a, b) => b.prob - a.prob);
   233→}
