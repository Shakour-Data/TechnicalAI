    17→function sma(closes: number[], period: number): number {
    18→  if (closes.length < period) return 0;
    19→  const slice = closes.slice(-period);
    20→  return slice.reduce((a, b) => a + b, 0) / period;
    21→}
    22→
    23→function emaCalc(closes: number[], period: number): number {
    24→  if (closes.length < period) return 0;
    25→  const k = 2 / (period + 1);
    26→  let v = sma(closes.slice(0, period), period);
    27→  for (let i = period; i < closes.length; i++) {
    28→    v = closes[i] * k + v * (1 - k);
    29→  }
    30→  return v;
    31→}

    33→function calcRSI(closes: number[], period = 14): number {
    34→  if (closes.length < period + 1) return 50;
    35→  let avgGain = 0, avgLoss = 0;
    36→  for (let i = 1; i <= period; i++) {
    37→    const diff = closes[i] - closes[i - 1];
    38→    if (diff > 0) avgGain += diff; else avgLoss += Math.abs(diff);
    39→  }
    40→  avgGain /= period; avgLoss /= period;
    41→  for (let i = period + 1; i < closes.length; i++) {
    42→    const diff = closes[i] - closes[i - 1];
    43→    avgGain = (avgGain * (period - 1) + (diff > 0 ? diff : 0)) / period;
    44→    avgLoss = (avgLoss * (period - 1) + (diff < 0 ? Math.abs(diff) : 0)) / period;
    45→  }
    46→  if (avgLoss === 0) return 100;
    47→  return 100 - 100 / (1 + avgGain / avgLoss);
    48→}

    50→function calcCCI(data: OHLCV[], period = 20): number {
    51→  if (data.length < period) return 0;
    52→  const tps: number[] = [];
    53→  for (let i = data.length - period; i < data.length; i++) {
    54→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
    55→  }
    56→  const mean = tps.reduce((a, b) => a + b, 0) / tps.length;
    57→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - mean), 0) / tps.length;
    58→  return meanDev === 0 ? 0 : (tps[tps.length - 1] - mean) / (0.015 * meanDev);
    59→}
    60→
    61→function calcMFI(data: OHLCV[], period = 14): number {
    62→  if (data.length < period + 1) return 50;
    63→  let posFlow = 0, negFlow = 0;
    64→  for (let i = data.length - period; i < data.length; i++) {
    65→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
    66→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
    67→    const mf = tp * data[i].volume;
    68→    if (tp > prevTp) posFlow += mf; else negFlow += mf;
    69→  }
    70→  return negFlow === 0 ? 100 : 100 - 100 / (1 + posFlow / negFlow);
    71→}

    73→function calcStochastic(data: OHLCV[], kPeriod = 14, smoothK = 3, smoothD = 3): { k: number; d: number } {
    74→  const rawKs: number[] = [];
    75→  for (let i = kPeriod - 1; i < data.length; i++) {
    76→    let lowest = Infinity, highest = -Infinity;
    77→    for (let j = i - kPeriod + 1; j <= i; j++) {
    78→      if (data[j].low < lowest) lowest = data[j].low;
    79→      if (data[j].high > highest) highest = data[j].high;
    80→    }
    81→    const range = highest - lowest;
    82→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
    83→  }
    84→  const smoothedK: number[] = [];
    85→  for (let i = smoothK - 1; i < rawKs.length; i++) {
    86→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
    87→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
    88→  }
    89→  const k = smoothedK[smoothedK.length - 1] ?? 50;
    90→  const dSlice = smoothedK.slice(-smoothD);
    91→  const d = dSlice.length > 0 ? dSlice.reduce((a, b) => a + b, 0) / dSlice.length : 50;
    92→  return { k, d };
    93→}

    95→function calcMACD(closes: number[], fast = 12, slow = 26, sig = 9): { line: number; signal: number; histogram: number } {
    96→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
    97→  const kFast = 2 / (fast + 1), kSlow = 2 / (slow + 1), kSig = 2 / (sig + 1);
    98→  let eFast = sma(closes.slice(0, fast), fast);
    99→  let eSlow = sma(closes.slice(0, slow), slow);
   100→  const macdVals: number[] = [];
   101→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   102→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   103→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   104→    macdVals.push(eFast - eSlow);
   105→  }
   106→  if (macdVals.length < sig) return { line: 0, signal: 0, histogram: 0 };
   107→  let s = sma(macdVals.slice(0, sig), sig);
   108→  for (let i = sig; i < macdVals.length; i++) {
   109→    s = macdVals[i] * kSig + s * (1 - kSig);
   110→  }
   111→  const line = macdVals[macdVals.length - 1];
   112→  return { line, signal: s, histogram: line - s };
   113→}

   115→function calcBollingerBands(closes: number[], period = 20, mult = 2): { upper: number; middle: number; lower: number } {
   116→  if (closes.length < period) return { upper: 0, middle: 0, lower: 0 };
   117→  const slice = closes.slice(-period);
   118→  const mid = slice.reduce((a, b) => a + b, 0) / period;
   119→  const variance = slice.reduce((a, b) => a + (b - mid) ** 2, 0) / period;
   120→  const sd = Math.sqrt(variance);
   121→  return { upper: mid + mult * sd, middle: mid, lower: mid - mult * sd };
   122→}
   123→
   124→function calcADX(data: OHLCV[], period = 14): { adx: number; diPlus: number; diMinus: number } {
   125→  if (data.length < period * 2) return { adx: 0, diPlus: 0, diMinus: 0 };
   126→  let sumPDM = 0, sumMDM = 0, sumTR = 0;
   127→  for (let i = data.length - period; i < data.length; i++) {
   128→    const upMove = data[i].high - data[i - 1].high;
   129→    const downMove = data[i - 1].low - data[i].low;
   130→    sumPDM += upMove > downMove && upMove > 0 ? upMove : 0;
   131→    sumMDM += downMove > upMove && downMove > 0 ? downMove : 0;
   132→    sumTR += Math.max(
   133→      data[i].high - data[i].low,
   134→      Math.abs(data[i].high - data[i - 1].close),
   135→      Math.abs(data[i].low - data[i - 1].close)
   136→    );
   137→  }
   138→  const diPlus = sumTR > 0 ? (sumPDM / sumTR) * 100 : 0;
   139→  const diMinus = sumTR > 0 ? (sumMDM / sumTR) * 100 : 0;
   140→  const dx = (diPlus + diMinus) > 0 ? Math.abs(diPlus - diMinus) / (diPlus + diMinus) * 100 : 0;
   141→  return { adx: dx, diPlus, diMinus };
   142→}
   143→
   144→function calcATR(data: OHLCV[], period = 14): number {
   145→  if (data.length < period + 1) return 0;
   146→  let sum = 0;
   147→  for (let i = data.length - period; i < data.length; i++) {
   148→    sum += Math.max(
   149→      data[i].high - data[i].low,
   150→      Math.abs(data[i].high - data[i - 1].close),
   151→      Math.abs(data[i].low - data[i - 1].close)
   152→    );
   153→  }
   154→  return sum / period;
   155→}

   157→function calcSAR(data: OHLCV[]): number {
   158→  if (data.length < 5) return 0;
   159→  // Simplified Parabolic SAR
   160→  return data[data.length - 1].low; // placeholder
   161→}

   163→// ─── 16-Feature Extraction for VDss ─────────────────────────────────────────

   165→const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));
   166→
   167→/**
   168→ * Extract the 16 VDSS (Volatility-Direction-Strength-Structure) features at a
   169→ * given point in the OHLCV history. Feature order must match
   170→ * {@link VDSS_FEATURE_NAMES} exactly.
   171→ *
   172→ * The extraction proceeds in two layers:
   173→ * - **Layer 1** — Raw indicator scores (`s_*`) mapped into [0,1] via
   174→ *   threshold tables or clamped normalisation.
   175→ * - **Layer 2** — Momentum-corrected scores (`f_*`) that apply a small delta
   176→ *   adjustment based on the 5-bar lookback change, plus binary cross-signal
   177→ *   features (Stochastic cross, MACD cross, RSI divergence).
   178→ *
   179→ * ### The 16 Features (in order)
   180→ * | Index | Name            | Source                       | Layer |
   181→ * |-------|-----------------|------------------------------|-------|
   182→ * |  0    | RSI             | `calcRSI(14)`                | 2     |
   183→ * |  1    | MFI             | `calcMFI(14)` (skip if no vol)| 2    |
   184→ * |  2    | CCI             | `calcCCI(20)`                | 2     |
   185→ * |  3    | ADX             | `calcADX(14)` DI± diff       | 1     |
   186→ * |  4    | MACD            | `calcMACD(12,26,9)` histogram| 2     |
   187→ * |  5    | Stochastic      | `calcStochastic(14,3,3)`     | 2     |
   188→ * |  6    | BB_position     | Bollinger %B (20,2)          | 1     |
   189→ * |  7    | MA21            | Price vs SMA-21              | 1     |
   190→ * |  8    | MA100           | Price vs SMA-100             | 1     |
   191→ * |  9    | EMA_gap         | (EMA12-EMA26)/EMA26          | 1     |
   192→ * | 10    | ATR_pct         | 1 − ATR/price×10, clamped    | 1     |
   193→ * | 11    | Trend           | R² × |angle| regression      | 1     |
   194→ * | 12    | SR_distance     | Nearest S/R proximity        | 1     |
   195→ * | 13    | Stoch_cross     | K/D bullish cross (binary)   | 2     |
   196→ * | 14    | MACD_cross      | Line/Signal bullish cross    | 2     |
   197→ * | 15    | Divergence      | Price/RSI divergence (binary)| 2     |
   198→ *
   199→ * @param data     - Full OHLCV history (chronological, oldest first)
   200→ * @param endIdx   - Index of the candle to compute features for
   201→ * @param hasVolume - Whether the data contains valid volume bars
   202→ * @returns Array of exactly 16 feature values, each in [0,1].
   203→ *          Returns `new Array(16).fill(0.5)` if the price at `endIdx` is ≤ 0.
   204→ */
   205→export function extractVDSSFeatures(data: OHLCV[], endIdx: number, hasVolume: boolean): number[] {
   206→  const slice = data.slice(0, endIdx + 1);
   207→  const closes = slice.map(d => d.close);
   208→  const price = closes[closes.length - 1];
   209→  if (price <= 0) return new Array(16).fill(0.5);
   210→
   211→  // ── Compute indicators ──
   212→  const rsi = calcRSI(closes);
   213→  const mfi = hasVolume ? calcMFI(slice) : 50;
   214→  const cci = calcCCI(slice);
   215→  const adxResult = calcADX(slice);
   216→  const macd = calcMACD(closes);
   217→  const stoch = calcStochastic(slice);
   218→  const bb = calcBollingerBands(closes);
   219→  const atr = calcATR(slice);
   220→  const ma21 = sma(closes, 21);
   221→  const ma100 = sma(closes, 100);
   222→  const ema12 = emaCalc(closes, 12);
   223→  const ema26 = emaCalc(closes, 26);
   224→
   225→  // ── LAYER 1: Raw scores (s_*) ──
   226→  const s_rsi = rsi > 80 ? 0.95 : rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   227→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   228→  const s_mfi = mfi > 80 ? 0.95 : mfi > 70 ? 0.85 : mfi > 60 ? 0.7 : mfi > 50 ? 0.55
   229→    : mfi > 40 ? 0.45 : mfi > 30 ? 0.3 : mfi > 20 ? 0.15 : 0.05;
   230→  const s_cci = cci > 200 ? 0.9 : cci > 100 ? 0.75 : cci > 0 ? 0.6
   231→    : cci > -100 ? 0.4 : cci > -200 ? 0.25 : 0.1;
   232→  const s_adx = adxResult.diPlus > adxResult.diMinus
   233→    ? clamp(0.5 + (adxResult.diPlus - adxResult.diMinus) / 100, 0.5, 1)
   234→    : clamp(0.5 - (adxResult.diMinus - adxResult.diPlus) / 100, 0, 0.5);
   235→  const s_macd = macd.histogram > 0 && macd.line > macd.signal ? 0.9
   236→    : macd.histogram > 0 ? 0.7 : macd.line > macd.signal ? 0.55
   237→    : macd.histogram < 0 && macd.line < macd.signal ? 0.1 : 0.3;
   238→  const s_stoch = clamp(
   239→    (stoch.k > 80 ? 0.9 : stoch.k > 70 ? 0.8 : stoch.k > 50 ? 0.6
   240→      : stoch.k > 30 ? 0.4 : stoch.k > 20 ? 0.2 : 0.1)
   241→    + (stoch.k > stoch.d ? 0.1 : -0.1), 0, 1);
   242→  const bbRange = bb.upper - bb.lower;
   243→  const s_bb = bbRange === 0 ? 0.5 : clamp((price - bb.lower) / bbRange, 0, 1);
   244→  const s_ma21 = clamp(price > ma21 ? 0.7 + ((price - ma21) / ma21) * 0.5 : 0.3 + ((price - ma21) / ma21) * 0.5, 0, 1);
   245→  const s_ma100 = clamp(price > ma100 ? 0.65 + ((price - ma100) / ma100) * 0.3 : 0.35 + ((price - ma100) / ma100) * 0.3, 0, 1);
   246→  const emaGap = ema12 > 0 && ema26 > 0 ? (ema12 - ema26) / ema26 : 0;
   247→  const s_ema = clamp(0.5 + emaGap * 5, 0, 1);
   248→  const s_atr = clamp(1 - atr / price * 10, 0.2, 1);
   249→
   250→  // Trend (R² + angle)
   251→  const trendPeriod = Math.min(21, closes.length);
   252→  const trendSlice = closes.slice(-trendPeriod);
   253→  const n = trendSlice.length;
   254→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   255→  for (let i = 0; i < n; i++) { sumX += i; sumY += trendSlice[i]; sumXY += i * trendSlice[i]; sumX2 += i * i; sumY2 += trendSlice[i] ** 2; }
   256→  const denom = n * sumX2 - sumX * sumX;
   257→  let r2 = 0, angle = 0;
   258→  if (denom !== 0) {
   259→    const slope = (n * sumXY - sumX * sumY) / denom;
   260→    const meanY = sumY / n;
   261→    let ssRes = 0, ssTot = 0;
   262→    for (let i = 0; i < n; i++) {
   263→      ssRes += (trendSlice[i] - (slope * i + meanY - slope * (n - 1) / 2)) ** 2;
   264→      ssTot += (trendSlice[i] - meanY) ** 2;
   265→    }
   266→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
   267→    // Convert slope/meanY to percentage before atan (same fix as ta-engine.ts)
   268→    // so that 1% daily rise → 45°, not ~0.57°
   269→    angle = meanY > 0 ? Math.atan((slope / meanY) * 100) * (180 / Math.PI) : 0;
   270→  }
   271→  const tStr = r2 * Math.abs(angle) / 45;
   272→  const direction = denom !== 0 ? (n * sumXY - sumX * sumY) / denom : 0;
   273→  const s_trend = direction > 0 ? clamp(0.5 + tStr / 2, 0.5, 1)
   274→    : direction < 0 ? clamp(0.5 - tStr / 2, 0, 0.5) : 0.5;
   275→
   276→  // Simplified S/R: distance to nearest local high/low
   277→  let s_sr = 0.5;
   278→  if (slice.length > 20) {
   279→    const lookback = slice.slice(-30);
   280→    let nearestHigh = Infinity, nearestLow = 0;
   281→    for (const d of lookback) {
   282→      if (d.high > price && d.high < nearestHigh) nearestHigh = d.high;
   283→      if (d.low < price && d.low > nearestLow) nearestLow = d.low;
   284→    }
   285→    const dR = nearestHigh < Infinity ? Math.abs(price - nearestHigh) / nearestHigh : 1;
   286→    const dS = nearestLow > 0 ? Math.abs(price - nearestLow) / nearestLow : 1;
   287→    s_sr = clamp(0.5 - dR * 1.5 + dS * 1.0, 0, 1);
   288→  }
   289→
   290→  // ── LAYER 2: Momentum correction ──
   291→  const TB = 5;
   292→  let f_rsi = s_rsi, f_mfi = s_mfi, f_cci = s_cci, f_macd = s_macd, f_stoch = s_stoch;
   293→  if (slice.length > TB + 15) {
   294→    const pastSlice = slice.slice(0, -TB);
   295→    const pastCloses = pastSlice.map(d => d.close);
   296→    const rsiPast = calcRSI(pastCloses);
   297→    const mfiPast = hasVolume ? calcMFI(pastSlice) : 50;
   298→    const cciPast = calcCCI(pastSlice);
   299→    const macdPast = calcMACD(pastCloses);
   300→    const stochPast = calcStochastic(pastSlice);
   301→
   302→    f_rsi = clamp(s_rsi + (rsi - rsiPast) / 100 * 0.08, 0, 1);
   303→    f_mfi = clamp(s_mfi + (mfi - mfiPast) / 100 * 0.08, 0, 1);
   304→    f_cci = clamp(s_cci + (cci - cciPast) / 300 * 0.06, 0, 1);
   305→    f_macd = clamp(s_macd + (macd.histogram - macdPast.histogram) / (Math.abs(macd.histogram) + Math.abs(macdPast.histogram) + 1) * 3 * 0.08, 0, 1);
   306→    f_stoch = clamp(s_stoch + (stoch.k - stochPast.k) / 50 * 0.06, 0, 1);
   307→  }
   308→
   309→  // ── LAYER 2: Cross signals (binary per spec) ──
   310→  let f_stochCross = 0.1;
   311→  let f_macdCross = 0.1;
   312→  let f_div = 0.15;
   313→
   314→  if (slice.length > 20) {
   315→    const prevSlice = slice.slice(0, -1);
   316→    const prevStoch = calcStochastic(prevSlice);
   317→    const prevMacd = calcMACD(prevSlice.map(d => d.close));
   318→
   319→    // Stochastic bullish cross: K_prev < D_prev AND K_curr > D_curr
   320→    f_stochCross = (prevStoch.k < prevStoch.d && stoch.k > stoch.d) ? 0.9 : 0.1;
   321→
   322→    // MACD bullish cross: line_prev < signal_prev AND line_curr > signal_curr
   323→    f_macdCross = (prevMacd.line < prevMacd.signal && macd.line > macd.signal) ? 0.9 : 0.1;
   324→
   325→    // Divergence detection
   326→    if (closes.length > 40) {
   327→      const p20 = closes[closes.length - 21];
   328→      const rsi20 = calcRSI(closes.slice(0, -20));
   329→      const pChange = price - p20;
   330→      const rsiChange = rsi - rsi20;
   331→      if (pChange !== 0 && rsiChange !== 0) {
   332→        f_div = (pChange > 0 && rsiChange < 0) ? 0.85 : (pChange < 0 && rsiChange > 0) ? 0.85 : 0.15;
   333→      }
   334→    }
   335→  }
   336→
   337→  // ── Return 16 features in exact order ──
   338→  return [
   339→    f_rsi, f_mfi, f_cci, s_adx,
   340→    f_macd, f_stoch, s_bb, s_ma21, s_ma100, s_ema,
   341→    s_atr, s_trend, s_sr,
   342→    f_stochCross, f_macdCross, f_div,
   343→  ];
   344→}

   346→// ─── Model Cache ────────────────────────────────────────────────────────────
   347→
   348→/** Cached model entry with timestamp for TTL-based invalidation. */
   349→interface CachedModel {
   350→  result: AdaptiveModelResult;
   351→  time: number;
   352→}
   353→
   354→const modelCache = new Map<string, CachedModel>();
   355→const CACHE_TTL = 60 * 60 * 1000; // 1 hour
   356→
   357→/**
   358→ * Retrieve a cached adaptive model for the given symbol if it has not expired.
   359→ * Cache TTL is 1 hour.
   360→ *
   361→ * @param symbol - Ticker/symbol identifier
   362→ * @returns The cached {@link AdaptiveModelResult}, or `null` if absent/expired
   363→ */
   364→export function getCachedAdaptiveModel(symbol: string): AdaptiveModelResult | null {
   365→  const cached = modelCache.get(symbol);
   366→  if (cached && Date.now() - cached.time < CACHE_TTL) return cached.result;
   367→  return null;
   368→}
   369→
   370→/**
   371→ * Store an adaptive model result in the cache with the current timestamp.
   372→ *
   373→ * @param symbol - Ticker/symbol identifier
   374→ * @param result - The model result to cache
   375→ */
   376→export function setCachedAdaptiveModel(symbol: string, result: AdaptiveModelResult): void {
   377→  modelCache.set(symbol, { result, time: Date.now() });
   378→}

   380→// ─── Main Training Function ─────────────────────────────────────────────────
   381→
   382→/**
   383→ * Train the Adaptive Weight Model for a given symbol using logistic regression
   384→ * with **TimeSeriesSplit** cross-validation.
   385→ *
   386→ * ### Training Pipeline
   387→ * 1. **Feature extraction** — For each candle from index `START_IDX` (40) to
   388→ *    `data.length − FORWARD_DAYS`, computes the 16 VDSS features via
   389→ *    {@link extractVDSSFeatures}.
   390→ * 2. **Binary labelling** — Labels each sample as `1` (bullish) if the
   391→ *    5-day forward return exceeds +1%, otherwise `0` (bearish).
   392→ * 3. **Model training** — Fits an {@link AdaptiveWeightModel} (L2-regularised
   393→ *    logistic regression) with TimeSeriesSplit validation on the feature/label
   394→ *    matrices.
   395→ * 4. **Prediction** — Scores the most recent candle's features and caches the
   396→ *    result (TTL = 1 hour).
   397→ *
   398→ * Returns `null` if there are insufficient data points, non-finite features,
   399→ * or if the underlying model fails to converge.
   400→ *
   401→ * @param data       - Full OHLCV history (chronological, oldest first)
   402→ * @param symbol     - Ticker/symbol identifier used as the model cache key
   403→ * @param minSamples - Minimum number of valid training samples required (default 70)
   404→ * @returns The trained {@link AdaptiveModelResult} on success, or `null` on failure
   405→ */
   406→export function trainAdaptiveModel(data: OHLCV[], symbol: string, minSamples = 70): AdaptiveModelResult | null {
   407→  const FORWARD_DAYS = 5;
   408→  const START_IDX = 40; // Need enough history for all indicators to stabilize
   409→
   410→  if (data.length < START_IDX + FORWARD_DAYS + minSamples) return null;
   411→
   412→  const hasVolume = data.some(d => d.volume > 0);
   413→
   414→  const X: number[][] = [];
   415→  const y: number[] = [];
   416→
   417→  for (let i = START_IDX; i < data.length - FORWARD_DAYS; i++) {
   418→    const features = extractVDSSFeatures(data, i, hasVolume);
   419→    const currentPrice = data[i].close;
   420→    const futurePrice = data[i + FORWARD_DAYS].close;
   421→
   422→    if (currentPrice <= 0) continue;
   423→    if (features.some(f => !isFinite(f))) continue;
   424→
   425→    // Label: 1 if 5-day forward return > 1%, else 0
   426→    const forwardReturn = (futurePrice - currentPrice) / currentPrice;
   427→    const label = forwardReturn > 0.01 ? 1 : 0;
   428→
   429→    X.push(features);
   430→    y.push(label);
   431→  }
   432→
   433→  if (X.length < minSamples) return null;
   434→
   435→  const model = new AdaptiveWeightModel(minSamples);
   436→  const success = model.train(X, y);
   437→
   438→  if (!success) return null;
   439→
   440→  // Predict current features
   441→  const currentFeatures = extractVDSSFeatures(data, data.length - 1, hasVolume);
   442→  model.predictScore(currentFeatures);
   443→
   444→  const result = model.getResult();
   445→  setCachedAdaptiveModel(symbol, result);
   446→
   447→  return result;
   448→}

   450→// ─── Calculate bullConsensus with ML Weights (Layer 3) ──────────────────────
   451→
   452→/**
   453→ * Calculate the ML-weighted bull consensus from the 16 VDSS features.
   454→ *
   455→ * ### ML Path (when `mlResult` is trained)
   456→ * Computes a weighted average: `Σ(w_j × f_j) / Σ(w_j)`, skipping the MFI
   457→ * feature (index 1) when `hasVolume` is false. The raw consensus is then
   458→ * blended with the ML direct prediction probability, with the blend weight
   459→ * proportional to `recentAccuracy × 0.30` (capped at 0.30).
   460→ *
   461→ * ### Fallback Heuristic (when ML is unavailable)
   462→ * Uses fixed default weights and the same weighted-average formula:
   463→ * ```
   464→ * DEFAULT_WEIGHTS = [
   465→ *   0.15, 0.12, 0.10, 0.08,  // rsi, mfi, cci, adx
   466→ *   0.14, 0.10, 0.08,         // macd, stoch, bb
   467→ *   0.10, 0.08, 0.10,         // ma21, ma100, ema
   468→ *   0.05, 0.10, 0.05,         // atr, trend, sr
   469→ *   0.06, 0.06, 0.08,         // stochCross, macdCross, div
   470→ * ]
   471→ * ```
   472→ *
   473→ * The fallback heuristic can also be expressed as a sigmoid of a raw bull
   474→ * score derived from individual indicator contributions:
   475→ * ```
   476→ * bullScore = (rsi<40 ? 0.3 : rsi>60 ? -0.1 : 0)
   477→ *           + (adx>25 && diPlus>diMinus ? 0.3 : 0)
   478→ *           + (macdHist>0 ? 0.2 : -0.2)
   479→ *           + (bbPos<0.3 ? 0.2 : bbPos>0.7 ? -0.2 : 0)
   480→ * bullConsensus = sigmoid(bullScore)
   481→ * ```
   482→ *
   483→ * @param features  - Array of 16 VDSS feature values for the current candle
   484→ * @param mlResult  - Trained adaptive model result, or `null` to use fallback weights
   485→ * @param hasVolume - Whether volume data is valid (disables MFI feature at index 1)
   486→ * @returns Object with `bullConsensus` clamped to [0,1] and `usedML` flag
   487→ */
   488→export function calculateBullConsensus(
   489→  features: number[],
   490→  mlResult: AdaptiveModelResult | null,
   491→  hasVolume: boolean,
   492→): { bullConsensus: number; usedML: boolean } {
   493→  if (mlResult && mlResult.isTrained && mlResult.weights.length === 16) {
   494→    const weights = mlResult.weights;
   495→    let bullConsensus = 0;
   496→    let totalWeight = 0;
   497→
   498→    for (let j = 0; j < 16; j++) {
   499→      // Skip MFI (index 1) if no volume
   500→      if (j === 1 && !hasVolume) continue;
   501→      bullConsensus += weights[j] * features[j];
   502→      totalWeight += weights[j];
   503→    }
   504→
   505→    if (totalWeight > 0) bullConsensus /= totalWeight;
   506→
   507→    // Blend with ML direct prediction (dynamic weight based on accuracy)
   508→    if (mlResult.predictionProb !== null) {
   509→      const mlWeight = Math.max(0, Math.min(mlResult.recentAccuracy * 0.30, 0.30));
   510→      bullConsensus = bullConsensus * (1 - mlWeight) + mlResult.predictionProb * mlWeight;
   511→    }
   512→
   513→    return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: true };
   514→  }
   515→
   516→  // Fallback: default weights (per spec, only when ML is unavailable)
   517→  const DEFAULT_WEIGHTS = [
   518→    0.15, 0.12, 0.10, 0.08,  // rsi, mfi, cci, adx
   519→    0.14, 0.10, 0.08,         // macd, stoch, bb
   520→    0.10, 0.08, 0.10,         // ma21, ma100, ema
   521→    0.05, 0.10, 0.05,         // atr, trend, sr
   522→    0.06, 0.06, 0.08,         // stochCross, macdCross, div
   523→  ];
   524→
   525→  let bullConsensus = 0;
   526→  let totalWeight = 0;
   527→  for (let j = 0; j < 16; j++) {
   528→    if (j === 1 && !hasVolume) continue;
   529→    bullConsensus += DEFAULT_WEIGHTS[j] * features[j];
   530→    totalWeight += DEFAULT_WEIGHTS[j];
   531→  }
   532→  if (totalWeight > 0) bullConsensus /= totalWeight;
   533→
   534→  return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: false };
   535→}

   599→// ─── Decision Graph Scenario Probabilities (3-Branch DAG) ──────────────────

   602→//   Each branch → 9 edges → 9 leaf scenarios

   604→//   Where P(Branch) is derived from EMV maximization,
   605→//   and P(Edge) is the conditional probability per edge formula.
   606→// ═══════════════════════════════════════════════════════════════════════════
   607→
   608→/**

   641→/**
   642→ * Calculate 9-scenario probabilities using the **3-branch decision DAG**.
   643→ *
   644→ * The DAG structure (per TechnicalAnalysisDssGraph.txt):
   645→ * - **Root** → 3 branches: Trend Following (A), Breakout (B), Reversal (C)
   646→ * - Each branch emits 9 conditional probabilities → 9 leaf scenarios
   647→ * - `P(Scenario_S) = Σ_branch P(Branch) × P(Edge_S | Branch)`
   648→ *
   649→ * ### Computation Steps
   650→ * 1. **P_Bull_Current / P_Bear_Current** — Bayesian-weighted aggregation of
   651→ *    current indicator signals (RSI, MFI, StochK, MA100 position, ADX, S/R
   652→ *    proximity) using category priors (trend=0.28, oscillator=0.25,
   653→ *    volume=0.15, volatility=0.10, leading=0.12, patterns=0.10).
   654→ * 2. **P_Bull_Forecast** — Uses ML `predictionProb` if available; otherwise
   655→ *    applies exponential decay toward 0.5 (mean-reversion).
   656→ * 3. **Final directional probs** — Combines current and forecast:
   657→ *    ```
   658→ *    P_Bull_Final  = α × P_Bull_Current  + (1-α) × P_Bull_Forecast
   659→ *    P_Bear_Final  = α × P_Bear_Current  + (1-α) × P_Bear_Forecast
   660→ *    P_Neutral_Final = max(0.04, 1 − P_Bull − P_Bear)
   661→ *    ```
   662→ * 4. **Conditional distributions** — CP(S_i | Bull/Bear/Neutral) computed
   663→ *    from ADX trend strength, overbought/oversold, volatility, divergence,
   664→ *    and ML-adaptive factors. Normalised to sum to 1 per state.
   665→ * 5. **Final probabilities** — Weighted sum across states, normalised to 100.
   666→ *
   667→ * ### The 9 Scenarios
   668→ * | i | Name              | Description                            |
   669→ * |---|-------------------|----------------------------------------|
   670→ * | 1 | Strong Bull       | ADX > 40, strong uptrend               |
   671→ * | 2 | Accelerating Bull | High momentum + volatility expansion   |
   672→ * | 3 | Cautious Bull     | Bullish but overbought risk            |
   673→ * | 4 | Strong Bear       | ADX > 40, strong downtrend             |
   674→ * | 5 | Accelerating Bear | High momentum + volatility expansion   |
   675→ * | 6 | Cautious Bear     | Bearish but oversold risk              |
   676→ * | 7 | Low-Vol Range     | ATR low, ADX < 20 → sideways           |
   677→ * | 8 | High-Vol Range    | ATR high, ADX < 30 → choppy            |
   678→ * | 9 | Shock             | Black-swan tail risk (2-5%)            |
   679→ *
   680→ * @param bullConsensus - Overall bull consensus ∈ [0,1]
   681→ * @param price        - Current close price
   682→ * @param SC1          - R1 resistance level
   683→ * @param S1           - S1 support level
   684→ * @param MA100        - 100-period simple moving average
   685→ * @param rsi          - Current RSI value (0-100)
   686→ * @param mfi          - Current MFI value (0-100)
   687→ * @param stochK       - Current Stochastic %K value
   688→ * @param hasVolume    - Whether volume data is valid
   689→ * @param mlResult     - Trained adaptive model result (or null)
   690→ * @param adx          - Current ADX value (default 25)
   691→ * @param atr          - Current ATR value (default 0)
   692→ * @returns Object with `pSC1`–`pSC9` (integers summing to 100) and adaptive `factors`
   693→ */
   694→export function calculateScenarioProbabilities(
   695→  bullConsensus: number,
   696→  price: number,
   697→  SC1: number,
   698→  S1: number,
   699→  MA100: number,
   700→  rsi: number,
   701→  mfi: number,
   702→  stochK: number,
   703→  hasVolume: boolean,
   704→  mlResult: AdaptiveModelResult | null,
   705→  adx: number = 25,
   706→  atr: number = 0,
   707→): { pSC1: number; pSC2: number; pSC3: number; pSC4: number; pSC5: number; pSC6: number; pSC7: number; pSC8: number; pSC9: number; factors: { momentum: number; volatility: number; trend: number } } {
   708→  // ── Adaptive ML factors ──
   709→  const mlParams = mlResult?.adaptiveParams;
   710→  const factors = {
   711→    momentum: mlParams?.momentumFactor ?? 0.7,
   712→    volatility: mlParams?.volatilityFactor ?? 0.5,
   713→    trend: mlParams?.trendFactor ?? 0.6,
   714→  };
   715→
   716→  const bull = bullConsensus;
   717→  const bear = 1 - bullConsensus;
   718→
   719→  // ═══════════════════════════════════════════════════════════════════════
   720→  // STEP 1: Compute P_Bull_Current, P_Bear_Current from current indicators
   721→  // ═══════════════════════════════════════════════════════════════════════
   722→  // Use Bayesian weighting: W_i = P(Success|Tool_i) × Prior_i / Σ
   723→  // For current signals, each indicator provides a directional score.
   724→
   725→  // RSI signal (0=bearish, 1=bullish)
   726→  const rsiBull = rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   727→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   728→
   729→  // MFI signal (volume-dependent)
   730→  const mfiBull = hasVolume
   731→    ? (mfi > 80 ? 0.85 : mfi > 60 ? 0.65 : mfi > 40 ? 0.45 : mfi > 20 ? 0.3 : 0.15)
   732→    : 0.5; // neutral if no volume
   733→
   734→  // Stochastic signal
   735→  const stochBull = stochK > 80 ? 0.85 : stochK > 60 ? 0.65 : stochK > 40 ? 0.45
   736→    : stochK > 20 ? 0.3 : 0.15;
   737→
   738→  // ADX trend strength normalized (0-1)
   739→  const adxNorm = Math.min(adx / 45, 1);
   740→
   741→  // Price vs MA100 position
   742→  const maPosBull = MA100 > 0 ? clamp((price - MA100) / (MA100 * 0.05), 0, 1) : 0.5;
   743→
   744→  // Distance to S/R (proximity = more likely to react)
   745→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0.5;
   746→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0.5;
   747→
   748→  // Priors per category (from instructions): trend=0.28, oscillator=0.25, volume=0.15, volatility=0.10, leading=0.12, patterns=0.10
   749→  // Adjusted: if no volume, redistribute 0.15 proportionally
   750→  let priorTrend = 0.28, priorOsc = 0.25, priorVol = hasVolume ? 0.15 : 0;
   751→  let priorVolatility = 0.10, priorLeading = 0.12, priorPattern = 0.10;
   752→  if (!hasVolume) {
   753→    const totalRemaining = priorTrend + priorOsc + priorVolatility + priorLeading + priorPattern;
   754→    const redistribute = 0.15;
   755→    priorTrend += redistribute * (priorTrend / totalRemaining);
   756→    priorOsc += redistribute * (priorOsc / totalRemaining);
   757→    priorVolatility += redistribute * (priorVolatility / totalRemaining);
   758→    priorLeading += redistribute * (priorLeading / totalRemaining);
   759→    priorPattern += redistribute * (priorPattern / totalRemaining);
   760→  }
   761→
   762→  // P_Bull_Current = Σ(W_i × Signal_Bull_i) / Σ(W_i)
   763→  const currentBullSignals = [
   764→    { signal: rsiBull, weight: priorOsc * 0.4 },           // RSI (oscillator)
   765→    { signal: mfiBull, weight: priorVol },                 // MFI (volume)
   766→    { signal: stochBull, weight: priorOsc * 0.35 },        // Stochastic (oscillator)
   767→    { signal: maPosBull, weight: priorTrend * 0.5 },       // MA position (trend)
   768→    { signal: adxNorm > 0.5 && bull > 0.5 ? bull : 0.5, weight: priorTrend * 0.3 }, // ADX trend (trend)
   769→    { signal: distS1, weight: priorPattern * 0.5 },        // Near support = bullish bounce
   770→  ];
   771→
   772→  let sumWeight = 0;
   773→  let sumWeightedBull = 0;
   774→  for (const { signal, weight } of currentBullSignals) {
   775→    if (weight <= 0) continue;
   776→    sumWeightedBull += weight * signal;
   777→    sumWeight += weight;
   778→  }
   779→  const P_Bull_Current = sumWeight > 0 ? clamp(sumWeightedBull / sumWeight, 0, 1) : bull;
   780→  const P_Bear_Current = 1 - P_Bull_Current;
   781→
   782→  // ═══════════════════════════════════════════════════════════════════════
   783→  // STEP 2: P_Bull_Forecast (from ML prediction, or decayed current)
   784→  // ═══════════════════════════════════════════════════════════════════════
   785→  // If ML prediction available, use predicted return to estimate future bull prob
   786→  // Otherwise use exponential decay of current state
   787→  const alpha = 0.5; // weight of current state (per instructions)
   788→
   789→  // Use ML predicted return if available, otherwise use consensus as proxy
   790→  const mlReturn = mlResult?.predictionProb;
   791→  let P_Bull_Forecast: number;
   792→  if (mlReturn !== null && mlReturn !== undefined) {
   793→    // ML gives direct P(bull) prediction — use it as forecast
   794→    P_Bull_Forecast = clamp(mlReturn, 0, 1);
   795→  } else {
   796→    // Decay: forecast = current × decay + 0.5 × (1-decay)  (mean-revert to 0.5)
   797→    const decayFactor = 0.85;
   798→    P_Bull_Forecast = P_Bull_Current * decayFactor + 0.5 * (1 - decayFactor);
   799→  }
   800→  const P_Bear_Forecast = 1 - P_Bull_Forecast;
   801→
   802→  // ═══════════════════════════════════════════════════════════════════════
   803→  // STEP 3: Final directional probabilities (combining current + forecast)
   804→  // ═══════════════════════════════════════════════════════════════════════
   805→  const P_Bull_Final = clamp(alpha * P_Bull_Current + (1 - alpha) * P_Bull_Forecast, 0.02, 0.98);
   806→  const P_Bear_Final = clamp(alpha * P_Bear_Current + (1 - alpha) * P_Bear_Forecast, 0.02, 0.98);
   807→  const P_Neutral_Final = Math.max(0.04, 1 - P_Bull_Final - P_Bear_Final);
   808→
   809→  // Normalize so they sum to 1
   810→  const dirSum = P_Bull_Final + P_Bear_Final + P_Neutral_Final;
   811→  const pBull = P_Bull_Final / dirSum;
   812→  const pBear = P_Bear_Final / dirSum;
   813→  const pNeutral = P_Neutral_Final / dirSum;
   814→
   815→  // ═══════════════════════════════════════════════════════════════════════
   816→
   817→  // Helper: Overbought/Oversold
   818→  const overbought = rsi > 70 ? (rsi - 70) / 30 : 0;
   819→  const oversold = rsi < 30 ? (30 - rsi) / 30 : 0;
   820→
   821→  // Helper: Volatility state
   822→  const atrPct = price > 0 ? (atr || price * 0.02) / price : 0.02;
   823→  const volHigh = clamp((atrPct - 0.02) / 0.03, 0, 1);
   824→  const volLow = clamp(1 - (atrPct - 0.01) / 0.02, 0, 1);
   825→
   826→  // Helper: Divergence detection
   827→  let divBull = 0;
   828→  let divBear = 0;
   829→  if (mlResult?.predictionProb !== undefined) {
   830→    const mlBull = mlResult.predictionProb ?? 0.5;
   831→    if (bull > 0.6 && mlBull < 0.4) divBear = (bull - mlBull) * 1.5;
   832→    if (bull < 0.4 && mlBull > 0.6) divBull = (mlBull - bull) * 1.5;
   833→  }
   834→  divBull = clamp(divBull, 0, 0.3);
   835→  divBear = clamp(divBear, 0, 0.3);
   836→
   837→  // Helper: Volume confirmation
   838→  const volConfirm = hasVolume ? 0.6 : 0.4;
   839→
   840→  // Helper: ADX > 40 = strong trend
   841→  const adxStrong = adx > 40 ? 1 : adx > 25 ? (adx - 25) / 15 : 0;
   842→
   843→  // ── Conditional probability distributions CP(S_i | State) ──
   844→  // Per attachment instructions: 3-branch decision graph
   845→  //   Branch A: Trend Following → scenarios A1-A9
   846→  //   Branch B: Breakout Strategy → scenarios B1-B9
   847→  //   Branch C: Reversal Strategy → scenarios C1-C9
   848→  // Final: P(S_i) = P(Bull)×CP_i|Bull + P(Bear)×CP_i|Bear + P(Neutral)×CP_i|Neutral
   849→  // Each CP distribution sums to 1 after normalization.
   850→  //
   851→  // ADX governs range/trend split (per attachment: ADX < 20 = weak trend)
   852→  // FIX: Range scenarios capped so they never dominate (>40% of any CP dist)
   853→
   854→  // ADX-based trend strength (per attachment: ADX_Normalized = min(ADX/60, 1))
   855→  const adxNormVal = Math.min(adx / 60, 1);
   856→  // Range weight: inversely proportional to ADX, but capped at 0.35 (was 0.6!)
   857→  const adxRange = adx < 20 ? 0.35 : adx < 30 ? 0.20 : 0.10;
   858→  const adxTrend = 1 - adxRange;
   859→
   860→  // ── Per attachment edge formulas (lines 657-716): ──
   861→  // A1: Strong Bull  = P_Bull × P_ADX>40 × P_Forecast_Bull_Strong
   862→  // A2: Accel Bull   = P_Bull × P_Volatility_High × P_Volume_High_Future
   863→  // A3: Cautious Bull= P_Bull × P_Divergence_Bearish_Future
   864→  // A4: Strong Bear  = P_Bear × P_ADX>40 × P_Forecast_Bear_Strong
   865→  // A5: Accel Bear   = P_Bear × P_Volatility_High × P_Volume_High_Future
   866→  // A6: Cautious Bear= P_Bear × P_Divergence_Bullish_Future
   867→  // A7: LowVol Range = P_Neutral × P_Volatility_Low_Future
   868→  // A8: HighVol Range= P_Neutral × P_Volatility_High_Future
   869→  // A9: Shock        = P_Shock × P_Forecast_Anomaly (0.02-0.05)
   870→
   871→  // CP_i|Bull: Bull scenarios dominate, range scenarios limited
   872→  const cpBull = [
   873→    // SC1: Strong Bull — ADX strength and trend quality
   874→    0.28 * adxTrend * (0.5 + 0.5 * adxNormVal),
   875→    // SC2: Accelerating Bull — momentum + volatility
   876→    0.15 * adxTrend * (0.3 + 0.7 * factors.momentum),
   877→    // SC3: Cautious Bull — reduced by overbought risk
   878→    0.12 * (1 - overbought * 0.6) * adxTrend,
   879→    // SC4: Strong Bear — small probability in bull market, driven by divergence
   880→    0.03 * (1 + divBear),
   881→    // SC5: Accelerating Bear — very small in bull market
   882→    0.02,
   883→    // SC6: Cautious Bear — driven by divergence warning
   884→    0.03 * (1 + divBear),
   885→    // SC7: Low-Vol Range — bounded, not dominant
   886→    adxRange * 0.45 * volLow,
   887→    // SC8: High-Vol Range — bounded
   888→    adxRange * 0.30 * volHigh,
   889→    // SC9: Shock — fixed small probability
   890→    0.04 * (1 + overbought * 0.5 + volHigh * 0.5),
   891→  ];
   892→
   893→  // CP_i|Bear: Mirror of bull (bear scenarios dominate)
   894→  const cpBear = [
   895→    // SC1: Strong Bull — small in bear market
   896→    0.03 * (1 + divBull),
   897→    // SC2: Accelerating Bull — very small
   898→    0.02,
   899→    // SC3: Cautious Bull — divergence-driven
   900→    0.03 * (1 + divBull),
   901→    // SC4: Strong Bear — ADX strength
   902→    0.28 * adxTrend * (0.5 + 0.5 * adxNormVal),
   903→    // SC5: Accelerating Bear — momentum
   904→    0.15 * adxTrend * (0.3 + 0.7 * factors.momentum),
   905→    // SC6: Cautious Bear — reduced by oversold risk
   906→    0.12 * (1 - oversold * 0.6) * adxTrend,
   907→    // SC7: Low-Vol Range — bounded
   908→    adxRange * 0.45 * volLow,
   909→    // SC8: High-Vol Range — bounded
   910→    adxRange * 0.30 * volHigh,
   911→    // SC9: Shock
   912→    0.04 * (1 + oversold * 0.5 + volHigh * 0.5),
   913→  ];
   914→
   915→  // CP_i|Neutral: More balanced distribution per attachment
   916→  // Per attachment: neutral market has trend signals too, just weaker
   917→  // Range gets adxRange but trend scenarios get reasonable weight
   918→  const cpNeutral = [
   919→    // SC1: Mild Bull — smaller in neutral
   920→    0.08 * adxTrend * (0.5 + 0.5 * adxNormVal),
   921→    // SC2: Bull Acceleration
   922→    0.04 * adxTrend * (0.3 + 0.7 * factors.momentum),
   923→    // SC3: Cautious Bull
   924→    0.07 * adxTrend,
   925→    // SC4: Mild Bear
   926→    0.08 * adxTrend * (0.5 + 0.5 * adxNormVal),
   927→    // SC5: Bear Acceleration
   928→    0.04 * adxTrend * (0.3 + 0.7 * factors.momentum),
   929→    // SC6: Cautious Bear
   930→    0.07 * adxTrend,
   931→    // SC7: Low-Vol Range — adxRange weighted, split between vol states
   932→    adxRange * 0.50 * volLow,
   933→    // SC8: High-Vol Range
   934→    adxRange * 0.35 * volHigh,
   935→    // SC9: Shock — small fixed
   936→    0.04,
   937→  ];
   938→
   939→  const normCP = (cp: number[]): number[] => {
   940→    const s = cp.reduce((a, b) => a + b, 0);
   941→    return s > 0 ? cp.map(v => v / s) : cp.map(() => 1 / 9);
   942→  };
   943→  const ncpBull = normCP(cpBull);
   944→  const ncpBear = normCP(cpBear);
   945→  const ncpNeutral = normCP(cpNeutral);
   946→
   947→  // ── Final probabilities per instructions formula ──
   948→  // P(S_i) = P(Bull)*CP_i|Bull + P(Bear)*CP_i|Bear + P(Neutral)*CP_i|Neutral
   949→  const rawS = [0, 0, 0, 0, 0, 0, 0, 0, 0];
   950→  for (let i = 0; i < 9; i++) {
   951→    rawS[i] = pBull * ncpBull[i] + pBear * ncpBear[i] + pNeutral * ncpNeutral[i];
   952→  }
   953→
   954→  // Normalize to sum = 100
   955→  const totalRaw = rawS.reduce((a, b) => a + b, 0);
   956→  if (totalRaw <= 0) {
   957→    return {
   958→      pSC1: 12, pSC2: 10, pSC3: 8, pSC4: 12, pSC5: 10, pSC6: 8,
   959→      pSC7: 15, pSC8: 15, pSC9: 10, factors,
   960→    };
   961→  }
   962→
   963→  const pSC1 = Math.round(rawS[0] / totalRaw * 100);
   964→  const pSC2 = Math.round(rawS[1] / totalRaw * 100);
   965→  const pSC3 = Math.round(rawS[2] / totalRaw * 100);
   966→  const pSC4 = Math.round(rawS[3] / totalRaw * 100);
   967→  const pSC5 = Math.round(rawS[4] / totalRaw * 100);
   968→  const pSC6 = Math.round(rawS[5] / totalRaw * 100);
   969→  const pSC7 = Math.round(rawS[6] / totalRaw * 100);
   970→  const pSC8 = Math.round(rawS[7] / totalRaw * 100);
   971→  const pSC9 = Math.max(1, 100 - (pSC1 + pSC2 + pSC3 + pSC4 + pSC5 + pSC6 + pSC7 + pSC8));
   972→  return { pSC1, pSC2, pSC3, pSC4, pSC5, pSC6, pSC7, pSC8, pSC9, factors };
   973→}

   975→/**
   976→ * Compute edge weights for the DAG (Directed Acyclic Graph) used in the
   977→ * decision-graph scoring model.
   978→ *
   979→ * The four edge weights represent the transition probabilities and risk
   980→ * parameters that govern how the DAG propagates scenario outcomes:
   981→ * - **up**        — Probability-weighted upside edge, driven by bull consensus
   982→ *                    and ML momentum coefficient.
   983→ * - **down**      — Probability-weighted downside edge, driven by bear consensus
   984→ *                    and ML trend coefficient.
   985→ * - **pullback**  — Mean-reversion pullback magnitude, modulated by the ML
   986→ *                    volatility coefficient.
   987→ * - **risk**      — Position-sizing risk parameter; inversely scaled by ADX
   988→ *                    strength and adjusted by the ML trend coefficient.
   989→ *
   990→ * When `mlResult` is trained, the coefficients for features at indices
   991→ * `[11]=Trend, [0]=RSI, [10]=ATR_pct` are used to derive ML-driven
   992→ * `{trendCoef, momentumCoef, volatilityCoef}` (normalised to sum to 1).
   993→ * Otherwise, all three coefficients default to 1/3 (equal weighting).
   994→ *
   995→ * @param bullConsensus - Overall bull consensus ∈ [0,1]
   996→ * @param adx          - Current ADX value (higher = stronger trend)
   997→ * @param mlResult     - Trained adaptive model result, or `null` for equal weights
   998→ * @returns Object with `{ up, down, pullback, risk }` — each clamped to a safe range
   999→ */
  1000→export function calculateEdgeWeights(
  1001→  bullConsensus: number,
  1002→  adx: number,
  1003→  mlResult: AdaptiveModelResult | null,
  1004→): { up: number; down: number; pullback: number; risk: number } {
  1005→  // Default coefficients
  1006→  let trendCoef = 1 / 3;
  1007→  let momentumCoef = 1 / 3;
  1008→  let volatilityCoef = 1 / 3;
  1009→
  1010→  if (mlResult?.isTrained && mlResult.coefficients.length === 16) {
  1011→    const ec = mlResult.coefficients;
  1012→    const absTrend = Math.abs(ec[11] ?? 0);
  1013→    const absMomentum = Math.abs(ec[0] ?? 0);
  1014→    const absVolatility = Math.abs(ec[10] ?? 0);
  1015→    const total = absTrend + absMomentum + absVolatility;
  1016→    if (total > 0) {
  1017→      trendCoef = absTrend / total;
  1018→      momentumCoef = absMomentum / total;
  1019→      volatilityCoef = absVolatility / total;
  1020→    }
  1021→  }
  1022→
  1023→  // Dynamic base calculations
  1024→  const up_base = bullConsensus * 0.7 + 0.15;
  1025→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
  1026→  const pullback_base = 0.25;
  1027→  const risk_base = 0.12 * (1.2 - Math.min(adx / 100, 1));
  1028→
  1029→  // Apply ML-driven adjustments
  1030→  const up = clamp(up_base * (1 + (momentumCoef - 1 / 3) * 0.3), 0.05, 0.95);
  1031→  const down = clamp(down_base * (1 + (trendCoef - 1 / 3) * 0.3), 0.05, 0.95);
  1032→  const pullback = clamp(pullback_base * (1 + (volatilityCoef - 1 / 3) * 0.2), 0.05, 0.50);
  1033→  const risk = clamp(risk_base * (1 - (trendCoef - 1 / 3) * 0.5), 0.02, 0.30);
  1034→
  1035→  return { up, down, pullback, risk };
  1036→}

  1128→/**
  1129→ * Backward-compatible price prediction using a simple linear-regression-based
  1130→ * trend + mean-reversion + momentum model.
  1131→ *
  1132→ * ### Prediction Method
  1133→ * 1. Computes 20-bar daily returns and derives short (3-bar) and medium
  1134→ *    (10-bar) momentum averages.
  1135→ * 2. Fits a linear regression on the last 20 closes to extract slope and R².
  1136→ * 3. Calculates ATR-based daily volatility.
  1137→ * 4. Combines three daily-return components with adaptive weights:
  1138→ *    - **Trend** (weight = R² × 0.6): regression slope / price
  1139→ *    - **Mean-reversion** (weight = min(|deviation|×5, 0.4)): −deviation × 0.05
  1140→ *    - **Momentum** (weight = 0.2): 0.4 × mom3 + 0.1 × mom10
  1141→ * 5. Projects 5 daily prices with exponential decay (λ = 0.1).
  1142→ *
  1143→ * @param data - Full OHLCV history (chronological, oldest first). Needs ≥ 40 bars.
  1144→ * @returns {@link PricePrediction} with 5 predicted prices, 5-day return,
  1145→ *          confidence ∈ [0.1, 0.9], and trend direction, or `null` if
  1146→ *          insufficient data or zero price.
  1147→ */
  1148→export function predictPrices(data: OHLCV[]): PricePrediction | null {
  1149→  if (data.length < 40) return null;
  1150→  const PREDICT_DAYS = 5;
  1151→  const closes = data.map(d => d.close);
  1152→  const price = closes[closes.length - 1];
  1153→  if (price <= 0) return null;
  1154→
  1155→  const LOOKBACK = 20;
  1156→  const returns: number[] = [];
  1157→  for (let i = closes.length - LOOKBACK; i < closes.length; i++) {
  1158→    returns.push(i > 0 && closes[i - 1] > 0 ? (closes[i] - closes[i - 1]) / closes[i - 1] : 0);
  1159→  }
  1160→
  1161→  const mom3 = returns.slice(-3).reduce((a, b) => a + b, 0) / 3;
  1162→  const mom10 = returns.slice(-10).reduce((a, b) => a + b, 0) / 10;
  1163→  const ma20 = sma(closes, 20);
  1164→  const deviation = price > 0 && ma20 > 0 ? (price - ma20) / ma20 : 0;
  1165→
  1166→  const trendSlice = closes.slice(-20);
  1167→  const tn = trendSlice.length;
  1168→  let sx = 0, sy = 0, sxy = 0, sx2 = 0;
  1169→  for (let i = 0; i < tn; i++) { sx += i; sy += trendSlice[i]; sxy += i * trendSlice[i]; sx2 += i * i; }
  1170→  const d = tn * sx2 - sx * sx;
  1171→  let slope = 0, r2 = 0;
  1172→  if (d !== 0) {
  1173→    slope = (tn * sxy - sx * sy) / d;
  1174→    const meanY = sy / tn;
  1175→    let ssRes = 0, ssTot = 0;
  1176→    for (let i = 0; i < tn; i++) { ssRes += (trendSlice[i] - (slope * i + meanY - slope * (tn - 1) / 2)) ** 2; ssTot += (trendSlice[i] - meanY) ** 2; }
  1177→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
  1178→  }
  1179→
  1180→  let atr = 0;
  1181→  for (let i = Math.max(1, data.length - 14); i < data.length; i++) {
  1182→    atr += Math.max(data[i].high - data[i].low, Math.abs(data[i].high - data[i - 1].close), Math.abs(data[i].low - data[i - 1].close));
  1183→  }
  1184→  atr /= Math.min(14, data.length - 1);
  1185→  const dailyVol = price > 0 ? atr / price : 0.02;
  1186→
  1187→  const trendDaily = price > 0 ? slope / price : 0;
  1188→  const meanRevDaily = -deviation * 0.05;
  1189→  const momentumDaily = mom3 * 0.4 + mom10 * 0.1;
  1190→  const trendW = r2 * 0.6, mrW = Math.min(Math.abs(deviation) * 5, 0.4), momW = 0.2;
  1191→  const predDaily = trendDaily * trendW + meanRevDaily * mrW + momentumDaily * momW;
  1192→
  1193→  const predictedPrices: number[] = [];
  1194→  let cp = price;
  1195→  for (let dd = 1; dd <= PREDICT_DAYS; dd++) {
  1196→    const decay = Math.exp(-0.1 * (dd - 1));
  1197→    cp = cp * (1 + predDaily * decay);
  1198→    predictedPrices.push(Math.round(cp));
  1199→  }
  1200→
  1201→  const predictedReturn5d = (predictedPrices[PREDICT_DAYS - 1] - price) / price;
  1202→  return {
  1203→    predictedPrices,
  1204→    predictedReturn5d,
  1205→    confidence: Math.min(0.9, Math.max(0.1, r2 * 0.7 + 0.2)),
  1206→    trendDirection: predictedReturn5d > 0.005 ? 'up' : predictedReturn5d < -0.005 ? 'down' : 'flat',
  1207→  };
  1208→}

  1210→export function evaluatePredictedIndicators(data: OHLCV[], prediction: PricePrediction): { mlScore: number; predictedBullConsensus: number } {
  1211→  const price = data[data.length - 1].close;
  1212→  if (price <= 0 || prediction.predictedPrices.length === 0) return { mlScore: 0.5, predictedBullConsensus: 0.5 };
  1213→  const avgP = prediction.predictedPrices.reduce((a, b) => a + b, 0) / prediction.predictedPrices.length;
  1214→  const closes = data.map(d => d.close);
  1215→  const ma21 = sma(closes, 21), ma100 = sma(closes, 100), rsi = calcRSI(closes);
  1216→  const rsiP = Math.max(0, Math.min(1, (rsi + (avgP > price ? 5 : -5)) / 100));
  1217→  const ma21S = ma21 > 0 ? (avgP > ma21 ? 0.7 : 0.3) : 0.5;
  1218→  const ma100S = ma100 > 0 ? (avgP > ma100 ? 0.65 : 0.35) : 0.5;
  1219→  const dirS = avgP > price ? 0.65 : 0.35;
  1220→  return {
  1221→    mlScore: Math.max(0, Math.min(1, 0.5 + prediction.predictedReturn5d * 10 * prediction.confidence)),
  1222→    predictedBullConsensus: Math.max(0, Math.min(1, rsiP * 0.3 + ma21S * 0.25 + ma100S * 0.2 + dirS * 0.25)),
  1223→  };
  1224→}
