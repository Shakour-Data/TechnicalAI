    49→function hexToRgba(hex: string, alpha: number): string {
    50→  const r = parseInt(hex.slice(1, 3), 16);
    51→  const g = parseInt(hex.slice(3, 5), 16);
    52→  const b = parseInt(hex.slice(5, 7), 16);
    53→  return `rgba(${r},${g},${b},${alpha})`;
    54→}
    55→
    56→const toFa = (n: number) => Math.round(n).toLocaleString('fa-IR');
    57→const toFa1 = (n: number) => n.toFixed(1).replace(/\./g, '/').replace(/-/g, '−');
    58→const toFa2 = (n: number) => n.toFixed(2).replace(/\./g, '/').replace(/-/g, '−');
    59→
    60→function rsiSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    61→  if (v > 70) return 'bearish';
    62→  if (v < 30) return 'bullish';
    63→  return 'neutral';
    64→}
    65→
    66→function macdSignal(h: number): 'bullish' | 'bearish' | 'neutral' {
    67→  if (h > 0) return 'bullish';
    68→  if (h < 0) return 'bearish';
    69→  return 'neutral';
    70→}
    71→
    72→function adxSignal(v: number): 'strong' | 'neutral' {
    73→  if (v > 25) return 'strong';
    74→  return 'neutral';
    75→}
    76→
    77→function stochSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    78→  if (v > 80) return 'bearish';
    79→  if (v < 20) return 'bullish';
    80→  return 'neutral';

    83→function signalDotBg(signal: 'bullish' | 'bearish' | 'neutral' | 'strong', bullColor: string, bearColor: string, neutralColor: string, cardSubFg: string): string {
    84→  switch (signal) {
    85→    case 'bullish': return bullColor;
    86→    case 'bearish': return bearColor;
    87→    case 'strong': return neutralColor;
    88→    default: return cardSubFg;
    89→  }
    90→}

    92→function gradeLabel(grade: string): string {
    93→  switch (grade) {
    94→    case 'Very Strong': return 'بسیار قوی';
    95→    case 'Strong': return 'قوی';
    96→    case 'Moderate': return 'متوسط';
    97→    default: return 'ضعیف';
    98→  }
    99→}
   100→
   101→function gradeStyle(grade: string, cardSubFg: string): React.CSSProperties {
   102→  switch (grade) {
   103→    case 'Very Strong': return { background: 'rgba(239,68,68,0.2)', color: '#f87171' };
   104→    case 'Strong': return { background: 'rgba(245,158,11,0.2)', color: '#fbbf24' };
   105→    case 'Moderate': return { background: 'rgba(14,165,233,0.2)', color: '#38bdf8' };
   106→    default: return { background: 'rgba(255,255,255,0.1)', color: cardSubFg };
   107→  }
   108→}
   109→
   110→function formatNumber(n: number): string {
   111→  if (Math.abs(n) >= 1e9) return (n / 1e9).toFixed(1) + 'B';
   112→  if (Math.abs(n) >= 1e6) return (n / 1e6).toFixed(1) + 'M';
   113→  if (Math.abs(n) >= 1e3) return (n / 1e3).toFixed(1) + 'K';
   114→  return toFa(n);
   115→}
   116→
   117→function pctDistance(from: number, to: number): number {
   118→  if (from === 0) return 0;
   119→  return ((to - from) / from) * 100;
   120→}
   121→
   122→function trendDirectionLabel(dir: string, bullColor: string, bearColor: string, neutralColor: string): { label: string; color: string } {
   123→  const d = dir?.toLowerCase() ?? '';
   124→  if (d.includes('up') || d === 'صعودی') return { label: 'صعودی', color: bullColor };
   125→  if (d.includes('down') || d === 'نزولی') return { label: 'نزولی', color: bearColor };
   126→  return { label: 'خنثی', color: neutralColor };
   127→}

   152→function ProgressBar({ value, max, color, height = 4 }: { value: number; max: number; color: string; height?: number }) {
   153→  const { colors: C } = useTheme();
   154→  const pct = Math.min(100, Math.max(0, (value / max) * 100));
   155→  return (
   156→    <div className="w-full rounded-full overflow-hidden" style={{ height, background: C.cardBorder }}>
   157→      <div className="h-full rounded-full transition-all duration-500" style={{ width: `${pct}%`, background: color }} />
   158→    </div>
   159→  );
   160→}
