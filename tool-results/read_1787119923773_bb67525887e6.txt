   106→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
   107→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,

   109→    isIndex, isTgju, currencyUnit,

   111→  } = props;
   112→
   113→  const unit = currencyUnit || 'ریال';
   114→
   115→  // ── Instrument Context (for intelligent naming) ─────────────────

   119→    return { name: `سهم ${symbolName}`, type: 'stock' as const, verb: 'قرار دارد' };
   120→  }, [symbolName, isIndex, isTgju]);
   121→
   122→  const IL = instrumentLabel;
   123→
   124→  const containerRef = useRef<HTMLDivElement>(null);
   125→  const textAnalysisRef = useRef<HTMLDivElement>(null);
   126→  const [exporting, setExporting] = useState(false);
   127→
   128→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   129→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   130→  const S1_level = supports[0] ?? currentPrice * 0.95;
   131→  const S2_level = supports[1] ?? currentPrice * 0.90;
   132→  const S3_level = supports[2] ?? currentPrice * 0.85;
   133→
   134→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   135→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   136→
   137→  // ── Dominant scenario ──────────────────────────────────────────
   138→  let highestKey = 'R3';
   139→  let highestProb = 0;
   140→  for (const key of SCENARIO_KEYS) {
   141→    if (scenarios[key].probability > highestProb) {
   142→      highestProb = scenarios[key].probability;
   143→      highestKey = key;
   144→    }
   145→  }
   146→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   147→
   148→  // ── Dynamic scenario labels (context-aware based on overallSignal) ─
   149→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal), [overallSignal]);
   150→  const strategy = dynamicMeta[highestKey];
   151→
   152→  // ── Text variety: hash-based selector for varied language ──────────

   156→    for (let i = 0; i < raw.length; i++) hash = ((hash << 5) - hash + raw.charCodeAt(i)) | 0;
   157→    return Math.abs(hash);
   158→  }, [rsi, macdHist, cci, adx, stochK, trendAngle, currentPrice]);
   159→

   161→  const rsiZone = rsi > 70 ? 'overbought' : rsi > 60 ? 'weak_ob' : rsi > 40 ? 'neutral' : rsi > 30 ? 'weak_os' : 'oversold';
   162→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   163→  const rsiColor = rsi > 70 ? 'text-red-700' : rsi > 60 ? 'text-amber-800' : rsi > 40 ? 'text-gray-600' : rsi > 30 ? 'text-amber-800' : 'text-emerald-700';
   164→
   165→  // ── Stochastic signal ──────────────────────────────────────────
   166→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   167→
   168→  // ── Trend text ─────────────────────────────────────────────────
   169→  const isUp = trendDirection === 'up';
   170→  const isDown = trendDirection === 'down';

   207→    if (!edgeWeights) return null;

   209→    const edgeProbs = calcEdgeProbabilities(edgeWeights);

   213→    const rawSums: Record<string, number> = {};
   214→    for (const key of SCENARIO_KEYS) rawSums[key] = 0;
   215→    for (const p of allPathsRaw) rawSums[p.target] += p.prob;
   216→

   218→      const rawSum = rawSums[p.target] || 0.0001;
   219→      const targetProb = scenarios[p.target as keyof typeof scenarios].probability / 100;
   220→      const scale = targetProb / rawSum;
   221→      return { ...p, prob: p.prob * scale };
   222→    });
   223→
   224→    // Find top path per scenario
   225→    const topPaths: Record<string, PathInfo | null> = {};
   226→    for (const key of SCENARIO_KEYS) {
   227→      const best = allPaths.filter(p => p.target === key).sort((a, b) => b.prob - a.prob)[0] ?? null;
   228→      topPaths[key] = best;
   229→    }
   230→
   231→    return { topPaths };
   232→  }, [edgeWeights, scenarios]);

   241→        `با توجه به ورود به منطقه اشباع خرید (RSI=${toFaGrouped(rsi)}${mfiWarning}) و نزدیکی به مقاومت ${toFaGrouped(R1_level)}، ریسک اصلاح کوتاه‌مدت بالا ارزیابی می‌شود. پیشنهاد: انتظار برای بازگشت قیمت به ${toFaGrouped(S1_level)}-${toFaGrouped(S2_level)} و سپس بررسی مجدد سیگنال‌ها.`,

   243→      ];
   244→      return variants[v];
   245→    }
   246→    if (rsi < 30 || (hasVolume && mfi < 20)) {
   247→      const variants = [
   248→        `اشباع فروش قابل توجه (RSI: ${toFaGrouped(rsi)}${mfiWarning}) در محدوده حمایت ${toFaGrouped(S1_level)} شناسایی شد. این شرایط می‌تواند فرصت مناسبی برای ورود باشد، با حد ضرر ${toFaGrouped(S2_level)}. هدف اول ${toFaGrouped(R1_level)} و هدف دوم ${toFaGrouped(R2_level)} ${unit}.`,
   249→        `${IL.name} به منطقه اشباع فروش رسیده (RSI=${toFaGrouped(rsi)}${mfiWarning}). فاصله تا حمایت کلیدی ${toFaGrouped(S1_level)} کم است. اگر واکنش مثبت به این حمایت مشاهده شود، بازگشت به ${toFaGrouped(R1_level)} محتمل خواهد بود.`,
   250→        `شاخص قدرت نسبی در محدوده اشباع فروش (مقدار ${toFaGrouped(rsi)}${mfiWarning}) ${IL.verb}. نزدیکی به حمایت ${toFaGrouped(S1_level)} و سیگنال فروش بیش از حد، احتمال بازگشت کوتاه‌مدت را تقویت می‌کند. حد ضرر پیشنهادی: ${toFaGrouped(S2_level)}.`,
   251→      ];
   252→      return variants[v];
   253→    }
   254→    const variants = [
   255→      `سیگنال‌ها در محدوده خنثی قرار دارند (RSI: ${toFaGrouped(rsi)}، قدرت روند: ${adxStrength}). محدوده رنج فعلی بین ${toFaGrouped(S1_level)} و ${toFaGrouped(R1_level)} است. تا خروج قیمت از این کانال، بهترین اقدام نظاره و عدم ورود است.`,
   256→      `وضعیت فعلی ${IL.name} نشان‌دهنده عدم قطعیت (RSI=${toFaGrouped(rsi)}، ADX=${toFaGrouped(adx)}). ${IL.name} در بازه ${toFaGrouped(S1_level)} تا ${toFaGrouped(R1_level)} در نوسان است. شکست معتبر یکی از این سطوح، مسیر آینده را مشخص خواهد کرد.`,

   861→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   862→function LevelBox({ index, label, price, strength, isTarget, type, unit }: {
   863→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   864→}) {
   865→  const isSupport = type === 'support';
   866→  const isTargetType = type === 'target';
   867→
   868→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   869→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   870→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   871→
   872→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   873→  const strengthLabel = strength >= 8 ? 'بسیار قوی' : strength >= 6 ? 'قوی' : strength >= 4 ? 'متوسط' : strength >= 2 ? 'ضعیف' : 'بسیار ضعیف';
   874→  const strengthColor = strength >= 8 ? '#34c98b' : strength >= 6 ? '#3ad5db' : strength >= 4 ? '#ffb11b' : strength >= 2 ? '#ff7b32' : '#ef4d62';
   875→
   876→  // Distance from current price (if available from context)
   877→  return (
   878→    <div
   879→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   880→      style={{
   881→        background: bgAccent,
   882→        border: `1px solid ${borderAccent}`,
   883→        borderRight: `3px solid ${accentColor}`,
   884→      }}
   885→    >
   886→      {/* Label + Price */}
   887→      <div className="flex-1 min-w-0">
   888→        <div className="flex items-center gap-2">
   889→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   890→          {isTarget && <span className="text-[9px]">🎯</span>}
   891→        </div>
   892→        <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   893→        <span className="text-[10px] text-gray-500 mr-1">{unit || 'ریال'}</span>
   894→      </div>
   895→
   896→      {/* Strength bar */}
   897→      <div className="flex flex-col items-end gap-1 min-w-[80px]">
   898→        <div className="flex items-center gap-1.5">
   899→          <span className="text-[9px] font-medium" style={{ color: strengthColor }}>{strengthLabel}</span>
   900→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(strength)}</span>
   901→          <span className="text-[9px] text-gray-500">/۱۰</span>
   902→        </div>
   903→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   904→          <div
   905→            className="h-full rounded-full transition-all duration-500"
   906→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   907→          />
   908→        </div>
   909→      </div>
   910→    </div>
   911→  );
   912→}
