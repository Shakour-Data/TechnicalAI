    17→    17→function clamp(v: number, min: number, max: number): number {
    18→    18→  return Math.max(min, Math.min(max, v));
    19→    19→}
    20→    20→
    21→    21→function normalize(arr: number[]): number[] {
    22→    22→  const sum = arr.reduce((a, b) => a + b, 0);
    23→    23→  if (sum === 0) return arr.map(() => 1 / arr.length);
    24→    24→  return arr.map((v) => v / sum);
    25→    25→}
    26→    26→
    27→    27→function sigmoid(x: number): number {
    28→    28→  return 1 / (1 + Math.exp(-x));
    29→    29→}

   997→   997→function buildContext(input: GraphInput): ProbContext {
   998→   998→  const adxNorm = clamp(input.adx / 60, 0, 1);
   999→   999→  const srProximity = clamp(
  1000→  1000→    1 - Math.min(input.distToR1, input.distToS1) * 10,
  1001→  1001→    0,
  1002→  1002→    1
  1003→  1003→  );
  1004→  1004→  const oscillatorExtreme = clamp(
  1005→  1005→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
  1006→  1006→    0,
  1007→  1007→    1
  1008→  1008→  );
  1009→  1009→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1010→  1010→
  1011→  1011→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1012→  1012→  const overboughtRisk =
  1013→  1013→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1014→  1014→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1015→  1015→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1016→  1016→
  1017→  1017→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1018→  1018→  const oversoldBounce =
  1019→  1019→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1020→  1020→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1021→  1021→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1022→  1022→
  1023→  1023→  return {
  1024→  1024→    input,
  1025→  1025→    adxNorm,
  1026→  1026→    srProximity,
  1027→  1027→    oscillatorExtreme,
  1028→  1028→    divergenceProxy,
  1029→  1029→    overboughtRisk,
  1030→  1030→    oversoldBounce,
  1031→  1031→  };
  1032→  1032→}

  1240→  1240→      const pR8 = 0.2 + input.strengthIndex * 0.15;
  1241→  1241→      const pR9 = 0.15 + input.strengthIndex * 0.1;
  1242→  1242→      return normalize([pR7, pR8, pR9]);
  1243→  1243→    }
  1244→  1244→
  1245→  1245→    // ── N_B_D_VOL_W → [R6, R4, R5] ──────────────────────────────────────
  1246→  1246→    // Weak volume bearish breakout — fake breakout risk, capped R5
  1247→  1247→    case 'N_B_D_VOL_W': {
  1248→  1248→      const pR6 = 0.4 + (1 - input.bullConsensus) * 0.15;
  1249→  1249→      const pR4 = 0.3 + input.bullConsensus * 0.15;
  1250→  1250→      const pR5 = clamp((1 - input.confidenceIndex) * 0.25, 0.05, 0.2);
  1251→  1251→      return normalize([pR6, pR4, pR5]);
  1252→  1252→    }
  1253→  1253→
  1254→  1254→    // ── N_B_NONE → [R5] ──────────────────────────────────────────────────
  1255→  1255→    case 'N_B_NONE': {
  1256→  1256→      return [1.0];
  1257→  1257→    }
  1258→  1258→
  1259→  1259→    // ── N_REVERSAL → [N_R_BULL, N_R_BEAR, N_R_NONE] ──────────────────────
  1260→  1260→    case 'N_REVERSAL': {
  1261→  1261→      const bullRev = overboughtRisk;
  1262→  1262→      const bearRev = oversoldBounce;
  1263→  1263→      const noneRev =
  1264→  1264→        1 - Math.max(overboughtRisk, oversoldBounce) * 0.7;
  1265→  1265→      return normalize([bullRev, bearRev, noneRev]);
  1266→  1266→    }
  1267→  1267→
  1268→  1268→    // ── N_R_BULL → [N_R_B_DIV, N_R_B_CANDLE, N_R_B_SR] ──────────────────
  1269→  1269→    case 'N_R_BULL': {
  1270→  1270→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1271→  1271→      const candleProb =
  1272→  1272→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1273→  1273→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1274→  1274→      return normalize([divProb, candleProb, srProb]);
  1275→  1275→    }
  1276→  1276→
  1277→  1277→    // ── N_R_B_DIV → [R4, R2, R5] ────────────────────────────────────────
  1278→  1278→    // Bullish divergence — explicit small R5
  1279→  1279→    case 'N_R_B_DIV': {
  1280→  1280→      const pR4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1281→  1281→      const pR2 = 0.2 + input.mlMomentum * 0.15;
  1282→  1282→      const pR5 = clamp((1 - adxNorm) * 0.25, 0.05, 0.2);
  1283→  1283→      return normalize([pR4, pR2, pR5]);
  1284→  1284→    }
  1285→  1285→
  1286→  1286→    // ── N_R_B_CANDLE → [R3, R4, R5] ─────────────────────────────────────
  1287→  1287→    // Bullish candle pattern — explicit small R5
  1288→  1288→    case 'N_R_B_CANDLE': {
  1289→  1289→      const pR3 = 0.5 * input.confidenceIndex + 0.2;
  1290→  1290→      const pR4 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1291→  1291→      const pR5 = clamp((1 - input.confidenceIndex) * 0.2, 0.05, 0.18);
  1292→  1292→      return normalize([pR3, pR4, pR5]);
  1293→  1293→    }
  1294→  1294→
  1295→  1295→    // ── N_R_B_SR → [R4, R5, R6] ─────────────────────────────────────────
  1296→  1296→    // Support bounce — explicit small R5
  1297→  1297→    // Edge order: R4, R5, R6 — probabilities computed then reordered
  1298→  1298→    case 'N_R_B_SR': {
  1299→  1299→      const pR4 = 0.5 + input.srAvgStrength * 0.2;
  1300→  1300→      const pR6 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1301→  1301→      const pR5 = clamp((1 - input.srAvgStrength) * 0.2, 0.05, 0.18);
  1302→  1302→      // Return in edge order: [R4, R5, R6]
  1303→  1303→      return normalize([pR4, pR5, pR6]);
  1304→  1304→    }
  1305→  1305→
  1306→  1306→    // ── N_R_BEAR → [N_R_BE_DIV, N_R_BE_CANDLE, N_R_BE_SR] ───────────────
  1307→  1307→    // Mirror of N_R_BULL
  1308→  1308→    case 'N_R_BEAR': {
  1309→  1309→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1310→  1310→      const candleProb =
  1311→  1311→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1312→  1312→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1313→  1313→      return normalize([divProb, candleProb, srProb]);
  1314→  1314→    }
  1315→  1315→
  1316→  1316→    // ── N_R_BE_DIV → [R6, R8, R5] ───────────────────────────────────────
  1317→  1317→    // Bearish divergence — explicit small R5
  1318→  1318→    case 'N_R_BE_DIV': {
  1319→  1319→      const pR6 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1320→  1320→      const pR8 = 0.2 + input.mlMomentum * 0.15;
  1321→  1321→      const pR5 = clamp((1 - adxNorm) * 0.25, 0.05, 0.2);
  1322→  1322→      return normalize([pR6, pR8, pR5]);
  1323→  1323→    }
  1324→  1324→
  1325→  1325→    // ── N_R_BE_CANDLE → [R7, R6, R5] ────────────────────────────────────
  1326→  1326→    // Bearish candle pattern — explicit small R5
  1327→  1327→    case 'N_R_BE_CANDLE': {
  1328→  1328→      const pR7 = 0.5 * input.confidenceIndex + 0.2;
  1329→  1329→      const pR6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1330→  1330→      const pR5 = clamp((1 - input.confidenceIndex) * 0.2, 0.05, 0.18);
  1331→  1331→      return normalize([pR7, pR6, pR5]);
  1332→  1332→    }
  1333→  1333→
  1334→  1334→    // ── N_R_BE_SR → [R6, R5, R4] ─────────────────────────────────────────
  1335→  1335→    // Resistance rejection — explicit small R5
  1336→  1336→    // Edge order: R6, R5, R4 — probabilities computed then reordered
  1337→  1337→    case 'N_R_BE_SR': {
  1338→  1338→      const pR6 = 0.5 + input.srAvgStrength * 0.2;
  1339→  1339→      const pR4 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1340→  1340→      const pR5 = clamp((1 - input.srAvgStrength) * 0.2, 0.05, 0.18);
  1341→  1341→      // Return in edge order: [R6, R5, R4]
  1342→  1342→      return normalize([pR6, pR5, pR4]);
  1343→  1343→    }
  1344→  1344→
  1345→  1345→    // ── N_R_NONE → [R5] ──────────────────────────────────────────────────
  1346→  1346→    case 'N_R_NONE': {
  1347→  1347→      return [1.0];
  1348→  1348→    }
  1349→  1349→
  1350→  1350→    // ── Terminal nodes (R1–R9) have no outgoing edges ─────────────────────
  1351→  1351→    default:
  1352→  1352→      return [];
  1353→  1353→  }
  1354→  1354→}

  1356→  1356→// === Graph Traversal ==========================================================
  1357→  1357→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1358→  1358→// =============================================================================
  1359→  1359→
  1360→  1360→function traverseGraph(
  1361→  1361→  nodeId: string,
  1362→  1362→  pathProb: number,
  1363→  1363→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1364→  1364→  ctx: ProbContext,
  1365→  1365→  adj: Map<string, number[]>,
  1366→  1366→  edges: GraphEdge[],
  1367→  1367→  edgeProbabilities: Record<number, number>,
  1368→  1368→  nodeAccum: Record<string, number>,
  1369→  1369→  scenarios: Record<string, number>,
  1370→  1370→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1371→  1371→): void {
  1372→  1372→  // Accumulate path probability at this node
  1373→  1373→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1374→  1374→
  1375→  1375→  const outEdgeIndices = adj.get(nodeId);
  1376→  1376→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1377→  1377→
  1378→  1378→  // Compute outgoing edge probabilities for this node
  1379→  1379→  const probs = computeNodeProbabilities(
  1380→  1380→    nodeId,
  1381→  1381→    ctx,
  1382→  1382→    outEdgeIndices.length
  1383→  1383→  );
  1384→  1384→
  1385→  1385→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1386→  1386→    const edgeIdx = outEdgeIndices[i];
  1387→  1387→    const edge = edges[edgeIdx];
  1388→  1388→    const prob = probs[i] || 0;
  1389→  1389→
  1390→  1390→    // Store edge probability
  1391→  1391→    edgeProbabilities[edgeIdx] = prob;
  1392→  1392→
  1393→  1393→    // Determine branch for child nodes
  1394→  1394→    let childBranch = currentBranch;
  1395→  1395→    if (nodeId === 'ROOT') {
  1396→  1396→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1397→  1397→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1398→  1398→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1399→  1399→    }
  1400→  1400→
  1401→  1401→    const childPathProb = pathProb * prob;
  1402→  1402→
  1403→  1403→    // Check if target is a terminal node
  1404→  1404→    const isTerminal =
  1405→  1405→      edge.to.length === 2 &&
  1406→  1406→      edge.to[0] === 'R' &&
  1407→  1407→      edge.to[1] >= '1' &&
  1408→  1408→      edge.to[1] <= '9';
  1409→  1409→
  1410→  1410→    if (isTerminal) {
  1411→  1411→      // Accumulate scenario probability
  1412→  1412→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1413→  1413→      // Accumulate branch contribution
  1414→  1414→      if (childBranch) {
  1415→  1415→        contributions[edge.to][childBranch] += childPathProb;
  1416→  1416→      }
  1417→  1417→    }
  1418→  1418→
  1419→  1419→    // Recurse into child
  1420→  1420→    traverseGraph(
  1421→  1421→      edge.to,
  1422→  1422→      childPathProb,
  1423→  1423→      childBranch,
  1424→  1424→      ctx,
  1425→  1425→      adj,
  1426→  1426→      edges,
  1427→  1427→      edgeProbabilities,
  1428→  1428→      nodeAccum,
  1429→  1429→      scenarios,
  1430→  1430→      contributions
  1431→  1431→    );
  1432→  1432→  }
  1433→  1433→}

  1435→  1435→// === Main Export: buildDecisionGraph ===========================================
  1436→  1436→
  1437→  1437→export function buildDecisionGraph(input: GraphInput): GraphData {
  1438→  1438→  // 1. Build static graph structure
  1439→  1439→  const nodes = createNodes();
  1440→  1440→  const edges = createEdges();
  1441→  1441→  const nodePositions = createPositions();
  1442→  1442→
  1443→  1443→  // 2. Build adjacency map: nodeId → array of edge indices
  1444→  1444→  const adj = new Map<string, number[]>();
  1445→  1445→  for (let i = 0; i < edges.length; i++) {
  1446→  1446→    const from = edges[i].from;
  1447→  1447→    if (!adj.has(from)) adj.set(from, []);
  1448→  1448→    adj.get(from)!.push(i);
  1449→  1449→  }
  1450→  1450→
  1451→  1451→  // 3. Build probability context
  1452→  1452→  const ctx = buildContext(input);
  1453→  1453→
  1454→  1454→  // 4. Initialize accumulation structures
  1455→  1455→  const edgeProbabilities: Record<number, number> = {};
  1456→  1456→  const nodeAccum: Record<string, number> = {};
  1457→  1457→  const scenarios: Record<string, number> = {};
  1458→  1458→  const contributions: Record<
  1459→  1459→    string,
  1460→  1460→    { trend: number; breakout: number; reversal: number }
  1461→  1461→  > = {};
  1462→  1462→
  1463→  1463→  for (let i = 1; i <= 9; i++) {
  1464→  1464→    const key = `R${i}`;
  1465→  1465→    scenarios[key] = 0;
  1466→  1466→    contributions[key] = { trend: 0, breakout: 0, reversal: 0 };
  1467→  1467→  }
  1468→  1468→
  1469→  1469→  // 5. Traverse graph from ROOT
  1470→  1470→  traverseGraph(
  1471→  1471→    'ROOT',
  1472→  1472→    1.0,
  1473→  1473→    null,
  1474→  1474→    ctx,
  1475→  1475→    adj,
  1476→  1476→    edges,
  1477→  1477→    edgeProbabilities,
  1478→  1478→    nodeAccum,
  1479→  1479→    scenarios,
  1480→  1480→    contributions
  1481→  1481→  );
  1482→  1482→
  1483→  1483→  // 6. Post-process scenario probabilities
  1484→  1484→  // Convert to percentage, clamp [2, 35], re-normalize to sum=100
  1485→  1485→  const rawPcts: Record<string, number> = {};
  1486→  1486→  for (let i = 1; i <= 9; i++) {
  1487→  1487→    const key = `R${i}`;
  1488→  1488→    rawPcts[key] = Math.round((scenarios[key] || 0) * 100);
  1489→  1489→  }
  1490→  1490→
  1491→  1491→  // Clamp each to [2, 35]
  1492→  1492→  const clamped = Object.entries(rawPcts).map(([k, v]) => ({
  1493→  1493→    key: k,
  1494→  1494→    value: clamp(v, 2, 35),
  1495→  1495→  }));
  1496→  1496→
  1497→  1497→  // Re-normalize to sum=100
  1498→  1498→  const clampedSum = clamped.reduce((s, c) => s + c.value, 0);
  1499→  1499→  const finalPcts: Record<string, number> = {};
  1500→  1500→  if (clampedSum > 0) {
  1501→  1501→    // Distribute proportionally and round, then fix rounding errors
  1502→  1502→    let runningTotal = 0;
  1503→  1503→    const entries = clamped
  1504→  1504→      .map((c) => {
  1505→  1505→        const raw = (c.value / clampedSum) * 100;
  1506→  1506→        return { key: c.key, raw };
  1507→  1507→      })
  1508→  1508→      .sort((a, b) => b.raw - a.raw);
  1509→  1509→
  1510→  1510→    for (let i = 0; i < entries.length; i++) {
  1511→  1511→      const isLast = i === entries.length - 1;
  1512→  1512→      if (isLast) {
  1513→  1513→        finalPcts[entries[i].key] = Math.round(100 - runningTotal);
  1514→  1514→      } else {
  1515→  1515→        const rounded = Math.round(entries[i].raw);
  1516→  1516→        finalPcts[entries[i].key] = rounded;
  1517→  1517→        runningTotal += rounded;
  1518→  1518→      }
  1519→  1519→    }
  1520→  1520→  }
  1521→  1521→
  1522→  1522→  // Ensure no negative values due to rounding
  1523→  1523→  for (const key of Object.keys(finalPcts)) {
  1524→  1524→    if (finalPcts[key] < 0) finalPcts[key] = 0;
  1525→  1525→  }
  1526→  1526→
  1527→  1527→  // 7. Compute branch probabilities (from ROOT outgoing edges)
  1528→  1528→  const rootEdgeIndices = adj.get('ROOT') || [];
  1529→  1529→  const rootProbs = computeNodeProbabilities('ROOT', ctx, rootEdgeIndices.length);
  1530→  1530→  const branchProbabilities = {
  1531→  1531→    trend: rootProbs[0] || 0,
  1532→  1532→    breakout: rootProbs[1] || 0,
  1533→  1533→    reversal: rootProbs[2] || 0,
  1534→  1534→  };
  1535→  1535→
  1536→  1536→  // 8. Build node display values (path probability as percentage)
  1537→  1537→  const nodeValues: Record<string, string> = {};
  1538→  1538→  for (const [nodeId, accum] of Object.entries(nodeAccum)) {
  1539→  1539→    nodeValues[nodeId] = `${Math.round(accum * 100)}%`;
  1540→  1540→  }
  1541→  1541→
  1542→  1542→  // Override terminal node values with final post-processed percentages
  1543→  1543→  for (const [key, pct] of Object.entries(finalPcts)) {
  1544→  1544→    nodeValues[key] = `${pct}%`;
  1545→  1545→  }
  1546→  1546→
  1547→  1547→  // 9. Build path contributions (as percentages)
  1548→  1548→  const pathContributions: Record<
  1549→  1549→    string,
  1550→  1550→    { trend: number; breakout: number; reversal: number }
  1551→  1551→  > = {};
  1552→  1552→  for (let i = 1; i <= 9; i++) {
  1553→  1553→    const key = `R${i}`;
  1554→  1554→    const c = contributions[key];
  1555→  1555→    const total = c.trend + c.breakout + c.reversal;
  1556→  1556→    if (total > 0) {
  1557→  1557→      pathContributions[key] = {
  1558→  1558→        trend: Math.round((c.trend / total) * 100),
  1559→  1559→        breakout: Math.round((c.breakout / total) * 100),
  1560→  1560→        reversal: Math.round((c.reversal / total) * 100),
  1561→  1561→      };
  1562→  1562→    } else {
  1563→  1563→      pathContributions[key] = { trend: 33, breakout: 34, reversal: 33 };
  1564→  1564→    }
  1565→  1565→  }
  1566→  1566→
  1567→  1567→  // 10. Compute 30-day probability trend
  1568→  1568→  const probFractions: Record<string, number> = {};
  1569→  1569→  for (const [k, v] of Object.entries(finalPcts)) {
  1570→  1570→    probFractions[k] = v / 100;
  1571→  1571→  }
  1572→  1572→  const probabilityTrend = calcTrendFromProbs(probFractions, 30);
  1573→  1573→
  1574→  1574→  // 11. Return complete GraphData
  1575→  1575→  return {
  1576→  1576→    nodes,
  1577→  1577→    edges,
  1578→  1578→    nodePositions,
  1579→  1579→    edgeProbabilities,
  1580→  1580→    nodeValues,
  1581→  1581→    branchProbabilities,
  1582→  1582→    scenarioProbabilities: finalPcts,
  1583→  1583→    pathContributions,
  1584→  1584→    probabilityTrend,
  1585→  1585→  };
  1586→  1586→}
