   777→      className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-gray-100 border border-gray-300 text-xs text-gray-600 hover:bg-gray-200 hover:text-gray-900 transition-all disabled:opacity-50 cursor-pointer">
   778→      {loading ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : icon}
   779→      {label}
   780→    </button>
   781→  );
   782→}

   784→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   785→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   786→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   787→  grade?: string; overlapCount?: number; touchCount?: number;
   788→}) {
   789→  const isSupport = type === 'support';
   790→  const isTargetType = type === 'target';
   791→
   792→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   793→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   794→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   795→
   796→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   797→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   798→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   799→
   800→  return (
   801→    <div
   802→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   803→      style={{
   804→        background: bgAccent,
   805→        border: `1px solid ${borderAccent}`,
   806→        borderRight: `3px solid ${accentColor}`,
   807→      }}
   808→    >
   809→      {/* Label + Price */}
   810→      <div className="flex-1 min-w-0">
   811→        <div className="flex items-center gap-2">
   812→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   813→          {isTarget && <span className="text-[9px]">🎯</span>}
   814→          {overlapCount !== undefined && overlapCount > 0 && (
   815→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   816→              {toPersianDigits(overlapCount)} هم‌پوشانی
   817→            </span>
   818→          )}
   819→        </div>
   820→        <div className="flex items-baseline gap-1">
   821→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   822→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   823→        </div>
   824→      </div>
   825→
   826→      {/* Strength + Grade */}
   827→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   828→        <div className="flex items-center gap-1.5">
   829→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   830→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   831→          <span className="text-[9px] text-gray-500">/۱۰</span>
   832→        </div>
   833→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   834→          <div
   835→            className="h-full rounded-full transition-all duration-500"
   836→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   837→          />
   838→        </div>
   839→        {touchCount !== undefined && touchCount > 0 && (
   840→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   841→        )}
   842→      </div>
   843→    </div>
   844→  );
   845→}
