   273→/* Advanced query parser — supports operators like country:US, exchange:NASD, sector:Tech */
   274→interface ParsedQuery {
   275→  country?: string;
   276→  exchange?: string;
   277→  sector?: string;
   278→  terms: string[];
   279→}
   280→
   281→function parseQuery(q: string): ParsedQuery {
   282→  const country: string[] = [];
   283→  const exchange: string[] = [];
   284→  const sector: string[] = [];
   285→  const terms: string[] = [];
   286→
   287→  for (const part of q.split(/\s+/)) {
   288→    if (!part) continue;
   289→    const lower = part.toLowerCase();
   290→    if (lower.startsWith('country:') || lower.startsWith('کشور:')) {
   291→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   292→      if (val) country.push(val.toLowerCase());
   293→    } else if (lower.startsWith('exchange:') || lower.startsWith('صرافی:')) {
   294→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   295→      if (val) exchange.push(val.toLowerCase());
   296→    } else if (lower.startsWith('sector:') || lower.startsWith('صنعت:')) {
   297→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   298→      if (val) sector.push(val.toLowerCase());
   299→    } else {
   300→      terms.push(lower);
   301→    }
   302→  }
   303→
   304→  return {
   305→    country: country.length > 0 ? country.join(' ') : undefined,
   306→    exchange: exchange.length > 0 ? exchange.join(' ') : undefined,
   307→    sector: sector.length > 0 ? sector.join(' ') : undefined,
   308→    terms,
   309→  };
   310→}

   316→const RECENT_KEY = 'fin-search-recent';
   317→const MAX_RECENT = 8;
   318→
   319→function getRecentSearches(): string[] {
   320→  if (typeof window === 'undefined') return [];
   321→  try { const raw = localStorage.getItem(RECENT_KEY); return raw ? JSON.parse(raw) : []; }
   322→  catch { return []; }
   323→}
   324→
   325→function addRecentSearch(name: string) {
   326→  if (typeof window === 'undefined' || !name) return;
   327→  try {
   328→    let list = getRecentSearches();
   329→    list = list.filter((r) => r !== name);
   330→    list.unshift(name);
   331→    if (list.length > MAX_RECENT) list = list.slice(0, MAX_RECENT);
   332→    localStorage.setItem(RECENT_KEY, JSON.stringify(list));
   333→  } catch { /* ignore */ }
   334→}

   551→  const doFilter = React.useCallback((items: InstrumentItem[], q: string) => {
   552→    const parsed = parseQuery(q);
   553→    const { country: opCountry, exchange: opExchange, sector: opSector, terms } = parsed;
   554→
   555→    const scoreAndFilter = (item: InstrumentItem) => {
   556→      /* Apply query operators */
   557→      if (opCountry) {
   558→        const match = (item.country?.toLowerCase().includes(opCountry)) ||
   559→          (item.countryEn?.toLowerCase().includes(opCountry));
   560→        if (!match) return null;
   561→      }
   562→      if (opExchange) {
   563→        const match = item.exchange?.toLowerCase().includes(opExchange);
   564→        if (!match) return null;
   565→      }
   566→      if (opSector) {
   567→        const match = item.sector?.toLowerCase().includes(opSector) ||
   568→          item.cs?.toLowerCase().includes(opSector);
   569→        if (!match) return null;
   570→      }
   571→
   572→      const fields = [item.l18, item.l30, item.cs, item.yahooSymbol || '', item.nameEn || '', item.exchange || '', item.country || '', item.countryEn || '', item.sector || ''];
   573→      const fullText = fields.join(' ').toLowerCase();
   574→      let score = 0;
   575→
   576→      for (const term of terms) {
   577→        if (!term) continue;
   578→        if (item.l18.toLowerCase().includes(term)) score += 10;
   579→        if (item.nameEn?.toLowerCase().includes(term)) score += 8;
   580→        if (item.yahooSymbol?.toLowerCase() === term) score += 20;
   581→        else if (item.yahooSymbol?.toLowerCase().includes(term)) score += 15;
   582→        if (item.l30.toLowerCase().includes(term)) score += 7;
   583→        if (item.exchange?.toLowerCase().includes(term)) score += 5;
   584→        if (item.country?.toLowerCase().includes(term)) score += 6;
   585→        if (item.countryEn?.toLowerCase() === term) score += 15;
   586→        else if (item.countryEn?.toLowerCase().includes(term)) score += 4;
   587→        if (item.sector?.toLowerCase().includes(term)) score += 5;
   588→        if (item.cs?.toLowerCase().includes(term)) score += 6;
   589→
   590→        if (!item.l18.toLowerCase().includes(term) && !item.l30.toLowerCase().includes(term) &&
   591→            !item.cs?.toLowerCase().includes(term) && !(item.yahooSymbol?.toLowerCase().includes(term)) &&
   592→            !(item.nameEn?.toLowerCase().includes(term)) && !(item.exchange?.toLowerCase().includes(term)) &&
   593→            !(item.country?.toLowerCase().includes(term)) && !(item.countryEn?.toLowerCase().includes(term)) &&
   594→            !(item.sector?.toLowerCase().includes(term))) {
   595→          return null;
   596→        }
   597→      }
   598→      /* If only operators (no terms), accept all filtered items */
   599→      if (terms.length === 0) score = 1;
   600→      return score;
   601→    };
   602→
   603→    const scored = items.map((item) => ({ item, score: scoreAndFilter(item) }))
   604→      .filter((r): r is { item: InstrumentItem; score: number } => r.score !== null && r.score > 0)
   605→      .sort((a, b) => b.score - a.score);
   606→
   607→    setTotalMatched(scored.length);
   608→    setResults(scored.slice(0, MAX_RESULTS).map((s) => s.item));
   609→    setActiveIndex(-1);
   610→  }, []);
