   130→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   131→function sma(closes: number[], period: number): number {
   132→  if (closes.length < period) return 0;
   133→  const slice = closes.slice(-period);
   134→  return slice.reduce((a, b) => a + b, 0) / period;
   135→}

   137→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   138→function emaCalc(closes: number[], period: number): number {
   139→  if (closes.length < period) return 0;
   140→  const k = 2 / (period + 1);
   141→  let emaVal = sma(closes.slice(0, period), period);
   142→  for (let i = period; i < closes.length; i++) {
   143→    emaVal = closes[i] * k + emaVal * (1 - k);
   144→  }
   145→  return emaVal;
   146→}

   148→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   149→function calcRSI(closes: number[], period: number = 14): number {
   150→  if (closes.length < period + 1) return 50;
   151→  let avgGain = 0;
   152→  let avgLoss = 0;
   153→  for (let i = 1; i <= period; i++) {
   154→    const diff = closes[i] - closes[i - 1];
   155→    if (diff > 0) avgGain += diff;
   156→    else avgLoss += Math.abs(diff);
   157→  }
   158→  avgGain /= period;
   159→  avgLoss /= period;
   160→  for (let i = period + 1; i < closes.length; i++) {
   161→    const diff = closes[i] - closes[i - 1];
   162→    const gain = diff > 0 ? diff : 0;
   163→    const loss = diff < 0 ? Math.abs(diff) : 0;
   164→    avgGain = (avgGain * (period - 1) + gain) / period;
   165→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   166→  }
   167→  if (avgLoss === 0) return 100;
   168→  const rs = avgGain / avgLoss;
   169→  return 100 - 100 / (1 + rs);
   170→}

   172→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   173→function calcMFI(data: OHLCV[], period: number = 14): number {
   174→  if (data.length < period + 1) return 50;
   175→  let posFlow = 0;
   176→  let negFlow = 0;
   177→  for (let i = data.length - period; i < data.length; i++) {
   178→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   179→    const mf = tp * data[i].volume;
   180→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   181→    if (tp > prevTp) posFlow += mf;
   182→    else negFlow += mf;
   183→  }
   184→  if (negFlow === 0) return 100;
   185→  const mfr = posFlow / negFlow;
   186→  return 100 - 100 / (1 + mfr);
   187→}

   189→// ─── Helper: CCI(20) ──────────────────────────────────────────────────────────
   190→function calcCCI(data: OHLCV[], period: number = 20): number {
   191→  if (data.length < period) return 0;
   192→  const tps: number[] = [];
   193→  for (let i = data.length - period; i < data.length; i++) {
   194→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   195→  }
   196→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   197→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   198→  if (meanDev === 0) return 0;
   199→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   200→}

   202→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   203→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   204→  if (data.length < kPeriod) return { k: 50, d: 50 };
   205→  const rawKs: number[] = [];
   206→  for (let i = kPeriod - 1; i < data.length; i++) {
   207→    let lowest = Infinity;
   208→    let highest = -Infinity;
   209→    for (let j = i - kPeriod + 1; j <= i; j++) {
   210→      if (data[j].low < lowest) lowest = data[j].low;
   211→      if (data[j].high > highest) highest = data[j].high;
   212→    }
   213→    const range = highest - lowest;
   214→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   215→  }
   216→  const smoothedK: number[] = [];
   217→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   218→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   219→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   220→  }
   221→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   222→  const dSlice = smoothedK.slice(-smoothD);
   223→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   224→  return { k, d };
   225→}

   227→// ─── Helper: Williams %R(14) ──────────────────────────────────────────────────
   228→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   229→  if (data.length < period) return -50;
   230→  const slice = data.slice(-period);
   231→  let highest = -Infinity;
   232→  let lowest = Infinity;
   233→  for (const c of slice) {
   234→    if (c.high > highest) highest = c.high;
   235→    if (c.low < lowest) lowest = c.low;
   236→  }
   237→  const range = highest - lowest;
   238→  if (range === 0) return -50;
   239→  return ((highest - data[data.length - 1].close) / range) * -100;
   240→}

   242→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   243→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   244→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   245→  const emaFast = emaCalc(closes, fast);
   246→  const emaSlow = emaCalc(closes, slow);
   247→  const macdLine = emaFast - emaSlow;
   248→  // Calculate signal as EMA of MACD values
   249→  const macdValues: number[] = [];
   250→  const kFast = 2 / (fast + 1);
   251→  const kSlow = 2 / (slow + 1);
   252→  let eFast = sma(closes.slice(0, fast), fast);
   253→  let eSlow = sma(closes.slice(0, slow), slow);
   254→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   255→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   256→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   257→    macdValues.push(eFast - eSlow);
   258→  }
   259→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   260→  const kSig = 2 / (sig + 1);
   261→  let signalVal = sma(macdValues.slice(0, sig), sig);
   262→  for (let i = sig; i < macdValues.length; i++) {
   263→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   264→  }
   265→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   266→}

   268→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   269→function calcATR(data: OHLCV[], period: number = 14): number {
   270→  if (data.length < 2) return 0;
   271→  const trs: number[] = [];
   272→  for (let i = 1; i < data.length; i++) {
   273→    const tr = Math.max(
   274→      data[i].high - data[i].low,
   275→      Math.abs(data[i].high - data[i - 1].close),
   276→      Math.abs(data[i].low - data[i - 1].close)
   277→    );
   278→    trs.push(tr);
   279→  }
   280→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   281→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   282→  for (let i = period; i < trs.length; i++) {
   283→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   284→  }
   285→  return atrVal;
   286→}

   288→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   289→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   290→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   291→  const plusDMs: number[] = [];
   292→  const minusDMs: number[] = [];
   293→  const trs: number[] = [];
   294→  for (let i = 1; i < data.length; i++) {
   295→    const upMove = data[i].high - data[i - 1].high;
   296→    const downMove = data[i - 1].low - data[i].low;
   297→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   298→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   299→    trs.push(Math.max(
   300→      data[i].high - data[i].low,
   301→      Math.abs(data[i].high - data[i - 1].close),
   302→      Math.abs(data[i].low - data[i - 1].close)
   303→    ));
   304→  }
   305→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   306→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   307→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   308→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   309→  const dxValues: number[] = [];
   310→  for (let i = period; i < trs.length; i++) {
   311→    smoothTR = smoothTR - smoothTR / period + trs[i];
   312→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   313→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   314→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   315→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   316→    const diSum = diPlus + diMinus;
   317→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   318→    dxValues.push(dx);
   319→  }
   320→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   321→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   322→  for (let i = period; i < dxValues.length; i++) {
   323→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   324→  }
   325→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   326→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   327→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   328→}

   330→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   331→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   332→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   333→  let isLong = true;
   334→  let af = step;
   335→  let ep = data[0].high;
   336→  let sar = data[0].low;
   337→  for (let i = 1; i < data.length; i++) {
   338→    const prevSar = sar;
   339→    sar = prevSar + af * (ep - prevSar);
   340→    // Ensure SAR is within bounds
   341→    if (isLong) {
   342→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   343→      if (data[i].low < sar) {
   344→        isLong = false;
   345→        sar = ep;
   346→        ep = data[i].low;
   347→        af = step;
   348→      } else {
   349→        if (data[i].high > ep) {
   350→          ep = data[i].high;
   351→          af = Math.min(af + step, max);
   352→        }
   353→      }
   354→    } else {
   355→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   356→      if (data[i].high > sar) {
   357→        isLong = true;
   358→        sar = ep;
   359→        ep = data[i].high;
   360→        af = step;
   361→      } else {
   362→        if (data[i].low < ep) {
   363→          ep = data[i].low;
   364→          af = Math.min(af + step, max);
   365→        }
   366→      }
   367→    }
   368→  }
   369→  return sar;
   370→}

   372→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   373→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   374→  if (closes.length < period) {
   375→    const c = closes[closes.length - 1] ?? 0;
   376→    return { upper: c, middle: c, lower: c };
   377→  }
   378→  const slice = closes.slice(-period);
   379→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   380→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   381→  const sd = Math.sqrt(variance);
   382→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   383→}

   385→// ─── Helper: OBV ──────────────────────────────────────────────────────────────
   386→function calcOBV(data: OHLCV[]): number {
   387→  let obv = 0;
   388→  for (let i = 1; i < data.length; i++) {
   389→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   390→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   391→  }
   392→  return obv;
   393→}

   395→// ─── Helper: VWAP ──────────────────────────────────────────────────────────────
   396→function calcVWAP(data: OHLCV[]): number {
   397→  if (data.length === 0) return 0;
   398→  let cumTPV = 0;
   399→  let cumVol = 0;
   400→  for (const d of data) {
   401→    const tp = (d.high + d.low + d.close) / 3;
   402→    cumTPV += tp * d.volume;
   403→    cumVol += d.volume;
   404→  }
   405→  return cumVol === 0 ? data[data.length - 1].close : cumTPV / cumVol;
   406→}

   421→// ─── Helper: Ichimoku Cloud (9, 26, 52) ──────────────────────────────────────
   422→function calcIchimoku(data: OHLCV[]): {
   423→  tenkan: number; kijun: number; senkouA: number; senkouB: number; chikou: number;
   424→} {
   425→  if (data.length < 52) {
   426→    const c = data[data.length - 1]?.close ?? 0;
   427→    return { tenkan: c, kijun: c, senkouA: c, senkouB: c, chikou: c };
   428→  }
   429→  const highs = data.map(d => d.high);
   430→  const lows = data.map(d => d.low);
   431→  const closes = data.map(d => d.close);
   432→
   433→  function donchian(h: number[], l: number[], end: number, period: number): number {
   434→    const start = end - period + 1;
   435→    if (start < 0) return (h[end] + l[end]) / 2;
   436→    let hi = -Infinity, lo = Infinity;
   437→    for (let i = start; i <= end; i++) {
   438→      if (h[i] > hi) hi = h[i];
   439→      if (l[i] < lo) lo = l[i];
   440→    }
   441→    return (hi + lo) / 2;
   442→  }
   443→
   444→  const n = data.length;
   445→  const tenkan = donchian(highs, lows, n - 1, 9);
   446→  const kijun = donchian(highs, lows, n - 1, 26);
   447→  // Senkou A = (tenkan + kijun) / 2, shifted 26 periods ahead
   448→  // We compute it at bar n-27 (shifted to current)
   449→  const senkouAIdx = Math.max(n - 1 - 26, 0);
   450→  const tA = donchian(highs, lows, senkouAIdx, 9);
   451→  const kA = donchian(highs, lows, senkouAIdx, 26);
   452→  const senkouA = (tA + kA) / 2;
   453→  // Senkou B = donchian(52), shifted 26 periods ahead
   454→  const senkouBIdx = Math.max(n - 1 - 26, 0);
   455→  const senkouB = donchian(highs, lows, senkouBIdx, 52);
   456→  // Chikou = current close plotted 26 periods back
   457→  const chikou = closes[n - 1];
   458→
   459→  return { tenkan, kijun, senkouA, senkouB, chikou };
   460→}

   462→function calcIchimokuArrays(data: OHLCV[]): {
   463→  tenkan: number[]; kijun: number[]; senkouA: number[]; senkouB: number[];
   464→} {
   465→  const n = data.length;
   466→  const highs = data.map(d => d.high);
   467→  const lows = data.map(d => d.low);
   468→
   469→  function donchian(end: number, period: number): number {
   470→    const start = end - period + 1;
   471→    if (start < 0) return (highs[end] + lows[end]) / 2;
   472→    let hi = -Infinity, lo = Infinity;
   473→    for (let i = start; i <= end; i++) {
   474→      if (highs[i] > hi) hi = highs[i];
   475→      if (lows[i] < lo) lo = lows[i];
   476→    }
   477→    return (hi + lo) / 2;
   478→  }
   479→
   480→  const tenkan: number[] = [];
   481→  const kijun: number[] = [];
   482→  const senkouA: number[] = [];
   483→  const senkouB: number[] = [];
   484→
   485→  for (let i = 0; i < n; i++) {
   486→    tenkan.push(donchian(i, 9));
   487→    kijun.push(donchian(i, 26));
   488→    // Senkou A shifted +26: at bar i, we need tenkan/kijun at bar i-26
   489→    const sIdx = i - 26;
   490→    if (sIdx >= 0) {
   491→      const t = donchian(sIdx, 9);
   492→      const k = donchian(sIdx, 26);
   493→      senkouA.push((t + k) / 2);
   494→    } else {
   495→      senkouA.push(donchian(i, 9)); // fallback
   496→    }
   497→    // Senkou B shifted +26: at bar i, we need donchian(52) at bar i-26
   498→    if (sIdx >= 0) {
   499→      senkouB.push(donchian(sIdx, 52));
   500→    } else {
   501→      senkouB.push(donchian(i, 52)); // fallback
   502→    }
   503→  }
   504→
   505→  return { tenkan, kijun, senkouA, senkouB };
   506→}

   508→// ─── Helper: SMA Array (full series) ──────────────────────────────────────
   509→function smaArray(closes: number[], period: number): number[] {
   510→  const result: number[] = [];
   511→  for (let i = 0; i < closes.length; i++) {
   512→    if (i < period - 1) { result.push(0); continue; }
   513→    let sum = 0;
   514→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   515→    result.push(sum / period);
   516→  }
   517→  return result;
   518→}

   520→// ─── Helper: EMA Array (full series) ──────────────────────────────────────
   521→function emaArrayCalc(closes: number[], period: number): number[] {
   522→  const result: number[] = [];
   523→  if (closes.length < period) return closes.map(() => 0);
   524→  const k = 2 / (period + 1);
   525→  let emaVal = 0;
   526→  for (let i = 0; i < period; i++) emaVal += closes[i];
   527→  emaVal /= period;
   528→  for (let i = 0; i < closes.length; i++) {
   529→    if (i < period) { result.push(0); continue; }
   530→    if (i === period) { result.push(emaVal); continue; }
   531→    emaVal = closes[i] * k + emaVal * (1 - k);
   532→    result.push(emaVal);
   533→  }
   534→  return result;
   535→}

   537→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   538→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   539→  const n = values.length;
   540→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   541→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   542→  for (let i = 0; i < n; i++) {
   543→    sumX += i;
   544→    sumY += values[i];
   545→    sumXY += i * values[i];
   546→    sumX2 += i * i;
   547→    sumY2 += values[i] * values[i];
   548→  }
   549→  const denom = n * sumX2 - sumX * sumX;
   550→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   551→  const slope = (n * sumXY - sumX * sumY) / denom;
   552→  const intercept = (sumY - slope * sumX) / n;
   553→  const meanY = sumY / n;
   554→  let ssRes = 0, ssTot = 0;
   555→  for (let i = 0; i < n; i++) {
   556→    const predicted = slope * i + intercept;
   557→    ssRes += (values[i] - predicted) ** 2;
   558→    ssTot += (values[i] - meanY) ** 2;
   559→  }
   560→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   561→  return { slope, intercept, r2 };
   562→}

   564→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   565→function roundToNice(price: number): number {
   566→  const abs = Math.abs(price);
   567→  let step: number;
   568→  if (abs >= 100000) step = 100;
   569→  else if (abs >= 10000) step = 50;
   570→  else if (abs >= 1000) step = 10;
   571→  else if (abs >= 100) step = 5;
   572→  else step = 1;
   573→  return Math.round(price / step) * step;
   574→}

   577→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: { price: number; idx: number }[]; lows: { price: number; idx: number }[] } {
   578→  const highs: { price: number; idx: number }[] = [];
   579→  const lows: { price: number; idx: number }[] = [];
   580→  for (let i = lookback; i < data.length - lookback; i++) {
   581→    let isHigh = true;
   582→    let isLow = true;
   583→    for (let j = i - lookback; j <= i + lookback; j++) {
   584→      if (j === i) continue;
   585→      if (data[j].high >= data[i].high) isHigh = false;
   586→      if (data[j].low <= data[i].low) isLow = false;
   587→    }
   588→    if (isHigh) highs.push({ price: data[i].high, idx: i });
   589→    if (isLow) lows.push({ price: data[i].low, idx: i });
   590→  }
   591→  return { highs, lows };
   592→}

   657→// ─── Helper: Linear System Solver (Gaussian Elimination) ─────────────────
   658→function solveLinearSystem(A: number[][], b: number[]): number[] | null {
   659→  const n = b.length;
   660→  const aug = A.map((row, i) => [...row, b[i]]);
   661→
   662→  for (let col = 0; col < n; col++) {
   663→    let maxVal = Math.abs(aug[col][col]);
   664→    let maxRow = col;
   665→    for (let row = col + 1; row < n; row++) {
   666→      if (Math.abs(aug[row][col]) > maxVal) {
   667→        maxVal = Math.abs(aug[row][col]);
   668→        maxRow = row;
   669→      }
   670→    }
   671→    if (maxVal < 1e-10) return null;
   672→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   673→
   674→    for (let row = col + 1; row < n; row++) {
   675→      const factor = aug[row][col] / aug[col][col];
   676→      for (let j = col; j <= n; j++) {
   677→        aug[row][j] -= factor * aug[col][j];
   678→      }
   679→    }
   680→  }
   681→
   682→  const x = new Array(n).fill(0);
   683→  for (let i = n - 1; i >= 0; i--) {
   684→    x[i] = aug[i][n];
   685→    for (let j = i + 1; j < n; j++) {
   686→      x[i] -= aug[i][j] * x[j];
   687→    }
   688→    if (Math.abs(aug[i][i]) < 1e-10) return null;
   689→    x[i] /= aug[i][i];
   690→  }
   691→
   692→  return x;
   693→}

   695→// ─── Helper: Generate All 7-Method S/R Levels ─────────────────────────────
   696→function generateAllLevels(data: OHLCV[], endIdx: number): RawSRLevel[] {
   697→  const levels: RawSRLevel[] = [];
   698→  if (endIdx < 10) return levels;
   699→
   700→  const slice = data.slice(0, endIdx + 1);
   701→  const currentPrice = slice[slice.length - 1].close;
   702→  const closes = slice.map(d => d.close);
   703→  const noFib: [string, string] = ['—', ''];
   704→
   705→  // ── Method 1: Swing High/Low (Price Action) ──────────────────────────
   706→  for (const lookback of [3, 5, 7]) {
   707→    const swings = findSwingLevels(slice, lookback);
   708→    const tol = currentPrice * 0.005; // 0.5% tolerance
   709→
   710→    // Connect swing highs → resistance
   711→    const usedHigh = new Set<number>();
   712→    for (const h of swings.highs) {
   713→      if (usedHigh.has(h.idx)) continue;
   714→      let sum = h.price;
   715→      let count = 1;
   716→      for (const h2 of swings.highs) {
   717→        if (h2 === h || usedHigh.has(h2.idx)) continue;
   718→        if (Math.abs(h2.price - h.price) < tol) {
   719→          sum += h2.price;
   720→          count++;
   721→          usedHigh.add(h2.idx);
   722→        }
   723→      }
   724→      usedHigh.add(h.idx);
   725→      if (sum / count > currentPrice * 1.005) {
   726→        levels.push({ price: sum / count, method: 'Swing_High', type: 'resistance', ...noFib });
   727→      }
   728→    }
   729→
   730→    // Connect swing lows → support
   731→    const usedLow = new Set<number>();
   732→    for (const l of swings.lows) {
   733→      if (usedLow.has(l.idx)) continue;
   734→      let sum = l.price;
   735→      let count = 1;
   736→      for (const l2 of swings.lows) {
   737→        if (l2 === l || usedLow.has(l2.idx)) continue;
   738→        if (Math.abs(l2.price - l.price) < tol) {
   739→          sum += l2.price;
   740→          count++;
   741→          usedLow.add(l2.idx);
   742→        }
   743→      }
   744→      usedLow.add(l.idx);
   745→      if (sum / count < currentPrice * 0.995) {
   746→        levels.push({ price: sum / count, method: 'Swing_Low', type: 'support', ...noFib });
   747→      }
   748→    }
   749→  }
   750→
   751→  // ── Method 2: SMA Dynamic S/R ──────────────────────────────────────
   752→  for (const period of [20, 50, 100, 200]) {
   753→    const smaVal = sma(closes, period);
   754→    if (smaVal <= 0) continue;
   755→    const type: 'support' | 'resistance' = currentPrice > smaVal ? 'support' : 'resistance';
   756→    levels.push({ price: smaVal, method: `SMA_${period}`, type, ...noFib });
   757→  }
   758→
   759→  // ── Method 3: Bollinger Bands ───────────────────────────────────────
   760→  const bb = calcBollingerBands(closes);
   761→  if (bb.upper > 0 && bb.lower > 0) {
   762→    levels.push({ price: bb.upper, method: 'BB_Upper', type: 'resistance', ...noFib });
   763→    levels.push({ price: bb.lower, method: 'BB_Lower', type: 'support', ...noFib });
   764→  }
   765→
   766→  // ── Method 4: Fibonacci Retracement + Extension ─────────────────────
   767→  interface SwingPair { high: number; highIdx: number; low: number; lowIdx: number; range: number }
   768→  let bestSwing: SwingPair | null = null;
   769→
   770→  for (const lb of [3, 5, 7, 10, 15, 20]) {
   771→    const sw = findSwingLevels(slice, lb);
   772→    if (sw.highs.length === 0 || sw.lows.length === 0) continue;
   773→    for (const h of sw.highs) {
   774→      for (const l of sw.lows) {
   775→        if (Math.abs(h.idx - l.idx) < 3) continue;
   776→        const range = Math.abs(h.price - l.price);
   777→        if (range / currentPrice < 0.03) continue;
   778→        if (!bestSwing || range > bestSwing.range) {
   779→          bestSwing = {
   780→            high: Math.max(h.price, l.price),
   781→            highIdx: h.price > l.price ? h.idx : l.idx,
   782→            low: Math.min(h.price, l.price),
   783→            lowIdx: h.price < l.price ? l.idx : h.idx,
   784→            range,
   785→          };
   786→        }
   787→      }
   788→    }
   789→  }
   790→
   791→  if (bestSwing) {
   792→    const H = bestSwing.high;
   793→    const L = bestSwing.low;
   794→    const range = H - L;
   795→
   796→    for (const ratio of FIB_RETRACEMENTS) {
   797→      const price = H - ratio * range;
   798→      if (price <= 0) continue;
   799→      const rounded = roundToNice(price);
   800→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   801→      levels.push({
   802→        price: rounded, method: `Fibonacci_${ratio}`, type,
   803→        fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   804→      });
   805→    }
   806→
   807→    for (const extRatio of FIB_EXTENSIONS) {
   808→      const price = H + (extRatio - 1) * range;
   809→      if (price <= 0) continue;
   810→      const rounded = roundToNice(price);
   811→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   812→      levels.push({
   813→        price: rounded, method: `Fibonacci_${extRatio}`, type,
   814→        fibRatio: String(extRatio), fibLabel: FIB_LABELS[String(extRatio)] || '',
   815→      });
   816→    }
   817→
   818→    // Inverse extensions (below L)
   819→    for (const extRatio of FIB_EXTENSIONS) {
   820→      const price = L - (extRatio - 1) * range;
   821→      if (price <= 0) continue;
   822→      const rounded = roundToNice(price);
   823→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   824→      levels.push({
   825→        price: rounded, method: `Fibonacci_${-extRatio}`, type,
   826→        fibRatio: String(-extRatio), fibLabel: '',
   827→      });
   828→    }
   829→
   830→    // Secondary swing confluence (check overlap with secondary swings)
   831→    const secondarySwings: SwingPair[] = [];
   832→    for (const lb of [3, 5, 7]) {
   833→      const sw = findSwingLevels(slice, lb);
   834→      for (const h of sw.highs) {
   835→        for (const l of sw.lows) {
   836→          if (Math.abs(h.idx - l.idx) < 3) continue;
   837→          const r = Math.abs(h.price - l.price);
   838→          if (r / currentPrice < 0.05) continue;
   839→          if (bestSwing && Math.abs(r - bestSwing.range) / bestSwing.range < 0.1) continue;
   840→          secondarySwings.push({
   841→            high: Math.max(h.price, l.price), highIdx: h.price > l.price ? h.idx : l.idx,
   842→            low: Math.min(h.price, l.price), lowIdx: h.price < l.price ? l.idx : h.idx,
   843→            range: r,
   844→          });
   845→        }
   846→      }
   847→    }
   848→    // Secondary swings' fib levels are merged via confluence (1% rule in mergeLevels)
   849→    const confluenceTol = currentPrice * 0.005;
   850→    for (const sec of secondarySwings.slice(0, 3)) {
   851→      const sH = sec.high, sL = sec.low, sR = sH - sL;
   852→      for (const ratio of FIB_RETRACEMENTS) {
   853→        const price = roundToNice(sH - ratio * sR);
   854→        if (price <= 0) continue;
   855→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   856→        levels.push({
   857→          price, method: `Fibonacci_${ratio}`, type,
   858→          fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   859→        });
   860→      }
   861→      for (const extR of FIB_EXTENSIONS) {
   862→        const price = roundToNice(sH + (extR - 1) * sR);
   863→        if (price <= 0) continue;
   864→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   865→        levels.push({
   866→          price, method: `Fibonacci_${extR}`, type,
   867→          fibRatio: String(extR), fibLabel: FIB_LABELS[String(extR)] || '',
   868→        });
   869→      }
   870→    }
   871→  }
   872→
   873→  // ── Method 5: Volume Profile (VAP) ──────────────────────────────────
   874→  const dataHasVolume = slice.some(d => d.volume > 0);
   875→  if (dataHasVolume) {
   876→    let minP = Infinity, maxP = -Infinity;
   877→    for (const d of slice) {
   878→      if (d.low < minP) minP = d.low;
   879→      if (d.high > maxP) maxP = d.high;
   880→    }
   881→    const numBins = 40;
   882→    const binSize = Math.max((maxP - minP) / numBins, currentPrice * 0.001);
   883→    const vap = new Array(numBins).fill(0);
   884→    const binCenters: number[] = [];
   885→
   886→    for (let b = 0; b < numBins; b++) {
   887→      binCenters.push(minP + (b + 0.5) * binSize);
   888→    }
   889→
   890→    for (const d of slice) {
   891→      for (let b = 0; b < numBins; b++) {
   892→        if (d.low <= binCenters[b] && binCenters[b] <= d.high) {
   893→          vap[b] += d.volume;
   894→        }
   895→      }
   896→    }
   897→
   898→    const maxVAP = Math.max(...vap);
   899→    if (maxVAP > 0) {
   900→      // POC (Point of Control)
   901→      let pocIdx = 0;
   902→      for (let b = 1; b < numBins; b++) {
   903→        if (vap[b] > vap[pocIdx]) pocIdx = b;
   904→      }
   905→      const pocPrice = binCenters[pocIdx];
   906→      const pocType: 'support' | 'resistance' = pocPrice > currentPrice ? 'resistance' : 'support';
   907→      levels.push({ price: pocPrice, method: 'VAP_POC', type: pocType, ...noFib });
   908→
   909→      // HVN (High Volume Nodes)
   910→      for (let b = 0; b < numBins; b++) {
   911→        if (b === pocIdx) continue;
   912→        if (vap[b] > 0.7 * maxVAP) {
   913→          const hvnPrice = binCenters[b];
   914→          const hvnType: 'support' | 'resistance' = hvnPrice > currentPrice ? 'resistance' : 'support';
   915→          levels.push({ price: hvnPrice, method: 'VAP_HVN', type: hvnType, ...noFib });
   916→        }
   917→      }
   918→    }
   919→  }
   920→
   921→  // ── Method 6: Pivot Points (4 algorithms) ───────────────────────────
   922→  if (endIdx >= 1) {
   923→    const prev = slice[endIdx - 1]; // Previous completed bar
   924→    const pH = prev.high, pL = prev.low, pC = prev.close, pO = prev.open;
   925→
   926→    // Standard Pivot
   927→    {
   928→      const PP = (pH + pL + pC) / 3;
   929→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   930→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   931→      const R3 = pH + 2 * (PP - pL), S3 = pL - 2 * (pH - PP);
   932→ if (R1 > 0) levels.push({ price: R1, method: 'Pivot_R1', type: 'resistance', ...noFib });
   933→      if (R2 > 0) levels.push({ price: R2, method: 'Pivot_R2', type: 'resistance', ...noFib });
   934→      if (R3 > 0) levels.push({ price: R3, method: 'Pivot_R3', type: 'resistance', ...noFib });
   935→      if (S1 > 0) levels.push({ price: S1, method: 'Pivot_S1', type: 'support', ...noFib });
   936→      if (S2 > 0) levels.push({ price: S2, method: 'Pivot_S2', type: 'support', ...noFib });
   937→      if (S3 > 0) levels.push({ price: S3, method: 'Pivot_S3', type: 'support', ...noFib });
   938→    }
   939→
   940→    // Fibonacci Pivot
   941→    {
   942→      const PP = (pH + pL + pC) / 3;
   943→      const range = pH - pL;
   944→      for (const [mult, suffix] of [[0.382, '1'], [0.618, '2'], [1.0, '3']] as [number, string][]) {
   945→        const R = PP + mult * range;
   946→        const S = PP - mult * range;
   947→        if (R > 0) levels.push({ price: R, method: `Fib_Pivot_R${suffix}`, type: 'resistance', ...noFib });
   948→        if (S > 0) levels.push({ price: S, method: `Fib_Pivot_S${suffix}`, type: 'support', ...noFib });
   949→      }
   950→    }
   951→
   952→    // Woodie's Pivot
   953→    {
   954→      const PP = (pH + pL + 2 * pC) / 4;
   955→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   956→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   957→      if (R1 > 0) levels.push({ price: R1, method: 'Woodie_R1', type: 'resistance', ...noFib });
   958→      if (R2 > 0) levels.push({ price: R2, method: 'Woodie_R2', type: 'resistance', ...noFib });
   959→      if (S1 > 0) levels.push({ price: S1, method: 'Woodie_S1', type: 'support', ...noFib });
   960→      if (S2 > 0) levels.push({ price: S2, method: 'Woodie_S2', type: 'support', ...noFib });
   961→    }
   962→
   963→    // DeMark Pivot
   964→    {
   965→      const X = pC < pO ? pH + 2 * pL + pC : pC > pO ? 2 * pH + pL + pC : pH + pL + 2 * pC;
   966→      const PP = X / 4;
   967→      const R1 = X / 2 - pL;
   968→      const S1 = X / 2 - pH;
   969→      if (R1 > 0) levels.push({ price: R1, method: 'DeMark_R1', type: 'resistance', ...noFib });
   970→      if (S1 > 0) levels.push({ price: S1, method: 'DeMark_S1', type: 'support', ...noFib });
   971→    }
   972→  }
   973→
   974→  // ── Method 7: Psychological Levels ──────────────────────────────────
   975→  {
   976→    const step = psychStep(currentPrice);
   977→    const base = psychRound(currentPrice);
   978→    for (const mult of [-2, -1, 1, 2]) {
   979→      const price = base + mult * step;
   980→      if (price <= 0) continue;
   981→      const type: 'support' | 'resistance' = price > currentPrice ? 'resistance' : 'support';
   982→      levels.push({ price, method: 'Psychological', type, ...noFib });
   983→    }
   984→  }
   985→
   986→  return levels;
   987→}

   989→// ─── Helper: Merge Nearby Levels (1% Confluence) ──────────────────────────
   990→function mergeLevels(rawLevels: RawSRLevel[], currentPrice: number): MergedSRLevel[] {
   991→  if (rawLevels.length === 0) return [];
   992→
   993→  const tol = currentPrice * 0.01; // 1% tolerance
   994→  const merged: MergedSRLevel[] = [];
   995→  const used = new Set<number>();
   996→
   997→  for (let i = 0; i < rawLevels.length; i++) {
   998→    if (used.has(i)) continue;
   999→
  1000→    let sumPrice = rawLevels[i].price;
  1001→    let count = 1;
  1002→    const methods = new Set<string>();
  1003→    methods.add(rawLevels[i].method);
  1004→    let bestFibRatio = rawLevels[i].fibRatio;
  1005→    let bestFibLabel = rawLevels[i].fibLabel;
  1006→
  1007→    for (let j = i + 1; j < rawLevels.length; j++) {
  1008→      if (used.has(j)) continue;
  1009→      if (Math.abs(rawLevels[j].price - rawLevels[i].price) < tol) {
  1010→        sumPrice += rawLevels[j].price;
  1011→        count++;
  1012→        methods.add(rawLevels[j].method);
  1013→        used.add(j);
  1014→        // Prefer Fibonacci label if available
  1015→        if (rawLevels[j].fibRatio !== '—' && bestFibRatio === '—') {
  1016→          bestFibRatio = rawLevels[j].fibRatio;
  1017→          bestFibLabel = rawLevels[j].fibLabel;
  1018→        }
  1019→      }
  1020→    }
  1021→    used.add(i);
  1022→
  1023→    const avgPrice = sumPrice / count;
  1024→    const type: 'support' | 'resistance' = avgPrice >= currentPrice ? 'resistance' : 'support';
  1025→
  1026→    merged.push({
  1027→      price: avgPrice,
  1028→      type,
  1029→      methods: [...methods],
  1030→      overlapCount: methods.size,
  1031→      fibRatio: bestFibRatio,
  1032→      fibLabel: bestFibLabel,
  1033→      score: 0,
  1034→      strength: 1,
  1035→      touchCount: 0,
  1036→      volumeRatio: 0,
  1037→      freshness: 0,
  1038→      distancePercent: 0,
  1039→    });
  1040→  }
  1041→
  1042→  return merged;
  1043→}

  1045→// ─── Helper: Compute Level Features (5 ML features) ──────────────────────
  1046→function computeLevelFeatures(
  1047→  levelPrice: number,
  1048→  data: OHLCV[],
  1049→  endIdx: number,
  1050→  currentPrice: number,
  1051→  overlapCount: number,
  1052→): number[] {
  1053→  const lookback = Math.min(30, endIdx);
  1054→  const startIdx = endIdx - lookback;
  1055→  const touchTol = currentPrice * 0.002; // 0.2%
  1056→
  1057→  let touchCount = 0;
  1058→  let totalVolAtTouches = 0;
  1059→  let totalVol = 0;
  1060→  let barsSinceLastTouch = lookback;
  1061→
  1062→  for (let i = startIdx; i <= endIdx; i++) {
  1063→    const bar = data[i];
  1064→    totalVol += bar.volume;
  1065→
  1066→    if (bar.low <= levelPrice + touchTol && bar.high >= levelPrice - touchTol) {
  1067→      touchCount++;
  1068→      totalVolAtTouches += bar.volume;
  1069→      barsSinceLastTouch = endIdx - i;
  1070→    }
  1071→  }
  1072→
  1073→  const avgVol = lookback > 0 ? totalVol / lookback : 0;
  1074→  const volumeRatio = (touchCount > 0 && avgVol > 0) ? (totalVolAtTouches / touchCount) / avgVol : 1;
  1075→  const freshness = Math.max(0, Math.min(1, 1 - barsSinceLastTouch / 30));
  1076→  const distancePercent = Math.min(1, Math.abs(levelPrice - currentPrice) / currentPrice);
  1077→
  1078→  // Features: [touch_count, volume_ratio, overlap_count, freshness, distance_percent]
  1079→  return [touchCount, volumeRatio, overlapCount - 1, freshness, distancePercent];
  1080→}

  1082→// ─── Helper: Train S/R Scoring Weights (Ridge OLS) ───────────────────────
  1083→function trainSRWeights(data: OHLCV[]): number[] {
  1084→  const defaultWeights = [0.25, 0.20, 0.25, 0.15, 0.15];
  1085→
  1086→  if (data.length < 50) return defaultWeights;
  1087→
  1088→  const X: number[][] = [];
  1089→  const y: number[] = [];
  1090→
  1091→  for (let i = 40; i <= data.length - 11; i += 5) {
  1092→    const barPrice = data[i].close;
  1093→    const rawLevels = generateAllLevels(data, i);
  1094→    if (rawLevels.length === 0) continue;
  1095→    const merged = mergeLevels(rawLevels, barPrice);
  1096→
  1097→    for (const level of merged) {
  1098→      const features = computeLevelFeatures(level.price, data, i, barPrice, level.overlapCount);
  1099→      X.push(features);
  1100→
  1101→      // Label: did price return within 10 bars?
  1102→      let returned = false;
  1103→      const retTol = barPrice * 0.005;
  1104→      for (let j = i + 1; j <= Math.min(i + 10, data.length - 1); j++) {
  1105→        if (data[j].low <= level.price + retTol && data[j].high >= level.price - retTol) {
  1106→          returned = true;
  1107→          break;
  1108→        }
  1109→      }
  1110→      y.push(returned ? 1 : 0);
  1111→    }
  1112→  }
  1113→
  1114→  if (X.length < 20) return defaultWeights;
  1115→
  1116→  // Ridge OLS: w = (X^T X + λI)^(-1) X^T y
  1117→  const n = X.length;
  1118→  const p = 5;
  1119→  const lambda = 0.01;
  1120→
  1121→  const XtX: number[][] = Array.from({ length: p }, () => new Array(p).fill(0));
  1122→  for (let i = 0; i < n; i++) {
  1123→    for (let a = 0; a < p; a++) {
  1124→      for (let b = 0; b < p; b++) {
  1125→        XtX[a][b] += X[i][a] * X[i][b];
  1126→      }
  1127→    }
  1128→  }
  1129→  for (let a = 0; a < p; a++) XtX[a][a] += lambda;
  1130→
  1131→  const Xty: number[] = new Array(p).fill(0);
  1132→  for (let i = 0; i < n; i++) {
  1133→    for (let a = 0; a < p; a++) {
  1134→      Xty[a] += X[i][a] * y[i];
  1135→    }
  1136→  }
  1137→
  1138→  const weights = solveLinearSystem(XtX, Xty);
  1139→  if (!weights) return defaultWeights;
  1140→  if (weights.some(w => isNaN(w) || !isFinite(w))) return defaultWeights;
  1141→
  1142→  for (let i = 0; i < p; i++) {
  1143→    if (weights[i] < 0) weights[i] = 0;
  1144→  }
  1145→
  1146→  const sum = weights.reduce((a, b) => a + b, 0);
  1147→  if (sum <= 0) return defaultWeights;
  1148→
  1149→  return weights.map(w => w / sum);
  1150→}

  1152→// ─── Support / Resistance (7-Method ML-Based System) ─────────────────────
  1153→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
  1154→  resistances: number[];
  1155→  supports: number[];
  1156→  supportStrengths: LevelStrength[];
  1157→  resistanceStrengths: LevelStrength[];
  1158→  priceTargets: LevelStrength[];
  1159→} {
  1160→  if (data.length < 10) {
  1161→    return {
  1162→      resistances: [], supports: [],
  1163→      supportStrengths: [], resistanceStrengths: [],
  1164→      priceTargets: [],
  1165→    };
  1166→  }
  1167→
  1168→  // ── Step 1: Generate all levels from 7 methods ────────────────────────
  1169→  const rawLevels = generateAllLevels(data, data.length - 1);
  1170→
  1171→  // ── Step 2: Merge nearby levels (confluence within 1%) ────────────────
  1172→  let merged = mergeLevels(rawLevels, currentPrice);
  1173→
  1174→  // ── Step 3: Train ML weights on historical data ───────────────────────
  1175→  const weights = trainSRWeights(data);
  1176→
  1177→  // ── Step 4: Compute features & score each merged level ────────────────
  1178→  for (const level of merged) {
  1179→    const features = computeLevelFeatures(level.price, data, data.length - 1, currentPrice, level.overlapCount);
  1180→
  1181→    const touchScore = Math.min(features[0] / 5, 1);
  1182→    const volumeScore = Math.min(features[1] / 2, 1);
  1183→    const overlapScore = Math.min(features[2] * 2, 10) / 10;
  1184→    const freshnessScore = features[3];
  1185→    const distanceScore = 1 - features[4];
  1186→
  1187→    const powerScore = (
  1188→      weights[0] * touchScore +
  1189→      weights[1] * volumeScore +
  1190→      weights[2] * overlapScore +
  1191→      weights[3] * freshnessScore +
  1192→      weights[4] * distanceScore
  1193→    ) * 10;
  1194→
  1195→    level.score = Math.max(0, Math.min(10, powerScore));
  1196→    level.strength = Math.max(1, Math.min(10, Math.round(level.score)));
  1197→    level.touchCount = features[0];
  1198→    level.volumeRatio = features[1];
  1199→    level.freshness = features[3];
  1200→    level.distancePercent = features[4];
  1201→  }
  1202→
  1203→  // ── Step 5: Psychological rounding of all level prices ─────────────────
  1204→  for (const level of merged) {
  1205→    level.price = psychRound(level.price);
  1206→  }
  1207→
  1208→  // Re-classify types after rounding
  1209→  for (const level of merged) {
  1210→    level.type = level.price >= currentPrice ? 'resistance' : 'support';
  1211→  }
  1212→
  1213→  // Deduplicate after rounding (keep strongest at same price)
  1214→  {
  1215→    const deduped: MergedSRLevel[] = [];
  1216→    for (const level of merged) {
  1217→      const existing = deduped.find(d => Math.abs(d.price - level.price) < currentPrice * 0.001);
  1218→      if (existing) {
  1219→        if (level.score > existing.score) {
  1220→          Object.assign(existing, level);
  1221→        }
  1222→      } else {
  1223→        deduped.push(level);
  1224→      }
  1225→    }
  1226→    merged = deduped;
  1227→  }
  1228→
  1229→  // ── Step 6: Separate into supports and resistances ─────────────────────
  1230→  let supports = merged.filter(l => l.type === 'support');
  1231→  let resistances = merged.filter(l => l.type === 'resistance');
  1232→
  1233→  // ── Step 7: Distance filtering (strict 5-10% gap rules) ─────────────
  1234→  // Rules:
  1235→  //   1. Consecutive lines of same type: 5% ≤ gap ≤ 10%
  1236→  //   2. Nearest support ≤ 10% below price (can be < 5%)
  1237→  //   3. Nearest resistance ≤ 10% above price (can be < 5%)
  1238→
  1239→  const filterAndSelectLevels = (
  1240→    levels: MergedSRLevel[],
  1241→    type: 'support' | 'resistance'
  1242→  ): MergedSRLevel[] => {
  1243→    if (levels.length === 0) return [];
  1244→
  1245→    // Sort: supports descending (nearest first), resistances ascending (nearest first)
  1246→    const sorted = [...levels].sort((a, b) =>
  1247→      type === 'support' ? b.price - a.price : a.price - b.price
  1248→    );
  1249→
  1250→    // ── Rule 2/3: Nearest level must be within 10% of price ──
  1251→    const nearest = sorted[0];
  1252→    const distFromPrice = Math.abs(nearest.price - currentPrice) / currentPrice;
  1253→    if (distFromPrice > 0.10) {
  1254→      // Nearest level is too far — generate a synthetic level at ~5% from price
  1255→      const syntheticPrice = type === 'support'
  1256→        ? Math.round(currentPrice * 0.95 / psychStep(currentPrice)) * psychStep(currentPrice)
  1257→        : Math.round(currentPrice * 1.05 / psychStep(currentPrice)) * psychStep(currentPrice);
  1258→      const synthetic: MergedSRLevel = {
  1259→        price: syntheticPrice,
  1260→        type,
  1261→        score: 3,
  1262→        strength: 3,
  1263→        overlapCount: 1,
  1264→        methods: ['Psychological'],
  1265→        fibRatio: '—',
  1266→        fibLabel: '',
  1267→      };
  1268→      sorted.unshift(synthetic);
  1269→    }
  1270→
  1271→    // ── Rule 1: Select levels with 5-10% gaps between consecutive ──
  1272→    const result: MergedSRLevel[] = [sorted[0]];
  1273→
  1274→    for (let i = 1; i < sorted.length && result.length < 6; i++) {
  1275→      const prevPrice = result[result.length - 1].price;
  1276→      const currPrice = sorted[i].price;
  1277→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
  1278→
  1279→      if (gap < 0.05) {
  1280→        // Too close — keep the stronger one
  1281→        if (sorted[i].score > result[result.length - 1].score) {
  1282→          result[result.length - 1] = sorted[i];
  1283→        }
  1284→      } else if (gap <= 0.10) {
  1285→        // Perfect gap (5-10%) — accept
  1286→        result.push(sorted[i]);
  1287→      } else {
  1288→        // Gap > 10% — try to find an intermediate level
  1289→        let found = false;
  1290→        for (let j = i + 1; j < sorted.length; j++) {
  1291→          const midPrice = sorted[j].price;
  1292→          const gapFromPrev = Math.abs(midPrice - prevPrice) / prevPrice;
  1293→          const gapToNext = Math.abs(currPrice - midPrice) / midPrice;
  1294→          if (gapFromPrev >= 0.05 && gapFromPrev <= 0.10) {
  1295→            result.push(sorted[j]);
  1296→            found = true;
  1297→            // Skip levels that are too close to the inserted one
  1298→            i = j;
  1299→            break;
  1300→          }
  1301→        }
  1302→        if (!found) {
  1303→          // No intermediate found — generate a synthetic level at ~7.5% from prev
  1304→          const synthGap = 0.075;
  1305→          const syntheticPrice = type === 'support'
  1306→            ? Math.round((prevPrice * (1 - synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice)
  1307→            : Math.round((prevPrice * (1 + synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice);
  1308→          // Make sure it's not too close to next level
  1309→          const gapToNext = Math.abs(currPrice - syntheticPrice) / syntheticPrice;
  1310→          if (gapToNext >= 0.05) {
  1311→            const synthetic: MergedSRLevel = {
  1312→              price: syntheticPrice,
  1313→              type,
  1314→              score: 2,
  1315→              strength: 2,
  1316→              overlapCount: 1,
  1317→              methods: ['Psychological'],
  1318→              fibRatio: '—',
  1319→              fibLabel: '',
  1320→            };
  1321→            result.push(synthetic);
  1322→          }
  1323→        }
  1324→      }
  1325→    }
  1326→
  1327→    return result;
  1328→  };
  1329→
  1330→  const finalSupports = filterAndSelectLevels(supports, 'support');
  1331→  const finalResistances = filterAndSelectLevels(resistances, 'resistance');
  1332→
  1333→  // ── Step 9: Build LevelStrength objects ────────────────────────────────
  1334→  const toLevelStrength = (m: MergedSRLevel): LevelStrength => {
  1335→    let grade: string;
  1336→    if (m.score >= 8.5) grade = 'Very Strong';
  1337→    else if (m.score >= 7) grade = 'Strong';
  1338→    else if (m.score >= 5) grade = 'Moderate';
  1339→    else grade = 'Weak';
  1340→
  1341→    return {
  1342→      price: m.price,
  1343→      strength: m.strength,
  1344→      score: m.score,
  1345→      grade,
  1346→      isTarget: m.score >= 7,
  1347→      overlapCount: m.overlapCount,
  1348→      methods: m.methods,
  1349→      fibRatio: m.fibRatio,
  1350→      fibLabel: m.fibLabel,
  1351→    };
  1352→  };
  1353→
  1354→  const supportStrengths = finalSupports.map(toLevelStrength);
  1355→  const resistanceStrengths = finalResistances.map(toLevelStrength);
  1356→
  1357→  // Price targets: all levels with isTarget=true
  1358→  const priceTargets = [...supportStrengths, ...resistanceStrengths].filter(l => l.isTarget);
  1359→
  1360→  return {
  1361→    resistances: resistanceStrengths.map(l => l.price),
  1362→    supports: supportStrengths.map(l => l.price),
  1363→    supportStrengths,
  1364→    resistanceStrengths,
  1365→    priceTargets,
  1366→  };
  1367→}

  1406→  // Find nearest support below price
  1407→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
  1408→  if (belowPrice.length > 0) nearestS = belowPrice[0];
  1409→

  1411→}

  1413→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
  1414→// Returns true if bullish divergence detected, false if bearish, null if none
  1415→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
  1416→  const lookback = 20;
  1417→  const start = Math.max(0, barIdx - lookback);
  1418→  const slice = data.slice(start, barIdx + 1);
  1419→  if (slice.length < 10) return null;
  1420→
  1421→  const mid = Math.floor(slice.length / 2);
  1422→  const firstHalf = slice.slice(0, mid);
  1423→  const secondHalf = slice.slice(mid);
  1424→
  1425→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
  1426→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
  1427→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
  1428→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
  1429→
  1430→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
  1431→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
  1432→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
  1433→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
  1434→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
  1435→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
  1436→
  1437→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
  1438→    return false; // bearish divergence
  1439→  }
  1440→
  1441→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
  1442→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
  1443→
  1444→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
  1445→    return true; // bullish divergence
  1446→  }
  1447→
  1448→  return null;
  1449→}

  1479→  const macd = calcMACD(closes);
  1480→  const adxResult = calcADX(slice);
  1481→  const atrVal = calcATR(slice);
  1482→  const bb = calcBollingerBands(closes);
  1483→  const ma21 = sma(closes, 21);
  1484→  const ma100 = sma(closes, 100);
  1485→  const ema12 = emaCalc(closes, 12);
  1486→  const ema26 = emaCalc(closes, 26);
  1487→  const trendResult = calcTrend(closes, 21);
  1488→  const sr = simplifiedSR(data, barIdx);
  1489→
  1490→  // ── Layer 1: Raw Scores (s_*) ────────────────────────────────────────────
  1491→
  1492→  // s_rsi — RSI mapped to 0-1
  1493→  const s_rsi = rsiVal > 80 ? 0.95
  1494→    : rsiVal > 70 ? 0.85
  1495→    : rsiVal > 60 ? 0.70
  1496→    : rsiVal > 50 ? 0.55
  1497→    : rsiVal > 40 ? 0.45
  1498→    : rsiVal > 30 ? 0.30
  1499→    : rsiVal > 20 ? 0.15
  1500→    : 0.05;
  1501→
  1502→  // s_mfi — MFI mapped to 0-1 (same formula as RSI)
  1503→  const s_mfi = mfiVal > 80 ? 0.95
  1504→    : mfiVal > 70 ? 0.85
  1505→    : mfiVal > 60 ? 0.70
  1506→    : mfiVal > 50 ? 0.55
  1507→    : mfiVal > 40 ? 0.45
  1508→    : mfiVal > 30 ? 0.30
  1509→    : mfiVal > 20 ? 0.15
  1510→    : 0.05;
  1511→
  1512→  // s_cci — CCI mapped to 0-1
  1513→  const s_cci = cciVal > 200 ? 0.90
  1514→    : cciVal > 100 ? 0.75
  1515→    : cciVal > 0 ? 0.60
  1516→    : cciVal > -100 ? 0.40
  1517→    : cciVal > -200 ? 0.25
  1518→    : 0.10;
  1519→
  1520→  // s_adx — Directional bias
  1521→  const s_adx = (adxResult.diPlus - adxResult.diMinus + 100) / 200;
  1522→
  1523→  // s_macd — MACD histogram + signal combined
  1524→  const absMacdLine = Math.abs(macd.line) + 1;
  1525→  const s_macd = macd.line > macd.signal
  1526→    ? 0.5 + Math.min(0.5, macd.histogram / absMacdLine)
  1527→    : 0.5 - Math.min(0.5, Math.abs(macd.histogram) / absMacdLine);
  1528→
  1529→  // s_stoch — Stochastic %K mapped to 0-1
  1530→  const s_stoch = stoch.k / 100;
  1531→
  1532→  // s_bb — Bollinger Band position
  1533→  const bbRange = bb.upper - bb.lower;
  1534→  const s_bb = bbRange > eps
  1535→    ? clamp((price - bb.lower) / bbRange, 0, 1)
  1536→    : 0.5;
  1537→
  1538→  // s_ma21 — Price relative to 21-period SMA
  1539→  const s_ma21 = ma21 > eps
  1540→    ? clamp(0.5 + (price - ma21) / ma21, 0, 1)
  1541→    : 0.5;
  1542→
  1543→  // s_ma100 — Price relative to 100-period SMA
  1544→  const s_ma100 = ma100 > eps
  1545→    ? clamp(0.5 + (price - ma100) / ma100, 0, 1)
  1546→    : 0.5;
  1547→
  1548→  // s_ema — EMA12 vs EMA26 spread
  1549→  const s_ema = ema26 > eps
  1550→    ? clamp(0.5 + (ema12 - ema26) / ema26 * 5, 0, 1)
  1551→    : 0.5;
  1552→
  1553→  // s_atr — Inverse volatility (lower ATR = higher score)
  1554→  const s_atr = price > eps ? 1 - Math.min(atrVal / price * 10, 1) : 0.5;
  1555→
  1556→  // s_trend — R² + angle combination
  1557→  const s_trend = (trendResult.r2 * 0.7) +
  1558→    (Math.max(0, Math.min(1, (trendResult.angle + 45) / 90)) * 0.3);
  1559→
  1560→  // s_sr — Support/Resistance proximity
  1561→  const dR1 = (sr.R1 - price) / (price > eps ? price : 1);
  1562→  const dS1 = (price - sr.S1) / (price > eps ? price : 1);
  1563→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
  1564→
  1565→  // ── Layer 2: Corrections (f_*) ───────────────────────────────────────────
  1566→
  1567→  // f_rsi — RSI with momentum correction
  1568→  const rsiCurrent = rsiVal;
  1569→  const rsi5ago = barIdx >= 5
  1570→    ? calcRSI(closes.slice(0, barIdx - 5 + 1))
  1571→    : rsiCurrent;
  1572→  const f_rsi = clamp(s_rsi + (rsiCurrent - rsi5ago) / 100 * 0.08, 0, 1);
  1573→
  1574→  // f_mfi — MFI with momentum correction (skip when no volume)
  1575→  const mfiCurrent = mfiVal;
  1576→  const mfi5ago = (barIdx >= 5 && sliceHasVolume)
  1577→    ? calcMFI(data.slice(0, barIdx - 5 + 1))
  1578→    : mfiCurrent;
  1579→  const f_mfi = sliceHasVolume
  1580→    ? clamp(s_mfi + (mfiCurrent - mfi5ago) / 100 * 0.08, 0, 1)
  1581→    : 0.5; // neutral when no volume
  1582→
  1583→  // f_cci — CCI with momentum correction (scale: /200 instead of /100)
  1584→  const cciCurrent = cciVal;
  1585→  const cci5ago = barIdx >= 5
  1586→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
  1587→    : cciCurrent;
  1588→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
  1589→
  1590→  // f_macd — Same as s_macd (no additional correction)
  1591→  const f_macd = s_macd;
  1592→
  1593→  // f_stoch — Same as s_stoch
  1594→  const f_stoch = s_stoch;
  1595→
  1596→  // f_stochCross — Stochastic K/D crossover detection
  1597→  let f_stochCross = 0.1;
  1598→  if (barIdx >= 1) {
  1599→    const prevSlice = data.slice(0, barIdx);
  1600→    const prevStoch = calcStochastic(prevSlice);
  1601→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
  1602→      f_stochCross = 0.9; // K crossed above D (bullish)
  1603→    }
  1604→  }
  1605→
  1606→  // f_macdCross — MACD/Signal crossover detection
  1607→  let f_macdCross = 0.1;
  1608→  if (barIdx >= 1) {
  1609→    const prevCloses = closes.slice(0, -1);
  1610→    if (prevCloses.length >= 35) {
  1611→      const prevMacd = calcMACD(prevCloses);
  1612→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
  1613→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
  1614→      }
  1615→    }
  1616→  }
  1617→
  1618→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
  1619→  const divResult = detectDivergenceSimple(data, barIdx);
  1620→  const f_div = divResult === true ? 0.85 : 0.15;
  1621→
  1622→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
  1623→  return {
  1624→    rsi: f_rsi,
  1625→    mfi: f_mfi,
  1626→    cci: f_cci,
  1627→    adx: s_adx,
  1628→    macd: f_macd,
  1629→    stoch: f_stoch,
  1630→    bb: s_bb,
  1631→    ma21: s_ma21,
  1632→    ma100: s_ma100,
  1633→    ema: s_ema,
  1634→    atr: s_atr,
  1635→    trend: s_trend,
  1636→    sr: s_sr,
  1637→    stochCross: f_stochCross,
  1638→    macdCross: f_macdCross,
  1639→    div: f_div,
  1640→  };
  1641→}

  1644→// TRAINING DATA CREATION (Layer 0 — Data Preparation)
  1645→// ═══════════════════════════════════════════════════════════════════════════════
  1646→
  1647→/**
  1648→ * Creates supervised training data from historical OHLCV data.
  1649→ * For each bar, computes features and labels based on future price movement.
  1650→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
  1651→ * Samples every `sampleStep` bars to reduce computation.
  1652→ */
  1653→function createTrainingData(
  1654→  data: OHLCV[],
  1655→  horizon: number = 5,
  1656→  sampleStep: number = 5
  1657→): { X: number[][]; y: number[] } {
  1658→  const X: number[][] = [];
  1659→  const y: number[] = [];
  1660→
  1661→  // Start from horizon+30 to ensure enough data for indicators
  1662→  // End at data.length - horizon - 1 to have future bars for labeling
  1663→  // Sample every sampleStep bars for performance
  1664→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
  1665→    const features = computeFeaturesAtBar(data, i);
  1666→
  1667→    // Convert to array following FEATURES_ORDER
  1668→    const featureVec: number[] = [];
  1669→    let valid = true;
  1670→    for (const key of FEATURES_ORDER) {
  1671→      const val = features[key];
  1672→      if (val === undefined || val === null || Number.isNaN(val)) {
  1673→        valid = false;
  1674→        break;
  1675→      }
  1676→      featureVec.push(val);
  1677→    }
  1678→    if (!valid) continue;
  1679→
  1680→    // Label: 1 if future close > 1% above current close
  1681→    const currentClose = data[i].close;
  1682→    const futureClose = data[i + horizon].close;
  1683→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
  1684→
  1685→    X.push(featureVec);
  1686→    y.push(label);
  1687→  }
  1688→
  1689→  return { X, y };
  1690→}

  1692→// ═══════════════════════════════════════════════════════════════════════════════
  1693→// VDss GRAPH STRUCTURE (Layer 5 — State Machine)
  1694→// ═══════════════════════════════════════════════════════════════════════════════
  1695→
  1696→/**
  1697→ * VDss Graph: Each node represents a market state.
  1698→ * Edges represent transitions with types: 'up', 'down', 'pullback', 'risk', 'terminal'.
  1699→ * Terminal nodes map to scenario results (R1=strong bullish, R5=deep correction).
  1700→ */
  1701→const VDSS_GRAPH = {
  1702→  // outgoing edges from each node: [target, type]
  1703→  outgoing: {
  1704→    A: [['B', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1705→    B: [['C', 'up'], ['G', 'pullback'], ['A', 'risk']] as [string, string][],
  1706→    C: [['D', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1707→    D: [['E', 'up'], ['B', 'pullback'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1708→    E: [['F', 'up'], ['D', 'pullback'], ['G', 'pullback'], ['H', 'down'], ['R1', 'terminal']] as [string, string][],
  1709→    F: [['R1', 'terminal'], ['E', 'pullback'], ['D', 'down'], ['G', 'risk']] as [string, string][],
  1710→    G: [['B', 'up'], ['A', 'pullback'], ['H', 'down'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1711→    H: [['G', 'pullback'], ['I', 'down'], ['B', 'up'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1712→    I: [['H', 'pullback'], ['J', 'down'], ['G', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1713→    J: [['I', 'pullback'], ['K', 'down'], ['H', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1714→    K: [['J', 'pullback'], ['I', 'up'], ['L', 'risk'], ['R4', 'terminal'], ['R5', 'terminal']] as [string, string][],
  1715→    L: [['C', 'up'], ['H', 'pullback'], ['J', 'down'], ['K', 'risk'], ['R5', 'terminal']] as [string, string][],
  1716→  },
  1717→  // which result scenario each terminal node maps to
  1718→  terminalMap: { R1: 'R1', R2: 'R2', R3: 'R3', R4: 'R4', R5: 'R5' } as Record<string, string>,
  1719→  startNode: 'A' as string,
  1720→};
  1721→
  1722→// ═══════════════════════════════════════════════════════════════════════════════
  1723→// LAYER 5 — Edge Weight Calculation
  1724→// ═══════════════════════════════════════════════════════════════════════════════
  1725→
  1726→/**
  1727→ * Calculates transition edge weights for the VDss graph.
  1728→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
  1729→ * to determine the relative probability of up/down/pullback/risk transitions.
  1730→ */
  1731→function calculateEdgeWeights(
  1732→  bullConsensus: number,
  1733→  adx: number,
  1734→  mlModel: AdaptiveWeightModel
  1735→): { up: number; down: number; pullback: number; risk: number } {
  1736→  // Extract ML coefficient insights if model is trained
  1737→  let trend_coef = 1 / 3;
  1738→  let momentum_coef = 1 / 3;
  1739→  let volatility_coef = 1 / 3;
  1740→
  1741→  if (mlModel.isTrained && mlModel.weights) {
  1742→    const coefs = mlModel.getCoefficients();
  1743→    // Map feature indices to coefficient categories
  1744→    // FEATURES_ORDER: ['rsi','mfi','cci','adx','macd','stoch','bb','ma21','ma100','ema','atr','trend','sr','stochCross','macdCross','div']
  1745→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER
  1746→    const rsiIdx = 0;     // 'rsi' in FEATURES_ORDER
  1747→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
  1748→
  1749→    const absTrend = Math.abs(coefs[trendIdx] ?? 0);
  1750→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
  1751→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
  1752→
  1753→    const coefSum = absTrend + absRsi + absAtr;
  1754→    if (coefSum > 1e-10) {
  1755→      trend_coef = absTrend / coefSum;
  1756→      momentum_coef = absRsi / coefSum;
  1757→      volatility_coef = absAtr / coefSum;
  1758→    }
  1759→  }
  1760→
  1761→  // Base weights driven by bull consensus
  1762→  const up_base = bullConsensus * 0.7 + 0.15;
  1763→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
  1764→  const pullback_base = 0.25;
  1765→  const risk_base = 0.12 * (1.2 - adx / 100);
  1766→
  1767→  // Apply ML coefficient adjustments
  1768→  const up = clamp(
  1769→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
  1770→    0.05, 0.95
  1771→  );
  1772→  const down = clamp(
  1773→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
  1774→    0.05, 0.95
  1775→  );
  1776→  const pullback = clamp(
  1777→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
  1778→    0.05, 0.50
  1779→  );
  1780→  const risk = clamp(
  1781→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
  1782→    0.02, 0.30
  1783→  );
  1784→
  1785→  return { up, down, pullback, risk };
  1786→}
  1787→
  1788→// ═══════════════════════════════════════════════════════════════════════════════
  1789→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)
  1790→// ═══════════════════════════════════════════════════════════════════════════════
  1791→
  1792→/**
  1793→ * Performs iterative DFS from the start node through the VDss graph,
  1794→ * computing raw path probabilities, then calibrating them against
  1795→ * Layer 4 scenario probabilities.
  1796→ *
  1797→ * Returns:
  1798→ * - calibratedPaths: Map<pathString, probability>
  1799→ * - calibrationFactors: per-scenario multiplier used
  1800→ * - scenarioSums: raw probability sums per scenario before calibration
  1801→ */
  1802→function calculatePathProbabilities(
  1803→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
  1804→  scenarioProbs: Record<string, number>
  1805→): {
  1806→  calibratedPaths: Map<string, number>;
  1807→  calibrationFactors: Record<string, number>;
  1808→  scenarioSums: Record<string, number>;
  1809→} {
  1810→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
  1811→  const MAX_PATHS = 200;
  1812→
  1813→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
  1814→  type StackEntry = [string, string, number];
  1815→
  1816→  // Collect all terminal paths with raw probabilities
  1817→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
  1818→
  1819→  // Iterative DFS
  1820→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
  1821→
  1822→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
  1823→    const [node, pathStr, pathProb] = stack.pop()!;
  1824→
  1825→    const edges = outgoing[node as keyof typeof outgoing];
  1826→    if (!edges) continue;
  1827→
  1828→    // Normalize outgoing edge weights to probabilities for this node
  1829→    // Map edge types to their weights
  1830→    let totalWeight = 0;
  1831→    const edgeWeightsList: number[] = [];
  1832→    for (const [, edgeType] of edges) {
  1833→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
  1834→      edgeWeightsList.push(w);
  1835→      totalWeight += w;
  1836→    }
  1837→
  1838→    if (totalWeight < 1e-10) continue;
  1839→
  1840→    // Expand edges
  1841→    for (let i = 0; i < edges.length; i++) {
  1842→      const [target, edgeType] = edges[i];
  1843→      const edgeProb = edgeWeightsList[i] / totalWeight;
  1844→      const newProb = pathProb * edgeProb;
  1845→      const newPath = pathStr + '→' + target;
  1846→
  1847→      if (edgeType === 'terminal') {
  1848→        // Reached a terminal node
  1849→        const scenario = terminalMap[target] ?? 'R3';
  1850→        rawPaths.push({ path: newPath, scenario, prob: newProb });
  1851→      } else {
  1852→        // Continue DFS
  1853→        stack.push([target, newPath, newProb]);
  1854→      }
  1855→    }
  1856→  }
  1857→
  1858→  // Group raw probabilities by scenario
  1859→  const rawSums: Record<string, number> = { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0 };
  1860→  for (const rp of rawPaths) {
  1861→    rawSums[rp.scenario] = (rawSums[rp.scenario] ?? 0) + rp.prob;
  1862→  }
  1863→
  1864→  // Compute calibration factors: scenarioProbs[scenario] / rawSum[scenario]
  1865→  const calibrationFactors: Record<string, number> = {};
  1866→  for (const scenario of ['R1', 'R2', 'R3', 'R4', 'R5']) {
  1867→    const rawSum = rawSums[scenario] ?? 0;
  1868→    const targetProb = (scenarioProbs[scenario] ?? 20) / 100; // convert from percentage
  1869→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
  1870→  }
  1871→
  1872→  // Apply calibration to each path
  1873→  const calibratedPaths = new Map<string, number>();
  1874→  for (const rp of rawPaths) {
  1875→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
  1876→    calibratedPaths.set(rp.path, rp.prob * factor);
  1877→  }
  1878→
  1879→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
  1880→}
  1881→
  1882→// ═══════════════════════════════════════════════════════════════════════════════
  1883→// SCENARIO TARGET BUILDER — ATR-based, S/R levels, max 1 ATR range
  1884→// ═══════════════════════════════════════════════════════════════════════════════
  1885→
  1886→function buildBullishTargets(price: number, atr: number, resistances: number[]): Array<{min: number; max: number}> {
  1887→  // Filter resistances above price, take up to 4
  1888→  const above = resistances.filter(r => r > price);
  1889→  const step = Math.max(10, Math.round(atr * 0.1)); // rounding step
  1890→  const rnd = (n: number) => Math.round(n / step) * step;
  1891→
  1892→  const results: Array<{min: number; max: number}> = [];
  1893→  let prevMax = price;
  1894→
  1895→  for (let i = 0; i < 4; i++) {
  1896→    const base = above[i] ?? (price + (i + 1) * atr);
  1897→    // Ensure base is above previous target and above price
  1898→    const safeBase = Math.max(base, prevMax + step);
  1899→    let min = rnd(Math.max(safeBase - 0.25 * atr, prevMax + step));
  1900→    let max = rnd(Math.min(safeBase + 0.5 * atr, min + atr));
  1901→    if (max <= min) max = rnd(min + 0.5 * atr);
  1902→    if (min <= prevMax) min = rnd(prevMax + step);
  1903→    if (max <= min) max = rnd(min + 0.5 * atr);
  1904→    results.push({min, max});
  1905→    prevMax = max;
  1906→  }
  1907→  return results;
  1908→}
  1909→
  1910→function buildBearishTargets(price: number, atr: number, supports: number[]): Array<{min: number; max: number}> {
  1911→  // Filter supports below price, take up to 4
  1912→  const below = supports.filter(s => s < price);
  1913→  const step = Math.max(10, Math.round(atr * 0.1));
  1914→  const rnd = (n: number) => Math.round(n / step) * step;
  1915→
  1916→  const results: Array<{min: number; max: number}> = [];
  1917→  let prevMin = price;
  1918→
  1919→  for (let i = 0; i < 4; i++) {
  1920→    const base = below[i] ?? (price - (i + 1) * atr);
  1921→    const safeBase = Math.min(base, prevMin - step);
  1922→    let max = rnd(Math.min(safeBase + 0.25 * atr, prevMin - step));
  1923→    let min = rnd(Math.max(safeBase - 0.5 * atr, max - atr));
  1924→    if (min >= max) min = rnd(max - 0.5 * atr);
  1925→    if (max >= prevMin) max = rnd(prevMin - step);
  1926→    if (min >= max) min = rnd(max - 0.5 * atr);
  1927→    results.push({min, max});
