    50→  1764→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
    51→  1765→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
    52→  1766→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
    53→  1767→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
    54→  1768→
    55→  1769→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
    56→  1770→    return false; // bearish divergence
    57→  1771→  }
    58→  1772→
    59→  1773→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
    60→  1774→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
    61→  1775→
    62→  1776→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
    63→  1777→    return true; // bullish divergence
    64→  1778→  }
    65→  1779→
    66→  1780→  return null;
    67→  1781→}
    68→  1782→
    69→  1783→// ═══════════════════════════════════════════════════════════════════════════════
    70→  1784→// LAYER 1 & 2 — Feature Computation at a Historical Bar
    71→  1785→// ═══════════════════════════════════════════════════════════════════════════════
    72→  1786→
    73→  1787→/**
    74→  1788→ * Computes all 16 ML features at a specific historical bar.
    75→  1789→ * Uses data[0..barIdx] (inclusive) for all indicator calculations.
    76→  1790→ * Returns an object keyed by feature name matching FEATURES_ORDER.
    77→  1791→ */
    78→  1792→function computeFeaturesAtBar(data: OHLCV[], barIdx: number): Record<string, number> {
    79→  1793→  // Need at least 30 bars for meaningful indicators
    80→  1794→  if (barIdx < 30) {
    81→  1795→    const empty: Record<string, number> = {};
    82→  1796→    for (const key of FEATURES_ORDER) empty[key] = 0.5;
    83→  1797→    return empty;
    84→  1798→  }
    85→  1799→
    86→  1800→  const slice = data.slice(0, barIdx + 1);
    87→  1801→  const closes = slice.map(d => d.close);
    88→  1802→  const price = closes[closes.length - 1];
    89→  1803→  const eps = 1e-10;
    90→  1804→
    91→  1805→  // ── Compute all base indicators ───────────────────────────────────────────
    92→  1806→  const rsiVal = calcRSI(closes);
    93→  1807→  const sliceHasVolume = slice.some(d => d.volume > 0);
    94→  1808→  const mfiVal = sliceHasVolume ? calcMFI(slice) : 50; // neutral when no volume
    95→  1809→  const cciVal = calcCCI(slice);
    96→  1810→  const stoch = calcStochastic(slice);
    97→  1811→  const macd = calcMACD(closes);
    98→  1812→  const adxResult = calcADX(slice);
    99→  1813→  const atrVal = calcATR(slice);
   100→  1814→  const bb = calcBollingerBands(closes);
   101→  1815→  const ma21 = sma(closes, 21);
   102→  1816→  const ma100 = sma(closes, 100);
   103→  1817→  const ema12 = emaCalc(closes, 12);
   104→  1818→  const ema26 = emaCalc(closes, 26);
   105→  1819→  const trendResult = calcTrend(closes, 21);
   106→  1820→  const sr = simplifiedSR(data, barIdx);
   107→  1821→
   108→  1822→  // ── Layer 1: Raw Scores (s_*) ────────────────────────────────────────────
   109→  1823→
   110→  1824→  // s_rsi — RSI mapped to 0-1
   111→  1825→  const s_rsi = rsiVal > 80 ? 0.95
   112→  1826→    : rsiVal > 70 ? 0.85
   113→  1827→    : rsiVal > 60 ? 0.70
   114→  1828→    : rsiVal > 50 ? 0.55
   115→  1829→    : rsiVal > 40 ? 0.45
   116→  1830→    : rsiVal > 30 ? 0.30
   117→  1831→    : rsiVal > 20 ? 0.15
   118→  1832→    : 0.05;
   119→  1833→
   120→  1834→  // s_mfi — MFI mapped to 0-1 (same formula as RSI)
   121→  1835→  const s_mfi = mfiVal > 80 ? 0.95
   122→  1836→    : mfiVal > 70 ? 0.85
   123→  1837→    : mfiVal > 60 ? 0.70
   124→  1838→    : mfiVal > 50 ? 0.55
   125→  1839→    : mfiVal > 40 ? 0.45
   126→  1840→    : mfiVal > 30 ? 0.30
   127→  1841→    : mfiVal > 20 ? 0.15
   128→  1842→    : 0.05;
   129→  1843→
   130→  1844→  // s_cci — CCI mapped to 0-1
   131→  1845→  const s_cci = cciVal > 200 ? 0.90
   132→  1846→    : cciVal > 100 ? 0.75
   133→  1847→    : cciVal > 0 ? 0.60
   134→  1848→    : cciVal > -100 ? 0.40
   135→  1849→    : cciVal > -200 ? 0.25
   136→  1850→    : 0.10;
   137→  1851→
   138→  1852→  // s_adx — Directional bias
   139→  1853→  const s_adx = (adxResult.diPlus - adxResult.diMinus + 100) / 200;
   140→  1854→
   141→  1855→  // s_macd — MACD histogram + signal combined
   142→  1856→  const absMacdLine = Math.abs(macd.line) + 1;
   143→  1857→  const s_macd = macd.line > macd.signal
   144→  1858→    ? 0.5 + Math.min(0.5, macd.histogram / absMacdLine)
   145→  1859→    : 0.5 - Math.min(0.5, Math.abs(macd.histogram) / absMacdLine);
   146→  1860→
   147→  1861→  // s_stoch — Stochastic %K mapped to 0-1
   148→  1862→  const s_stoch = stoch.k / 100;
   149→  1863→
   150→  1864→  // s_bb — Bollinger Band position
   151→  1865→  const bbRange = bb.upper - bb.lower;
   152→  1866→  const s_bb = bbRange > eps
   153→  1867→    ? clamp((price - bb.lower) / bbRange, 0, 1)
   154→  1868→    : 0.5;
   155→  1869→
   156→  1870→  // s_ma21 — Price relative to 21-period SMA
   157→  1871→  const s_ma21 = ma21 > eps
   158→  1872→    ? clamp(0.5 + (price - ma21) / ma21, 0, 1)
   159→  1873→    : 0.5;
   160→  1874→
   161→  1875→  // s_ma100 — Price relative to 100-period SMA
   162→  1876→  const s_ma100 = ma100 > eps
   163→  1877→    ? clamp(0.5 + (price - ma100) / ma100, 0, 1)
   164→  1878→    : 0.5;
   165→  1879→
   166→  1880→  // s_ema — EMA12 vs EMA26 spread
   167→  1881→  const s_ema = ema26 > eps
   168→  1882→    ? clamp(0.5 + (ema12 - ema26) / ema26 * 5, 0, 1)
   169→  1883→    : 0.5;
   170→  1884→
   171→  1885→  // s_atr — Inverse volatility (lower ATR = higher score)
   172→  1886→  const s_atr = price > eps ? 1 - Math.min(atrVal / price * 10, 1) : 0.5;
   173→  1887→
   174→  1888→  // s_trend — R² + angle combination
   175→  1889→  const s_trend = (trendResult.r2 * 0.7) +
   176→  1890→    (Math.max(0, Math.min(1, (trendResult.angle + 45) / 90)) * 0.3);
   177→  1891→
   178→  1892→  // s_sr — Support/Resistance proximity
   179→  1893→  const dR1 = (sr.R1 - price) / (price > eps ? price : 1);
   180→  1894→  const dS1 = (price - sr.S1) / (price > eps ? price : 1);
   181→  1895→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
   182→  1896→
   183→  1897→  // ── Layer 2: Corrections (f_*) ───────────────────────────────────────────
   184→  1898→
   185→  1899→  // f_rsi — RSI with momentum correction
   186→  1900→  const rsiCurrent = rsiVal;
   187→  1901→  const rsi5ago = barIdx >= 5
   188→  1902→    ? calcRSI(closes.slice(0, barIdx - 5 + 1))
   189→  1903→    : rsiCurrent;
   190→  1904→  const f_rsi = clamp(s_rsi + (rsiCurrent - rsi5ago) / 100 * 0.08, 0, 1);
   191→  1905→
   192→  1906→  // f_mfi — MFI with momentum correction (skip when no volume)
   193→  1907→  const mfiCurrent = mfiVal;
   194→  1908→  const mfi5ago = (barIdx >= 5 && sliceHasVolume)
   195→  1909→    ? calcMFI(data.slice(0, barIdx - 5 + 1))
   196→  1910→    : mfiCurrent;
   197→  1911→  const f_mfi = sliceHasVolume
   198→  1912→    ? clamp(s_mfi + (mfiCurrent - mfi5ago) / 100 * 0.08, 0, 1)
   199→  1913→    : 0.5; // neutral when no volume
   200→  1914→
   201→  1915→  // f_cci — CCI with momentum correction (scale: /200 instead of /100)
   202→  1916→  const cciCurrent = cciVal;
   203→  1917→  const cci5ago = barIdx >= 5
   204→  1918→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
   205→  1919→    : cciCurrent;
   206→  1920→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
   207→  1921→
   208→  1922→  // f_macd — Same as s_macd (no additional correction)
   209→  1923→  const f_macd = s_macd;
   210→  1924→
   211→  1925→  // f_stoch — Same as s_stoch
   212→  1926→  const f_stoch = s_stoch;
   213→  1927→
   214→  1928→  // f_stochCross — Stochastic K/D crossover detection
   215→  1929→  let f_stochCross = 0.1;
   216→  1930→  if (barIdx >= 1) {
   217→  1931→    const prevSlice = data.slice(0, barIdx);
   218→  1932→    const prevStoch = calcStochastic(prevSlice);
   219→  1933→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
   220→  1934→      f_stochCross = 0.9; // K crossed above D (bullish)
   221→  1935→    }
   222→  1936→  }
   223→  1937→
   224→  1938→  // f_macdCross — MACD/Signal crossover detection
   225→  1939→  let f_macdCross = 0.1;
   226→  1940→  if (barIdx >= 1) {
   227→  1941→    const prevCloses = closes.slice(0, -1);
   228→  1942→    if (prevCloses.length >= 35) {
   229→  1943→      const prevMacd = calcMACD(prevCloses);
   230→  1944→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
   231→  1945→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
   232→  1946→      }
   233→  1947→    }
   234→  1948→  }
   235→  1949→
   236→  1950→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
   237→  1951→  const divResult = detectDivergenceSimple(data, barIdx);
   238→  1952→  const f_div = divResult === true ? 0.85 : 0.15;
   239→  1953→
   240→  1954→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
   241→  1955→  return {
   242→  1956→    rsi: f_rsi,
   243→  1957→    mfi: f_mfi,
   244→  1958→    cci: f_cci,
   245→  1959→    adx: s_adx,
   246→  1960→    macd: f_macd,
   247→  1961→    stoch: f_stoch,
   248→  1962→    bb: s_bb,
   249→  1963→    ma21: s_ma21,
   250→  1964→    ma100: s_ma100,
   251→  1965→    ema: s_ema,
   252→  1966→    atr: s_atr,
   253→  1967→    trend: s_trend,
   254→  1968→    sr: s_sr,
   255→  1969→    stochCross: f_stochCross,
   256→  1970→    macdCross: f_macdCross,
   257→  1971→    div: f_div,
   258→  1972→  };
   259→  1973→}
   260→  1974→
   261→  1975→// ═══════════════════════════════════════════════════════════════════════════════
   262→  1976→// TRAINING DATA CREATION (Layer 0 — Data Preparation)
   263→  1977→// ═══════════════════════════════════════════════════════════════════════════════
   264→  1978→
   265→  1979→/**
   266→  1980→ * Creates supervised training data from historical OHLCV data.
   267→  1981→ * For each bar, computes features and labels based on future price movement.
   268→  1982→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
   269→  1983→ * Samples every `sampleStep` bars to reduce computation.
   270→  1984→ */
   271→  1985→function createTrainingData(
   272→  1986→  data: OHLCV[],
   273→  1987→  horizon: number = 5,
   274→  1988→  sampleStep: number = 5
   275→  1989→): { X: number[][]; y: number[] } {
   276→  1990→  const X: number[][] = [];
   277→  1991→  const y: number[] = [];
   278→  1992→
   279→  1993→  // Start from horizon+30 to ensure enough data for indicators
   280→  1994→  // End at data.length - horizon - 1 to have future bars for labeling
   281→  1995→  // Sample every sampleStep bars for performance
   282→  1996→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
   283→  1997→    const features = computeFeaturesAtBar(data, i);
   284→  1998→
   285→  1999→    // Convert to array following FEATURES_ORDER
   286→  2000→    const featureVec: number[] = [];
   287→  2001→    let valid = true;
   288→  2002→    for (const key of FEATURES_ORDER) {
   289→  2003→      const val = features[key];
   290→  2004→      if (val === undefined || val === null || Number.isNaN(val)) {
   291→  2005→        valid = false;
   292→  2006→        break;
   293→  2007→      }
   294→  2008→      featureVec.push(val);
   295→  2009→    }
   296→  2010→    if (!valid) continue;
   297→  2011→
   298→  2012→    // Label: 1 if future close > 1% above current close
   299→  2013→    const currentClose = data[i].close;
   300→  2014→    const futureClose = data[i + horizon].close;
   301→  2015→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
   302→  2016→
   303→  2017→    X.push(featureVec);
   304→  2018→    y.push(label);
   305→  2019→  }
   306→  2020→
   307→  2021→  return { X, y };
   308→  2022→}
   309→  2023→
   310→  2024→// ═══════════════════════════════════════════════════════════════════════════════
   311→  2025→// VDss GRAPH STRUCTURE (Layer 5 — State Machine)
   312→  2026→// ═══════════════════════════════════════════════════════════════════════════════
   313→  2027→
   314→  2028→/**
   315→  2029→ * VDss Graph: Each node represents a market state.
   316→  2030→ * Edges represent transitions with types: 'up', 'down', 'pullback', 'risk', 'terminal'.
   317→  2031→ * Terminal nodes map to scenario results (R1=strong bullish, R5=deep correction).
   318→  2032→ */
   319→  2033→const VDSS_GRAPH = {
   320→  2034→  // outgoing edges from each node: [target, type]
   321→  2035→  // v11: 9 scenario terminals — R1(most bullish) to R9(most bearish)
   322→  2036→  outgoing: {
   323→  2037→    A: [['B', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
   324→  2038→    B: [['C', 'up'], ['G', 'pullback'], ['A', 'risk'], ['R3', 'terminal']] as [string, string][],
   325→  2039→    C: [['D', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
   326→  2040→    D: [['E', 'up'], ['B', 'pullback'], ['G', 'pullback'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
   327→  2041→    E: [['F', 'up'], ['D', 'pullback'], ['G', 'pullback'], ['H', 'down'], ['R2', 'terminal']] as [string, string][],
   328→  2042→    F: [['R1', 'terminal'], ['E', 'pullback'], ['D', 'down'], ['G', 'risk']] as [string, string][],
   329→  2043→    G: [['B', 'up'], ['A', 'pullback'], ['H', 'down'], ['L', 'risk'], ['R4', 'terminal']] as [string, string][],
   330→  2044→    H: [['G', 'pullback'], ['I', 'down'], ['B', 'up'], ['L', 'risk'], ['R5', 'terminal']] as [string, string][],
   331→  2045→    I: [['H', 'pullback'], ['J', 'down'], ['G', 'up'], ['L', 'risk'], ['R6', 'terminal']] as [string, string][],
   332→  2046→    J: [['I', 'pullback'], ['K', 'down'], ['H', 'up'], ['L', 'risk'], ['R7', 'terminal']] as [string, string][],
   333→  2047→    K: [['J', 'pullback'], ['I', 'up'], ['L', 'risk'], ['R8', 'terminal'], ['R9', 'terminal']] as [string, string][],
   334→  2048→    L: [['C', 'up'], ['H', 'pullback'], ['J', 'down'], ['K', 'risk'], ['R9', 'terminal']] as [string, string][],
   335→  2049→  },
   336→  2050→  // which result scenario each terminal node maps to
   337→  2051→  terminalMap: { R1: 'R1', R2: 'R2', R3: 'R3', R4: 'R4', R5: 'R5', R6: 'R6', R7: 'R7', R8: 'R8', R9: 'R9' } as Record<string, string>,
   338→  2052→  startNode: 'A' as string,
   339→  2053→};
   340→  2054→
   341→  2055→// ═══════════════════════════════════════════════════════════════════════════════
   342→  2056→// LAYER 5 — Edge Weight Calculation
   343→  2057→// ═══════════════════════════════════════════════════════════════════════════════
   344→  2058→
   345→  2059→/**
   346→  2060→ * Calculates transition edge weights for the VDss graph.
   347→  2061→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
   348→  2062→ * to determine the relative probability of up/down/pullback/risk transitions.
   349→  2063→ */
   350→  2064→function calculateEdgeWeights(
   351→  2065→  bullConsensus: number,
   352→  2066→  adx: number,
   353→  2067→  mlModel: AdaptiveWeightModel
   354→  2068→): { up: number; down: number; pullback: number; risk: number } {
   355→  2069→  // Extract ML coefficient insights if model is trained
   356→  2070→  let trend_coef = 1 / 3;
   357→  2071→  let momentum_coef = 1 / 3;
   358→  2072→  let volatility_coef = 1 / 3;
   359→  2073→
   360→  2074→  if (mlModel.isTrained && mlModel.weights) {
   361→  2075→    const coefs = mlModel.getCoefficients();
   362→  2076→    // Map feature indices to coefficient categories
   363→  2077→    // FEATURES_ORDER: ['rsi','mfi','cci','adx','macd','stoch','bb','ma21','ma100','ema','atr','trend','sr','stochCross','macdCross','div']
   364→  2078→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER
   365→  2079→    const rsiIdx = 0;     // 'rsi' in FEATURES_ORDER
   366→  2080→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
   367→  2081→
   368→  2082→    const absTrend = Math.abs(coefs[trendIdx] ?? 0);
   369→  2083→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
   370→  2084→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
   371→  2085→
   372→  2086→    const coefSum = absTrend + absRsi + absAtr;
   373→  2087→    if (coefSum > 1e-10) {
   374→  2088→      trend_coef = absTrend / coefSum;
   375→  2089→      momentum_coef = absRsi / coefSum;
   376→  2090→      volatility_coef = absAtr / coefSum;
   377→  2091→    }
   378→  2092→  }
   379→  2093→
   380→  2094→  // Base weights driven by bull consensus
   381→  2095→  const up_base = bullConsensus * 0.7 + 0.15;
   382→  2096→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
   383→  2097→  const pullback_base = 0.25;
   384→  2098→  const risk_base = 0.12 * (1.2 - adx / 100);
   385→  2099→
   386→  2100→  // Apply ML coefficient adjustments
   387→  2101→  const up = clamp(
   388→  2102→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
   389→  2103→    0.05, 0.95
   390→  2104→  );
   391→  2105→  const down = clamp(
   392→  2106→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
   393→  2107→    0.05, 0.95
   394→  2108→  );
   395→  2109→  const pullback = clamp(
   396→  2110→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
   397→  2111→    0.05, 0.50
   398→  2112→  );
   399→  2113→  const risk = clamp(
   400→  2114→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
   401→  2115→    0.02, 0.30
   402→  2116→  );
   403→  2117→
   404→  2118→  return { up, down, pullback, risk };
   405→  2119→}
   406→  2120→
   407→  2121→// ═══════════════════════════════════════════════════════════════════════════════
   408→  2122→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)
   409→  2123→// ═══════════════════════════════════════════════════════════════════════════════
   410→  2124→
   411→  2125→/**
   412→  2126→ * Performs iterative DFS from the start node through the VDss graph,
   413→  2127→ * computing raw path probabilities, then calibrating them against
   414→  2128→ * Layer 4 scenario probabilities.
   415→  2129→ *
   416→  2130→ * Returns:
   417→  2131→ * - calibratedPaths: Map<pathString, probability>
   418→  2132→ * - calibrationFactors: per-scenario multiplier used
   419→  2133→ * - scenarioSums: raw probability sums per scenario before calibration
   420→  2134→ */
   421→  2135→function calculatePathProbabilities(
   422→  2136→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
   423→  2137→  scenarioProbs: Record<string, number>
   424→  2138→): {
   425→  2139→  calibratedPaths: Map<string, number>;
   426→  2140→  calibrationFactors: Record<string, number>;
   427→  2141→  scenarioSums: Record<string, number>;
   428→  2142→} {
   429→  2143→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
   430→  2144→  const MAX_PATHS = 200;
   431→  2145→
   432→  2146→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
   433→  2147→  type StackEntry = [string, string, number];
   434→  2148→
   435→  2149→  // Collect all terminal paths with raw probabilities
   436→  2150→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
   437→  2151→
   438→  2152→  // Iterative DFS
   439→  2153→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
   440→  2154→
   441→  2155→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
   442→  2156→    const [node, pathStr, pathProb] = stack.pop()!;
   443→  2157→
   444→  2158→    const edges = outgoing[node as keyof typeof outgoing];
   445→  2159→    if (!edges) continue;
   446→  2160→
   447→  2161→    // Normalize outgoing edge weights to probabilities for this node
   448→  2162→    // Map edge types to their weights
   449→  2163→    let totalWeight = 0;
   450→  2164→    const edgeWeightsList: number[] = [];
   451→  2165→    for (const [, edgeType] of edges) {
   452→  2166→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
   453→  2167→      edgeWeightsList.push(w);
   454→  2168→      totalWeight += w;
   455→  2169→    }
   456→  2170→
   457→  2171→    if (totalWeight < 1e-10) continue;
   458→  2172→
   459→  2173→    // Expand edges
   460→  2174→    for (let i = 0; i < edges.length; i++) {
   461→  2175→      const [target, edgeType] = edges[i];
   462→  2176→      const edgeProb = edgeWeightsList[i] / totalWeight;
   463→  2177→      const newProb = pathProb * edgeProb;
   464→  2178→      const newPath = pathStr + '→' + target;
   465→  2179→
   466→  2180→      if (edgeType === 'terminal') {
   467→  2181→        // Reached a terminal node
   468→  2182→        const scenario = terminalMap[target] ?? 'R3';
   469→  2183→        rawPaths.push({ path: newPath, scenario, prob: newProb });
   470→  2184→      } else {
   471→  2185→        // Continue DFS
   472→  2186→        stack.push([target, newPath, newProb]);
   473→  2187→      }
   474→  2188→    }
   475→  2189→  }
   476→  2190→
   477→  2191→  // Group raw probabilities by scenario
   478→  2192→  const rawSums: Record<string, number> = { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0, R6: 0, R7: 0, R8: 0, R9: 0 };
   479→  2193→  for (const rp of rawPaths) {
   480→  2194→    rawSums[rp.scenario] = (rawSums[rp.scenario] ?? 0) + rp.prob;
   481→  2195→  }
   482→  2196→
   483→  2197→  // Compute calibration factors: scenarioProbs[scenario] / rawSum[scenario]
   484→  2198→  const calibrationFactors: Record<string, number> = {};
   485→  2199→  for (const scenario of ['R1', 'R2', 'R3', 'R4', 'R5', 'R6', 'R7', 'R8', 'R9']) {
   486→  2200→    const rawSum = rawSums[scenario] ?? 0;
   487→  2201→    const targetProb = (scenarioProbs[scenario] ?? 20) / 100; // convert from percentage
   488→  2202→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
   489→  2203→  }
   490→  2204→
   491→  2205→  // Apply calibration to each path
   492→  2206→  const calibratedPaths = new Map<string, number>();
   493→  2207→  for (const rp of rawPaths) {
   494→  2208→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
   495→  2209→    calibratedPaths.set(rp.path, rp.prob * factor);
   496→  2210→  }
   497→  2211→
   498→  2212→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
   499→  2213→}

   505→  2219→function buildBullishTargets(price: number, atr: number, resistances: number[]): Array<{min: number; max: number}> {

   513→  2227→
   514→  2228→  for (let i = 0; i < 4; i++) {
   515→  2229→    const base = above[i] ?? (price + (i + 1) * 0.25 * atr);

   517→  2231→    const safeBase = Math.max(base, prevMax + 1);
   518→  2232→    let min = rnd(Math.max(safeBase - 0.25 * atr, prevMax + 1));
   519→  2233→    let max = rnd(Math.min(safeBase + 0.5 * atr, min + atr));
   520→  2234→    if (max <= min) max = rnd(min + 0.25 * atr);
   521→  2235→    if (min <= prevMax) min = rnd(prevMax + 1);
   522→  2236→    if (max <= min) max = rnd(min + 0.25 * atr);
   523→  2237→    // Enforce max 1 ATR range
   524→  2238→    if (max - min > atr) max = rnd(min + atr);
   525→  2239→    results.push({min, max});
   526→  2240→    prevMax = max;
   527→  2241→  }
   528→  2242→  return results;
