   700→function scoreSchool(s: SchoolDef, inp: NarrativeInput): number {
   701→  let s_ = 0;
   702→  const { trendDirection, rsi, adx, bbPosition, diPlus, diMinus } = inp;
   703→  switch (s.id) {
   704→    case 'trend': s_ = adx > 25 ? 8 : adx > 15 ? 5 : 2; break;
   705→    case 'classical': s_ = (inp.nearResistance || inp.nearSupport) ? 9 : 5; break;
   706→    case 'candlestick': s_ = (inp.nearResistance || inp.nearSupport) ? 8 : 4; break;
   707→    case 'oscillator': s_ = (rsi > 70 || rsi < 30) ? 9 : Math.abs(rsi - 50) > 15 ? 6 : 3; break;
   708→    case 'harmonic': s_ = (inp.nearResistance || inp.nearSupport) ? 7 : 3; break;
   709→    case 'elliott': s_ = inp.trendR2 > 0.7 ? 9 : inp.trendR2 > 0.4 ? 5 : 2; break;
   710→    case 'volume': s_ = inp.hasVolume ? 8 : 2; break;
   711→    case 'sr': s_ = (inp.nearResistance || inp.nearSupport) ? 9 : 4; break;
   712→    case 'volatility': s_ = inp.atr / inp.price > 0.03 ? 8 : inp.atr / inp.price > 0.01 ? 5 : 2; break;
   713→    case 'hybrid': s_ = 6; break;
   714→    default: s_ = 4;
   715→  }
   716→  if (s.id === 'trend' && trendDirection === 'up') s_ += 1;
   717→  if (s.id === 'trend' && diPlus > diMinus) s_ += 1;
   718→  return s_;
   719→}

   721→function scoreStyle(st: StyleDef, inp: NarrativeInput): number {
   722→  let s_ = 4;
   723→  const { trendDirection, adx, rsi, dominantScenarioKey } = inp;
   724→  switch (st.id) {
   725→    case 'executive': s_ = adx > 25 ? 7 : 4; break;
   726→    case 'technical': s_ = 6; break;
   727→    case 'trading': s_ = (dominantScenarioKey === 'R1' || dominantScenarioKey === 'R2') ? 9 : 4; break;
   728→    case 'forecast': s_ = 6; break;
   729→    case 'educational': s_ = 5; break;
   730→  }
   731→  return s_;
   732→}

   734→function scoreTone(t: ToneDef, inp: NarrativeInput): number {
   735→  let s_ = 3;
   736→  const { trendDirection, adx, rsi, dominantScenarioKey, macdHist } = inp;
   737→  switch (t.id) {
   738→    case 'conservative': s_ = 5; if (rsi > 65 || rsi < 35) s_ += 3; break;
   739→    case 'aggressive': s_ = 3; if (adx > 30 && (rsi > 60 || rsi < 40)) s_ += 4; break;
   740→    case 'balanced': s_ = 6; break;
   741→    case 'alert': s_ = 4; if (rsi > 75 || rsi < 25) s_ += 3; break;
   742→    case 'optimistic': s_ = 3; if (trendDirection === 'up' && macdHist > 0) s_ += 4; break;
   743→    case 'realistic': s_ = 6; break;
   744→  }
   745→  if (dominantScenarioKey === 'R1' || dominantScenarioKey === 'R2') s_ += 2;
   746→  return s_ + 1;
   747→}

   749→interface Scored<T> { item: T; score: number; }
   750→function scoreAll<T>(items: T[], inp: NarrativeInput, fn: (item: T, inp: NarrativeInput) => number): Scored<T>[] {
   751→  return items.map(item => ({ item, score: fn(item, inp) })).sort((a, b) => b.score - a.score);
   752→}

   773→export function buildNarrativeInput(body: {
   774→  currentPrice: number; trendDirection: string; rsi: number; adx: number;
   775→  stochK: number; stochD: number; macdLine: number; macdSignal: number; macdHist: number;
   776→  diPlus: number; diMinus: number; obv: number; hasVolume: boolean;
   777→  cci: number; mfi: number; atr: number; bollingerUpper: number; bollingerMiddle: number; bollingerLower: number;
   778→  trendAngle: number; trendR2: number; overallSignal: string;
   779→  scenarios: Record<string, { probability: number }>;
   780→  resistances: number[]; supports: number[]; ma21: number; ma100: number;
   781→}, isV10Scenarios: boolean = false): NarrativeInput {
   782→  const bbRange = body.bollingerUpper - body.bollingerLower;
   783→  const bbPosition = bbRange > 0 ? Math.max(0, Math.min(100, ((body.currentPrice - body.bollingerLower) / bbRange) * 100)) : 50;
   784→  const R1 = body.resistances[0] || body.currentPrice * 1.05;
   785→  const S1 = body.supports[0] || body.currentPrice * 0.95;
   786→  // Detect scenario keys from the data itself
   787→  const allScenarioKeys = (isV10Scenarios || Object.keys(body.scenarios).some(k => k.startsWith('S')))
   788→    ? ['S1', 'S2', 'S3', 'S4', 'S5', 'S6', 'S7', 'S8', 'S9'] as const
   789→    : ['R1', 'R2', 'R3', 'R4', 'R5', 'R6', 'R7', 'R8', 'R9'] as const;
   790→  let dominantKey = allScenarioKeys[0]; let dominantProb = 0;
   791→  for (const k of allScenarioKeys) {
   792→    const p = body.scenarios[k]?.probability ?? 0;
   793→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
   794→  }
   795→  return {
   796→    price: body.currentPrice, trendDirection: body.trendDirection, rsi: body.rsi, adx: body.adx,
   797→    stochK: body.stochK, stochD: body.stochD, macdLine: body.macdLine, macdSignal: body.macdSignal,

   800→    bbPosition, bbUpper: body.bollingerUpper, bbLower: body.bollingerLower,
   801→    dominantScenarioKey: dominantKey,
   802→    scenarioProbabilities: Object.fromEntries(allScenarioKeys.map(k => [k, body.scenarios[k]?.probability ?? 0])),
   803→    nearResistance: R1 > 0 ? Math.abs(body.currentPrice - R1) / R1 < 0.03 : false,
   804→    nearSupport: S1 > 0 ? Math.abs(body.currentPrice - S1) / S1 < 0.03 : false,
   805→    priceVsMa21: body.currentPrice > body.ma21 ? 'above' : 'below',
   806→    priceVsMa100: body.currentPrice > body.ma100 ? 'above' : 'below',
   807→    trendR2: body.trendR2,
   808→  };
   809→}

   815→/** Simple deterministic hash for date+instrument rotation */
   816→function simpleHash(str: string): number {
   817→  let hash = 0;
   818→  for (let i = 0; i < str.length; i++) {
   819→    const ch = str.charCodeAt(i);
   820→    hash = ((hash << 5) - hash) + ch;
   821→    hash |= 0;
   822→  }
   823→  return Math.abs(hash);
   824→}

   826→/**
   827→ * Selects a V10 persona, template format, and variation seed
   828→ * based on deterministic hash of date + instrument + school.
   829→ * This ensures: different persona per day, different per instrument,
   830→ * and school affinity matching.
   831→ */
   832→export function selectV10Persona(
   833→  instrument: string,
   834→  date: string,
   835→  schoolId: string
   836→): { persona: V10Persona; template: V10TemplateFormat; variationSeed: number } {
   837→  const hashInput = `${date}:${instrument}:v10:2024`;
   838→  const dayHash = simpleHash(hashInput);
   839→
   840→  // Step 1: Find personas matching school affinity
   841→  const matching = PERSONAS.filter(p => p.schoolAffinity === schoolId);
   842→  // Step 2: If no match, use all personas
   843→  const pool = matching.length > 0 ? matching : PERSONAS;
   844→  // Step 3: Select persona by hash
   845→  const personaIndex = dayHash % pool.length;
   846→  const persona = pool[personaIndex];
   847→
   848→  // Step 4: Select template — vary by hash % 7
   849→  let templateIndex: number;
   850→  const templateMod = dayHash % 7;
   851→  if (templateMod < 2) templateIndex = 1; // technical (most common)
   852→  else if (templateMod < 3) templateIndex = 0; // executive
   853→  else if (templateMod < 4) templateIndex = 3; // forecast
   854→  else if (templateMod < 5) templateIndex = 4; // educational
   855→  else templateIndex = 2; // trading
   856→  // Occasional variation: hash % 5 == 0 → use a different template
   857→  if (dayHash % 5 === 0) {
   858→    templateIndex = dayHash % TEMPLATE_FORMATS.length;
   859→  }
   860→  const template = TEMPLATE_FORMATS[templateIndex];
   861→
   862→  return {
   863→    persona,
   864→    template,
   865→    variationSeed: dayHash,
   866→  };
   867→}

   873→/**
   874→ * Selects N phrases from a section of the phrase bank, rotated by seed.
   875→ * Returns a comma-separated string of selected phrases.
   876→ */
   877→function selectPhrases(section: string, subsection: string, count: number, seed: number): string {
   878→  const sectionData = PHRASE_BANK[section];
   879→  if (!sectionData) return '';
   880→  const phrases = sectionData[subsection] || sectionData[Object.keys(sectionData)[0]] || [];
   881→  if (phrases.length === 0) return '';
   882→  const startIdx = seed % phrases.length;
   883→  const selected: string[] = [];
   884→  for (let i = 0; i < Math.min(count, phrases.length); i++) {
   885→    selected.push(phrases[(startIdx + i) % phrases.length]);
   886→  }
   887→  return selected.join('\n- ');
   888→}
