   101→export default function VdesAnalysis(props: VdesAnalysisProps) {
   102→  const {
   103→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
   104→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
   105→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
   106→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
   107→    supportStrengths, resistanceStrengths, priceTargets,
   108→    isIndex, isTgju, currencyUnit,
   109→  } = props;
   110→
   111→  const unit = currencyUnit || 'ریال';
   112→
   113→  // ── Instrument Context (for intelligent naming) ─────────────────
   114→  const instrumentLabel = useMemo(() => {
   115→    if (isIndex) return { name: `شاخص ${symbolName}`, type: 'index' as const, verb: 'قرار دارد' };
   116→    if (isTgju) return { name: symbolName, type: 'tgju' as const, verb: 'قرار دارد' };
   117→    return { name: `سهم ${symbolName}`, type: 'stock' as const, verb: 'قرار دارد' };
   118→  }, [symbolName, isIndex, isTgju]);
   119→
   120→  const IL = instrumentLabel;
   121→
   122→  const containerRef = useRef<HTMLDivElement>(null);
   123→  const textAnalysisRef = useRef<HTMLDivElement>(null);
   124→  const [exporting, setExporting] = useState(false);
   125→
   126→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   127→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   128→  const S1_level = supports[0] ?? currentPrice * 0.95;
   129→  const S2_level = supports[1] ?? currentPrice * 0.90;
   130→  const S3_level = supports[2] ?? currentPrice * 0.85;
   131→
   132→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   133→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   134→
   135→  // ── Dominant scenario ──────────────────────────────────────────
   136→  let highestKey = 'R3';
   137→  let highestProb = 0;
   138→  for (const key of SCENARIO_KEYS) {
   139→    if (scenarios[key].probability > highestProb) {
   140→      highestProb = scenarios[key].probability;
   141→      highestKey = key;
   142→    }
   143→  }
   144→  const strategy = STRATEGY_MAP[highestKey];
   145→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   146→
   147→  // ── Text variety: hash-based selector for varied language ──────────
   148→  const varietySeed = useMemo(() => {
   149→    const raw = `${rsi}${macdHist}${cci}${adx}${stochK}${trendAngle}${Math.round(currentPrice)}`;
   150→    let hash = 0;
   151→    for (let i = 0; i < raw.length; i++) hash = ((hash << 5) - hash + raw.charCodeAt(i)) | 0;
   152→    return Math.abs(hash);
   153→  }, [rsi, macdHist, cci, adx, stochK, trendAngle, currentPrice]);
   154→

   158→  const rsiColor = rsi > 70 ? 'text-red-400' : rsi > 60 ? 'text-amber-400' : rsi > 40 ? 'text-gray-300' : rsi > 30 ? 'text-amber-400' : 'text-emerald-400';
   159→
   160→  // ── Stochastic signal ──────────────────────────────────────────
   161→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   162→
   163→  // ── Trend text ─────────────────────────────────────────────────
   164→  const isUp = trendDirection === 'up';

   166→  const trendLabel = isUp ? 'صعودی' : isDown ? 'نزولی' : 'خنثی';
   167→  const trendColorCls = isUp ? 'text-emerald-400' : isDown ? 'text-red-400' : 'text-amber-400';
   168→  const adxStrength = adx > 40 ? 'بسیار قوی' : adx > 25 ? 'قوی' : adx > 15 ? 'متوسط' : 'ضعیف';
   169→  const adxColor = adx > 40 ? 'text-emerald-400' : adx > 25 ? 'text-amber-400' : 'text-gray-500';
   170→  const r2Pct = (trendR2 * 100).toFixed(1);
   171→
   172→  // ── Bollinger position ─────────────────────────────────────────
   173→  const bbRange = bollingerUpper - bollingerLower;
   174→  const bbPos = bbRange > 0 ? toPersianDigits(((currentPrice - bollingerLower) / bbRange * 100).toFixed(0)) : '۵۰';
   175→  const bbAbove = currentPrice > bollingerUpper;
   176→  const bbBelow = currentPrice < bollingerLower;
   177→  const bbSignal = bbAbove ? 'بالای باند بالایی' : bbBelow ? 'زیر باند پایینی' : `داخل باندها (${bbPos}٪)`;

   179→  // ── Distance to MA ─────────────────────────────────────────────
   180→  const distMa21 = ((currentPrice - ma21) / ma21 * 100).toFixed(1);
   181→  const distMa100 = ((currentPrice - ma100) / ma100 * 100).toFixed(1);
   182→
   183→  // ── TradingView scenario colors ────────────────────────────────
   184→  const chartCandles = useMemo(() => {
   185→    if (candles.length <= 2000) return candles;
   186→    return candles.slice(-2000);
   187→  }, [candles]);
   188→
   189→  const tvScenarios = useMemo(() => {
   190→    const colors: Record<string, string> = { R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62' };
   191→    return Object.fromEntries(
   192→      SCENARIO_KEYS.map(k => [k, { targetMin: scenarios[k].targetMin, targetMax: scenarios[k].targetMax, name: scenarios[k].name, probability: scenarios[k].probability, color: colors[k] }])
   193→    );
   194→  }, [scenarios]);
   195→

   197→  const mfiWarning = hasVolume ? ` (مقدار ${toFa(mfi)})` : '';
   198→  const strategyText = useMemo(() => {
   199→    const v = varietySeed % 3; // 3 variants per signal type
   200→    if (rsi > 70 || (hasVolume && mfi > 80)) {
   201→      const variants = [
   202→        `هشدار اشباع خرید فعال است (RSI: ${toFa(rsi)}${mfiWarning}). فاصله تا مقاومت اول ${toFa(R1_level)} ${unit} است. بهترین استراتژی در این شرایط، صبر برای اصلاح قیمت تا محدوده حمایت ${toFa(S1_level)} تا ${toFa(S2_level)} و ورود در نقطه بهینه‌تر می‌باشد.`,
   203→        `با توجه به ورود به منطقه اشباع خرید (RSI=${toFa(rsi)}${mfiWarning}) و نزدیکی به مقاومت ${toFa(R1_level)}، ریسک اصلاح کوتاه‌مدت بالا ارزیابی می‌شود. پیشنهاد: انتظار برای بازگشت قیمت به ${toFa(S1_level)}-${toFa(S2_level)} و سپس بررسی مجدد سیگنال‌ها.`,
   204→        `${IL.name} در محدوده اشباع خرید (RSI: ${toFa(rsi)}${mfiWarning}) ${IL.verb}. با در نظر گرفتن مقاومت ${toFa(R1_level)} در بالا و حمایت ${toFa(S1_level)} در پایین، ورود جدید در این نقطه توصیه نمی‌شود. بهتر است منتظر عقبنشینی به ${toFa(S2_level)} بمانید.`,
   205→      ];
   206→      return variants[v];
   207→    }
   208→    if (rsi < 30 || (hasVolume && mfi < 20)) {
   209→      const variants = [
   210→        `اشباع فروش قابل توجه (RSI: ${toFa(rsi)}${mfiWarning}) در محدوده حمایت ${toFa(S1_level)} شناسایی شد. این شرایط می‌تواند فرصت مناسبی برای ورود باشد، با حد ضرر ${toFa(S2_level)}. هدف اول ${toFa(R1_level)} و هدف دوم ${toFa(R2_level)} ${unit}.`,
   211→        `${IL.name} به منطقه اشباع فروش رسیده (RSI=${toFa(rsi)}${mfiWarning}). فاصله تا حمایت کلیدی ${toFa(S1_level)} کم است. اگر واکنش مثبت به این حمایت مشاهده شود، بازگشت به ${toFa(R1_level)} محتمل خواهد بود.`,
   212→        `شاخص قدرت نسبی در محدوده اشباع فروش (مقدار ${toFa(rsi)}${mfiWarning}) ${IL.verb}. نزدیکی به حمایت ${toFa(S1_level)} و سیگنال فروش بیش از حد، احتمال بازگشت کوتاه‌مدت را تقویت می‌کند. حد ضرر پیشنهادی: ${toFa(S2_level)}.`,
   213→      ];
   214→      return variants[v];
   215→    }
   216→    const variants = [
   217→      `سیگنال‌ها در محدوده خنثی قرار دارند (RSI: ${toFa(rsi)}، قدرت روند: ${adxStrength}). محدوده رنج فعلی بین ${toFa(S1_level)} و ${toFa(R1_level)} است. تا خروج قیمت از این کانال، بهترین اقدام نظاره و عدم ورود است.`,
   218→      `وضعیت فعلی ${IL.name} نشان‌دهنده عدم قطعیت (RSI=${toFa(rsi)}، ADX=${toFa(adx)}). ${IL.name} در بازه ${toFa(S1_level)} تا ${toFa(R1_level)} در نوسان است. شکست معتبر یکی از این سطوح، مسیر آینده را مشخص خواهد کرد.`,
   219→      `شاخص‌ها هم‌جهت نیستند (RSI: ${toFa(rsi)}) و قدرت روند ${adxStrength} ارزیابی می‌شود. تا زمانی که ${IL.name} از محدوده ${toFa(S1_level)}-${toFa(R1_level)} خارج نشود، ورود با ریسک بالا همراه خواهد بود.`,
   220→    ];
   221→    return variants[v];
   222→  }, [rsi, mfi, hasVolume, R1_level, R2_level, S1_level, S2_level, varietySeed, IL.name, IL.verb]);
   223→
   224→  // ═══════════════════════════════════════════════════════════════════
   225→  // Export handlers

   229→  const captureContainer = useCallback(async (): Promise<string | null> => {
   230→    const el = containerRef.current;
   231→    if (!el) return null;

   236→      const { toPng } = await import('html-to-image');
   237→      return await toPng(el, { backgroundColor: '#060a13', pixelRatio: 2, quality: 0.95 });
   238→    } finally {
   239→      if (toolbar) toolbar.style.visibility = '';

   241→  }, []);

   245→    setExporting(true);

   272→<html lang="fa" dir="rtl">
   273→<head>
   274→<meta charset="UTF-8">
   275→<meta name="viewport" content="width=device-width, initial-scale=1.0">

   278→  * { margin: 0; padding: 0; box-sizing: border-box; }

   281→    display: flex;
   282→    justify-content: center;
   283→    padding: 20px;
   284→    min-height: 100vh;
   285→  }
   286→  .page-wrapper {
   287→    width: 100%;
   288→    max-width: 960px;
   289→  }
   290→  .page-wrapper img {
   291→    width: 100%;
   292→    height: auto;
   293→    display: block;
   294→    border-radius: 16px;
   295→  }
   296→  .footer {
   297→    text-align: center;
   298→    font-family: Tahoma, Arial, sans-serif;
   299→    font-size: 11px;
   300→    color: #4b5563;
   301→    margin-top: 16px;
   302→    padding-top: 12px;
   303→    border-top: 1px solid rgba(255,255,255,0.05);

   305→  @media print {

   307→    .footer { color: #999; }
   308→  }
   309→</style>
   310→</head>
   311→<body>
   312→<div class="page-wrapper">
   313→  <img src="data:image/png;base64,${base64}" alt="تحلیل تکنیکال ${IL.name}" />

   315→</div>
   316→</body>
   317→</html>`;
   318→      const blob = new Blob([html], { type: 'text/html;charset=utf-8' });
   319→      const url = URL.createObjectURL(blob);
   320→      const a = document.createElement('a'); a.href = url; a.download = `${IL.name}-تحلیل-تکنیکال.html`; a.click();
   321→      URL.revokeObjectURL(url);
   322→    } catch (err) { console.error('HTML export failed:', err); }
   323→    finally { setExporting(false); }
   324→  }, [IL.name, captureContainer]);
   325→
   326→  // ── Export Plain Text ──────────────────────────────────────────
   327→  const handleExportText = useCallback(() => {
   328→    const el = textAnalysisRef.current;
   329→    if (!el) return;
   330→    setExporting(true);
   331→    try {
   332→      const now = new Date();
   333→      const dateStr = toShamsi(now, 'long');
   334→      // Extract text content, clean up whitespace
   335→      const rawText = el.innerText || el.textContent || '';
   336→      const lines = rawText.split('\n').map(l => l.trim()).filter(Boolean);
   337→      const text = `تحلیل تکنیکال ${IL.name}
   338→${'═'.repeat(50)}
   339→تاریخ تهیه: ${dateStr}
   340→
   341→${lines.join('\n')}
   342→
   343→${'─'.repeat(50)}
   344→صرفاً جنبه تحلیلی دارد و توصیه سرمایه‌گذاری نیست.`;
   345→      const blob = new Blob([`\uFEFF${text}`], { type: 'text/plain;charset=utf-8' });
   346→      const url = URL.createObjectURL(blob);
   347→      const a = document.createElement('a'); a.href = url; a.download = `${IL.name}-تحلیل-متنی.txt`; a.click();
   348→      URL.revokeObjectURL(url);
   349→    } catch (err) { console.error('Text export failed:', err); }
   350→    finally { setExporting(false); }
   351→  }, [IL.name]);
   352→
   353→  return (
   354→    <div ref={containerRef} className="space-y-5" dir="rtl">
   355→      {/* ═══ HEADER ═══ */}

   704→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   705→function LevelBox({ index, label, price, strength, isTarget, type, unit }: {
   706→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   707→}) {
   708→  const isSupport = type === 'support';
   709→  const isTargetType = type === 'target';
   710→
   711→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   712→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   713→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   714→
   715→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   716→  const strengthLabel = strength >= 8 ? 'بسیار قوی' : strength >= 6 ? 'قوی' : strength >= 4 ? 'متوسط' : strength >= 2 ? 'ضعیف' : 'بسیار ضعیف';
   717→  const strengthColor = strength >= 8 ? '#34c98b' : strength >= 6 ? '#3ad5db' : strength >= 4 ? '#ffb11b' : strength >= 2 ? '#ff7b32' : '#ef4d62';
   718→
   719→  // Distance from current price (if available from context)
   720→  return (
   721→    <div
   722→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   723→      style={{
   724→        background: bgAccent,
   725→        border: `1px solid ${borderAccent}`,
   726→        borderRight: `3px solid ${accentColor}`,
   727→      }}
   728→    >
   729→      {/* Label + Price */}
   730→      <div className="flex-1 min-w-0">
   731→        <div className="flex items-center gap-2">
   732→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   733→          {isTarget && <span className="text-[9px]">🎯</span>}
   734→        </div>
   735→        <span className="text-sm font-black text-gray-100 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   736→        <span className="text-[10px] text-gray-500 mr-1">{unit || 'ریال'}</span>
   737→      </div>
   738→
   739→      {/* Strength bar */}
   740→      <div className="flex flex-col items-end gap-1 min-w-[80px]">
   741→        <div className="flex items-center gap-1.5">
   742→          <span className="text-[9px] font-medium" style={{ color: strengthColor }}>{strengthLabel}</span>
   743→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(strength)}</span>
   744→          <span className="text-[9px] text-gray-500">/۱۰</span>
   745→        </div>
   746→        <div className="w-full h-1.5 rounded-full bg-white/5 overflow-hidden">
   747→          <div
   748→            className="h-full rounded-full transition-all duration-500"
   749→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   750→          />
   751→        </div>
   752→      </div>
   753→    </div>
   754→  );
   755→}
