    81→  R3: { text: 'بازار رنج — منتظر خروج از محدوده بمانید', tagCls: 'bg-amber-500/15 text-amber-400 border-amber-500/30' },
    82→  R4: { text: 'اصلاحی — احتیاط و کاهش حجم معاملات', tagCls: 'bg-orange-500/15 text-orange-400 border-orange-500/30' },
    83→  R5: { text: 'نزولی قوی — خروج فوری توصیه می‌شود', tagCls: 'bg-red-500/15 text-red-400 border-red-500/30' },
    84→};

    87→// Component
    88→// ═══════════════════════════════════════════════════════════════════════════════
    89→
    90→export default function VdesAnalysis(props: VdesAnalysisProps) {
    91→  const {
    92→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
    93→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
    94→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
    95→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
    96→    supportStrengths, resistanceStrengths, priceTargets,
    97→  } = props;
    98→
    99→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   100→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   101→  const S1_level = supports[0] ?? currentPrice * 0.95;
   102→  const S2_level = supports[1] ?? currentPrice * 0.90;
   103→  const S3_level = supports[2] ?? currentPrice * 0.85;
   104→
   105→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   106→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   107→
   108→  // ── Dominant scenario ──────────────────────────────────────────
   109→  let highestKey = 'R3';
   110→  let highestProb = 0;
   111→  for (const key of SCENARIO_KEYS) {
   112→    if (scenarios[key].probability > highestProb) {
   113→      highestProb = scenarios[key].probability;
   114→      highestKey = key;
   115→    }
   116→  }
   117→  const strategy = STRATEGY_MAP[highestKey];
   118→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   119→
   120→  // ── RSI signal ─────────────────────────────────────────────────
   121→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   122→  const rsiColor = rsi > 70 ? 'text-red-400' : rsi > 60 ? 'text-amber-400' : rsi > 40 ? 'text-gray-300' : rsi > 30 ? 'text-amber-400' : 'text-emerald-400';
   123→
   124→  // ── Stochastic signal ──────────────────────────────────────────
   125→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   126→
   127→  // ── MACD signal ────────────────────────────────────────────────
   128→  const macdSignalText = macdHist > 0 ? 'مومنتوم مثبت (هیستوگرام بالای صفر)' : 'مومنتوم منفی (هیستوگرام زیر صفر)';
   129→  const macdBullish = macdLine > macdSignal;
   130→
   131→  // ── Trend text ─────────────────────────────────────────────────
   132→  const trendText = trendDirection === 'up'
   133→    ? `صعودی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${(trendR2 * 100).toFixed(1)}%`
   134→    : trendDirection === 'down'
   135→    ? `نزولی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${(trendR2 * 100).toFixed(1)}%`
   136→    : 'خنثی و بدون جهت مشخص';
   137→
   138→
   139→
   140→  // ── Bollinger Band position (simplified) ────────────────────────
   141→  const bbRange = bollingerUpper - bollingerLower;
   142→  const bbPos = bbRange > 0 ? ((currentPrice - bollingerLower) / bbRange * 100).toFixed(0) : '50';
   143→  const bbSignal = currentPrice > bollingerUpper ? 'بالای باند بالایی (اشباع خرید)'
   144→    : currentPrice < bollingerLower ? 'زیر باند پایینی (اشباع فروش)'
   145→    : `داخل باندها (${bbPos}٪ از بازه)`;
   146→
   147→
   148→  // ── TradingView scenario colors ────────────────────────────────
   149→  // Limit candles for chart rendering to prevent freeze on large datasets (e.g. derivatives with 10k+ candles)
   150→  const chartCandles = useMemo(() => {
   151→    if (candles.length <= 2000) return candles;
   152→    return candles.slice(-2000);
   153→  }, [candles]);
   154→
   155→  const tvScenarios = useMemo(() => {
   156→    const colors: Record<string, string> = {
   157→      R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62',
   158→    };
   159→    return Object.fromEntries(
   160→      SCENARIO_KEYS.map(k => [k, {
   161→        targetMin: scenarios[k].targetMin,
   162→        targetMax: scenarios[k].targetMax,
   163→        name: scenarios[k].name,
   164→        probability: scenarios[k].probability,
   165→        color: colors[k],
   166→      }])
   167→    );
   168→  }, [scenarios]);
   169→
   170→  // ── Strategy recommendation text ────────────────────────────────
   171→  const mfiWarning = hasVolume ? ` (مقدار ${toFa(mfi)})` : '';
   172→  const strategyText = rsi > 70 || (hasVolume && mfi > 80)
   173→    ? `با توجه به هشدار اشباع خرید (قدرت نسبی: ${toFa(rsi)}${mfiWarning}) و فاصله قیمت تا مقاومت ${toFa(R1_level)}، استراتژی محتاطانه، انتظار برای اصلاح قیمت و ورود در محدوده حمایت ${toFa(S1_level)} تا ${toFa(S2_level)} ریال می‌باشد. در این محدوده می‌توان با حد ضرر ${toFa(S2_level)} ریال وارد موقعیت خرید شد.`
   174→    : rsi < 30 || (hasVolume && mfi < 20)
   175→    ? `با توجه به اشباع فروش (قدرت نسبی: ${toFa(rsi)}${mfiWarning}) و نزدیکی به حمایت ${toFa(S1_level)}، فرصت خرید در محدوده فعلی با حد ضرر ${toFa(S2_level)} ریال قابل بررسی است. هدف اولیه ${toFa(R1_level)} و هدف ثانویه ${toFa(R2_level)} ریال تعیین می‌شود.`
   176→    : `با توجه به وضعیت خنثی شاخص‌ها (قدرت نسبی: ${toFa(rsi)}، قدرت روند: ${adx > 25 ? 'قوی' : 'ضعیف'})، انتظار برای خروج قیمت از محدوده ${toFa(S1_level)} تا ${toFa(R1_level)} ریال و سپس تصمیم‌گیری توصیه می‌شود.${hasVolume ? ' حجم معاملات و مومنتوم کلی را برای تأیید سیگنال پایش کنید.' : ' مومنتوم کلی و حرکات قیمتی را برای تأیید سیگنال پایش کنید.'}`;
   177→
   178→  return (
   179→    <div className="space-y-5" dir="rtl">
   180→      {/* ═══ HEADER ═══ */}
   181→      <div className="rounded-[32px] px-6 py-5 border border-amber-500/20"
   182→        style={{
   183→          background: 'rgba(18,28,46,0.7)',
   184→          backdropFilter: 'blur(14px)',
   185→          boxShadow: '0 25px 50px -12px rgba(0,0,0,0.6)',
   186→        }}>
   187→        <h1 className="text-xl font-bold mb-1" style={{ color: '#f8e365' }}>
   188→          📈 تحلیل تکنیکال {symbolName}
   189→        </h1>
   190→        <div className="flex flex-wrap gap-3 mt-2">
   191→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   192→            📍 قیمت مرجع: <b className="text-gray-200">{toFa(currentPrice)} ریال</b>
   193→          </span>
   194→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   195→            🎯 هدف کوتاه‌مدت: <b className="text-gray-200">{toFa(targetMin)} — {toFa(targetMax)} ریال</b>
   196→          </span>
   197→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   198→            📊 روند: <b className="text-gray-200">{trendText}</b>
   199→          </span>
   200→          <span className={`px-3.5 py-1 rounded-full border text-xs font-medium ${
   201→            rsi > 70 ? 'bg-red-500/15 text-red-400 border-red-500/30'
   202→            : rsi < 30 ? 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30'
   203→            : 'bg-amber-500/15 text-amber-400 border-amber-500/30'
   204→          }`}>
   205→            قدرت نسبی: {toFa(rsi)} ({rsiSignal})
   206→          </span>
   207→        </div>
   208→      </div>
   209→
   210→      {/* ═══ TRADINGVIEW CHART ═══ */}
   211→      <div className="rounded-[28px] p-4 border border-white/6"
   212→        style={{
   213→          background: 'rgba(14,22,40,0.7)',
   214→          backdropFilter: 'blur(8px)',
   215→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   216→        }}>
   217→        <h2 className="text-sm font-semibold mb-3 text-gray-200 flex items-center gap-3">
   218→          <span style={{ color: '#f8e365' }}>📊</span>
   219→          نمودار قیمتی سهم (روزانه) — به همراه MA21، MA100، سطوح حمایت/مقاومت و اهداف قیمتی
   220→        </h2>
   221→        <TradingViewChart
   222→          symbolName={symbolName}
   223→          candles={chartCandles}
   224→          supports={supports}
   225→          resistances={resistances}
   226→          supportStrengths={supportStrengths}
   227→          resistanceStrengths={resistanceStrengths}
   228→          ma21={ma21}
   229→          ma100={ma100}
   230→          scenarios={tvScenarios}
   231→        />
   232→        <div className="mt-3 flex flex-wrap gap-4 text-[10px] text-gray-500">
   233→          <span>🔴 خطوط قرمز: مقاومت‌ها ({resistances.map(toFa).join(' ، ')})</span>
   234→          <span>🟢 خطوط سبز: حمایت‌ها ({supports.map(toFa).join(' ، ')})</span>
   235→          <span>🟣 خط بنفش: میانگین ۱۰۰ روزه ({toFa(ma100)})</span>
   236→          <span>🔵 خط آبی: میانگین ۲۱ روزه ({toFa(ma21)})</span>
   237→          <span>🟡 خطوط زرد: اهداف قیمتی سناریوها (رند شده)</span>
   238→        </div>
   239→      </div>
   240→
   241→      {/* ═══ DUAL ANALYSIS LAYOUT ═══ */}
   242→      <div className="rounded-[28px] p-5 border border-white/6"
   243→        style={{
   244→          background: 'rgba(14,22,40,0.7)',
   245→          backdropFilter: 'blur(8px)',
   246→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   247→        }}>
   248→        <h2 className="text-sm font-semibold mb-4 text-gray-200 flex items-center gap-3">
   249→          <span style={{ color: '#f8e365' }}>🧠</span>
   250→          تحلیل جامع روند و سطوح کلیدی
   251→        </h2>
   252→
   253→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-5">
   254→          {/* ── Text Analysis (Right in RTL) ── */}
   255→          <div className="rounded-2xl p-4 border border-white/4 space-y-4 flex flex-col"
   256→            style={{ background: 'rgba(8,20,35,0.5)' }}>
   257→
   258→            {/* ── PARAGRAPH 1: General Trend & Price Position ── */}
   259→            <p className="text-sm text-gray-300 leading-[1.85]">
   260→              <strong style={{ color: '#f8e365' }}>روند کلی و موقعیت قیمت:</strong>{' '}
   261→              سهم {symbolName} در حال حاضر در محدوده{' '}
   262→              <b className="text-gray-100">{toFa(currentPrice)} ریال</b> معامله می‌شود و روند میان‌مدت{' '}
   263→              <b className={trendDirection === 'up' ? 'text-emerald-400' : trendDirection === 'down' ? 'text-red-400' : 'text-amber-400'}>{trendDirection === 'up' ? 'صعودی' : trendDirection === 'down' ? 'نزولی' : 'خنثی'}</b>
   264→              {' '}است (زاویه {toFa(Math.abs(trendAngle))}°، ضریب تعیین={+(trendR2 * 100).toFixed(1)}%).
   265→              قیمت نسبت به میانگین متحرک ۲۱ روزه ({toFa(ma21)} ریال){' '}
   266→              {currentPrice > ma21 ? <span className="text-emerald-400">بالاتر</span> : <span className="text-red-400">پایین‌تر</span>}
   267→              {' '}و نسبت به میانگین ۱۰۰ روزه ({toFa(ma100)} ریال){' '}
   268→              {currentPrice > ma100 ? <span className="text-emerald-400">بالاتر</span> : <span className="text-red-400">پایین‌تر</span>}
   269→              {' '}قرار دارد. {' '}
   270→              {sar < currentPrice
   271→                ? <span>توقف زیان پارابولیک زیر قیمت قرار دارد و <b className="text-emerald-400">تأیید‌کننده روند صعودی</b> است.</span>
   272→                : <span>توقف زیان پارابولیک بالای قیمت قرار دارد و <b className="text-red-400">هشدار نزولی</b> صادر می‌کند.</span>
   273→              }
   274→              {' '}قدرت روند{' '}
   275→              {adx > 40 ? <b className="text-emerald-400">بسیار قوی</b> : adx > 25 ? <b className="text-amber-400">متوسط</b> : <b className="text-gray-500">ضعیف یا رنج</b>}
   276→              {' '}ارزیابی می‌شود (مقدار {toFa(adx)}). {' '}
   277→              {diPlus > diMinus
   278→                ? <span>فشار خریداری ({toFa(diPlus)}) نسبت به فروشندگان ({toFa(diMinus)}) غالب است.</span>
   279→                : <span>فشار فروشندگان ({toFa(diMinus)}) نسبت به خریداران ({toFa(diPlus)}) غالب است.</span>
   280→              }
   281→            </p>
   282→
   283→            {/* ── PARAGRAPH 2: Momentum & Oscillator Results ── */}
   284→            <p className="text-sm text-gray-300 leading-[1.85]">
   285→              <strong style={{ color: '#f8e365' }}>وضعیت مومنتوم و قدرت خرید/فروش:</strong>{' '}
   286→              شاخص قدرت نسبی در ناحیه{' '}
   287→              <b className={rsiColor}>{rsiSignal}</b>{' '}
   288→              قرار دارد (مقدار {toFa(rsi)}).
   289→              {hasVolume && <>
   290→              {' '}شاخص جریان نقدی{' '}
   291→              {mfi > 80 ? <span className="text-red-400">اشباع خرید را تأیید می‌کند</span> : mfi < 20 ? <span className="text-emerald-400">اشباع فروش را نشان می‌دهد</span> : <span>در محدوده عادی است</span>}
   292→              {' '}(مقدار {toFa(mfi)}).{' '}</>}
   293→              شاخص تمایل کانال کالا در محدوده{' '}
   294→              {cci > 100 ? <span className="text-red-400">بالاتر از حد معمول (قدرت خریداری)</span> : cci < -100 ? <span className="text-emerald-400">پایین‌تر از حد معمول (قدرت فروشندگان)</span> : <span>عادی</span>}
   295→              {' '}(مقدار {toFa(cci)}).{' '}
   296→              نوسان‌نما در وضعیت{' '}
   297→              <b className={stochK > 80 ? 'text-red-400' : stochK < 20 ? 'text-emerald-400' : 'text-gray-300'}>{stochSignal}</b>{' '}
   298→              قرار دارد (K={toFa(stochK)}، D={toFa(stochD)}).{' '}
   299→              مومنتوم کلی{' '}
   300→              {macdHist > 0 && macdLine > macdSignal
   301→                ? <span className="text-emerald-400">مثبت و صعودی است — سیگنال خرید</span>
   302→                : macdHist > 0
   303→                ? <span className="text-amber-400">مثبت اما ضعیف‌شونده</span>
   304→                : macdLine < macdSignal
   305→                ? <span className="text-red-400">منفی و نزولی است — سیگنال فروش</span>
   306→                : <span className="text-gray-400">منفی اما در حال بهبود</span>
   307→              }
   308→              {' '}(هیستوگرام: {toFa(macdHist)}).
   309→            </p>
   310→
   311→            {/* ── PARAGRAPH 3: Volatility Analysis ── */}
   312→            <p className="text-sm text-gray-300 leading-[1.85]">
   313→              <strong style={{ color: '#f8e365' }}>تحلیل نوسانات:</strong>{' '}
   314→              قیمت نسبت به باندهای نوسان{' '}
   315→              <b className="text-gray-200">{bbSignal}</b>
   316→              {' '}قرار دارد.
   317→              {currentPrice > bollingerUpper
