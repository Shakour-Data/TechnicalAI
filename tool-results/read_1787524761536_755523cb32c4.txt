   105→// ─── Mini Sparkline SVG ───────────────────────────────────────────────────────
   106→
   107→function MiniSparkline({ predictions, color, width = 120, height = 40 }: { predictions: PredictionPoint[]; color: string; width?: number; height?: number }) {
   108→  if (predictions.length < 2) return null;
   109→
   110→  const prices = predictions.map(p => p.price);
   111→  const min = Math.min(...prices, ...predictions.map(p => p.lower));
   112→  const max = Math.max(...prices, ...predictions.map(p => p.upper));
   113→  const range = max - min || 1;
   114→  const pad = 4;
   115→
   116→  const toX = (i: number) => pad + (i / (predictions.length - 1)) * (width - 2 * pad);
   117→  const toY = (v: number) => pad + (1 - (v - min) / range) * (height - 2 * pad);
   118→
   119→  // Main line
   120→  const linePoints = predictions.map((p, i) => `${toX(i)},${toY(p.price)}`).join(' ');
   121→  // Upper band
   122→  const upperPoints = predictions.map((p, i) => `${toX(i)},${toY(p.upper)}`).join(' ');
   123→  // Lower band
   124→  const lowerPoints = predictions.map((p, i) => `${toX(i)},${toY(p.lower)}`).reverse().join(' ');
   125→  // Fill area
   126→  const fillPath = `M${upperPoints} L${lowerPoints} Z`;
   127→
   128→  return (
   129→    <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-10" style={{ minWidth: width }}>
   130→      <defs>
   131→        <linearGradient id={`fill-${color.replace('#', '')}`} x1="0" y1="0" x2="0" y2="1">
   132→          <stop offset="0%" stopColor={color} stopOpacity="0.15" />
   133→          <stop offset="100%" stopColor={color} stopOpacity="0.02" />
   134→        </linearGradient>
   135→      </defs>
   136→      <path d={fillPath} fill={`url(#fill-${color.replace('#', '')})`} />
   137→      <polyline points={upperPoints} fill="none" stroke={color} strokeOpacity="0.3" strokeWidth="0.5" strokeDasharray="2,2" />
   138→      <polyline points={predictions.map((p, i) => `${toX(i)},${toY(p.lower)}`).join(' ')} fill="none" stroke={color} strokeOpacity="0.3" strokeWidth="0.5" strokeDasharray="2,2" />
   139→      <polyline points={linePoints} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   140→      {predictions.map((p, i) => (
   141→        <circle key={i} cx={toX(i)} cy={toY(p.price)} r="1.5" fill={color} />
   142→      ))}
   143→    </svg>
   144→  );
   145→}

   156→  const [expandedModel, setExpandedModel] = useState<string | null>(null);
   157→  const [showTable, setShowTable] = useState(false);
   158→
   159→  // Check ML service health on mount (auto-starts service if down)
   160→  useEffect(() => {
   161→    fetch('/api/ml-predict')
   162→      .then(r => r.json())
   163→      .then(data => setServiceAvailable(data.status === 'ok'))
   164→      .catch(() => setServiceAvailable(false));
   165→  }, []);
   166→
   167→  const runPrediction = useCallback(async () => {
   168→    if (candles.length < 60) {
   169→      setError('حداقل ۶۰ کندل داده برای پیش‌بینی ML نیاز است. داده‌های فعلی کافی نیست.');
   170→      return;
   171→    }
   172→    setLoading(true);
   173→    setError(null);
   174→    setResult(null);
   175→    try {
   176→      const resp = await fetch('/api/ml-predict', {
   177→        method: 'POST',
   178→        headers: { 'Content-Type': 'application/json' },
   179→        body: JSON.stringify({ candles, sessions, models: selectedModels }),
   180→      });
   181→      const json = await resp.json();
   182→      if (json.error) {
   183→        setError(json.error);
   184→      } else {
   185→        setResult(json);
   186→      }
   187→    } catch (err) {
   188→      const msg = err instanceof Error ? err.message : 'خطای ناشناخته';
   189→      // Auto-retry if service was starting
   190→      if (msg.includes('راه‌اندازی') || msg.includes('دسترس نیست')) {
   191→        setError('سرویس ML در حال راه‌اندازی است. لطفاً پس از ۱۰ ثانیه دوباره تلاش کنید.');
   192→        // Re-check health after 10s
   193→        setTimeout(() => {
   194→          fetch('/api/ml-predict')
   195→            .then(r => r.json())
   196→            .then(data => setServiceAvailable(data.status === 'ok'))
   197→            .catch(() => setServiceAvailable(false));
   198→        }, 10000);
   199→      } else {
   200→        setError(msg);
   201→      }
   202→    } finally {
   203→      setLoading(false);
   204→    }
   205→  }, [candles, sessions, selectedModels]);
   206→
   207→  // Feature importance sorted
   208→  const topFeatures = useMemo(() => {
   209→    if (!result?.feature_importance) return [];
   210→    return Object.entries(result.feature_importance)
   211→      .sort(([, a], [, b]) => b - a)
   212→      .slice(0, 10);
   213→  }, [result]);
   214→
   215→  // Ensemble predictions for chart
   216→  const ensemblePredictions = result?.ensemble?.predictions ?? [];
   217→
   218→  // Determine if ensemble is overall bullish/bearish
   219→  const ensembleTrend = useMemo(() => {
   220→    if (ensemblePredictions.length < 1) return 'neutral';
   221→    const lastPred = ensemblePredictions[ensemblePredictions.length - 1];
   222→    if (lastPred.change_pct > 1) return 'bullish';
   223→    if (lastPred.change_pct < -1) return 'bearish';
   224→    return 'neutral';
   225→  }, [ensemblePredictions]);
   226→
   227→  const trendIcon = ensembleTrend === 'bullish' ? TrendingUp : ensembleTrend === 'bearish' ? TrendingDown : BarChart3;
   228→  const TrendIcon = trendIcon;
   229→  const trendLabel = ensembleTrend === 'bullish' ? 'صعودی' : ensembleTrend === 'bearish' ? 'نزولی' : 'خنثی';
   230→  const trendColor = ensembleTrend === 'bullish' ? 'text-emerald-700' : ensembleTrend === 'bearish' ? 'text-red-700' : 'text-amber-700';
   231→
   232→  return (
   233→    <div className="space-y-4">
   234→      {/* ═══ HEADER ═══ */}
   235→      <div className="rounded-2xl px-6 py-5 border border-violet-200 bg-gradient-to-l from-violet-50 to-white shadow-sm">
   236→        <div className="flex items-center justify-between flex-wrap gap-3">
   237→          <div className="flex items-center gap-3">
   238→            <div className="w-10 h-10 rounded-xl bg-violet-100 border border-violet-200 flex items-center justify-center">
   239→              <BrainCircuit className="w-5 h-5 text-violet-700" />
   240→            </div>
   241→            <div>
   242→              <h2 className="text-base font-bold text-gray-900">پیش‌بینی قیمت با هوش مصنوعی</h2>
   243→              <p className="text-[10px] text-gray-500">۵ مدل ML × ۳۴ ویژگی تکنیکال — scikit-learn, XGBoost, LightGBM</p>
   244→            </div>
   245→          </div>
   246→          <div className="flex items-center gap-2">
   247→            {serviceAvailable === false && (
   248→              <span className="px-3 py-1 rounded-full bg-amber-50 border border-amber-200 text-[10px] text-amber-600 flex items-center gap-1">
   249→                <RefreshCw className="w-3 h-3 animate-spin" />
   250→                در حال راه‌اندازی...
   251→              </span>

   253→            {serviceAvailable === true && (
   254→              <span className="px-3 py-1 rounded-full bg-emerald-50 border border-emerald-200 text-[10px] text-emerald-600 flex items-center gap-1">

   257→              </span>

   259→          </div>
   260→        </div>
   261→      </div>
   262→
   263→      {/* ═══ CONTROLS ═══ */}

   265→        <div className="flex flex-wrap items-end gap-4">
   266→          {/* Sessions selector */}
   267→          <div className="flex flex-col gap-1.5">
   268→            <label className="text-[10px] text-gray-500 font-medium">تعداد جلسات پیش‌بینی</label>
   269→            <div className="flex items-center gap-1 bg-[#f3f4f6] rounded-lg p-0.5 border border-[#e5e7eb]">
   270→              {SESSION_OPTIONS.map(s => (
   271→                <button
   272→                  key={s}
   273→                  onClick={() => setSessions(s)}
   274→                  className={`px-3 py-1.5 rounded-md text-xs font-medium transition-all cursor-pointer ${
   275→                    sessions === s
   276→                      ? 'bg-white text-violet-700 shadow-sm border border-[#e5e7eb]'

   278→                  }`}

   303→                    background: selectedModels.includes(key) ? MODEL_COLORS[key] + '10' : undefined,
   304→                  }}
   305→                >
   306→                  {label}
   307→                </button>
   308→              ))}
   309→            </div>
   310→          </div>
   311→
   312→          {/* Run button */}
   313→          <button
   314→            onClick={runPrediction}
   315→            disabled={loading || selectedModels.length === 0 || candles.length < 60}
   316→            className="mr-auto inline-flex items-center gap-2 px-5 py-2 rounded-lg bg-violet-600 text-white text-xs font-bold hover:bg-violet-700 transition-colors disabled:opacity-50 disabled:cursor-not-allowed cursor-pointer"
   317→          >
   318→            <Zap className={`w-3.5 h-3.5 ${loading ? 'animate-pulse' : ''}`} />
   319→            {loading ? 'در حال تحلیل...' : 'شروع پیش‌بینی'}
   320→          </button>
   321→        </div>
   322→
   323→        {candles.length < 60 && (
   324→          <p className="mt-3 text-[10px] text-amber-600 flex items-center gap-1">
   325→            <AlertTriangle className="w-3 h-3" />
   326→            تعداد کندل‌های فعلی ({toPersianDigits(String(candles.length))}) کمتر از حداقل مورد نیاز (۶۰) است.
   327→          </p>
   328→        )}
   329→      </div>
   330→
   331→      {/* ═══ LOADING ═══ */}
   332→      {loading && (
   333→        <div className="rounded-2xl p-8 border border-violet-200 bg-violet-50/30">
   334→          <div className="flex flex-col items-center justify-center gap-4">
   335→            <div className="relative">
   336→              <div className="w-12 h-12 rounded-full border-4 border-violet-200 border-t-violet-600 animate-spin" />
   337→              <BrainCircuit className="w-5 h-5 text-violet-600 absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2" />
   338→            </div>
   339→            <div className="text-center">

   343→              </p>
   344→            </div>
   345→          </div>
   346→        </div>
   347→      )}
   348→
   349→      {/* ═══ ERROR ═══ */}

   351→        <div className="rounded-2xl p-5 border border-red-200 bg-red-50">
   352→          <div className="flex items-start gap-3">

   362→      {/* ═══ RESULTS ═══ */}

   367→            <div className="rounded-xl p-3 border border-[#e5e7eb] bg-[#ffffff] shadow-sm">
