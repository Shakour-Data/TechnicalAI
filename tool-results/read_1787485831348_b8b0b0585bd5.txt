  1114→  1114→/**
  1115→  1115→ * Deterministic seeded shuffle using Fisher-Yates.
  1116→  1116→ */
  1117→  1117→function seededShuffle<T>(array: T[], seed: number): T[] {
  1118→  1118→  const result = [...array];
  1119→  1119→  let s = seed;
  1120→  1120→  for (let i = result.length - 1; i > 0; i--) {
  1121→  1121→    s = (s * 1664525 + 1013904223) & 0xffffffff;
  1122→  1122→    const j = s % (i + 1);
  1123→  1123→    [result[i], result[j]] = [result[j], result[i]];
  1124→  1124→  }
  1125→  1125→  return result;
  1126→  1126→}

  1129→  1129→ * Get `count` phrases from a section/subcategory, deterministically shuffled by seed.
