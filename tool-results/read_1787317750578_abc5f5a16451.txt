   132→   132→function sma(closes: number[], period: number): number {
   133→   133→  if (closes.length < period) return 0;
   134→   134→  const slice = closes.slice(-period);
   135→   135→  return slice.reduce((a, b) => a + b, 0) / period;
   136→   136→}

   138→   138→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   139→   139→function emaCalc(closes: number[], period: number): number {
   140→   140→  if (closes.length < period) return 0;
   141→   141→  const k = 2 / (period + 1);
   142→   142→  let emaVal = sma(closes.slice(0, period), period);
   143→   143→  for (let i = period; i < closes.length; i++) {
   144→   144→    emaVal = closes[i] * k + emaVal * (1 - k);
   145→   145→  }
   146→   146→  return emaVal;
   147→   147→}

   149→   149→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   150→   150→function calcRSI(closes: number[], period: number = 14): number {
   151→   151→  if (closes.length < period + 1) return 50;
   152→   152→  let avgGain = 0;
   153→   153→  let avgLoss = 0;
   154→   154→  for (let i = 1; i <= period; i++) {
   155→   155→    const diff = closes[i] - closes[i - 1];
   156→   156→    if (diff > 0) avgGain += diff;
   157→   157→    else avgLoss += Math.abs(diff);
   158→   158→  }
   159→   159→  avgGain /= period;
   160→   160→  avgLoss /= period;
   161→   161→  for (let i = period + 1; i < closes.length; i++) {
   162→   162→    const diff = closes[i] - closes[i - 1];
   163→   163→    const gain = diff > 0 ? diff : 0;
   164→   164→    const loss = diff < 0 ? Math.abs(diff) : 0;
   165→   165→    avgGain = (avgGain * (period - 1) + gain) / period;
   166→   166→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   167→   167→  }
   168→   168→  if (avgLoss === 0) return 100;
   169→   169→  const rs = avgGain / avgLoss;
   170→   170→  return 100 - 100 / (1 + rs);
   171→   171→}

   173→   173→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   174→   174→function calcMFI(data: OHLCV[], period: number = 14): number {
   175→   175→  if (data.length < period + 1) return 50;
   176→   176→  let posFlow = 0;
   177→   177→  let negFlow = 0;
   178→   178→  for (let i = data.length - period; i < data.length; i++) {
   179→   179→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   180→   180→    const mf = tp * data[i].volume;
   181→   181→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   182→   182→    if (tp > prevTp) posFlow += mf;
   183→   183→    else negFlow += mf;
   184→   184→  }
   185→   185→  if (negFlow === 0) return 100;
   186→   186→  const mfr = posFlow / negFlow;
   187→   187→  return 100 - 100 / (1 + mfr);
   188→   188→}

   191→   191→function calcCCI(data: OHLCV[], period: number = 20): number {
   192→   192→  if (data.length < period) return 0;
   193→   193→  const tps: number[] = [];
   194→   194→  for (let i = data.length - period; i < data.length; i++) {
   195→   195→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   196→   196→  }
   197→   197→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   198→   198→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   199→   199→  if (meanDev === 0) return 0;
   200→   200→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   201→   201→}

   204→   204→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   205→   205→  if (data.length < kPeriod) return { k: 50, d: 50 };
   206→   206→  const rawKs: number[] = [];
   207→   207→  for (let i = kPeriod - 1; i < data.length; i++) {
   208→   208→    let lowest = Infinity;
   209→   209→    let highest = -Infinity;
   210→   210→    for (let j = i - kPeriod + 1; j <= i; j++) {
   211→   211→      if (data[j].low < lowest) lowest = data[j].low;
   212→   212→      if (data[j].high > highest) highest = data[j].high;
   213→   213→    }
   214→   214→    const range = highest - lowest;
   215→   215→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   216→   216→  }
   217→   217→  const smoothedK: number[] = [];
   218→   218→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   219→   219→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   220→   220→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   221→   221→  }
   222→   222→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   223→   223→  const dSlice = smoothedK.slice(-smoothD);
   224→   224→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   225→   225→  return { k, d };
   226→   226→}

   229→   229→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   230→   230→  if (data.length < period) return -50;
   231→   231→  const slice = data.slice(-period);
   232→   232→  let highest = -Infinity;
   233→   233→  let lowest = Infinity;
   234→   234→  for (const c of slice) {
   235→   235→    if (c.high > highest) highest = c.high;
   236→   236→    if (c.low < lowest) lowest = c.low;
   237→   237→  }
   238→   238→  const range = highest - lowest;
   239→   239→  if (range === 0) return -50;
   240→   240→  return ((highest - data[data.length - 1].close) / range) * -100;
   241→   241→}

   243→   243→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   244→   244→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   245→   245→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   246→   246→  const emaFast = emaCalc(closes, fast);
   247→   247→  const emaSlow = emaCalc(closes, slow);
   248→   248→  const macdLine = emaFast - emaSlow;
   249→   249→  // Calculate signal as EMA of MACD values
   250→   250→  const macdValues: number[] = [];
   251→   251→  const kFast = 2 / (fast + 1);
   252→   252→  const kSlow = 2 / (slow + 1);
   253→   253→  let eFast = sma(closes.slice(0, fast), fast);
   254→   254→  let eSlow = sma(closes.slice(0, slow), slow);
   255→   255→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   256→   256→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   257→   257→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   258→   258→    macdValues.push(eFast - eSlow);
   259→   259→  }
   260→   260→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   261→   261→  const kSig = 2 / (sig + 1);
   262→   262→  let signalVal = sma(macdValues.slice(0, sig), sig);
   263→   263→  for (let i = sig; i < macdValues.length; i++) {
   264→   264→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   265→   265→  }
   266→   266→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   267→   267→}

   269→   269→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   270→   270→function calcATR(data: OHLCV[], period: number = 14): number {
   271→   271→  if (data.length < 2) return 0;
   272→   272→  const trs: number[] = [];
   273→   273→  for (let i = 1; i < data.length; i++) {
   274→   274→    const tr = Math.max(
   275→   275→      data[i].high - data[i].low,
   276→   276→      Math.abs(data[i].high - data[i - 1].close),
   277→   277→      Math.abs(data[i].low - data[i - 1].close)
   278→   278→    );
   279→   279→    trs.push(tr);
   280→   280→  }
   281→   281→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   282→   282→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   283→   283→  for (let i = period; i < trs.length; i++) {
   284→   284→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   285→   285→  }
   286→   286→  return atrVal;
   287→   287→}

   289→   289→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   290→   290→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   291→   291→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   292→   292→  const plusDMs: number[] = [];
   293→   293→  const minusDMs: number[] = [];
   294→   294→  const trs: number[] = [];
   295→   295→  for (let i = 1; i < data.length; i++) {
   296→   296→    const upMove = data[i].high - data[i - 1].high;
   297→   297→    const downMove = data[i - 1].low - data[i].low;
   298→   298→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   299→   299→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   300→   300→    trs.push(Math.max(
   301→   301→      data[i].high - data[i].low,
   302→   302→      Math.abs(data[i].high - data[i - 1].close),
   303→   303→      Math.abs(data[i].low - data[i - 1].close)
   304→   304→    ));
   305→   305→  }
   306→   306→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   307→   307→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   308→   308→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   309→   309→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   310→   310→  const dxValues: number[] = [];
   311→   311→  for (let i = period; i < trs.length; i++) {
   312→   312→    smoothTR = smoothTR - smoothTR / period + trs[i];
   313→   313→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   314→   314→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   315→   315→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   316→   316→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   317→   317→    const diSum = diPlus + diMinus;
   318→   318→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   319→   319→    dxValues.push(dx);
   320→   320→  }
   321→   321→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   322→   322→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   323→   323→  for (let i = period; i < dxValues.length; i++) {
   324→   324→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   325→   325→  }
   326→   326→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   327→   327→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   328→   328→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   329→   329→}

   331→   331→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   332→   332→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   333→   333→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   334→   334→  let isLong = true;
   335→   335→  let af = step;
   336→   336→  let ep = data[0].high;
   337→   337→  let sar = data[0].low;
   338→   338→  for (let i = 1; i < data.length; i++) {
   339→   339→    const prevSar = sar;
   340→   340→    sar = prevSar + af * (ep - prevSar);
   341→   341→    // Ensure SAR is within bounds
   342→   342→    if (isLong) {
   343→   343→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   344→   344→      if (data[i].low < sar) {
   345→   345→        isLong = false;
   346→   346→        sar = ep;
   347→   347→        ep = data[i].low;
   348→   348→        af = step;
   349→   349→      } else {
   350→   350→        if (data[i].high > ep) {
   351→   351→          ep = data[i].high;
   352→   352→          af = Math.min(af + step, max);
   353→   353→        }
   354→   354→      }
   355→   355→    } else {
   356→   356→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   357→   357→      if (data[i].high > sar) {
   358→   358→        isLong = true;
   359→   359→        sar = ep;
   360→   360→        ep = data[i].high;
   361→   361→        af = step;
   362→   362→      } else {
   363→   363→        if (data[i].low < ep) {
   364→   364→          ep = data[i].low;
   365→   365→          af = Math.min(af + step, max);
   366→   366→        }
   367→   367→      }
   368→   368→    }
   369→   369→  }
   370→   370→  return sar;
   371→   371→}

   374→   374→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   375→   375→  if (closes.length < period) {
   376→   376→    const c = closes[closes.length - 1] ?? 0;
   377→   377→    return { upper: c, middle: c, lower: c };
   378→   378→  }
   379→   379→  const slice = closes.slice(-period);
   380→   380→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   381→   381→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   382→   382→  const sd = Math.sqrt(variance);
   383→   383→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   384→   384→}

   387→   387→function calcOBV(data: OHLCV[]): number {
   388→   388→  let obv = 0;
   389→   389→  for (let i = 1; i < data.length; i++) {
   390→   390→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   391→   391→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   392→   392→  }
   393→   393→  return obv;
   394→   394→}

   396→   396→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   397→   397→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   398→   398→  const n = values.length;
   399→   399→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   400→   400→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   401→   401→  for (let i = 0; i < n; i++) {
   402→   402→    sumX += i;
   403→   403→    sumY += values[i];
   404→   404→    sumXY += i * values[i];
   405→   405→    sumX2 += i * i;
   406→   406→    sumY2 += values[i] * values[i];
   407→   407→  }
   408→   408→  const denom = n * sumX2 - sumX * sumX;
   409→   409→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   410→   410→  const slope = (n * sumXY - sumX * sumY) / denom;
   411→   411→  const intercept = (sumY - slope * sumX) / n;
   412→   412→  const meanY = sumY / n;
   413→   413→  let ssRes = 0, ssTot = 0;
   414→   414→  for (let i = 0; i < n; i++) {
   415→   415→    const predicted = slope * i + intercept;
   416→   416→    ssRes += (values[i] - predicted) ** 2;
   417→   417→    ssTot += (values[i] - meanY) ** 2;
   418→   418→  }
   419→   419→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   420→   420→  return { slope, intercept, r2 };
   421→   421→}

   423→   423→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   424→   424→function roundToNice(price: number): number {
   425→   425→  const abs = Math.abs(price);
   426→   426→  let step: number;
   427→   427→  if (abs >= 100000) step = 100;
   428→   428→  else if (abs >= 10000) step = 50;
   429→   429→  else if (abs >= 1000) step = 10;
   430→   430→  else if (abs >= 100) step = 5;
   431→   431→  else step = 1;
   432→   432→  return Math.round(price / step) * step;
   433→   433→}

   477→   477→function calcTrend(closes: number[], period: number): TrendResult {
   478→   478→  const slice = closes.slice(-Math.min(period, closes.length));
   479→   479→  const { slope, r2 } = linearRegression(slice);
   480→   480→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   481→   481→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   482→   482→  const priceThreshold = avgPrice * 0.001; // 0.1%
   483→   483→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   484→   484→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   485→   485→}

   564→   564→  const R5_level = resistances[4] ?? price * 1.30;
