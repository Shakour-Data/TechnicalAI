    84→    84→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
    85→    85→function sma(closes: number[], period: number): number {
    86→    86→  if (closes.length < period) return 0;
    87→    87→  const slice = closes.slice(-period);
    88→    88→  return slice.reduce((a, b) => a + b, 0) / period;

    91→    91→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
    92→    92→function emaCalc(closes: number[], period: number): number {
    93→    93→  if (closes.length < period) return 0;
    94→    94→  const k = 2 / (period + 1);
    95→    95→  let emaVal = sma(closes.slice(0, period), period);
    96→    96→  for (let i = period; i < closes.length; i++) {
    97→    97→    emaVal = closes[i] * k + emaVal * (1 - k);
    98→    98→  }
    99→    99→  return emaVal;
   100→   100→}

   102→   102→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   103→   103→function calcRSI(closes: number[], period: number = 14): number {
   104→   104→  if (closes.length < period + 1) return 50;
   105→   105→  let avgGain = 0;
   106→   106→  let avgLoss = 0;
   107→   107→  for (let i = 1; i <= period; i++) {
   108→   108→    const diff = closes[i] - closes[i - 1];
   109→   109→    if (diff > 0) avgGain += diff;
   110→   110→    else avgLoss += Math.abs(diff);
   111→   111→  }
   112→   112→  avgGain /= period;
   113→   113→  avgLoss /= period;
   114→   114→  for (let i = period + 1; i < closes.length; i++) {
   115→   115→    const diff = closes[i] - closes[i - 1];
   116→   116→    const gain = diff > 0 ? diff : 0;
   117→   117→    const loss = diff < 0 ? Math.abs(diff) : 0;
   118→   118→    avgGain = (avgGain * (period - 1) + gain) / period;
   119→   119→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   120→   120→  }
   121→   121→  if (avgLoss === 0) return 100;
   122→   122→  const rs = avgGain / avgLoss;
   123→   123→  return 100 - 100 / (1 + rs);
   124→   124→}

   126→   126→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   127→   127→function calcMFI(data: OHLCV[], period: number = 14): number {
   128→   128→  if (data.length < period + 1) return 50;
   129→   129→  let posFlow = 0;
   130→   130→  let negFlow = 0;
   131→   131→  for (let i = data.length - period; i < data.length; i++) {
   132→   132→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   133→   133→    const mf = tp * data[i].volume;
   134→   134→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   135→   135→    if (tp > prevTp) posFlow += mf;
   136→   136→    else negFlow += mf;
   137→   137→  }
   138→   138→  if (negFlow === 0) return 100;
   139→   139→  const mfr = posFlow / negFlow;
   140→   140→  return 100 - 100 / (1 + mfr);
   141→   141→}

   144→   144→function calcCCI(data: OHLCV[], period: number = 20): number {
   145→   145→  if (data.length < period) return 0;
   146→   146→  const tps: number[] = [];
   147→   147→  for (let i = data.length - period; i < data.length; i++) {
   148→   148→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   149→   149→  }
   150→   150→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   151→   151→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   152→   152→  if (meanDev === 0) return 0;
   153→   153→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   154→   154→}

   157→   157→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   158→   158→  if (data.length < kPeriod) return { k: 50, d: 50 };
   159→   159→  const rawKs: number[] = [];
   160→   160→  for (let i = kPeriod - 1; i < data.length; i++) {
   161→   161→    let lowest = Infinity;
   162→   162→    let highest = -Infinity;
   163→   163→    for (let j = i - kPeriod + 1; j <= i; j++) {
   164→   164→      if (data[j].low < lowest) lowest = data[j].low;
   165→   165→      if (data[j].high > highest) highest = data[j].high;
   166→   166→    }
   167→   167→    const range = highest - lowest;
   168→   168→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   169→   169→  }
   170→   170→  const smoothedK: number[] = [];
   171→   171→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   172→   172→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   173→   173→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   174→   174→  }
   175→   175→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   176→   176→  const dSlice = smoothedK.slice(-smoothD);
   177→   177→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   178→   178→  return { k, d };
   179→   179→}

   182→   182→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   183→   183→  if (data.length < period) return -50;
   184→   184→  const slice = data.slice(-period);
   185→   185→  let highest = -Infinity;
   186→   186→  let lowest = Infinity;
   187→   187→  for (const c of slice) {
   188→   188→    if (c.high > highest) highest = c.high;
   189→   189→    if (c.low < lowest) lowest = c.low;
   190→   190→  }
   191→   191→  const range = highest - lowest;
   192→   192→  if (range === 0) return -50;
   193→   193→  return ((highest - data[data.length - 1].close) / range) * -100;
   194→   194→}

   196→   196→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   197→   197→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   198→   198→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   199→   199→  const emaFast = emaCalc(closes, fast);
   200→   200→  const emaSlow = emaCalc(closes, slow);
   201→   201→  const macdLine = emaFast - emaSlow;
   202→   202→  // Calculate signal as EMA of MACD values
   203→   203→  const macdValues: number[] = [];
   204→   204→  const kFast = 2 / (fast + 1);
   205→   205→  const kSlow = 2 / (slow + 1);
   206→   206→  let eFast = sma(closes.slice(0, fast), fast);
   207→   207→  let eSlow = sma(closes.slice(0, slow), slow);
   208→   208→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   209→   209→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   210→   210→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   211→   211→    macdValues.push(eFast - eSlow);
   212→   212→  }
   213→   213→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   214→   214→  const kSig = 2 / (sig + 1);
   215→   215→  let signalVal = sma(macdValues.slice(0, sig), sig);
   216→   216→  for (let i = sig; i < macdValues.length; i++) {
   217→   217→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   218→   218→  }
   219→   219→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   220→   220→}

   222→   222→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   223→   223→function calcATR(data: OHLCV[], period: number = 14): number {
   224→   224→  if (data.length < 2) return 0;
   225→   225→  const trs: number[] = [];
   226→   226→  for (let i = 1; i < data.length; i++) {
   227→   227→    const tr = Math.max(
   228→   228→      data[i].high - data[i].low,
   229→   229→      Math.abs(data[i].high - data[i - 1].close),
   230→   230→      Math.abs(data[i].low - data[i - 1].close)
   231→   231→    );
   232→   232→    trs.push(tr);
   233→   233→  }
   234→   234→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   235→   235→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   236→   236→  for (let i = period; i < trs.length; i++) {
   237→   237→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   238→   238→  }
   239→   239→  return atrVal;
   240→   240→}

   242→   242→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   243→   243→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   244→   244→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   245→   245→  const plusDMs: number[] = [];
   246→   246→  const minusDMs: number[] = [];
   247→   247→  const trs: number[] = [];
   248→   248→  for (let i = 1; i < data.length; i++) {
   249→   249→    const upMove = data[i].high - data[i - 1].high;
   250→   250→    const downMove = data[i - 1].low - data[i].low;
   251→   251→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   252→   252→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   253→   253→    trs.push(Math.max(
   254→   254→      data[i].high - data[i].low,
   255→   255→      Math.abs(data[i].high - data[i - 1].close),
   256→   256→      Math.abs(data[i].low - data[i - 1].close)
   257→   257→    ));
   258→   258→  }
   259→   259→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   260→   260→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   261→   261→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   262→   262→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   263→   263→  const dxValues: number[] = [];
   264→   264→  for (let i = period; i < trs.length; i++) {
   265→   265→    smoothTR = smoothTR - smoothTR / period + trs[i];
   266→   266→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   267→   267→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   268→   268→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   269→   269→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   270→   270→    const diSum = diPlus + diMinus;
   271→   271→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   272→   272→    dxValues.push(dx);
   273→   273→  }
   274→   274→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   275→   275→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   276→   276→  for (let i = period; i < dxValues.length; i++) {
   277→   277→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   278→   278→  }
   279→   279→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   280→   280→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   281→   281→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   282→   282→}

   284→   284→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   285→   285→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   286→   286→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   287→   287→  let isLong = true;
   288→   288→  let af = step;
   289→   289→  let ep = data[0].high;
   290→   290→  let sar = data[0].low;
   291→   291→  for (let i = 1; i < data.length; i++) {
   292→   292→    const prevSar = sar;
   293→   293→    sar = prevSar + af * (ep - prevSar);
   294→   294→    // Ensure SAR is within bounds
   295→   295→    if (isLong) {
   296→   296→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   297→   297→      if (data[i].low < sar) {
   298→   298→        isLong = false;
   299→   299→        sar = ep;
   300→   300→        ep = data[i].low;
   301→   301→        af = step;
   302→   302→      } else {
   303→   303→        if (data[i].high > ep) {
   304→   304→          ep = data[i].high;
   305→   305→          af = Math.min(af + step, max);
   306→   306→        }
   307→   307→      }
   308→   308→    } else {
   309→   309→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   310→   310→      if (data[i].high > sar) {
   311→   311→        isLong = true;
   312→   312→        sar = ep;
   313→   313→        ep = data[i].high;
   314→   314→        af = step;
   315→   315→      } else {
   316→   316→        if (data[i].low < ep) {
   317→   317→          ep = data[i].low;
   318→   318→          af = Math.min(af + step, max);
   319→   319→        }
   320→   320→      }
   321→   321→    }
   322→   322→  }
   323→   323→  return sar;
   324→   324→}

   327→   327→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   328→   328→  if (closes.length < period) {
   329→   329→    const c = closes[closes.length - 1] ?? 0;
   330→   330→    return { upper: c, middle: c, lower: c };
   331→   331→  }
   332→   332→  const slice = closes.slice(-period);
   333→   333→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   334→   334→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   335→   335→  const sd = Math.sqrt(variance);
   336→   336→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   337→   337→}

   340→   340→function calcOBV(data: OHLCV[]): number {

   342→   342→  for (let i = 1; i < data.length; i++) {
   343→   343→    if (data[i].close > data[i - 1].close) obv += data[i].volume;

   345→   345→  }
   346→   346→  return obv;

   349→   349→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   350→   350→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   351→   351→  const n = values.length;
   352→   352→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   353→   353→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   354→   354→  for (let i = 0; i < n; i++) {
   355→   355→    sumX += i;
   356→   356→    sumY += values[i];
   357→   357→    sumXY += i * values[i];
   358→   358→    sumX2 += i * i;
   359→   359→    sumY2 += values[i] * values[i];
   360→   360→  }
   361→   361→  const denom = n * sumX2 - sumX * sumX;
   362→   362→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   363→   363→  const slope = (n * sumXY - sumX * sumY) / denom;
   364→   364→  const intercept = (sumY - slope * sumX) / n;
   365→   365→  const meanY = sumY / n;
   366→   366→  let ssRes = 0, ssTot = 0;
   367→   367→  for (let i = 0; i < n; i++) {
   368→   368→    const predicted = slope * i + intercept;
   369→   369→    ssRes += (values[i] - predicted) ** 2;
   370→   370→    ssTot += (values[i] - meanY) ** 2;
   371→   371→  }
   372→   372→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   373→   373→  return { slope, intercept, r2 };
   374→   374→}

   377→   377→function roundToNice(price: number): number {
   378→   378→  const abs = Math.abs(price);
   379→   379→  let step: number;
   380→   380→  if (abs >= 100000) step = 100;
   381→   381→  else if (abs >= 10000) step = 50;
   382→   382→  else if (abs >= 1000) step = 10;
   383→   383→  else if (abs >= 100) step = 5;
   384→   384→  else step = 1;
   385→   385→  return Math.round(price / step) * step;
   386→   386→}

   388→   388→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
   389→   389→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: number[]; lows: number[] } {
   390→   390→  const highs: number[] = [];
   391→   391→  const lows: number[] = [];
   392→   392→  for (let i = lookback; i < data.length - lookback; i++) {
   393→   393→    let isHigh = true;
   394→   394→    let isLow = true;
   395→   395→    for (let j = i - lookback; j <= i + lookback; j++) {
   396→   396→      if (j === i) continue;
   397→   397→      if (data[j].high >= data[i].high) isHigh = false;
   398→   398→      if (data[j].low <= data[i].low) isLow = false;
   399→   399→    }
   400→   400→    if (isHigh) highs.push(data[i].high);
   401→   401→    if (isLow) lows.push(data[i].low);
   402→   402→  }
   403→   403→  return { highs, lows };
   404→   404→}

   406→   406→// ─── Support / Resistance (6 each, adaptive gap, multi-factor strength) ───
   407→   407→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   408→   408→  resistances: number[];
   409→   409→  supports: number[];
   410→   410→  supportStrengths: LevelStrength[];
   411→   411→  resistanceStrengths: LevelStrength[];
   412→   412→  priceTargets: LevelStrength[];
   413→   413→} {
   414→   414→  const closes = data.map(d => d.close);
   415→   415→  const avgVolume = data.reduce((s, d) => s + d.volume, 0) / data.length;
   416→   416→
   417→   417→  // Tag each raw level with its source type for confluence scoring
   418→   418→  interface TaggedLevel { price: number; source: string };
   419→   419→  const allRaw: TaggedLevel[] = [];
   420→   420→
   421→   421→  // 1. Pivot Points (source: 'pivot')
   422→   422→  const prev = data[data.length - 2] ?? data[data.length - 1];
   423→   423→  if (prev) {
   424→   424→    const pp = (prev.high + prev.low + prev.close) / 3;
   425→   425→    const hl = prev.high - prev.low;
   426→   426→    const pivotR = [
   427→   427→      2 * pp - prev.low, pp + hl,
   428→   428→      prev.high + 2 * (pp - prev.low),
   429→   429→      prev.high + 2 * (pp - prev.low) + hl,
   430→   430→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   431→   431→    ];
   432→   432→    const pivotS = [
   433→   433→      2 * pp - prev.high, pp - hl,
   434→   434→      prev.low - 2 * (prev.high - pp),
   435→   435→      prev.low - 2 * (prev.high - pp) - hl,
   436→   436→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   437→   437→    ];
   438→   438→    for (const p of pivotR) allRaw.push({ price: p, source: 'pivot' });
   439→   439→    for (const p of pivotS) allRaw.push({ price: p, source: 'pivot' });
   440→   440→  }
   441→   441→
   442→   442→  // 2. Swing levels (source: 'swing')
   443→   443→  for (const lb of [3, 5, 7, 10]) {
   444→   444→    const swings = findSwingLevels(data, lb);
   445→   445→    for (const h of swings.highs) allRaw.push({ price: h, source: 'swing' });
   446→   446→    for (const l of swings.lows) allRaw.push({ price: l, source: 'swing' });
   447→   447→  }
   448→   448→
   449→   449→  // 3. Moving Average levels (source: 'ma')
   450→   450→  for (const period of [5, 10, 21, 50, 100, 200]) {
   451→   451→    const v = sma(closes, period);
   452→   452→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01)
   453→   453→      allRaw.push({ price: v, source: 'ma' });
   454→   454→  }
   455→   455→  const ema12 = emaCalc(closes, 12);
   456→   456→  const ema26 = emaCalc(closes, 26);
   457→   457→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01)
   458→   458→    allRaw.push({ price: ema12, source: 'ma' });
   459→   459→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01)
   460→   460→    allRaw.push({ price: ema26, source: 'ma' });
   461→   461→
   462→   462→  // 4. Bollinger Band levels (source: 'bb')
   463→   463→  const bbLevels = calcBollingerBands(closes);
   464→   464→  if (bbLevels.upper > 0) allRaw.push({ price: bbLevels.upper, source: 'bb' });
   465→   465→  if (bbLevels.lower > 0) allRaw.push({ price: bbLevels.lower, source: 'bb' });
   466→   466→
   467→   467→  // 5. Round number / psychological levels (source: 'round')
   468→   468→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
   469→   469→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
   470→   470→    const p = Math.round(currentPrice * mult / magnitude) * magnitude;
   471→   471→    allRaw.push({ price: p, source: 'round' });
   472→   472→  }
   473→   473→
   474→   474→  // 6. Recent high/low zones (source: 'hilo')
   475→   475→  for (const w of [5, 22, 66]) {
   476→   476→    const slice = data.slice(-Math.min(w, data.length));
   477→   477→    if (slice.length > 0) {
   478→   478→      allRaw.push({ price: Math.max(...slice.map(d => d.high)), source: 'hilo' });
   479→   479→      allRaw.push({ price: Math.min(...slice.map(d => d.low)), source: 'hilo' });
   480→   480→    }
   481→   481→  }
   482→   482→
   483→   483→  // 7. Round all and filter invalid
   484→   484→  const rounded: TaggedLevel[] = allRaw
   485→   485→    .map(l => ({ price: roundToNice(l.price), source: l.source }))
   486→   486→    .filter(l => l.price > 0);
   487→   487→
   488→   488→  // 8. Cluster nearby levels (within 0.5%) — merge sources, keep most common price
   489→   489→  interface Cluster {
   490→   490→    price: number;
   491→   491→    sources: Set<string>;
   492→   492→    touchCount: number;
   493→   493→    totalVolumeAtTouches: number;
   494→   494→    mostRecentTouchBar: number;
   495→   495→  }
   496→   496→  const clusters: Cluster[] = [];
   497→   497→  for (const lvl of rounded) {
   498→   498→    let merged = false;
   499→   499→    for (const c of clusters) {
   500→   500→      if (Math.abs(c.price - lvl.price) / Math.max(c.price, 1) < 0.005) {
   501→   501→        c.sources.add(lvl.source);
   502→   502→        merged = true;
   503→   503→        break;
   504→   504→      }
   505→   505→    }
   506→   506→    if (!merged) {
   507→   507→      clusters.push({ price: lvl.price, sources: new Set([lvl.source]), touchCount: 0, totalVolumeAtTouches: 0, mostRecentTouchBar: -1 });
   508→   508→    }
   509→   509→  }
   510→   510→
   511→   511→  // 9. Count actual price TOUCHES (rejections) and volume at each cluster level
   512→   512→  // A real touch = bar wick reached the level AND bar body closed away from it
   513→   513→  const touchThreshold = currentPrice * 0.004; // 0.4% tight proximity
   514→   514→  for (let i = 0; i < data.length; i++) {
   515→   515→    const bar = data[i];
   516→   516→    for (const c of clusters) {
   517→   517→      const isResistance = c.price > currentPrice;
   518→   518→      if (isResistance) {
   519→   519→        // Resistance: bar high reached near the level AND closed below it (rejection)
   520→   520→        if (bar.high >= c.price - touchThreshold && bar.close < c.price) {
   521→   521→          c.touchCount++;
   522→   522→          c.totalVolumeAtTouches += bar.volume;
   523→   523→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   524→   524→        }
   525→   525→      } else {
   526→   526→        // Support: bar low reached near the level AND closed above it (rejection)
   527→   527→        if (bar.low <= c.price + touchThreshold && bar.close > c.price) {
   528→   528→          c.touchCount++;
   529→   529→          c.totalVolumeAtTouches += bar.volume;
   530→   530→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   531→   531→        }
   532→   532→      }
   533→   533→    }
   534→   534→  }
   535→   535→
   536→   536→  // 10. Calculate multi-factor strength (0-10)
   537→   537→  const levelStrengths: LevelStrength[] = clusters.map(c => {
   538→   538→    // Factor 1: Source Confluence (0-2.5 pts)
   539→   539→    const sourceScore = Math.min(2.5, c.sources.size * 0.5);
   540→   540→    const hasSwingAndPivot = c.sources.has('swing') && c.sources.has('pivot') ? 0.5 : 0;
   541→   541→    const maConfluence = c.sources.has('ma') ? 0.5 : 0;
   542→   542→    const confluenceScore = Math.min(2.5, sourceScore + hasSwingAndPivot + maConfluence);
   543→   543→
   544→   544→    // Factor 2: Touch Count — CAPPED by recency (0-2 pts)
   545→   545→    // Only count touches in the most recent 50% of data to avoid ancient history inflating
   546→   546→    const recentBarStart = Math.floor(data.length * 0.5);
   547→   547→    const recentTouches = c.mostRecentTouchBar >= recentBarStart ? c.touchCount : Math.max(0, c.touchCount - 1);
   548→   548→    let touchScore: number;
   549→   549→    if (recentTouches === 0) touchScore = 0;
   550→   550→    else if (recentTouches === 1) touchScore = 0.4;
   551→   551→    else if (recentTouches <= 2) touchScore = 0.8;
   552→   552→    else if (recentTouches <= 4) touchScore = 1.2;
   553→   553→    else if (recentTouches <= 7) touchScore = 1.6;
   554→   554→    else touchScore = 2.0;
   555→   555→
   556→   556→    // Factor 3: Volume Confirmation (0-1.5 pts)
   557→   557→    let volumeScore = 0;
   558→   558→    if (c.touchCount > 0) {
   559→   559→      const avgVolAtTouch = c.totalVolumeAtTouches / c.touchCount;
   560→   560→      const volRatio = avgVolume > 0 ? avgVolAtTouch / avgVolume : 1;
   561→   561→      if (volRatio > 2.0) volumeScore = 1.5;
   562→   562→      else if (volRatio > 1.5) volumeScore = 1.2;
   563→   563→      else if (volRatio > 1.0) volumeScore = 0.8;
   564→   564→      else if (volRatio > 0.7) volumeScore = 0.4;
   565→   565→      else volumeScore = 0.2;
   566→   566→    }
   567→   567→
   568→   568→    // Factor 4: Freshness / Recency (0-1.5 pts)
   569→   569→    let freshnessScore = 0;
   570→   570→    if (c.touchCount > 0 && data.length > 0) {
   571→   571→      const barsAgo = data.length - 1 - c.mostRecentTouchBar;
   572→   572→      if (barsAgo <= 3) freshnessScore = 1.5;
   573→   573→      else if (barsAgo <= 10) freshnessScore = 1.2;
   574→   574→      else if (barsAgo <= 25) freshnessScore = 0.8;
   575→   575→      else if (barsAgo <= 50) freshnessScore = 0.4;
   576→   576→      else freshnessScore = 0.1;
   577→   577→    }
   578→   578→
   579→   579→    // Factor 5: Proximity to current price (0-1 pt)
   580→   580→    const distRatio = Math.abs(c.price - currentPrice) / currentPrice;
   581→   581→    let proximityScore = 0;
   582→   582→    if (distRatio < 0.015) proximityScore = 1.0;
   583→   583→    else if (distRatio < 0.03) proximityScore = 0.7;
   584→   584→    else if (distRatio < 0.06) proximityScore = 0.4;
   585→   585→    else if (distRatio < 0.10) proximityScore = 0.2;
   586→   586→
   587→   587→    const total = confluenceScore + touchScore + volumeScore + freshnessScore + proximityScore;
   588→   588→    const strength = Math.min(10, Math.max(1, Math.round(total)));
   589→   589→
   590→   590→    return { price: c.price, strength, isTarget: false };
   591→   591→  });
   592→   592→
   593→   593→  // 11. Separate above/below current price
   594→   594→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   595→   595→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   596→   596→
   597→   597→  // 12. Adaptive gap enforcement — keep highest strength in cluster
   598→   598→  function enforceGap(levels: LevelStrength[], targetCount: number): LevelStrength[] {
   599→   599→    const gapSteps = [0.05, 0.04, 0.03, 0.025, 0.02, 0.015, 0.01, 0.008, 0.005];
   600→   600→    for (const gapPct of gapSteps) {
   601→   601→      if (levels.length === 0) return [];
   602→   602→      const result: LevelStrength[] = [levels[0]];
   603→   603→      for (let i = 1; i < levels.length; i++) {
   604→   604→        if (result.length >= targetCount) break;
   605→   605→        const prevPrice = result[result.length - 1].price;
   606→   606→        const currPrice = levels[i].price;
   607→   607→        const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   608→   608→        if (gap >= gapPct) {
   609→   609→          result.push(levels[i]);
   610→   610→        } else {
   611→   611→          // Merge: keep stronger level's price, add 0.5 for confluence
   612→   612→          const keep = levels[i].strength >= result[result.length - 1].strength ? levels[i] : result[result.length - 1];
   613→   613→          const mergedStrength = Math.min(10, Math.max(keep.strength, result[result.length - 1].strength) + 0.5);
   614→   614→          result[result.length - 1] = { ...keep, strength: Math.round(mergedStrength) };
   615→   615→        }
   616→   616→      }
   617→   617→      if (result.length >= targetCount) return result;
   618→   618→    }
   619→   619→    return levels.slice(0, targetCount);
   620→   620→  }
   621→   621→
   622→   622→  const finalResistances = enforceGap(above, 6);
   623→   623→  const finalSupports = enforceGap(below, 6);
   624→   624→
   625→   625→  // 13. Determine price targets: top 2 by strength, or any above 7
   626→   626→  const allLevels = [...finalResistances, ...finalSupports];
   627→   627→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   628→   628→  let targets: LevelStrength[];
   629→   629→  if (sortedByStrength.some(l => l.strength >= 7)) {
   630→   630→    targets = sortedByStrength.filter(l => l.strength >= 7).slice(0, 3);
   631→   631→  } else {
   632→   632→    targets = sortedByStrength.slice(0, 2);
   633→   633→  }
   634→   634→
   635→   635→  // Mark targets
   636→   636→  targets = targets.map(t => ({ ...t, isTarget: true }));
   637→   637→
   638→   638→  // Also mark in final arrays
   639→   639→  finalResistances.forEach(r => {
   640→   640→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   641→   641→  });
   642→   642→  finalSupports.forEach(s => {
   643→   643→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   644→   644→  });
   645→   645→
   646→   646→  return {
   647→   647→    resistances: finalResistances.map(l => l.price),
   648→   648→    supports: finalSupports.map(l => l.price),
   649→   649→    supportStrengths: finalSupports,
   650→   650→    resistanceStrengths: finalResistances,
   651→   651→    priceTargets: targets,
   652→   652→  };
   653→   653→}

   656→   656→function calcTrend(closes: number[], period: number): TrendResult {
   657→   657→  const slice = closes.slice(-Math.min(period, closes.length));
   658→   658→  const { slope, r2 } = linearRegression(slice);
   659→   659→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   660→   660→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   661→   661→  const priceThreshold = avgPrice * 0.001; // 0.1%
   662→   662→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   663→   663→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   664→   664→}
