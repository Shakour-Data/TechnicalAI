   181→   181→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   182→   182→function sma(closes: number[], period: number): number {
   183→   183→  if (closes.length < period) return 0;
   184→   184→  const slice = closes.slice(-period);
   185→   185→  return slice.reduce((a, b) => a + b, 0) / period;
   186→   186→}

   188→   188→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   189→   189→function emaCalc(closes: number[], period: number): number {
   190→   190→  if (closes.length < period) return 0;
   191→   191→  const k = 2 / (period + 1);
   192→   192→  let emaVal = sma(closes.slice(0, period), period);
   193→   193→  for (let i = period; i < closes.length; i++) {
   194→   194→    emaVal = closes[i] * k + emaVal * (1 - k);
   195→   195→  }
   196→   196→  return emaVal;
   197→   197→}

   199→   199→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   200→   200→function calcRSI(closes: number[], period: number = 14): number {
   201→   201→  if (closes.length < period + 1) return 50;
   202→   202→  let avgGain = 0;
   203→   203→  let avgLoss = 0;
   204→   204→  for (let i = 1; i <= period; i++) {
   205→   205→    const diff = closes[i] - closes[i - 1];
   206→   206→    if (diff > 0) avgGain += diff;
   207→   207→    else avgLoss += Math.abs(diff);
   208→   208→  }
   209→   209→  avgGain /= period;
   210→   210→  avgLoss /= period;
   211→   211→  for (let i = period + 1; i < closes.length; i++) {
   212→   212→    const diff = closes[i] - closes[i - 1];
   213→   213→    const gain = diff > 0 ? diff : 0;
   214→   214→    const loss = diff < 0 ? Math.abs(diff) : 0;
   215→   215→    avgGain = (avgGain * (period - 1) + gain) / period;
   216→   216→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   217→   217→  }
   218→   218→  if (avgLoss === 0) return 100;
   219→   219→  const rs = avgGain / avgLoss;
   220→   220→  return 100 - 100 / (1 + rs);
   221→   221→}

   223→   223→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   224→   224→function calcMFI(data: OHLCV[], period: number = 14): number {
   225→   225→  if (data.length < period + 1) return 50;
   226→   226→  let posFlow = 0;
   227→   227→  let negFlow = 0;
   228→   228→  for (let i = data.length - period; i < data.length; i++) {
   229→   229→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   230→   230→    const mf = tp * data[i].volume;
   231→   231→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   232→   232→    if (tp > prevTp) posFlow += mf;
   233→   233→    else negFlow += mf;
   234→   234→  }
   235→   235→  if (negFlow === 0) return 100;
   236→   236→  const mfr = posFlow / negFlow;
   237→   237→  return 100 - 100 / (1 + mfr);
   238→   238→}

   241→   241→function calcCCI(data: OHLCV[], period: number = 20): number {
   242→   242→  if (data.length < period) return 0;
   243→   243→  const tps: number[] = [];
   244→   244→  for (let i = data.length - period; i < data.length; i++) {
   245→   245→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   246→   246→  }
   247→   247→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   248→   248→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   249→   249→  if (meanDev === 0) return 0;
   250→   250→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   251→   251→}

   254→   254→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   255→   255→  if (data.length < kPeriod) return { k: 50, d: 50 };
   256→   256→  const rawKs: number[] = [];
   257→   257→  for (let i = kPeriod - 1; i < data.length; i++) {
   258→   258→    let lowest = Infinity;
   259→   259→    let highest = -Infinity;
   260→   260→    for (let j = i - kPeriod + 1; j <= i; j++) {
   261→   261→      if (data[j].low < lowest) lowest = data[j].low;
   262→   262→      if (data[j].high > highest) highest = data[j].high;
   263→   263→    }
   264→   264→    const range = highest - lowest;
   265→   265→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   266→   266→  }
   267→   267→  const smoothedK: number[] = [];
   268→   268→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   269→   269→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   270→   270→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   271→   271→  }
   272→   272→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   273→   273→  const dSlice = smoothedK.slice(-smoothD);
   274→   274→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   275→   275→  return { k, d };
   276→   276→}

   279→   279→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   280→   280→  if (data.length < period) return -50;
   281→   281→  const slice = data.slice(-period);
   282→   282→  let highest = -Infinity;
   283→   283→  let lowest = Infinity;
   284→   284→  for (const c of slice) {
   285→   285→    if (c.high > highest) highest = c.high;
   286→   286→    if (c.low < lowest) lowest = c.low;
   287→   287→  }
   288→   288→  const range = highest - lowest;
   289→   289→  if (range === 0) return -50;
   290→   290→  return ((highest - data[data.length - 1].close) / range) * -100;
   291→   291→}

   293→   293→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   294→   294→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   295→   295→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   296→   296→  const emaFast = emaCalc(closes, fast);
   297→   297→  const emaSlow = emaCalc(closes, slow);
   298→   298→  const macdLine = emaFast - emaSlow;
   299→   299→  // Calculate signal as EMA of MACD values
   300→   300→  const macdValues: number[] = [];
   301→   301→  const kFast = 2 / (fast + 1);
   302→   302→  const kSlow = 2 / (slow + 1);
   303→   303→  let eFast = sma(closes.slice(0, fast), fast);
   304→   304→  let eSlow = sma(closes.slice(0, slow), slow);
   305→   305→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   306→   306→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   307→   307→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   308→   308→    macdValues.push(eFast - eSlow);
   309→   309→  }
   310→   310→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   311→   311→  const kSig = 2 / (sig + 1);
   312→   312→  let signalVal = sma(macdValues.slice(0, sig), sig);
   313→   313→  for (let i = sig; i < macdValues.length; i++) {
   314→   314→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   315→   315→  }
   316→   316→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   317→   317→}

   319→   319→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   320→   320→function calcATR(data: OHLCV[], period: number = 14): number {
   321→   321→  if (data.length < 2) return 0;
   322→   322→  const trs: number[] = [];
   323→   323→  for (let i = 1; i < data.length; i++) {
   324→   324→    const tr = Math.max(
   325→   325→      data[i].high - data[i].low,
   326→   326→      Math.abs(data[i].high - data[i - 1].close),
   327→   327→      Math.abs(data[i].low - data[i - 1].close)
   328→   328→    );
   329→   329→    trs.push(tr);
   330→   330→  }
   331→   331→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   332→   332→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   333→   333→  for (let i = period; i < trs.length; i++) {
   334→   334→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   335→   335→  }
   336→   336→  return atrVal;
   337→   337→}

   339→   339→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   340→   340→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   341→   341→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   342→   342→  const plusDMs: number[] = [];
   343→   343→  const minusDMs: number[] = [];
   344→   344→  const trs: number[] = [];
   345→   345→  for (let i = 1; i < data.length; i++) {
   346→   346→    const upMove = data[i].high - data[i - 1].high;
   347→   347→    const downMove = data[i - 1].low - data[i].low;
   348→   348→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   349→   349→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   350→   350→    trs.push(Math.max(
   351→   351→      data[i].high - data[i].low,
   352→   352→      Math.abs(data[i].high - data[i - 1].close),
   353→   353→      Math.abs(data[i].low - data[i - 1].close)
   354→   354→    ));
   355→   355→  }
   356→   356→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   357→   357→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   358→   358→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   359→   359→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   360→   360→  const dxValues: number[] = [];
   361→   361→  for (let i = period; i < trs.length; i++) {
   362→   362→    smoothTR = smoothTR - smoothTR / period + trs[i];
   363→   363→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   364→   364→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   365→   365→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   366→   366→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   367→   367→    const diSum = diPlus + diMinus;
   368→   368→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   369→   369→    dxValues.push(dx);
   370→   370→  }
   371→   371→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   372→   372→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   373→   373→  for (let i = period; i < dxValues.length; i++) {
   374→   374→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   375→   375→  }
   376→   376→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   377→   377→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   378→   378→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   379→   379→}

   381→   381→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   382→   382→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   383→   383→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   384→   384→  let isLong = true;
   385→   385→  let af = step;
   386→   386→  let ep = data[0].high;
   387→   387→  let sar = data[0].low;
   388→   388→  for (let i = 1; i < data.length; i++) {
   389→   389→    const prevSar = sar;
   390→   390→    sar = prevSar + af * (ep - prevSar);
   391→   391→    // Ensure SAR is within bounds
   392→   392→    if (isLong) {
   393→   393→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   394→   394→      if (data[i].low < sar) {
   395→   395→        isLong = false;
   396→   396→        sar = ep;
   397→   397→        ep = data[i].low;
   398→   398→        af = step;
   399→   399→      } else {
   400→   400→        if (data[i].high > ep) {
   401→   401→          ep = data[i].high;
   402→   402→          af = Math.min(af + step, max);
   403→   403→        }
   404→   404→      }
   405→   405→    } else {
   406→   406→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   407→   407→      if (data[i].high > sar) {
   408→   408→        isLong = true;
   409→   409→        sar = ep;
   410→   410→        ep = data[i].high;
   411→   411→        af = step;
   412→   412→      } else {
   413→   413→        if (data[i].low < ep) {
   414→   414→          ep = data[i].low;
   415→   415→          af = Math.min(af + step, max);
   416→   416→        }
   417→   417→      }
   418→   418→    }
   419→   419→  }
   420→   420→  return sar;
   421→   421→}

   425→   425→  if (closes.length < period) {

   432→   432→  const sd = Math.sqrt(variance);
   433→   433→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   434→   434→}

   473→   473→function calcIchimoku(data: OHLCV[]): {
   474→   474→  tenkan: number; kijun: number; senkouA: number; senkouB: number; chikou: number;
   475→   475→} {
   476→   476→  if (data.length < 52) {
   477→   477→    const c = data[data.length - 1]?.close ?? 0;
   478→   478→    return { tenkan: c, kijun: c, senkouA: c, senkouB: c, chikou: c };
   479→   479→  }
   480→   480→  const highs = data.map(d => d.high);
   481→   481→  const lows = data.map(d => d.low);
   482→   482→  const closes = data.map(d => d.close);
   483→   483→
   484→   484→  function donchian(h: number[], l: number[], end: number, period: number): number {
   485→   485→    const start = end - period + 1;
   486→   486→    if (start < 0) return (h[end] + l[end]) / 2;
   487→   487→    let hi = -Infinity, lo = Infinity;
   488→   488→    for (let i = start; i <= end; i++) {
   489→   489→      if (h[i] > hi) hi = h[i];
   490→   490→      if (l[i] < lo) lo = l[i];
   491→   491→    }
   492→   492→    return (hi + lo) / 2;
   493→   493→  }
   494→   494→
   495→   495→  const n = data.length;
   496→   496→  const tenkan = donchian(highs, lows, n - 1, 9);
   497→   497→  const kijun = donchian(highs, lows, n - 1, 26);
   498→   498→  // Senkou A = (tenkan + kijun) / 2, shifted 26 periods ahead
   499→   499→  // We compute it at bar n-27 (shifted to current)
   500→   500→  const senkouAIdx = Math.max(n - 1 - 26, 0);
   501→   501→  const tA = donchian(highs, lows, senkouAIdx, 9);
   502→   502→  const kA = donchian(highs, lows, senkouAIdx, 26);
   503→   503→  const senkouA = (tA + kA) / 2;
   504→   504→  // Senkou B = donchian(52), shifted 26 periods ahead
   505→   505→  const senkouBIdx = Math.max(n - 1 - 26, 0);
   506→   506→  const senkouB = donchian(highs, lows, senkouBIdx, 52);
   507→   507→  // Chikou = current close plotted 26 periods back
   508→   508→  const chikou = closes[n - 1];
   509→   509→
   510→   510→  return { tenkan, kijun, senkouA, senkouB, chikou };
   511→   511→}

   520→   520→  function donchian(end: number, period: number): number {

   546→   546→      senkouA.push(donchian(i, 9)); // fallback
   547→   547→    }
   548→   548→    // Senkou B shifted +26: at bar i, we need donchian(52) at bar i-26
   549→   549→    if (sIdx >= 0) {
   550→   550→      senkouB.push(donchian(sIdx, 52));
   551→   551→    } else {
   552→   552→      senkouB.push(donchian(i, 52)); // fallback
   553→   553→    }
   554→   554→  }
   555→   555→
   556→   556→  return { tenkan, kijun, senkouA, senkouB };
   557→   557→}

   572→   572→function emaArrayCalc(closes: number[], period: number): number[] {
   573→   573→  const result: number[] = [];
   574→   574→  if (closes.length < period) return closes.map(() => 0);
   575→   575→  const k = 2 / (period + 1);
   576→   576→  let emaVal = 0;
   577→   577→  for (let i = 0; i < period; i++) emaVal += closes[i];
   578→   578→  emaVal /= period;
   579→   579→  for (let i = 0; i < closes.length; i++) {
   580→   580→    if (i < period) { result.push(0); continue; }
   581→   581→    if (i === period) { result.push(emaVal); continue; }
   582→   582→    emaVal = closes[i] * k + emaVal * (1 - k);
   583→   583→    result.push(emaVal);
   584→   584→  }
   585→   585→  return result;
   586→   586→}

   588→   588→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   589→   589→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   590→   590→  const n = values.length;
   591→   591→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   592→   592→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   593→   593→  for (let i = 0; i < n; i++) {
   594→   594→    sumX += i;
   595→   595→    sumY += values[i];
   596→   596→    sumXY += i * values[i];
   597→   597→    sumX2 += i * i;
   598→   598→    sumY2 += values[i] * values[i];
   599→   599→  }
   600→   600→  const denom = n * sumX2 - sumX * sumX;
   601→   601→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   602→   602→  const slope = (n * sumXY - sumX * sumY) / denom;
   603→   603→  const intercept = (sumY - slope * sumX) / n;
   604→   604→  const meanY = sumY / n;
   605→   605→  let ssRes = 0, ssTot = 0;
   606→   606→  for (let i = 0; i < n; i++) {
   607→   607→    const predicted = slope * i + intercept;
   608→   608→    ssRes += (values[i] - predicted) ** 2;
   609→   609→    ssTot += (values[i] - meanY) ** 2;
   610→   610→  }
   611→   611→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   612→   612→  return { slope, intercept, r2 };
   613→   613→}

   708→   708→// ─── Helper: Linear System Solver (Gaussian Elimination) ─────────────────
   709→   709→function solveLinearSystem(A: number[][], b: number[]): number[] | null {
   710→   710→  const n = b.length;
   711→   711→  const aug = A.map((row, i) => [...row, b[i]]);
   712→   712→
   713→   713→  for (let col = 0; col < n; col++) {
   714→   714→    let maxVal = Math.abs(aug[col][col]);
   715→   715→    let maxRow = col;
   716→   716→    for (let row = col + 1; row < n; row++) {
   717→   717→      if (Math.abs(aug[row][col]) > maxVal) {
   718→   718→        maxVal = Math.abs(aug[row][col]);
   719→   719→        maxRow = row;
   720→   720→      }
   721→   721→    }
   722→   722→    if (maxVal < 1e-10) return null;
   723→   723→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   724→   724→
   725→   725→    for (let row = col + 1; row < n; row++) {
   726→   726→      const factor = aug[row][col] / aug[col][col];
   727→   727→      for (let j = col; j <= n; j++) {
   728→   728→        aug[row][j] -= factor * aug[col][j];
   729→   729→      }
   730→   730→    }
   731→   731→  }
   732→   732→
   733→   733→  const x = new Array(n).fill(0);
   734→   734→  for (let i = n - 1; i >= 0; i--) {
   735→   735→    x[i] = aug[i][n];
   736→   736→    for (let j = i + 1; j < n; j++) {
   737→   737→      x[i] -= aug[i][j] * x[j];
   738→   738→    }
   739→   739→    if (Math.abs(aug[i][i]) < 1e-10) return null;
   740→   740→    x[i] /= aug[i][i];
   741→   741→  }
   742→   742→
   743→   743→  return x;
   744→   744→}

   746→   746→// ─── Helper: Generate All 7-Method S/R Levels ─────────────────────────────
   747→   747→function generateAllLevels(data: OHLCV[], endIdx: number): RawSRLevel[] {
   748→   748→  const levels: RawSRLevel[] = [];
   749→   749→  if (endIdx < 10) return levels;
   750→   750→
   751→   751→  const slice = data.slice(0, endIdx + 1);
   752→   752→  const currentPrice = slice[slice.length - 1].close;
   753→   753→  const closes = slice.map(d => d.close);
   754→   754→  const noFib: [string, string] = ['—', ''];
   755→   755→
   756→   756→  // ── Method 1: Swing High/Low (Price Action) ──────────────────────────
   757→   757→  for (const lookback of [3, 5, 7]) {
   758→   758→    const swings = findSwingLevels(slice, lookback);
   759→   759→    const tol = currentPrice * 0.005; // 0.5% tolerance
   760→   760→
   761→   761→    // Connect swing highs → resistance
   762→   762→    const usedHigh = new Set<number>();
   763→   763→    for (const h of swings.highs) {
   764→   764→      if (usedHigh.has(h.idx)) continue;
   765→   765→      let sum = h.price;
   766→   766→      let count = 1;
   767→   767→      for (const h2 of swings.highs) {
   768→   768→        if (h2 === h || usedHigh.has(h2.idx)) continue;
   769→   769→        if (Math.abs(h2.price - h.price) < tol) {
   770→   770→          sum += h2.price;
   771→   771→          count++;
   772→   772→          usedHigh.add(h2.idx);
   773→   773→        }
   774→   774→      }
   775→   775→      usedHigh.add(h.idx);
   776→   776→      if (sum / count > currentPrice * 1.005) {
   777→   777→        levels.push({ price: sum / count, method: 'Swing_High', type: 'resistance', ...noFib });
   778→   778→      }
   779→   779→    }
   780→   780→
   781→   781→    // Connect swing lows → support
   782→   782→    const usedLow = new Set<number>();
   783→   783→    for (const l of swings.lows) {
   784→   784→      if (usedLow.has(l.idx)) continue;
   785→   785→      let sum = l.price;
   786→   786→      let count = 1;
   787→   787→      for (const l2 of swings.lows) {
   788→   788→        if (l2 === l || usedLow.has(l2.idx)) continue;
   789→   789→        if (Math.abs(l2.price - l.price) < tol) {
   790→   790→          sum += l2.price;
   791→   791→          count++;
   792→   792→          usedLow.add(l2.idx);
   793→   793→        }
   794→   794→      }
   795→   795→      usedLow.add(l.idx);
   796→   796→      if (sum / count < currentPrice * 0.995) {
   797→   797→        levels.push({ price: sum / count, method: 'Swing_Low', type: 'support', ...noFib });
   798→   798→      }
   799→   799→    }
   800→   800→  }
   801→   801→
   802→   802→  // ── Method 2: SMA Dynamic S/R ──────────────────────────────────────
   803→   803→  for (const period of [20, 50, 100, 200]) {
   804→   804→    const smaVal = sma(closes, period);
   805→   805→    if (smaVal <= 0) continue;
   806→   806→    const type: 'support' | 'resistance' = currentPrice > smaVal ? 'support' : 'resistance';
   807→   807→    levels.push({ price: smaVal, method: `SMA_${period}`, type, ...noFib });
   808→   808→  }
   809→   809→
   810→   810→  // ── Method 3: Bollinger Bands ───────────────────────────────────────
   811→   811→  const bb = calcBollingerBands(closes);
   812→   812→  if (bb.upper > 0 && bb.lower > 0) {
   813→   813→    levels.push({ price: bb.upper, method: 'BB_Upper', type: 'resistance', ...noFib });
   814→   814→    levels.push({ price: bb.lower, method: 'BB_Lower', type: 'support', ...noFib });
   815→   815→  }
   816→   816→
   817→   817→  // ── Method 4: Fibonacci Retracement + Extension ─────────────────────
   818→   818→  interface SwingPair { high: number; highIdx: number; low: number; lowIdx: number; range: number }
   819→   819→  let bestSwing: SwingPair | null = null;
   820→   820→
   821→   821→  for (const lb of [3, 5, 7, 10, 15, 20]) {
   822→   822→    const sw = findSwingLevels(slice, lb);
   823→   823→    if (sw.highs.length === 0 || sw.lows.length === 0) continue;
   824→   824→    for (const h of sw.highs) {
   825→   825→      for (const l of sw.lows) {
   826→   826→        if (Math.abs(h.idx - l.idx) < 3) continue;
   827→   827→        const range = Math.abs(h.price - l.price);
   828→   828→        if (range / currentPrice < 0.03) continue;
   829→   829→        if (!bestSwing || range > bestSwing.range) {
   830→   830→          bestSwing = {
   831→   831→            high: Math.max(h.price, l.price),
   832→   832→            highIdx: h.price > l.price ? h.idx : l.idx,
   833→   833→            low: Math.min(h.price, l.price),
   834→   834→            lowIdx: h.price < l.price ? l.idx : h.idx,
   835→   835→            range,
   836→   836→          };
   837→   837→        }
   838→   838→      }
   839→   839→    }
   840→   840→  }
   841→   841→
   842→   842→  if (bestSwing) {
   843→   843→    const H = bestSwing.high;
   844→   844→    const L = bestSwing.low;
   845→   845→    const range = H - L;
   846→   846→
   847→   847→    for (const ratio of FIB_RETRACEMENTS) {
   848→   848→      const price = H - ratio * range;
   849→   849→      if (price <= 0) continue;
   850→   850→      const rounded = roundToNice(price);
   851→   851→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   852→   852→      levels.push({
   853→   853→        price: rounded, method: `Fibonacci_${ratio}`, type,
   854→   854→        fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   855→   855→      });
   856→   856→    }
   857→   857→
   858→   858→    for (const extRatio of FIB_EXTENSIONS) {
   859→   859→      const price = H + (extRatio - 1) * range;
   860→   860→      if (price <= 0) continue;
   861→   861→      const rounded = roundToNice(price);
   862→   862→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   863→   863→      levels.push({
   864→   864→        price: rounded, method: `Fibonacci_${extRatio}`, type,
   865→   865→        fibRatio: String(extRatio), fibLabel: FIB_LABELS[String(extRatio)] || '',
   866→   866→      });
   867→   867→    }
   868→   868→
   869→   869→    // Inverse extensions (below L)
   870→   870→    for (const extRatio of FIB_EXTENSIONS) {
   871→   871→      const price = L - (extRatio - 1) * range;
   872→   872→      if (price <= 0) continue;
   873→   873→      const rounded = roundToNice(price);
   874→   874→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   875→   875→      levels.push({
   876→   876→        price: rounded, method: `Fibonacci_${-extRatio}`, type,
   877→   877→        fibRatio: String(-extRatio), fibLabel: '',
   878→   878→      });
   879→   879→    }
   880→   880→
   881→   881→    // Secondary swing confluence (check overlap with secondary swings)
   882→   882→    const secondarySwings: SwingPair[] = [];
   883→   883→    for (const lb of [3, 5, 7]) {
   884→   884→      const sw = findSwingLevels(slice, lb);
   885→   885→      for (const h of sw.highs) {
   886→   886→        for (const l of sw.lows) {
   887→   887→          if (Math.abs(h.idx - l.idx) < 3) continue;
   888→   888→          const r = Math.abs(h.price - l.price);
   889→   889→          if (r / currentPrice < 0.05) continue;
   890→   890→          if (bestSwing && Math.abs(r - bestSwing.range) / bestSwing.range < 0.1) continue;
   891→   891→          secondarySwings.push({
   892→   892→            high: Math.max(h.price, l.price), highIdx: h.price > l.price ? h.idx : l.idx,
   893→   893→            low: Math.min(h.price, l.price), lowIdx: h.price < l.price ? l.idx : h.idx,
   894→   894→            range: r,
   895→   895→          });
   896→   896→        }
   897→   897→      }
   898→   898→    }
   899→   899→    // Secondary swings' fib levels are merged via confluence (1% rule in mergeLevels)
   900→   900→    const confluenceTol = currentPrice * 0.005;
   901→   901→    for (const sec of secondarySwings.slice(0, 3)) {
   902→   902→      const sH = sec.high, sL = sec.low, sR = sH - sL;
   903→   903→      for (const ratio of FIB_RETRACEMENTS) {
   904→   904→        const price = roundToNice(sH - ratio * sR);
   905→   905→        if (price <= 0) continue;
   906→   906→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   907→   907→        levels.push({
   908→   908→          price, method: `Fibonacci_${ratio}`, type,
   909→   909→          fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   910→   910→        });
   911→   911→      }
   912→   912→      for (const extR of FIB_EXTENSIONS) {
   913→   913→        const price = roundToNice(sH + (extR - 1) * sR);
   914→   914→        if (price <= 0) continue;
   915→   915→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   916→   916→        levels.push({
   917→   917→          price, method: `Fibonacci_${extR}`, type,
   918→   918→          fibRatio: String(extR), fibLabel: FIB_LABELS[String(extR)] || '',
   919→   919→        });
   920→   920→      }
   921→   921→    }
   922→   922→  }
   923→   923→
   924→   924→  // ── Method 5: Volume Profile (VAP) ──────────────────────────────────
   925→   925→  const dataHasVolume = slice.some(d => d.volume > 0);
   926→   926→  if (dataHasVolume) {
   927→   927→    let minP = Infinity, maxP = -Infinity;
   928→   928→    for (const d of slice) {
   929→   929→      if (d.low < minP) minP = d.low;
   930→   930→      if (d.high > maxP) maxP = d.high;
   931→   931→    }
   932→   932→    const numBins = 40;
   933→   933→    const binSize = Math.max((maxP - minP) / numBins, currentPrice * 0.001);
   934→   934→    const vap = new Array(numBins).fill(0);
   935→   935→    const binCenters: number[] = [];
   936→   936→
   937→   937→    for (let b = 0; b < numBins; b++) {
   938→   938→      binCenters.push(minP + (b + 0.5) * binSize);
   939→   939→    }
   940→   940→
   941→   941→    for (const d of slice) {
   942→   942→      for (let b = 0; b < numBins; b++) {
   943→   943→        if (d.low <= binCenters[b] && binCenters[b] <= d.high) {
   944→   944→          vap[b] += d.volume;
   945→   945→        }
   946→   946→      }
   947→   947→    }
   948→   948→
   949→   949→    const maxVAP = Math.max(...vap);
   950→   950→    if (maxVAP > 0) {
   951→   951→      // POC (Point of Control)
   952→   952→      let pocIdx = 0;
   953→   953→      for (let b = 1; b < numBins; b++) {
   954→   954→        if (vap[b] > vap[pocIdx]) pocIdx = b;
   955→   955→      }
   956→   956→      const pocPrice = binCenters[pocIdx];
   957→   957→      const pocType: 'support' | 'resistance' = pocPrice > currentPrice ? 'resistance' : 'support';
   958→   958→      levels.push({ price: pocPrice, method: 'VAP_POC', type: pocType, ...noFib });
   959→   959→
   960→   960→      // HVN (High Volume Nodes)
   961→   961→      for (let b = 0; b < numBins; b++) {
   962→   962→        if (b === pocIdx) continue;
   963→   963→        if (vap[b] > 0.7 * maxVAP) {
   964→   964→          const hvnPrice = binCenters[b];
   965→   965→          const hvnType: 'support' | 'resistance' = hvnPrice > currentPrice ? 'resistance' : 'support';
   966→   966→          levels.push({ price: hvnPrice, method: 'VAP_HVN', type: hvnType, ...noFib });
   967→   967→        }
   968→   968→      }
   969→   969→    }
   970→   970→  }
   971→   971→
   972→   972→  // ── Method 6: Pivot Points (4 algorithms) ───────────────────────────
   973→   973→  if (endIdx >= 1) {
   974→   974→    const prev = slice[endIdx - 1]; // Previous completed bar
   975→   975→    const pH = prev.high, pL = prev.low, pC = prev.close, pO = prev.open;
   976→   976→
   977→   977→    // Standard Pivot
   978→   978→    {
   979→   979→      const PP = (pH + pL + pC) / 3;
   980→   980→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   981→   981→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   982→   982→      const R3 = pH + 2 * (PP - pL), S3 = pL - 2 * (pH - PP);
   983→   983→ if (R1 > 0) levels.push({ price: R1, method: 'Pivot_R1', type: 'resistance', ...noFib });
   984→   984→      if (R2 > 0) levels.push({ price: R2, method: 'Pivot_R2', type: 'resistance', ...noFib });
   985→   985→      if (R3 > 0) levels.push({ price: R3, method: 'Pivot_R3', type: 'resistance', ...noFib });
   986→   986→      if (S1 > 0) levels.push({ price: S1, method: 'Pivot_S1', type: 'support', ...noFib });
   987→   987→      if (S2 > 0) levels.push({ price: S2, method: 'Pivot_S2', type: 'support', ...noFib });
   988→   988→      if (S3 > 0) levels.push({ price: S3, method: 'Pivot_S3', type: 'support', ...noFib });
   989→   989→    }
   990→   990→
   991→   991→    // Fibonacci Pivot
   992→   992→    {
   993→   993→      const PP = (pH + pL + pC) / 3;
   994→   994→      const range = pH - pL;
   995→   995→      for (const [mult, suffix] of [[0.382, '1'], [0.618, '2'], [1.0, '3']] as [number, string][]) {
   996→   996→        const R = PP + mult * range;
   997→   997→        const S = PP - mult * range;
   998→   998→        if (R > 0) levels.push({ price: R, method: `Fib_Pivot_R${suffix}`, type: 'resistance', ...noFib });
   999→   999→        if (S > 0) levels.push({ price: S, method: `Fib_Pivot_S${suffix}`, type: 'support', ...noFib });
  1000→  1000→      }
  1001→  1001→    }
  1002→  1002→
  1003→  1003→    // Woodie's Pivot
  1004→  1004→    {
  1005→  1005→      const PP = (pH + pL + 2 * pC) / 4;
  1006→  1006→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
  1007→  1007→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
  1008→  1008→      if (R1 > 0) levels.push({ price: R1, method: 'Woodie_R1', type: 'resistance', ...noFib });
  1009→  1009→      if (R2 > 0) levels.push({ price: R2, method: 'Woodie_R2', type: 'resistance', ...noFib });
  1010→  1010→      if (S1 > 0) levels.push({ price: S1, method: 'Woodie_S1', type: 'support', ...noFib });
  1011→  1011→      if (S2 > 0) levels.push({ price: S2, method: 'Woodie_S2', type: 'support', ...noFib });
  1012→  1012→    }
  1013→  1013→
  1014→  1014→    // DeMark Pivot
  1015→  1015→    {
  1016→  1016→      const X = pC < pO ? pH + 2 * pL + pC : pC > pO ? 2 * pH + pL + pC : pH + pL + 2 * pC;
  1017→  1017→      const PP = X / 4;
  1018→  1018→      const R1 = X / 2 - pL;
  1019→  1019→      const S1 = X / 2 - pH;
  1020→  1020→      if (R1 > 0) levels.push({ price: R1, method: 'DeMark_R1', type: 'resistance', ...noFib });
  1021→  1021→      if (S1 > 0) levels.push({ price: S1, method: 'DeMark_S1', type: 'support', ...noFib });
  1022→  1022→    }
  1023→  1023→  }
  1024→  1024→
  1025→  1025→  // ── Method 7: Psychological Levels ──────────────────────────────────
  1026→  1026→  {
  1027→  1027→    const step = psychStep(currentPrice);
  1028→  1028→    const base = psychRound(currentPrice);
  1029→  1029→    for (const mult of [-2, -1, 1, 2]) {
  1030→  1030→      const price = base + mult * step;
  1031→  1031→      if (price <= 0) continue;
  1032→  1032→      const type: 'support' | 'resistance' = price > currentPrice ? 'resistance' : 'support';
  1033→  1033→      levels.push({ price, method: 'Psychological', type, ...noFib });
  1034→  1034→    }
  1035→  1035→  }
  1036→  1036→
  1037→  1037→  return levels;
  1038→  1038→}

  1040→  1040→// ─── Helper: Merge Nearby Levels (1% Confluence) ──────────────────────────
  1041→  1041→function mergeLevels(rawLevels: RawSRLevel[], currentPrice: number): MergedSRLevel[] {
  1042→  1042→  if (rawLevels.length === 0) return [];
  1043→  1043→
  1044→  1044→  const tol = currentPrice * 0.01; // 1% tolerance
  1045→  1045→  const merged: MergedSRLevel[] = [];
  1046→  1046→  const used = new Set<number>();
  1047→  1047→
  1048→  1048→  for (let i = 0; i < rawLevels.length; i++) {
  1049→  1049→    if (used.has(i)) continue;
  1050→  1050→
  1051→  1051→    let sumPrice = rawLevels[i].price;
  1052→  1052→    let count = 1;
  1053→  1053→    const methods = new Set<string>();
  1054→  1054→    methods.add(rawLevels[i].method);
  1055→  1055→    let bestFibRatio = rawLevels[i].fibRatio;
  1056→  1056→    let bestFibLabel = rawLevels[i].fibLabel;
  1057→  1057→
  1058→  1058→    for (let j = i + 1; j < rawLevels.length; j++) {
  1059→  1059→      if (used.has(j)) continue;
  1060→  1060→      if (Math.abs(rawLevels[j].price - rawLevels[i].price) < tol) {
  1061→  1061→        sumPrice += rawLevels[j].price;
  1062→  1062→        count++;
  1063→  1063→        methods.add(rawLevels[j].method);
  1064→  1064→        used.add(j);
  1065→  1065→        // Prefer Fibonacci label if available
  1066→  1066→        if (rawLevels[j].fibRatio !== '—' && bestFibRatio === '—') {
  1067→  1067→          bestFibRatio = rawLevels[j].fibRatio;
  1068→  1068→          bestFibLabel = rawLevels[j].fibLabel;
  1069→  1069→        }
  1070→  1070→      }
  1071→  1071→    }
  1072→  1072→    used.add(i);
  1073→  1073→
  1074→  1074→    const avgPrice = sumPrice / count;
  1075→  1075→    const type: 'support' | 'resistance' = avgPrice >= currentPrice ? 'resistance' : 'support';
  1076→  1076→
  1077→  1077→    merged.push({
  1078→  1078→      price: avgPrice,
  1079→  1079→      type,
  1080→  1080→      methods: [...methods],
  1081→  1081→      overlapCount: methods.size,
  1082→  1082→      fibRatio: bestFibRatio,
  1083→  1083→      fibLabel: bestFibLabel,
  1084→  1084→      score: 0,
  1085→  1085→      strength: 1,
  1086→  1086→      touchCount: 0,
  1087→  1087→      volumeRatio: 0,
  1088→  1088→      freshness: 0,
  1089→  1089→      distancePercent: 0,
  1090→  1090→    });
  1091→  1091→  }
  1092→  1092→
  1093→  1093→  return merged;
  1094→  1094→}

  1096→  1096→// ─── Helper: Compute Level Features (5 ML features) ──────────────────────
  1097→  1097→function computeLevelFeatures(
  1098→  1098→  levelPrice: number,
  1099→  1099→  data: OHLCV[],
  1100→  1100→  endIdx: number,
  1101→  1101→  currentPrice: number,
  1102→  1102→  overlapCount: number,
  1103→  1103→): number[] {
  1104→  1104→  const lookback = Math.min(30, endIdx);
  1105→  1105→  const startIdx = endIdx - lookback;
  1106→  1106→  const touchTol = currentPrice * 0.002; // 0.2%
  1107→  1107→
  1108→  1108→  let touchCount = 0;
  1109→  1109→  let totalVolAtTouches = 0;
  1110→  1110→  let totalVol = 0;
  1111→  1111→  let barsSinceLastTouch = lookback;
  1112→  1112→
  1113→  1113→  for (let i = startIdx; i <= endIdx; i++) {
  1114→  1114→    const bar = data[i];
  1115→  1115→    totalVol += bar.volume;
  1116→  1116→
  1117→  1117→    if (bar.low <= levelPrice + touchTol && bar.high >= levelPrice - touchTol) {
  1118→  1118→      touchCount++;
  1119→  1119→      totalVolAtTouches += bar.volume;
  1120→  1120→      barsSinceLastTouch = endIdx - i;
  1121→  1121→    }
  1122→  1122→  }
  1123→  1123→
  1124→  1124→  const avgVol = lookback > 0 ? totalVol / lookback : 0;
  1125→  1125→  const volumeRatio = (touchCount > 0 && avgVol > 0) ? (totalVolAtTouches / touchCount) / avgVol : 1;
  1126→  1126→  const freshness = Math.max(0, Math.min(1, 1 - barsSinceLastTouch / 30));
  1127→  1127→  const distancePercent = Math.min(1, Math.abs(levelPrice - currentPrice) / currentPrice);
  1128→  1128→
  1129→  1129→  // Features: [touch_count, volume_ratio, overlap_count, freshness, distance_percent]
  1130→  1130→  return [touchCount, volumeRatio, overlapCount - 1, freshness, distancePercent];
  1131→  1131→}

  1133→  1133→// ─── Helper: Train S/R Scoring Weights (Ridge OLS) ───────────────────────
  1134→  1134→function trainSRWeights(data: OHLCV[]): number[] {
  1135→  1135→  const defaultWeights = [0.25, 0.20, 0.25, 0.15, 0.15];
  1136→  1136→
  1137→  1137→  if (data.length < 50) return defaultWeights;
  1138→  1138→
  1139→  1139→  const X: number[][] = [];
  1140→  1140→  const y: number[] = [];
  1141→  1141→
  1142→  1142→  for (let i = 40; i <= data.length - 11; i += 5) {
  1143→  1143→    const barPrice = data[i].close;
  1144→  1144→    const rawLevels = generateAllLevels(data, i);
  1145→  1145→    if (rawLevels.length === 0) continue;
  1146→  1146→    const merged = mergeLevels(rawLevels, barPrice);
  1147→  1147→
  1148→  1148→    for (const level of merged) {
  1149→  1149→      const features = computeLevelFeatures(level.price, data, i, barPrice, level.overlapCount);
  1150→  1150→      X.push(features);
  1151→  1151→
  1152→  1152→      // Label: did price return within 10 bars?
  1153→  1153→      let returned = false;
  1154→  1154→      const retTol = barPrice * 0.005;
  1155→  1155→      for (let j = i + 1; j <= Math.min(i + 10, data.length - 1); j++) {
  1156→  1156→        if (data[j].low <= level.price + retTol && data[j].high >= level.price - retTol) {
  1157→  1157→          returned = true;
  1158→  1158→          break;
  1159→  1159→        }
  1160→  1160→      }
  1161→  1161→      y.push(returned ? 1 : 0);
  1162→  1162→    }
  1163→  1163→  }
  1164→  1164→
  1165→  1165→  if (X.length < 20) return defaultWeights;
  1166→  1166→
  1167→  1167→  // Ridge OLS: w = (X^T X + λI)^(-1) X^T y
  1168→  1168→  const n = X.length;
  1169→  1169→  const p = 5;
  1170→  1170→  const lambda = 0.01;
  1171→  1171→
  1172→  1172→  const XtX: number[][] = Array.from({ length: p }, () => new Array(p).fill(0));
  1173→  1173→  for (let i = 0; i < n; i++) {
  1174→  1174→    for (let a = 0; a < p; a++) {
  1175→  1175→      for (let b = 0; b < p; b++) {
  1176→  1176→        XtX[a][b] += X[i][a] * X[i][b];
  1177→  1177→      }
  1178→  1178→    }
  1179→  1179→  }
  1180→  1180→  for (let a = 0; a < p; a++) XtX[a][a] += lambda;
  1181→  1181→
  1182→  1182→  const Xty: number[] = new Array(p).fill(0);
  1183→  1183→  for (let i = 0; i < n; i++) {
  1184→  1184→    for (let a = 0; a < p; a++) {
  1185→  1185→      Xty[a] += X[i][a] * y[i];
  1186→  1186→    }
  1187→  1187→  }
  1188→  1188→
  1189→  1189→  const weights = solveLinearSystem(XtX, Xty);
  1190→  1190→  if (!weights) return defaultWeights;
  1191→  1191→  if (weights.some(w => isNaN(w) || !isFinite(w))) return defaultWeights;
  1192→  1192→
  1193→  1193→  for (let i = 0; i < p; i++) {
  1194→  1194→    if (weights[i] < 0) weights[i] = 0;
  1195→  1195→  }
  1196→  1196→
  1197→  1197→  const sum = weights.reduce((a, b) => a + b, 0);

  1564→  1564→      if (existing) {
  1565→  1565→        if (level.score > existing.score) {
  1566→  1566→          Object.assign(existing, level);
  1567→  1567→        }
  1568→  1568→      } else {
  1569→  1569→        deduped.push(level);
  1570→  1570→      }
  1571→  1571→    }
  1572→  1572→    merged = deduped;
  1573→  1573→  }
  1574→  1574→
  1575→  1575→  // ── Step 6: Separate into supports and resistances ─────────────────────
  1576→  1576→  let supports = merged.filter(l => l.type === 'support');
  1577→  1577→  let resistances = merged.filter(l => l.type === 'resistance');
  1578→  1578→
  1579→  1579→  // ── Step 7: Distance filtering (strict 5-10% gap rules) ─────────────
  1580→  1580→  // Rules:
  1581→  1581→  //   1. Consecutive lines of same type: 5% ≤ gap ≤ 10%
  1582→  1582→  //   2. Nearest support ≤ 10% below price (can be < 5%)
  1583→  1583→  //   3. Nearest resistance ≤ 10% above price (can be < 5%)
  1584→  1584→
  1585→  1585→  const filterAndSelectLevels = (
  1586→  1586→    levels: MergedSRLevel[],
  1587→  1587→    type: 'support' | 'resistance'
  1588→  1588→  ): MergedSRLevel[] => {
  1589→  1589→    if (levels.length === 0) return [];

  1591→  1591→    // Sort: supports descending (nearest first), resistances ascending (nearest first)

  1593→  1593→      type === 'support' ? b.price - a.price : a.price - b.price
  1594→  1594→    );
  1595→  1595→
  1596→  1596→    // ── Rule 2/3: Nearest level must be within 10% of price ──
  1597→  1597→    const nearest = sorted[0];

  1599→  1599→    if (distFromPrice > 0.10) {
  1600→  1600→      // Nearest level is too far — generate a synthetic level at ~5% from price
  1601→  1601→      const syntheticPrice = type === 'support'
  1602→  1602→        ? Math.round(currentPrice * 0.95 / psychStep(currentPrice)) * psychStep(currentPrice)
  1603→  1603→        : Math.round(currentPrice * 1.05 / psychStep(currentPrice)) * psychStep(currentPrice);
  1604→  1604→      const synthetic: MergedSRLevel = {
  1605→  1605→        price: syntheticPrice,
  1606→  1606→        type,
  1607→  1607→        score: 3,
  1608→  1608→        strength: 3,
  1609→  1609→        overlapCount: 1,
  1610→  1610→        methods: ['Psychological'],
  1611→  1611→        fibRatio: '—',
  1612→  1612→        fibLabel: '',
  1613→  1613→      };
  1614→  1614→      sorted.unshift(synthetic);
  1615→  1615→    }
  1616→  1616→
  1617→  1617→    // ── Rule 1: Select levels with 5-10% gaps between consecutive ──
  1618→  1618→    const result: MergedSRLevel[] = [sorted[0]];
  1619→  1619→

  1621→  1621→      const prevPrice = result[result.length - 1].price;
  1622→  1622→      const currPrice = sorted[i].price;
  1623→  1623→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
  1624→  1624→
  1625→  1625→      if (gap < 0.05) {
  1626→  1626→        // Too close — keep the stronger one
  1627→  1627→        if (sorted[i].score > result[result.length - 1].score) {
  1628→  1628→          result[result.length - 1] = sorted[i];
  1629→  1629→        }
  1630→  1630→      } else if (gap <= 0.10) {
  1631→  1631→        // Perfect gap (5-10%) — accept
  1632→  1632→        result.push(sorted[i]);
  1633→  1633→      } else {
  1634→  1634→        // Gap > 10% — try to find an intermediate level
  1635→  1635→        let found = false;
  1636→  1636→        for (let j = i + 1; j < sorted.length; j++) {
  1637→  1637→          const midPrice = sorted[j].price;
  1638→  1638→          const gapFromPrev = Math.abs(midPrice - prevPrice) / prevPrice;
  1639→  1639→          const gapToNext = Math.abs(currPrice - midPrice) / midPrice;
  1640→  1640→          if (gapFromPrev >= 0.05 && gapFromPrev <= 0.10) {
  1641→  1641→            result.push(sorted[j]);
  1642→  1642→            found = true;
  1643→  1643→            // Skip levels that are too close to the inserted one
  1644→  1644→            i = j;
  1645→  1645→            break;
  1646→  1646→          }
  1647→  1647→        }
  1648→  1648→        if (!found) {
  1649→  1649→          // No intermediate found — generate a synthetic level at ~7.5% from prev
  1650→  1650→          const synthGap = 0.075;
  1651→  1651→          const syntheticPrice = type === 'support'
  1652→  1652→            ? Math.round((prevPrice * (1 - synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice)
  1653→  1653→            : Math.round((prevPrice * (1 + synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice);
  1654→  1654→          // Make sure it's not too close to next level
  1655→  1655→          const gapToNext = Math.abs(currPrice - syntheticPrice) / syntheticPrice;
  1656→  1656→          if (gapToNext >= 0.05) {
  1657→  1657→            const synthetic: MergedSRLevel = {
  1658→  1658→              price: syntheticPrice,
  1659→  1659→              type,
  1660→  1660→              score: 2,
  1661→  1661→              strength: 2,
  1662→  1662→              overlapCount: 1,
  1663→  1663→              methods: ['Psychological'],
  1664→  1664→              fibRatio: '—',
  1665→  1665→              fibLabel: '',
  1666→  1666→            };
  1667→  1667→            result.push(synthetic);
  1668→  1668→          }
  1669→  1669→        }
  1670→  1670→      }
  1671→  1671→    }
  1672→  1672→
  1673→  1673→    return result;
  1674→  1674→  };
  1675→  1675→
  1676→  1676→  const finalSupports = filterAndSelectLevels(supports, 'support');
  1677→  1677→  const finalResistances = filterAndSelectLevels(resistances, 'resistance');
  1678→  1678→
  1679→  1679→  // ── Step 9: Build LevelStrength objects ────────────────────────────────
  1680→  1680→  const toLevelStrength = (m: MergedSRLevel): LevelStrength => {
  1681→  1681→    let grade: string;
  1682→  1682→    if (m.score >= 8.5) grade = 'Very Strong';
  1683→  1683→    else if (m.score >= 7) grade = 'Strong';
  1684→  1684→    else if (m.score >= 5) grade = 'Moderate';
  1685→  1685→    else grade = 'Weak';
  1686→  1686→
  1687→  1687→    return {
  1688→  1688→      price: m.price,
  1689→  1689→      strength: m.strength,
  1690→  1690→      score: m.score,
  1691→  1691→      grade,
  1692→  1692→      isTarget: m.score >= 7,
  1693→  1693→      overlapCount: m.overlapCount,
  1694→  1694→      methods: m.methods,
  1695→  1695→      fibRatio: m.fibRatio,
  1696→  1696→      fibLabel: m.fibLabel,
  1697→  1697→    };
  1698→  1698→  };
  1699→  1699→
  1700→  1700→  const supportStrengths = finalSupports.map(toLevelStrength);
  1701→  1701→  const resistanceStrengths = finalResistances.map(toLevelStrength);
  1702→  1702→
  1703→  1703→  // Price targets: all levels with isTarget=true
  1704→  1704→  const priceTargets = [...supportStrengths, ...resistanceStrengths].filter(l => l.isTarget);
  1705→  1705→
  1706→  1706→  return {
  1707→  1707→    resistances: resistanceStrengths.map(l => l.price),
  1708→  1708→    supports: supportStrengths.map(l => l.price),
  1709→  1709→    supportStrengths,
  1710→  1710→    resistanceStrengths,
  1711→  1711→    priceTargets,
  1712→  1712→  };
  1713→  1713→}

  1717→  1717→function calcTrend(closes: number[], period: number): TrendResult {
  1718→  1718→  const slice = closes.slice(-Math.min(period, closes.length));
  1719→  1719→  const { slope, r2 } = linearRegression(slice);
  1720→  1720→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
  1721→  1721→  // Convert slope/avgPrice to percentage before atan so that:
  1722→  1722→  //   1% daily rise → atan(1) = 45°, 0.5% → ~26.6°, 2% → ~63.4°
  1723→  1723→  // Without the ×100 factor, atan(slope/avgPrice) always yields near-0° for
  1724→  1724→  // large prices (e.g. index at 6.7M with 1% daily change → 0.57°).
  1725→  1725→  const angle = avgPrice === 0 ? 0 : Math.atan((slope / avgPrice) * 100) * (180 / Math.PI);
  1726→  1726→  const priceThreshold = avgPrice * 0.001; // 0.1%
  1727→  1727→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
  1728→  1728→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
  1729→  1729→}

  1736→  1736→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
  1737→  1737→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
  1738→  1738→  const slice = data.slice(0, barIdx + 1);
  1739→  1739→  const price = slice[slice.length - 1].close;
  1740→  1740→  let nearestR = price * 1.05;
  1741→  1741→  let nearestS = price * 0.95;
  1742→  1742→
  1743→  1743→  // Collect all swing highs (resistance) and swing lows (support)
  1744→  1744→  const resistances: number[] = [];
  1745→  1745→  const supports: number[] = [];
  1746→  1746→  for (const lb of [3, 5, 7]) {
  1747→  1747→    const swings = findSwingLevels(slice, lb);
  1748→  1748→    resistances.push(...swings.highs);
  1749→  1749→    supports.push(...swings.lows);
  1750→  1750→  }
  1751→  1751→
  1752→  1752→  // Find nearest resistance above price
  1753→  1753→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
  1754→  1754→  if (abovePrice.length > 0) nearestR = abovePrice[0];
  1755→  1755→
  1756→  1756→  // Find nearest support below price
  1757→  1757→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
  1758→  1758→  if (belowPrice.length > 0) nearestS = belowPrice[0];
  1759→  1759→
  1760→  1760→  return { R1: nearestR, S1: nearestS };
  1761→  1761→}

  1764→  1764→// Returns true if bullish divergence detected, false if bearish, null if none
  1765→  1765→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
  1766→  1766→  const lookback = 20;
  1767→  1767→  const start = Math.max(0, barIdx - lookback);
  1768→  1768→  const slice = data.slice(start, barIdx + 1);
  1769→  1769→  if (slice.length < 10) return null;
  1770→  1770→
  1771→  1771→  const mid = Math.floor(slice.length / 2);
  1772→  1772→  const firstHalf = slice.slice(0, mid);
  1773→  1773→  const secondHalf = slice.slice(mid);
  1774→  1774→
  1775→  1775→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
  1776→  1776→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
  1777→  1777→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
  1778→  1778→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
  1779→  1779→
  1780→  1780→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
  1781→  1781→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
  1782→  1782→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
  1783→  1783→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
  1784→  1784→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
  1785→  1785→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
  1786→  1786→
  1787→  1787→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
  1788→  1788→    return false; // bearish divergence
  1789→  1789→  }
  1790→  1790→
  1791→  1791→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
  1792→  1792→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
  1793→  1793→
  1794→  1794→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
  1795→  1795→    return true; // bullish divergence
  1796→  1796→  }
  1797→  1797→
  1798→  1798→  return null;
  1799→  1799→}
