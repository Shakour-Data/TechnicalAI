   134→// Component
   135→// ═══════════════════════════════════════════════════════════════════════════════
   136→
   137→export default function VdssGraph(props: VdssGraphProps) {
   138→  const { colors: C } = useTheme();
   139→  const {
   140→    symbolName,
   141→    currentPrice,
   142→    scenarios,
   143→    decisionGraph,
   144→  } = props;
   145→
   146→  const [activeFilter, setActiveFilter] = useState('all');
   147→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   148→  const graphRef = useRef<HTMLDivElement>(null);
   149→  const svgRef = useRef<SVGSVGElement>(null);
   150→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   151→  const shellRef = useRef<HTMLDivElement>(null);
   152→
   153→  // ── Derived data from decisionGraph ──────────────────────────────
   154→  const edges = decisionGraph?.edges ?? [];
   155→  const nodes = decisionGraph?.nodes ?? [];
   156→  const nodePositions = decisionGraph?.nodePositions ?? {};
   157→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   158→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   159→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   160→  const pathContributions = decisionGraph?.pathContributions ?? {};
   161→  const nodeValues = decisionGraph?.nodeValues ?? {};
   162→
   163→  // Build node lookup map
   164→  const nodeMap = useMemo(() => {
   165→    const map: Record<string, (typeof nodes)[number]> = {};
   166→    for (const n of nodes) map[n.id] = n;
   167→    return map;
   168→  }, [nodes]);
   169→
   170→  // ── Visible edges/nodes based on filter ──────────────────────────
   171→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   172→    if (activeFilter === 'all') {
   173→      const allEdgeIdx = edges.map((_, i) => i);
   174→      const allNodes = new Set(nodes.map(n => n.id));
   175→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   176→    }
   177→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   178→      // Show edges going TO this scenario node, plus their source nodes
   179→      const edgeSet = new Set<number>();
   180→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   181→      edges.forEach((e, i) => {
   182→        if (e.to === activeFilter) {
   183→          edgeSet.add(i);
   184→          nodeSet.add(e.from);
   185→        }
   186→      });
   187→      // Also show intermediate nodes connected to those sources
   188→      edges.forEach((e, i) => {
   189→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   190→          edgeSet.add(i);
   191→          nodeSet.add(e.from);
   192→        }
   193→      });
   194→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   195→    }
   196→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   197→      const branchType = `branch-${activeFilter}` as const;
   198→      const edgeSet = new Set<number>();
   199→      const nodeSet = new Set<string>(['ROOT']);
   200→      edges.forEach((e, i) => {
   201→        if (e.type === branchType || (e.type !== `branch-trend` && e.type !== `branch-breakout` && e.type !== `branch-reversal` && nodeSet.has(e.from))) {
   202→          // Include branch-type edges
   203→          if (e.type === branchType) {
   204→            edgeSet.add(i);
   205→            nodeSet.add(e.from);
   206→            nodeSet.add(e.to);
   207→          }
   208→        }
   209→      });
   210→      // Now add all edges whose source is already in nodeSet
   211→      const expanded = new Set<number>();
   212→      const expandedNodes = new Set<string>(nodeSet);
   213→      edges.forEach((e, i) => {
   214→        if (expandedNodes.has(e.from)) {
   215→          expanded.add(i);
   216→          expandedNodes.add(e.to);
   217→        }
   218→      });
   219→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   220→    }
   221→    // Type filter (up/pullback/down/risk)
   222→    const edgeSet = new Set<number>();
   223→    const nodeSet = new Set<string>(['ROOT']);
   224→    edges.forEach((e, i) => {
   225→      if (e.type === activeFilter) {
   226→        edgeSet.add(i);
   227→        nodeSet.add(e.from);
   228→        nodeSet.add(e.to);
   229→      }
   230→    });
   231→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   232→  }, [activeFilter, edges, nodes]);
   233→
   234→  // ── Draw SVG edges ───────────────────────────────────────────────
   235→  const drawEdges = useCallback(() => {
   236→    const svg = svgRef.current;
   237→    const graph = graphRef.current;
   238→    if (!svg || !graph || edges.length === 0) return;
   239→
   240→    let markersSvg = '';
   241→    const seenColors = new Set<string>();
   242→    for (const e of edges) {
   243→      const c = EDGE_COLORS[e.type];
   244→      if (c && !seenColors.has(e.type)) {
   245→        seenColors.add(e.type);
   246→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   247→      }
   248→    }
   249→
   250→    let pathsSvg = '';
   251→    const graphRect = graph.getBoundingClientRect();
   252→
   253→    for (let ei = 0; ei < edges.length; ei++) {
   254→      const { from: fromId, to: toId, label, type } = edges[ei];
   255→      const isVisible = visibleEdgeIndices.includes(ei);
   256→      const fromEl = nodeRefs.current[fromId];
   257→      const toEl = nodeRefs.current[toId];
   258→      if (!fromEl || !toEl) continue;
   259→
   260→      const fRect = fromEl.getBoundingClientRect();
   261→      const tRect = toEl.getBoundingClientRect();
   262→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   263→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   264→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   265→      const by = tRect.top - graphRect.top + tRect.height / 2;
   266→      const dx = bx - ax;
   267→      const dy = by - ay;
   268→      const dist = Math.sqrt(dx * dx + dy * dy);
   269→
   270→      if (dist === 0) continue;
   271→
   272→      // Curved edges with bend proportional to distance
   273→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   274→      const mx = (ax + bx) / 2;
   275→      const my = (ay + by) / 2;
   276→      const nx = -dy / dist;
   277→      const ny = dx / dist;
   278→      const cx = mx + nx * bend;
   279→      const cy = my + ny * bend;
   280→
   281→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   282→      const prob = edgeProbabilities[ei] ?? 0;
   283→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   284→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   285→
   286→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.72 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   287→
   288→      // Edge labels — skip deterministic root→branch and branch→event edges (index < 6)
   289→      if (isVisible && ei >= 6) {
   290→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#cde4ef" font-size="10" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="4" stroke-linejoin="round" opacity="0.9" data-type="${type}" class="edge-label">${label}</text>`;
   291→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3" stroke-linejoin="round" opacity="0.85" class="edge-prob">${probLabel}</text>`;
   292→      }
   293→    }
   294→
   295→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   296→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   297→
   298→  useEffect(() => { drawEdges(); }, [drawEdges]);
   299→
   300→  useEffect(() => {
   301→    const el = shellRef.current;
   302→    if (!el) return;
   303→    const observer = new ResizeObserver(() => drawEdges());
   304→    observer.observe(el);
   305→    return () => observer.disconnect();
   306→  }, [drawEdges]);
   307→
   308→
   309→  // ── Detail panel content ─────────────────────────────────────────
   310→  const detailContent = useMemo(() => {
   311→    if (!selectedNode) return null;
   312→    const node = nodeMap[selectedNode];
   313→    if (!node) return null;
   314→
   315→    // For terminal scenario nodes, show branch breakdown
   316→    if (node.isTerminal && SCENARIO_KEYS.includes(selectedNode as typeof SCENARIO_KEYS[number])) {
   317→      const contrib = pathContributions[selectedNode] ?? { trend: 0, breakout: 0, reversal: 0 };
   318→      const meta = SCENARIO_META[selectedNode];
   319→      const s = scenarios[selectedNode as keyof typeof scenarios];
   320→      return (
   321→        <div style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}>
   322→          <h3 style={{ fontSize: '14px', fontWeight: 700, margin: 0, color: meta.color }}>{node.title}</h3>
   323→          <p style={{ fontSize: '12px', color: '#c4d8e3', lineHeight: 1.95, margin: 0 }}>{node.desc}</p>
   324→          <span style={{
   325→            display: 'inline-block', padding: '4px 8px', margin: '3px 2px', borderRadius: 8,
   326→            color: '#d7eaf1', fontSize: '10px',
   327→            background: 'rgba(255,255,255,.06)', border: '1px solid rgba(255,255,255,.09)',
   328→          }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   329→          <div style={{ fontSize: '12px', color: '#c4d8e3' }} dir="ltr">{nodeValues[selectedNode] ?? '--'}</div>
   330→
   331→          <div style={{ marginTop: 12, paddingTop: 12, borderTop: `1px solid ${D.line}` }}>
   332→            <p style={{ fontSize: '12px', fontWeight: 700, marginBottom: 8, color: '#d8eff9' }}>سهم هر استراتژی:</p>
   333→            {Object.entries(BRANCH_META).map(([bKey, bMeta]) => {
   334→              const val = contrib[bKey as 'trend' | 'breakout' | 'reversal'];
   335→              const pct = (val * 100).toFixed(1);
   336→              return (
   337→                <div key={bKey} style={{
   338→                  display: 'flex', alignItems: 'center', justifyContent: 'space-between',
   339→                  padding: '9px 0', borderBottom: `1px dashed rgba(175,210,225,.15)`,
   340→                }}>
   341→                  <div style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
   342→                    <span style={{ display: 'inline-block', width: 8, height: 8, borderRadius: '50%', background: bMeta.color }} />
   343→                    <span style={{ fontSize: '11px', color: '#d8eff9' }}>{bMeta.label}</span>
   344→                  </div>
   345→                  <span style={{ fontSize: '11px', fontWeight: 700, color: bMeta.color }}>{toPersianDigits(pct)}٪</span>
   346→                </div>
   347→              );
   348→            })}
   349→            <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '9px 0' }}>
   350→              <span style={{ fontSize: '11px', fontWeight: 700, color: '#d8eff9' }}>مجموع</span>
   351→              <span style={{ fontSize: '11px', fontWeight: 900, color: meta.color }}>{toFa(s?.probability ?? 0)}٪</span>
   352→            </div>
   353→          </div>
   354→        </div>
   355→      );
   356→    }
   357→
   358→    // For non-terminal nodes, show incoming/outgoing edges
   359→    const inputs = edges.filter(e => e.to === selectedNode);
   360→    const outputs = edges.filter(e => e.from === selectedNode);
   361→
   362→    return (
   363→      <div style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}>
   364→        <h3 style={{ fontSize: '14px', fontWeight: 700, margin: 0, color: '#d8eff9' }}>{node.title}</h3>
   365→        <p style={{ fontSize: '12px', color: '#c4d8e3', lineHeight: 1.95, margin: 0 }}><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   366→        <span style={{
   367→          display: 'inline-block', padding: '4px 8px', margin: '3px 2px', borderRadius: 8,
   368→          color: '#d7eaf1', fontSize: '10px',
   369→          background: 'rgba(255,255,255,.06)', border: '1px solid rgba(255,255,255,.09)',
   370→        }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   371→        <p style={{ fontSize: '12px', color: '#c4d8e3', lineHeight: 1.95, margin: 0 }}>{node.desc}</p>
   372→        {inputs.length > 0 && (
   373→          <div>
   374→            <p style={{ fontSize: '12px', fontWeight: 600, marginBottom: 4, color: '#d8eff9' }}>مسیرهای ورودی ({toFa(inputs.length)}):</p>
   375→            <ul style={{ margin: '12px 0 0', padding: 0, listStyle: 'none' }}>
   376→              {inputs.map((e, i) => {
   377→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   378→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   379→                return (
   380→                  <li key={i} style={{
   381→                    padding: '9px 0', borderTop: '1px dashed rgba(175,210,225,.15)',
   382→                    fontSize: '11px', lineHeight: 1.8, color: '#b8cfdb',
   383→                  }}>
   384→                    <b style={{ color: '#d8eff9' }}>{SCENARIO_DISPLAY[e.from] || e.from} ← {SCENARIO_DISPLAY[e.to] || e.to}</b>
   385→                    <span style={{
   386→                      marginRight: 8, padding: '2px 6px', borderRadius: 4,
   387→                      fontSize: '9px', fontWeight: 700,
   388→                      background: `${edgeColor}22`, color: edgeColor,
   389→                    }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   390→                    <br />{e.label}
   391→                  </li>
   392→                );
   393→              })}
   394→            </ul>
   395→          </div>
   396→        )}
   397→        {outputs.length > 0 && (
   398→          <div>
   399→            <p style={{ fontSize: '12px', fontWeight: 600, marginBottom: 4, color: '#d8eff9' }}>مسیرهای خروجی ({toFa(outputs.length)}):</p>
   400→            <ul style={{ margin: '12px 0 0', padding: 0, listStyle: 'none' }}>
   401→              {outputs.map((e, i) => {
   402→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   403→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   404→                return (
   405→                  <li key={i} style={{
   406→                    padding: '9px 0', borderTop: '1px dashed rgba(175,210,225,.15)',
   407→                    fontSize: '11px', lineHeight: 1.8, color: '#b8cfdb',
   408→                  }}>
   409→                    <b style={{ color: '#d8eff9' }}>{SCENARIO_DISPLAY[e.from] || e.from} → {SCENARIO_DISPLAY[e.to] || e.to}</b>
   410→                    <span style={{
   411→                      marginRight: 8, padding: '2px 6px', borderRadius: 4,
   412→                      fontSize: '9px', fontWeight: 700,
   413→                      background: `${edgeColor}22`, color: edgeColor,
   414→                    }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   415→                    <br />{e.label}
   416→                  </li>
   417→                );
   418→              })}
   419→            </ul>
   420→          </div>
   421→        )}
   422→      </div>
   423→    );
   424→  }, [selectedNode, nodeMap, nodeValues, edges, edgeProbabilities, pathContributions, scenarios]);
   425→
   426→  // ── All filter buttons ────────────────────────────────────────────
   427→  const allFilters = [
   428→    ...TYPE_FILTERS,
   429→    { key: 'sep1', label: '│', isSep: true as const },
   430→    ...Object.entries(BRANCH_META).map(([k, v]) => ({ key: k, label: v.label, branchKey: k, branchColor: v.color })),
   431→    { key: 'sep2', label: '│', isSep: true as const },
   432→    ...SCENARIO_KEYS.map(k => ({ key: k, label: SCENARIO_META[k].label, scenarioKey: k })),
   433→  ];
   434→
   435→  // ═══ Design dimensions ═══

   841→function ProbabilityTrendTable({ data }: { data: ProbabilityTrendResult }) {
   842→  const { colors: C } = useTheme();
   843→  const [open, setOpen] = useState(true);
   844→
   845→  // Build a lookup: scenarioKey -> trend array
   846→  const scenarioMap = useMemo(() => {
   847→    const m: Record<string, ProbabilityTrendResult['scenarios'][number]> = {};
   848→    for (const s of data.scenarios) m[s.scenarioKey] = s;
   849→    return m;
   850→  }, [data.scenarios]);
   851→
   852→  // Group map
   853→  const groupMap = useMemo(() => {
   854→    const m: Record<string, ProbabilityTrendResult['groups'][number]> = {};
   855→    for (const g of data.groups) m[g.group] = g;
   856→    return m;
   857→  }, [data.groups]);
   858→
   859→  // Get DayPoint by day number (1-indexed)
   860→  const getDay = (trend: DayPoint[], day: number) =>
   861→    trend.find(d => d.day === day);
   862→
   863→  // Format probability as percentage string with 2 decimals
   864→  const fmtPct = (v: number) => toPersianDigits((v * 100).toFixed(2)) + '٪';
   865→
   866→  return (
   867→    <Collapsible open={open} onOpenChange={setOpen} dir="rtl">
   868→      <CollapsibleTrigger className="w-full mt-4 flex items-center justify-between px-4 py-3 rounded-2xl border border-[#e5e7eb] bg-[#ffffff] cursor-pointer hover:bg-[#f9fafb] transition-colors"
   869→        style={{ boxShadow: '0 4px 16px rgba(0,0,0,.06)' }}>
   870→        <div className="flex items-center gap-3">
   871→          <div className="w-10 h-10 rounded-xl border border-emerald-200 flex items-center justify-center text-emerald-800 text-xl font-bold"
   872→            style={{ boxShadow: 'inset 0 0 22px rgba(5,150,105,.06), 0 0 22px rgba(5,150,105,.04)' }}>📈</div>
   873→          <div className="text-right">
   874→            <h2 className="text-sm font-bold text-[#111827]">روند ۳۰ روزه احتمالات</h2>
   875→            <p className="text-[11px] text-[#6b7280]">توزیع روزانه احتمال سناریوها — مدل زوال نمایی دوگانه + احتمال تجمعی CDF</p>
   876→          </div>
   877→        </div>
   878→        <span className={`text-[#6b7280] transition-transform duration-200 ${open ? 'rotate-180' : ''}`}>▼</span>
   879→      </CollapsibleTrigger>
   880→
   881→      <CollapsibleContent>
   882→        <div className="mt-2 rounded-2xl border border-[#e5e7eb] bg-[#ffffff] p-4 overflow-x-auto"
   883→          style={{ boxShadow: '0 4px 16px rgba(0,0,0,.06)' }}>
   884→          <Table className="text-[11px] min-w-[900px]">
   885→            <TableHeader>
   886→              <TableRow className="border-b border-[#e5e7eb]">
   887→                <TableHead className="text-right text-[#374151] font-bold px-2 py-2 min-w-[120px]">سناریو</TableHead>
   888→                {TREND_DAYS.map(day => (
   889→                  <TableHead key={day} colSpan={2} className="text-center text-[#374151] font-bold px-1.5 py-2 border-r border-[#e5e7eb]">
   890→                    روز {toPersianDigits(day.toString())}
   891→                  </TableHead>
   892→                ))}
   893→                <TableHead className="text-center text-[#374151] font-bold px-2 py-2 border-r border-[#e5e7eb]">اوج</TableHead>
   894→              </TableRow>
   895→              <TableRow className="border-b border-[#e5e7eb] bg-[#f9fafb]">
   896→                <TableHead className="px-2 py-1" />
   897→                {TREND_DAYS.map(day => (
   898→                  <React.Fragment key={day}>
   899→                    <TableHead className="text-center text-[10px] text-[#6b7280] font-medium px-1 py-1">اختصاصی</TableHead>
   900→                    <TableHead className="text-center text-[10px] text-[#6b7280] font-medium px-1 py-1">تجمعی</TableHead>
   901→                  </React.Fragment>
   902→                ))}
   903→                <TableHead className="text-center text-[10px] text-[#6b7280] font-medium px-2 py-1">روز</TableHead>
   904→              </TableRow>
   905→            </TableHeader>
   906→            <TableBody>
   907→              {/* Scenario rows */}
   908→              {['R1','R2','R3','R4','R5','R6','R7','R8','R9'].map(key => {
   909→                const sc = scenarioMap[key];
   910→                if (!sc) return null;
   911→                const gc = TREND_GROUP_COLOR[sc.group];
   912→                return (
   913→                  <TableRow key={key} className="border-b border-[#e5e7eb]/60 hover:bg-[#f9fafb]">
   914→                    <TableCell className="px-2 py-1.5 font-bold" style={{ color: gc.text, minWidth: 120 }}>
   915→                      <span className="inline-block w-2 h-2 rounded-full ml-1.5" style={{ background: gc.text }} />
   916→                      {sc.label}
   917→                    </TableCell>
   918→                    {TREND_DAYS.map(day => {
   919→                      const dp = getDay(sc.trend, day);
   920→                      const isPeak = sc.peakDay === day;
   921→                      return (
   922→                        <React.Fragment key={day}>
   923→                          <TableCell
   924→                            className={`text-center px-1 py-1 tabular-nums ${isPeak ? 'font-black' : 'font-medium'}`}
   925→                            style={isPeak ? { background: gc.badge, color: gc.text, borderRadius: 4 } : { color: '#374151' }}
   926→                          >
   927→                            {dp ? fmtPct(dp.individualProb) : '—'}
   928→                          </TableCell>
   929→                          <TableCell
   930→                            className={`text-center px-1 py-1 tabular-nums ${isPeak ? 'font-black' : 'font-medium'}`}
   931→                            style={isPeak ? { background: gc.badge, color: gc.text, borderRadius: 4 } : { color: '#6b7280' }}
   932→                          >
   933→                            {dp ? fmtPct(dp.cumulativeProb) : '—'}
   934→                          </TableCell>
   935→                        </React.Fragment>
   936→                      );
   937→                    })}
   938→                    <TableCell className="text-center px-2 py-1 font-bold tabular-nums" style={{ color: gc.text }}>
   939→                      {toPersianDigits(sc.peakDay.toString())}
   940→                    </TableCell>
   941→                  </TableRow>
   942→                );
   943→              })}
   944→
   945→              {/* Group trend rows */}
   946→              <TableRow className="border-t-2 border-[#e5e7eb] bg-[#f3f4f6]">
   947→                <TableCell className="px-2 py-2 font-black text-[#111827]" colSpan={1}>
   948→                  روند گروهی
   949→                </TableCell>
   950→                {TREND_DAYS.map(day => {
   951→                  const bearG = groupMap['bearish'];
   952→                  const neutG = groupMap['neutral'];
   953→                  const bullG = groupMap['bullish'];
   954→                  const bCum = getDay(bearG?.trend ?? [], day)?.cumulativeProb ?? 0;
   955→                  const nCum = getDay(neutG?.trend ?? [], day)?.cumulativeProb ?? 0;
   956→                  const buCum = getDay(bullG?.trend ?? [], day)?.cumulativeProb ?? 0;
   957→                  const total = bCum + nCum + buCum;
   958→                  const bullPct = total > 0 ? buCum / total : 0;
   959→                  const barColor = bullPct > 0.55 ? '#16a34a' : bullPct < 0.45 ? '#dc2626' : '#6b7280';
   960→                  return (
   961→                    <TableCell key={day} colSpan={2} className="text-center px-1 py-2">
   962→                      <div className="flex items-center justify-center gap-1">
   963→                        <div className="w-12 h-1.5 rounded-full bg-[#e5e7eb] overflow-hidden">

   966→                        <span className="text-[10px] font-bold tabular-nums" style={{ color: barColor }}>
   967→                          {fmtPct(bullPct)}
   968→                        </span>

   970→                    </TableCell>

   977→                const gData = groupMap[g];
   978→                if (!gData) return null;
   979→                const gc = TREND_GROUP_COLOR[g];
   980→                const lastDay = getDay(gData.trend, 30);

   983→                    <TableCell className="px-2 py-1.5 font-bold" style={{ color: gc.text }}>
   984→                      <span className="inline-block w-1.5 h-1.5 rounded-full ml-1" style={{ background: gc.text }} />
   985→                      {TREND_GROUP_LABEL[g]}
   986→                    </TableCell>
   987→                    {TREND_DAYS.map(day => {
   988→                      const dp = getDay(gData.trend, day);
   989→                      return (
   990→                        <TableCell key={day} colSpan={2} className="text-center px-1 py-1 tabular-nums font-medium" style={{ color: gc.text }}>
   991→                          {dp ? fmtPct(dp.cumulativeProb) : '—'}
   992→                        </TableCell>
   993→                      );
   994→                    })}
   995→                    <TableCell className="text-center px-2 py-1 font-bold tabular-nums text-[10px]" style={{ color: gc.text }}>
   996→                      {lastDay ? fmtPct(lastDay.cumulativeProb) : '—'}
   997→                    </TableCell>
   998→                  </TableRow>
   999→                );
  1000→              })}
  1001→            </TableBody>
  1002→          </Table>
  1003→
  1004→          <div className="mt-3 space-y-2">
  1005→            <div className="px-4 py-2.5 rounded-lg border-r-3 border-emerald-700/60 bg-emerald-50 text-[11px] text-[#374151] leading-relaxed">
  1006→              <b>توضیح:</b> احتمال اختصاصی = احتمال وقوع سناریو در آن روز خاص.

  1011→            </div>

  1013→            {data.scenarios.filter(s => s.interpretation).map(s => (
  1014→              <div key={s.scenarioKey} className="px-4 py-1.5 text-[10px] text-[#6b7280]">
  1015→                <b>{s.label}:</b> {s.interpretation}
