    76→    76→function sma(closes: number[], period: number): number {
    77→    77→  if (closes.length < period) return 0;
    78→    78→  const slice = closes.slice(-period);
    79→    79→  return slice.reduce((a, b) => a + b, 0) / period;

    82→    82→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
    83→    83→function emaCalc(closes: number[], period: number): number {
    84→    84→  if (closes.length < period) return 0;
    85→    85→  const k = 2 / (period + 1);
    86→    86→  let emaVal = sma(closes.slice(0, period), period);
    87→    87→  for (let i = period; i < closes.length; i++) {
    88→    88→    emaVal = closes[i] * k + emaVal * (1 - k);
    89→    89→  }
    90→    90→  return emaVal;
    91→    91→}

    93→    93→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
    94→    94→function calcRSI(closes: number[], period: number = 14): number {
    95→    95→  if (closes.length < period + 1) return 50;
    96→    96→  let avgGain = 0;
    97→    97→  let avgLoss = 0;
    98→    98→  for (let i = 1; i <= period; i++) {
    99→    99→    const diff = closes[i] - closes[i - 1];
   100→   100→    if (diff > 0) avgGain += diff;
   101→   101→    else avgLoss += Math.abs(diff);
   102→   102→  }
   103→   103→  avgGain /= period;
   104→   104→  avgLoss /= period;
   105→   105→  for (let i = period + 1; i < closes.length; i++) {
   106→   106→    const diff = closes[i] - closes[i - 1];
   107→   107→    const gain = diff > 0 ? diff : 0;
   108→   108→    const loss = diff < 0 ? Math.abs(diff) : 0;
   109→   109→    avgGain = (avgGain * (period - 1) + gain) / period;
   110→   110→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   111→   111→  }
   112→   112→  if (avgLoss === 0) return 100;
   113→   113→  const rs = avgGain / avgLoss;
   114→   114→  return 100 - 100 / (1 + rs);
   115→   115→}

   117→   117→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   118→   118→function calcMFI(data: OHLCV[], period: number = 14): number {
   119→   119→  if (data.length < period + 1) return 50;
   120→   120→  let posFlow = 0;
   121→   121→  let negFlow = 0;
   122→   122→  for (let i = data.length - period; i < data.length; i++) {
   123→   123→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   124→   124→    const mf = tp * data[i].volume;
   125→   125→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   126→   126→    if (tp > prevTp) posFlow += mf;
   127→   127→    else negFlow += mf;
   128→   128→  }
   129→   129→  if (negFlow === 0) return 100;
   130→   130→  const mfr = posFlow / negFlow;
   131→   131→  return 100 - 100 / (1 + mfr);
   132→   132→}

   135→   135→function calcCCI(data: OHLCV[], period: number = 20): number {
   136→   136→  if (data.length < period) return 0;
   137→   137→  const tps: number[] = [];
   138→   138→  for (let i = data.length - period; i < data.length; i++) {
   139→   139→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   140→   140→  }
   141→   141→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   142→   142→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   143→   143→  if (meanDev === 0) return 0;
   144→   144→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   145→   145→}

   148→   148→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   149→   149→  if (data.length < kPeriod) return { k: 50, d: 50 };
   150→   150→  const rawKs: number[] = [];
   151→   151→  for (let i = kPeriod - 1; i < data.length; i++) {
   152→   152→    let lowest = Infinity;
   153→   153→    let highest = -Infinity;
   154→   154→    for (let j = i - kPeriod + 1; j <= i; j++) {
   155→   155→      if (data[j].low < lowest) lowest = data[j].low;
   156→   156→      if (data[j].high > highest) highest = data[j].high;
   157→   157→    }
   158→   158→    const range = highest - lowest;
   159→   159→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   160→   160→  }
   161→   161→  const smoothedK: number[] = [];
   162→   162→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   163→   163→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   164→   164→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   165→   165→  }
   166→   166→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   167→   167→  const dSlice = smoothedK.slice(-smoothD);
   168→   168→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   169→   169→  return { k, d };
   170→   170→}

   173→   173→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   174→   174→  if (data.length < period) return -50;
   175→   175→  const slice = data.slice(-period);
   176→   176→  let highest = -Infinity;
   177→   177→  let lowest = Infinity;
   178→   178→  for (const c of slice) {
   179→   179→    if (c.high > highest) highest = c.high;
   180→   180→    if (c.low < lowest) lowest = c.low;
   181→   181→  }
   182→   182→  const range = highest - lowest;
   183→   183→  if (range === 0) return -50;
   184→   184→  return ((highest - data[data.length - 1].close) / range) * -100;
   185→   185→}

   187→   187→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   188→   188→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   189→   189→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   190→   190→  const emaFast = emaCalc(closes, fast);
   191→   191→  const emaSlow = emaCalc(closes, slow);
   192→   192→  const macdLine = emaFast - emaSlow;
   193→   193→  // Calculate signal as EMA of MACD values
   194→   194→  const macdValues: number[] = [];
   195→   195→  const kFast = 2 / (fast + 1);
   196→   196→  const kSlow = 2 / (slow + 1);
   197→   197→  let eFast = sma(closes.slice(0, fast), fast);
   198→   198→  let eSlow = sma(closes.slice(0, slow), slow);
   199→   199→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   200→   200→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   201→   201→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   202→   202→    macdValues.push(eFast - eSlow);
   203→   203→  }
   204→   204→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   205→   205→  const kSig = 2 / (sig + 1);
   206→   206→  let signalVal = sma(macdValues.slice(0, sig), sig);
   207→   207→  for (let i = sig; i < macdValues.length; i++) {
   208→   208→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   209→   209→  }
   210→   210→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   211→   211→}

   213→   213→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   214→   214→function calcATR(data: OHLCV[], period: number = 14): number {
   215→   215→  if (data.length < 2) return 0;
   216→   216→  const trs: number[] = [];
   217→   217→  for (let i = 1; i < data.length; i++) {
   218→   218→    const tr = Math.max(
   219→   219→      data[i].high - data[i].low,
   220→   220→      Math.abs(data[i].high - data[i - 1].close),
   221→   221→      Math.abs(data[i].low - data[i - 1].close)
   222→   222→    );
   223→   223→    trs.push(tr);
   224→   224→  }
   225→   225→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   226→   226→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   227→   227→  for (let i = period; i < trs.length; i++) {
   228→   228→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   229→   229→  }
   230→   230→  return atrVal;
   231→   231→}

   233→   233→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   234→   234→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   235→   235→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   236→   236→  const plusDMs: number[] = [];
   237→   237→  const minusDMs: number[] = [];
   238→   238→  const trs: number[] = [];
   239→   239→  for (let i = 1; i < data.length; i++) {
   240→   240→    const upMove = data[i].high - data[i - 1].high;
   241→   241→    const downMove = data[i - 1].low - data[i].low;
   242→   242→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   243→   243→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   244→   244→    trs.push(Math.max(
   245→   245→      data[i].high - data[i].low,
   246→   246→      Math.abs(data[i].high - data[i - 1].close),
   247→   247→      Math.abs(data[i].low - data[i - 1].close)
   248→   248→    ));
   249→   249→  }
   250→   250→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   251→   251→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   252→   252→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   253→   253→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   254→   254→  const dxValues: number[] = [];
   255→   255→  for (let i = period; i < trs.length; i++) {
   256→   256→    smoothTR = smoothTR - smoothTR / period + trs[i];
   257→   257→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   258→   258→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   259→   259→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   260→   260→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   261→   261→    const diSum = diPlus + diMinus;
   262→   262→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   263→   263→    dxValues.push(dx);
   264→   264→  }
   265→   265→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   266→   266→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   267→   267→  for (let i = period; i < dxValues.length; i++) {
   268→   268→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   269→   269→  }
   270→   270→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   271→   271→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   272→   272→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   273→   273→}

   275→   275→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   276→   276→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   277→   277→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   278→   278→  let isLong = true;
   279→   279→  let af = step;
   280→   280→  let ep = data[0].high;
   281→   281→  let sar = data[0].low;
   282→   282→  for (let i = 1; i < data.length; i++) {
   283→   283→    const prevSar = sar;
   284→   284→    sar = prevSar + af * (ep - prevSar);
   285→   285→    // Ensure SAR is within bounds
   286→   286→    if (isLong) {
   287→   287→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   288→   288→      if (data[i].low < sar) {
   289→   289→        isLong = false;
   290→   290→        sar = ep;
   291→   291→        ep = data[i].low;
   292→   292→        af = step;
   293→   293→      } else {
   294→   294→        if (data[i].high > ep) {
   295→   295→          ep = data[i].high;
   296→   296→          af = Math.min(af + step, max);
   297→   297→        }
   298→   298→      }
   299→   299→    } else {
   300→   300→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   301→   301→      if (data[i].high > sar) {
   302→   302→        isLong = true;
   303→   303→        sar = ep;
   304→   304→        ep = data[i].high;
   305→   305→        af = step;
   306→   306→      } else {
   307→   307→        if (data[i].low < ep) {
   308→   308→          ep = data[i].low;
   309→   309→          af = Math.min(af + step, max);
   310→   310→        }
   311→   311→      }
   312→   312→    }
   313→   313→  }
   314→   314→  return sar;
   315→   315→}

   318→   318→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   319→   319→  if (closes.length < period) {
   320→   320→    const c = closes[closes.length - 1] ?? 0;
   321→   321→    return { upper: c, middle: c, lower: c };
   322→   322→  }
   323→   323→  const slice = closes.slice(-period);
   324→   324→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   325→   325→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   326→   326→  const sd = Math.sqrt(variance);
   327→   327→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   328→   328→}

   340→   340→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   341→   341→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   342→   342→  const n = values.length;
   343→   343→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   344→   344→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   345→   345→  for (let i = 0; i < n; i++) {
   346→   346→    sumX += i;
   347→   347→    sumY += values[i];
   348→   348→    sumXY += i * values[i];
   349→   349→    sumX2 += i * i;
   350→   350→    sumY2 += values[i] * values[i];
   351→   351→  }
   352→   352→  const denom = n * sumX2 - sumX * sumX;
   353→   353→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   354→   354→  const slope = (n * sumXY - sumX * sumY) / denom;
   355→   355→  const intercept = (sumY - slope * sumX) / n;
   356→   356→  const meanY = sumY / n;
   357→   357→  let ssRes = 0, ssTot = 0;
   358→   358→  for (let i = 0; i < n; i++) {
   359→   359→    const predicted = slope * i + intercept;
   360→   360→    ssRes += (values[i] - predicted) ** 2;
   361→   361→    ssTot += (values[i] - meanY) ** 2;
   362→   362→  }
   363→   363→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   364→   364→  return { slope, intercept, r2 };
   365→   365→}

   368→   368→function roundToNice(price: number): number {
   369→   369→  const abs = Math.abs(price);
   370→   370→  let step: number;
   371→   371→  if (abs >= 100000) step = 100;
   372→   372→  else if (abs >= 10000) step = 50;
   373→   373→  else if (abs >= 1000) step = 10;
   374→   374→  else if (abs >= 100) step = 5;
   375→   375→  else step = 1;
   376→   376→  return Math.round(price / step) * step;
   377→   377→}

   380→   380→function findSwingLevels(data: OHLCV[]): { highs: number[]; lows: number[] } {
   381→   381→  const highs: number[] = [];
   382→   382→  const lows: number[] = [];
   383→   383→  const lookback = 3;
   384→   384→  for (let i = lookback; i < data.length - lookback; i++) {
   385→   385→    let isHigh = true;
   386→   386→    let isLow = true;
   387→   387→    for (let j = i - lookback; j <= i + lookback; j++) {
   388→   388→      if (j === i) continue;
   389→   389→      if (data[j].high >= data[i].high) isHigh = false;
   390→   390→      if (data[j].low <= data[i].low) isLow = false;
   391→   391→    }
   392→   392→    if (isHigh) highs.push(data[i].high);
   393→   393→    if (isLow) lows.push(data[i].low);
   394→   394→  }
   395→   395→  return { highs, lows };
   396→   396→}

   398→   398→// ─── Support / Resistance (5 each, rounded) ──────────────────────────────────
   399→   399→function calcSupportResistance(data: OHLCV[], currentPrice: number): { resistances: number[]; supports: number[] } {
   400→   400→  const allLevels: number[] = [];
   401→   401→
   402→   402→  // 1. Pivot Points from previous day
   403→   403→  const prev = data[data.length - 2] ?? data[data.length - 1];
   404→   404→  if (prev) {
   405→   405→    const pp = (prev.high + prev.low + prev.close) / 3;
   406→   406→    const hl = prev.high - prev.low;
   407→   407→    const pivotR = [
   408→   408→      2 * pp - prev.low,
   409→   409→      pp + hl,
   410→   410→      prev.high + 2 * (pp - prev.low),
   411→   411→      prev.high + 2 * (pp - prev.low) + hl,
   412→   412→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   413→   413→    ];
   414→   414→    const pivotS = [
   415→   415→      2 * pp - prev.high,
   416→   416→      pp - hl,
   417→   417→      prev.low - 2 * (prev.high - pp),
   418→   418→      prev.low - 2 * (prev.high - pp) - hl,
   419→   419→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   420→   420→    ];
   421→   421→    allLevels.push(...pivotR, ...pivotS);
   422→   422→  }
   423→   423→
   424→   424→  // 2. Swing levels
   425→   425→  const swings = findSwingLevels(data);
   426→   426→  allLevels.push(...swings.highs, ...swings.lows);
   427→   427→
   428→   428→  // 3. Round all
   429→   429→  const rounded = allLevels.map(roundToNice);
   430→   430→
   431→   431→  // 4. Remove duplicates (within 0.5%)
   432→   432→  const unique: number[] = [];
   433→   433→  for (const level of rounded) {
   434→   434→    if (level <= 0) continue;
   435→   435→    const isDup = unique.some(u => Math.abs(u - level) / u < 0.005);
   436→   436→    if (!isDup) unique.push(level);
   437→   437→  }
   438→   438→
   439→   439→  // 5. Separate and sort
   440→   440→  const above = unique.filter(l => l > currentPrice).sort((a, b) => a - b);
   441→   441→  const below = unique.filter(l => l < currentPrice).sort((a, b) => b - a);
   442→   442→
   443→   443→  return {
   444→   444→    resistances: above.slice(0, 5),
   445→   445→    supports: below.slice(0, 5),
   446→   446→  };
   447→   447→}

   450→   450→function calcTrend(closes: number[], period: number): TrendResult {
   451→   451→  const slice = closes.slice(-Math.min(period, closes.length));
   452→   452→  const { slope, r2 } = linearRegression(slice);
   453→   453→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   454→   454→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   455→   455→  const priceThreshold = avgPrice * 0.001; // 0.1%
   456→   456→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   457→   457→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   458→   458→}
