   206→   206→        `اشباع فروش قابل توجه (RSI: ${toFa(rsi)}${mfiWarning}) در محدوده حمایت ${toFa(S1_level)} شناسایی شد. این شرایط می‌تواند فرصت مناسبی برای ورود باشد، با حد ضرر ${toFa(S2_level)}. هدف اول ${toFa(R1_level)} و هدف دوم ${toFa(R2_level)} ریال.`,

   666→   666→function LevelBox({ index, label, price, strength, isTarget, type }: {
   667→   667→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target';
   668→   668→}) {
   669→   669→  const isSupport = type === 'support';
   670→   670→  const isTargetType = type === 'target';
   671→   671→
   672→   672→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   673→   673→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   674→   674→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   675→   675→
   676→   676→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   677→   677→  const strengthLabel = strength >= 8 ? 'بسیار قوی' : strength >= 6 ? 'قوی' : strength >= 4 ? 'متوسط' : strength >= 2 ? 'ضعیف' : 'بسیار ضعیف';
   678→   678→  const strengthColor = strength >= 8 ? '#34c98b' : strength >= 6 ? '#3ad5db' : strength >= 4 ? '#ffb11b' : strength >= 2 ? '#ff7b32' : '#ef4d62';
   679→   679→
   680→   680→  // Distance from current price (if available from context)
   681→   681→  return (
   682→   682→    <div
   683→   683→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   684→   684→      style={{
   685→   685→        background: bgAccent,
   686→   686→        border: `1px solid ${borderAccent}`,
   687→   687→        borderRight: `3px solid ${accentColor}`,
   688→   688→      }}
   689→   689→    >
   690→   690→      {/* Label + Price */}
   691→   691→      <div className="flex-1 min-w-0">
   692→   692→        <div className="flex items-center gap-2">
   693→   693→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   694→   694→          {isTarget && <span className="text-[9px]">🎯</span>}
   695→   695→        </div>
   696→   696→        <span className="text-sm font-black text-gray-100 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   697→   697→        <span className="text-[10px] text-gray-500 mr-1">ریال</span>
   698→   698→      </div>
   699→   699→
   700→   700→      {/* Strength bar */}
   701→   701→      <div className="flex flex-col items-end gap-1 min-w-[80px]">
   702→   702→        <div className="flex items-center gap-1.5">
   703→   703→          <span className="text-[9px] font-medium" style={{ color: strengthColor }}>{strengthLabel}</span>
   704→   704→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(strength)}</span>
   705→   705→          <span className="text-[9px] text-gray-500">/۱۰</span>
   706→   706→        </div>
   707→   707→        <div className="w-full h-1.5 rounded-full bg-white/5 overflow-hidden">

   714→   714→    </div>
