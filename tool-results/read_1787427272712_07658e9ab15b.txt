   140→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,

   144→    supportStrengths, resistanceStrengths, priceTargets,
   145→    isIndex, isTgju, currencyUnit,
   146→    edgeWeights,
   147→  } = props;
   148→
   149→  const unit = currencyUnit || 'ریال';
   150→
   151→  // ── Instrument Context (for intelligent naming) ─────────────────
   152→  const instrumentLabel = useMemo(() => {
   153→    if (isIndex) return { name: `شاخص ${symbolName}`, type: 'index' as const, verb: 'قرار دارد' };
   154→    if (isTgju) return { name: symbolName, type: 'tgju' as const, verb: 'قرار دارد' };
   155→    return { name: `سهم ${symbolName}`, type: 'stock' as const, verb: 'قرار دارد' };
   156→  }, [symbolName, isIndex, isTgju]);
   157→
   158→  const IL = instrumentLabel;
   159→
   160→  const containerRef = useRef<HTMLDivElement>(null);
   161→  const textAnalysisRef = useRef<HTMLDivElement>(null);
   162→  const chartApiRef = useRef<IChartApi | null>(null);
   163→  const [exporting, setExporting] = useState(false);
   164→  const [exportMenuOpen, setExportMenuOpen] = useState(false);
   165→  const exportMenuRef = useRef<HTMLDivElement>(null);
   166→
   167→  // ── AI Analysis state ───────────────────────────────────────────
   168→  const [aiAnalysis, setAiAnalysis] = useState<string | null>(null);
   169→  const [aiLoading, setAiLoading] = useState(false);
   170→  const [aiError, setAiError] = useState<string | null>(null);
   171→  const aiFetchIdRef = useRef(0);
   172→  // Track the current AbortController + its timeout for proper cleanup
   173→  const abortControllerRef = useRef<{ controller: AbortController; timeoutId: ReturnType<typeof setTimeout> } | null>(null);
   174→
   175→  // Cleanup any in-flight fetch (abort + clear timeout)
   176→  const cancelInFlightFetch = useCallback(() => {
   177→    if (abortControllerRef.current) {
   178→      clearTimeout(abortControllerRef.current.timeoutId);
   179→      try {
   180→        abortControllerRef.current.controller.abort('cancelled');
   181→      } catch { /* ignore */ }
   182→      abortControllerRef.current = null;
   183→    }
   184→  }, []);
   185→
   186→  // Cleanup on unmount
   187→  useEffect(() => {
   188→    return () => cancelInFlightFetch();
   189→  }, [cancelInFlightFetch]);
   190→
   191→  // ── Analysis version selector (V7 locked / V8 / V9 locked / V10 locked / V11 active) ─────────
   192→  const [analysisVersion, setAnalysisVersion] = useState<7 | 8 | 9 | 10 | 11>(11);
   193→  // ── V11 Debug data ──────────────────────────────────────────────
   194→  const [v11Debug, setV11Debug] = useState<any>(null);
   195→
   196→  // ── V11 Scenarios (built from TA data + S/R levels) ───────────────
   197→  const v11Result = useMemo(() => {
   198→    if (analysisVersion !== 11) return null;
   199→    return buildV11Scenarios({
   200→      currentPrice,
   201→      atr,
   202→      overallSignal: overallSignal as 'bullish' | 'bearish' | 'neutral',
   203→      v9Scenarios: scenarios,
   204→      supports,
   205→      resistances,
   206→      bollingerUpper,
   207→      bollingerLower,
   208→      bollingerMiddle,
   209→      ma21,
   210→      ma100,
   211→    });
   212→  }, [analysisVersion, currentPrice, atr, overallSignal, scenarios, supports, resistances,
   213→      bollingerUpper, bollingerLower, bollingerMiddle, ma21, ma100]);
   214→
   215→  // ── V10 Scenarios (LOCKED — backward compat) ───────────────
   216→  const v10Result = useMemo(() => {
   217→    if (analysisVersion !== 10) return null;
   218→    return buildV10Locked({
   219→      currentPrice,
   220→      atr,
   221→      overallSignal: overallSignal as 'bullish' | 'bearish' | 'neutral',
   222→      v9Scenarios: scenarios,
   223→      supports,
   224→      resistances,
   225→      bollingerUpper,
   226→      bollingerLower,
   227→      bollingerMiddle,
   228→      ma21,
   229→      ma100,
   230→    });
   231→  }, [analysisVersion, currentPrice, atr, overallSignal, scenarios, supports, resistances,
   232→      bollingerUpper, bollingerLower, bollingerMiddle, ma21, ma100]);
   233→
   234→  // ── Fetch AI analysis (robust with logging + auto-retry) ─────────────
   235→  const fetchAiAnalysis = useCallback(async (isRetry = false) => {
   236→    // Cancel any previous in-flight fetch first
   237→    cancelInFlightFetch();
   238→
   239→    const fetchId = ++aiFetchIdRef.current;
   240→    setAiLoading(true);
   241→    setAiError(null);
   242→    try {
   243→      const apiUrl = analysisVersion === 11 ? '/api/v11-analysis' : analysisVersion === 10 ? '/api/v10-analysis' : analysisVersion === 9 ? '/api/v9-analysis' : analysisVersion === 8 ? '/api/v8-analysis' : '/api/vdes-analysis';
   244→      // V10/V11: use S-key scenarios, otherwise use V9 R-key scenarios
   245→      const sKeyResult = analysisVersion === 11 ? v11Result : analysisVersion === 10 ? v10Result : null;
   246→      const activeScenarios = sKeyResult ? sKeyResult.scenarios : scenarios;
   247→      const requestBody: any = {
   248→          symbolName, currentPrice, ma21, ma100,
   249→          rsi, mfi, cci, adx, stochK, stochD,
   250→          macdLine, macdSignal, macdHist,
   251→          diPlus, diMinus, sar, atr, obv, hasVolume,
   252→          bollingerUpper, bollingerMiddle, bollingerLower,
   253→          trendDirection, trendAngle, trendR2, overallSignal,
   254→          scenarios: activeScenarios, resistances, supports,
   255→          resistanceStrengths, supportStrengths,
   256→          isIndex, isTgju, currencyUnit,
   257→        };
   258→      // V10/V11: pass dominant + highest cumulative keys for narrative consistency
   259→      if (sKeyResult) {
   260→        if (analysisVersion === 11) {
   261→          requestBody.v11DominantKey = sKeyResult.dominantKey;
   262→          requestBody.v11HighestCumulativeKey = sKeyResult.highestCumulativeKey;
   263→        } else if (analysisVersion === 10) {
   264→          requestBody.v10DominantKey = sKeyResult.dominantKey;
   265→          requestBody.v10HighestCumulativeKey = sKeyResult.highestCumulativeKey;
   266→        }
   267→      }
   268→      // V9/V10/V11: include OHLCV history for ML training/prediction
   269→      // Cap at last 500 candles to keep request payload small (ML needs ~120, patterns ~50)
   270→      if ((analysisVersion === 9 || analysisVersion === 10 || analysisVersion === 11) && candles.length > 0) {
   271→        const recentCandles = candles.length > 500 ? candles.slice(-500) : candles;
   272→        requestBody.ohlcvHistory = recentCandles.map(c => ({
   273→          date: c.date, open: c.open, high: c.high,
   274→          low: c.low, close: c.close, volume: c.volume,
   275→        }));
   276→      }
   277→
   278→      // AbortController for 120s client-side timeout
   279→      const controller = new AbortController();
   280→      const timeoutId = setTimeout(() => {
   281→        controller.abort('Client timeout after 120s');
   282→      }, 120000);
   283→      // Store for cleanup on re-fetch / unmount
   284→      abortControllerRef.current = { controller, timeoutId };
   285→
   286→      console.log(`[VdesAnalysis v${analysisVersion}] Fetching ${apiUrl} for ${symbolName} (retry=${isRetry})`);
   287→      const res = await fetch(apiUrl, {
   288→        method: 'POST',
   289→        headers: { 'Content-Type': 'application/json' },
   290→        body: JSON.stringify(requestBody),
   291→        signal: controller.signal,
   292→      });
   293→      // Fetch completed — clear timeout and ref
   294→      clearTimeout(timeoutId);
   295→      abortControllerRef.current = null;
   296→
   297→      if (!res.ok) {
   298→        let errorMsg = `خطای سرور (${res.status})`;
   299→        try {
   300→          const errData = await res.json();
   301→          errorMsg = errData.error || errorMsg;
   302→        } catch {
   303→          // Response body is not JSON (e.g. 502 from Caddy)
   304→          if (res.status === 502) errorMsg = 'سرور در حال بارگذاری مجدد است. لطفاً چند ثانیه صبر کنید و دوباره تلاش کنید.';
   305→        }
   306→        console.error(`[VdesAnalysis v${analysisVersion}] API error ${res.status}:`, errorMsg);
   307→        throw new Error(errorMsg);
   308→      }
   309→      const data = await res.json();
   310→      if (fetchId === aiFetchIdRef.current) {
   311→        if (data.analysis && data.analysis.trim().length > 0) {
   312→          setAiAnalysis(data.analysis);
   313→          console.log(`[VdesAnalysis v${analysisVersion}] Got analysis for ${symbolName}: ${data.analysis.length} chars`);
   314→        } else {
   315→          console.error(`[VdesAnalysis v${analysisVersion}] Empty analysis for ${symbolName}`, data);
   316→          setAiError('مدل پاسخ خالی تولید کرد. لطفاً دوباره تلاش کنید.');
   317→        }
   318→        if (data._debug?.v11) setV11Debug({ ...data._debug.v11, v11Persona: data._debug.v11Persona, v11Template: data._debug.v11Template });
   319→        else if (data._debug?.v10) setV11Debug({ ...data._debug.v10, v11Persona: data._debug.v10Persona, v10Template: data._debug.v10Template });
   320→        else setV11Debug(null);
   321→      }
   322→    } catch (err) {
   323→      // Clear timeout and ref on error too
   324→      clearTimeout(abortControllerRef.current?.timeoutId);
   325→      abortControllerRef.current = null;
   326→
   327→      // Determine if this is an abort
   328→      const errName = err instanceof Error ? err.name : '';
   329→      const errMsg = err instanceof Error ? err.message : '';
   330→      const isAbort = errName === 'AbortError';
   331→      const isIntentionalCancel = isAbort && errMsg === 'cancelled';
   332→
   333→      // Silently ignore intentional cancellations (superseded by new fetch or unmount)
   334→      if (isIntentionalCancel) {
   335→        return;
   336→      }
   337→
   338→      console.error(`[VdesAnalysis v${analysisVersion}] Error for ${symbolName}:`, err);
   339→      if (fetchId === aiFetchIdRef.current) {
   340→        if (isAbort) {
   341→          setAiError('زمان پاسخگویی سرور به پایان رسید. لطفاً دوباره تلاش کنید.');
   342→        } else {
   343→          setAiError(errMsg || 'خطای ناشناخته');
   344→        }
   345→        // Auto-retry once on transient errors (503/429/timeout/abort) after 5s
   346→        if (!isRetry) {
   347→          const isTransient = errMsg.includes('503') || errMsg.includes('429') || errMsg.includes('پایان رسید') || errMsg.includes('بارگذاری مجدد') || errMsg.includes('timeout') || isAbort;
   348→          if (isTransient) {
   349→            console.log(`[VdesAnalysis] Auto-retry in 5s for ${symbolName}`);
   350→            setTimeout(() => {
   351→              if (fetchId === aiFetchIdRef.current) fetchAiAnalysis(true).catch(() => {});
   352→            }, 5000);
   353→          }
   354→        }
   355→      }
   356→    } finally {
   357→      if (fetchId === aiFetchIdRef.current) {
   358→        setAiLoading(false);
   359→      }
   360→    }
   361→  }, [symbolName, currentPrice, ma21, ma100, rsi, mfi, cci, adx, stochK, stochD,
   362→    macdLine, macdSignal, macdHist, diPlus, diMinus, sar, atr, obv, hasVolume,
   363→    bollingerUpper, bollingerMiddle, bollingerLower, trendDirection, trendAngle, trendR2,
   364→    overallSignal, scenarios, resistances, supports, resistanceStrengths, supportStrengths,
   365→    isIndex, isTgju, currencyUnit, analysisVersion, v11Result, v10Result, cancelInFlightFetch]);
   366→
   367→  // ── Auto-fetch AI analysis only on symbol or version change ──
   368→  // NOT on every data refresh (auto-refresh every 15min would re-trigger
   369→  // a 10-25s LLM call, making the text disappear each time)
   370→  const fetchTriggerKey = `${symbolName}:${analysisVersion}`;
   371→  const lastAutoFetchKeyRef = useRef('');
   372→  // Keep a ref to fetchAiAnalysis so the effect always calls the latest version
   373→  const fetchAiAnalysisRef = useRef(fetchAiAnalysis);
   374→  fetchAiAnalysisRef.current = fetchAiAnalysis;
   375→  useEffect(() => {
   376→    if (fetchTriggerKey === lastAutoFetchKeyRef.current) return;
   377→    if (!symbolName || !currentPrice || currentPrice <= 0) {
   378→      console.log('[VdesAnalysis] Skipping fetch: no symbol or invalid price', { symbolName, currentPrice });
   379→      return;
   380→    }
   381→    lastAutoFetchKeyRef.current = fetchTriggerKey;
   382→    // Reset old text when switching symbols/versions
   383→    setAiAnalysis(null);
   384→    setAiError(null);
   385→    // Small delay to ensure all useMemo values (v11Result etc.) are computed
   386→    const timer = setTimeout(() => {
   387→      fetchAiAnalysisRef.current().catch(() => {});
   388→    }, 150);
   389→    return () => {
   390→      clearTimeout(timer);
   391→      // Also cancel any in-flight fetch when this effect re-runs or unmounts
   392→      cancelInFlightFetch();
   393→    };
   394→  }, [fetchTriggerKey, symbolName, currentPrice, cancelInFlightFetch]);
   395→
   396→  // ── Close export menu on outside click ───────────────────────
   397→  useEffect(() => {
   398→    if (!exportMenuOpen) return;
   399→    const handler = (e: MouseEvent) => {
   400→      if (exportMenuRef.current && !exportMenuRef.current.contains(e.target as Node)) {
   401→        setExportMenuOpen(false);
   402→      }
   403→    };
   404→    document.addEventListener('mousedown', handler);
   405→    return () => document.removeEventListener('mousedown', handler);
   406→  }, [exportMenuOpen]);
   407→
   408→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   409→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   410→  const S1_level = supports[0] ?? currentPrice * 0.95;
   411→  const S2_level = supports[1] ?? currentPrice * 0.90;
   412→  const S3_level = supports[2] ?? currentPrice * 0.85;
   413→
   414→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   415→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   416→
   417→  // ── Dominant scenario (V9) ──────────────────────────────────────
   418→  let highestKey = 'R3';
   419→  let highestProb = 0;
   420→  for (const key of SCENARIO_KEYS) {
   421→    if (scenarios[key].probability > highestProb) {
   422→      highestProb = scenarios[key].probability;
   423→      highestKey = key;
   424→    }
   425→  }

  1056→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
  1057→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
  1058→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
  1059→  grade?: string; overlapCount?: number; touchCount?: number;
  1060→}) {
  1061→  const isSupport = type === 'support';
  1062→  const isTargetType = type === 'target';
  1063→
  1064→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
  1065→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
  1066→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
  1067→
  1068→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
  1069→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
  1070→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
  1071→
  1072→  return (
  1073→    <div
  1074→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
  1075→      style={{
  1076→        background: bgAccent,
  1077→        border: `1px solid ${borderAccent}`,
  1078→        borderRight: `3px solid ${accentColor}`,
  1079→      }}
