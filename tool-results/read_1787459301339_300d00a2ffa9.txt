   228→export default function Home() {
   229→  const [data, setData] = useState<AnalysisData | null>(null);
   230→  const [loading, setLoading] = useState(false);
   231→  const [loadingMessage, setLoadingMessage] = useState<string | null>(null);
   232→  const [error, setError] = useState<string | null>(null);
   233→  const [activePanel, setActivePanel] = useState<SidebarItem>('visual');
   234→  const [refreshing, setRefreshing] = useState(false);
   235→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   236→  const [currentPage, setCurrentPage] = useState<PageId>('analysis');
   237→
   238→  // Store last fetch params for auto-refresh
   239→  const lastFetchRef = useRef<{ symbol: string; category?: string; insCode?: string; tgjuKey?: string; finpySector?: string; finpyIndex?: string; webId?: number } | null>(null);
   240→  const refreshTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   241→
   242→  // Chart container ref for image export
   243→  const chartWrapperRef = useRef<HTMLDivElement>(null);
   244→
   245→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number) => {
   246→    // TGJU instrument: fetch historical data via tgju.org chart API
   247→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   248→      setLoading(true);
   249→      setLoadingMessage('در حال دریافت داده‌های تاریخی ... (حدود ۱۵ ثانیه)');
   250→      setError(null);
   251→      setData(null);
   252→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId };
   253→      try {
   254→        const res = await fetch('/api/tgju-analysis?key=' + encodeURIComponent(tgjuKey));
   255→        if (!res.ok) {
   256→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   257→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   258→          throw new Error(errMsg);
   259→        }
   260→        const json = await res.json();
   261→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   262→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   263→        }
   264→        setData({ ...json, isTgju: true });
   265→        setActivePanel('visual');
   266→      } catch (err) {
   267→        setError(String(err));
   268→      } finally {
   269→        setLoading(false);
   270→        setLoadingMessage(null);
   271→      }
   272→      return;
   273→    }
   274→
   275→    // Index: fetch historical data via TSETMC CDN (webId), then finpy-tse (finpyIndex/finpySector)
   276→    if (category === 'index') {
   277→      setLoading(true);
   278→      setLoadingMessage('در حال دریافت داده‌های تاریخی شاخص ...');
   279→      setError(null);
   280→      setData(null);
   281→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId };
   282→      let gotError = false;
   283→      try {
   284→        // 1. Try webId (TSETMC CDN)
   285→        if (webId) {
   286→          setLoadingMessage('در حال دریافت داده‌های شاخص از TSETMC ...');
   287→          const res = await fetch('/api/finpy-sector?webId=' + encodeURIComponent(String(webId)));
   288→          if (res.ok) {
   289→            try {
   290→              const json = await res.json();
   291→              if (json.candles && json.candles.length > 0 && json.ta) {
   292→                setData(json);
   293→                setActivePanel('visual');
   294→                setLoading(false);
   295→                setLoadingMessage(null);
   296→                return;
   297→              }
   298→              if (json.error) { setError(json.error); gotError = true; }
   299→            } catch (_) { /* invalid JSON */ }
   300→          } else { gotError = true; }
   301→        }
   302→
   303→        // 2. Main indices via finpy-tse
   304→        if (finpyIndex) {
   305→          setLoadingMessage('در حال دریافت داده‌های شاخص اصلی ...');
   306→          const res = await fetch('/api/finpy-sector?indexKey=' + encodeURIComponent(finpyIndex));
   307→          if (res.ok) {
   308→            try {
   309→              const json = await res.json();
   310→              if (json.candles && json.candles.length > 0 && json.ta) {
   311→                setData(json);
   312→                setActivePanel('visual');
   313→                setLoading(false);
   314→                setLoadingMessage(null);
   315→                return;
   316→              }
   317→              if (json.error) { setError(json.error); gotError = true; }
   318→            } catch (_) { /* invalid JSON */ }
   319→          } else { gotError = true; }
   320→        }
   321→
   322→        // 3. Sector indices via finpy-tse
   323→        if (finpySector) {
   324→          setLoadingMessage('در حال دریافت داده‌های شاخص گروه ...');
   325→          const res = await fetch('/api/finpy-sector?sector=' + encodeURIComponent(finpySector));
   326→          if (res.ok) {
   327→            try {
   328→              const json = await res.json();
   329→              if (json.candles && json.candles.length > 0 && json.ta) {
   330→                setData(json);
   331→                setActivePanel('visual');
   332→                setLoading(false);
   333→                setLoadingMessage(null);
   334→                return;
   335→              }
   336→              if (json.error) { setError(json.error); gotError = true; }
   337→            } catch (_) { /* invalid JSON */ }
   338→          } else { gotError = true; }
   339→        }
