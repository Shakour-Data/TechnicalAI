   272→/* Advanced query parser — supports operators like country:US, exchange:NASD, sector:Tech */
   273→interface ParsedQuery {
   274→  country?: string;
   275→  exchange?: string;
   276→  sector?: string;
   277→  terms: string[];
   278→}
   279→
   280→function parseQuery(q: string): ParsedQuery {
   281→  const country: string[] = [];
   282→  const exchange: string[] = [];
   283→  const sector: string[] = [];
   284→  const terms: string[] = [];
   285→
   286→  for (const part of q.split(/\s+/)) {
   287→    if (!part) continue;
   288→    const lower = part.toLowerCase();
   289→    if (lower.startsWith('country:') || lower.startsWith('کشور:')) {
   290→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   291→      if (val) country.push(val.toLowerCase());
   292→    } else if (lower.startsWith('exchange:') || lower.startsWith('صرافی:')) {
   293→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   294→      if (val) exchange.push(val.toLowerCase());
   295→    } else if (lower.startsWith('sector:') || lower.startsWith('صنعت:')) {
   296→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   297→      if (val) sector.push(val.toLowerCase());
   298→    } else {
   299→      terms.push(lower);
   300→    }
   301→  }
   302→
   303→  return {
   304→    country: country.length > 0 ? country.join(' ') : undefined,
   305→    exchange: exchange.length > 0 ? exchange.join(' ') : undefined,
   306→    sector: sector.length > 0 ? sector.join(' ') : undefined,
   307→    terms,
   308→  };
   309→}

   315→const RECENT_KEY = 'fin-search-recent';
   316→const MAX_RECENT = 8;
   317→
   318→function getRecentSearches(): string[] {
   319→  if (typeof window === 'undefined') return [];
   320→  try { const raw = localStorage.getItem(RECENT_KEY); return raw ? JSON.parse(raw) : []; }
   321→  catch { return []; }
   322→}
   323→
   324→function addRecentSearch(name: string) {
   325→  if (typeof window === 'undefined' || !name) return;
   326→  try {
   327→    let list = getRecentSearches();
   328→    list = list.filter((r) => r !== name);
   329→    list.unshift(name);
   330→    if (list.length > MAX_RECENT) list = list.slice(0, MAX_RECENT);
   331→    localStorage.setItem(RECENT_KEY, JSON.stringify(list));
   332→  } catch { /* ignore */ }
   333→}

   532→    const { country: opCountry, exchange: opExchange, sector: opSector, terms } = parsed;
   533→
   534→    const scoreAndFilter = (item: InstrumentItem) => {
   535→      /* Apply query operators */
   536→      if (opCountry) {
   537→        const match = (item.country?.toLowerCase().includes(opCountry)) ||
   538→          (item.countryEn?.toLowerCase().includes(opCountry));
   539→        if (!match) return null;
   540→      }
   541→      if (opExchange) {
   542→        const match = item.exchange?.toLowerCase().includes(opExchange);
   543→        if (!match) return null;
   544→      }
   545→      if (opSector) {
   546→        const match = item.sector?.toLowerCase().includes(opSector) ||
   547→          item.cs?.toLowerCase().includes(opSector);
   548→        if (!match) return null;
   549→      }
   550→
   551→      const fields = [item.l18, item.l30, item.cs, item.yahooSymbol || '', item.nameEn || '', item.exchange || '', item.country || '', item.countryEn || '', item.sector || ''];
   552→      const fullText = fields.join(' ').toLowerCase();
   553→      let score = 0;
   554→
   555→      for (const term of terms) {
   556→        if (!term) continue;
   557→        if (item.l18.toLowerCase().includes(term)) score += 10;
   558→        if (item.nameEn?.toLowerCase().includes(term)) score += 8;
   559→        if (item.yahooSymbol?.toLowerCase() === term) score += 20;
   560→        else if (item.yahooSymbol?.toLowerCase().includes(term)) score += 15;
   561→        if (item.l30.toLowerCase().includes(term)) score += 7;
   562→        if (item.exchange?.toLowerCase().includes(term)) score += 5;
   563→        if (item.country?.toLowerCase().includes(term)) score += 6;
   564→        if (item.countryEn?.toLowerCase() === term) score += 15;
   565→        else if (item.countryEn?.toLowerCase().includes(term)) score += 4;
   566→        if (item.sector?.toLowerCase().includes(term)) score += 5;
   567→        if (item.cs?.toLowerCase().includes(term)) score += 6;
   568→
   569→        if (!item.l18.toLowerCase().includes(term) && !item.l30.toLowerCase().includes(term) &&
   570→            !item.cs?.toLowerCase().includes(term) && !(item.yahooSymbol?.toLowerCase().includes(term)) &&
   571→            !(item.nameEn?.toLowerCase().includes(term)) && !(item.exchange?.toLowerCase().includes(term)) &&
   572→            !(item.country?.toLowerCase().includes(term)) && !(item.countryEn?.toLowerCase().includes(term)) &&
   573→            !(item.sector?.toLowerCase().includes(term))) {
   574→          return null;
   575→        }
   576→      }
   577→      /* If only operators (no terms), accept all filtered items */
   578→      if (terms.length === 0) score = 1;
   579→      return score;
   580→    };
   581→
   582→    const scored = items.map((item) => ({ item, score: scoreAndFilter(item) }))
   583→      .filter((r): r is { item: InstrumentItem; score: number } => r.score !== null && r.score > 0)
   584→      .sort((a, b) => b.score - a.score);

   586→    setTotalMatched(scored.length);
   587→    setResults(scored.slice(0, MAX_RESULTS).map((s) => s.item));

   589→  }, []);
