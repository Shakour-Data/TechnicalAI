    17→function safeNum(v: number, fallback: number = 0): number {
    18→  return (typeof v === 'number' && isFinite(v)) ? v : fallback;
    19→}
    20→
    21→function clamp(v: number, min: number, max: number): number {
    22→  return Math.max(min, Math.min(max, safeNum(v, (min + max) / 2)));
    23→}
    24→
    25→function normalize(arr: number[]): number[] {
    26→  const safe = arr.map(v => safeNum(v, 0));
    27→  const sum = safe.reduce((a, b) => a + b, 0);
    28→  if (sum <= 0) return arr.map(() => 1 / arr.length);
    29→  return safe.map((v) => v / sum);
    30→}
    31→
    32→function sigmoid(x: number): number {
    33→  const sx = safeNum(x, 0);
    34→  if (Math.abs(sx) > 500) return sx > 0 ? 1 : 0; // prevent Infinity
    35→  return 1 / (1 + Math.exp(-sx));
    36→}

    38→/**
    39→ * IRON LAW: Round proportional floats to integers that sum to EXACTLY 100,
    40→ * with each value constrained to [minVal, maxVal].
    41→ *
    42→ * Algorithm: Largest Remainder Method (LRM) + iterative bound enforcement.
    43→ * This function GUARANTEES sum(result) === 100 and minVal <= result[i] <= maxVal
    44→ * for all i, as long as n*minVal <= 100 <= n*maxVal.
    45→ *
    46→ * ⚠️  NEVER bypass this function when producing integer scenario percentages.
    47→ */
    48→function enforceSumTo100(
    49→  rawFloats: number[],
    50→  minVal: number,
    51→  maxVal: number
    52→): number[] {
    53→  const n = rawFloats.length;
    54→
    55→  // Normalize floats to sum exactly to 100 first
    56→  const floatSum = rawFloats.reduce((a, b) => a + b, 0);
    57→  const normalized =
    58→    floatSum > 0
    59→      ? rawFloats.map((v) => (v / floatSum) * 100)
    60→      : rawFloats.map(() => 100 / n);
    61→
    62→  // Step 1: LRM — floor all, distribute deficit by largest fractional part
    63→  const result = normalized.map((v) => Math.max(Math.floor(v), 0));
    64→  let deficit = 100 - result.reduce((a, b) => a + b, 0);
    65→
    66→  const byFrac = Array.from({ length: n }, (_, i) => ({
    67→    idx: i,
    68→    frac: normalized[i] - Math.floor(normalized[i]),
    69→  })).sort((a, b) => b.frac - a.frac);
    70→
    71→  // Give +1 to top `deficit` entries (LRM)
    72→  for (let d = 0; d < deficit && d < n; d++) {
    73→    result[byFrac[d].idx]++;
    74→  }
    75→  // Now result sums to exactly 100
    76→
    77→  // Step 2: Iteratively enforce [minVal, maxVal] bounds while preserving sum = 100
    78→  // Each iteration: move 1 from a violator to a suitable partner.
    79→  // Converges because each move strictly reduces the total violation magnitude.
    80→  let changed = true;
    81→  let safety = 0;
    82→  while (changed && safety < 100) {
    83→    changed = false;
    84→    safety++;
    85→
    86→    // Enforce minimum: raise any value below minVal by taking from the largest above minVal
    87→    for (let i = 0; i < n; i++) {
    88→      while (result[i] < minVal) {
    89→        let donor = -1;
    90→        for (let j = 0; j < n; j++) {
    91→          if (j !== i && result[j] > minVal && (donor === -1 || result[j] > result[donor])) {
    92→            donor = j;
    93→          }
    94→        }
    95→        if (donor === -1) break; // no donor available — shouldn't happen if n*minVal <= 100
    96→        result[donor]--;
    97→        result[i]++;
    98→        changed = true;
    99→      }
   100→    }
   101→
   102→    // Enforce maximum: lower any value above maxVal by giving to the smallest below maxVal
   103→    for (let i = 0; i < n; i++) {
   104→      while (result[i] > maxVal) {
   105→        let receiver = -1;
   106→        for (let j = 0; j < n; j++) {
   107→          if (j !== i && result[j] < maxVal && (receiver === -1 || result[j] < result[receiver])) {
   108→            receiver = j;
   109→          }
   110→        }
   111→        if (receiver === -1) break; // no receiver — shouldn't happen if 100 <= n*maxVal
   112→        result[i]--;
   113→        result[receiver]++;
   114→        changed = true;
   115→      }
   116→    }
   117→  }
   118→
   119→  // Step 3: Final assertion — sum MUST be 100 (safety net, should never trigger)
   120→  const finalSum = result.reduce((a, b) => a + b, 0);
   121→  if (finalSum !== 100 && n > 0) {
   122→    // Adjust the entry farthest from its bound
   123→    const diff = 100 - finalSum;
   124→    let bestIdx = 0;
   125→    let bestRoom = -1;
   126→    for (let i = 0; i < n; i++) {
   127→      const room = diff > 0 ? maxVal - result[i] : result[i] - minVal;
   128→      if (room > bestRoom) {
   129→        bestRoom = room;
   130→        bestIdx = i;
   131→      }
   132→    }
   133→    result[bestIdx] += diff;
   134→  }
   135→
   136→  return result;
   137→}

  1123→// === Probability Context Builder ==============================================

  1125→function buildContext(input: GraphInput): ProbContext {
  1126→  const adxNorm = clamp(input.adx / 60, 0, 1);
  1127→  const srProximity = clamp(
  1128→    1 - Math.min(input.distToR1, input.distToS1) * 10,
  1129→    0,
  1130→    1
  1131→  );
  1132→  const oscillatorExtreme = clamp(
  1133→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
  1134→    0,
  1135→    1
  1136→  );
  1137→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1138→
  1139→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1140→  const overboughtRisk =
  1141→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1142→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1143→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1144→
  1145→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1146→  const oversoldBounce =
  1147→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1148→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1149→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1150→
  1151→  return {
  1152→    input,
  1153→    adxNorm,
  1154→    srProximity,
  1155→    oscillatorExtreme,
  1156→    divergenceProxy,
  1157→    overboughtRisk,
  1158→    oversoldBounce,
  1159→  };
  1160→}

  1162→// === Node Probability Computation =============================================
  1163→// For each non-terminal node, returns an array of probabilities for its
  1164→// outgoing edges. The order MUST match the order of edges in createEdges().
  1165→// =============================================================================
  1166→
  1167→function computeNodeProbabilities(
  1168→  nodeId: string,
  1169→  ctx: ProbContext,
  1170→  _numChildren: number
  1171→): number[] {
  1172→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1173→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1174→
  1175→  switch (nodeId) {
  1176→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1177→    case 'ROOT': {
  1178→      const trendW = adxNorm * 0.5 + input.mlTrend * 0.18 + 0.1;
  1179→      const breakW = srProximity * 0.3 + (1 - adxNorm) * 0.08 + 0.08;
  1180→      const revW = ctx.divergenceProxy * 0.35 + (1 - adxNorm) * 0.12 + 0.06;
  1181→      return normalize([trendW, breakW, revW]);
  1182→    }
  1183→
  1184→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1185→    case 'N_TREND': {
  1186→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1187→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1188→      const flatUncertainty = (1 - Math.abs(input.bullConsensus - 0.5) * 2);
  1189→      const flat = flatUncertainty * clamp(1 - adxNorm * 1.5, 0, 1) * 0.45;
  1190→      return normalize([bull, bear, flat]);

  1192→
  1193→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1194→    case 'N_T_BULL': {
  1195→      const momHigh =
  1196→        clamp(
  1197→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1198→            (input.macdHist > 0
  1199→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1200→              : 0),
  1201→          0,
  1202→          1
  1203→        ) * input.mlMomentum;
  1204→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1205→      return normalize([momHigh, momMod, overboughtRisk]);
  1206→    }
  1207→
  1208→    // ── N_T_B_MOM_HIGH → [SC8, SC9, SC7] ────────────────────────────────────
  1209→    // Strong bullish momentum — all three outcomes are bullish (no range)
  1210→    case 'N_T_B_MOM_HIGH': {

  1212→      const pSC9 = 0.15 + input.strengthIndex * 0.2;
  1213→      const pSC7 = 0.35 * input.confidenceIndex + 0.15;
  1214→      return normalize([pSC8, pSC9, pSC7]);
  1215→    }
  1216→
  1217→    // ── N_T_B_MOM_MOD → [SC7, SC6, SC5] ─────────────────────────────────────
  1218→    // Moderate bullish — small explicit SC5 based on uncertainty
  1219→    case 'N_T_B_MOM_MOD': {

  1221→      const pSC6 = 0.55 * (1 - input.confidenceIndex) + 0.15;

  1441→    // Mirror of N_R_BULL

  1444→      const candleProb =
  1445→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1446→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1447→      return normalize([divProb, candleProb, srProb]);
  1448→    }
  1449→
  1450→    // ── N_R_BE_DIV → [SC4, SC2, SC5] ───────────────────────────────────────
  1451→    // Bearish divergence — explicit small SC5
  1452→    case 'N_R_BE_DIV': {
  1453→      const pSC4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1454→      const pSC2 = 0.2 + input.mlMomentum * 0.15;
  1455→      const pSC5 = clamp((1 - adxNorm) * 0.1, 0.02, 0.1);
  1456→      return normalize([pSC4, pSC2, pSC5]);
  1457→    }
  1458→
  1459→    // ── N_R_BE_CANDLE → [SC3, SC4, SC5] ────────────────────────────────────
  1460→    // Bearish candle pattern — explicit small SC5
  1461→    case 'N_R_BE_CANDLE': {
  1462→      const pSC3 = 0.5 * input.confidenceIndex + 0.2;
  1463→      const pSC4 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1464→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.1, 0.02, 0.1);
  1465→      return normalize([pSC3, pSC4, pSC5]);
  1466→    }
  1467→
  1468→    // ── N_R_BE_SR → [SC4, SC5, SC6] ─────────────────────────────────────────
  1469→    // Resistance rejection — explicit small SC5
  1470→    // Edge order: SC4, SC5, SC6 — probabilities computed then reordered
  1471→    case 'N_R_BE_SR': {
  1472→      const pSC4 = 0.5 + input.srAvgStrength * 0.2;
  1473→      const pSC6 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1474→      const pSC5 = clamp((1 - input.srAvgStrength) * 0.1, 0.02, 0.1);
  1475→      // Return in edge order: [SC4, SC5, SC6]
  1476→      return normalize([pSC4, pSC5, pSC6]);
  1477→    }
  1478→
  1479→    // ── N_R_NONE → [SC4, SC5, SC6] ──────────────────────────────────────────
  1480→    case 'N_R_NONE': {
  1481→      const pSC5 = 0.4 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1482→      const pSC6 = 0.2 * input.bullConsensus + 0.1;
  1483→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.1;
  1484→      return normalize([pSC4, pSC5, pSC6]);
  1485→    }
  1486→
  1487→    // ── Terminal nodes (SC1–SC9) have no outgoing edges ─────────────────────
  1488→    default:
  1489→      return [];
  1490→  }
  1491→}

  1493→// === Graph Traversal ==========================================================
  1494→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1495→// =============================================================================
  1496→
  1497→function traverseGraph(
  1498→  nodeId: string,
  1499→  pathProb: number,
  1500→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1501→  ctx: ProbContext,
  1502→  adj: Map<string, number[]>,
  1503→  edges: GraphEdge[],
  1504→  edgeProbabilities: Record<number, number>,
  1505→  nodeAccum: Record<string, number>,
  1506→  scenarios: Record<string, number>,
  1507→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1508→): void {
  1509→  // Accumulate path probability at this node
  1510→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1511→
  1512→  const outEdgeIndices = adj.get(nodeId);
  1513→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1514→
  1515→  // Compute outgoing edge probabilities for this node
  1516→  const probs = computeNodeProbabilities(
  1517→    nodeId,
  1518→    ctx,
  1519→    outEdgeIndices.length
  1520→  );
  1521→
  1522→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1523→    const edgeIdx = outEdgeIndices[i];
  1524→    const edge = edges[edgeIdx];
  1525→    const prob = probs[i] || 0;
  1526→
  1527→    // Store edge probability
  1528→    edgeProbabilities[edgeIdx] = prob;
  1529→
  1530→    // Determine branch for child nodes
  1531→    let childBranch = currentBranch;
  1532→    if (nodeId === 'ROOT') {
  1533→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1534→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1535→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1536→    }
  1537→
  1538→    const childPathProb = pathProb * prob;
  1539→
  1540→    // Check if target is a terminal node (SC1–SC9)
  1541→    const isTerminal =
  1542→      edge.to.length === 3 &&
  1543→      edge.to[0] === 'S' &&
  1544→      edge.to[1] === 'C' &&
  1545→      edge.to[2] >= '1' &&
  1546→      edge.to[2] <= '9';
  1547→
  1548→    if (isTerminal) {
  1549→      // Accumulate scenario probability
  1550→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1551→      // Accumulate branch contribution
  1552→      if (childBranch) {
  1553→        contributions[edge.to][childBranch] += childPathProb;
  1554→      }
  1555→    }
  1556→
  1557→    // Recurse into child
  1558→    traverseGraph(
  1559→      edge.to,
  1560→      childPathProb,
  1561→      childBranch,
  1562→      ctx,
  1563→      adj,
  1564→      edges,
  1565→      edgeProbabilities,
  1566→      nodeAccum,
  1567→      scenarios,
  1568→      contributions
  1569→    );
  1570→  }
  1571→}

  1575→export function buildDecisionGraph(input: GraphInput): GraphData {
  1576→  // 0. Sanitize all numeric inputs — replace NaN/Infinity/undefined with safe defaults
  1577→  const price = safeNum(input.price, 1000);
  1578→  const bullConsensus = clamp(safeNum(input.bullConsensus, 0.5), 0, 1);
  1579→  const rsi = clamp(safeNum(input.rsi, 50), 0, 100);

  1582→  const stochK = clamp(safeNum(input.stochK, 50), 0, 100);
  1583→  const stochD = clamp(safeNum(input.stochD, 50), 0, 100);
  1584→  const adx = clamp(safeNum(input.adx, 20), 0, 100);
  1585→  const diPlus = clamp(safeNum(input.diPlus, 20), 0, 100);
  1586→  const diMinus = clamp(safeNum(input.diMinus, 20), 0, 100);
  1587→  const macdHist = safeNum(input.macdHist, 0);
  1588→  const atr = safeNum(input.atr, price * 0.02);
  1589→  const bbUpper = safeNum(input.bbUpper, price * 1.03);
  1590→  const bbMiddle = safeNum(input.bbMiddle, price);
  1591→  const bbLower = safeNum(input.bbLower, price * 0.97);
  1592→  const sar = safeNum(input.sar, price);
  1593→  const ichimokuTenkan = safeNum(input.ichimokuTenkan, price);
  1594→  const ichimokuKijun = safeNum(input.ichimokuKijun, price);
  1595→  const ichimokuSenkouA = safeNum(input.ichimokuSenkouA, price);
  1596→  const ichimokuSenkouB = safeNum(input.ichimokuSenkouB, price);
  1597→  const maAlignment = clamp(safeNum(input.maAlignment, 0.5), 0, 1);
  1598→  const momentum = safeNum(input.momentum, 0);
  1599→  const awesomeOsc = safeNum(input.awesomeOsc, 0);
  1600→  const fisherTransform = clamp(safeNum(input.fisherTransform, 0), -5, 5);
  1601→  const confidenceIndex = clamp(safeNum(input.confidenceIndex, 0.5), 0, 1);
  1602→  const strengthIndex = clamp(safeNum(input.strengthIndex, 0.5), 0, 1);
  1603→  const hasVolume = input.hasVolume === true;
  1604→  const distToR1 = safeNum(input.distToR1, 0.03);
  1605→  const distToS1 = safeNum(input.distToS1, 0.03);
  1606→  const srAvgStrength = clamp(safeNum(input.srAvgStrength, 0.3), 0, 1);
  1607→  const mlMomentum = clamp(safeNum(input.mlMomentum, 0.7), 0, 1);
  1608→  const mlVolatility = clamp(safeNum(input.mlVolatility, 0.5), 0, 1);
  1609→  const mlTrend = clamp(safeNum(input.mlTrend, 0.6), 0, 1);
  1610→
  1611→  // Create sanitized input object
  1612→  const sanitizedInput: GraphInput = {
  1613→    price, bullConsensus, rsi, mfi, cci, stochK, stochD, adx, diPlus, diMinus,
  1614→    macdHist, atr, bbUpper, bbMiddle, bbLower, sar,
  1615→    ichimokuTenkan, ichimokuKijun, ichimokuSenkouA, ichimokuSenkouB,
  1616→    maAlignment, momentum, awesomeOsc, fisherTransform,
  1617→    confidenceIndex, strengthIndex, hasVolume,
  1618→    distToR1, distToS1, srAvgStrength,
  1619→    mlMomentum, mlVolatility, mlTrend,
  1620→  };
  1621→
  1622→  // 1. Build static graph structure
  1623→  const nodes = createNodes();
  1624→  const edges = createEdges();
  1625→  const nodePositions = createPositions();
  1626→
  1627→  // 2. Build adjacency map: nodeId → array of edge indices
  1628→  const adj = new Map<string, number[]>();
  1629→  for (let i = 0; i < edges.length; i++) {
  1630→    const from = edges[i].from;
  1631→    if (!adj.has(from)) adj.set(from, []);
  1632→    adj.get(from)!.push(i);
  1633→  }
  1634→
  1635→  // 3. Build probability context (use sanitized input)
  1636→  const ctx = buildContext(sanitizedInput);

  1639→  const edgeProbabilities: Record<number, number> = {};

  1642→  const contributions: Record<
  1643→    string,
  1644→    { trend: number; breakout: number; reversal: number }
  1645→  > = {};
  1646→
  1647→  for (let i = 1; i <= 9; i++) {
  1648→    const key = `SC${i}`;

  1656→    1.0,
  1657→    null,

  1662→    nodeAccum,
  1663→    scenarios,
  1664→    contributions
  1665→  );
  1666→
  1667→  // 6. Post-process scenario probabilities — IRON LAW: integers summing to EXACTLY 100
  1668→  // Convert raw traversal fractions to percentages, then use enforceSumTo100 for
  1669→  // guaranteed sum=100 with each value in [2, 35].
  1670→  const rawFloats: number[] = [];
  1671→  for (let i = 1; i <= 9; i++) {
  1672→    const key = `SC${i}`;
  1673→    rawFloats.push(safeNum(scenarios[key], 0) * 100);
  1674→  }
  1675→
  1676→  const roundedInts = enforceSumTo100(rawFloats, 2, 35);
  1677→
  1678→  const finalPcts: Record<string, number> = {};
  1679→  for (let i = 1; i <= 9; i++) {
  1680→    const key = `SC${i}`;
  1681→    finalPcts[key] = roundedInts[i - 1];
  1682→  }
  1683→
  1684→  // 7. Compute branch probabilities (from ROOT outgoing edges)
  1685→  const rootEdgeIndices = adj.get('ROOT') || [];
  1686→  const rootProbs = computeNodeProbabilities('ROOT', ctx, rootEdgeIndices.length);
  1687→  const branchProbabilities = {
  1688→    trend: rootProbs[0] || 0,
  1689→    breakout: rootProbs[1] || 0,
  1690→    reversal: rootProbs[2] || 0,
  1691→  };
  1692→
  1693→  // 8. Build node display values (path probability as percentage)
  1694→  const nodeValues: Record<string, string> = {};
  1695→  for (const [nodeId, accum] of Object.entries(nodeAccum)) {
  1696→    nodeValues[nodeId] = `${Math.round(accum * 100)}%`;
  1697→  }
  1698→
  1699→  // Override terminal node values with final post-processed percentages
  1700→  for (const [key, pct] of Object.entries(finalPcts)) {
  1701→    nodeValues[key] = `${pct}%`;
  1702→  }
  1703→
  1704→  // 9. Build path contributions (as percentages)
  1705→  const pathContributions: Record<
  1706→    string,
  1707→    { trend: number; breakout: number; reversal: number }
  1708→  > = {};
  1709→  for (let i = 1; i <= 9; i++) {
  1710→    const key = `SC${i}`;
  1711→    const c = contributions[key];
  1712→    const total = c.trend + c.breakout + c.reversal;
  1713→    if (total > 0) {
  1714→      pathContributions[key] = {
  1715→        trend: Math.round((c.trend / total) * 100),
  1716→        breakout: Math.round((c.breakout / total) * 100),
  1717→        reversal: Math.round((c.reversal / total) * 100),
  1718→      };
  1719→    } else {
  1720→      pathContributions[key] = { trend: 33, breakout: 34, reversal: 33 };
  1721→    }
  1722→  }
  1723→
  1724→  // 10. Compute 30-day probability trend
  1725→  const probFractions: Record<string, number> = {};
  1726→  for (const [k, v] of Object.entries(finalPcts)) {
  1727→    probFractions[k] = v / 100;
  1728→  }
  1729→  const probabilityTrend = calcTrendFromProbs(probFractions, 30);
  1730→
  1731→  // 11. Return complete GraphData
  1732→  return {
  1733→    nodes,
  1734→    edges,
  1735→    nodePositions,
  1736→    edgeProbabilities,
  1737→    nodeValues,
  1738→    branchProbabilities,
  1739→    scenarioProbabilities: finalPcts,
  1740→    pathContributions,
  1741→    probabilityTrend,
  1742→  };
  1743→}
