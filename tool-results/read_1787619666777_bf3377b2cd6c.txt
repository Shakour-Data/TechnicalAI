    10→    10→// === Helper Functions =========================================================
    11→    11→
    12→    12→function clamp(v: number, min: number, max: number): number {
    13→    13→  return Math.max(min, Math.min(max, v));
    14→    14→}
    15→    15→
    16→    16→function normalize(arr: number[]): number[] {
    17→    17→  const sum = arr.reduce((a, b) => a + b, 0);
    18→    18→  if (sum === 0) return arr.map(() => 1 / arr.length);
    19→    19→  return arr.map((v) => v / sum);
    20→    20→}
    21→    21→
    22→    22→function sigmoid(x: number): number {
    23→    23→  return 1 / (1 + Math.exp(-x));
    24→    24→}

   988→   988→// === Probability Context Builder ==============================================
   989→   989→
   990→   990→function buildContext(input: GraphInput): ProbContext {
   991→   991→  const adxNorm = clamp(input.adx / 60, 0, 1);
   992→   992→  const srProximity = clamp(
   993→   993→    1 - Math.min(input.distToR1, input.distToS1) * 10,
   994→   994→    0,
   995→   995→    1
   996→   996→  );
   997→   997→  const oscillatorExtreme = clamp(
   998→   998→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
   999→   999→    0,
  1000→  1000→    1
  1001→  1001→  );
  1002→  1002→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1003→  1003→
  1004→  1004→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1005→  1005→  const overboughtRisk =
  1006→  1006→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1007→  1007→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1008→  1008→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1009→  1009→
  1010→  1010→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1011→  1011→  const oversoldBounce =
  1012→  1012→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1013→  1013→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1014→  1014→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1015→  1015→
  1016→  1016→  return {
  1017→  1017→    input,
  1018→  1018→    adxNorm,
  1019→  1019→    srProximity,
  1020→  1020→    oscillatorExtreme,
  1021→  1021→    divergenceProxy,
  1022→  1022→    overboughtRisk,
  1023→  1023→    oversoldBounce,
  1024→  1024→  };
  1025→  1025→}

  1027→  1027→// === Node Probability Computation =============================================
  1028→  1028→// For each non-terminal node, returns an array of probabilities for its
  1029→  1029→// outgoing edges. The order MUST match the order of edges in createEdges().
  1030→  1030→// =============================================================================
  1031→  1031→
  1032→  1032→function computeNodeProbabilities(
  1033→  1033→  nodeId: string,
  1034→  1034→  ctx: ProbContext,
  1035→  1035→  _numChildren: number
  1036→  1036→): number[] {
  1037→  1037→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1038→  1038→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1039→  1039→
  1040→  1040→  switch (nodeId) {
  1041→  1041→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1042→  1042→    case 'ROOT': {
  1043→  1043→      const trendW = adxNorm * 0.45 + input.mlTrend * 0.15 + 0.12;
  1044→  1044→      const breakW = srProximity * 0.35 + (1 - adxNorm) * 0.1 + 0.1;
  1045→  1045→      const revW = ctx.divergenceProxy * 0.3 + (1 - adxNorm) * 0.15 + 0.08;
  1046→  1046→      return normalize([trendW, breakW, revW]);
  1047→  1047→    }
  1048→  1048→
  1049→  1049→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1050→  1050→    case 'N_TREND': {
  1051→  1051→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1052→  1052→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1053→  1053→      const flat =
  1054→  1054→        (1 - Math.abs(input.bullConsensus - 0.5) * 2) * (1 - adxNorm * 0.7);
  1055→  1055→      return normalize([bull, bear, flat]);
  1056→  1056→    }
  1057→  1057→
  1058→  1058→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1059→  1059→    case 'N_T_BULL': {
  1060→  1060→      const momHigh =
  1061→  1061→        clamp(
  1062→  1062→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1063→  1063→            (input.macdHist > 0
  1064→  1064→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1065→  1065→              : 0),
  1066→  1066→          0,
  1067→  1067→          1
  1068→  1068→        ) * input.mlMomentum;
  1069→  1069→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1070→  1070→      return normalize([momHigh, momMod, overboughtRisk]);
  1071→  1071→    }
  1072→  1072→
  1073→  1073→    // ── N_T_B_MOM_HIGH → [R3, R4, R2] ────────────────────────────────────
  1074→  1074→    // Strong bullish momentum — all three outcomes are bullish (no range)
  1075→  1075→    case 'N_T_B_MOM_HIGH': {

  1077→  1077→      const pR4 = 0.15 + input.strengthIndex * 0.2;
  1078→  1078→      const pR2 = 0.35 * input.confidenceIndex + 0.15;

  1080→  1080→    }
  1081→  1081→
  1082→  1082→    // ── N_T_B_MOM_MOD → [R2, R1, R5] ─────────────────────────────────────
  1083→  1083→    // Moderate bullish — small explicit R5 based on uncertainty
  1084→  1084→    case 'N_T_B_MOM_MOD': {
  1085→  1085→      const pR2 = 0.5 * input.confidenceIndex + 0.2;
  1086→  1086→      const pR1 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1087→  1087→      const pR5 = clamp((1 - adxNorm) * 0.35, 0.05, 0.25);

  1104→  1104→          (input.momentum < 0 ? Math.abs(input.momentum) / 50 : 0) +
  1105→  1105→            (input.macdHist < 0
  1106→  1106→              ? (Math.abs(input.macdHist) / (atrSafe * 0.5)) * 0.3
  1107→  1107→              : 0),
  1108→  1108→          0,
  1109→  1109→          1
  1110→  1110→        ) * input.mlMomentum;
  1111→  1111→      const momMod = (1 - momLow) * 0.6 * input.mlMomentum;
  1112→  1112→      return normalize([momLow, momMod, oversoldBounce]);

  1128→  1128→      const pR6 = 0.5 * (1 - input.confidenceIndex) + 0.15;

  1236→  1236→    }
  1237→  1237→
  1238→  1238→    // ── N_B_D_VOL_W → [R6, R1, R5] ──────────────────────────────────────
  1239→  1239→    // Weak volume bearish breakout — fake breakout risk, capped R5
  1240→  1240→    case 'N_B_D_VOL_W': {
  1241→  1241→      const pR6 = 0.4 + (1 - input.bullConsensus) * 0.15;
  1242→  1242→      const pR1 = 0.3 + input.bullConsensus * 0.15;
  1243→  1243→      const pR5 = clamp((1 - input.confidenceIndex) * 0.25, 0.05, 0.2);
  1244→  1244→      return normalize([pR6, pR1, pR5]);
  1245→  1245→    }
  1246→  1246→
  1247→  1247→    // ── N_B_NONE → [R5] ──────────────────────────────────────────────────
  1248→  1248→    case 'N_B_NONE': {
  1249→  1249→      return [1.0];
  1250→  1250→    }
  1251→  1251→
  1252→  1252→    // ── N_REVERSAL → [N_R_BULL, N_R_BEAR, N_R_NONE] ──────────────────────
  1253→  1253→    case 'N_REVERSAL': {
  1254→  1254→      const bullRev = overboughtRisk;
  1255→  1255→      const bearRev = oversoldBounce;
  1256→  1256→      const noneRev =
  1257→  1257→        1 - Math.max(overboughtRisk, oversoldBounce) * 0.7;
  1258→  1258→      return normalize([bullRev, bearRev, noneRev]);
  1259→  1259→    }
  1260→  1260→
  1261→  1261→    // ── N_R_BULL → [N_R_B_DIV, N_R_B_CANDLE, N_R_B_SR] ──────────────────
  1262→  1262→    case 'N_R_BULL': {
  1263→  1263→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1264→  1264→      const candleProb =
  1265→  1265→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1266→  1266→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1267→  1267→      return normalize([divProb, candleProb, srProb]);
  1268→  1268→    }
  1269→  1269→
  1270→  1270→    // ── N_R_B_DIV → [R1, R3, R5] ────────────────────────────────────────
  1271→  1271→    // Bullish divergence — explicit small R5
  1272→  1272→    case 'N_R_B_DIV': {
  1273→  1273→      const pR1 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1274→  1274→      const pR3 = 0.2 + input.mlMomentum * 0.15;
  1275→  1275→      const pR5 = clamp((1 - adxNorm) * 0.25, 0.05, 0.2);
  1276→  1276→      return normalize([pR1, pR3, pR5]);
  1277→  1277→    }
  1278→  1278→
  1279→  1279→    // ── N_R_B_CANDLE → [R2, R1, R5] ─────────────────────────────────────
  1280→  1280→    // Bullish candle pattern — explicit small R5
  1281→  1281→    case 'N_R_B_CANDLE': {
  1282→  1282→      const pR2 = 0.5 * input.confidenceIndex + 0.2;
  1283→  1283→      const pR1 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1284→  1284→      const pR5 = clamp((1 - input.confidenceIndex) * 0.2, 0.05, 0.18);
  1285→  1285→      return normalize([pR2, pR1, pR5]);
  1286→  1286→    }
  1287→  1287→
  1288→  1288→    // ── N_R_B_SR → [R1, R5, R6] ─────────────────────────────────────────
  1289→  1289→    // Support bounce — explicit small R5
  1290→  1290→    // Edge order: R1, R5, R6 — probabilities computed then reordered
  1291→  1291→    case 'N_R_B_SR': {
  1292→  1292→      const pR1 = 0.5 + input.srAvgStrength * 0.2;
  1293→  1293→      const pR6 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1294→  1294→      const pR5 = clamp((1 - input.srAvgStrength) * 0.2, 0.05, 0.18);
  1295→  1295→      // Return in edge order: [R1, R5, R6]
  1296→  1296→      return normalize([pR1, pR5, pR6]);
  1297→  1297→    }
  1298→  1298→
  1299→  1299→    // ── N_R_BEAR → [N_R_BE_DIV, N_R_BE_CANDLE, N_R_BE_SR] ───────────────
  1300→  1300→    // Mirror of N_R_BULL
  1301→  1301→    case 'N_R_BEAR': {
  1302→  1302→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1303→  1303→      const candleProb =
  1304→  1304→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1305→  1305→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1306→  1306→      return normalize([divProb, candleProb, srProb]);
  1307→  1307→    }
  1308→  1308→
  1309→  1309→    // ── N_R_BE_DIV → [R6, R8, R5] ───────────────────────────────────────
  1310→  1310→    // Bearish divergence — explicit small R5
  1311→  1311→    case 'N_R_BE_DIV': {
  1312→  1312→      const pR6 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1313→  1313→      const pR8 = 0.2 + input.mlMomentum * 0.15;
  1314→  1314→      const pR5 = clamp((1 - adxNorm) * 0.25, 0.05, 0.2);
  1315→  1315→      return normalize([pR6, pR8, pR5]);
  1316→  1316→    }
  1317→  1317→
  1318→  1318→    // ── N_R_BE_CANDLE → [R7, R6, R5] ────────────────────────────────────
  1319→  1319→    // Bearish candle pattern — explicit small R5
  1320→  1320→    case 'N_R_BE_CANDLE': {
  1321→  1321→      const pR7 = 0.5 * input.confidenceIndex + 0.2;
  1322→  1322→      const pR6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1323→  1323→      const pR5 = clamp((1 - input.confidenceIndex) * 0.2, 0.05, 0.18);
  1324→  1324→      return normalize([pR7, pR6, pR5]);
  1325→  1325→    }
  1326→  1326→
  1327→  1327→    // ── N_R_BE_SR → [R6, R5, R1] ─────────────────────────────────────────
  1328→  1328→    // Resistance rejection — explicit small R5
  1329→  1329→    // Edge order: R6, R5, R1 — probabilities computed then reordered
  1330→  1330→    case 'N_R_BE_SR': {
  1331→  1331→      const pR6 = 0.5 + input.srAvgStrength * 0.2;
  1332→  1332→      const pR1 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1333→  1333→      const pR5 = clamp((1 - input.srAvgStrength) * 0.2, 0.05, 0.18);
  1334→  1334→      // Return in edge order: [R6, R5, R1]
  1335→  1335→      return normalize([pR6, pR5, pR1]);
  1336→  1336→    }
  1337→  1337→
  1338→  1338→    // ── N_R_NONE → [R5] ──────────────────────────────────────────────────
  1339→  1339→    case 'N_R_NONE': {
  1340→  1340→      return [1.0];
  1341→  1341→    }
  1342→  1342→
  1343→  1343→    // ── Terminal nodes (R1–R9) have no outgoing edges ─────────────────────
  1344→  1344→    default:
  1345→  1345→      return [];
  1346→  1346→  }

  1349→  1349→// === Graph Traversal ==========================================================
  1350→  1350→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1351→  1351→// =============================================================================
  1352→  1352→
  1353→  1353→function traverseGraph(
  1354→  1354→  nodeId: string,
  1355→  1355→  pathProb: number,
  1356→  1356→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1357→  1357→  ctx: ProbContext,
  1358→  1358→  adj: Map<string, number[]>,
  1359→  1359→  edges: GraphEdge[],
  1360→  1360→  edgeProbabilities: Record<number, number>,
  1361→  1361→  nodeAccum: Record<string, number>,
  1362→  1362→  scenarios: Record<string, number>,
  1363→  1363→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1364→  1364→): void {
  1365→  1365→  // Accumulate path probability at this node
  1366→  1366→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1367→  1367→
  1368→  1368→  const outEdgeIndices = adj.get(nodeId);
  1369→  1369→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1370→  1370→
  1371→  1371→  // Compute outgoing edge probabilities for this node
  1372→  1372→  const probs = computeNodeProbabilities(
  1373→  1373→    nodeId,
  1374→  1374→    ctx,
  1375→  1375→    outEdgeIndices.length
  1376→  1376→  );
  1377→  1377→
  1378→  1378→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1379→  1379→    const edgeIdx = outEdgeIndices[i];
  1380→  1380→    const edge = edges[edgeIdx];
  1381→  1381→    const prob = probs[i] || 0;
  1382→  1382→
  1383→  1383→    // Store edge probability
  1384→  1384→    edgeProbabilities[edgeIdx] = prob;
  1385→  1385→
  1386→  1386→    // Determine branch for child nodes
  1387→  1387→    let childBranch = currentBranch;
  1388→  1388→    if (nodeId === 'ROOT') {
  1389→  1389→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1390→  1390→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1391→  1391→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1392→  1392→    }
  1393→  1393→
  1394→  1394→    const childPathProb = pathProb * prob;
  1395→  1395→
  1396→  1396→    // Check if target is a terminal node
  1397→  1397→    const isTerminal =
  1398→  1398→      edge.to.length === 2 &&
  1399→  1399→      edge.to[0] === 'R' &&
  1400→  1400→      edge.to[1] >= '1' &&
  1401→  1401→      edge.to[1] <= '9';
  1402→  1402→
  1403→  1403→    if (isTerminal) {
  1404→  1404→      // Accumulate scenario probability
  1405→  1405→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1406→  1406→      // Accumulate branch contribution
  1407→  1407→      if (childBranch) {
  1408→  1408→        contributions[edge.to][childBranch] += childPathProb;
  1409→  1409→      }
  1410→  1410→    }
  1411→  1411→
  1412→  1412→    // Recurse into child
  1413→  1413→    traverseGraph(
  1414→  1414→      edge.to,
  1415→  1415→      childPathProb,
  1416→  1416→      childBranch,
  1417→  1417→      ctx,
  1418→  1418→      adj,
  1419→  1419→      edges,
  1420→  1420→      edgeProbabilities,
  1421→  1421→      nodeAccum,
  1422→  1422→      scenarios,
  1423→  1423→      contributions
  1424→  1424→    );
  1425→  1425→  }
  1426→  1426→}

  1430→  1430→export function buildDecisionGraph(input: GraphInput): GraphData {
  1431→  1431→  // 1. Build static graph structure
  1432→  1432→  const nodes = createNodes();
  1433→  1433→  const edges = createEdges();
  1434→  1434→  const nodePositions = createPositions();
  1435→  1435→
  1436→  1436→  // 2. Build adjacency map: nodeId → array of edge indices
  1437→  1437→  const adj = new Map<string, number[]>();
  1438→  1438→  for (let i = 0; i < edges.length; i++) {
  1439→  1439→    const from = edges[i].from;
  1440→  1440→    if (!adj.has(from)) adj.set(from, []);
  1441→  1441→    adj.get(from)!.push(i);
  1442→  1442→  }
  1443→  1443→
  1444→  1444→  // 3. Build probability context
  1445→  1445→  const ctx = buildContext(input);
  1446→  1446→
  1447→  1447→  // 4. Initialize accumulation structures
  1448→  1448→  const edgeProbabilities: Record<number, number> = {};
  1449→  1449→  const nodeAccum: Record<string, number> = {};
  1450→  1450→  const scenarios: Record<string, number> = {};
  1451→  1451→  const contributions: Record<
  1452→  1452→    string,
  1453→  1453→    { trend: number; breakout: number; reversal: number }
  1454→  1454→  > = {};
  1455→  1455→
  1456→  1456→  for (let i = 1; i <= 9; i++) {
  1457→  1457→    const key = `R${i}`;
  1458→  1458→    scenarios[key] = 0;
  1459→  1459→    contributions[key] = { trend: 0, breakout: 0, reversal: 0 };
  1460→  1460→  }
  1461→  1461→
  1462→  1462→  // 5. Traverse graph from ROOT
  1463→  1463→  traverseGraph(
  1464→  1464→    'ROOT',
  1465→  1465→    1.0,
  1466→  1466→    null,
  1467→  1467→    ctx,
  1468→  1468→    adj,
  1469→  1469→    edges,
  1470→  1470→    edgeProbabilities,
  1471→  1471→    nodeAccum,
  1472→  1472→    scenarios,
  1473→  1473→    contributions
  1474→  1474→  );
  1475→  1475→
  1476→  1476→  // 6. Post-process scenario probabilities
  1477→  1477→  // Convert to percentage, clamp [2, 35], re-normalize to sum=100
  1478→  1478→  const rawPcts: Record<string, number> = {};
  1479→  1479→  for (let i = 1; i <= 9; i++) {
  1480→  1480→    const key = `R${i}`;
  1481→  1481→    rawPcts[key] = Math.round((scenarios[key] || 0) * 100);
  1482→  1482→  }
  1483→  1483→
  1484→  1484→  // Clamp each to [2, 35]
  1485→  1485→  const clamped = Object.entries(rawPcts).map(([k, v]) => ({
  1486→  1486→    key: k,
  1487→  1487→    value: clamp(v, 2, 35),
  1488→  1488→  }));
  1489→  1489→
  1490→  1490→  // Re-normalize to sum=100
  1491→  1491→  const clampedSum = clamped.reduce((s, c) => s + c.value, 0);
  1492→  1492→  const finalPcts: Record<string, number> = {};
  1493→  1493→  if (clampedSum > 0) {
  1494→  1494→    // Distribute proportionally and round, then fix rounding errors
  1495→  1495→    let runningTotal = 0;
  1496→  1496→    const entries = clamped
  1497→  1497→      .map((c) => {
  1498→  1498→        const raw = (c.value / clampedSum) * 100;
  1499→  1499→        return { key: c.key, raw };
  1500→  1500→      })
  1501→  1501→      .sort((a, b) => b.raw - a.raw);
  1502→  1502→
  1503→  1503→    for (let i = 0; i < entries.length; i++) {
  1504→  1504→      const isLast = i === entries.length - 1;
  1505→  1505→      if (isLast) {
  1506→  1506→        finalPcts[entries[i].key] = Math.round(100 - runningTotal);
  1507→  1507→      } else {
  1508→  1508→        const rounded = Math.round(entries[i].raw);
  1509→  1509→        finalPcts[entries[i].key] = rounded;
  1510→  1510→        runningTotal += rounded;
  1511→  1511→      }
  1512→  1512→    }
  1513→  1513→  }
  1514→  1514→
  1515→  1515→  // Ensure no negative values due to rounding
  1516→  1516→  for (const key of Object.keys(finalPcts)) {
  1517→  1517→    if (finalPcts[key] < 0) finalPcts[key] = 0;
  1518→  1518→  }
  1519→  1519→
  1520→  1520→  // 7. Compute branch probabilities (from ROOT outgoing edges)
  1521→  1521→  const rootEdgeIndices = adj.get('ROOT') || [];
  1522→  1522→  const rootProbs = computeNodeProbabilities('ROOT', ctx, rootEdgeIndices.length);
  1523→  1523→  const branchProbabilities = {
  1524→  1524→    trend: rootProbs[0] || 0,
  1525→  1525→    breakout: rootProbs[1] || 0,
  1526→  1526→    reversal: rootProbs[2] || 0,
  1527→  1527→  };
  1528→  1528→
  1529→  1529→  // 8. Build node display values (path probability as percentage)
  1530→  1530→  const nodeValues: Record<string, string> = {};
  1531→  1531→  for (const [nodeId, accum] of Object.entries(nodeAccum)) {
  1532→  1532→    nodeValues[nodeId] = `${Math.round(accum * 100)}%`;
  1533→  1533→  }
  1534→  1534→
  1535→  1535→  // Override terminal node values with final post-processed percentages
  1536→  1536→  for (const [key, pct] of Object.entries(finalPcts)) {
  1537→  1537→    nodeValues[key] = `${pct}%`;
  1538→  1538→  }
  1539→  1539→
  1540→  1540→  // 9. Build path contributions (as percentages)
  1541→  1541→  const pathContributions: Record<
  1542→  1542→    string,
  1543→  1543→    { trend: number; breakout: number; reversal: number }
  1544→  1544→  > = {};
  1545→  1545→  for (let i = 1; i <= 9; i++) {
  1546→  1546→    const key = `R${i}`;
  1547→  1547→    const c = contributions[key];
  1548→  1548→    const total = c.trend + c.breakout + c.reversal;
  1549→  1549→    if (total > 0) {
  1550→  1550→      pathContributions[key] = {
  1551→  1551→        trend: Math.round((c.trend / total) * 100),
  1552→  1552→        breakout: Math.round((c.breakout / total) * 100),
  1553→  1553→        reversal: Math.round((c.reversal / total) * 100),
  1554→  1554→      };
  1555→  1555→    } else {
  1556→  1556→      pathContributions[key] = { trend: 33, breakout: 34, reversal: 33 };
  1557→  1557→    }
  1558→  1558→  }
  1559→  1559→
  1560→  1560→  // 10. Return complete GraphData
  1561→  1561→  return {
  1562→  1562→    nodes,
  1563→  1563→    edges,
  1564→  1564→    nodePositions,
  1565→  1565→    edgeProbabilities,
  1566→  1566→    nodeValues,
  1567→  1567→    branchProbabilities,
  1568→  1568→    scenarioProbabilities: finalPcts,
  1569→  1569→    pathContributions,
  1570→  1570→  };
  1571→  1571→}

  1573→  1573→// === 30-Day Probability Trend Calculation ======================================
  1574→  1574→
  1575→  1575→// Dual exponential decay for forecast weighting
  1576→  1576→// W_forecast_h = α * exp(-h/τ₁) + (1-α) * exp(-h/τ₂)
  1577→  1577→export function dualDecay(h: number, alpha = 0.7, tauShort = 5, tauLong = 20): number {
  1578→  1578→  return alpha * Math.exp(-h / tauShort) + (1 - alpha) * Math.exp(-h / tauLong);
  1579→  1579→}
  1580→  1580→
  1581→  1581→// Normalize decay factors so they sum to 1
  1582→  1582→export function normalizedDecayFactors(horizon: number): number[] {
  1583→  1583→  const factors: number[] = [];
  1584→  1584→  for (let h = 1; h <= horizon; h++) {
  1585→  1585→    factors.push(dualDecay(h));
  1586→  1586→  }
  1587→  1587→  const total = factors.reduce((a, b) => a + b, 0);
  1588→  1588→  return factors.map(f => f / total);
  1589→  1589→}

  1621→  1621→
  1622→  1622→export function calculateProbabilityTrend(
  1623→  1623→  scenarioProbabilities: Array<{key: string; name: string; nameEn: string; probability: number; isBullish?: boolean; isBearish?: boolean; isNeutral?: boolean}>,
  1624→  1624→  horizon: number = 30
  1625→  1625→): ProbabilityTrendResult {
  1626→  1626→  const decay = normalizedDecayFactors(horizon);
  1627→  1627→  
  1628→  1628→  const scenarios: ScenarioTrend[] = scenarioProbabilities.map(sp => {
  1629→  1629→    const trend: DayProbability[] = [];
  1630→  1630→    let cumProb = 0;
  1631→  1631→    let peakDay = 1;
  1632→  1632→    let peakProb = 0;
  1633→  1633→    
  1634→  1634→    for (let h = 1; h <= horizon; h++) {
