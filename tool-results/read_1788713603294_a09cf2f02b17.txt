   110→// ─── Mini Sparkline SVG ───────────────────────────────────────────────────────
   111→
   112→function MiniSparkline({ predictions, color, width = 120, height = 40 }: { predictions: PredictionPoint[]; color: string; width?: number; height?: number }) {
   113→  if (predictions.length < 2) return null;
   114→
   115→  const prices = predictions.map(p => p.price);
   116→  const min = Math.min(...prices, ...predictions.map(p => p.lower));
   117→  const max = Math.max(...prices, ...predictions.map(p => p.upper));
   118→  const range = max - min || 1;
   119→  const pad = 4;
   120→
   121→  const toX = (i: number) => pad + (i / (predictions.length - 1)) * (width - 2 * pad);
   122→  const toY = (v: number) => pad + (1 - (v - min) / range) * (height - 2 * pad);
   123→
   124→  // Main line
   125→  const linePoints = predictions.map((p, i) => `${toX(i)},${toY(p.price)}`).join(' ');
   126→  // Upper band
   127→  const upperPoints = predictions.map((p, i) => `${toX(i)},${toY(p.upper)}`).join(' ');
   128→  // Lower band
   129→  const lowerPoints = predictions.map((p, i) => `${toX(i)},${toY(p.lower)}`).reverse().join(' ');
   130→  // Fill area
   131→  const fillPath = `M${upperPoints} L${lowerPoints} Z`;
   132→
   133→  return (
   134→    <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-10" style={{ minWidth: width }}>
   135→      <defs>
   136→        <linearGradient id={`fill-${color.replace('#', '')}`} x1="0" y1="0" x2="0" y2="1">
   137→          <stop offset="0%" stopColor={color} stopOpacity="0.15" />
   138→          <stop offset="100%" stopColor={color} stopOpacity="0.02" />
   139→        </linearGradient>
   140→      </defs>
   141→      <path d={fillPath} fill={`url(#fill-${color.replace('#', '')})`} />
   142→      <polyline points={upperPoints} fill="none" stroke={color} strokeOpacity="0.3" strokeWidth="0.5" strokeDasharray="2,2" />
   143→      <polyline points={predictions.map((p, i) => `${toX(i)},${toY(p.lower)}`).join(' ')} fill="none" stroke={color} strokeOpacity="0.3" strokeWidth="0.5" strokeDasharray="2,2" />
   144→      <polyline points={linePoints} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   145→      {predictions.map((p, i) => (
   146→        <circle key={i} cx={toX(i)} cy={toY(p.price)} r="1.5" fill={color} />
   147→      ))}
   148→    </svg>
   149→  );
   150→}

   154→export default function MLForecast({ symbolName, candles, currentPrice, priceDecimals }: MLForecastProps) {

   161→  const [expandedModel, setExpandedModel] = useState<string | null>(null);
   162→  const [showTable, setShowTable] = useState(false);
   163→
   164→  // Derive decimals from prop (default 0 for TSE integers)
   165→  const decimals = priceDecimals ?? 0;
   166→  const toFa = (n: number) => toFaDyn(n, decimals);
   167→  const toFaDecimal = (n: number) => toFaDecimalDyn(n, decimals);
   168→
   169→  // Check ML service health on mount (auto-starts service if down)
   170→  useEffect(() => {
   171→    fetch('/api/ml-predict')
   172→      .then(r => r.json())
   173→      .then(data => setServiceAvailable(data.status === 'ok'))
   174→      .catch(() => setServiceAvailable(false));
   175→  }, []);
   176→
   177→  const runPrediction = useCallback(async () => {
   178→    if (candles.length < 60) {
   179→      setError('حداقل ۶۰ کندل داده برای پیش‌بینی ML نیاز است. داده‌های فعلی کافی نیست.');
   180→      return;
   181→    }
   182→    setLoading(true);
   183→    setError(null);
   184→    setResult(null);
   185→    try {
   186→      const resp = await fetch('/api/ml-predict', {
   187→        method: 'POST',
   188→        headers: { 'Content-Type': 'application/json' },
   189→        body: JSON.stringify({ candles, sessions, models: selectedModels }),
   190→      });
   191→      const json = await resp.json();
   192→      if (json.error) {
   193→        setError(json.error);
   194→      } else {
   195→        setResult(json);
   196→      }
   197→    } catch (err) {
   198→      const msg = err instanceof Error ? err.message : 'خطای ناشناخته';
   199→      // Auto-retry if service was starting
   200→      if (msg.includes('راه‌اندازی') || msg.includes('دسترس نیست')) {
   201→        setError('سرویس ML در حال راه‌اندازی است. لطفاً پس از ۱۰ ثانیه دوباره تلاش کنید.');
   202→        // Re-check health after 10s
   203→        setTimeout(() => {
   204→          fetch('/api/ml-predict')
   205→            .then(r => r.json())
   206→            .then(data => setServiceAvailable(data.status === 'ok'))
   207→            .catch(() => setServiceAvailable(false));
   208→        }, 10000);
   209→      } else {
   210→        setError(msg);
   211→      }
   212→    } finally {
   213→      setLoading(false);
   214→    }
   215→  }, [candles, sessions, selectedModels]);
   216→
   217→  // Feature importance sorted
   218→  const topFeatures = useMemo(() => {
   219→    if (!result?.feature_importance) return [];
   220→    return Object.entries(result.feature_importance)
   221→      .sort(([, a], [, b]) => b - a)
   222→      .slice(0, 10);
   223→  }, [result]);
   224→
   225→  // Ensemble predictions for chart
   226→  const ensemblePredictions = result?.ensemble?.predictions ?? [];

   228→  // Determine if ensemble is overall bullish/bearish
   229→  const ensembleTrend = useMemo(() => {
   230→    if (ensemblePredictions.length < 1) return 'neutral';
   231→    const lastPred = ensemblePredictions[ensemblePredictions.length - 1];
   232→    if (lastPred.change_pct > 1) return 'bullish';
   233→    if (lastPred.change_pct < -1) return 'bearish';
   234→    return 'neutral';
   235→  }, [ensemblePredictions]);
   236→
   237→  const trendIcon = ensembleTrend === 'bullish' ? TrendingUp : ensembleTrend === 'bearish' ? TrendingDown : BarChart3;
   238→  const TrendIcon = trendIcon;
   239→  const trendLabel = ensembleTrend === 'bullish' ? 'صعودی' : ensembleTrend === 'bearish' ? 'نزولی' : 'خنثی';
   240→  const trendColor = ensembleTrend === 'bullish' ? 'text-emerald-700' : ensembleTrend === 'bearish' ? 'text-red-700' : 'text-amber-700';
   241→
   242→  return (
   243→    <div className="space-y-4">
   244→      {/* ═══ HEADER ═══ */}
   245→      <div className="rounded-2xl px-6 py-5 border border-violet-200 bg-gradient-to-l from-violet-50 to-white shadow-sm">
   246→        <div className="flex items-center justify-between flex-wrap gap-3">
   247→          <div className="flex items-center gap-3">
   248→            <div className="w-10 h-10 rounded-xl bg-violet-100 border border-violet-200 flex items-center justify-center">
   249→              <BrainCircuit className="w-5 h-5 text-violet-700" />
   250→            </div>
   251→            <div>
   252→              <h2 className="text-base font-bold text-gray-900">پیش‌بینی قیمت با هوش مصنوعی</h2>
   253→              <p className="text-[10px] text-gray-500">۵ مدل ML × ۳۴ ویژگی تکنیکال — scikit-learn, XGBoost, LightGBM</p>
   254→            </div>
   255→          </div>
   256→          <div className="flex items-center gap-2">
   257→            {serviceAvailable === false && (
   258→              <span className="px-3 py-1 rounded-full bg-amber-50 border border-amber-200 text-[10px] text-amber-600 flex items-center gap-1">
   259→                <RefreshCw className="w-3 h-3 animate-spin" />
   260→                در حال راه‌اندازی...
   261→              </span>
   262→            )}
   263→            {serviceAvailable === true && (
   264→              <span className="px-3 py-1 rounded-full bg-emerald-50 border border-emerald-200 text-[10px] text-emerald-600 flex items-center gap-1">
   265→                <CheckCircle2 className="w-3 h-3" />
   266→                آنلاین
   267→              </span>
   268→            )}
   269→          </div>
   270→        </div>
   271→      </div>
   272→
   273→      {/* ═══ CONTROLS ═══ */}
   274→      <div className="rounded-2xl p-5 border border-[#e5e7eb] bg-[#ffffff] shadow-sm">
   275→        <div className="flex flex-wrap items-end gap-4">
   276→          {/* Sessions selector */}
   277→          <div className="flex flex-col gap-1.5">
   278→            <label className="text-[10px] text-gray-500 font-medium">تعداد جلسات پیش‌بینی</label>
   279→            <div className="flex items-center gap-1 bg-[#f3f4f6] rounded-lg p-0.5 border border-[#e5e7eb]">
   280→              {SESSION_OPTIONS.map(s => (
   281→                <button
   282→                  key={s}
   283→                  onClick={() => setSessions(s)}
   284→                  className={`px-3 py-1.5 rounded-md text-xs font-medium transition-all cursor-pointer ${
   285→                    sessions === s
   286→                      ? 'bg-white text-violet-700 shadow-sm border border-[#e5e7eb]'

   290→                  {toPersianDigits(String(s))}

   294→          </div>
   295→
   296→          {/* Model toggles */}
   297→          <div className="flex flex-col gap-1.5">

   299→            <div className="flex items-center gap-1.5 flex-wrap">
   300→              {Object.entries(MODEL_LABELS).map(([key, label]) => (

   302→                  key={key}
   303→                  onClick={() => setSelectedModels(prev =>
   304→                    prev.includes(key) ? prev.filter(k => k !== key) : [...prev, key]
   305→                  )}
   306→                  className={`px-2.5 py-1 rounded-lg text-[10px] font-medium border transition-all cursor-pointer ${
   307→                    selectedModels.includes(key)
   308→                      ? 'border-current/30'

   311→                  style={{
   312→                    color: selectedModels.includes(key) ? MODEL_COLORS[key] : undefined,
   313→                    background: selectedModels.includes(key) ? MODEL_COLORS[key] + '10' : undefined,
   314→                  }}
   315→                >
   316→                  {label}
   317→                </button>
   318→              ))}
   319→            </div>
   320→          </div>
   321→
   322→          {/* Run button */}
   323→          <button
   324→            onClick={runPrediction}
   325→            disabled={loading || selectedModels.length === 0 || candles.length < 60}
   326→            className="mr-auto inline-flex items-center gap-2 px-5 py-2 rounded-lg bg-violet-600 text-white text-xs font-bold hover:bg-violet-700 transition-colors disabled:opacity-50 disabled:cursor-not-allowed cursor-pointer"
   327→          >
   328→            <Zap className={`w-3.5 h-3.5 ${loading ? 'animate-pulse' : ''}`} />
   329→            {loading ? 'در حال تحلیل...' : 'شروع پیش‌بینی'}
   330→          </button>
   331→        </div>
   332→
   333→        {candles.length < 60 && (
   334→          <p className="mt-3 text-[10px] text-amber-600 flex items-center gap-1">
   335→            <AlertTriangle className="w-3 h-3" />
   336→            تعداد کندل‌های فعلی ({toPersianDigits(String(candles.length))}) کمتر از حداقل مورد نیاز (۶۰) است.
   337→          </p>
   338→        )}
   339→      </div>
   340→
   341→      {/* ═══ LOADING ═══ */}
   342→      {loading && (
   343→        <div className="rounded-2xl p-8 border border-violet-200 bg-violet-50/30">
   344→          <div className="flex flex-col items-center justify-center gap-4">
   345→            <div className="relative">
   346→              <div className="w-12 h-12 rounded-full border-4 border-violet-200 border-t-violet-600 animate-spin" />

   353→              </p>
   354→            </div>
   355→          </div>
   356→        </div>
   357→      )}
   358→
   359→      {/* ═══ ERROR ═══ */}
   360→      {error && !loading && (
   361→        <div className="rounded-2xl p-5 border border-red-200 bg-red-50">
   362→          <div className="flex items-start gap-3">
   363→            <AlertTriangle className="w-5 h-5 text-red-500 mt-0.5 shrink-0" />
