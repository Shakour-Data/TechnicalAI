  1006→      // Don't finalize yet, keep collecting points
  1007→      setCurrentDrawing(completed);
  1008→      return;
  1009→    }
  1010→    if (completed.points.length >= minPoints) {
  1011→      setDrawings((prev) => [...prev, completed]);
  1012→    }
  1013→
  1014→    setCurrentDrawing(null);
  1015→    setIsDrawing(false);
  1016→  }, [isDrawing, currentDrawing]);
  1017→
  1018→  /* - Text input submit ------------------- */
  1019→  const handleTextSubmit = useCallback(() => {
  1020→    if (!textInput || !textValue.trim() || !currentDrawing) {
  1021→      setTextInput(null);
  1022→      setCurrentDrawing(null);
  1023→      setIsDrawing(false);
  1024→      return;
  1025→    }
  1026→
  1027→    const completed: Drawing = {
  1028→      ...currentDrawing,
  1029→      text: textValue.trim(),
  1030→      completed: true,
  1031→    };
  1032→    setDrawings((prev) => [...prev, completed]);
  1033→    setTextInput(null);
  1034→    setTextValue('');
  1035→    setCurrentDrawing(null);
  1036→    setIsDrawing(false);
  1037→  }, [textInput, textValue, currentDrawing]);
  1038→
  1039→  const handleTextCancel = useCallback(() => {
  1040→    setTextInput(null);
  1041→    setTextValue('');
  1042→    setCurrentDrawing(null);
  1043→    setIsDrawing(false);
  1044→  }, []);
  1045→
  1046→  /* - Delete / Clear -------------------- */
  1047→  const handleDeleteLast = useCallback(() => {
  1048→    setDrawings((prev) => prev.slice(0, -1));
  1049→  }, []);
  1050→
  1051→  const handleClearAll = useCallback(() => {
  1052→    setDrawings([]);
  1053→  }, []);
  1054→
  1055→  /* - Render ------------------------- */
  1056→  const fmt = (n: number) => toPersianDigits(n.toLocaleString('en', { maximumFractionDigits: 0 }));
  1057→
  1058→  return (
  1059→    <div className="w-full">
  1060→      {/* - CROSSHAIR LEGEND (Shamsi date + OHLC) ---------- */}
  1061→      {hoverInfo && (
  1062→        <div className="flex items-center gap-3 px-3 py-1.5 border border-[#e5e7eb] rounded-t-lg bg-[#ffffff] text-[11px]" dir="rtl" style={{ fontFamily: 'Vazirmatn, sans-serif' }}>
  1063→          <span className="text-gray-500 font-medium">{hoverInfo.date}</span>
  1064→          <div className="w-px h-3.5 bg-gray-200" />
  1065→          <span className="text-gray-500">باز: <span className={chgColor(hoverInfo.o, hoverInfo.c) === BULL ? 'text-emerald-700 font-bold' : 'text-red-700 font-bold'}>{fmt(hoverInfo.o)}</span></span>
  1066→          <span className="text-gray-500">بالا: <span className="text-emerald-700 font-bold">{fmt(hoverInfo.h)}</span></span>
  1067→          <span className="text-gray-500">پایین: <span className="text-red-700 font-bold">{fmt(hoverInfo.l)}</span></span>
  1068→          <span className="text-gray-500">بسته: <span className={chgColor(hoverInfo.o, hoverInfo.c) === BULL ? 'text-emerald-700 font-bold' : 'text-red-700 font-bold'}>{fmt(hoverInfo.c)}</span></span>
  1069→          {hoverInfo.v > 0 && (
  1070→            <span className="text-gray-500">حجم: <span className="text-gray-900 font-bold">{toPersianDigits((hoverInfo.v / 1e6).toFixed(1))}M</span></span>
  1071→          )}
  1072→          <span className={`font-bold ${hoverInfo.chg >= 0 ? 'text-emerald-700' : 'text-red-700'}`}>
  1073→            {hoverInfo.chg >= 0 ? '▲' : '▼'} {toPersianDigits(Math.abs(hoverInfo.chg).toFixed(2))}٪
  1074→          </span>
  1075→        </div>
  1076→      )}
  1077→      {/* - TOOLBAR --------------------- */}
  1078→      <div
  1079→        className={`flex items-center gap-1 px-2 py-1.5 overflow-x-auto flex-nowrap border ${hoverInfo ? 'border-t-0 rounded-b-lg' : 'rounded-t-lg border-b-0'}`}
  1080→        style={{ backgroundColor: TOOLBAR_BG, borderColor: TOOLBAR_BORDER }}
  1081→      >
  1082→        {TOOLS.map((tool) => (
  1083→          <button
  1084→            key={tool.id}
  1085→            onClick={() => {
  1086→              setActiveTool(tool.id);
  1087→              if (isDrawing) {
  1088→                setIsDrawing(false);
  1089→                setCurrentDrawing(null);
  1090→                setTextInput(null);
  1091→              }
  1092→            }}
  1093→            title={tool.label}
  1094→            className={`shrink-0 flex items-center justify-center w-8 h-8 rounded-md transition-all cursor-pointer ${
  1095→              activeTool === tool.id
  1096→                ? 'bg-amber-500/20 text-amber-800 ring-1 ring-amber-500/30'
  1097→                : 'text-gray-500 hover:bg-gray-100 hover:text-gray-700'
  1098→            }`}
  1099→          >
  1100→            {tool.icon}
  1101→          </button>
  1102→        ))}
  1103→
  1104→        <div className="w-px h-6 bg-gray-200 mx-1 shrink-0" />
  1105→
  1106→        {PALETTE.map((c) => (
  1107→          <button
  1108→            key={c}
  1109→            onClick={() => setActiveColor(c)}
  1110→            title={c}
  1111→            className={`shrink-0 w-6 h-6 rounded-full border-2 transition-all cursor-pointer ${
  1112→              activeColor === c ? 'border-gray-900 scale-110' : 'border-transparent hover:scale-105'
  1113→            }`}
  1114→            style={{ backgroundColor: c }}
  1115→          />
  1116→        ))}
  1117→
  1118→        <div className="w-px h-6 bg-gray-200 mx-1 shrink-0" />
  1119→
  1120→        <button
  1121→          onClick={handleDeleteLast}
  1122→          title="حذف آخر"
  1123→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-500 hover:bg-red-50 hover:text-red-700 transition-all cursor-pointer"
  1124→        >
  1125→          <Trash2 className="w-3.5 h-3.5" />
  1126→        </button>
  1127→
  1128→        <button
  1129→          onClick={handleClearAll}
  1130→          title="پاک کردن همه"
  1131→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-500 hover:bg-red-50 hover:text-red-700 transition-all cursor-pointer"
  1132→        >
  1133→          <X className="w-3.5 h-3.5" />
  1134→        </button>
  1135→
  1136→        <div className="mr-auto shrink-0 text-xs text-gray-500 font-medium" dir="rtl">
  1137→          {TOOLS.find((t) => t.id === activeTool)?.label}
  1138→        </div>
  1139→      </div>
  1140→
  1141→      <div
  1142→        className="relative rounded-b-lg overflow-hidden border border-gray-200"
  1143→        style={{ height }}
  1144→      >
  1145→        <div
  1146→          ref={chartContainerRef}
  1147→          className="w-full h-full"
  1148→          data-chart
  1149→        />
  1150→
  1151→        <svg
  1152→          ref={svgRef}
  1153→          className="absolute top-0 left-0"
  1154→          style={{
  1155→            width: chartSize.w || '100%',
  1156→            height: chartSize.h || '100%',
  1157→            pointerEvents: activeTool !== 'cursor' ? 'auto' : 'none',
  1158→            zIndex: 10,
  1159→          }}
  1160→          onMouseDown={handleMouseDown}
  1161→          onMouseMove={handleMouseMove}
  1162→          onMouseUp={handleMouseUp}
  1163→          onMouseLeave={handleMouseUp}
  1164→        />
  1165→
  1166→        {textInput && (
  1167→          <div
  1168→            className="absolute z-20"
  1169→            style={{
  1170→              left: textInput.x,
  1171→              top: textInput.y - 8,
  1172→            }}
  1173→          >
  1174→            <input
  1175→              type="text"
  1176→              value={textValue}
  1177→              onChange={(e) => setTextValue(e.target.value)}
  1178→              onKeyDown={(e) => {
  1179→                if (e.key === 'Enter') handleTextSubmit();
  1180→                if (e.key === 'Escape') handleTextCancel();
  1181→              }}
  1182→              placeholder="متن..."
  1183→              autoFocus
  1184→              dir="rtl"
  1185→              className="bg-white border border-gray-200 rounded px-2 py-1 text-sm text-gray-900 outline-none focus:border-amber-500 w-32"
  1186→              onBlur={handleTextSubmit}
  1187→            />
  1188→          </div>
  1189→        )}
  1190→      </div>
  1191→    </div>
  1192→  );
  1193→}
