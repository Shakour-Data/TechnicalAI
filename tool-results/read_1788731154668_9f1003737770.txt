   274→/* Advanced query parser — supports operators like country:US, exchange:NASD, sector:Tech */
   275→interface ParsedQuery {
   276→  country?: string;
   277→  exchange?: string;
   278→  sector?: string;
   279→  terms: string[];
   280→}
   281→
   282→function parseQuery(q: string): ParsedQuery {
   283→  const country: string[] = [];
   284→  const exchange: string[] = [];
   285→  const sector: string[] = [];
   286→  const terms: string[] = [];
   287→
   288→  for (const part of q.split(/\s+/)) {
   289→    if (!part) continue;
   290→    const lower = part.toLowerCase();
   291→    if (lower.startsWith('country:') || lower.startsWith('کشور:')) {
   292→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   293→      if (val) country.push(val.toLowerCase());
   294→    } else if (lower.startsWith('exchange:') || lower.startsWith('صرافی:')) {
   295→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   296→      if (val) exchange.push(val.toLowerCase());
   297→    } else if (lower.startsWith('sector:') || lower.startsWith('صنعت:')) {
   298→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   299→      if (val) sector.push(val.toLowerCase());
   300→    } else {
   301→      terms.push(lower);
   302→    }
   303→  }
   304→
   305→  return {
   306→    country: country.length > 0 ? country.join(' ') : undefined,
   307→    exchange: exchange.length > 0 ? exchange.join(' ') : undefined,
   308→    sector: sector.length > 0 ? sector.join(' ') : undefined,
   309→    terms,
   310→  };
   311→}

   317→const RECENT_KEY = 'fin-search-recent';
   318→const MAX_RECENT = 8;
   319→
   320→function getRecentSearches(): string[] {
   321→  if (typeof window === 'undefined') return [];
   322→  try { const raw = localStorage.getItem(RECENT_KEY); return raw ? JSON.parse(raw) : []; }
   323→  catch { return []; }
   324→}
   325→
   326→function addRecentSearch(name: string) {
   327→  if (typeof window === 'undefined' || !name) return;
   328→  try {
   329→    let list = getRecentSearches();
   330→    list = list.filter((r) => r !== name);
   331→    list.unshift(name);
   332→    if (list.length > MAX_RECENT) list = list.slice(0, MAX_RECENT);
   333→    localStorage.setItem(RECENT_KEY, JSON.stringify(list));
   334→  } catch { /* ignore */ }
   335→}

   552→  const doFilter = React.useCallback((items: InstrumentItem[], q: string) => {
   553→    const parsed = parseQuery(q);
   554→    const { country: opCountry, exchange: opExchange, sector: opSector, terms } = parsed;
   555→
   556→    const scoreAndFilter = (item: InstrumentItem) => {
   557→      /* Apply query operators */
   558→      if (opCountry) {
   559→        const match = (item.country?.toLowerCase().includes(opCountry)) ||
   560→          (item.countryEn?.toLowerCase().includes(opCountry));
   561→        if (!match) return null;
   562→      }
   563→      if (opExchange) {
   564→        const match = item.exchange?.toLowerCase().includes(opExchange);
   565→        if (!match) return null;
   566→      }
   567→      if (opSector) {
   568→        const match = item.sector?.toLowerCase().includes(opSector) ||
   569→          item.cs?.toLowerCase().includes(opSector);
   570→        if (!match) return null;
   571→      }
   572→
   573→      const fields = [item.l18, item.l30, item.cs, item.yahooSymbol || '', item.nameEn || '', item.exchange || '', item.country || '', item.countryEn || '', item.sector || ''];
   574→      const fullText = fields.join(' ').toLowerCase();
   575→      let score = 0;
   576→
   577→      for (const term of terms) {
   578→        if (!term) continue;
   579→        if (item.l18.toLowerCase().includes(term)) score += 10;
   580→        if (item.nameEn?.toLowerCase().includes(term)) score += 8;
   581→        if (item.yahooSymbol?.toLowerCase() === term) score += 20;
   582→        else if (item.yahooSymbol?.toLowerCase().includes(term)) score += 15;
   583→        if (item.l30.toLowerCase().includes(term)) score += 7;
   584→        if (item.exchange?.toLowerCase().includes(term)) score += 5;
   585→        if (item.country?.toLowerCase().includes(term)) score += 6;
   586→        if (item.countryEn?.toLowerCase() === term) score += 15;
   587→        else if (item.countryEn?.toLowerCase().includes(term)) score += 4;
   588→        if (item.sector?.toLowerCase().includes(term)) score += 5;
   589→        if (item.cs?.toLowerCase().includes(term)) score += 6;
   590→
   591→        if (!item.l18.toLowerCase().includes(term) && !item.l30.toLowerCase().includes(term) &&
   592→            !item.cs?.toLowerCase().includes(term) && !(item.yahooSymbol?.toLowerCase().includes(term)) &&
   593→            !(item.nameEn?.toLowerCase().includes(term)) && !(item.exchange?.toLowerCase().includes(term)) &&
   594→            !(item.country?.toLowerCase().includes(term)) && !(item.countryEn?.toLowerCase().includes(term)) &&
   595→            !(item.sector?.toLowerCase().includes(term))) {
   596→          return null;
   597→        }
   598→      }
   599→      /* If only operators (no terms), accept all filtered items */
   600→      if (terms.length === 0) score = 1;
   601→      return score;
   602→    };
   603→
   604→    const scored = items.map((item) => ({ item, score: scoreAndFilter(item) }))
   605→      .filter((r): r is { item: InstrumentItem; score: number } => r.score !== null && r.score > 0)
   606→      .sort((a, b) => b.score - a.score);
   607→
   608→    setTotalMatched(scored.length);
   609→    setResults(scored.slice(0, MAX_RESULTS).map((s) => s.item));

   611→  }, []);
