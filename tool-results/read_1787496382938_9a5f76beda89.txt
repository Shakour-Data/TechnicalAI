   595→  return { slope, intercept, r2 };
   596→}

   598→// ─── Helper: Round to nice number (per spec price-range rules) ─────────────
   599→function roundToNice(price: number): number {
   600→  const abs = Math.abs(price);
   601→  let step: number;
   602→  if (abs >= 100000) step = 100;
   603→  else if (abs >= 10000) step = 10;
   604→  else if (abs >= 1000) step = 5;
   605→  else if (abs >= 100) step = 1;
   606→  else step = 0.5;
   607→  return Math.round(price / step) * step;
   608→}

   610→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
   611→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: { price: number; idx: number }[]; lows: { price: number; idx: number }[] } {
   612→  const highs: { price: number; idx: number }[] = [];
   613→  const lows: { price: number; idx: number }[] = [];
   614→  for (let i = lookback; i < data.length - lookback; i++) {
   615→    let isHigh = true;
   616→    let isLow = true;
   617→    for (let j = i - lookback; j <= i + lookback; j++) {
   618→      if (j === i) continue;
   619→      if (data[j].high >= data[i].high) isHigh = false;
   620→      if (data[j].low <= data[i].low) isLow = false;
   621→    }
   622→    if (isHigh) highs.push({ price: data[i].high, idx: i });
   623→    if (isLow) lows.push({ price: data[i].low, idx: i });
   624→  }
   625→  return { highs, lows };
   626→}

   679→  if (price >= 10000) return 1000;

   691→// ─── Helper: Linear System Solver (Gaussian Elimination) ─────────────────
   692→function solveLinearSystem(A: number[][], b: number[]): number[] | null {
   693→  const n = b.length;
   694→  const aug = A.map((row, i) => [...row, b[i]]);
   695→
   696→  for (let col = 0; col < n; col++) {
   697→    let maxVal = Math.abs(aug[col][col]);
   698→    let maxRow = col;
   699→    for (let row = col + 1; row < n; row++) {
   700→      if (Math.abs(aug[row][col]) > maxVal) {
   701→        maxVal = Math.abs(aug[row][col]);
   702→        maxRow = row;
   703→      }
   704→    }
   705→    if (maxVal < 1e-10) return null;
   706→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   707→
   708→    for (let row = col + 1; row < n; row++) {
   709→      const factor = aug[row][col] / aug[col][col];
   710→      for (let j = col; j <= n; j++) {
   711→        aug[row][j] -= factor * aug[col][j];
   712→      }
   713→    }
   714→  }
   715→
   716→  const x = new Array(n).fill(0);
   717→  for (let i = n - 1; i >= 0; i--) {
   718→    x[i] = aug[i][n];
   719→    for (let j = i + 1; j < n; j++) {
   720→      x[i] -= aug[i][j] * x[j];
   721→    }
   722→    if (Math.abs(aug[i][i]) < 1e-10) return null;
   723→    x[i] /= aug[i][i];
   724→  }
   725→
   726→  return x;
   727→}

   729→// ─── Helper: Generate All 7-Method S/R Levels ─────────────────────────────
   730→function generateAllLevels(data: OHLCV[], endIdx: number): RawSRLevel[] {
   731→  const levels: RawSRLevel[] = [];
   732→  if (endIdx < 10) return levels;
   733→
   734→  const slice = data.slice(0, endIdx + 1);
   735→  const currentPrice = slice[slice.length - 1].close;
   736→  const closes = slice.map(d => d.close);
   737→  const noFib: [string, string] = ['—', ''];
   738→
   739→  // ── Method 1: Swing High/Low (Price Action) ──────────────────────────
   740→  for (const lookback of [3, 5, 7]) {
   741→    const swings = findSwingLevels(slice, lookback);
   742→    const tol = currentPrice * 0.005; // 0.5% tolerance
   743→
   744→    // Connect swing highs → resistance
   745→    const usedHigh = new Set<number>();
   746→    for (const h of swings.highs) {
   747→      if (usedHigh.has(h.idx)) continue;
   748→      let sum = h.price;
   749→      let count = 1;
   750→      for (const h2 of swings.highs) {
   751→        if (h2 === h || usedHigh.has(h2.idx)) continue;
   752→        if (Math.abs(h2.price - h.price) < tol) {
   753→          sum += h2.price;
   754→          count++;
   755→          usedHigh.add(h2.idx);
   756→        }
   757→      }
   758→      usedHigh.add(h.idx);
   759→      if (sum / count > currentPrice * 1.005) {
   760→        levels.push({ price: sum / count, method: 'Swing_High', type: 'resistance', ...noFib });
   761→      }
   762→    }
   763→
   764→    // Connect swing lows → support
   765→    const usedLow = new Set<number>();
   766→    for (const l of swings.lows) {
   767→      if (usedLow.has(l.idx)) continue;
   768→      let sum = l.price;
   769→      let count = 1;
   770→      for (const l2 of swings.lows) {
   771→        if (l2 === l || usedLow.has(l2.idx)) continue;
   772→        if (Math.abs(l2.price - l.price) < tol) {
   773→          sum += l2.price;
   774→          count++;
   775→          usedLow.add(l2.idx);
   776→        }
   777→      }
   778→      usedLow.add(l.idx);
   779→      if (sum / count < currentPrice * 0.995) {
   780→        levels.push({ price: sum / count, method: 'Swing_Low', type: 'support', ...noFib });
   781→      }
   782→    }
   783→  }
   784→
   785→  // ── Method 2: SMA Dynamic S/R ──────────────────────────────────────
   786→  for (const period of [20, 50, 100, 200]) {
   787→    const smaVal = sma(closes, period);
   788→    if (smaVal <= 0) continue;
   789→    const type: 'support' | 'resistance' = currentPrice > smaVal ? 'support' : 'resistance';
   790→    levels.push({ price: smaVal, method: `SMA_${period}`, type, ...noFib });
   791→  }
   792→
   793→  // ── Method 3: Bollinger Bands ───────────────────────────────────────
   794→  const bb = calcBollingerBands(closes);
   795→  if (bb.upper > 0 && bb.lower > 0) {
   796→    levels.push({ price: bb.upper, method: 'BB_Upper', type: 'resistance', ...noFib });
   797→    levels.push({ price: bb.lower, method: 'BB_Lower', type: 'support', ...noFib });
   798→  }
   799→
   800→  // ── Method 4: Fibonacci Retracement + Extension ─────────────────────
   801→  interface SwingPair { high: number; highIdx: number; low: number; lowIdx: number; range: number }
   802→  let bestSwing: SwingPair | null = null;
   803→
   804→  for (const lb of [3, 5, 7, 10, 15, 20]) {
   805→    const sw = findSwingLevels(slice, lb);
   806→    if (sw.highs.length === 0 || sw.lows.length === 0) continue;
   807→    for (const h of sw.highs) {
   808→      for (const l of sw.lows) {
   809→        if (Math.abs(h.idx - l.idx) < 3) continue;
   810→        const range = Math.abs(h.price - l.price);
   811→        if (range / currentPrice < 0.03) continue;
   812→        if (!bestSwing || range > bestSwing.range) {
   813→          bestSwing = {
   814→            high: Math.max(h.price, l.price),
   815→            highIdx: h.price > l.price ? h.idx : l.idx,
   816→            low: Math.min(h.price, l.price),
   817→            lowIdx: h.price < l.price ? l.idx : h.idx,
   818→            range,
   819→          };
   820→        }
   821→      }
   822→    }
   823→  }
   824→
   825→  if (bestSwing) {
   826→    const H = bestSwing.high;
   827→    const L = bestSwing.low;
   828→    const range = H - L;
   829→
   830→    for (const ratio of FIB_RETRACEMENTS) {
   831→      const price = H - ratio * range;
   832→      if (price <= 0) continue;
   833→      const rounded = roundToNice(price);
   834→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   835→      levels.push({
   836→        price: rounded, method: `Fibonacci_${ratio}`, type,
   837→        fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   838→      });
   839→    }
   840→
   841→    for (const extRatio of FIB_EXTENSIONS) {
   842→      const price = H + (extRatio - 1) * range;
   843→      if (price <= 0) continue;
   844→      const rounded = roundToNice(price);
   845→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   846→      levels.push({
   847→        price: rounded, method: `Fibonacci_${extRatio}`, type,
   848→        fibRatio: String(extRatio), fibLabel: FIB_LABELS[String(extRatio)] || '',
   849→      });
   850→    }
   851→
   852→    // Inverse extensions (below L)
   853→    for (const extRatio of FIB_EXTENSIONS) {
   854→      const price = L - (extRatio - 1) * range;
   855→      if (price <= 0) continue;
   856→      const rounded = roundToNice(price);
   857→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   858→      levels.push({
   859→        price: rounded, method: `Fibonacci_${-extRatio}`, type,
   860→        fibRatio: String(-extRatio), fibLabel: '',
   861→      });
   862→    }
   863→
   864→    // Secondary swing confluence (check overlap with secondary swings)
   865→    const secondarySwings: SwingPair[] = [];
   866→    for (const lb of [3, 5, 7]) {
   867→      const sw = findSwingLevels(slice, lb);
   868→      for (const h of sw.highs) {
   869→        for (const l of sw.lows) {
   870→          if (Math.abs(h.idx - l.idx) < 3) continue;
   871→          const r = Math.abs(h.price - l.price);
   872→          if (r / currentPrice < 0.05) continue;
   873→          if (bestSwing && Math.abs(r - bestSwing.range) / bestSwing.range < 0.1) continue;
   874→          secondarySwings.push({
   875→            high: Math.max(h.price, l.price), highIdx: h.price > l.price ? h.idx : l.idx,
   876→            low: Math.min(h.price, l.price), lowIdx: h.price < l.price ? l.idx : h.idx,
   877→            range: r,
   878→          });
   879→        }
   880→      }
   881→    }
   882→    // Secondary swings' fib levels are merged via confluence (1% rule in mergeLevels)
   883→    const confluenceTol = currentPrice * 0.005;
   884→    for (const sec of secondarySwings.slice(0, 3)) {
   885→      const sH = sec.high, sL = sec.low, sR = sH - sL;
   886→      for (const ratio of FIB_RETRACEMENTS) {
   887→        const price = roundToNice(sH - ratio * sR);
   888→        if (price <= 0) continue;
   889→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   890→        levels.push({
   891→          price, method: `Fibonacci_${ratio}`, type,
   892→          fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   893→        });
   894→      }
   895→      for (const extR of FIB_EXTENSIONS) {
   896→        const price = roundToNice(sH + (extR - 1) * sR);
   897→        if (price <= 0) continue;
   898→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   899→        levels.push({
   900→          price, method: `Fibonacci_${extR}`, type,
   901→          fibRatio: String(extR), fibLabel: FIB_LABELS[String(extR)] || '',
   902→        });
   903→      }
   904→    }
   905→  }
   906→
   907→  // ── Method 5: Volume Profile (VAP) ──────────────────────────────────
   908→  const dataHasVolume = slice.some(d => d.volume > 0);
   909→  if (dataHasVolume) {
   910→    let minP = Infinity, maxP = -Infinity;
   911→    for (const d of slice) {
   912→      if (d.low < minP) minP = d.low;
   913→      if (d.high > maxP) maxP = d.high;
   914→    }
   915→    const numBins = 40;
   916→    const binSize = Math.max((maxP - minP) / numBins, currentPrice * 0.001);
   917→    const vap = new Array(numBins).fill(0);
   918→    const binCenters: number[] = [];
   919→
   920→    for (let b = 0; b < numBins; b++) {
   921→      binCenters.push(minP + (b + 0.5) * binSize);
   922→    }
   923→
   924→    for (const d of slice) {
   925→      for (let b = 0; b < numBins; b++) {
   926→        if (d.low <= binCenters[b] && binCenters[b] <= d.high) {
   927→          vap[b] += d.volume;
   928→        }
   929→      }
   930→    }
   931→
   932→    const maxVAP = Math.max(...vap);
   933→    if (maxVAP > 0) {
   934→      // POC (Point of Control)
   935→      let pocIdx = 0;
   936→      for (let b = 1; b < numBins; b++) {
   937→        if (vap[b] > vap[pocIdx]) pocIdx = b;
   938→      }
   939→      const pocPrice = binCenters[pocIdx];
   940→      const pocType: 'support' | 'resistance' = pocPrice > currentPrice ? 'resistance' : 'support';
   941→      levels.push({ price: pocPrice, method: 'VAP_POC', type: pocType, ...noFib });
   942→
   943→      // HVN (High Volume Nodes)
   944→      for (let b = 0; b < numBins; b++) {
   945→        if (b === pocIdx) continue;
   946→        if (vap[b] > 0.7 * maxVAP) {
   947→          const hvnPrice = binCenters[b];
   948→          const hvnType: 'support' | 'resistance' = hvnPrice > currentPrice ? 'resistance' : 'support';
   949→          levels.push({ price: hvnPrice, method: 'VAP_HVN', type: hvnType, ...noFib });
   950→        }
   951→      }
   952→    }
   953→  }
   954→
   955→  // ── Method 6: Pivot Points (4 algorithms) ───────────────────────────
   956→  if (endIdx >= 1) {
   957→    const prev = slice[endIdx - 1]; // Previous completed bar
   958→    const pH = prev.high, pL = prev.low, pC = prev.close, pO = prev.open;
   959→
   960→    // Standard Pivot
   961→    {
   962→      const PP = (pH + pL + pC) / 3;
   963→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   964→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   965→      const R3 = pH + 2 * (PP - pL), S3 = pL - 2 * (pH - PP);
   966→ if (R1 > 0) levels.push({ price: R1, method: 'Pivot_R1', type: 'resistance', ...noFib });
   967→      if (R2 > 0) levels.push({ price: R2, method: 'Pivot_R2', type: 'resistance', ...noFib });
   968→      if (R3 > 0) levels.push({ price: R3, method: 'Pivot_R3', type: 'resistance', ...noFib });
   969→      if (S1 > 0) levels.push({ price: S1, method: 'Pivot_S1', type: 'support', ...noFib });
   970→      if (S2 > 0) levels.push({ price: S2, method: 'Pivot_S2', type: 'support', ...noFib });
   971→      if (S3 > 0) levels.push({ price: S3, method: 'Pivot_S3', type: 'support', ...noFib });
   972→    }
   973→
   974→    // Fibonacci Pivot
   975→    {
   976→      const PP = (pH + pL + pC) / 3;
   977→      const range = pH - pL;
   978→      for (const [mult, suffix] of [[0.382, '1'], [0.618, '2'], [1.0, '3']] as [number, string][]) {
   979→        const R = PP + mult * range;
   980→        const S = PP - mult * range;
   981→        if (R > 0) levels.push({ price: R, method: `Fib_Pivot_R${suffix}`, type: 'resistance', ...noFib });
   982→        if (S > 0) levels.push({ price: S, method: `Fib_Pivot_S${suffix}`, type: 'support', ...noFib });
   983→      }
   984→    }
   985→
   986→    // Woodie's Pivot
   987→    {
   988→      const PP = (pH + pL + 2 * pC) / 4;
   989→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   990→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   991→      if (R1 > 0) levels.push({ price: R1, method: 'Woodie_R1', type: 'resistance', ...noFib });
   992→      if (R2 > 0) levels.push({ price: R2, method: 'Woodie_R2', type: 'resistance', ...noFib });
   993→      if (S1 > 0) levels.push({ price: S1, method: 'Woodie_S1', type: 'support', ...noFib });
   994→      if (S2 > 0) levels.push({ price: S2, method: 'Woodie_S2', type: 'support', ...noFib });
   995→    }
   996→
   997→    // DeMark Pivot
   998→    {
   999→      const X = pC < pO ? pH + 2 * pL + pC : pC > pO ? 2 * pH + pL + pC : pH + pL + 2 * pC;
  1000→      const PP = X / 4;
  1001→      const R1 = X / 2 - pL;
  1002→      const S1 = X / 2 - pH;
  1003→      if (R1 > 0) levels.push({ price: R1, method: 'DeMark_R1', type: 'resistance', ...noFib });
  1004→      if (S1 > 0) levels.push({ price: S1, method: 'DeMark_S1', type: 'support', ...noFib });
  1005→    }
  1006→  }
  1007→
  1008→  // ── Method 7: Psychological Levels ──────────────────────────────────
  1009→  {
  1010→    const step = psychStep(currentPrice);
  1011→    const base = psychRound(currentPrice);
  1012→    for (const mult of [-2, -1, 1, 2]) {
  1013→      const price = base + mult * step;
  1014→      if (price <= 0) continue;
  1015→      const type: 'support' | 'resistance' = price > currentPrice ? 'resistance' : 'support';
  1016→      levels.push({ price, method: 'Psychological', type, ...noFib });
  1017→    }
  1018→  }
  1019→
  1020→  return levels;
  1021→}

  1023→// ─── Helper: Merge Nearby Levels (1% Confluence) ──────────────────────────
  1024→function mergeLevels(rawLevels: RawSRLevel[], currentPrice: number): MergedSRLevel[] {
  1025→  if (rawLevels.length === 0) return [];
  1026→
  1027→  const tol = currentPrice * 0.01; // 1% tolerance
  1028→  const merged: MergedSRLevel[] = [];
  1029→  const used = new Set<number>();
  1030→
  1031→  for (let i = 0; i < rawLevels.length; i++) {
  1032→    if (used.has(i)) continue;
  1033→
  1034→    let sumPrice = rawLevels[i].price;
  1035→    let count = 1;
  1036→    const methods = new Set<string>();
  1037→    methods.add(rawLevels[i].method);
  1038→    let bestFibRatio = rawLevels[i].fibRatio;
  1039→    let bestFibLabel = rawLevels[i].fibLabel;
  1040→
  1041→    for (let j = i + 1; j < rawLevels.length; j++) {
  1042→      if (used.has(j)) continue;
  1043→      if (Math.abs(rawLevels[j].price - rawLevels[i].price) < tol) {
  1044→        sumPrice += rawLevels[j].price;
  1045→        count++;
  1046→        methods.add(rawLevels[j].method);
  1047→        used.add(j);
  1048→        // Prefer Fibonacci label if available
  1049→        if (rawLevels[j].fibRatio !== '—' && bestFibRatio === '—') {
  1050→          bestFibRatio = rawLevels[j].fibRatio;
  1051→          bestFibLabel = rawLevels[j].fibLabel;
  1052→        }
  1053→      }
  1054→    }
  1055→    used.add(i);
  1056→
  1057→    const avgPrice = sumPrice / count;
  1058→    const type: 'support' | 'resistance' = avgPrice >= currentPrice ? 'resistance' : 'support';
  1059→
  1060→    merged.push({
  1061→      price: avgPrice,
  1062→      type,
  1063→      methods: [...methods],
  1064→      overlapCount: methods.size,
  1065→      fibRatio: bestFibRatio,
  1066→      fibLabel: bestFibLabel,
  1067→      score: 0,
  1068→      strength: 1,
  1069→      touchCount: 0,
  1070→      volumeRatio: 0,
  1071→      freshness: 0,
  1072→      distancePercent: 0,
  1073→    });
  1074→  }
  1075→
  1076→  return merged;
  1077→}

  1079→// ─── Helper: Compute Level Features (5 ML features) ──────────────────────
  1080→function computeLevelFeatures(
  1081→  levelPrice: number,
  1082→  data: OHLCV[],
  1083→  endIdx: number,
  1084→  currentPrice: number,
  1085→  overlapCount: number,
  1086→): number[] {
  1087→  const lookback = Math.min(30, endIdx);
  1088→  const startIdx = endIdx - lookback;
  1089→  const touchTol = currentPrice * 0.002; // 0.2%
  1090→
  1091→  let touchCount = 0;
  1092→  let totalVolAtTouches = 0;
  1093→  let totalVol = 0;
  1094→  let barsSinceLastTouch = lookback;
  1095→
  1096→  for (let i = startIdx; i <= endIdx; i++) {
  1097→    const bar = data[i];
  1098→    totalVol += bar.volume;
  1099→
  1100→    if (bar.low <= levelPrice + touchTol && bar.high >= levelPrice - touchTol) {
  1101→      touchCount++;
  1102→      totalVolAtTouches += bar.volume;
  1103→      barsSinceLastTouch = endIdx - i;
  1104→    }
  1105→  }
  1106→
  1107→  const avgVol = lookback > 0 ? totalVol / lookback : 0;
  1108→  const volumeRatio = (touchCount > 0 && avgVol > 0) ? (totalVolAtTouches / touchCount) / avgVol : 1;
  1109→  const freshness = Math.max(0, Math.min(1, 1 - barsSinceLastTouch / 30));
  1110→  const distancePercent = Math.min(1, Math.abs(levelPrice - currentPrice) / currentPrice);
  1111→
  1112→  // Features: [touch_count, volume_ratio, overlap_count, freshness, distance_percent]
  1113→  return [touchCount, volumeRatio, overlapCount - 1, freshness, distancePercent];
  1114→}

  1116→// ─── Helper: Train S/R Scoring Weights (Ridge OLS) ───────────────────────
  1117→function trainSRWeights(data: OHLCV[]): number[] {
  1118→  const defaultWeights = [0.25, 0.20, 0.25, 0.15, 0.15];
  1119→
  1120→  if (data.length < 50) return defaultWeights;
  1121→
  1122→  const X: number[][] = [];
  1123→  const y: number[] = [];
  1124→
  1125→  for (let i = 40; i <= data.length - 11; i += 5) {
  1126→    const barPrice = data[i].close;
  1127→    const rawLevels = generateAllLevels(data, i);
  1128→    if (rawLevels.length === 0) continue;
  1129→    const merged = mergeLevels(rawLevels, barPrice);
  1130→
  1131→    for (const level of merged) {
  1132→      const features = computeLevelFeatures(level.price, data, i, barPrice, level.overlapCount);
  1133→      X.push(features);
  1134→
  1135→      // Label: did price return within 10 bars?
  1136→      let returned = false;
  1137→      const retTol = barPrice * 0.005;
  1138→      for (let j = i + 1; j <= Math.min(i + 10, data.length - 1); j++) {
  1139→        if (data[j].low <= level.price + retTol && data[j].high >= level.price - retTol) {
  1140→          returned = true;
  1141→          break;
  1142→        }
  1143→      }
  1144→      y.push(returned ? 1 : 0);
  1145→    }
  1146→  }
  1147→
  1148→  if (X.length < 20) return defaultWeights;
  1149→
  1150→  // Ridge OLS: w = (X^T X + λI)^(-1) X^T y
  1151→  const n = X.length;
  1152→  const p = 5;
  1153→  const lambda = 0.01;
  1154→
  1155→  const XtX: number[][] = Array.from({ length: p }, () => new Array(p).fill(0));
  1156→  for (let i = 0; i < n; i++) {
  1157→    for (let a = 0; a < p; a++) {
  1158→      for (let b = 0; b < p; b++) {
  1159→        XtX[a][b] += X[i][a] * X[i][b];
  1160→      }
  1161→    }
  1162→  }
  1163→  for (let a = 0; a < p; a++) XtX[a][a] += lambda;
  1164→
  1165→  const Xty: number[] = new Array(p).fill(0);
  1166→  for (let i = 0; i < n; i++) {
  1167→    for (let a = 0; a < p; a++) {
  1168→      Xty[a] += X[i][a] * y[i];
  1169→    }
  1170→  }
  1171→
  1172→  const weights = solveLinearSystem(XtX, Xty);
  1173→  if (!weights) return defaultWeights;
  1174→  if (weights.some(w => isNaN(w) || !isFinite(w))) return defaultWeights;
  1175→
  1176→  for (let i = 0; i < p; i++) {
  1177→    if (weights[i] < 0) weights[i] = 0;
  1178→  }
  1179→
  1180→  const sum = weights.reduce((a, b) => a + b, 0);
  1181→  if (sum <= 0) return defaultWeights;
  1182→
  1183→  return weights.map(w => w / sum);
  1184→}

  1186→// ─── Helper: WMA Array (full series) ─────────────────────────────────────
  1187→function calcWMAArray(closes: number[], period: number): number[] {
  1188→  const result: number[] = [];
  1189→  if (closes.length < period) return closes.map(() => 0);
  1190→  for (let i = 0; i < closes.length; i++) {
  1191→    if (i < period - 1) { result.push(0); continue; }
  1192→    let numer = 0;
  1193→    let denom = 0;
  1194→    for (let j = 0; j < period; j++) {
  1195→      const weight = period - j;
  1196→      numer += closes[i - period + 1 + j] * weight;
  1197→      denom += weight;
  1198→    }
  1199→    result.push(numer / denom);
  1200→  }
  1201→  return result;
  1202→}

  1319→function calcFisherTransform(data: OHLCV[], period: number = 9): number {
  1320→  if (data.length < period) return 0;
  1321→  let fisher = 0;
  1322→  for (let i = period; i < data.length; i++) {
  1323→    let maxH = -Infinity;
  1324→    let minL = Infinity;
  1325→    for (let j = i - period + 1; j <= i; j++) {
  1326→      if (data[j].high > maxH) maxH = data[j].high;
  1327→      if (data[j].low < minL) minL = data[j].low;
  1328→    }
  1329→    const range = maxH - minL;
  1330→    if (range === 0) continue;
  1331→    const raw = 2 * ((data[i].close - minL) / range - 0.5);
  1332→    const val = Math.max(0.0001, Math.min(0.9999, (raw + 1) / 2));
  1333→    fisher = 0.5 * Math.log(val / (1 - val)) + 0.5 * fisher;
  1334→  }
  1335→  return fisher;
  1336→}

  1481→// ─── Support / Resistance (7-Method ML-Based System) ─────────────────────
  1482→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
  1483→  resistances: number[];
  1484→  supports: number[];
  1485→  supportStrengths: LevelStrength[];
  1486→  resistanceStrengths: LevelStrength[];
  1487→  priceTargets: LevelStrength[];
  1488→} {
  1489→  if (data.length < 10) {
  1490→    return {
  1491→      resistances: [], supports: [],
  1492→      supportStrengths: [], resistanceStrengths: [],
  1493→      priceTargets: [],
  1494→    };
  1495→  }
  1496→
  1497→  // ── Step 1: Generate all levels from 7 methods ────────────────────────
  1498→  const rawLevels = generateAllLevels(data, data.length - 1);
  1499→
  1500→  // ── Step 2: Merge nearby levels (confluence within 1%) ────────────────
  1501→  let merged = mergeLevels(rawLevels, currentPrice);
  1502→
  1503→  // ── Step 3: Train ML weights on historical data ───────────────────────
  1504→  const weights = trainSRWeights(data);
  1505→
  1506→  // ── Step 4: Compute features & score each merged level ────────────────
  1507→  for (const level of merged) {
  1508→    const features = computeLevelFeatures(level.price, data, data.length - 1, currentPrice, level.overlapCount);
  1509→
  1510→    const touchScore = Math.min(features[0] / 5, 1);
  1511→    const volumeScore = Math.min(features[1] / 2, 1);
  1512→    const overlapScore = Math.min(features[2] * 2, 10) / 10;
  1513→    const freshnessScore = features[3];
  1514→    const distanceScore = 1 - features[4];
  1515→
  1516→    const powerScore = (
  1517→      weights[0] * touchScore +
  1518→      weights[1] * volumeScore +
  1519→      weights[2] * overlapScore +
  1520→      weights[3] * freshnessScore +
  1521→      weights[4] * distanceScore
  1522→    ) * 10;
  1523→
  1524→    level.score = Math.max(0, Math.min(10, powerScore));
  1525→    level.strength = Math.max(1, Math.min(10, Math.round(level.score)));
  1526→    level.touchCount = features[0];
  1527→    level.volumeRatio = features[1];
  1528→    level.freshness = features[3];
  1529→    level.distancePercent = features[4];
  1530→  }
  1531→
  1532→  // ── Step 5: Psychological rounding of all level prices ─────────────────
  1533→  for (const level of merged) {
  1534→    level.price = psychRound(level.price);
  1535→  }
  1536→
  1537→  // Re-classify types after rounding
  1538→  for (const level of merged) {
  1539→    level.type = level.price >= currentPrice ? 'resistance' : 'support';
  1540→  }
  1541→
  1542→  // Deduplicate after rounding (keep strongest at same price)
  1543→  {
  1544→    const deduped: MergedSRLevel[] = [];
  1545→    for (const level of merged) {
  1546→      const existing = deduped.find(d => Math.abs(d.price - level.price) < currentPrice * 0.001);
  1547→      if (existing) {
  1548→        if (level.score > existing.score) {
  1549→          Object.assign(existing, level);
  1550→        }
  1551→      } else {
  1552→        deduped.push(level);
  1553→      }
  1554→    }
  1555→    merged = deduped;
  1556→  }
  1557→
  1558→  // ── Step 6: Separate into supports and resistances ─────────────────────
  1559→  let supports = merged.filter(l => l.type === 'support');
  1560→  let resistances = merged.filter(l => l.type === 'resistance');
  1561→
  1562→  // ── Step 7: Distance filtering (strict 5-10% gap rules) ─────────────
  1563→  // Rules:
  1564→  //   1. Consecutive lines of same type: 5% ≤ gap ≤ 10%
  1565→  //   2. Nearest support ≤ 10% below price (can be < 5%)
  1566→  //   3. Nearest resistance ≤ 10% above price (can be < 5%)
  1567→
  1568→  const filterAndSelectLevels = (
  1569→    levels: MergedSRLevel[],
  1570→    type: 'support' | 'resistance'
  1571→  ): MergedSRLevel[] => {
  1572→    if (levels.length === 0) return [];
  1573→
  1574→    // Sort: supports descending (nearest first), resistances ascending (nearest first)
  1575→    const sorted = [...levels].sort((a, b) =>
  1576→      type === 'support' ? b.price - a.price : a.price - b.price
  1577→    );
  1578→
  1579→    // ── Rule 2/3: Nearest level must be within 10% of price ──
  1580→    const nearest = sorted[0];
  1581→    const distFromPrice = Math.abs(nearest.price - currentPrice) / currentPrice;
  1582→    if (distFromPrice > 0.10) {
  1583→      // Nearest level is too far — generate a synthetic level at ~5% from price
  1584→      const syntheticPrice = type === 'support'
  1585→        ? Math.round(currentPrice * 0.95 / psychStep(currentPrice)) * psychStep(currentPrice)
  1586→        : Math.round(currentPrice * 1.05 / psychStep(currentPrice)) * psychStep(currentPrice);
  1587→      const synthetic: MergedSRLevel = {
  1588→        price: syntheticPrice,
  1589→        type,
  1590→        score: 3,
  1591→        strength: 3,
  1592→        overlapCount: 1,
  1593→        methods: ['Psychological'],
  1594→        fibRatio: '—',
  1595→        fibLabel: '',
  1596→      };
  1597→      sorted.unshift(synthetic);
  1598→    }
  1599→
  1600→    // ── Rule 1: Select levels with 5-10% gaps between consecutive ──
  1601→    const result: MergedSRLevel[] = [sorted[0]];
  1602→
  1603→    for (let i = 1; i < sorted.length && result.length < 6; i++) {
  1604→      const prevPrice = result[result.length - 1].price;
  1605→      const currPrice = sorted[i].price;
  1606→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
  1607→
  1608→      if (gap < 0.05) {
  1609→        // Too close — keep the stronger one
  1610→        if (sorted[i].score > result[result.length - 1].score) {
  1611→          result[result.length - 1] = sorted[i];
  1612→        }
  1613→      } else if (gap <= 0.10) {
  1614→        // Perfect gap (5-10%) — accept
  1615→        result.push(sorted[i]);
  1616→      } else {
  1617→        // Gap > 10% — try to find an intermediate level
  1618→        let found = false;
  1619→        for (let j = i + 1; j < sorted.length; j++) {
  1620→          const midPrice = sorted[j].price;
  1621→          const gapFromPrev = Math.abs(midPrice - prevPrice) / prevPrice;
  1622→          const gapToNext = Math.abs(currPrice - midPrice) / midPrice;
  1623→          if (gapFromPrev >= 0.05 && gapFromPrev <= 0.10) {
  1624→            result.push(sorted[j]);
  1625→            found = true;
  1626→            // Skip levels that are too close to the inserted one
  1627→            i = j;
  1628→            break;
  1629→          }
  1630→        }
  1631→        if (!found) {
  1632→          // No intermediate found — generate a synthetic level at ~7.5% from prev
  1633→          const synthGap = 0.075;
  1634→          const syntheticPrice = type === 'support'
  1635→            ? Math.round((prevPrice * (1 - synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice)
  1636→            : Math.round((prevPrice * (1 + synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice);
  1637→          // Make sure it's not too close to next level
  1638→          const gapToNext = Math.abs(currPrice - syntheticPrice) / syntheticPrice;
  1639→          if (gapToNext >= 0.05) {
  1640→            const synthetic: MergedSRLevel = {
  1641→              price: syntheticPrice,
  1642→              type,
  1643→              score: 2,
  1644→              strength: 2,
  1645→              overlapCount: 1,
  1646→              methods: ['Psychological'],
  1647→              fibRatio: '—',
  1648→              fibLabel: '',
  1649→            };
  1650→            result.push(synthetic);
  1651→          }
  1652→        }
  1653→      }
  1654→    }
  1655→
  1656→    return result;
  1657→  };
  1658→
  1659→  const finalSupports = filterAndSelectLevels(supports, 'support');
  1660→  const finalResistances = filterAndSelectLevels(resistances, 'resistance');
  1661→
  1662→  // ── Step 9: Build LevelStrength objects ────────────────────────────────
  1663→  const toLevelStrength = (m: MergedSRLevel): LevelStrength => {
  1664→    let grade: string;
  1665→    if (m.score >= 8.5) grade = 'Very Strong';
  1666→    else if (m.score >= 7) grade = 'Strong';
  1667→    else if (m.score >= 5) grade = 'Moderate';
  1668→    else grade = 'Weak';
  1669→
  1670→    return {
  1671→      price: m.price,
  1672→      strength: m.strength,
  1673→      score: m.score,
  1674→      grade,
  1675→      isTarget: m.score >= 7,
  1676→      overlapCount: m.overlapCount,
  1677→      methods: m.methods,
  1678→      fibRatio: m.fibRatio,
  1679→      fibLabel: m.fibLabel,
  1680→    };
  1681→  };
  1682→
  1683→  const supportStrengths = finalSupports.map(toLevelStrength);
  1684→  const resistanceStrengths = finalResistances.map(toLevelStrength);
  1685→
  1686→  // Price targets: all levels with isTarget=true
  1687→  const priceTargets = [...supportStrengths, ...resistanceStrengths].filter(l => l.isTarget);
  1688→
  1689→  return {
  1690→    resistances: resistanceStrengths.map(l => l.price),
  1691→    supports: supportStrengths.map(l => l.price),
  1692→    supportStrengths,
  1693→    resistanceStrengths,
  1694→    priceTargets,
  1695→  };
  1696→}

  1700→function calcTrend(closes: number[], period: number): TrendResult {
  1701→  const slice = closes.slice(-Math.min(period, closes.length));
  1702→  const { slope, r2 } = linearRegression(slice);
  1703→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
  1704→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
  1705→  const priceThreshold = avgPrice * 0.001; // 0.1%
  1706→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
  1707→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
  1708→}

  1715→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
  1716→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
  1717→  const slice = data.slice(0, barIdx + 1);
  1718→  const price = slice[slice.length - 1].close;
  1719→  let nearestR = price * 1.05;
  1720→  let nearestS = price * 0.95;
  1721→
  1722→  // Collect all swing highs (resistance) and swing lows (support)
  1723→  const resistances: number[] = [];
  1724→  const supports: number[] = [];
  1725→  for (const lb of [3, 5, 7]) {
  1726→    const swings = findSwingLevels(slice, lb);
  1727→    resistances.push(...swings.highs);
  1728→    supports.push(...swings.lows);
  1729→  }
  1730→
  1731→  // Find nearest resistance above price
  1732→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
  1733→  if (abovePrice.length > 0) nearestR = abovePrice[0];
  1734→
  1735→  // Find nearest support below price
  1736→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
  1737→  if (belowPrice.length > 0) nearestS = belowPrice[0];
  1738→
  1739→  return { R1: nearestR, S1: nearestS };
  1740→}

  1742→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
  1743→// Returns true if bullish divergence detected, false if bearish, null if none
  1744→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
  1745→  const lookback = 20;
  1746→  const start = Math.max(0, barIdx - lookback);
  1747→  const slice = data.slice(start, barIdx + 1);
  1748→  if (slice.length < 10) return null;
  1749→
  1750→  const mid = Math.floor(slice.length / 2);
  1751→  const firstHalf = slice.slice(0, mid);
  1752→  const secondHalf = slice.slice(mid);
  1753→
  1754→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
  1755→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
  1756→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
  1757→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
  1758→
  1759→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
  1760→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
  1761→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
  1762→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
  1763→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
  1764→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
  1765→
  1766→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
  1767→    return false; // bearish divergence
  1768→  }
  1769→
  1770→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
  1771→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
  1772→
  1773→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
  1774→    return true; // bullish divergence
  1775→  }
  1776→
  1777→  return null;
  1778→}

  1784→/**
  1785→ * Computes all 16 ML features at a specific historical bar.
  1786→ * Uses data[0..barIdx] (inclusive) for all indicator calculations.
  1787→ * Returns an object keyed by feature name matching FEATURES_ORDER.
  1788→ */
  1789→function computeFeaturesAtBar(data: OHLCV[], barIdx: number): Record<string, number> {
  1790→  // Need at least 30 bars for meaningful indicators
  1791→  if (barIdx < 30) {
  1792→    const empty: Record<string, number> = {};
  1793→    for (const key of FEATURES_ORDER) empty[key] = 0.5;
  1794→    return empty;
  1795→  }
  1796→
  1797→  const slice = data.slice(0, barIdx + 1);
  1798→  const closes = slice.map(d => d.close);
  1799→  const price = closes[closes.length - 1];
  1800→  const eps = 1e-10;
  1801→
  1802→  // ── Compute all base indicators ───────────────────────────────────────────
  1803→  const rsiVal = calcRSI(closes);
  1804→  const sliceHasVolume = slice.some(d => d.volume > 0);
  1805→  const mfiVal = sliceHasVolume ? calcMFI(slice) : 50; // neutral when no volume
  1806→  const cciVal = calcCCI(slice);
  1807→  const stoch = calcStochastic(slice);
  1808→  const macd = calcMACD(closes);
  1809→  const adxResult = calcADX(slice);
  1810→  const atrVal = calcATR(slice);
  1811→  const bb = calcBollingerBands(closes);
  1812→  const ma21 = sma(closes, 21);
  1813→  const ma100 = sma(closes, 100);
  1814→  const ema12 = emaCalc(closes, 12);
  1815→  const ema26 = emaCalc(closes, 26);
  1816→  const trendResult = calcTrend(closes, 21);
  1817→  const sr = simplifiedSR(data, barIdx);
  1818→
  1819→  // ── Layer 1: Raw Scores (s_*) ────────────────────────────────────────────
  1820→
  1821→  // s_rsi — RSI mapped to 0-1
  1822→  const s_rsi = rsiVal > 80 ? 0.95
  1823→    : rsiVal > 70 ? 0.85
  1824→    : rsiVal > 60 ? 0.70
  1825→    : rsiVal > 50 ? 0.55
  1826→    : rsiVal > 40 ? 0.45
  1827→    : rsiVal > 30 ? 0.30
  1828→    : rsiVal > 20 ? 0.15
  1829→    : 0.05;
  1830→
  1831→  // s_mfi — MFI mapped to 0-1 (same formula as RSI)
  1832→  const s_mfi = mfiVal > 80 ? 0.95
  1833→    : mfiVal > 70 ? 0.85
  1834→    : mfiVal > 60 ? 0.70
  1835→    : mfiVal > 50 ? 0.55
  1836→    : mfiVal > 40 ? 0.45
  1837→    : mfiVal > 30 ? 0.30
  1838→    : mfiVal > 20 ? 0.15
  1839→    : 0.05;
  1840→
  1841→  // s_cci — CCI mapped to 0-1
  1842→  const s_cci = cciVal > 200 ? 0.90
  1843→    : cciVal > 100 ? 0.75
  1844→    : cciVal > 0 ? 0.60
  1845→    : cciVal > -100 ? 0.40
  1846→    : cciVal > -200 ? 0.25
  1847→    : 0.10;
  1848→
  1849→  // s_adx — Directional bias
  1850→  const s_adx = (adxResult.diPlus - adxResult.diMinus + 100) / 200;
  1851→
  1852→  // s_macd — MACD histogram + signal combined
  1853→  const absMacdLine = Math.abs(macd.line) + 1;
  1854→  const s_macd = macd.line > macd.signal
  1855→    ? 0.5 + Math.min(0.5, macd.histogram / absMacdLine)
  1856→    : 0.5 - Math.min(0.5, Math.abs(macd.histogram) / absMacdLine);
  1857→
  1858→  // s_stoch — Stochastic %K mapped to 0-1
  1859→  const s_stoch = stoch.k / 100;
  1860→
  1861→  // s_bb — Bollinger Band position
  1862→  const bbRange = bb.upper - bb.lower;
  1863→  const s_bb = bbRange > eps
  1864→    ? clamp((price - bb.lower) / bbRange, 0, 1)
  1865→    : 0.5;
  1866→
  1867→  // s_ma21 — Price relative to 21-period SMA
  1868→  const s_ma21 = ma21 > eps
  1869→    ? clamp(0.5 + (price - ma21) / ma21, 0, 1)
  1870→    : 0.5;
  1871→
  1872→  // s_ma100 — Price relative to 100-period SMA
  1873→  const s_ma100 = ma100 > eps
  1874→    ? clamp(0.5 + (price - ma100) / ma100, 0, 1)
  1875→    : 0.5;
  1876→
  1877→  // s_ema — EMA12 vs EMA26 spread
  1878→  const s_ema = ema26 > eps
  1879→    ? clamp(0.5 + (ema12 - ema26) / ema26 * 5, 0, 1)
  1880→    : 0.5;
  1881→
  1882→  // s_atr — Inverse volatility (lower ATR = higher score)
  1883→  const s_atr = price > eps ? 1 - Math.min(atrVal / price * 10, 1) : 0.5;
  1884→
  1885→  // s_trend — R² + angle combination
  1886→  const s_trend = (trendResult.r2 * 0.7) +
  1887→    (Math.max(0, Math.min(1, (trendResult.angle + 45) / 90)) * 0.3);
  1888→
  1889→  // s_sr — Support/Resistance proximity
  1890→  const dR1 = (sr.R1 - price) / (price > eps ? price : 1);
  1891→  const dS1 = (price - sr.S1) / (price > eps ? price : 1);
  1892→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
  1893→
  1894→  // ── Layer 2: Corrections (f_*) ───────────────────────────────────────────
  1895→
  1896→  // f_rsi — RSI with momentum correction
  1897→  const rsiCurrent = rsiVal;
  1898→  const rsi5ago = barIdx >= 5
  1899→    ? calcRSI(closes.slice(0, barIdx - 5 + 1))
  1900→    : rsiCurrent;
  1901→  const f_rsi = clamp(s_rsi + (rsiCurrent - rsi5ago) / 100 * 0.08, 0, 1);
  1902→
  1903→  // f_mfi — MFI with momentum correction (skip when no volume)
  1904→  const mfiCurrent = mfiVal;
  1905→  const mfi5ago = (barIdx >= 5 && sliceHasVolume)
  1906→    ? calcMFI(data.slice(0, barIdx - 5 + 1))
  1907→    : mfiCurrent;
  1908→  const f_mfi = sliceHasVolume
  1909→    ? clamp(s_mfi + (mfiCurrent - mfi5ago) / 100 * 0.08, 0, 1)
  1910→    : 0.5; // neutral when no volume
  1911→
  1912→  // f_cci — CCI with momentum correction (scale: /200 instead of /100)
  1913→  const cciCurrent = cciVal;
  1914→  const cci5ago = barIdx >= 5
  1915→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
  1916→    : cciCurrent;
  1917→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
  1918→
  1919→  // f_macd — Same as s_macd (no additional correction)
  1920→  const f_macd = s_macd;
  1921→
  1922→  // f_stoch — Same as s_stoch
  1923→  const f_stoch = s_stoch;
  1924→
  1925→  // f_stochCross — Stochastic K/D crossover detection
  1926→  let f_stochCross = 0.1;
  1927→  if (barIdx >= 1) {
  1928→    const prevSlice = data.slice(0, barIdx);
  1929→    const prevStoch = calcStochastic(prevSlice);
  1930→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
  1931→      f_stochCross = 0.9; // K crossed above D (bullish)
  1932→    }
  1933→  }
  1934→
  1935→  // f_macdCross — MACD/Signal crossover detection
  1936→  let f_macdCross = 0.1;
  1937→  if (barIdx >= 1) {
  1938→    const prevCloses = closes.slice(0, -1);
  1939→    if (prevCloses.length >= 35) {
  1940→      const prevMacd = calcMACD(prevCloses);
  1941→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
  1942→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
  1943→      }
  1944→    }
  1945→  }
  1946→
  1947→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
  1948→  const divResult = detectDivergenceSimple(data, barIdx);
  1949→  const f_div = divResult === true ? 0.85 : 0.15;
  1950→
  1951→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
  1952→  return {
  1953→    rsi: f_rsi,
  1954→    mfi: f_mfi,
  1955→    cci: f_cci,
  1956→    adx: s_adx,
  1957→    macd: f_macd,
  1958→    stoch: f_stoch,
  1959→    bb: s_bb,
  1960→    ma21: s_ma21,
  1961→    ma100: s_ma100,
  1962→    ema: s_ema,
  1963→    atr: s_atr,
  1964→    trend: s_trend,
  1965→    sr: s_sr,
  1966→    stochCross: f_stochCross,
  1967→    macdCross: f_macdCross,
  1968→    div: f_div,
  1969→  };
  1970→}

  1973→// TRAINING DATA CREATION (Layer 0 — Data Preparation)

  1976→/**
  1977→ * Creates supervised training data from historical OHLCV data.
  1978→ * For each bar, computes features and labels based on future price movement.
  1979→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
  1980→ * Samples every `sampleStep` bars to reduce computation.
  1981→ */
  1982→function createTrainingData(
  1983→  data: OHLCV[],
  1984→  horizon: number = 5,
  1985→  sampleStep: number = 5
  1986→): { X: number[][]; y: number[] } {
  1987→  const X: number[][] = [];
  1988→  const y: number[] = [];
  1989→
  1990→  // Start from horizon+30 to ensure enough data for indicators
  1991→  // End at data.length - horizon - 1 to have future bars for labeling
  1992→  // Sample every sampleStep bars for performance
  1993→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
  1994→    const features = computeFeaturesAtBar(data, i);
  1995→
  1996→    // Convert to array following FEATURES_ORDER
  1997→    const featureVec: number[] = [];
  1998→    let valid = true;
  1999→    for (const key of FEATURES_ORDER) {
  2000→      const val = features[key];
  2001→      if (val === undefined || val === null || Number.isNaN(val)) {
  2002→        valid = false;
  2003→        break;
  2004→      }

  2055→/**
  2056→ * Calculates transition edge weights for the VDss graph.
  2057→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
  2058→ * to determine the relative probability of up/down/pullback/risk transitions.

  2060→function calculateEdgeWeights(
  2061→  bullConsensus: number,
  2062→  adx: number,
  2063→  mlModel: AdaptiveWeightModel
  2064→): { up: number; down: number; pullback: number; risk: number } {

  2068→  let volatility_coef = 1 / 3;
  2069→

  2074→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER

  2076→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
  2077→

  2079→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
  2080→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
  2081→
  2082→    const coefSum = absTrend + absRsi + absAtr;
  2083→    if (coefSum > 1e-10) {
  2084→      trend_coef = absTrend / coefSum;
  2085→      momentum_coef = absRsi / coefSum;
  2086→      volatility_coef = absAtr / coefSum;
  2087→    }
  2088→  }
  2089→
  2090→  // Base weights driven by bull consensus
  2091→  const up_base = bullConsensus * 0.7 + 0.15;
  2092→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
  2093→  const pullback_base = 0.25;
  2094→  const risk_base = 0.12 * (1.2 - adx / 100);
  2095→
  2096→  // Apply ML coefficient adjustments
  2097→  const up = clamp(
  2098→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
  2099→    0.05, 0.95
  2100→  );
  2101→  const down = clamp(
  2102→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
  2103→    0.05, 0.95
  2104→  );
  2105→  const pullback = clamp(
  2106→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
  2107→    0.05, 0.50
  2108→  );
  2109→  const risk = clamp(
  2110→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
  2111→    0.02, 0.30
  2112→  );
  2113→
  2114→  return { up, down, pullback, risk };
  2115→}

  2118→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)

  2120→
  2121→/**
  2122→ * Performs iterative DFS from the start node through the VDss graph,
  2123→ * computing raw path probabilities, then calibrating them against
  2124→ * Layer 4 scenario probabilities.
  2125→ *
  2126→ * Returns:
  2127→ * - calibratedPaths: Map<pathString, probability>
  2128→ * - calibrationFactors: per-scenario multiplier used
  2129→ * - scenarioSums: raw probability sums per scenario before calibration
  2130→ */
  2131→function calculatePathProbabilities(
  2132→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
  2133→  scenarioProbs: Record<string, number>
  2134→): {
  2135→  calibratedPaths: Map<string, number>;
  2136→  calibrationFactors: Record<string, number>;
  2137→  scenarioSums: Record<string, number>;
  2138→} {
  2139→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
  2140→  const MAX_PATHS = 200;
  2141→
  2142→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
  2143→  type StackEntry = [string, string, number];
  2144→
  2145→  // Collect all terminal paths with raw probabilities
  2146→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
  2147→
  2148→  // Iterative DFS
  2149→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
  2150→
  2151→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
  2152→    const [node, pathStr, pathProb] = stack.pop()!;
  2153→
  2154→    const edges = outgoing[node as keyof typeof outgoing];
  2155→    if (!edges) continue;
  2156→
  2157→    // Normalize outgoing edge weights to probabilities for this node
  2158→    // Map edge types to their weights
  2159→    let totalWeight = 0;
  2160→    const edgeWeightsList: number[] = [];
  2161→    for (const [, edgeType] of edges) {
  2162→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
  2163→      edgeWeightsList.push(w);
  2164→      totalWeight += w;
  2165→    }
  2166→
  2167→    if (totalWeight < 1e-10) continue;
  2168→
  2169→    // Expand edges
  2170→    for (let i = 0; i < edges.length; i++) {
  2171→      const [target, edgeType] = edges[i];
  2172→      const edgeProb = edgeWeightsList[i] / totalWeight;
  2173→      const newProb = pathProb * edgeProb;
  2174→      const newPath = pathStr + '→' + target;
  2175→
  2176→      if (edgeType === 'terminal') {
  2177→        // Reached a terminal node
  2178→        const scenario = terminalMap[target] ?? 'R3';
  2179→        rawPaths.push({ path: newPath, scenario, prob: newProb });
  2180→      } else {
  2181→        // Continue DFS
  2182→        stack.push([target, newPath, newProb]);
  2183→      }
  2184→    }
  2185→  }
  2186→
  2187→  // Group raw probabilities by scenario
  2188→  const rawSums: Record<string, number> = { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0, R6: 0, R7: 0, R8: 0, R9: 0 };
  2189→  for (const rp of rawPaths) {
  2190→    rawSums[rp.scenario] = (rawSums[rp.scenario] ?? 0) + rp.prob;
  2191→  }
  2192→
  2193→  // Compute calibration factors: scenarioProbs[scenario] / rawSum[scenario]
  2194→  const calibrationFactors: Record<string, number> = {};
  2195→  for (const scenario of ['R1', 'R2', 'R3', 'R4', 'R5', 'R6', 'R7', 'R8', 'R9']) {
  2196→    const rawSum = rawSums[scenario] ?? 0;
  2197→    const targetProb = (scenarioProbs[scenario] ?? 20) / 100; // convert from percentage
  2198→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
  2199→  }
  2200→
  2201→  // Apply calibration to each path
  2202→  const calibratedPaths = new Map<string, number>();
  2203→  for (const rp of rawPaths) {
  2204→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
  2205→    calibratedPaths.set(rp.path, rp.prob * factor);
  2206→  }
  2207→
  2208→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
  2209→}

  2215→function buildBullishTargets(price: number, atr: number, resistances: number[]): Array<{min: number; max: number}> {
  2216→  // Filter resistances above price, take up to 4
  2217→  const above = resistances.filter(r => r > price);

  2225→    const base = above[i] ?? (price + (i + 1) * 0.25 * atr);
  2226→    // Ensure base is above previous target and above price
  2227→    const safeBase = Math.max(base, prevMax + 1);
  2228→    let min = rnd(Math.max(safeBase - 0.25 * atr, prevMax + 1));

  2230→    if (max <= min) max = rnd(min + 0.25 * atr);
  2231→    if (min <= prevMax) min = rnd(prevMax + 1);
  2232→    if (max <= min) max = rnd(min + 0.25 * atr);
  2233→    // Enforce max 1 ATR range
  2234→    if (max - min > atr) max = rnd(min + atr);
  2235→    results.push({min, max});
  2236→    prevMax = max;
  2237→  }
  2238→  return results;
