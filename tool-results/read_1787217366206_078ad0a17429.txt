    21→function cacheKey(body: Record<string, unknown>): string {
    22→  // Use a stable subset of inputs for caching
    23→  const k = {
    24→    s: body.symbolName,

    36→// ─── Helpers ──────────────────────────────────────────────────────────
    37→function toPersianNum(n: number): string {
    38→  if (!isFinite(n) || isNaN(n)) return '\u06f0';
    39→  return Math.round(n).toLocaleString('fa-IR');
    40→}
    41→
    42→function srGrade(strength: number): string {
    43→  if (strength >= 8.5) return '\u0628\u0633\u06cc\u0627\u0631 \u0642\u0648\u06cc';
    44→  if (strength >= 7) return '\u0642\u0648\u06cc';
    45→  if (strength >= 5) return '\u0645\u062a\u0648\u0633\u0637';
    46→  if (strength >= 3) return '\u0636\u0639\u06cc\u0641';
    47→  return '\u0628\u0633\u06cc\u0627\u0631 \u0636\u0639\u06cc\u0641';
    48→}

    50→// ─── Build ML Selector Input ──────────────────────────────────────────
    51→function buildMLInput(body: Record<string, unknown>): MLSelectorInput {
    52→  const price = (body.currentPrice as number) || 0;
    53→  const trendDir = (body.trendDirection as string) || 'range';
    54→  const adx = (body.adx as number) || 0;
    55→  const rsi = (body.rsi as number) || 50;
    56→  const stochK = (body.stochK as number) || 50;
    57→  const macdHist = (body.macdHist as number) || 0;
    58→  const obv = (body.obv as number) || 0;
    59→  const bollingerUpper = (body.bollingerUpper as number) || 0;
    60→  const bollingerLower = (body.bollingerLower as number) || 0;
    61→  const resistance = (body.resistanceStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
    62→  const support = (body.supportStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
    63→  const atr = (body.atr as number) || 0;
    64→
    65→  const bbRange = bollingerUpper - bollingerLower;
    66→  const bbPosition = bbRange > 0 ? Math.min(100, Math.max(0, Math.round((price - bollingerLower) / bbRange * 100))) : 50;
    67→
    68→  const scenarios = body.scenarios as Record<string, { probability: number }> | undefined;
    69→  let dominantKey = 'R3';
    70→  let dominantProb = 0;
    71→  for (const k of ['R1', 'R2', 'R3', 'R4', 'R5'] as const) {
    72→    const p = scenarios?.[k]?.probability ?? 0;
    73→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
    74→  }
    75→
    76→  return {
    77→    price,
    78→    trend: trendDir === 'up' ? 'up' : trendDir === 'down' ? 'down' : 'range',
    79→    adx,
    80→    diPlus: (body.diPlus as number) || 0,
    81→    diMinus: (body.diMinus as number) || 0,
    82→    rsi,
    83→    stochK,
    84→    macdHist,
    85→    obv,
    86→    bbPosition,
    87→    resistance,
    88→    support,
    89→    atr,
    90→    scenarioDominant: dominantKey,
    91→    hasVolume: (body.hasVolume as boolean) ?? false,
    92→  };
    93→}

   324→// ─── Retry helper (FIXED: exponential backoff with jitter) ───────
   325→function sleep(ms: number) { return new Promise(r => setTimeout(r, ms)); }
   326→
   327→function jitter(base: number): number {
   328→  return base + Math.random() * base * 0.3;
   329→}
   330→
   331→async function aiCompletionWithRetry(
   332→  zai: Awaited<ReturnType<typeof getZai>>,
   333→  messages: { role: string; content: string }[],
   334→  maxRetries = 3,
   335→): Promise<string> {
   336→  for (let attempt = 1; attempt <= maxRetries; attempt++) {
   337→    try {
   338→      const completion = await zai.chat.completions.create({
   339→        messages,
   340→        thinking: { type: 'disabled' },
   341→      });
   342→      const content = completion.choices[0]?.message?.content;
   343→      if (!content || content.trim().length === 0) {
   344→        throw new Error('Empty AI response');
   345→      }
   346→      return content;
   347→    } catch (err: unknown) {
   348→      const msg = err instanceof Error ? err.message : String(err);
   349→      if (msg.includes('429') && attempt < maxRetries) {
   350→        const waitMs = Math.round(jitter(15000 * Math.pow(2, attempt - 1))); // 15s, 30s, 60s + jitter
   351→        console.warn(`[AI Analysis v5] 429 retry ${attempt}/${maxRetries}, waiting ${waitMs}ms...`);
   352→        await sleep(waitMs);
   353→        continue;
   354→      }
   355→      throw err;
   356→    }
   357→  }
   358→  throw new Error('Max retries exceeded');
   359→}
