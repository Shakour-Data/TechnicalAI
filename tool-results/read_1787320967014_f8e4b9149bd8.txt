   698→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   699→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   700→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   701→  grade?: string; overlapCount?: number; touchCount?: number;
   702→}) {
   703→  const isSupport = type === 'support';
   704→  const isTargetType = type === 'target';
   705→
   706→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   707→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   708→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   709→
   710→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   711→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   712→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   713→
   714→  return (
   715→    <div
   716→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   717→      style={{
   718→        background: bgAccent,
   719→        border: `1px solid ${borderAccent}`,
   720→        borderRight: `3px solid ${accentColor}`,
   721→      }}
   722→    >
   723→      {/* Label + Price */}
   724→      <div className="flex-1 min-w-0">
   725→        <div className="flex items-center gap-2">
   726→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   727→          {isTarget && <span className="text-[9px]">🎯</span>}
   728→          {overlapCount !== undefined && overlapCount > 0 && (
   729→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   730→              {toPersianDigits(overlapCount)} هم‌پوشانی
   731→            </span>
   732→          )}
   733→        </div>
   734→        <div className="flex items-baseline gap-1">
   735→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   736→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   737→        </div>
   738→      </div>
   739→
   740→      {/* Strength + Grade */}
   741→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   742→        <div className="flex items-center gap-1.5">
   743→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   744→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   745→          <span className="text-[9px] text-gray-500">/۱۰</span>
   746→        </div>
   747→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   748→          <div
   749→            className="h-full rounded-full transition-all duration-500"
   750→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   751→          />
   752→        </div>
   753→        {touchCount !== undefined && touchCount > 0 && (
   754→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   755→        )}
   756→      </div>
   757→    </div>
   758→  );
   759→}
