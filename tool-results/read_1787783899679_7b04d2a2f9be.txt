   248→// Determine the data source of an item
   249→function getSource(item: InstrumentItem): 'tse' | 'tgju' | 'yahoo' {
   250→  if (item.yahooSymbol || item.category === 'yahoo_stock' || item.category === 'yahoo_etf') return 'yahoo';

   252→  return 'tse';

   255→/* --─ Recent Searches ------------------------------------ */
   256→
   257→const RECENT_KEY = 'fin-search-recent';
   258→const MAX_RECENT = 8;
   259→
   260→function getRecentSearches(): string[] {
   261→  if (typeof window === 'undefined') return [];
   262→  try {
   263→    const raw = localStorage.getItem(RECENT_KEY);
   264→    return raw ? JSON.parse(raw) : [];
   265→  } catch { return []; }
   266→}
   267→
   268→function addRecentSearch(name: string) {
   269→  if (typeof window === 'undefined' || !name) return;
   270→  try {
   271→    let list = getRecentSearches();
   272→    list = list.filter((r) => r !== name);
   273→    list.unshift(name);
   274→    if (list.length > MAX_RECENT) list = list.slice(0, MAX_RECENT);
   275→    localStorage.setItem(RECENT_KEY, JSON.stringify(list));
   276→  } catch { /* ignore */ }
   277→}

   285→  compact = false,
   286→}: SymbolSearchProps) {

   289→  const [results, setResults] = React.useState<InstrumentItem[]>([]);

   297→  const [showIndustryPicker, setShowIndustryPicker] = React.useState(false);

   300→
   301→  const cacheRef = React.useRef<InstrumentsData | null>(null);
   302→  const tgjuCacheRef = React.useRef<TgjuData | null>(null);
   303→  const yahooCacheRef = React.useRef<YahooData | null>(null);
   304→  const fetchRef = React.useRef<Promise<InstrumentsData> | null>(null);
   305→  const tgjuFetchRef = React.useRef<Promise<TgjuData | null> | null>(null);
   306→  const yahooFetchRef = React.useRef<Promise<YahooData | null> | null>(null);
   307→  const inputRef = React.useRef<HTMLInputElement>(null);
   308→  const dropdownRef = React.useRef<HTMLDivElement>(null);

   316→    const stocks = activeCategory === 'yahoo_stock' ? yahooCacheRef.current.yahooStocks
   317→      : activeCategory === 'yahoo_etf' ? yahooCacheRef.current.yahooEtfs
   318→      : [...yahooCacheRef.current.yahooStocks, ...yahooCacheRef.current.yahooEtfs];

   326→    const stocks = activeCategory === 'yahoo_stock' ? yahooCacheRef.current.yahooStocks
   327→      : activeCategory === 'yahoo_etf' ? yahooCacheRef.current.yahooEtfs
   328→      : [...yahooCacheRef.current.yahooStocks, ...yahooCacheRef.current.yahooEtfs];

   332→
   333→  /* -- Data fetching -- */
   334→  const fetchData = React.useCallback(async (): Promise<InstrumentsData> => {
   335→    if (cacheRef.current) return cacheRef.current;
   336→    if (fetchRef.current) return fetchRef.current;
   337→    const p = async () => {
   338→      try {
   339→        const r = await fetch('/api/instruments');
   340→        if (!r.ok) throw new Error(`${r.status}`);
   341→        const d: InstrumentsData = await r.json();
   342→        cacheRef.current = d;
   343→        return d;
   344→      } finally {
   345→        fetchRef.current = null;
   346→      }
   347→    };
   348→    fetchRef.current = p();
   349→    return fetchRef.current;
   350→  }, []);
   351→
   352→  const fetchTgjuData = React.useCallback(async (): Promise<TgjuData | null> => {
   353→    if (tgjuCacheRef.current) return tgjuCacheRef.current;
   354→    if (tgjuFetchRef.current) return tgjuFetchRef.current;
   355→    const p = async () => {
   356→      try {
   357→        const r = await fetch('/api/tgju-instruments');
   358→        if (!r.ok) return null;
   359→        const d: TgjuData = await r.json();
   360→        tgjuCacheRef.current = d;
   361→        return d;
   362→      } catch { return null; } finally { tgjuFetchRef.current = null; }
   363→    };
   364→    tgjuFetchRef.current = p();
   365→    return p();
   366→  }, []);
   367→
   368→  const fetchYahooData = React.useCallback(async (): Promise<YahooData | null> => {
   369→    if (yahooCacheRef.current) return yahooCacheRef.current;
   370→    if (yahooFetchRef.current) return yahooFetchRef.current;
   371→    const p = async () => {
   372→      try {
   373→        const r = await fetch('/api/yahoo-instruments');
   374→        if (!r.ok) return null;
   375→        const d = await r.json() as YahooData;
   376→        yahooCacheRef.current = d;
   377→        return d;
   378→      } catch { return null; } finally { yahooFetchRef.current = null; }
   379→    };
   380→    yahooFetchRef.current = p();
   381→    return p();
   382→  }, []);
   383→
   384→  /* -- Get all items for current category (with sub-filters) -- */
   385→  const getItemsForCategory = React.useCallback(async (cat: CategoryKey): Promise<InstrumentItem[]> => {
   386→    if (cat === 'yahoo_stock') {
   387→      const yahoo = yahooCacheRef.current || await fetchYahooData();
   388→      let items = yahoo?.yahooStocks || [];
   389→      if (activeCountry) items = items.filter((s) => s.country === activeCountry);
   390→      if (activeSector) items = items.filter((s) => s.sector === activeSector);
   391→      return items;
   392→    }
   393→    if (cat === 'yahoo_etf') {
   394→      const yahoo = yahooCacheRef.current || await fetchYahooData();
   395→      let items = yahoo?.yahooEtfs || [];
   396→      if (activeCountry) items = items.filter((s) => s.country === activeCountry);
   397→      return items;
   398→    }
   399→
   400→    if (TGJU_CATEGORIES.has(cat)) {
   401→      const tgju = tgjuCacheRef.current || await fetchTgjuData();
   402→      const yahoo = yahooCacheRef.current || await fetchYahooData();
   403→      if (!tgju && !yahoo) return [];
   404→      let items: InstrumentItem[] = [];
   405→      switch (cat) {
   406→        case 'currency': items = tgju?.currencies || []; break;
   407→        case 'gold': items = [...(tgju?.gold || []), ...(tgju?.silver || [])]; break;
   408→        case 'gold_etf': items = tgju?.goldEtfs || []; break;
   409→        case 'crypto': items = [...(tgju?.crypto || []), ...(yahoo?.yahooCrypto || [])]; break;
   410→        case 'world_index': items = [...(tgju?.worldIndices || []), ...(tgju?.foreignStocks || []), ...(yahoo?.yahooIndices || [])]; break;
   411→        case 'forex': items = [...(tgju?.forex || []), ...(yahoo?.yahooForex || [])]; break;
   412→        case 'energy': items = [...(tgju?.energy || []), ...(yahoo?.yahooEnergy || [])]; break;
   413→        case 'metal': items = [...(tgju?.metals || []), ...(yahoo?.yahooMetals || [])]; break;
   414→        case 'commodity': items = [...(tgju?.commodities || []), ...(yahoo?.yahooCommodities || [])]; break;
   415→        default: items = [];
   416→      }
   417→      return items;
   418→    }
   419→
   420→    // TSE categories
   421→    const data = cacheRef.current || await fetchData();
   422→    switch (cat) {
   423→      case 'indices': return data.indices;
   424→      case 'stocks':
   425→        return activeIndustry
   426→          ? data.stocks.filter((s) => s.cs === activeIndustry)
   427→          : data.stocks;
   428→      case 'etf': return data.etfs;
   429→      case 'bond': return [...data.bonds, ...data.mortgage];
   430→      case 'derivative': return [...data.futures, ...data.salaf];
   431→      case 'all':
   432→      default: {

   472→      // Build searchable text from all available fields
   473→      const fields = [

   475→        item.l30,                    // English name + symbol (e.g., "Apple (AAPL)")

   483→      ];
   484→
   485→      const fullText = fields.join(' ').toLowerCase();
   486→      let score = 0;
   487→
   488→      // Score: exact matches get higher score
   489→      for (const term of terms) {
   490→        if (!term) continue;
   491→        // Check Persian name
   492→        if (item.l18.toLowerCase().includes(term)) score += 10;
   493→        // Check English name
   494→        if (item.nameEn?.toLowerCase().includes(term)) score += 8;
   495→        // Check Yahoo symbol (exact or partial)
   496→        if (item.yahooSymbol?.toLowerCase() === term) score += 20; // exact ticker
   497→        else if (item.yahooSymbol?.toLowerCase().includes(term)) score += 15;
   498→        // Check l30 (English name + symbol)
   499→        if (item.l30.toLowerCase().includes(term)) score += 7;
   500→        // Check exchange
   501→        if (item.exchange?.toLowerCase().includes(term)) score += 5;
   502→        // Check country
   503→        if (item.country?.toLowerCase().includes(term)) score += 6;
   504→        if (item.countryEn?.toLowerCase() === term) score += 15;
   505→        else if (item.countryEn?.toLowerCase().includes(term)) score += 4;
   506→        // Check sector
   507→        if (item.sector?.toLowerCase().includes(term)) score += 5;
   508→        // Check Persian sector/group
   509→        if (item.cs?.toLowerCase().includes(term)) score += 6;
   510→
   511→        // If none matched, reject
   512→        if (
   513→          !item.l18.toLowerCase().includes(term) &&
   514→          !item.l30.toLowerCase().includes(term) &&
   515→          !item.cs?.toLowerCase().includes(term) &&
   516→          !(item.yahooSymbol?.toLowerCase().includes(term)) &&

   521→          !(item.sector?.toLowerCase().includes(term))
   522→        ) {
   523→          return null; // no match
   524→        }
   525→      }
   526→      return score;
   527→    };
   528→
   529→    const scored = items
   530→      .map((item) => ({ item, score: scoreAndFilter(item) }))
   531→      .filter((r): r is { item: InstrumentItem; score: number } => r.score !== null && r.score > 0)
