   353→   353→/**
   354→   354→ * Parse a number that might contain Persian digits, commas, or other formatting.
   355→   355→ * Handles null/undefined/empty strings gracefully.
   356→   356→ */
   357→   357→function parsePersianNum(str: string | null | undefined | number): number {
   358→   358→  if (str === null || str === undefined) return 0;
   359→   359→  let s = String(str).trim();
   360→   360→  if (!s || s === '-' || s === '') return 0;
   361→   361→  // Strip trailing % sign (e.g. "0.05%" → "0.05")
   362→   362→  if (s.endsWith('%')) s = s.slice(0, -1).trim();
   363→   363→  // Replace commas and Persian digits
   364→   364→  const cleaned = s.replace(/[,۰-۹]/g, (c: string) => {
   365→   365→    const persianDigits = '۰۱۲۳۴۵۶۷۸۹';
   366→   366→    const idx = persianDigits.indexOf(c);
   367→   367→    return idx >= 0 ? String(idx) : c === ',' ? '' : c;
   368→   368→  });
   369→   369→  const n = Number(cleaned);
   370→   370→  return isFinite(n) ? n : 0;
   371→   371→}

   373→   373→/**
   374→   374→ * Check if a value looks like a date string (contains / or - and is not a number).
   375→   375→ */
   376→   376→function looksLikeDate(val: string | null | undefined | number): boolean {
   377→   377→  if (!val) return false;
   378→   378→  const s = String(val).trim();
   379→   379→  return /[\/\-]/.test(s) && isNaN(Number(s.replace(/[,۰-۹\/\-]/g, '')));
   380→   380→}
   381→   381→
   382→   382→/**
   383→   383→ * Check if a value looks like a percentage (ends with % or is small like 0.5 to 50).
   384→   384→ */
   385→   385→function looksLikePercentage(val: string | null | undefined | number): boolean {
   386→   386→  if (!val) return false;
   387→   387→  const s = String(val).trim();
   388→   388→  if (s.endsWith('%')) return true;
   389→   389→  const n = parsePersianNum(val);
   390→   390→  return !isNaN(n) && Math.abs(n) < 100 && n !== 0;
   391→   391→}
   392→   392→
   393→   393→/**
   394→   394→ * Auto-detect column mapping for TGJU API response.
   395→   395→ * Expected format: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   396→   396→ * But the API may change column order.
   397→   397→ */
   398→   398→function detectColumnMap(row: (string | number)[]): {
   399→   399→  openIdx: number; highIdx: number; lowIdx: number; closeIdx: number;
   400→   400→  jalaliDateIdx: number; gregorianDateIdx: number;
   401→   401→} | null {
   402→   402→  if (!row || row.length < 6) return null;
   403→   403→
   404→   404→  let jalaliDateIdx = -1;
   405→   405→  let gregorianDateIdx = -1;
   406→   406→  const numericIndices: number[] = [];
   407→   407→  const percentageIndices: number[] = [];
   408→   408→
   409→   409→  for (let i = 0; i < row.length; i++) {
   410→   410→    const val = row[i];
   411→   411→    if (looksLikeDate(val)) {
   412→   412→      // Jalali dates have Persian digits or format like 1404/05/25
   413→   413→      const s = String(val);
   414→   414→      if (/[۰-۹]/.test(s) || /^14\d{2}/.test(s.replace(/[۰-۹]/g, (d: string) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d))))) {
   415→   415→        if (jalaliDateIdx === -1) jalaliDateIdx = i;
   416→   416→      } else {
   417→   417→        if (gregorianDateIdx === -1) gregorianDateIdx = i;
   418→   418→      }
   419→   419→    } else if (looksLikePercentage(val)) {
   420→   420→      percentageIndices.push(i);
   421→   421→    } else {
   422→   422→      const n = parsePersianNum(val);
   423→   423→      if (isFinite(n) && n !== 0) {
   424→   424→        numericIndices.push(i);
   425→   425→      }
   426→   426→    }
   427→   427→  }
   428→   428→
   429→   429→  // If we couldn't auto-detect dates, fall back to assumed positions
   430→   430→  if (jalaliDateIdx === -1 && row.length >= 8) {
   431→   431→    // Default assumption: row[6] = gregorian, row[7] = jalali
   432→   432→    if (looksLikeDate(row[7])) jalaliDateIdx = 7;
   433→   433→    else if (looksLikeDate(row[6])) jalaliDateIdx = 6;
   434→   434→    if (gregorianDateIdx === -1 && looksLikeDate(row[6]) && jalaliDateIdx !== 6) gregorianDateIdx = 6;
   435→   435→    else if (gregorianDateIdx === -1 && looksLikeDate(row[7]) && jalaliDateIdx !== 7) gregorianDateIdx = 7;
   436→   436→  }
   437→   437→
   438→   438→  // Numeric columns (excluding percentages) should be OHLC + change
   439→   439→  // OHLC: typically 4 largest absolute values among non-percentage numerics
   440→   440→  // change: the remaining one
   441→   441→  const ohlcCandidates = numericIndices.filter(i => !percentageIndices.includes(i));
   442→   442→
   443→   443→  if (ohlcCandidates.length < 4) {
   444→   444→    // Fallback: assume first 4 numeric columns are OHLC
   445→   445→    if (numericIndices.length >= 4) {
   446→   446→      return {
   447→   447→        openIdx: numericIndices[0],
   448→   448→        highIdx: numericIndices[2],
   449→   449→        lowIdx: numericIndices[1],
   450→   450→        closeIdx: numericIndices[3],
   451→   451→        jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : 7,
   452→   452→        gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : 6,
   453→   453→      };
   454→   454→    }
   455→   455→    return null;
   456→   456→  }
   457→   457→
   458→   458→  // Sort by absolute value descending to identify OHLC (largest) vs change (smallest)
   459→   459→  const sorted = ohlcCandidates
   460→   460→    .map(i => ({ idx: i, val: Math.abs(parsePersianNum(row[i])) }))
   461→   461→    .sort((a, b) => b.val - a.val);
   462→   462→
   463→   463→  // The 4 largest should be OHLC. Among them, identify high (max) and low (min).
   464→   464→  // open and close are the other two.
   465→   465→  const top4 = sorted.slice(0, 4);
   466→   466→  const maxIdx = top4.reduce((a, b) => a.val > b.val ? a : b).idx;
   467→   467→  const minIdx = top4.reduce((a, b) => a.val < b.val ? a : b).idx;
   468→   468→  const others = top4.filter(x => x.idx !== maxIdx && x.idx !== minIdx);
   469→   469→
   470→   470→  // Among the two remaining, the one closer to high is open, the other is close
   471→   471→  // (heuristic: in an uptrend open < close, but we can't assume direction)
   472→   472→  // Default: first = open, second = close
   473→   473→  const openIdx = others[0]?.idx ?? ohlcCandidates[0];
   474→   474→  const closeIdx = others[1]?.idx ?? ohlcCandidates[3];
   475→   475→
   476→   476→  return {
   477→   477→    openIdx,
   478→   478→    highIdx: maxIdx,
   479→   479→    lowIdx: minIdx,
   480→   480→    closeIdx,
   481→   481→    jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : row.length - 1,
   482→   482→    gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : row.length - 2,
   483→   483→  };
   484→   484→}

   486→   486→/**
   487→   487→ * Validate a candle's OHLC values for sanity.
   488→   488→ * Returns true if the candle is valid.
   489→   489→ */
   490→   490→function isValidCandle(o: number, h: number, l: number, c: number): boolean {
   491→   491→  if (!isFinite(o) || !isFinite(h) || !isFinite(l) || !isFinite(c)) return false;
   492→   492→  if (o <= 0 || h <= 0 || l <= 0 || c <= 0) return false;
   493→   493→  // High must be >= Low, Open, Close
   494→   494→  if (h < l || h < o || h < c) return false;
   495→   495→  // Low must be <= High, Open, Close
   496→   496→  if (l > o || l > c) return false;
   497→   497→  // No infinite or extreme values
   498→   498→  if (o > 1e15 || h > 1e15 || l > 1e15 || c > 1e15) return false;
   499→   499→  return true;
   500→   500→}

   938→   938→/**
   939→   939→ * Fetch TGJU historical data via queued page_reader with retry and rate limiting.
   940→   940→ * This prevents 429 errors by serializing requests and adding delays between calls.
   941→   941→ */
   942→   942→async function fetchViaQueuedPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   943→   943→  // Enqueue this request — ensures only one page_reader call at a time
   944→   944→  let result: TgjuOHLC[] | null = null;
   945→   945→  const taskPromise = (async () => {
   946→   946→    // Rate limit: wait if last call was too recent
   947→   947→    const timeSinceLastCall = Date.now() - lastPageReaderCall;
   948→   948→    if (timeSinceLastCall < PAGE_READER_MIN_INTERVAL) {
   949→   949→      const waitTime = PAGE_READER_MIN_INTERVAL - timeSinceLastCall;
   950→   950→      console.log(`[TGJU] Rate limiting page_reader: waiting ${waitTime}ms...`);
   951→   951→      await new Promise(r => setTimeout(r, waitTime));
   952→   952→    }
   953→   953→
   954→   954→    // Retry with exponential backoff
   955→   955→    for (let attempt = 1; attempt <= PAGE_READER_MAX_RETRIES; attempt++) {
   956→   956→      try {
   957→   957→        lastPageReaderCall = Date.now();
   958→   958→        const fetchResult = await fetchTgjuHistoryViaPageReader(tgjuKey);
   959→   959→        if (fetchResult && fetchResult.length >= 30) {
   960→   960→          result = fetchResult;
   961→   961→          return; // success
   962→   962→        }
   963→   963→        // Got data but not enough candles — don't retry
   964→   964→        if (fetchResult && fetchResult.length > 0) {
   965→   965→          console.warn(`[TGJU] page_reader returned only ${fetchResult.length} candles for ${tgjuKey}`);
   966→   966→          return;
   967→   967→        }
   968→   968→      } catch (err) {
   969→   969→        const errMsg = err instanceof Error ? err.message : String(err);
   970→   970→        const isRateLimit = errMsg.includes('429') || errMsg.includes('Too many requests');
   971→   971→        if (isRateLimit && attempt < PAGE_READER_MAX_RETRIES) {
   972→   972→          const delay = PAGE_READER_RETRY_BASE_DELAY * Math.pow(3, attempt - 1); // 3s, 9s, 27s
   973→   973→          console.warn(`[TGJU] page_reader 429 for ${tgjuKey}, retry ${attempt}/${PAGE_READER_MAX_RETRIES} in ${delay}ms...`);
   974→   974→          await new Promise(r => setTimeout(r, delay));
   975→   975→          continue;
   976→   976→        }
   977→   977→        console.error(`[TGJU] page_reader error for ${tgjuKey} (attempt ${attempt}):`, errMsg);
   978→   978→        return;
   979→   979→      }
   980→   980→    }
   981→   981→  })();
   982→   982→
   983→   983→  // Chain onto the queue
   984→   984→  pageReaderQueue = pageReaderQueue.then(() => taskPromise, () => taskPromise);
   985→   985→  await pageReaderQueue;
   986→   986→  return result;
   987→   987→}

   994→   994→async function fetchTgjuHistoryViaPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   995→   995→  const apiUrl = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=365`;

  1000→  1000→  const invokeResult = await zai.functions.invoke('page_reader', { url: apiUrl });
  1001→  1001→
  1002→  1002→  if (invokeResult.code !== 200 || !invokeResult.data?.html) {
  1003→  1003→    const errMsg = invokeResult.error || `code=${invokeResult.code}`;

  1007→  1007→  // Extract JSON from <pre> tag in the HTML
  1008→  1008→  const html = invokeResult.data.html;
  1009→  1009→  const preMatch = html.match(/<pre[^>]*>([\s\S]*?)<\/pre>/);
  1010→  1010→  if (!preMatch) {
  1011→  1011→    throw new Error('no <pre> block found in page_reader response');
  1012→  1012→  }
  1013→  1013→
  1014→  1014→  // Unescape HTML entities and strip HTML tags from values
  1015→  1015→  const jsonStr = preMatch[1]
  1016→  1016→    .replace(/<[^>]*>/g, '')        // Strip ALL HTML tags first
  1017→  1017→    .replace(/&quot;/g, '"')
  1018→  1018→    .replace(/&amp;/g, '&')
  1019→  1019→    .replace(/&lt;/g, '<')
  1020→  1020→    .replace(/&gt;/g, '>')
  1021→  1021→    .replace(/\s+/g, ' ')           // Normalize whitespace
  1022→  1022→    .trim();
  1023→  1023→
  1024→  1024→  if (!jsonStr.startsWith('{')) {
  1025→  1025→    throw new Error('extracted content is not JSON');
  1026→  1026→  }
  1027→  1027→
  1028→  1028→  return parseTgjuChartData(tgjuKey, jsonStr);
  1029→  1029→}

  1031→  1031→/**
  1032→  1032→ * Parse the JSON data from TGJU chart API response.
  1033→  1033→ * Used by both direct fetch and page_reader paths.
  1034→  1034→ */
  1035→  1035→function parseTgjuChartData(tgjuKey: string, jsonStr: string): TgjuOHLC[] | null {
  1036→  1036→  try {
  1037→  1037→    const parsed = JSON.parse(jsonStr);
  1038→  1038→    const rawRows: (string | number)[][] = parsed.data;
  1039→  1039→
  1040→  1040→    if (!rawRows || rawRows.length === 0) {
  1041→  1041→      console.warn(`[TGJU] Empty data array for ${tgjuKey}`);
  1042→  1042→      return null;
  1043→  1043→    }
  1044→  1044→
  1045→  1045→    // Sanitize row values: strip any residual HTML tags and trim
  1046→  1046→    const rows = rawRows.map(row =>
  1047→  1047→      row.map(cell => {
  1048→  1048→        const s = String(cell).trim();
  1049→  1049→        // Strip HTML tags if any remain (safety for direct fetch path)
  1050→  1050→        return s.replace(/<[^>]*>/g, '').replace(/\s+/g, ' ');
  1051→  1051→      })
  1052→  1052→    );
  1053→  1053→
  1054→  1054→    // ── Auto-detect column mapping from first valid row ──
  1055→  1055→    let colMap = detectColumnMap(rows[0]);
  1056→  1056→    if (!colMap) {
  1057→  1057→      for (let r = 1; r < Math.min(5, rows.length); r++) {
  1058→  1058→        colMap = detectColumnMap(rows[r]);
  1059→  1059→        if (colMap) break;
  1060→  1060→      }
  1061→  1061→    }
  1062→  1062→    if (!colMap) {
  1063→  1063→      console.warn(`[TGJU] Could not auto-detect columns for ${tgjuKey}, using default order`);
  1064→  1064→      colMap = { openIdx: 0, highIdx: 2, lowIdx: 1, closeIdx: 3, jalaliDateIdx: 7, gregorianDateIdx: 6 };
  1065→  1065→    } else {
  1066→  1066→      console.log(`[TGJU] Column map for ${tgjuKey}: O=${colMap.openIdx} H=${colMap.highIdx} L=${colMap.lowIdx} C=${colMap.closeIdx} Date=${colMap.jalaliDateIdx}`);
  1067→  1067→    }
  1068→  1068→
  1069→  1069→    const { openIdx, highIdx, lowIdx, closeIdx, jalaliDateIdx, gregorianDateIdx } = colMap;
  1070→  1070→
  1071→  1071→    // ── Parse rows ──
  1072→  1072→    const candles: TgjuOHLC[] = [];
  1073→  1073→    let invalidCount = 0;
  1074→  1074→
  1075→  1075→    for (const row of rows) {
  1076→  1076→      const rawDate = String(row[jalaliDateIdx] || row[gregorianDateIdx] || '');
  1077→  1077→      const date = rawDate.replace(/[\/]/g, '-');
  1078→  1078→      const o = parsePersianNum(row[openIdx]);
  1079→  1079→      const h = parsePersianNum(row[highIdx]);
  1080→  1080→      const l = parsePersianNum(row[lowIdx]);
  1081→  1081→      const c = parsePersianNum(row[closeIdx]);
  1082→  1082→
  1083→  1083→      if (!isValidCandle(o, h, l, c) || date.length === 0) {
  1084→  1084→        invalidCount++;
  1085→  1085→        continue;
  1086→  1086→      }
  1087→  1087→
  1088→  1088→      candles.push({ date, open: o, high: h, low: l, close: c });
  1089→  1089→    }
  1090→  1090→
  1091→  1091→    if (invalidCount > 0) {
  1092→  1092→      console.warn(`[TGJU] ${tgjuKey}: ${invalidCount}/${rows.length} invalid candles skipped`);
  1093→  1093→    }
  1094→  1094→
  1095→  1095→    if (candles.length === 0) {
  1096→  1096→      return null;
  1097→  1097→    }
  1098→  1098→
  1099→  1099→    // ── Detect and fix data drift ──
  1100→  1100→    let driftCount = 0;
  1101→  1101→    for (let i = 1; i < candles.length; i++) {
  1102→  1102→      const prevClose = candles[i - 1].close;
  1103→  1103→      const currClose = candles[i].close;
  1104→  1104→      if (prevClose > 0) {
  1105→  1105→        const pctChange = Math.abs(currClose - prevClose) / prevClose;
  1106→  1106→        if (pctChange > 0.5) {
  1107→  1107→          driftCount++;
  1108→  1108→        }
  1109→  1109→      }
  1110→  1110→    }
  1111→  1111→    if (driftCount > candles.length * 0.1) {
  1112→  1112→      console.warn(`[TGJU] ${tgjuKey}: Possible data drift detected (${driftCount} suspicious jumps out of ${candles.length} candles)`);
  1113→  1113→    }
  1114→  1114→
  1115→  1115→    // Reverse: API returns oldest-first (asc), we want newest-last for charting
  1116→  1116→    const firstDate = candles[0].date;
  1117→  1117→    const lastDate = candles[candles.length - 1].date;
  1118→  1118→    if (firstDate > lastDate) {
  1119→  1119→      candles.reverse();
  1120→  1120→    }
  1121→  1121→
  1122→  1122→    console.log(`[TGJU] ${tgjuKey}: Parsed ${candles.length} candles (${candles[0]?.date} to ${candles[candles.length - 1]?.date})`);
  1123→  1123→
  1124→  1124→    return candles;
  1125→  1125→  } catch (err) {
  1126→  1126→    console.error(`[TGJU] Failed to parse chart data for ${tgjuKey}:`, err);
  1127→  1127→    return null;
  1128→  1128→  }
  1129→  1129→}

  1148→  1148→      ts: now,
  1149→  1149→    });
  1150→  1150→

  1152→  1152→    if (liveInstrumentsCache.data) {

  1157→  1157→        inst.lowPrice = last.low;
