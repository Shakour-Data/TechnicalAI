   168→   168→export default function SymbolSearch({
   169→   169→  onSelect,
   170→   170→  placeholder = 'جستجوی ابزار مالی ...',
   171→   171→  className,
   172→   172→}: SymbolSearchProps) {
   173→   173→  const [query, setQuery] = React.useState('');
   174→   174→  const [open, setOpen] = React.useState(false);
   175→   175→  const [results, setResults] = React.useState<InstrumentItem[]>([]);
   176→   176→  const [loading, setLoading] = React.useState(false);
   177→   177→  const [activeIndex, setActiveIndex] = React.useState(-1);
   178→   178→  const [totalMatched, setTotalMatched] = React.useState(0);
   179→   179→  const [activeCategory, setActiveCategory] = React.useState<CategoryKey>('all');
   180→   180→  const [activeIndustry, setActiveIndustry] = React.useState<string | null>(null);
   181→   181→  const [showIndustryPicker, setShowIndustryPicker] = React.useState(false);
   182→   182→  const [indexSubFilter, setIndexSubFilter] = React.useState<'all' | 'main' | 'sector'>('all');
   183→   183→
   184→   184→  const cacheRef = React.useRef<InstrumentsData | null>(null);
   185→   185→  const tgjuCacheRef = React.useRef<TgjuData | null>(null);
   186→   186→  const fetchRef = React.useRef<Promise<InstrumentsData> | null>(null);
   187→   187→  const tgjuFetchRef = React.useRef<Promise<TgjuData | null> | null>(null);
   188→   188→  const inputRef = React.useRef<HTMLInputElement>(null);
   189→   189→  const dropdownRef = React.useRef<HTMLDivElement>(null);
   190→   190→  const itemRefs = React.useRef<(HTMLDivElement | null)[]>([]);
   191→   191→  const industryScrollRef = React.useRef<HTMLDivElement>(null);
   192→   192→
   193→   193→  /* -- Data fetching -- */
   194→   194→  const fetchData = React.useCallback(async (): Promise<InstrumentsData> => {
   195→   195→    if (cacheRef.current) return cacheRef.current;
   196→   196→    if (fetchRef.current) return fetchRef.current;
   197→   197→    const p = async () => {
   198→   198→      try {
   199→   199→        const r = await fetch('/api/instruments');
   200→   200→        if (!r.ok) { console.warn('[SymbolSearch] /api/instruments failed:', r.status); return { indices: [], stocks: [], etfs: [], bonds: [], futures: [], salaf: [], mortgage: [], industries: [] }; }
   201→   201→        const d: InstrumentsData = await r.json();
   202→   202→        cacheRef.current = d;
   203→   203→        return d;
   204→   204→      } catch (e) {
   205→   205→        console.warn('[SymbolSearch] fetchData error:', e);
   206→   206→        return { indices: [], stocks: [], etfs: [], bonds: [], futures: [], salaf: [], mortgage: [], industries: [] };
   207→   207→      } finally {
   208→   208→        fetchRef.current = null;
   209→   209→      }
   210→   210→    };
   211→   211→    fetchRef.current = p();
   212→   212→    return fetchRef.current;
   213→   213→  }, []);
   214→   214→
   215→   215→  const fetchTgjuData = React.useCallback(async (): Promise<TgjuData | null> => {
   216→   216→    if (tgjuCacheRef.current) return tgjuCacheRef.current;
   217→   217→    if (tgjuFetchRef.current) return tgjuFetchRef.current;
   218→   218→    const p = async () => {
   219→   219→      try {
   220→   220→        const r = await fetch('/api/tgju-instruments');
   221→   221→        if (!r.ok) return null;
   222→   222→        const d: TgjuData = await r.json();
   223→   223→        tgjuCacheRef.current = d;
   224→   224→        return d;
   225→   225→      } catch {
   226→   226→        return null;
   227→   227→      } finally {
   228→   228→        tgjuFetchRef.current = null;
   229→   229→      }
   230→   230→    };
   231→   231→    tgjuFetchRef.current = p();
   232→   232→    return p();
   233→   233→  }, []);
   234→   234→
   235→   235→  /* -- Check if category is TGJU-based -- */
   236→   236→  const isTgjuCategory = (cat: CategoryKey) => TGJU_CATEGORIES.has(cat);
   237→   237→
   238→   238→  /* -- Get all items for current category -- */
   239→   239→  const getItemsForCategory = React.useCallback(async (cat: CategoryKey): Promise<InstrumentItem[]> => {
   240→   240→    if (isTgjuCategory(cat)) {
   241→   241→      const tgju = tgjuCacheRef.current || await fetchTgjuData();
   242→   242→      if (!tgju) return [];
   243→   243→      switch (cat) {
   244→   244→        case 'currency': return tgju.currencies;
   245→   245→        case 'gold': return [...tgju.gold, ...tgju.silver];
   246→   246→        case 'gold_etf': return tgju.goldEtfs;
   247→   247→        case 'silver': return tgju.silver;
   248→   248→        case 'crypto': return tgju.crypto;
   249→   249→        case 'world_index': return tgju.worldIndices;
   250→   250→        case 'foreign_stock': return tgju.foreignStocks;
   251→   251→        case 'forex': return tgju.forex;
   252→   252→        case 'energy': return tgju.energy;
   253→   253→        case 'metal': return tgju.metals;
   254→   254→        case 'commodity': return tgju.commodities;
   255→   255→        default: return [];
   256→   256→      }
   257→   257→    }
   258→   258→
   259→   259→    // TSE categories
   260→   260→    const data = cacheRef.current || await fetchData();
   261→   261→    switch (cat) {
   262→   262→      case 'indices': {
   263→   263→        let idxItems = data.indices;
   264→   264→        if (indexSubFilter === 'main') {
   265→   265→          idxItems = idxItems.filter((i) => i.cs === 'شاخص اصلی');
   266→   266→        } else if (indexSubFilter === 'sector') {
   267→   267→          idxItems = idxItems.filter((i) => i.cs === 'صنعت');
   268→   268→        }
   269→   269→        return idxItems;
   270→   270→      }
   271→   271→      case 'stocks':
   272→   272→        return activeIndustry
   273→   273→          ? data.stocks.filter((s) => s.cs === activeIndustry)
   274→   274→          : data.stocks;
   275→   275→      case 'etf': return data.etfs;
   276→   276→      case 'bond': return [...data.bonds, ...data.mortgage];
   277→   277→      case 'derivative': return [...data.futures, ...data.salaf];
   278→   278→      case 'all':
   279→   279→      default: {
   280→   280→        const tgju = tgjuCacheRef.current;
   281→   281→        return [
   282→   282→          ...data.indices,
   283→   283→          ...data.stocks,
   284→   284→          ...data.etfs,
   285→   285→          ...(tgju?.currencies || []),
   286→   286→          ...(tgju?.forex || []),
   287→   287→          ...(tgju?.crypto || []),
   288→   288→          ...(tgju?.gold || []),
   289→   289→          ...(tgju?.silver || []),
   290→   290→          ...(tgju?.goldEtfs || []),
   291→   291→          ...(tgju?.worldIndices || []),
   292→   292→          ...(tgju?.foreignStocks || []),
   293→   293→          ...(tgju?.energy || []),
   294→   294→          ...(tgju?.metals || []),
   295→   295→          ...(tgju?.commodities || []),
   296→   296→          ...data.bonds,
   297→   297→          ...data.futures,
   298→   298→          ...data.salaf,
   299→   299→          ...data.mortgage,
   300→   300→        ];
   301→   301→      }
   302→   302→    }
   303→   303→  }, [activeIndustry, indexSubFilter, fetchData, fetchTgjuData]);
   304→   304→
   305→   305→  /* -- Popular items -- */
   306→   306→  const getPopular = React.useCallback(async () => {
   307→   307→    const data = cacheRef.current || await fetchData();
   308→   308→    const tgju = tgjuCacheRef.current;
   309→   309→    const all = [
   310→   310→      ...(data.indices.map((i) => ({ ...i, tval: i.index || 0 }))),
   311→   311→      ...data.stocks,
   312→   312→      ...data.etfs,
   313→   313→      ...(tgju?.currencies || []),
   314→   314→      ...(tgju?.crypto?.slice(0, 5) || []),
   315→   315→      ...(tgju?.foreignStocks?.slice(0, 3) || []),
   316→   316→      ...(tgju?.gold?.slice(0, 3) || []),
   317→   317→    ];
   318→   318→    const sorted = [...all].sort((a, b) => (b.tval || 0) - (a.tval || 0)).slice(0, 15);
   319→   319→    setResults(sorted);
   320→   320→    setTotalMatched(all.length);
   321→   321→  }, [fetchData]);
   322→   322→
   323→   323→  /* -- Filter by search query -- */
   324→   324→  const doFilter = React.useCallback((items: InstrumentItem[], q: string) => {
   325→   325→    const n = q.trim().toLowerCase();
   326→   326→    const f = items.filter(
   327→   327→      (s) =>
   328→   328→        s.l18?.toLowerCase().includes(n) ||
   329→   329→        s.l30?.toLowerCase().includes(n) ||
   330→   330→        s.cs?.toLowerCase().includes(n),
   331→   331→    );
   332→   332→    setTotalMatched(f.length);
   333→   333→    setResults(f.slice(0, MAX_RESULTS));
   334→   334→    setActiveIndex(-1);
   335→   335→  }, []);
   336→   336→
   337→   337→  /* -- Show dropdown with data -- */
   338→   338→  const showDropdown = React.useCallback(async (q: string) => {
   339→   339→    setOpen(true);
   340→   340→    await Promise.allSettled([fetchData(), fetchTgjuData()]);
   341→   341→    const items = await getItemsForCategory(activeCategory);
   342→   342→    if (q.trim().length === 0) {
   343→   343→      if (activeCategory === 'all') {
   344→   344→        await getPopular();
   345→   345→      } else {
   346→   346→        setResults(items.slice(0, MAX_RESULTS));
   347→   347→        setTotalMatched(items.length);
   348→   348→      }
   349→   349→    } else {
   350→   350→      doFilter(items, q);
   351→   351→    }
   352→   352→    setLoading(false);
   353→   353→  }, [fetchData, fetchTgjuData, getItemsForCategory, activeCategory, getPopular, doFilter]);
   354→   354→
   355→   355→  /* -- Category change -- */
   356→   356→  const handleCategoryChange = React.useCallback(async (cat: CategoryKey) => {
   357→   357→    setActiveCategory(cat);
   358→   358→    if (cat !== 'stocks') setActiveIndustry(null);
   359→   359→    if (cat !== 'indices') setIndexSubFilter('all');
   360→   360→    setShowIndustryPicker(false);
   361→   361→
   362→   362→    const items = await getItemsForCategory(cat);
   363→   363→    if (query.trim().length > 0) {
   364→   364→      doFilter(items, query);
   365→   365→    } else if (cat === 'all') {
   366→   366→      await getPopular();
   367→   367→    } else {
   368→   368→      setResults(items.slice(0, MAX_RESULTS));
   369→   369→      setTotalMatched(items.length);
   370→   370→    }
   371→   371→    setActiveIndex(-1);
   372→   372→  }, [getItemsForCategory, query, doFilter, getPopular]);
   373→   373→
   374→   374→  /* -- Industry change -- */
   375→   375→  const handleIndustryChange = React.useCallback((industry: string | null) => {
   376→   376→    setActiveIndustry(industry);
   377→   377→    setShowIndustryPicker(false);
   378→   378→    if (cacheRef.current) {
   379→   379→      const items = industry
   380→   380→        ? cacheRef.current.stocks.filter((s) => s.cs === industry)
   381→   381→        : cacheRef.current.stocks;
   382→   382→      if (query.trim().length > 0) {
   383→   383→        doFilter(items, query);
   384→   384→      } else {
   385→   385→        setResults(items.slice(0, MAX_RESULTS));
   386→   386→        setTotalMatched(items.length);
   387→   387→      }
   388→   388→      setActiveIndex(-1);
   389→   389→    }
   390→   390→  }, [query, doFilter]);
   391→   391→
   392→   392→  const handleChange = React.useCallback((value: string) => {
   393→   393→    setQuery(value);
   394→   394→    getItemsForCategory(activeCategory).then((items) => {
   395→   395→      if (value.trim().length === 0) {
   396→   396→        if (activeCategory === 'all') getPopular();
   397→   397→        else {
   398→   398→          setResults(items.slice(0, MAX_RESULTS));
   399→   399→          setTotalMatched(items.length);
   400→   400→        }
   401→   401→      } else {
   402→   402→        doFilter(items, value);
   403→   403→      }
   404→   404→    });
   405→   405→  }, [activeCategory, getItemsForCategory, getPopular, doFilter]);
   406→   406→
   407→   407→  const handleFocus = React.useCallback(() => {
   408→   408→    setLoading(true);
   409→   409→    showDropdown(query);
   410→   410→  }, [query, showDropdown]);
   411→   411→
   412→   412→  const selectSymbol = React.useCallback((s: InstrumentItem) => {
   413→   413→    setQuery(s.l18);
   414→   414→    setOpen(false);
   415→   415→    onSelect?.(s.l18, s.category, s.insCode, s.tgjuKey);
   416→   416→    inputRef.current?.blur();
   417→   417→  }, [onSelect]);
   418→   418→
   419→   419→  const handleKeyDown = React.useCallback((e: React.KeyboardEvent) => {
   420→   420→    if (!open || results.length === 0) {
   421→   421→      if (e.key === 'ArrowDown' || e.key === 'Enter') { e.preventDefault(); handleFocus(); }
   422→   422→      return;
   423→   423→    }
   424→   424→    if (e.key === 'ArrowDown') { e.preventDefault(); setActiveIndex((p) => p < results.length - 1 ? p + 1 : 0); }
   425→   425→    else if (e.key === 'ArrowUp') { e.preventDefault(); setActiveIndex((p) => p > 0 ? p - 1 : results.length - 1); }
   426→   426→    else if (e.key === 'Enter') { e.preventDefault(); if (activeIndex >= 0) selectSymbol(results[activeIndex]); }
   427→   427→    else if (e.key === 'Escape') { e.preventDefault(); setOpen(false); inputRef.current?.blur(); }
