   133→def flush_cache(symbol: Optional[str] = None) -> None:
   134→    """Flush model cache for a symbol or all."""
   135→    with _cache_lock:
   136→        if symbol:
   137→            _model_cache.pop(symbol, None)

   149→def _safe_div(a, b, fill: float = 0.0):
   150→    """Divide with zero-safety for scalars and Series."""
   151→    if isinstance(a, pd.Series) or isinstance(b, pd.Series):
   152→        result = np.where(np.abs(b) > 1e-12, a / b, fill)
   153→        idx = a.index if isinstance(a, pd.Series) else b.index
   154→        return pd.Series(result, index=idx)
   155→    return a / b if abs(b) > 1e-12 else fill

   158→def extract_features(df: pd.DataFrame) -> pd.DataFrame:
   159→    """
   160→    Build 40+ technical features from OHLCV data.
   161→
   162→    Parameters
   163→    ----------
   164→    df : pd.DataFrame
   165→        Must contain columns: open, high, low, close, volume (lowercase).
   166→        Index should be sequential integers.
   167→
   168→    Returns
   169→    -------
   170→    pd.DataFrame  —  same row count, NaNs at the head where windows can't fill.
   171→    """
   172→    out = pd.DataFrame(index=df.index)
   173→    c = df["close"].astype(float)
   174→    h = df["high"].astype(float)
   175→    l = df["low"].astype(float)
   176→    o = df["open"].astype(float)
   177→    v = df["volume"].astype(float)
   178→
   179→    # ── Price returns ──
   180→    for period in (1, 3, 5, 10, 20):
   181→        out[f"return_{period}d"] = c.pct_change(period)
   182→
   183→    # ── Rolling statistics ──
   184→    for w in (5, 10, 20, 60):
   185→        roll = c.rolling(w, min_periods=1)
   186→        out[f"close_mean_{w}"] = roll.mean()
   187→        out[f"close_std_{w}"] = roll.std()
   188→        out[f"close_min_{w}"] = roll.min()
   189→        out[f"close_max_{w}"] = roll.max()
   190→
   191→    # ── RSI(14) ──
   192→    delta = c.diff()
   193→    gain = delta.clip(lower=0).rolling(14, min_periods=1).mean()
   194→    loss = (-delta.clip(upper=0)).rolling(14, min_periods=1).mean()
   195→    rs = _safe_div(gain, loss, fill=100.0)
   196→    out["rsi_14"] = 100.0 - (100.0 / (1.0 + rs))
   197→
   198→    # ── MACD(12,26,9) ──
   199→    ema12 = c.ewm(span=12, adjust=False).mean()
   200→    ema26 = c.ewm(span=26, adjust=False).mean()
   201→    out["macd"] = ema12 - ema26
   202→    out["macd_signal"] = out["macd"].ewm(span=9, adjust=False).mean()
   203→    out["macd_hist"] = out["macd"] - out["macd_signal"]
   204→
   205→    # ── Stochastic(14,3,3) ──
   206→    low14 = l.rolling(14, min_periods=1).min()
   207→    high14 = h.rolling(14, min_periods=1).max()
   208→    out["stoch_k"] = 100.0 * _safe_div(c - low14, high14 - low14)
   209→    out["stoch_d"] = out["stoch_k"].rolling(3, min_periods=1).mean()
   210→
   211→    # ── CCI(20) ──
   212→    typical = (h + l + c) / 3.0
   213→    sma20 = typical.rolling(20, min_periods=1).mean()
   214→    mad20 = typical.rolling(20, min_periods=1).apply(lambda x: np.mean(np.abs(x - x.mean())), raw=True)
   215→    out["cci_20"] = _safe_div(typical - sma20, 0.015 * mad20)
   216→
   217→    # ── ATR(14) ──
   218→    tr = pd.concat([h - l, (h - c.shift(1)).abs(), (l - c.shift(1)).abs()], axis=1).max(axis=1)
   219→    out["atr_14"] = tr.rolling(14, min_periods=1).mean()
   220→
   221→    # ── ADX(14) ──
   222→    plus_dm = h.diff()
   223→    minus_dm = -l.diff()
   224→    plus_dm = plus_dm.where((plus_dm > minus_dm) & (plus_dm > 0), 0.0)
   225→    minus_dm = minus_dm.where((minus_dm > plus_dm) & (minus_dm > 0), 0.0)
   226→    tr_smooth = tr.ewm(span=14, adjust=False).mean()
   227→    plus_di = 100.0 * _safe_div(plus_dm.ewm(span=14, adjust=False).mean(), tr_smooth)
   228→    minus_di = 100.0 * _safe_div(minus_dm.ewm(span=14, adjust=False).mean(), tr_smooth)
   229→    dx_sum = _safe_div(abs(plus_di - minus_di), plus_di + minus_di) * 100.0
   230→    out["adx_14"] = dx_sum.ewm(span=14, adjust=False).mean()
   231→
   232→    # ── Bollinger Bands position ──
   233→    bb_mid = c.rolling(20, min_periods=1).mean()
   234→    bb_std = c.rolling(20, min_periods=1).std()
   235→    out["bb_position"] = _safe_div(c - bb_mid, 2.0 * bb_std)
   236→
   237→    # ── Volume features ──
   238→    out["volume_sma_20"] = v.rolling(20, min_periods=1).mean()
   239→    out["volume_ratio"] = _safe_div(v, out["volume_sma_20"])
   240→    out["volume_zscore"] = _safe_div(v - out["volume_sma_20"], v.rolling(20, min_periods=1).std())
   241→
   242→    # OBV (vectorized)
   243→    direction = pd.Series(np.sign(c.diff()), index=df.index).fillna(0)
   244→    obv = (direction * v).cumsum()
   245→    out["obv"] = obv
   246→
   247→    # ── Price patterns (rolling 10-bar) ──
   248→    is_hh = (h > h.shift(1)).astype(float)
   249→    is_ll = (l < l.shift(1)).astype(float)
   250→    is_inside = ((h <= h.shift(1)) & (l >= l.shift(1))).astype(float)
   251→    pw = 10
   252→    out["higher_highs"] = is_hh.rolling(pw, min_periods=1).mean()
   253→    out["lower_lows"] = is_ll.rolling(pw, min_periods=1).mean()
   254→    out["inside_bar_ratio"] = is_inside.rolling(pw, min_periods=1).mean()
   255→
   256→    # ── Momentum ──
   257→    out["roc_10"] = _safe_div(c - c.shift(10), c.shift(10)) * 100.0
   258→    momentum_score = (
   259→        out.get("rsi_14", pd.Series(50, index=df.index)) - 50.0
   260→        + out.get("macd_hist", pd.Series(0, index=df.index)) * 10.0
   261→        + out.get("stoch_k", pd.Series(50, index=df.index)) - 50.0
   262→    ) / 3.0
   263→    out["momentum_score"] = momentum_score
   264→
   265→    # ── Lag features (returns & RSI for last 5 periods) ──
   266→    for lag in range(1, 6):
   267→        out[f"return_1d_lag{lag}"] = out["return_1d"].shift(lag)
   268→        out[f"rsi_14_lag{lag}"] = out["rsi_14"].shift(lag)
   269→
   270→    return out
   271→

   278→def build_ohlcv_df(ohlcv: List[List]) -> pd.DataFrame:
   279→    """Convert raw OHLCV list to DataFrame with lowercase columns."""
   280→    df = pd.DataFrame(ohlcv, columns=["date", "open", "high", "low", "close", "volume"])

   282→        df[col] = pd.to_numeric(df[col], errors="coerce")
   283→    df["date"] = df["date"].astype(str)
   284→    return df

   289→# ─────────────────────────────────────────────────────────────────────────────

   291→
   292→def build_direction_target(df: pd.DataFrame) -> pd.Series:
   293→    """Binary: 1 if 5-day forward return > 0.5%, else 0."""
   294→    fwd = df["close"].shift(-FORWARD_DAYS) / df["close"] - 1.0
   295→    return (fwd > DIRECTION_THRESHOLD).astype(int)
   296→
   297→
   298→def build_regression_target(df: pd.DataFrame) -> pd.Series:
   299→    """5-day forward return as percentage."""
   300→    return (df["close"].shift(-FORWARD_DAYS) / df["close"] - 1.0) * 100.0
   301→
   302→
   303→def build_volatility_target(df: pd.DataFrame) -> pd.Series:
   304→    """Next-10-day realized volatility (std of daily returns * sqrt(252) * 100)."""
   305→    ret = df["close"].pct_change()
   306→    return ret.shift(-VOLATILITY_WINDOW).rolling(VOLATILITY_WINDOW).std() * np.sqrt(252) * 100.0

   314→def _save_model_artifact(model: Any, path: str, input_dim: int) -> None:
   315→    """Save model as ONNX (preferred) or pickle fallback."""
   316→    import onnx
   317→    os.makedirs(os.path.dirname(path), exist_ok=True)
   318→    if ONNX_EXPORT_AVAILABLE:
   319→        try:
   320→            if isinstance(model, xgb.XGBModel):
   321→                # XGBoost via onnxmltools (uses its own data types)
   322→                from onnxmltools.convert.common.data_types import FloatTensorType as XGBFloatType
   323→                initial_type = [('float_input', XGBFloatType([None, input_dim]))]
   324→                onnx_model = onnxmltools.convert_xgboost(model, initial_types=initial_type)
   325→            else:
   326→                # sklearn model via skl2onnx
   327→                initial_type = [('float_input', FloatTensorType([None, input_dim]))]
   328→                onnx_model = skl2onnx.convert_sklearn(model, initial_types=initial_type)
   329→            onnx.save(onnx_model, path)
   330→            return
   331→        except Exception as e:
   332→            log("WARN", f"ONNX export failed for {path}: {e}, falling back to pickle")
   333→    # Pickle fallback
   334→    pkl_path = path.replace(".onnx", ".pkl")
   335→    with open(pkl_path, "wb") as f:
   336→        pickle.dump(model, f)

   338→
   339→def _load_model_artifact(path: str) -> Tuple[Any, bool]:
   340→    """Load model from ONNX or pickle. Returns (model_or_session, is_onnx)."""
   341→    if os.path.exists(path):
   342→        try:
   343→            session = _get_ort().InferenceSession(path)
   344→            return session, True
   345→        except Exception:
   346→            pass
   347→    pkl_path = path.replace(".onnx", ".pkl")
   348→    if os.path.exists(pkl_path):
   349→        with open(pkl_path, "rb") as f:
   350→            return pickle.load(f), False
   351→    return None, False

   374→def _load_scaler(symbol: str) -> Optional[StandardScaler]:
   375→    cached = get_cached_scaler(symbol)
   376→    if cached is not None:
   377→        return cached
   378→    path = os.path.join(MODELS_DIR, symbol, "scaler.pkl")
   379→    if os.path.exists(path):
   380→        with open(path, "rb") as f:
   381→            scaler = pickle.load(f)
   382→        cache_scaler(symbol, scaler)
   383→        return scaler
   384→    return None

   391→
   392→def train_models(
   393→    symbol: str,
   394→    ohlcv: List[List],
   395→    model_names: Optional[List[str]] = None,
   396→) -> Dict[str, Any]:
   397→    """
   398→    Train requested models for a symbol.
   399→
   400→    Returns a dict with metrics and metadata.
   401→    """
   402→    if model_names is None:
   403→        model_names = MODEL_TYPES
   404→
   405→    # Validate
   406→    for mn in model_names:
   407→        if mn not in MODEL_TYPES:
   408→            raise ValueError(f"Unknown model type: {mn}")
   409→
   410→    df = build_ohlcv_df(ohlcv)
   411→    if len(df) < MIN_CANDLES:
   412→        raise ValueError(f"Need at least {MIN_CANDLES} candles, got {len(df)}")
   413→
   414→    log("INFO", f"Training {model_names} for {symbol} with {len(df)} candles")
   415→
   416→    # Feature engineering
   417→    features_df = extract_features(df)
   418→    feature_cols = [c for c in features_df.columns]
   419→
   420→    # Drop rows with NaN
   421→    valid_mask = features_df[feature_cols].notna().all(axis=1)
   422→    X = features_df.loc[valid_mask, feature_cols].values
   423→    feature_index = features_df.index[valid_mask]
   424→
   425→    # Build targets
   426→    dir_target = build_direction_target(df).loc[feature_index].values
   427→    reg_target = build_regression_target(df).loc[feature_index].values
   428→    vol_target = build_volatility_target(df).loc[feature_index].values
   429→
   430→    # Drop rows where targets are NaN
   431→    valid2 = ~(np.isnan(dir_target) | np.isnan(reg_target) | np.isnan(vol_target))
   432→    X = X[valid2]
   433→    dir_target = dir_target[valid2]
   434→    reg_target = reg_target[valid2]
   435→    vol_target = vol_target[valid2]
   436→
   437→    if len(X) < MIN_CANDLES:
   438→        raise ValueError(f"After cleaning, only {len(X)} valid samples (need {MIN_CANDLES})")
   439→
   440→    # Scale features
   441→    scaler = StandardScaler()
   442→    X_scaled = scaler.fit_transform(X)
   443→    input_dim = X_scaled.shape[1]
   444→
   445→    # Save scaler
   446→    _save_scaler(scaler, symbol)
   447→    cache_scaler(symbol, scaler)
   448→
   449→    # TimeSeriesSplit
   450→    tscv = TimeSeriesSplit(n_splits=5)
   451→    metrics: Dict[str, Dict] = {}
   452→    all_feature_importance: Dict[str, float] = {}
   453→
   454→    sym_dir = os.path.join(MODELS_DIR, symbol)
   455→    os.makedirs(sym_dir, exist_ok=True)
   456→
   457→    for mn in model_names:
   458→        log("INFO", f"  Training model: {mn}")
   459→
   460→        if mn == "xgboost_direction":
   461→            model = xgb.XGBClassifier(
   462→                n_estimators=200,
   463→                max_depth=5,
   464→                learning_rate=0.05,
   465→                subsample=0.8,
   466→                colsample_bytree=0.8,
   467→                use_label_encoder=False,
   468→                eval_metric="logloss",
   469→                verbosity=0,
   470→                n_jobs=-1,
   471→            )
   472→            target = dir_target
   473→            fold_metrics = {"accuracy": [], "f1": [], "auc": []}
   474→
   475→            for train_idx, test_idx in tscv.split(X_scaled):
   476→                model.fit(X_scaled[train_idx], target[train_idx])
   477→                preds = model.predict(X_scaled[test_idx])
   478→                proba = model.predict_proba(X_scaled[test_idx])[:, 1]
   479→                fold_metrics["accuracy"].append(accuracy_score(target[test_idx], preds))
   480→                fold_metrics["f1"].append(f1_score(target[test_idx], preds, zero_division=0))
   481→                try:
   482→                    fold_metrics["auc"].append(roc_auc_score(target[test_idx], proba))
   483→                except Exception:
   484→                    fold_metrics["auc"].append(0.5)
   485→
   486→            # Final fit on all data
   487→            model.fit(X_scaled, target)
   488→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   489→            imp = model.feature_importances_
   490→
   491→        elif mn == "xgboost_regression":
   492→            model = xgb.XGBRegressor(
   493→                n_estimators=200,
   494→                max_depth=5,
   495→                learning_rate=0.05,
   496→                subsample=0.8,
   497→                colsample_bytree=0.8,
   498→                verbosity=0,
   499→                n_jobs=-1,
   500→            )
   501→            target = reg_target
   502→            fold_metrics = {"mae": [], "rmse": [], "r2": []}
   503→
   504→            for train_idx, test_idx in tscv.split(X_scaled):
   505→                model.fit(X_scaled[train_idx], target[train_idx])
   506→                preds = model.predict(X_scaled[test_idx])
   507→                fold_metrics["mae"].append(mean_absolute_error(target[test_idx], preds))
   508→                fold_metrics["rmse"].append(np.sqrt(mean_squared_error(target[test_idx], preds)))
   509→                fold_metrics["r2"].append(r2_score(target[test_idx], preds))
   510→
   511→            model.fit(X_scaled, target)
   512→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   513→            imp = model.feature_importances_
   514→
   515→        elif mn == "ensemble_direction":
   516→            model = GradientBoostingClassifier(
   517→                n_estimators=200,
   518→                max_depth=4,
   519→                learning_rate=0.05,
   520→                subsample=0.8,
   521→                max_features="sqrt",
   522→                random_state=42,

   524→            target = dir_target

   526→
   527→            for train_idx, test_idx in tscv.split(X_scaled):
   528→                model.fit(X_scaled[train_idx], target[train_idx])
   529→                preds = model.predict(X_scaled[test_idx])
   530→                proba = model.predict_proba(X_scaled[test_idx])[:, 1]
   531→                fold_metrics["accuracy"].append(accuracy_score(target[test_idx], preds))
   532→                fold_metrics["f1"].append(f1_score(target[test_idx], preds, zero_division=0))
   533→                try:
   534→                    fold_metrics["auc"].append(roc_auc_score(target[test_idx], proba))
   535→                except Exception:
   536→                    fold_metrics["auc"].append(0.5)
   537→
   538→            model.fit(X_scaled, target)
   539→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   540→            imp = model.feature_importances_
   541→
   542→        elif mn == "ensemble_regression":
   543→            model = GradientBoostingRegressor(
   544→                n_estimators=200,
   545→                max_depth=4,
   546→                learning_rate=0.05,
   547→                subsample=0.8,
   548→                max_features="sqrt",
   549→                random_state=42,
   550→            )
   551→            target = reg_target
   552→            fold_metrics = {"mae": [], "rmse": [], "r2": []}
   553→
   554→            for train_idx, test_idx in tscv.split(X_scaled):
   555→                model.fit(X_scaled[train_idx], target[train_idx])
   556→                preds = model.predict(X_scaled[test_idx])
   557→                fold_metrics["mae"].append(mean_absolute_error(target[test_idx], preds))
   558→                fold_metrics["rmse"].append(np.sqrt(mean_squared_error(target[test_idx], preds)))
   559→                fold_metrics["r2"].append(r2_score(target[test_idx], preds))
   560→
   561→            model.fit(X_scaled, target)
   562→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   563→            imp = model.feature_importances_
   564→
   565→        elif mn == "volatility_model":
   566→            model = RandomForestRegressor(
   567→                n_estimators=200,
   568→                max_depth=6,
   569→                min_samples_leaf=5,
   570→                random_state=42,
   571→                n_jobs=-1,
   572→            )
   573→            target = vol_target
   574→            fold_metrics = {"mae": [], "rmse": [], "r2": []}
   575→
   576→            for train_idx, test_idx in tscv.split(X_scaled):
   577→                model.fit(X_scaled[train_idx], target[train_idx])
   578→                preds = model.predict(X_scaled[test_idx])
   579→                fold_metrics["mae"].append(mean_absolute_error(target[test_idx], preds))
   580→                fold_metrics["rmse"].append(np.sqrt(mean_squared_error(target[test_idx], preds)))
   581→                fold_metrics["r2"].append(r2_score(target[test_idx], preds))
   582→
   583→            model.fit(X_scaled, target)

   585→            imp = model.feature_importances_
   586→
   587→        else:
   588→            continue
   589→
   590→        # Save model artifact
   591→        onnx_path = os.path.join(sym_dir, f"{mn}.onnx")
   592→        _save_model_artifact(model, onnx_path, input_dim)
   593→
   594→        # Cache in memory
   595→        cache_model(symbol, mn, model, metrics.get(mn, {}))
   596→
   597→        # Save metadata
   598→        meta = {
   599→            "model_type": mn,
   600→            "symbol": symbol,
   601→            "metrics": metrics.get(mn, {}),
   602→            "feature_cols": feature_cols,
   603→            "input_dim": input_dim,

   607→        }
   608→        _save_meta(meta, os.path.join(sym_dir, f"{mn}_meta.json"))
   609→
   610→        # Accumulate feature importance (average across models)
   611→        for i, col in enumerate(feature_cols):
   612→            all_feature_importance[col] = all_feature_importance.get(col, 0.0) + float(imp[i])
   613→
   614→    # Normalize feature importance
   615→    total_imp = sum(all_feature_importance.values())
   616→    if total_imp > 0:

   621→

   638→def _predict_with_model(model: Any, features: np.ndarray, is_onnx: bool) -> np.ndarray:
   639→    """Run prediction on a single row of features."""
   640→    if is_onnx:
   641→        session = model  # ONNX Runtime session
   642→        input_name = session.get_inputs()[0].name
   643→        result = session.run(None, {input_name: features.astype(np.float32)})
   644→        # Classification: result[1] is probabilities; Regression: result[0] is values
   645→        if len(result) == 2 and result[1].shape[1] >= 2:
   646→            return result[1]  # probabilities
   647→        return result[0]  # regression
   648→    else:
   649→        # sklearn or xgboost native
   650→        if hasattr(model, "predict_proba"):
   651→            return model.predict_proba(features)  # type: ignore
   652→        return model.predict(features)  # type: ignore

   654→
   655→def predict_sessions(
   656→    symbol: str,
   657→    ohlcv: List[List],
   658→    horizon: int = 30,
   659→) -> Dict[str, Any]:
   660→    """
   661→    Generate multi-session prediction for a symbol.
   662→
   663→    Uses iterative prediction: predict 5 sessions, append, re-extract, repeat.
   664→    """
   665→    df = build_ohlcv_df(ohlcv)
   666→    if len(df) < MIN_CANDLES:
   667→        raise ValueError(f"Need at least {MIN_CANDLES} candles for prediction, got {len(df)}")
   668→
   669→    # Load scaler
   670→    scaler = _load_scaler(symbol)
   671→    if scaler is None:
   672→        raise ValueError(f"No trained scaler for symbol {symbol}")
   673→
   674→    # Determine which models are available
   675→    sym_dir = os.path.join(MODELS_DIR, symbol)
   676→    available_models = []
   677→    dir_model = None
   678→    dir_is_onnx = False
   679→    reg_model = None
   680→    reg_is_onnx = False
   681→
   682→    for mn in ("xgboost_direction", "ensemble_direction"):
   683→        model, is_onnx = _load_model_artifact(os.path.join(sym_dir, f"{mn}.onnx"))
   684→        if model is not None:
   685→            dir_model = model
   686→            dir_is_onnx = is_onnx
   687→            available_models.append(mn)
   688→            break
   689→
   690→    for mn in ("xgboost_regression", "ensemble_regression"):
   691→        model, is_onnx = _load_model_artifact(os.path.join(sym_dir, f"{mn}.onnx"))
   692→        if model is not None:
   693→            reg_model = model
   694→            reg_is_onnx = is_onnx
   695→            available_models.append(mn)
   696→            break
   697→
   698→    if dir_model is None and reg_model is None:
   699→        raise ValueError(f"No trained model for symbol {symbol}")
   700→
   701→    # Extract current features for the last candle
   702→    features_df = extract_features(df)
   703→    feature_cols = list(features_df.columns)
   704→    last_row = features_df.iloc[-1:].values
   705→    if np.any(np.isnan(last_row)):
   706→        # Forward-fill NaNs with column means
   707→        col_means = features_df.mean().values
   708→        nan_mask = np.isnan(last_row)
   709→        last_row[nan_mask] = np.take(col_means, np.where(nan_mask)[1])
   710→
   711→    current_features = {}
   712→    for i, col in enumerate(feature_cols):
   713→        current_features[col] = round(float(last_row[0, i]), 4)
   714→
   715→    # Iterative prediction
   716→    working_df = df.copy()
   717→    sessions: List[Dict] = []
   718→    last_close = float(df["close"].iloc[-1])
   719→    iterations = (horizon + PREDICTION_STEPS - 1) // PREDICTION_STEPS
   720→
   721→    for it in range(iterations):
   722→        remaining = horizon - len(sessions)
   723→        steps = min(PREDICTION_STEPS, remaining)
   724→
   725→        # Extract features from current working dataframe
   726→        feat_df = extract_features(working_df)
   727→        feat_cols = list(feat_df.columns)
   728→        last_features = feat_df.iloc[-1:].values
   729→
   730→        # Handle NaNs
   731→        if np.any(np.isnan(last_features)):
   732→            col_means = feat_df.mean().values
   733→            for j in range(last_features.shape[1]):
   734→                if np.isnan(last_features[0, j]):
   735→                    last_features[0, j] = col_means[j]
   736→
   737→        scaled = scaler.transform(last_features)
   738→
   739→        predicted_change = 0.0
   740→        direction = "flat"
   741→        confidence = 0.5
   742→
   743→        if reg_model is not None:
   744→            reg_result = _predict_with_model(reg_model, scaled, reg_is_onnx)
   745→            predicted_change = float(reg_result[0])
   746→
   747→        if dir_model is not None:
   748→            dir_result = _predict_with_model(dir_model, scaled, dir_is_onnx)
   749→            # dir_result is probability array [p_down, p_up]
   750→            if dir_result.ndim == 2 and dir_result.shape[1] >= 2:
   751→                p_up = float(dir_result[0, 1])
   752→            else:
   753→                p_up = float(dir_result[0]) if float(dir_result[0]) > 0.5 else 0.5
   754→            confidence = p_up if p_up > 0.5 else (1.0 - p_up)
   755→            direction = "up" if p_up > 0.5 else "down"
   756→
   757→        # If only regression model, derive direction from predicted change
   758→        if dir_model is None and reg_model is not None:
   759→            if predicted_change > DIRECTION_THRESHOLD * 100:
   760→                direction = "up"
   761→                confidence = min(0.9, 0.5 + abs(predicted_change) / 5.0)
   762→            elif predicted_change < -DIRECTION_THRESHOLD * 100:
   763→                direction = "down"
   764→                confidence = min(0.9, 0.5 + abs(predicted_change) / 5.0)
   765→
   766→        # Generate steps sessions
   767→        for s in range(steps):
   768→            session_num = len(sessions) + 1
   769→            step_change = predicted_change / steps
   770→            new_close = last_close * (1.0 + step_change / 100.0)
   771→            change_pct = round((new_close - last_close) / last_close * 100, 4)
   772→
   773→            sessions.append({
   774→                "session": session_num,
   775→                "predicted_close": round(new_close, 2),
   776→                "confidence": round(confidence, 4),
   777→                "direction": direction,
   778→                "change_pct": change_pct,
   779→            })
   780→            last_close = new_close
   781→
   782→        # Append predicted candle to working_df for next iteration
   783→        avg_change = predicted_change / steps
   784→        new_close_row = last_close * (1.0 + avg_change / 100.0)
   785→        new_row = {
   786→            "date": f"pred_{len(sessions)}",
   787→            "open": last_close,
   788→            "high": max(last_close, new_close_row) * (1.0 + abs(avg_change) / 200.0),
   789→            "low": min(last_close, new_close_row) * (1.0 - abs(avg_change) / 200.0),
   790→            "close": new_close_row,
   791→            "volume": 0.0,
   792→        }
   793→        working_df = pd.concat([working_df, pd.DataFrame([new_row])], ignore_index=True)
   794→
   795→    # Aggregate results
   796→    closes = [s["predicted_close"] for s in sessions]
   797→    min_price = min(closes)
   798→    max_price = max(closes)
   799→    final_close = closes[-1]
   800→    initial_close = float(df["close"].iloc[-1])
   801→    overall_change = (final_close - initial_close) / initial_close * 100
   802→
   803→    if overall_change > 1.0:
   804→        overall_direction = "up"
   805→    elif overall_change < -1.0:
   806→        overall_direction = "down"
   807→    else:
   808→        overall_direction = "flat"
   809→
   810→    avg_confidence = np.mean([s["confidence"] for s in sessions])
   811→    price_range = max_price - min_price
   812→    price_range_pct = price_range / initial_close * 100
   813→
   814→    if price_range_pct > 5.0:
   815→        risk_level = "high"
   816→    elif price_range_pct > 2.5:
   817→        risk_level = "medium"
   818→    else:
   819→        risk_level = "low"
   820→
   821→    return {
   822→        "status": "ok",
   823→        "symbol": symbol,
   824→        "prediction": {
   825→            "sessions": sessions,
   826→            "overall_direction": overall_direction,
   827→            "overall_confidence": round(float(avg_confidence), 4),
   828→            "target_price_min": round(min_price, 2),
   829→            "target_price_max": round(max_price, 2),
   830→            "risk_level": risk_level,
   831→        },
   832→        "models_used": available_models,
   833→        "current_features": current_features,
   834→    }
   835→

   842→def list_all_models() -> Dict[str, Any]:
   843→    """List all trained models across all symbols."""
   844→    result: Dict[str, List[Dict]] = {}
   845→    if not os.path.exists(MODELS_DIR):
   846→        return {"status": "ok", "models": {}}
   847→
   848→    for sym in sorted(os.listdir(MODELS_DIR)):
   849→        sym_path = os.path.join(MODELS_DIR, sym)
   850→        if not os.path.isdir(sym_path):
   851→            continue
   852→        models = []
   853→        for mn in MODEL_TYPES:
   854→            meta_path = os.path.join(sym_path, f"{mn}_meta.json")
   855→            if os.path.exists(meta_path):
   856→                meta = _load_meta(meta_path)
   857→                if meta:
   858→                    models.append(meta)
   859→        if models:
   860→            result[sym] = models
   861→
   862→    return {"status": "ok", "models": result}

   865→def get_symbol_models(symbol: str) -> Dict[str, Any]:
   866→    """Get models for a specific symbol."""
   867→    sym_path = os.path.join(MODELS_DIR, symbol)
   868→    if not os.path.isdir(sym_path):
   869→        return {"status": "error", "message": f"No trained model for symbol {symbol}"}
   870→
   871→    models = []
   872→    for mn in MODEL_TYPES:
   873→        meta_path = os.path.join(sym_path, f"{mn}_meta.json")
   874→        if os.path.exists(meta_path):
   875→            meta = _load_meta(meta_path)
   876→            if meta:
   877→                models.append(meta)
   878→
   879→    if not models:
   880→        return {"status": "error", "message": f"No trained model for symbol {symbol}"}
   881→
   882→    return {"status": "ok", "symbol": symbol, "models": models}

   891→    """Send a JSON response."""

   894→    handler.send_header("Content-Type", "application/json; charset=utf-8")
   895→    handler.send_header("Content-Length", str(len(body)))
   896→    handler.end_headers()
   897→    handler.wfile.write(body)
