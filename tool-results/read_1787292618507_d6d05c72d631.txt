   131→// Component
   132→// ═══════════════════════════════════════════════════════════════════════════════
   133→
   134→export default function VdssGraph(props: VdssGraphProps) {
   135→  const { symbolName, currentPrice, resistances, supports, ma100, scenarios, rsi, adx, bullScore, edgeWeights, currencyUnit, overallSignal } = props;
   136→
   137→  // ── Dynamic scenario labels (context-aware) ──────────────────
   138→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal || 'neutral'), [overallSignal]);
   139→
   140→  // ── Dynamic NODE_DEFS: R1-R5 titles/descriptions adapt to signal ──
   141→  const nodeDefs = useMemo(() => ({
   142→    ...NODE_DEFS,
   143→    R1: { ...NODE_DEFS.R1, title: dynamicMeta.R1.nodeTitle, desc: dynamicMeta.R1.nodeDesc },
   144→    R2: { ...NODE_DEFS.R2, title: dynamicMeta.R2.nodeTitle, desc: dynamicMeta.R2.nodeDesc },
   145→    R3: { ...NODE_DEFS.R3, title: dynamicMeta.R3.nodeTitle, desc: dynamicMeta.R3.nodeDesc },
   146→    R4: { ...NODE_DEFS.R4, title: dynamicMeta.R4.nodeTitle, desc: dynamicMeta.R4.nodeDesc },
   147→    R5: { ...NODE_DEFS.R5, title: dynamicMeta.R5.nodeTitle, desc: dynamicMeta.R5.nodeDesc },
   148→  }), [dynamicMeta]);
   149→  const unit = currencyUnit || 'ریال';
   150→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   151→  const S1_level = supports[0] ?? currentPrice * 0.95;
   152→
   153→  const [activeFilter, setActiveFilter] = useState('all');
   154→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   155→  const graphRef = useRef<HTMLDivElement>(null);
   156→  const svgRef = useRef<SVGSVGElement>(null);
   157→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   158→  const shellRef = useRef<HTMLDivElement>(null);
   159→
   160→  // ── Edge probabilities (using TA engine's bullScore) ──────────────
   161→  const edgeProbs = useMemo(() =>
   162→    edgeWeights ? calcEdgeProbabilities(edgeWeights) : {},
   163→  [edgeWeights]
   164→  );
   165→
   166→  // ── All paths with raw probabilities ─────────────────────────────
   167→  const allPathsRaw = useMemo(() => {
   168→    if (Object.keys(edgeProbs).length === 0) return [];
   169→    return findAllPaths(edgeProbs);
   170→  }, [edgeProbs]);
   171→
   172→  // ── Calibrate path probabilities to match TA engine scenario probabilities ──
   173→  const allPaths = useMemo(() => {
   174→    // Compute raw sum per target
   175→    const rawSums: Record<string, number> = {};
   176→    for (const key of SCENARIO_KEYS) rawSums[key] = 0;
   177→    for (const p of allPathsRaw) rawSums[p.target] += p.prob;
   178→
   179→    // Scale each path so the sum per target matches scenario probability
   180→    return allPathsRaw.map(p => {
   181→      const rawSum = rawSums[p.target] || 0.0001;
   182→      const targetProb = scenarios[p.target as keyof typeof scenarios].probability / 100;
   183→      const scale = targetProb / rawSum;
   184→      return { ...p, prob: p.prob * scale };
   185→    });
   186→  }, [allPathsRaw, scenarios]);
   187→
   188→  // ── Filtered paths ────────────────────────────────────────────────
   189→  const filteredPaths = useMemo(() => {
   190→    if (activeFilter === 'all') return allPaths;
   191→    // Check if filter is a scenario key
   192→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   193→      return allPaths.filter(p => p.target === activeFilter);
   194→    }
   195→    // Type filter: paths must use only edges of this type
   196→    return allPaths.filter(p =>
   197→      p.edges.every(ei => EDGES[ei][3] === activeFilter)
   198→    );
   199→  }, [allPaths, activeFilter]);
   200→
   201→  // ── Visible edges/nodes based on filter ────────────────────────────
   202→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   203→    if (activeFilter === 'all') {
   204→      const allEdgeIdx = EDGES.map((_, i) => i);
   205→      const allNodes = new Set(Object.keys(NODE_POSITIONS));
   206→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   207→    }
   208→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   209→      // Show all edges/nodes on paths to this scenario
   210→      const edgeSet = new Set<number>();
   211→      const nodeSet = new Set<string>();
   212→      for (const p of filteredPaths) {
   213→        p.edges.forEach(ei => edgeSet.add(ei));
   214→        p.nodes.forEach(n => nodeSet.add(n));
   215→      }
   216→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   217→    }
   218→    // Type filter
   219→    const edgeSet = new Set<number>();
   220→    const nodeSet = new Set<string>();
   221→    EDGES.forEach((e, i) => {
   222→      if (e[3] === activeFilter) {
   223→        edgeSet.add(i);
   224→        nodeSet.add(e[0]);
   225→        nodeSet.add(e[1]);
   226→      }
   227→    });
   228→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   229→  }, [activeFilter, filteredPaths]);
   230→
   231→  // ── Node values ──────────────────────────────────────────────────
   232→  const nodeValues: Record<string, string> = {
   233→    A: toFa(currentPrice),
   234→    B: toFa(R1_level),
   235→    D: toFa(resistances[1] ?? currentPrice * 1.10),
   236→    E: `${toFa(Math.round(R1_level + (resistances[1] ?? R1_level * 1.05 - R1_level) * 0.5))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 0.8))}`,
   237→    F: `${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 1.2))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 2.0))}`,
   238→    G: toFa(S1_level),
   239→    H: toFa(supports[1] ?? currentPrice * 0.90),
   240→    I: toFa(supports[2] ?? currentPrice * 0.85),
   241→    J: toFa(supports[3] ?? currentPrice * 0.80),
   242→    K: toFa(ma100),
   243→    L: props.trendDirection === 'up' ? 'صعودی' : props.trendDirection === 'down' ? 'نزولی' : 'خنثی',
   244→  };
   245→  for (const key of SCENARIO_KEYS) {
   246→    const s = scenarios[key];
   247→    nodeValues[key] = `${toFa(s.targetMin)} — ${toFa(s.targetMax)} ${unit}`;
   248→  }
   249→
   250→  // ── Draw SVG edges ───────────────────────────────────────────────
   251→  const drawEdges = useCallback(() => {
   252→    const svg = svgRef.current;
   253→    const graph = graphRef.current;
   254→    if (!svg || !graph) return;
   255→
   256→    let markersSvg = '';
   257→    for (const [k, c] of Object.entries(EDGE_COLORS)) {
   258→      markersSvg += `<marker id="arrow-${k}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   259→    }
   260→
   261→    let pathsSvg = '';
   262→    const graphRect = graph.getBoundingClientRect();
   263→
   264→    for (let ei = 0; ei < EDGES.length; ei++) {
   265→      const [fromId, toId, label, type] = EDGES[ei];
   266→      const isVisible = visibleEdgeIndices.includes(ei);
   267→      const fromEl = nodeRefs.current[fromId];
   268→      const toEl = nodeRefs.current[toId];
   269→      if (!fromEl || !toEl) continue;
   270→
   271→      const fRect = fromEl.getBoundingClientRect();
   272→      const tRect = toEl.getBoundingClientRect();
   273→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   274→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   275→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   276→      const by = tRect.top - graphRect.top + tRect.height / 2;
   277→      const dx = bx - ax;
   278→      const dy = by - ay;
   279→      const dist = Math.sqrt(dx * dx + dy * dy);
   280→      if (dist === 0) continue;
   281→
   282→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   283→      const mx = (ax + bx) / 2;
   284→      const my = (ay + by) / 2;
   285→      const nx = -dy / dist;
   286→      const ny = dx / dist;
   287→      const cx = mx + nx * bend;
   288→      const cy = my + ny * bend;
   289→
   290→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   291→      const prob = edgeProbs[String(ei)] ?? 0;
   292→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   293→
   294→      pathsSvg += `<path d="${d}" stroke="${EDGE_COLORS[type]}" stroke-width="${isVisible ? 2 : 1}" opacity="${isVisible ? 0.72 : 0.06}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   295→
   296→      // Edge probability label (only for visible edges)
   297→      if (isVisible) {
   298→        pathsSvg += `<text x="${cx}" y="${cy - 6}" fill="#374151" font-size="9" text-anchor="middle" paint-order="stroke" stroke="#f3f4f6" stroke-width="3.5" stroke-linejoin="round" opacity="0.85" data-type="${type}" class="edge-label">${label}</text>`;
   299→        pathsSvg += `<text x="${cx}" y="${cy + 8}" fill="${EDGE_COLORS[type]}" font-size="10" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#f3f4f6" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   300→      }
   301→    }
   302→
   303→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   304→  }, [visibleEdgeIndices, edgeProbs]);
   305→
   306→  useEffect(() => { drawEdges(); }, [drawEdges]);
   307→  useEffect(() => {
   308→    const el = shellRef.current;
   309→    if (!el) return;
   310→    const observer = new ResizeObserver(() => drawEdges());
   311→    observer.observe(el);
   312→    return () => observer.disconnect();
   313→  }, [drawEdges]);
   314→
   315→  // ── Aggregated path probabilities per target (calibrated = matches scenario prob) ──
   316→  const pathProbsByTarget = useMemo(() => {
   317→    const result: Record<string, number> = {};
   318→    for (const key of SCENARIO_KEYS) {
   319→      result[key] = 0;
   320→      for (const p of filteredPaths) {
   321→        if (p.target === key) result[key] = (result[key] ?? 0) + p.prob;
   322→      }
   323→      // Fallback: if path probability is 0 but scenario has probability, use scenario directly
   324→      if (result[key] === 0 && scenarios[key].probability > 0) {
   325→        result[key] = scenarios[key].probability / 100;
   326→      }
   327→    }
   328→    return result;
   329→  }, [filteredPaths, scenarios]);
   330→
   331→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   332→
   333→  // ── Detail panel content ─────────────────────────────────────────
   334→  const detailContent = useMemo(() => {
   335→    if (!selectedNode) return null;
   336→    const def = nodeDefs[selectedNode];
   337→    if (!def) return null;
   338→    const relations = EDGES.filter(e => e[0] === selectedNode || e[1] === selectedNode);
   339→    const inputs = relations.filter(e => e[1] === selectedNode);
   340→    const outputs = relations.filter(e => e[0] === selectedNode);
   341→
   342→    return (
   343→      <div className="space-y-3">
   344→        <h3 className="text-sm font-bold text-gray-900">{def.title}</h3>
   345→        <p className="text-xs text-gray-600"><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   346→        <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border border-gray-300 bg-gray-100 text-gray-600">{def.type}</span>
   347→        <p className="text-xs text-gray-500 leading-relaxed">{def.desc}</p>
   348→        {inputs.length > 0 && (
   349→          <div>
   350→            <p className="text-xs font-medium text-gray-600 mb-1">مسیرهای ورودی ({inputs.length}):</p>
   351→            <ul className="space-y-1">
   352→              {inputs.map((e, i) => {
   353→                const ei = EDGES.indexOf(e);
   354→                const ep = edgeProbs[String(ei)] ?? 0;
   355→                return (
   356→                  <li key={i} className="text-[11px] text-gray-500 leading-relaxed border-t border-dashed border-gray-300 pt-1.5">
   357→                    <b className="text-gray-500">{e[0]} ← {e[1]}</b>
   358→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${EDGE_COLORS[e[3]]}20`, color: EDGE_COLORS[e[3]] }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   359→                    <br />{e[2]}
   360→                  </li>
   361→                );
   362→              })}
   363→            </ul>
   364→          </div>
   365→        )}
   366→        {outputs.length > 0 && (
   367→          <div>
   368→            <p className="text-xs font-medium text-gray-600 mb-1">مسیرهای خروجی ({outputs.length}):</p>
   369→            <ul className="space-y-1">
   370→              {outputs.map((e, i) => {
   371→                const ei = EDGES.indexOf(e);
   372→                const ep = edgeProbs[String(ei)] ?? 0;
   373→                return (
   374→                  <li key={i} className="text-[11px] text-gray-500 leading-relaxed border-t border-dashed border-gray-300 pt-1.5">
   375→                    <b className="text-gray-500">{e[0]} → {e[1]}</b>
   376→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${EDGE_COLORS[e[3]]}20`, color: EDGE_COLORS[e[3]] }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   377→                    <br />{e[2]}
   378→                  </li>
   379→                );
   380→              })}
   381→            </ul>
   382→          </div>
   383→        )}
