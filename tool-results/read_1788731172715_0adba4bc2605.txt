   360→   360→/**
   361→   361→ * Parse a number that might contain Persian digits, commas, or other formatting.
   362→   362→ * Handles null/undefined/empty strings gracefully.
   363→   363→ */
   364→   364→function parsePersianNum(str: string | null | undefined | number): number {
   365→   365→  if (str === null || str === undefined) return 0;
   366→   366→  let s = String(str).trim();
   367→   367→  if (!s || s === '-' || s === '') return 0;
   368→   368→  // Strip trailing % sign (e.g. "0.05%" → "0.05")
   369→   369→  if (s.endsWith('%')) s = s.slice(0, -1).trim();
   370→   370→  // Replace commas and Persian digits
   371→   371→  const cleaned = s.replace(/[,۰-۹]/g, (c: string) => {
   372→   372→    const persianDigits = '۰۱۲۳۴۵۶۷۸۹';
   373→   373→    const idx = persianDigits.indexOf(c);
   374→   374→    return idx >= 0 ? String(idx) : c === ',' ? '' : c;
   375→   375→  });
   376→   376→  const n = Number(cleaned);
   377→   377→  return isFinite(n) ? n : 0;
   378→   378→}

   380→   380→/**
   381→   381→ * Check if a value looks like a date string (contains / or - and is not a number).
   382→   382→ */
   383→   383→function looksLikeDate(val: string | null | undefined | number): boolean {
   384→   384→  if (!val) return false;
   385→   385→  const s = String(val).trim();
   386→   386→  return /[\/\-]/.test(s) && isNaN(Number(s.replace(/[,۰-۹\/\-]/g, '')));
   387→   387→}
   388→   388→
   389→   389→/**
   390→   390→ * Check if a value looks like a percentage (ends with % or is small like 0.5 to 50).
   391→   391→ */
   392→   392→function looksLikePercentage(val: string | null | undefined | number): boolean {
   393→   393→  if (!val) return false;
   394→   394→  const s = String(val).trim();
   395→   395→  if (s.endsWith('%')) return true;
   396→   396→  const n = parsePersianNum(val);
   397→   397→  return !isNaN(n) && Math.abs(n) < 100 && n !== 0;
   398→   398→}

   400→   400→/**
   401→   401→ * Auto-detect column mapping for TGJU API response.
   402→   402→ * Expected format: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   403→   403→ * But the API may change column order.
   404→   404→ */
   405→   405→function detectColumnMap(row: (string | number)[]): {
   406→   406→  openIdx: number; highIdx: number; lowIdx: number; closeIdx: number;
   407→   407→  jalaliDateIdx: number; gregorianDateIdx: number;
   408→   408→} | null {
   409→   409→  if (!row || row.length < 6) return null;
   410→   410→
   411→   411→  let jalaliDateIdx = -1;
   412→   412→  let gregorianDateIdx = -1;
   413→   413→  const numericIndices: number[] = [];
   414→   414→  const percentageIndices: number[] = [];
   415→   415→
   416→   416→  for (let i = 0; i < row.length; i++) {
   417→   417→    const val = row[i];
   418→   418→    if (looksLikeDate(val)) {
   419→   419→      // Jalali dates have Persian digits or format like 1404/05/25
   420→   420→      const s = String(val);
   421→   421→      if (/[۰-۹]/.test(s) || /^14\d{2}/.test(s.replace(/[۰-۹]/g, (d: string) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d))))) {
   422→   422→        if (jalaliDateIdx === -1) jalaliDateIdx = i;
   423→   423→      } else {
   424→   424→        if (gregorianDateIdx === -1) gregorianDateIdx = i;
   425→   425→      }
   426→   426→    } else if (looksLikePercentage(val)) {
   427→   427→      percentageIndices.push(i);
   428→   428→    } else {
   429→   429→      const n = parsePersianNum(val);
   430→   430→      if (isFinite(n) && n !== 0) {
   431→   431→        numericIndices.push(i);
   432→   432→      }
   433→   433→    }
   434→   434→  }
   435→   435→
   436→   436→  // If we couldn't auto-detect dates, fall back to assumed positions
   437→   437→  if (jalaliDateIdx === -1 && row.length >= 8) {
   438→   438→    // Default assumption: row[6] = gregorian, row[7] = jalali
   439→   439→    if (looksLikeDate(row[7])) jalaliDateIdx = 7;
   440→   440→    else if (looksLikeDate(row[6])) jalaliDateIdx = 6;
   441→   441→    if (gregorianDateIdx === -1 && looksLikeDate(row[6]) && jalaliDateIdx !== 6) gregorianDateIdx = 6;
   442→   442→    else if (gregorianDateIdx === -1 && looksLikeDate(row[7]) && jalaliDateIdx !== 7) gregorianDateIdx = 7;
   443→   443→  }
   444→   444→
   445→   445→  // Numeric columns (excluding percentages) should be OHLC + change
   446→   446→  // OHLC: typically 4 largest absolute values among non-percentage numerics
   447→   447→  // change: the remaining one
   448→   448→  const ohlcCandidates = numericIndices.filter(i => !percentageIndices.includes(i));
   449→   449→
   450→   450→  if (ohlcCandidates.length < 4) {
   451→   451→    // Fallback: assume first 4 numeric columns are OHLC
   452→   452→    if (numericIndices.length >= 4) {
   453→   453→      return {
   454→   454→        openIdx: numericIndices[0],
   455→   455→        highIdx: numericIndices[2],
   456→   456→        lowIdx: numericIndices[1],
   457→   457→        closeIdx: numericIndices[3],
   458→   458→        jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : 7,
   459→   459→        gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : 6,
   460→   460→      };
   461→   461→    }
   462→   462→    return null;
   463→   463→  }
   464→   464→
   465→   465→  // Sort by absolute value descending to identify OHLC (largest) vs change (smallest)
   466→   466→  const sorted = ohlcCandidates
   467→   467→    .map(i => ({ idx: i, val: Math.abs(parsePersianNum(row[i])) }))
   468→   468→    .sort((a, b) => b.val - a.val);
   469→   469→
   470→   470→  // The 4 largest should be OHLC. Among them, identify high (max) and low (min).
   471→   471→  // open and close are the other two.
   472→   472→  const top4 = sorted.slice(0, 4);
   473→   473→  const maxIdx = top4.reduce((a, b) => a.val > b.val ? a : b).idx;
   474→   474→  const minIdx = top4.reduce((a, b) => a.val < b.val ? a : b).idx;
   475→   475→  const others = top4.filter(x => x.idx !== maxIdx && x.idx !== minIdx);
   476→   476→
   477→   477→  // Among the two remaining, the one closer to high is open, the other is close
   478→   478→  // (heuristic: in an uptrend open < close, but we can't assume direction)
   479→   479→  // Default: first = open, second = close
   480→   480→  const openIdx = others[0]?.idx ?? ohlcCandidates[0];
   481→   481→  const closeIdx = others[1]?.idx ?? ohlcCandidates[3];
   482→   482→
   483→   483→  return {
   484→   484→    openIdx,
   485→   485→    highIdx: maxIdx,
   486→   486→    lowIdx: minIdx,
   487→   487→    closeIdx,
   488→   488→    jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : row.length - 1,
   489→   489→    gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : row.length - 2,
   490→   490→  };
   491→   491→}

   493→   493→/**
   494→   494→ * Validate a candle's OHLC values for sanity.
   495→   495→ * Returns true if the candle is valid.
   496→   496→ */
   497→   497→function isValidCandle(o: number, h: number, l: number, c: number): boolean {
   498→   498→  if (!isFinite(o) || !isFinite(h) || !isFinite(l) || !isFinite(c)) return false;
   499→   499→  if (o <= 0 || h <= 0 || l <= 0 || c <= 0) return false;
   500→   500→  // High must be >= Low, Open, Close
   501→   501→  if (h < l || h < o || h < c) return false;
   502→   502→  // Low must be <= High, Open, Close
   503→   503→  if (l > o || l > c) return false;
   504→   504→  // No infinite or extreme values
   505→   505→  if (o > 1e15 || h > 1e15 || l > 1e15 || c > 1e15) return false;
   506→   506→  return true;
   507→   507→}

   509→   509→/**
   510→   510→ * Check if parsed candle data is reasonably fresh.
   511→   511→ * Rejects data where the most recent candle is older than 2 years.
   512→   512→ * This prevents serving data from e.g. 1392 when current year is 1404.
   513→   513→ */
   514→   514→function isDataFresh(candles: TgjuOHLC[]): boolean {
   515→   515→  if (candles.length === 0) return false;
   516→   516→  const lastDateStr = candles[candles.length - 1].date;
   517→   517→  const yearMatch = lastDateStr.match(/(\d{4})/);
   518→   518→  if (!yearMatch) return true; // Can't determine year, assume fresh
   519→   519→
   520→   520→  const year = parseInt(yearMatch[1], 10);
   521→   521→  const now = new Date();
   522→   522→  const gregorianYear = now.getFullYear();
   523→   523→  // Approximate current Jalali year (Gregorian - 621 or 622)
   524→   524→  const approxJalaliYear = gregorianYear - 621;
   525→   525→
   526→   526→  // Jalali years: 1390-1500 range
   527→   527→  if (year >= 1390 && year <= 1500) {
   528→   528→    return year >= approxJalaliYear - 2; // Allow up to 2 years old
   529→   529→  }
   530→   530→
   531→   531→  // Gregorian years
   532→   532→  if (year >= 2020 && year <= 2035) {
   533→   533→    return year >= gregorianYear - 2;
   534→   534→  }
   535→   535→
   536→   536→  // Unknown format, don't reject
   537→   537→  return true;
   538→   538→}

   620→   620→          change: 0,
   621→   621→          changePercent: 0,
   622→   622→          category: 'currency',

   624→   624→      }
   625→   625→    }
   626→   626→
   627→   627→    if (goldRes.ok) {
   628→   628→      const goldCategories: TgjuGoldCategory[] = await goldRes.json();
   629→   629→      for (const cat of goldCategories) {
   630→   630→        for (const item of cat.prices) {

   868→   868→ *
   869→   869→ * Strategy (permanent fix):
   870→   870→ * 1. Check in-memory cache (4h TTL)
   871→   871→ * 2. For global instruments with Yahoo mapping → use Yahoo Finance as PRIMARY
   872→   872→ * 3. For Iranian instruments → try direct fetch, then queued page_reader with retry
   873→   873→ * 4. If all fail → return stale cache (up to 24h) or file-based cache

   875→   875→export async function fetchTgjuHistory(tgjuKey: string): Promise<TgjuOHLC[]> {
   876→   876→  const now = Date.now();
   877→   877→
   878→   878→  // ── Step 0: Check in-memory cache (with freshness validation) ──
   879→   879→  const cached = historyCache.get(tgjuKey);
   880→   880→  if (cached && now - cached.ts < HISTORY_TTL) {
   881→   881→    // Validate cached data is not stale (e.g. 1392-era data from a previous bug)
   882→   882→    if (isDataFresh(cached.data)) {
   883→   883→      return cached.data;
   884→   884→    } else {
   885→   885→      console.warn(`[TGJU] In-memory cache for ${tgjuKey} is STALE (latest: ${cached.data[cached.data.length - 1]?.date}), invalidating`);
   886→   886→      historyCache.delete(tgjuKey);
   887→   887→    }
   888→   888→  }
   889→   889→
   890→   890→  // ── Step 1: For global instruments, use Yahoo Finance as PRIMARY source ──
   891→   891→  const yahooSymbol = TGJU_TO_YAHOO_MAP[tgjuKey];
   892→   892→  if (yahooSymbol) {
   893→   893→    try {
   894→   894→      const YahooFinance = (await import('yahoo-finance2')).default;
   895→   895→      const yf = new YahooFinance({ suppressNotices: ['yahooSurvey'], validation: { logErrors: false } });
   896→   896→      const result = await yf.chart(yahooSymbol, {
   897→   897→        period1: new Date(Date.now() - 400 * 24 * 60 * 60 * 1000),
   898→   898→        period2: new Date(),
   899→   899→        interval: '1d',
   900→   900→      });
   901→   901→      if (result.quotes && result.quotes.length >= 30) {
   902→   902→        const candles: TgjuOHLC[] = result.quotes
   903→   903→          .filter((q: any) => q.close != null && q.open != null)
   904→   904→          .map((q: any) => ({
   905→   905→            date: q.date?.toISOString().split('T')[0] || '',
   906→   906→            open: q.open!,
   907→   907→            high: q.high || q.close!,
   908→   908→            low: q.low || q.close!,
   909→   909→            close: q.close!,
   910→   910→          }));
   911→   911→        if (candles.length >= 30) {
   912→   912→          console.log(`[TGJU] ${tgjuKey}: Got ${candles.length} candles from Yahoo (${yahooSymbol})`);
   913→   913→          yahooFetchedKeys.add(tgjuKey);
   914→   914→          cacheHistory(tgjuKey, candles, now);
   915→   915→          return candles;
   916→   916→        }
   917→   917→      }
   918→   918→    } catch (err) {
   919→   919→      console.warn(`[TGJU] Yahoo historical fetch failed for ${tgjuKey} (${yahooSymbol}):`, err instanceof Error ? err.message : err);
   920→   920→    }
   921→   921→  }
   922→   922→

   925→   925→  const directResult = await tryDirectFetch(tgjuKey);
   926→   926→  if (directResult && directResult.length >= 30) {
   927→   927→    cacheHistory(tgjuKey, directResult, now);
   928→   928→    return directResult;
   929→   929→  }
   930→   930→

   933→   933→  const pageReaderResult = await fetchViaQueuedPageReader(tgjuKey);
   934→   934→  if (pageReaderResult && pageReaderResult.length >= 30) {

   936→   936→    return pageReaderResult;

   938→   938→
   939→   939→  // ── Step 3: All methods failed — return stale cache (only if fresh) ──

   962→   962→async function tryDirectFetch(tgjuKey: string): Promise<TgjuOHLC[] | null> {

   991→   991→/**
   992→   992→ * Fetch TGJU historical data via queued page_reader with retry and rate limiting.
   993→   993→ * This prevents 429 errors by serializing requests and adding delays between calls.
   994→   994→ */
   995→   995→async function fetchViaQueuedPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   996→   996→  // Enqueue this request — ensures only one page_reader call at a time
   997→   997→  let result: TgjuOHLC[] | null = null;
   998→   998→  const taskPromise = (async () => {
   999→   999→    // Rate limit: wait if last call was too recent
  1000→  1000→    const timeSinceLastCall = Date.now() - lastPageReaderCall;
  1001→  1001→    if (timeSinceLastCall < PAGE_READER_MIN_INTERVAL) {
  1002→  1002→      const waitTime = PAGE_READER_MIN_INTERVAL - timeSinceLastCall;
  1003→  1003→      console.log(`[TGJU] Rate limiting page_reader: waiting ${waitTime}ms...`);
  1004→  1004→      await new Promise(r => setTimeout(r, waitTime));
  1005→  1005→    }
  1006→  1006→
  1007→  1007→    // Retry with exponential backoff
  1008→  1008→    for (let attempt = 1; attempt <= PAGE_READER_MAX_RETRIES; attempt++) {
  1009→  1009→      try {
  1010→  1010→        lastPageReaderCall = Date.now();
  1011→  1011→        const fetchResult = await fetchTgjuHistoryViaPageReader(tgjuKey);
  1012→  1012→        if (fetchResult && fetchResult.length >= 30) {
  1013→  1013→          result = fetchResult;
  1014→  1014→          return; // success
  1015→  1015→        }
  1016→  1016→        // Got data but not enough candles — don't retry
  1017→  1017→        if (fetchResult && fetchResult.length > 0) {
  1018→  1018→          console.warn(`[TGJU] page_reader returned only ${fetchResult.length} candles for ${tgjuKey}`);
  1019→  1019→          return;
  1020→  1020→        }
  1021→  1021→      } catch (err) {
  1022→  1022→        const errMsg = err instanceof Error ? err.message : String(err);
  1023→  1023→        const isRateLimit = errMsg.includes('429') || errMsg.includes('Too many requests');
  1024→  1024→        if (isRateLimit && attempt < PAGE_READER_MAX_RETRIES) {
  1025→  1025→          const delay = PAGE_READER_RETRY_BASE_DELAY * Math.pow(3, attempt - 1); // 3s, 9s, 27s
  1026→  1026→          console.warn(`[TGJU] page_reader 429 for ${tgjuKey}, retry ${attempt}/${PAGE_READER_MAX_RETRIES} in ${delay}ms...`);
  1027→  1027→          await new Promise(r => setTimeout(r, delay));
  1028→  1028→          continue;
  1029→  1029→        }
  1030→  1030→        console.error(`[TGJU] page_reader error for ${tgjuKey} (attempt ${attempt}):`, errMsg);
  1031→  1031→        return;
  1032→  1032→      }
  1033→  1033→    }
  1034→  1034→  })();
  1035→  1035→
  1036→  1036→  // Chain onto the queue
  1037→  1037→  pageReaderQueue = pageReaderQueue.then(() => taskPromise, () => taskPromise);
  1038→  1038→  await pageReaderQueue;
  1039→  1039→  return result;
  1040→  1040→}

  1042→  1042→/**
  1043→  1043→ * Single page_reader call to fetch TGJU historical data.
  1044→  1044→ * Uses z-ai SDK to bypass Cloudflare challenges.
  1045→  1045→ * Throws on failure so the caller (fetchViaQueuedPageReader) can retry.
  1046→  1046→ */
  1047→  1047→async function fetchTgjuHistoryViaPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
  1048→  1048→  // CRITICAL: order_dir=desc to get the MOST RECENT candles, not the oldest.
  1049→  1049→  const apiUrl = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=desc&start=0&length=365`;
  1050→  1050→
  1051→  1051→  const ZAI = (await import('z-ai-web-dev-sdk')).default;
  1052→  1052→  const zai = await ZAI.create();
  1053→  1053→  console.log(`[TGJU] Using z-ai SDK page_reader for ${tgjuKey}...`);
  1054→  1054→  const invokeResult = await zai.functions.invoke('page_reader', { url: apiUrl });
  1055→  1055→
  1056→  1056→  if (invokeResult.code !== 200 || !invokeResult.data?.html) {
  1057→  1057→    const errMsg = invokeResult.error || `code=${invokeResult.code}`;
  1058→  1058→    throw new Error(errMsg);
  1059→  1059→  }
  1060→  1060→
  1061→  1061→  // Extract JSON from <pre> tag in the HTML
  1062→  1062→  const html = invokeResult.data.html;
  1063→  1063→  const preMatch = html.match(/<pre[^>]*>([\s\S]*?)<\/pre>/);
  1064→  1064→  if (!preMatch) {
  1065→  1065→    throw new Error('no <pre> block found in page_reader response');
  1066→  1066→  }
  1067→  1067→
  1068→  1068→  // Unescape HTML entities and strip HTML tags from values
  1069→  1069→  const jsonStr = preMatch[1]
  1070→  1070→    .replace(/<[^>]*>/g, '')        // Strip ALL HTML tags first
  1071→  1071→    .replace(/&quot;/g, '"')
  1072→  1072→    .replace(/&amp;/g, '&')
  1073→  1073→    .replace(/&lt;/g, '<')
  1074→  1074→    .replace(/&gt;/g, '>')
  1075→  1075→    .replace(/\s+/g, ' ')           // Normalize whitespace
  1076→  1076→    .trim();
  1077→  1077→
  1078→  1078→  if (!jsonStr.startsWith('{')) {
  1079→  1079→    throw new Error('extracted content is not JSON');
  1080→  1080→  }
  1081→  1081→
  1082→  1082→  return parseTgjuChartData(tgjuKey, jsonStr);
  1083→  1083→}

  1085→  1085→/**
  1086→  1086→ * Parse the JSON data from TGJU chart API response.
  1087→  1087→ * Used by both direct fetch and page_reader paths.
  1088→  1088→ */
  1089→  1089→function parseTgjuChartData(tgjuKey: string, jsonStr: string): TgjuOHLC[] | null {
  1090→  1090→  try {
  1091→  1091→    const parsed = JSON.parse(jsonStr);
  1092→  1092→    const rawRows: (string | number)[][] = parsed.data;
  1093→  1093→
  1094→  1094→    if (!rawRows || rawRows.length === 0) {
  1095→  1095→      console.warn(`[TGJU] Empty data array for ${tgjuKey}`);
  1096→  1096→      return null;
  1097→  1097→    }
  1098→  1098→
  1099→  1099→    // Sanitize row values: strip any residual HTML tags and trim
  1100→  1100→    const rows = rawRows.map(row =>
  1101→  1101→      row.map(cell => {
  1102→  1102→        const s = String(cell).trim();
  1103→  1103→        // Strip HTML tags if any remain (safety for direct fetch path)
  1104→  1104→        return s.replace(/<[^>]*>/g, '').replace(/\s+/g, ' ');
  1105→  1105→      })
  1106→  1106→    );
  1107→  1107→
  1108→  1108→    // ── Auto-detect column mapping from first valid row ──
  1109→  1109→    let colMap = detectColumnMap(rows[0]);
  1110→  1110→    if (!colMap) {
  1111→  1111→      for (let r = 1; r < Math.min(5, rows.length); r++) {
  1112→  1112→        colMap = detectColumnMap(rows[r]);
  1113→  1113→        if (colMap) break;
  1114→  1114→      }
  1115→  1115→    }
  1116→  1116→    if (!colMap) {
  1117→  1117→      console.warn(`[TGJU] Could not auto-detect columns for ${tgjuKey}, using default order`);
  1118→  1118→      colMap = { openIdx: 0, highIdx: 2, lowIdx: 1, closeIdx: 3, jalaliDateIdx: 7, gregorianDateIdx: 6 };
  1119→  1119→    } else {
  1120→  1120→      console.log(`[TGJU] Column map for ${tgjuKey}: O=${colMap.openIdx} H=${colMap.highIdx} L=${colMap.lowIdx} C=${colMap.closeIdx} Date=${colMap.jalaliDateIdx}`);
  1121→  1121→    }
  1122→  1122→
  1123→  1123→    const { openIdx, highIdx, lowIdx, closeIdx, jalaliDateIdx, gregorianDateIdx } = colMap;
  1124→  1124→
  1125→  1125→    // ── Parse rows ──
  1126→  1126→    const candles: TgjuOHLC[] = [];
  1127→  1127→    let invalidCount = 0;
  1128→  1128→
  1129→  1129→    for (const row of rows) {
  1130→  1130→      const rawDate = String(row[jalaliDateIdx] || row[gregorianDateIdx] || '');
  1131→  1131→      const date = rawDate.replace(/[\/]/g, '-');
  1132→  1132→      const o = parsePersianNum(row[openIdx]);
  1133→  1133→      const h = parsePersianNum(row[highIdx]);
  1134→  1134→      const l = parsePersianNum(row[lowIdx]);
  1135→  1135→      const c = parsePersianNum(row[closeIdx]);
  1136→  1136→
  1137→  1137→      if (!isValidCandle(o, h, l, c) || date.length === 0) {
  1138→  1138→        invalidCount++;
  1139→  1139→        continue;
  1140→  1140→      }
  1141→  1141→
  1142→  1142→      candles.push({ date, open: o, high: h, low: l, close: c });
  1143→  1143→    }
  1144→  1144→
  1145→  1145→    if (invalidCount > 0) {
  1146→  1146→      console.warn(`[TGJU] ${tgjuKey}: ${invalidCount}/${rows.length} invalid candles skipped`);
  1147→  1147→    }
  1148→  1148→
  1149→  1149→    if (candles.length === 0) {
  1150→  1150→      return null;
  1151→  1151→    }
  1152→  1152→
  1153→  1153→    // ── Detect and fix data drift ──
  1154→  1154→    let driftCount = 0;
  1155→  1155→    for (let i = 1; i < candles.length; i++) {
  1156→  1156→      const prevClose = candles[i - 1].close;
  1157→  1157→      const currClose = candles[i].close;
  1158→  1158→      if (prevClose > 0) {
  1159→  1159→        const pctChange = Math.abs(currClose - prevClose) / prevClose;
  1160→  1160→        if (pctChange > 0.5) {
  1161→  1161→          driftCount++;
  1162→  1162→        }
  1163→  1163→      }
  1164→  1164→    }
  1165→  1165→    if (driftCount > candles.length * 0.1) {
  1166→  1166→      console.warn(`[TGJU] ${tgjuKey}: Possible data drift detected (${driftCount} suspicious jumps out of ${candles.length} candles)`);
  1167→  1167→    }
  1168→  1168→
  1169→  1169→    // Reverse: API returns oldest-first (asc), we want newest-last for charting
  1170→  1170→    const firstDate = candles[0].date;
  1171→  1171→    const lastDate = candles[candles.length - 1].date;
  1172→  1172→    if (firstDate > lastDate) {
  1173→  1173→      candles.reverse();
  1174→  1174→    }
  1175→  1175→
  1176→  1176→    // ── Freshness validation: reject data that is too old (e.g. from 1392) ──
  1177→  1177→    if (!isDataFresh(candles)) {
  1178→  1178→      console.warn(`[TGJU] ${tgjuKey}: Data is STALE (latest candle: ${candles[candles.length - 1]?.date}), REJECTING`);
  1179→  1179→      return null;
  1180→  1180→    }
  1181→  1181→
  1182→  1182→    console.log(`[TGJU] ${tgjuKey}: Parsed ${candles.length} candles (${candles[0]?.date} to ${candles[candles.length - 1]?.date})`);
  1183→  1183→
  1184→  1184→    return candles;
  1185→  1185→  } catch (err) {
  1186→  1186→    console.error(`[TGJU] Failed to parse chart data for ${tgjuKey}:`, err);
  1187→  1187→    return null;
  1188→  1188→  }
  1189→  1189→}
