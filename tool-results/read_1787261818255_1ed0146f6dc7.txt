   218→export default function Home() {

   227→  // Store last fetch params for auto-refresh

   229→  const refreshTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   230→
   231→  // Chart container ref for image export

   233→
   234→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number) => {
   235→    // TGJU instrument: fetch historical data via tgju.org chart API
   236→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   237→      setLoading(true);
   238→      setLoadingMessage('در حال دریافت داده‌های تاریخی ... (حدود ۱۵ ثانیه)');
   239→      setError(null);
   240→      setData(null);
   241→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId };
   242→      try {
   243→        const res = await fetch(`/api/tgju-analysis?key=${encodeURIComponent(tgjuKey)}`);
   244→        if (!res.ok) {
   245→          const err = await res.json();
   246→          throw new Error(err.error || 'خطا در دریافت داده‌های تاریخی');
   247→        }
   248→        const json = await res.json();
   249→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   250→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   251→        }
   252→        setData({ ...json, isTgju: true });
   253→        setActivePanel('visual');
   254→      } catch (err) {
   255→        setError(String(err));
   256→      } finally {
   257→        setLoading(false);
   258→        setLoadingMessage(null);
   259→      }
   260→      return;
   261→    }
   262→
   263→    // Index: fetch historical data via TSETMC CDN (webId), then finpy-tse (finpyIndex/finpySector)
   264→    if (category === 'index') {
   265→      setLoading(true);
   266→      setLoadingMessage('در حال دریافت داده‌های تاریخی شاخص ...');
   267→      setError(null);
   268→      setData(null);
   269→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId };
   270→      let gotError = false;
   271→      try {
   272→        // 1. Try webId (TSETMC CDN)
   273→        if (webId) {
   274→          setLoadingMessage('در حال دریافت داده‌های شاخص از TSETMC ...');
   275→          const res = await fetch(`/api/finpy-sector?webId=${encodeURIComponent(String(webId))}`);
   276→          if (res.ok) {
   277→            try {
   278→              const json = await res.json();
   279→              if (json.candles && json.candles.length > 0 && json.ta) {
   280→                setData(json);
   281→                setActivePanel('visual');
   282→                setLoading(false);
   283→                setLoadingMessage(null);
   284→                return;
   285→              }
   286→              if (json.error) { setError(json.error); gotError = true; }
   287→            } catch { /* invalid JSON */ }
   288→          } else { gotError = true; }
   289→        }
   290→
   291→        // 2. Main indices via finpy-tse
   292→        if (finpyIndex) {
   293→          setLoadingMessage('در حال دریافت داده‌های شاخص اصلی ...');
   294→          const res = await fetch(`/api/finpy-sector?indexKey=${encodeURIComponent(finpyIndex)}`);
   295→          if (res.ok) {
   296→            try {
   297→              const json = await res.json();
   298→              if (json.candles && json.candles.length > 0 && json.ta) {
   299→                setData(json);
   300→                setActivePanel('visual');
   301→                setLoading(false);
   302→                setLoadingMessage(null);
   303→                return;
   304→              }
   305→              if (json.error) { setError(json.error); gotError = true; }
   306→            } catch { /* invalid JSON */ }
   307→          } else { gotError = true; }
   308→        }
   309→
   310→        // 3. Sector indices via finpy-tse
   311→        if (finpySector) {
   312→          setLoadingMessage('در حال دریافت داده‌های شاخص گروه ...');
   313→          const res = await fetch(`/api/finpy-sector?sector=${encodeURIComponent(finpySector)}`);
   314→          if (res.ok) {
   315→            try {
   316→              const json = await res.json();
   317→              if (json.candles && json.candles.length > 0 && json.ta) {
   318→                setData(json);
   319→                setActivePanel('visual');
   320→                setLoading(false);
   321→                setLoadingMessage(null);
   322→                return;
   323→              }
   324→              if (json.error) { setError(json.error); gotError = true; }
   325→            } catch { /* invalid JSON */ }
   326→          } else { gotError = true; }
   327→        }
   328→
   329→        if (!gotError) {
   330→          setError('داده‌های تاریخی این شاخص در حال حاضر قابل دسترسی نیست.');
   331→        }
   332→      } catch (err) {
   333→        setError(String(err));
   334→      }
   335→      setLoading(false);
   336→      setLoadingMessage(null);
   337→      return;
   338→    }
   339→
   340→    // Regular TSE instrument: full TA analysis (تعدیل شده / adjusted prices only)
   341→    setLoading(true);
   342→    setError(null);
   343→    setData(null);
   344→    lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId };
