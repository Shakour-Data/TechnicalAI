    85→  const lastFetchRef = useRef<FetchParams | null>(null);
    86→  const refreshingRef = useRef(false);
    87→
    88→  // ── doRefresh callback ──
    89→  const doRefresh = useCallback(async () => {
    90→    if (refreshingRef.current || !lastFetchRef.current) return;

    92→    refreshingRef.current = true;
    93→    setRefreshing(true);

    95→      // TGJU

    97→        const res = await fetch(`/api/tgju-analysis?key=${encodeURIComponent(params.tgjuKey)}`);
    98→        if (res.ok) {
    99→          const json = await res.json();
   100→          if (json.candles && json.candles.length > 0 && json.ta) {
   101→            setData({ ...json, isTgju: true, category: params.category });
   102→          }
   103→        }
   104→        return;
   105→      }
   106→      // TSE Index (finpy-tse)
   107→      if (params.category === 'tse_index' && params.insCode) {
   108→        const res = await fetch(`/api/index-analysis?webId=${encodeURIComponent(params.insCode)}`);
   109→        if (res.ok) {
   110→          const json = await res.json();
   111→          if (json.candles && json.candles.length > 0 && json.ta) {
   112→            setData({ ...json, isIndex: true, category: params.category });
   113→          }
   114→        }
   115→        return;
   116→      }
   117→      // Index via finpy-sector
   118→      if (params.category === 'index') {
   119→        if (params.webId) {
   120→          const r = await fetch(`/api/finpy-sector?indexKey=${encodeURIComponent(params.webId)}`);
   121→          if (r.ok) {
   122→            const json = await r.json();
   123→            if (json.candles && json.candles.length > 0 && json.ta) {
   124→              setData(json);
   125→              return;
   126→            }
   127→          }
   128→        }
   129→        if (params.finpySector) {
   130→          const r = await fetch(`/api/finpy-sector?sector=${encodeURIComponent(params.finpySector)}`);
   131→          if (r.ok) {
   132→            const json = await r.json();
   133→            if (json.candles && json.candles.length > 0 && json.ta) {
   134→              setData(json);
   135→              return;
   136→            }
   137→          }
   138→        }
   139→        return;
   140→      }
   141→      // Regular TSE
   142→      const res = await fetch(`/api/analysis?symbol=${encodeURIComponent(params.symbol)}`);
   143→      if (res.ok) {
   144→        const json = await res.json();
   145→        setData(json);
   146→      }
   147→    } finally {
   148→      refreshingRef.current = false;
   149→      setRefreshing(false);
   150→    }

   277→      setError(String(err));
   278→    } finally {
   279→      setLoading(false);
   280→    }

   296→          <div className="flex items-center gap-3 shrink-0">

   317→              className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-medium transition-colors ${currentPage === 'help' ? 'bg-sky-50 text-sky-700 border border-sky-200' : 'text-gray-500 hover:text-gray-700 hover:bg-gray-50'}`}

   334→                </div>
   335→              </div>
   336→              {isTgjuData && (
   337→                <span className="px-2 py-1 rounded-lg bg-teal-50 border border-teal-200 text-teal-700 text-[10px] font-bold flex items-center gap-1">
   338→                  <Coins className="w-3 h-3" /> TGJU
   339→                </span>
   340→              )}
   341→              {isIndexData && (
   342→                <span className="px-2 py-1 rounded-lg bg-rose-50 border border-rose-200 text-rose-700 text-[10px] font-bold flex items-center gap-1">
   343→                  <BarChart3 className="w-3 h-3" /> شاخص (finpy-tse)

   346→              <div className={`px-3 py-1.5 rounded-lg border ${signalBg}`}>
   347→                <div className={`flex items-center gap-1.5 text-xs font-bold ${signalColor}`}>
   348→                  <SignalIcon className="w-3.5 h-3.5" />

   350→                </div>
