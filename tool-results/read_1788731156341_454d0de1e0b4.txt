   125→// ─── EMA (full array) ───────────────────────────────────────────────────────────
   126→function emaArray(data: number[], period: number): number[] {
   127→  const result: number[] = new Array(data.length).fill(0);
   128→  if (data.length < period) return result;
   129→  let sum = 0;
   130→  for (let i = 0; i < period; i++) sum += data[i];
   131→  result[period - 1] = sum / period;
   132→  const k = 2 / (period + 1);
   133→  for (let i = period; i < data.length; i++) {
   134→    result[i] = data[i] * k + result[i - 1] * (1 - k);
   135→  }
   136→  return result;
   137→}

   143→// ─── SMA (full array) ──────────────────────────────────────────────────────────
   144→function smaArray(closes: number[], period: number): number[] {
   145→  const result: number[] = new Array(closes.length).fill(0);
   146→  for (let i = period - 1; i < closes.length; i++) {
   147→    let sum = 0;
   148→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   149→    result[i] = sum / period;
   150→  }
   151→  return result;
   152→}

   154→// ─── WMA — Weighted Moving Average ─────────────────────────────────────────────
   155→function wmaArray(data: number[], period: number): number[] {
   156→  const result: number[] = new Array(data.length).fill(0);
   157→  if (data.length < period) return result;
   158→  const denom = period * (period + 1) / 2;
   159→  for (let i = period - 1; i < data.length; i++) {
   160→    let sum = 0;
   161→    for (let j = 0; j < period; j++) sum += data[i - j] * (period - j);
   162→    result[i] = sum / denom;
   163→  }
   164→  return result;
   165→}

   167→// ─── HMA — Hull Moving Average ─────────────────────────────────────────────────
   168→function hmaArray(closes: number[], period: number): number[] {
   169→  const halfPeriod = Math.max(1, Math.floor(period / 2));
   170→  const sqrtPeriod = Math.max(2, Math.floor(Math.sqrt(period)));
   171→  const wmaHalf = wmaArray(closes, halfPeriod);
   172→  const wmaFull = wmaArray(closes, period);
   173→  const diff = new Array(closes.length).fill(0);
   174→  for (let i = 0; i < closes.length; i++) diff[i] = 2 * wmaHalf[i] - wmaFull[i];
   175→  return wmaArray(diff, sqrtPeriod);
   176→}

   178→// ─── TMA — Triangular Moving Average (SMA of SMA) ──────────────────────────────
   179→function tmaArray(closes: number[], period: number): number[] {
   180→  const halfPeriod = Math.max(2, Math.ceil(period / 2));
   181→  const firstPass = smaArray(closes, period);
   182→  return smaArray(firstPass, halfPeriod);
   183→}

   185→// ─── LMA — Linear Regression (Least Squares) Moving Average ────────────────────
   186→function lmaArray(closes: number[], period: number): number[] {
   187→  const result: number[] = new Array(closes.length).fill(0);
   188→  if (closes.length < period) return result;
   189→  const sumX = period * (period - 1) / 2;
   190→  const sumX2 = (period - 1) * period * (2 * period - 1) / 6;
   191→  const denom = period * sumX2 - sumX * sumX;
   192→  if (denom === 0) return result;
   193→  for (let i = period - 1; i < closes.length; i++) {
   194→    let sumY = 0;
   195→    let sumXY = 0;
   196→    for (let j = 0; j < period; j++) {
   197→      const idx = i - period + 1 + j;
   198→      sumY += closes[idx];
   199→      sumXY += j * closes[idx];
   200→    }
   201→    const b = (period * sumXY - sumX * sumY) / denom;
   202→    const a = (sumY - b * sumX) / period;
   203→    result[i] = a + b * (period - 1);
   204→  }
   205→  return result;
   206→}

   208→// ─── VWMA — Volume Weighted Moving Average ──────────────────────────────────────
   209→function vwmaArray(closes: number[], volumes: number[], period: number): number[] {
   210→  const result: number[] = new Array(closes.length).fill(0);
   211→  if (closes.length < period) return result;
   212→  for (let i = period - 1; i < closes.length; i++) {
   213→    let pvSum = 0;
   214→    let vSum = 0;
   215→    for (let j = i - period + 1; j <= i; j++) {
   216→      pvSum += closes[j] * volumes[j];
   217→      vSum += volumes[j];
   218→    }
   219→    result[i] = vSum === 0 ? 0 : pvSum / vSum;
   220→  }
   221→  return result;
   222→}

   228→function ichimokuArray(
   229→  highs: number[],
   230→  lows: number[],
   231→  tenkanP: number = 9,
   232→  kijunP: number = 26,
   233→  senkouBP: number = 52,
   234→): { tenkan: number[]; kijun: number[]; senkouA: number[]; senkouB: number[] } {
   235→  const len = highs.length;
   236→  const tenkan: number[] = new Array(len).fill(0);
   237→  const kijun: number[] = new Array(len).fill(0);
   238→  const senkouA: number[] = new Array(len).fill(0);
   239→  const senkouB: number[] = new Array(len).fill(0);
   240→
   241→  for (let i = tenkanP - 1; i < len; i++) {
   242→    let hh = -Infinity, ll = Infinity;
   243→    for (let j = i - tenkanP + 1; j <= i; j++) {
   244→      if (highs[j] > hh) hh = highs[j];
   245→      if (lows[j] < ll) ll = lows[j];
   246→    }
   247→    tenkan[i] = (hh + ll) / 2;
   248→  }
   249→
   250→  for (let i = kijunP - 1; i < len; i++) {
   251→    let hh = -Infinity, ll = Infinity;
   252→    for (let j = i - kijunP + 1; j <= i; j++) {
   253→      if (highs[j] > hh) hh = highs[j];
   254→      if (lows[j] < ll) ll = lows[j];
   255→    }
   256→    kijun[i] = (hh + ll) / 2;
   257→  }
   258→
   259→  for (let i = kijunP - 1; i < len; i++) {
   260→    senkouA[i] = (tenkan[i] + kijun[i]) / 2;
   261→  }
   262→
   263→  for (let i = senkouBP - 1; i < len; i++) {
   264→    let hh = -Infinity, ll = Infinity;
   265→    for (let j = i - senkouBP + 1; j <= i; j++) {
   266→      if (highs[j] > hh) hh = highs[j];
   267→      if (lows[j] < ll) ll = lows[j];
   268→    }
   269→    senkouB[i] = (hh + ll) / 2;
   270→  }
   271→
   272→  return { tenkan, kijun, senkouA, senkouB };
   273→}

   301→    haLow[i] = Math.min(lows[i], haOpen[i], haClose[i]);
   302→  }

   304→  return { haOpen, haHigh, haLow, haClose };

   311→// ─── RSI (full array) ──────────────────────────────────────────────────────────
   312→function rsiArray(closes: number[], period: number = 14): number[] {
   313→  const result: number[] = new Array(closes.length).fill(0);
   314→  if (closes.length < period + 1) return result;
   315→
   316→  let gainSum = 0,
   317→    lossSum = 0;
   318→  for (let i = 1; i <= period; i++) {
   319→    const diff = closes[i] - closes[i - 1];
   320→    if (diff > 0) gainSum += diff;
   321→    else lossSum += Math.abs(diff);
   322→  }
   323→  let avgGain = gainSum / period;
   324→  let avgLoss = lossSum / period;
   325→  result[period] = avgLoss === 0 ? 100 : 100 - 100 / (1 + avgGain / avgLoss);
   326→
   327→  for (let i = period + 1; i < closes.length; i++) {
   328→    const diff = closes[i] - closes[i - 1];
   329→    const gain = diff > 0 ? diff : 0;
   330→    const loss = diff < 0 ? Math.abs(diff) : 0;
   331→    avgGain = (avgGain * (period - 1) + gain) / period;
   332→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   333→    result[i] = avgLoss === 0 ? 100 : 100 - 100 / (1 + avgGain / avgLoss);
   334→  }
   335→  return result;
   336→}

   338→// ─── Stochastic (full array) ───────────────────────────────────────────────────
   339→function stochasticArray(
   340→  highs: number[],
   341→  lows: number[],
   342→  closes: number[],
   343→  kPeriod: number = 14,
   344→  dPeriod: number = 3,
   345→): { k: number[]; d: number[] } {
   346→  const kArr: number[] = new Array(closes.length).fill(0);
   347→  const dArr: number[] = new Array(closes.length).fill(0);
   348→
   349→  for (let i = kPeriod - 1; i < closes.length; i++) {
   350→    let highest = -Infinity,
   351→      lowest = Infinity;
   352→    for (let j = i - kPeriod + 1; j <= i; j++) {
   353→      if (highs[j] > highest) highest = highs[j];
   354→      if (lows[j] < lowest) lowest = lows[j];
   355→    }
   356→    kArr[i] = highest === lowest ? 50 : ((closes[i] - lowest) / (highest - lowest)) * 100;
   357→  }
   358→
   359→  for (let i = kPeriod - 1 + dPeriod - 1; i < closes.length; i++) {
   360→    let sum = 0;
   361→    for (let j = i - dPeriod + 1; j <= i; j++) sum += kArr[j];
   362→    dArr[i] = sum / dPeriod;
   363→  }
   364→  return { k: kArr, d: dArr };
   365→}

   367→// ─── Stochastic RSI ────────────────────────────────────────────────────────────
   368→function stochRsiArray(
   369→  closes: number[],
   370→  rsiPeriod: number = 14,
   371→  stochPeriod: number = 14,
   372→  smoothPeriod: number = 3,
   373→): { k: number[]; d: number[] } {
   374→  const rsiValues = rsiArray(closes, rsiPeriod);
   375→  const kArr: number[] = new Array(closes.length).fill(0);
   376→  const dArr: number[] = new Array(closes.length).fill(0);
   377→  const startIdx = rsiPeriod - 1 + stochPeriod - 1;
   378→
   379→  for (let i = startIdx; i < closes.length; i++) {
   380→    let minRsi = Infinity,
   381→      maxRsi = -Infinity;
   382→    for (let j = i - stochPeriod + 1; j <= i; j++) {
   383→      if (rsiValues[j] < minRsi) minRsi = rsiValues[j];
   384→      if (rsiValues[j] > maxRsi) maxRsi = rsiValues[j];
   385→    }
   386→    kArr[i] = maxRsi === minRsi ? 50 : ((rsiValues[i] - minRsi) / (maxRsi - minRsi)) * 100;
   387→  }
   388→
   389→  for (let i = startIdx + smoothPeriod - 1; i < closes.length; i++) {
   390→    let sum = 0;
   391→    for (let j = i - smoothPeriod + 1; j <= i; j++) sum += kArr[j];
   392→    dArr[i] = sum / smoothPeriod;
   393→  }
   394→  return { k: kArr, d: dArr };
   395→}

   435→// ─── TRIX — Triple EMA Rate of Change ──────────────────────────────────────────
   436→function trixArray(closes: number[], period: number = 9): number[] {
   437→  const result: number[] = new Array(closes.length).fill(0);
   438→  const startIdx = 3 * (period - 1) + 1;
   439→  if (closes.length < startIdx) return result;
   440→
   441→  const ema1 = emaArray(closes, period);
   442→  const ema2 = emaArray(ema1, period);
   443→  const ema3 = emaArray(ema2, period);
   444→
   445→  for (let i = startIdx; i < closes.length; i++) {
   446→    if (ema3[i - 1] === 0) {
   447→      result[i] = 0;
   448→    } else {
   449→      result[i] = ((ema3[i] - ema3[i - 1]) / Math.abs(ema3[i - 1])) * 100;
   450→    }
   451→  }
   452→  return result;
   453→}

   455→// ─── CCI (full array) ──────────────────────────────────────────────────────────
   456→function cciArray(
   457→  highs: number[],
   458→  lows: number[],
   459→  closes: number[],
   460→  period: number = 20,
   461→): number[] {
   462→  const len = closes.length;
   463→  const result: number[] = new Array(len).fill(0);
   464→
   465→  for (let i = period - 1; i < len; i++) {
   466→    let tpSum = 0;
   467→    for (let j = i - period + 1; j <= i; j++) {
   468→      tpSum += (highs[j] + lows[j] + closes[j]) / 3;
   469→    }
   470→    const tpMean = tpSum / period;
   471→    let md = 0;
   472→    for (let j = i - period + 1; j <= i; j++) {
   473→      md += Math.abs((highs[j] + lows[j] + closes[j]) / 3 - tpMean);
   474→    }
   475→    md /= period;
   476→    result[i] = md === 0 ? 0 : ((highs[i] + lows[i] + closes[i]) / 3 - tpMean) / (0.015 * md);
   477→  }
   478→  return result;
   479→}

   481→// ─── MFI (full array) ──────────────────────────────────────────────────────────
   482→function mfiArray(
   483→  highs: number[],
   484→  lows: number[],
   485→  closes: number[],
   486→  volumes: number[],
   487→  period: number = 14,
   488→): number[] {
   489→  const len = closes.length;
   490→  const result: number[] = new Array(len).fill(0);
   491→
   492→  for (let i = period; i < len; i++) {
   493→    let posFlow = 0,
   494→      negFlow = 0;
   495→    for (let j = i - period + 1; j <= i; j++) {
   496→      const tp = (highs[j] + lows[j] + closes[j]) / 3;
   497→      const prevTp = (highs[j - 1] + lows[j - 1] + closes[j - 1]) / 3;
   498→      const mf = tp * volumes[j];
   499→      if (tp > prevTp) posFlow += mf;
   500→      else negFlow += mf;
   501→    }
   502→    result[i] = negFlow === 0 ? 100 : 100 - 100 / (1 + posFlow / negFlow);
   503→  }
   504→  return result;
   505→}

   507→// ─── MACD (full array) ─────────────────────────────────────────────────────────
   508→function macdArray(
   509→  closes: number[],
   510→  fast: number = 12,
   511→  slow: number = 26,
   512→  signal: number = 9,
   513→): { line: number[]; signalArr: number[]; histogram: number[] } {
   514→  const len = closes.length;
   515→  const line: number[] = new Array(len).fill(0);
   516→  const signalArr: number[] = new Array(len).fill(0);
   517→  const histogram: number[] = new Array(len).fill(0);
   518→
   519→  const emaFast = emaArray(closes, fast);
   520→  const emaSlow = emaArray(closes, slow);
   521→
   522→  for (let i = 0; i < len; i++) {
   523→    line[i] = emaFast[i] - emaSlow[i];
   524→  }
   525→
   526→  const macdSignalEma = emaArray(line, signal);
   527→  for (let i = 0; i < len; i++) {
   528→    signalArr[i] = macdSignalEma[i];
   529→    histogram[i] = line[i] - macdSignalEma[i];
   530→  }
   531→
   532→  return { line, signalArr, histogram };
   533→}

   535→// ─── ADX + DMI (full array) ────────────────────────────────────────────────────
   536→function adxArray(
   537→  highs: number[],
   538→  lows: number[],
   539→  closes: number[],
   540→  period: number = 14,
   541→): { adx: number[]; dmiPlus: number[]; dmiMinus: number[] } {
   542→  const len = closes.length;
   543→  const adxResult: number[] = new Array(len).fill(0);
   544→  const dmiPlusResult: number[] = new Array(len).fill(0);
   545→  const dmiMinusResult: number[] = new Array(len).fill(0);
   546→  if (len < period * 2 + 1) return { adx: adxResult, dmiPlus: dmiPlusResult, dmiMinus: dmiMinusResult };
   547→
   548→  const tr: number[] = new Array(len).fill(0);
   549→  const plusDM: number[] = new Array(len).fill(0);
   550→  const minusDM: number[] = new Array(len).fill(0);
   551→
   552→  for (let i = 1; i < len; i++) {
   553→    tr[i] = Math.max(highs[i] - lows[i], Math.abs(highs[i] - closes[i - 1]), Math.abs(lows[i] - closes[i - 1]));
   554→    const upMove = highs[i] - highs[i - 1];
   555→    const downMove = lows[i - 1] - lows[i];
   556→    plusDM[i] = upMove > downMove && upMove > 0 ? upMove : 0;
   557→    minusDM[i] = downMove > upMove && downMove > 0 ? downMove : 0;
   558→  }
   559→
   560→  const smooth = (arr: number[], start: number) => {
   561→    let sum = 0;
   562→    for (let i = start; i < start + period; i++) sum += arr[i];
   563→    const res: number[] = new Array(len).fill(0);
   564→    res[start + period - 1] = sum;
   565→    for (let i = start + period; i < len; i++) {
   566→      res[i] = res[i - 1] - res[i - 1] / period + arr[i];
   567→    }
   568→    return res;
   569→  };
   570→
   571→  const atrSmooth = smooth(tr, 1);
   572→  const sPlusDM = smooth(plusDM, 1);
   573→  const sMinusDM = smooth(minusDM, 1);
   574→
   575→  // Compute DI+ and DI-
   576→  for (let i = 1; i < len; i++) {
   577→    const atrVal = atrSmooth[i];
   578→    dmiPlusResult[i] = atrVal === 0 ? 0 : (sPlusDM[i] / atrVal) * 100;
   579→    dmiMinusResult[i] = atrVal === 0 ? 0 : (sMinusDM[i] / atrVal) * 100;
   580→  }
   581→
   582→  const dx: number[] = new Array(len).fill(0);
   583→  for (let i = 1; i < len; i++) {
   584→    const sum = sPlusDM[i] + sMinusDM[i];
   585→    dx[i] = sum === 0 ? 0 : (Math.abs(sPlusDM[i] - sMinusDM[i]) / sum) * 100;
   586→  }
   587→
   588→  // ADX = smoothed DX
   589→  let adxSum = 0;
   590→  const startIdx = period * 2;
   591→  for (let i = period + 1; i <= startIdx; i++) adxSum += dx[i];
   592→  adxResult[startIdx] = adxSum / period;
   593→  for (let i = startIdx + 1; i < len; i++) {
   594→    adxResult[i] = (adxResult[i - 1] * (period - 1) + dx[i]) / period;
   595→  }
   596→
   597→  return { adx: adxResult, dmiPlus: dmiPlusResult, dmiMinus: dmiMinusResult };
   598→}

   600→// ─── Parabolic SAR (full array) ────────────────────────────────────────────────
   601→function sarArray(highs: number[], lows: number[], closes: number[]): number[] {
   602→  const len = closes.length;
   603→  const result: number[] = new Array(len).fill(0);
   604→  if (len < 2) return result;
   605→
   606→  let isLong = closes[1] > closes[0];
   607→  let af = 0.02;
   608→  let ep = isLong ? highs[1] : lows[1];
   609→  result[0] = lows[0];
   610→  result[1] = isLong ? lows[0] : highs[0];
   611→
   612→  for (let i = 2; i < len; i++) {
   613→    result[i] = result[i - 1] + af * (ep - result[i - 1]);
   614→
   615→    if (isLong) {
   616→      if (lows[i] < result[i]) {
   617→        isLong = false;
   618→        result[i] = ep;
   619→        ep = lows[i];
   620→        af = 0.02;
   621→      } else {
   622→        if (highs[i] > ep) {
   623→          ep = highs[i];
   624→          af = Math.min(af + 0.02, 0.2);
   625→        }
   626→      }
   627→    } else {
   628→      if (highs[i] > result[i]) {
   629→        isLong = true;
   630→        result[i] = ep;
   631→        ep = highs[i];
   632→        af = 0.02;
   633→      } else {
   634→        if (lows[i] < ep) {
   635→          ep = lows[i];
   636→          af = Math.min(af + 0.02, 0.2);
   637→        }
   638→      }
   639→    }
   640→  }
   641→  return result;
   642→}

   644→// ─── Awesome Oscillator ────────────────────────────────────────────────────────
   645→function awesomeOscArray(highs: number[], lows: number[]): number[] {
   646→  const len = highs.length;
   647→  const median = new Array(len).fill(0);
   648→  for (let i = 0; i < len; i++) median[i] = (highs[i] + lows[i]) / 2;
   649→  const sma5 = smaArray(median, 5);
   650→  const sma34 = smaArray(median, 34);
   651→  const result: number[] = new Array(len).fill(0);
   652→  for (let i = 0; i < len; i++) result[i] = sma5[i] - sma34[i];
   653→  return result;
   654→}

   656→// ─── Fisher Transform ──────────────────────────────────────────────────────────
   657→function fisherTransformArray(highs: number[], lows: number[], period: number = 9): number[] {
   658→  const result: number[] = new Array(highs.length).fill(0);
   659→  if (highs.length < period) return result;
   660→
   661→  for (let i = period - 1; i < highs.length; i++) {
   662→    let maxH = -Infinity,
   663→      minL = Infinity;
   664→    for (let j = i - period + 1; j <= i; j++) {
   665→      if (highs[j] > maxH) maxH = highs[j];
   666→      if (lows[j] < minL) minL = lows[j];
   667→    }
   668→    const range = maxH - minL;
   669→    if (range === 0) {
   670→      result[i] = 0;
   671→      continue;
   672→    }
   673→    const raw = ((highs[i] + lows[i]) / 2 - minL) / range;
   674→    const x = Math.max(-0.999, Math.min(0.999, 2 * raw - 1));
   675→    result[i] = 0.5 * Math.log((1 + x) / (1 - x));
   676→  }
   677→  return result;
   678→}

   684→// ─── ATR (full array) ──────────────────────────────────────────────────────────
   685→function atrArray(highs: number[], lows: number[], closes: number[], period: number = 14): number[] {
   686→  const len = closes.length;
   687→  const tr: number[] = new Array(len).fill(0);
   688→  const result: number[] = new Array(len).fill(0);
   689→
   690→  for (let i = 1; i < len; i++) {
   691→    tr[i] = Math.max(
   692→      highs[i] - lows[i],
   693→      Math.abs(highs[i] - closes[i - 1]),
   694→      Math.abs(lows[i] - closes[i - 1]),
   695→    );
   696→  }
   697→
   698→  if (len < period + 1) return result;
   699→
   700→  let sum = 0;
   701→  for (let i = 1; i <= period; i++) sum += tr[i];
   702→  result[period] = sum / period;
   703→  for (let i = period + 1; i < len; i++) {
   704→    result[i] = (result[i - 1] * (period - 1) + tr[i]) / period;
   705→  }
   706→  return result;
   707→}

   709→// ─── Bollinger Bands (full array) ──────────────────────────────────────────────
   710→function bollingerArray(
   711→  closes: number[],
   712→  period: number = 20,
   713→  mult: number = 2,
   714→): { upper: number[]; middle: number[]; lower: number[]; width: number[] } {
   715→  const middle = smaArray(closes, period);
   716→  const upper: number[] = new Array(closes.length).fill(0);
   717→  const lower: number[] = new Array(closes.length).fill(0);
   718→  const width: number[] = new Array(closes.length).fill(0);
   719→
   720→  for (let i = period - 1; i < closes.length; i++) {
   721→    let sumSq = 0;
   722→    for (let j = i - period + 1; j <= i; j++) {
   723→      sumSq += (closes[j] - middle[i]) ** 2;
   724→    }
   725→    const stdDev = Math.sqrt(sumSq / period);
   726→    upper[i] = middle[i] + mult * stdDev;
   727→    lower[i] = middle[i] - mult * stdDev;
   728→    width[i] = middle[i] === 0 ? 0 : ((upper[i] - lower[i]) / middle[i]) * 100;
   729→  }
   730→  return { upper, middle, lower, width };
   731→}

   734→function keltnerArray(
   735→  closes: number[],
   736→  highs: number[],
   737→  lows: number[],
   738→  emaPeriod: number = 20,
   739→  atrPeriod: number = 20,
   740→  mult: number = 2,
   741→): { upper: number[]; middle: number[]; lower: number[] } {
   742→  const middle = emaArray(closes, emaPeriod);
   743→  const atr = atrArray(highs, lows, closes, atrPeriod);
   744→  const upper: number[] = new Array(closes.length).fill(0);
   745→  const lower: number[] = new Array(closes.length).fill(0);
   746→  for (let i = 0; i < closes.length; i++) {
   747→    upper[i] = middle[i] + mult * atr[i];
   748→    lower[i] = middle[i] - mult * atr[i];
   749→  }
   750→  return { upper, middle, lower };
   751→}

   753→// ─── Envelopes ──────────────────────────────────────────────────────────────────
   754→function envelopeArray(
   755→  closes: number[],
   756→  period: number = 20,
   757→  pct: number = 2.5,
   758→): { upper: number[]; lower: number[] } {
   759→  const middle = smaArray(closes, period);
   760→  const upper: number[] = new Array(closes.length).fill(0);
   761→  const lower: number[] = new Array(closes.length).fill(0);
   762→  const factorUp = 1 + pct / 100;
   763→  const factorDn = 1 - pct / 100;
   764→  for (let i = 0; i < closes.length; i++) {
   765→    upper[i] = middle[i] * factorUp;
   766→    lower[i] = middle[i] * factorDn;
   767→  }
   768→  return { upper, lower };
   769→}

   771→// ─── Historical Volatility (annualized) ────────────────────────────────────────
   772→function histVolatilityArray(closes: number[], period: number = 20): number[] {
   773→  const result: number[] = new Array(closes.length).fill(0);
   774→  if (closes.length < period + 1) return result;
   775→
   776→  const logReturns: number[] = new Array(closes.length).fill(0);
   777→  for (let i = 1; i < closes.length; i++) {
   778→    logReturns[i] = closes[i - 1] === 0 ? 0 : Math.log(closes[i] / closes[i - 1]);
   779→  }
   780→
   781→  for (let i = period; i < closes.length; i++) {
   782→    let sum = 0;
   783→    for (let j = i - period + 1; j <= i; j++) sum += logReturns[j];
   784→    const mean = sum / period;
   785→    let sumSq = 0;
   786→    for (let j = i - period + 1; j <= i; j++) sumSq += (logReturns[j] - mean) ** 2;
   787→    result[i] = Math.sqrt(sumSq / period) * Math.sqrt(252);
   788→  }
   789→  return result;
   790→}

   792→// ─── Standard Deviation ─────────────────────────────────────────────────────────
   793→function stdDevArray(closes: number[], period: number = 20): number[] {
   794→  const result: number[] = new Array(closes.length).fill(0);
   795→  const sma = smaArray(closes, period);
   796→  for (let i = period - 1; i < closes.length; i++) {
   797→    let sumSq = 0;
   798→    for (let j = i - period + 1; j <= i; j++) {
   799→      sumSq += (closes[j] - sma[i]) ** 2;
   800→    }
   801→    result[i] = Math.sqrt(sumSq / period);
   802→  }
   803→  return result;
   804→}

   810→// ─── OBV — On Balance Volume ────────────────────────────────────────────────────
   811→function obvArray(closes: number[], volumes: number[]): number[] {
   812→  const result: number[] = new Array(closes.length).fill(0);
   813→  if (closes.length === 0) return result;
   814→  result[0] = volumes[0];
   815→  for (let i = 1; i < closes.length; i++) {
   816→    if (closes[i] > closes[i - 1]) result[i] = result[i - 1] + volumes[i];
   817→    else if (closes[i] < closes[i - 1]) result[i] = result[i - 1] - volumes[i];
   818→    else result[i] = result[i - 1];
   819→  }
   820→  return result;
   821→}

   823→// ─── VPT — Volume Price Trend ──────────────────────────────────────────────────
   824→function vptArray(closes: number[], volumes: number[]): number[] {
   825→  const result: number[] = new Array(closes.length).fill(0);
   826→  for (let i = 1; i < closes.length; i++) {
   827→    const pctChange = closes[i - 1] === 0 ? 0 : (closes[i] - closes[i - 1]) / closes[i - 1];
   828→    result[i] = result[i - 1] + volumes[i] * pctChange;
   829→  }
   830→  return result;
   831→}

   840→  }

   844→// ─── Chaikin A/D Line ──────────────────────────────────────────────────────────
   845→function chaikinADArray(highs: number[], lows: number[], closes: number[], volumes: number[]): number[] {
   846→  const result: number[] = new Array(closes.length).fill(0);
   847→  for (let i = 0; i < closes.length; i++) {
   848→    const range = highs[i] - lows[i];
   849→    const clv = range === 0 ? 0 : ((closes[i] - lows[i]) - (highs[i] - closes[i])) / range;
   850→    result[i] = (i === 0 ? 0 : result[i - 1]) + clv * volumes[i];
   851→  }
   852→  return result;

   860→  const raw = new Array(len).fill(0);

   866→// ─── VAP — Volume at Price (simplified: rolling VWAP using typical price) ──────
   867→function vapArray(highs: number[], lows: number[], closes: number[], volumes: number[], period: number = 20): number[] {
   868→  const result: number[] = new Array(closes.length).fill(0);
   869→  if (closes.length < period) return result;
   870→  for (let i = period - 1; i < closes.length; i++) {
   871→    let pvSum = 0;
   872→    let vSum = 0;
   873→    for (let j = i - period + 1; j <= i; j++) {
   874→      const tp = (highs[j] + lows[j] + closes[j]) / 3;
   875→      pvSum += tp * volumes[j];
   876→      vSum += volumes[j];
   877→    }
   878→    result[i] = vSum === 0 ? 0 : pvSum / vSum;
   879→  }
   880→  return result;
   881→}

   883→// ─── VWAP — Volume Weighted Average Price (cumulative) ─────────────────────────
   884→function vwapArray(highs: number[], lows: number[], closes: number[], volumes: number[]): number[] {
   885→  const result: number[] = new Array(closes.length).fill(0);
   886→  if (closes.length === 0) return result;
   887→  let cumPV = 0;
   888→  let cumV = 0;
   889→  for (let i = 0; i < closes.length; i++) {
   890→    const tp = (highs[i] + lows[i] + closes[i]) / 3;
   891→    cumPV += tp * volumes[i];
   892→    cumV += volumes[i];
   893→    result[i] = cumV === 0 ? 0 : cumPV / cumV;
   894→  }
   895→  return result;
   896→}
