   441→/* ─── In-memory cache ────────────────────────────────── */
   442→
   443→let quotesCache: YahooQuote[] | null = null;
   444→let quotesCacheTime = 0;
   445→const CACHE_TTL = 120_000; // 2 minutes
   446→
   447→/* ─── Fetch live quotes ──────────────────────────────── */
   448→
   449→export async function fetchYahooQuotes(): Promise<YahooQuote[]> {
   450→  const now = Date.now();
   451→  if (quotesCache && now - quotesCacheTime < CACHE_TTL) {
   452→    return quotesCache;
   453→  }
   454→
   455→  const symbols = ALL_YAHOO_INSTRUMENTS.map((s) => s.symbol);
   456→  const quotes: YahooQuote[] = [];
   457→
   458→  // Fetch in batches of 20 (Yahoo API limit)
   459→  const BATCH_SIZE = 20;
   460→  for (let i = 0; i < symbols.length; i += BATCH_SIZE) {
   461→    const batch = symbols.slice(i, i + BATCH_SIZE);
   462→    try {
   463→      const results = await yahooFinance.quote(batch, {
   464→        fields: [
   465→          'symbol', 'regularMarketPrice', 'regularMarketChange', 'regularMarketChangePercent',
   466→          'regularMarketPreviousClose', 'regularMarketOpen', 'regularMarketDayHigh',
   467→          'regularMarketDayLow', 'regularMarketVolume', 'marketCap', 'currency',
   468→          'shortName', 'longName',
   469→        ],
   470→      });
   471→
   472→      for (const r of results) {
   473→        const def = ALL_YAHOO_INSTRUMENTS.find((s) => s.symbol === r.symbol);
   474→        if (!def) continue;
   475→        const unit = r.currency || 'USD';
   476→        quotes.push({
   477→          symbol: r.symbol,
   478→          name: def.name,
   479→          nameEn: def.nameEn,
   480→          country: def.country,
   481→          countryEn: def.countryEn,
   482→          exchange: def.exchange,
   483→          groupTitle: def.groupTitle,
   484→          category: def.category,
   485→          unit,
   486→          price: r.regularMarketPrice ?? 0,
   487→          change: r.regularMarketChange ?? 0,
   488→          changePercent: r.regularMarketChangePercent ?? 0,
   489→          previousClose: r.regularMarketPreviousClose ?? 0,
   490→          open: r.regularMarketOpen ?? 0,
   491→          high: r.regularMarketDayHigh ?? 0,
   492→          low: r.regularMarketDayLow ?? 0,
   493→          volume: r.regularMarketVolume ?? 0,
   494→          marketCap: r.marketCap ?? 0,
   495→          currency: r.currency || 'USD',
   496→        });
   497→      }
   498→    } catch (err) {
   499→      console.error(`[Yahoo Finance] Error fetching batch ${Math.floor(i / BATCH_SIZE) + 1}:`, err);
   500→    }
   501→  }
   502→
   503→  quotesCache = quotes;
   504→  quotesCacheTime = now;
   505→  return quotes;
   506→}

   508→/* ─── Fetch historical data ──────────────────────────── */
   509→
   510→export async function fetchYahooHistory(symbol: string, days: number = 365): Promise<YahooOHLC[]> {
   511→  const end = new Date();
   512→  const start = new Date();
   513→  start.setDate(end.getDate() - days);
   514→
   515→  const result = await yahooFinance.chart(symbol, {
   516→    period1: start,
   517→    period2: end,
   518→    interval: '1d',
   519→  });
   520→
   521→  if (!result.quotes || result.quotes.length === 0) {
   522→    return [];
   523→  }
   524→
   525→  return result.quotes
   526→    .filter((q) => q.close != null && q.open != null)
   527→    .map((q) => ({
   528→      date: q.date?.toISOString().split('T')[0] || '',
   529→      open: q.open!,
   530→      high: q.high || q.close!,
   531→      low: q.low || q.close!,
   532→      close: q.close!,
   533→      volume: q.volume || 0,
   534→    }));
   535→}
