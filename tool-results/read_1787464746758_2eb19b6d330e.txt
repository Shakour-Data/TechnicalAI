   130→   130→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   131→   131→function sma(closes: number[], period: number): number {
   132→   132→  if (closes.length < period) return 0;
   133→   133→  const slice = closes.slice(-period);
   134→   134→  return slice.reduce((a, b) => a + b, 0) / period;
   135→   135→}

   137→   137→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   138→   138→function emaCalc(closes: number[], period: number): number {
   139→   139→  if (closes.length < period) return 0;
   140→   140→  const k = 2 / (period + 1);
   141→   141→  let emaVal = sma(closes.slice(0, period), period);
   142→   142→  for (let i = period; i < closes.length; i++) {
   143→   143→    emaVal = closes[i] * k + emaVal * (1 - k);
   144→   144→  }
   145→   145→  return emaVal;
   146→   146→}

   148→   148→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   149→   149→function calcRSI(closes: number[], period: number = 14): number {
   150→   150→  if (closes.length < period + 1) return 50;
   151→   151→  let avgGain = 0;
   152→   152→  let avgLoss = 0;
   153→   153→  for (let i = 1; i <= period; i++) {
   154→   154→    const diff = closes[i] - closes[i - 1];
   155→   155→    if (diff > 0) avgGain += diff;
   156→   156→    else avgLoss += Math.abs(diff);
   157→   157→  }
   158→   158→  avgGain /= period;
   159→   159→  avgLoss /= period;
   160→   160→  for (let i = period + 1; i < closes.length; i++) {
   161→   161→    const diff = closes[i] - closes[i - 1];
   162→   162→    const gain = diff > 0 ? diff : 0;
   163→   163→    const loss = diff < 0 ? Math.abs(diff) : 0;
   164→   164→    avgGain = (avgGain * (period - 1) + gain) / period;
   165→   165→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   166→   166→  }
   167→   167→  if (avgLoss === 0) return 100;
   168→   168→  const rs = avgGain / avgLoss;
   169→   169→  return 100 - 100 / (1 + rs);
   170→   170→}

   172→   172→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   173→   173→function calcMFI(data: OHLCV[], period: number = 14): number {
   174→   174→  if (data.length < period + 1) return 50;
   175→   175→  let posFlow = 0;
   176→   176→  let negFlow = 0;
   177→   177→  for (let i = data.length - period; i < data.length; i++) {
   178→   178→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   179→   179→    const mf = tp * data[i].volume;
   180→   180→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   181→   181→    if (tp > prevTp) posFlow += mf;
   182→   182→    else negFlow += mf;
   183→   183→  }
   184→   184→  if (negFlow === 0) return 100;
   185→   185→  const mfr = posFlow / negFlow;
   186→   186→  return 100 - 100 / (1 + mfr);
   187→   187→}

   190→   190→function calcCCI(data: OHLCV[], period: number = 20): number {
   191→   191→  if (data.length < period) return 0;
   192→   192→  const tps: number[] = [];
   193→   193→  for (let i = data.length - period; i < data.length; i++) {
   194→   194→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   195→   195→  }
   196→   196→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   197→   197→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   198→   198→  if (meanDev === 0) return 0;
   199→   199→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   200→   200→}

   202→   202→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   203→   203→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   204→   204→  if (data.length < kPeriod) return { k: 50, d: 50 };
   205→   205→  const rawKs: number[] = [];
   206→   206→  for (let i = kPeriod - 1; i < data.length; i++) {
   207→   207→    let lowest = Infinity;
   208→   208→    let highest = -Infinity;
   209→   209→    for (let j = i - kPeriod + 1; j <= i; j++) {
   210→   210→      if (data[j].low < lowest) lowest = data[j].low;
   211→   211→      if (data[j].high > highest) highest = data[j].high;
   212→   212→    }
   213→   213→    const range = highest - lowest;
   214→   214→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   215→   215→  }
   216→   216→  const smoothedK: number[] = [];
   217→   217→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   218→   218→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   219→   219→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   220→   220→  }
   221→   221→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   222→   222→  const dSlice = smoothedK.slice(-smoothD);
   223→   223→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   224→   224→  return { k, d };
   225→   225→}

   228→   228→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   229→   229→  if (data.length < period) return -50;
   230→   230→  const slice = data.slice(-period);
   231→   231→  let highest = -Infinity;
   232→   232→  let lowest = Infinity;
   233→   233→  for (const c of slice) {
   234→   234→    if (c.high > highest) highest = c.high;
   235→   235→    if (c.low < lowest) lowest = c.low;
   236→   236→  }
   237→   237→  const range = highest - lowest;
   238→   238→  if (range === 0) return -50;
   239→   239→  return ((highest - data[data.length - 1].close) / range) * -100;
   240→   240→}

   242→   242→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   243→   243→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   244→   244→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   245→   245→  const emaFast = emaCalc(closes, fast);
   246→   246→  const emaSlow = emaCalc(closes, slow);
   247→   247→  const macdLine = emaFast - emaSlow;
   248→   248→  // Calculate signal as EMA of MACD values
   249→   249→  const macdValues: number[] = [];
   250→   250→  const kFast = 2 / (fast + 1);
   251→   251→  const kSlow = 2 / (slow + 1);
   252→   252→  let eFast = sma(closes.slice(0, fast), fast);
   253→   253→  let eSlow = sma(closes.slice(0, slow), slow);
   254→   254→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   255→   255→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   256→   256→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   257→   257→    macdValues.push(eFast - eSlow);
   258→   258→  }
   259→   259→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   260→   260→  const kSig = 2 / (sig + 1);
   261→   261→  let signalVal = sma(macdValues.slice(0, sig), sig);
   262→   262→  for (let i = sig; i < macdValues.length; i++) {
   263→   263→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   264→   264→  }
   265→   265→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   266→   266→}

   268→   268→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   269→   269→function calcATR(data: OHLCV[], period: number = 14): number {
   270→   270→  if (data.length < 2) return 0;
   271→   271→  const trs: number[] = [];
   272→   272→  for (let i = 1; i < data.length; i++) {
   273→   273→    const tr = Math.max(
   274→   274→      data[i].high - data[i].low,
   275→   275→      Math.abs(data[i].high - data[i - 1].close),
   276→   276→      Math.abs(data[i].low - data[i - 1].close)
   277→   277→    );
   278→   278→    trs.push(tr);
   279→   279→  }
   280→   280→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   281→   281→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   282→   282→  for (let i = period; i < trs.length; i++) {
   283→   283→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   284→   284→  }
   285→   285→  return atrVal;
   286→   286→}

   288→   288→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   289→   289→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   290→   290→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   291→   291→  const plusDMs: number[] = [];
   292→   292→  const minusDMs: number[] = [];
   293→   293→  const trs: number[] = [];
   294→   294→  for (let i = 1; i < data.length; i++) {
   295→   295→    const upMove = data[i].high - data[i - 1].high;
   296→   296→    const downMove = data[i - 1].low - data[i].low;
   297→   297→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   298→   298→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   299→   299→    trs.push(Math.max(
   300→   300→      data[i].high - data[i].low,
   301→   301→      Math.abs(data[i].high - data[i - 1].close),
   302→   302→      Math.abs(data[i].low - data[i - 1].close)
   303→   303→    ));
   304→   304→  }
   305→   305→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   306→   306→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   307→   307→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   308→   308→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   309→   309→  const dxValues: number[] = [];
   310→   310→  for (let i = period; i < trs.length; i++) {
   311→   311→    smoothTR = smoothTR - smoothTR / period + trs[i];
   312→   312→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   313→   313→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   314→   314→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   315→   315→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   316→   316→    const diSum = diPlus + diMinus;
   317→   317→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   318→   318→    dxValues.push(dx);
   319→   319→  }
   320→   320→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   321→   321→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   322→   322→  for (let i = period; i < dxValues.length; i++) {
   323→   323→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   324→   324→  }
   325→   325→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   326→   326→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   327→   327→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   328→   328→}

   330→   330→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   331→   331→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   332→   332→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   333→   333→  let isLong = true;
   334→   334→  let af = step;
   335→   335→  let ep = data[0].high;
   336→   336→  let sar = data[0].low;
   337→   337→  for (let i = 1; i < data.length; i++) {
   338→   338→    const prevSar = sar;
   339→   339→    sar = prevSar + af * (ep - prevSar);
   340→   340→    // Ensure SAR is within bounds
   341→   341→    if (isLong) {
   342→   342→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   343→   343→      if (data[i].low < sar) {
   344→   344→        isLong = false;
   345→   345→        sar = ep;
   346→   346→        ep = data[i].low;
   347→   347→        af = step;
   348→   348→      } else {
   349→   349→        if (data[i].high > ep) {
   350→   350→          ep = data[i].high;
   351→   351→          af = Math.min(af + step, max);
   352→   352→        }
   353→   353→      }
   354→   354→    } else {
   355→   355→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   356→   356→      if (data[i].high > sar) {
   357→   357→        isLong = true;
   358→   358→        sar = ep;
   359→   359→        ep = data[i].high;
   360→   360→        af = step;
   361→   361→      } else {
   362→   362→        if (data[i].low < ep) {
   363→   363→          ep = data[i].low;
   364→   364→          af = Math.min(af + step, max);
   365→   365→        }
   366→   366→      }
   367→   367→    }
   368→   368→  }
   369→   369→  return sar;
   370→   370→}

   372→   372→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   373→   373→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   374→   374→  if (closes.length < period) {
   375→   375→    const c = closes[closes.length - 1] ?? 0;
   376→   376→    return { upper: c, middle: c, lower: c };
   377→   377→  }
   378→   378→  const slice = closes.slice(-period);
   379→   379→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   380→   380→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   381→   381→  const sd = Math.sqrt(variance);
   382→   382→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   383→   383→}

   386→   386→function calcOBV(data: OHLCV[]): number {
   387→   387→  let obv = 0;
   388→   388→  for (let i = 1; i < data.length; i++) {
   389→   389→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   390→   390→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   391→   391→  }
   392→   392→  return obv;
   393→   393→}

   395→   395→// ─── Helper: VWAP ──────────────────────────────────────────────────────────────
   396→   396→function calcVWAP(data: OHLCV[]): number {
   397→   397→  if (data.length === 0) return 0;
   398→   398→  let cumTPV = 0;
   399→   399→  let cumVol = 0;
   400→   400→  for (const d of data) {
   401→   401→    const tp = (d.high + d.low + d.close) / 3;
   402→   402→    cumTPV += tp * d.volume;
   403→   403→    cumVol += d.volume;
   404→   404→  }
   405→   405→  return cumVol === 0 ? data[data.length - 1].close : cumTPV / cumVol;
   406→   406→}

   408→   408→function calcVWAPArray(data: OHLCV[]): number[] {

   422→   422→function calcIchimoku(data: OHLCV[]): {
   423→   423→  tenkan: number; kijun: number; senkouA: number; senkouB: number; chikou: number;
   424→   424→} {
   425→   425→  if (data.length < 52) {
   426→   426→    const c = data[data.length - 1]?.close ?? 0;
   427→   427→    return { tenkan: c, kijun: c, senkouA: c, senkouB: c, chikou: c };
   428→   428→  }
   429→   429→  const highs = data.map(d => d.high);
   430→   430→  const lows = data.map(d => d.low);
   431→   431→  const closes = data.map(d => d.close);
   432→   432→
   433→   433→  function donchian(h: number[], l: number[], end: number, period: number): number {
   434→   434→    const start = end - period + 1;
   435→   435→    if (start < 0) return (h[end] + l[end]) / 2;
   436→   436→    let hi = -Infinity, lo = Infinity;
   437→   437→    for (let i = start; i <= end; i++) {
   438→   438→      if (h[i] > hi) hi = h[i];
   439→   439→      if (l[i] < lo) lo = l[i];
   440→   440→    }
   441→   441→    return (hi + lo) / 2;
   442→   442→  }
   443→   443→
   444→   444→  const n = data.length;
   445→   445→  const tenkan = donchian(highs, lows, n - 1, 9);
   446→   446→  const kijun = donchian(highs, lows, n - 1, 26);
   447→   447→  // Senkou A = (tenkan + kijun) / 2, shifted 26 periods ahead
   448→   448→  // We compute it at bar n-27 (shifted to current)
   449→   449→  const senkouAIdx = Math.max(n - 1 - 26, 0);
   450→   450→  const tA = donchian(highs, lows, senkouAIdx, 9);
   451→   451→  const kA = donchian(highs, lows, senkouAIdx, 26);
   452→   452→  const senkouA = (tA + kA) / 2;
   453→   453→  // Senkou B = donchian(52), shifted 26 periods ahead
   454→   454→  const senkouBIdx = Math.max(n - 1 - 26, 0);
   455→   455→  const senkouB = donchian(highs, lows, senkouBIdx, 52);
   456→   456→  // Chikou = current close plotted 26 periods back
   457→   457→  const chikou = closes[n - 1];
   458→   458→
   459→   459→  return { tenkan, kijun, senkouA, senkouB, chikou };
   460→   460→}

   470→   470→    const start = end - period + 1;

   489→   489→    const sIdx = i - 26;

   491→   491→      const t = donchian(sIdx, 9);
   492→   492→      const k = donchian(sIdx, 26);
   493→   493→      senkouA.push((t + k) / 2);

   495→   495→      senkouA.push(donchian(i, 9)); // fallback
   496→   496→    }
   497→   497→    // Senkou B shifted +26: at bar i, we need donchian(52) at bar i-26
   498→   498→    if (sIdx >= 0) {
   499→   499→      senkouB.push(donchian(sIdx, 52));
   500→   500→    } else {
   501→   501→      senkouB.push(donchian(i, 52)); // fallback
   502→   502→    }
   503→   503→  }
   504→   504→
   505→   505→  return { tenkan, kijun, senkouA, senkouB };

   508→   508→// ─── Helper: SMA Array (full series) ──────────────────────────────────────
   509→   509→function smaArray(closes: number[], period: number): number[] {
   510→   510→  const result: number[] = [];
   511→   511→  for (let i = 0; i < closes.length; i++) {
   512→   512→    if (i < period - 1) { result.push(0); continue; }
   513→   513→    let sum = 0;
   514→   514→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   515→   515→    result.push(sum / period);
   516→   516→  }
   517→   517→  return result;
   518→   518→}

   520→   520→// ─── Helper: EMA Array (full series) ──────────────────────────────────────
   521→   521→function emaArrayCalc(closes: number[], period: number): number[] {
   522→   522→  const result: number[] = [];
   523→   523→  if (closes.length < period) return closes.map(() => 0);
   524→   524→  const k = 2 / (period + 1);
   525→   525→  let emaVal = 0;
   526→   526→  for (let i = 0; i < period; i++) emaVal += closes[i];
   527→   527→  emaVal /= period;
   528→   528→  for (let i = 0; i < closes.length; i++) {
   529→   529→    if (i < period) { result.push(0); continue; }
   530→   530→    if (i === period) { result.push(emaVal); continue; }
   531→   531→    emaVal = closes[i] * k + emaVal * (1 - k);
   532→   532→    result.push(emaVal);
   533→   533→  }
   534→   534→  return result;
   535→   535→}

   537→   537→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   538→   538→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   539→   539→  const n = values.length;
   540→   540→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   541→   541→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   542→   542→  for (let i = 0; i < n; i++) {
   543→   543→    sumX += i;
   544→   544→    sumY += values[i];
   545→   545→    sumXY += i * values[i];
   546→   546→    sumX2 += i * i;
   547→   547→    sumY2 += values[i] * values[i];
   548→   548→  }
   549→   549→  const denom = n * sumX2 - sumX * sumX;
   550→   550→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   551→   551→  const slope = (n * sumXY - sumX * sumY) / denom;
   552→   552→  const intercept = (sumY - slope * sumX) / n;
   553→   553→  const meanY = sumY / n;
   554→   554→  let ssRes = 0, ssTot = 0;
   555→   555→  for (let i = 0; i < n; i++) {
   556→   556→    const predicted = slope * i + intercept;
   557→   557→    ssRes += (values[i] - predicted) ** 2;
   558→   558→    ssTot += (values[i] - meanY) ** 2;
   559→   559→  }
   560→   560→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   561→   561→  return { slope, intercept, r2 };
   562→   562→}

   657→   657→// ─── Helper: Linear System Solver (Gaussian Elimination) ─────────────────
   658→   658→function solveLinearSystem(A: number[][], b: number[]): number[] | null {
   659→   659→  const n = b.length;
   660→   660→  const aug = A.map((row, i) => [...row, b[i]]);
   661→   661→
   662→   662→  for (let col = 0; col < n; col++) {
   663→   663→    let maxVal = Math.abs(aug[col][col]);
   664→   664→    let maxRow = col;
   665→   665→    for (let row = col + 1; row < n; row++) {
   666→   666→      if (Math.abs(aug[row][col]) > maxVal) {
   667→   667→        maxVal = Math.abs(aug[row][col]);
   668→   668→        maxRow = row;
   669→   669→      }
   670→   670→    }
   671→   671→    if (maxVal < 1e-10) return null;
   672→   672→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   673→   673→
   674→   674→    for (let row = col + 1; row < n; row++) {
   675→   675→      const factor = aug[row][col] / aug[col][col];
   676→   676→      for (let j = col; j <= n; j++) {
   677→   677→        aug[row][j] -= factor * aug[col][j];
   678→   678→      }
   679→   679→    }
   680→   680→  }
   681→   681→
   682→   682→  const x = new Array(n).fill(0);
   683→   683→  for (let i = n - 1; i >= 0; i--) {
   684→   684→    x[i] = aug[i][n];
   685→   685→    for (let j = i + 1; j < n; j++) {
   686→   686→      x[i] -= aug[i][j] * x[j];
   687→   687→    }
   688→   688→    if (Math.abs(aug[i][i]) < 1e-10) return null;
   689→   689→    x[i] /= aug[i][i];
   690→   690→  }
   691→   691→
   692→   692→  return x;
   693→   693→}

   695→   695→// ─── Helper: Generate All 7-Method S/R Levels ─────────────────────────────
   696→   696→function generateAllLevels(data: OHLCV[], endIdx: number): RawSRLevel[] {
   697→   697→  const levels: RawSRLevel[] = [];
   698→   698→  if (endIdx < 10) return levels;
   699→   699→
   700→   700→  const slice = data.slice(0, endIdx + 1);
   701→   701→  const currentPrice = slice[slice.length - 1].close;
   702→   702→  const closes = slice.map(d => d.close);
   703→   703→  const noFib: [string, string] = ['—', ''];
   704→   704→
   705→   705→  // ── Method 1: Swing High/Low (Price Action) ──────────────────────────
   706→   706→  for (const lookback of [3, 5, 7]) {
   707→   707→    const swings = findSwingLevels(slice, lookback);
   708→   708→    const tol = currentPrice * 0.005; // 0.5% tolerance
   709→   709→
   710→   710→    // Connect swing highs → resistance
   711→   711→    const usedHigh = new Set<number>();
   712→   712→    for (const h of swings.highs) {
   713→   713→      if (usedHigh.has(h.idx)) continue;
   714→   714→      let sum = h.price;
   715→   715→      let count = 1;
   716→   716→      for (const h2 of swings.highs) {
   717→   717→        if (h2 === h || usedHigh.has(h2.idx)) continue;
   718→   718→        if (Math.abs(h2.price - h.price) < tol) {
   719→   719→          sum += h2.price;
   720→   720→          count++;
   721→   721→          usedHigh.add(h2.idx);
   722→   722→        }
   723→   723→      }
   724→   724→      usedHigh.add(h.idx);
   725→   725→      if (sum / count > currentPrice * 1.005) {
   726→   726→        levels.push({ price: sum / count, method: 'Swing_High', type: 'resistance', ...noFib });
   727→   727→      }
   728→   728→    }
   729→   729→
   730→   730→    // Connect swing lows → support
   731→   731→    const usedLow = new Set<number>();
   732→   732→    for (const l of swings.lows) {
   733→   733→      if (usedLow.has(l.idx)) continue;
   734→   734→      let sum = l.price;
   735→   735→      let count = 1;
   736→   736→      for (const l2 of swings.lows) {
   737→   737→        if (l2 === l || usedLow.has(l2.idx)) continue;
   738→   738→        if (Math.abs(l2.price - l.price) < tol) {
   739→   739→          sum += l2.price;
   740→   740→          count++;
   741→   741→          usedLow.add(l2.idx);
   742→   742→        }
   743→   743→      }
   744→   744→      usedLow.add(l.idx);
   745→   745→      if (sum / count < currentPrice * 0.995) {
   746→   746→        levels.push({ price: sum / count, method: 'Swing_Low', type: 'support', ...noFib });
   747→   747→      }
   748→   748→    }
   749→   749→  }
   750→   750→
   751→   751→  // ── Method 2: SMA Dynamic S/R ──────────────────────────────────────
   752→   752→  for (const period of [20, 50, 100, 200]) {
   753→   753→    const smaVal = sma(closes, period);
   754→   754→    if (smaVal <= 0) continue;
   755→   755→    const type: 'support' | 'resistance' = currentPrice > smaVal ? 'support' : 'resistance';
   756→   756→    levels.push({ price: smaVal, method: `SMA_${period}`, type, ...noFib });
   757→   757→  }
   758→   758→
   759→   759→  // ── Method 3: Bollinger Bands ───────────────────────────────────────
   760→   760→  const bb = calcBollingerBands(closes);
   761→   761→  if (bb.upper > 0 && bb.lower > 0) {
   762→   762→    levels.push({ price: bb.upper, method: 'BB_Upper', type: 'resistance', ...noFib });
   763→   763→    levels.push({ price: bb.lower, method: 'BB_Lower', type: 'support', ...noFib });
   764→   764→  }
   765→   765→
   766→   766→  // ── Method 4: Fibonacci Retracement + Extension ─────────────────────
   767→   767→  interface SwingPair { high: number; highIdx: number; low: number; lowIdx: number; range: number }
   768→   768→  let bestSwing: SwingPair | null = null;
   769→   769→
   770→   770→  for (const lb of [3, 5, 7, 10, 15, 20]) {
   771→   771→    const sw = findSwingLevels(slice, lb);
   772→   772→    if (sw.highs.length === 0 || sw.lows.length === 0) continue;
   773→   773→    for (const h of sw.highs) {
   774→   774→      for (const l of sw.lows) {
   775→   775→        if (Math.abs(h.idx - l.idx) < 3) continue;
   776→   776→        const range = Math.abs(h.price - l.price);
   777→   777→        if (range / currentPrice < 0.03) continue;
   778→   778→        if (!bestSwing || range > bestSwing.range) {
   779→   779→          bestSwing = {
   780→   780→            high: Math.max(h.price, l.price),
   781→   781→            highIdx: h.price > l.price ? h.idx : l.idx,
   782→   782→            low: Math.min(h.price, l.price),
   783→   783→            lowIdx: h.price < l.price ? l.idx : h.idx,
   784→   784→            range,
   785→   785→          };
   786→   786→        }
   787→   787→      }
   788→   788→    }
   789→   789→  }
   790→   790→
   791→   791→  if (bestSwing) {
   792→   792→    const H = bestSwing.high;
   793→   793→    const L = bestSwing.low;
   794→   794→    const range = H - L;
   795→   795→
   796→   796→    for (const ratio of FIB_RETRACEMENTS) {
   797→   797→      const price = H - ratio * range;
   798→   798→      if (price <= 0) continue;
   799→   799→      const rounded = roundToNice(price);
   800→   800→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   801→   801→      levels.push({
   802→   802→        price: rounded, method: `Fibonacci_${ratio}`, type,
   803→   803→        fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   804→   804→      });
   805→   805→    }
   806→   806→
   807→   807→    for (const extRatio of FIB_EXTENSIONS) {
   808→   808→      const price = H + (extRatio - 1) * range;
   809→   809→      if (price <= 0) continue;
   810→   810→      const rounded = roundToNice(price);
   811→   811→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   812→   812→      levels.push({
   813→   813→        price: rounded, method: `Fibonacci_${extRatio}`, type,
   814→   814→        fibRatio: String(extRatio), fibLabel: FIB_LABELS[String(extRatio)] || '',
   815→   815→      });
   816→   816→    }
   817→   817→
   818→   818→    // Inverse extensions (below L)
   819→   819→    for (const extRatio of FIB_EXTENSIONS) {
   820→   820→      const price = L - (extRatio - 1) * range;
   821→   821→      if (price <= 0) continue;
   822→   822→      const rounded = roundToNice(price);
   823→   823→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   824→   824→      levels.push({
   825→   825→        price: rounded, method: `Fibonacci_${-extRatio}`, type,
   826→   826→        fibRatio: String(-extRatio), fibLabel: '',
   827→   827→      });
   828→   828→    }
   829→   829→
   830→   830→    // Secondary swing confluence (check overlap with secondary swings)
   831→   831→    const secondarySwings: SwingPair[] = [];
   832→   832→    for (const lb of [3, 5, 7]) {
   833→   833→      const sw = findSwingLevels(slice, lb);
   834→   834→      for (const h of sw.highs) {
   835→   835→        for (const l of sw.lows) {
   836→   836→          if (Math.abs(h.idx - l.idx) < 3) continue;
   837→   837→          const r = Math.abs(h.price - l.price);
   838→   838→          if (r / currentPrice < 0.05) continue;
   839→   839→          if (bestSwing && Math.abs(r - bestSwing.range) / bestSwing.range < 0.1) continue;
   840→   840→          secondarySwings.push({
   841→   841→            high: Math.max(h.price, l.price), highIdx: h.price > l.price ? h.idx : l.idx,
   842→   842→            low: Math.min(h.price, l.price), lowIdx: h.price < l.price ? l.idx : h.idx,
   843→   843→            range: r,
   844→   844→          });
   845→   845→        }
   846→   846→      }
   847→   847→    }
   848→   848→    // Secondary swings' fib levels are merged via confluence (1% rule in mergeLevels)
   849→   849→    const confluenceTol = currentPrice * 0.005;
   850→   850→    for (const sec of secondarySwings.slice(0, 3)) {
   851→   851→      const sH = sec.high, sL = sec.low, sR = sH - sL;
   852→   852→      for (const ratio of FIB_RETRACEMENTS) {
   853→   853→        const price = roundToNice(sH - ratio * sR);
   854→   854→        if (price <= 0) continue;
   855→   855→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   856→   856→        levels.push({
   857→   857→          price, method: `Fibonacci_${ratio}`, type,
   858→   858→          fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   859→   859→        });
   860→   860→      }
   861→   861→      for (const extR of FIB_EXTENSIONS) {
   862→   862→        const price = roundToNice(sH + (extR - 1) * sR);
   863→   863→        if (price <= 0) continue;
   864→   864→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   865→   865→        levels.push({
   866→   866→          price, method: `Fibonacci_${extR}`, type,
   867→   867→          fibRatio: String(extR), fibLabel: FIB_LABELS[String(extR)] || '',
   868→   868→        });
   869→   869→      }
   870→   870→    }
   871→   871→  }
   872→   872→
   873→   873→  // ── Method 5: Volume Profile (VAP) ──────────────────────────────────
   874→   874→  const dataHasVolume = slice.some(d => d.volume > 0);
   875→   875→  if (dataHasVolume) {
   876→   876→    let minP = Infinity, maxP = -Infinity;
   877→   877→    for (const d of slice) {
   878→   878→      if (d.low < minP) minP = d.low;
   879→   879→      if (d.high > maxP) maxP = d.high;
   880→   880→    }
   881→   881→    const numBins = 40;
   882→   882→    const binSize = Math.max((maxP - minP) / numBins, currentPrice * 0.001);
   883→   883→    const vap = new Array(numBins).fill(0);
   884→   884→    const binCenters: number[] = [];
   885→   885→
   886→   886→    for (let b = 0; b < numBins; b++) {
   887→   887→      binCenters.push(minP + (b + 0.5) * binSize);
   888→   888→    }
   889→   889→
   890→   890→    for (const d of slice) {
   891→   891→      for (let b = 0; b < numBins; b++) {
   892→   892→        if (d.low <= binCenters[b] && binCenters[b] <= d.high) {
   893→   893→          vap[b] += d.volume;
   894→   894→        }
   895→   895→      }
   896→   896→    }
   897→   897→
   898→   898→    const maxVAP = Math.max(...vap);
   899→   899→    if (maxVAP > 0) {
   900→   900→      // POC (Point of Control)
   901→   901→      let pocIdx = 0;
   902→   902→      for (let b = 1; b < numBins; b++) {
   903→   903→        if (vap[b] > vap[pocIdx]) pocIdx = b;
   904→   904→      }
   905→   905→      const pocPrice = binCenters[pocIdx];
   906→   906→      const pocType: 'support' | 'resistance' = pocPrice > currentPrice ? 'resistance' : 'support';
   907→   907→      levels.push({ price: pocPrice, method: 'VAP_POC', type: pocType, ...noFib });
   908→   908→
   909→   909→      // HVN (High Volume Nodes)
   910→   910→      for (let b = 0; b < numBins; b++) {
   911→   911→        if (b === pocIdx) continue;
   912→   912→        if (vap[b] > 0.7 * maxVAP) {
   913→   913→          const hvnPrice = binCenters[b];
   914→   914→          const hvnType: 'support' | 'resistance' = hvnPrice > currentPrice ? 'resistance' : 'support';
   915→   915→          levels.push({ price: hvnPrice, method: 'VAP_HVN', type: hvnType, ...noFib });
   916→   916→        }
   917→   917→      }
   918→   918→    }
   919→   919→  }
   920→   920→
   921→   921→  // ── Method 6: Pivot Points (4 algorithms) ───────────────────────────
   922→   922→  if (endIdx >= 1) {
   923→   923→    const prev = slice[endIdx - 1]; // Previous completed bar
   924→   924→    const pH = prev.high, pL = prev.low, pC = prev.close, pO = prev.open;
   925→   925→
   926→   926→    // Standard Pivot
   927→   927→    {
   928→   928→      const PP = (pH + pL + pC) / 3;
   929→   929→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   930→   930→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   931→   931→      const R3 = pH + 2 * (PP - pL), S3 = pL - 2 * (pH - PP);
   932→   932→ if (R1 > 0) levels.push({ price: R1, method: 'Pivot_R1', type: 'resistance', ...noFib });
   933→   933→      if (R2 > 0) levels.push({ price: R2, method: 'Pivot_R2', type: 'resistance', ...noFib });
   934→   934→      if (R3 > 0) levels.push({ price: R3, method: 'Pivot_R3', type: 'resistance', ...noFib });
   935→   935→      if (S1 > 0) levels.push({ price: S1, method: 'Pivot_S1', type: 'support', ...noFib });
   936→   936→      if (S2 > 0) levels.push({ price: S2, method: 'Pivot_S2', type: 'support', ...noFib });
   937→   937→      if (S3 > 0) levels.push({ price: S3, method: 'Pivot_S3', type: 'support', ...noFib });
   938→   938→    }
   939→   939→
   940→   940→    // Fibonacci Pivot
   941→   941→    {
   942→   942→      const PP = (pH + pL + pC) / 3;
   943→   943→      const range = pH - pL;
   944→   944→      for (const [mult, suffix] of [[0.382, '1'], [0.618, '2'], [1.0, '3']] as [number, string][]) {
   945→   945→        const R = PP + mult * range;
   946→   946→        const S = PP - mult * range;
   947→   947→        if (R > 0) levels.push({ price: R, method: `Fib_Pivot_R${suffix}`, type: 'resistance', ...noFib });
   948→   948→        if (S > 0) levels.push({ price: S, method: `Fib_Pivot_S${suffix}`, type: 'support', ...noFib });
   949→   949→      }
   950→   950→    }
   951→   951→
   952→   952→    // Woodie's Pivot
   953→   953→    {
   954→   954→      const PP = (pH + pL + 2 * pC) / 4;
   955→   955→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   956→   956→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   957→   957→      if (R1 > 0) levels.push({ price: R1, method: 'Woodie_R1', type: 'resistance', ...noFib });
   958→   958→      if (R2 > 0) levels.push({ price: R2, method: 'Woodie_R2', type: 'resistance', ...noFib });
   959→   959→      if (S1 > 0) levels.push({ price: S1, method: 'Woodie_S1', type: 'support', ...noFib });
   960→   960→      if (S2 > 0) levels.push({ price: S2, method: 'Woodie_S2', type: 'support', ...noFib });
   961→   961→    }
   962→   962→
   963→   963→    // DeMark Pivot
   964→   964→    {
   965→   965→      const X = pC < pO ? pH + 2 * pL + pC : pC > pO ? 2 * pH + pL + pC : pH + pL + 2 * pC;
   966→   966→      const PP = X / 4;
   967→   967→      const R1 = X / 2 - pL;
   968→   968→      const S1 = X / 2 - pH;
   969→   969→      if (R1 > 0) levels.push({ price: R1, method: 'DeMark_R1', type: 'resistance', ...noFib });
   970→   970→      if (S1 > 0) levels.push({ price: S1, method: 'DeMark_S1', type: 'support', ...noFib });
   971→   971→    }
   972→   972→  }
   973→   973→
   974→   974→  // ── Method 7: Psychological Levels ──────────────────────────────────
   975→   975→  {
   976→   976→    const step = psychStep(currentPrice);
   977→   977→    const base = psychRound(currentPrice);
   978→   978→    for (const mult of [-2, -1, 1, 2]) {
   979→   979→      const price = base + mult * step;
   980→   980→      if (price <= 0) continue;
   981→   981→      const type: 'support' | 'resistance' = price > currentPrice ? 'resistance' : 'support';
   982→   982→      levels.push({ price, method: 'Psychological', type, ...noFib });
   983→   983→    }
   984→   984→  }
   985→   985→
   986→   986→  return levels;
   987→   987→}

   989→   989→// ─── Helper: Merge Nearby Levels (1% Confluence) ──────────────────────────
   990→   990→function mergeLevels(rawLevels: RawSRLevel[], currentPrice: number): MergedSRLevel[] {
   991→   991→  if (rawLevels.length === 0) return [];
   992→   992→
   993→   993→  const tol = currentPrice * 0.01; // 1% tolerance
   994→   994→  const merged: MergedSRLevel[] = [];
   995→   995→  const used = new Set<number>();
   996→   996→
   997→   997→  for (let i = 0; i < rawLevels.length; i++) {
   998→   998→    if (used.has(i)) continue;
   999→   999→
  1000→  1000→    let sumPrice = rawLevels[i].price;
  1001→  1001→    let count = 1;
  1002→  1002→    const methods = new Set<string>();
  1003→  1003→    methods.add(rawLevels[i].method);
  1004→  1004→    let bestFibRatio = rawLevels[i].fibRatio;
  1005→  1005→    let bestFibLabel = rawLevels[i].fibLabel;
  1006→  1006→
  1007→  1007→    for (let j = i + 1; j < rawLevels.length; j++) {
  1008→  1008→      if (used.has(j)) continue;
  1009→  1009→      if (Math.abs(rawLevels[j].price - rawLevels[i].price) < tol) {
  1010→  1010→        sumPrice += rawLevels[j].price;
  1011→  1011→        count++;
  1012→  1012→        methods.add(rawLevels[j].method);
  1013→  1013→        used.add(j);
  1014→  1014→        // Prefer Fibonacci label if available
  1015→  1015→        if (rawLevels[j].fibRatio !== '—' && bestFibRatio === '—') {
  1016→  1016→          bestFibRatio = rawLevels[j].fibRatio;
  1017→  1017→          bestFibLabel = rawLevels[j].fibLabel;
  1018→  1018→        }
  1019→  1019→      }
  1020→  1020→    }
  1021→  1021→    used.add(i);
  1022→  1022→
  1023→  1023→    const avgPrice = sumPrice / count;
  1024→  1024→    const type: 'support' | 'resistance' = avgPrice >= currentPrice ? 'resistance' : 'support';
  1025→  1025→
  1026→  1026→    merged.push({
  1027→  1027→      price: avgPrice,
  1028→  1028→      type,
  1029→  1029→      methods: [...methods],
  1030→  1030→      overlapCount: methods.size,
  1031→  1031→      fibRatio: bestFibRatio,
  1032→  1032→      fibLabel: bestFibLabel,
  1033→  1033→      score: 0,
  1034→  1034→      strength: 1,
  1035→  1035→      touchCount: 0,
  1036→  1036→      volumeRatio: 0,
  1037→  1037→      freshness: 0,
  1038→  1038→      distancePercent: 0,
  1039→  1039→    });
  1040→  1040→  }
  1041→  1041→
  1042→  1042→  return merged;
  1043→  1043→}

  1045→  1045→// ─── Helper: Compute Level Features (5 ML features) ──────────────────────
  1046→  1046→function computeLevelFeatures(
  1047→  1047→  levelPrice: number,
  1048→  1048→  data: OHLCV[],
  1049→  1049→  endIdx: number,
  1050→  1050→  currentPrice: number,
  1051→  1051→  overlapCount: number,
  1052→  1052→): number[] {
  1053→  1053→  const lookback = Math.min(30, endIdx);
  1054→  1054→  const startIdx = endIdx - lookback;
  1055→  1055→  const touchTol = currentPrice * 0.002; // 0.2%
  1056→  1056→
  1057→  1057→  let touchCount = 0;
  1058→  1058→  let totalVolAtTouches = 0;
  1059→  1059→  let totalVol = 0;
  1060→  1060→  let barsSinceLastTouch = lookback;
  1061→  1061→
  1062→  1062→  for (let i = startIdx; i <= endIdx; i++) {
  1063→  1063→    const bar = data[i];
  1064→  1064→    totalVol += bar.volume;
  1065→  1065→
  1066→  1066→    if (bar.low <= levelPrice + touchTol && bar.high >= levelPrice - touchTol) {
  1067→  1067→      touchCount++;
  1068→  1068→      totalVolAtTouches += bar.volume;
  1069→  1069→      barsSinceLastTouch = endIdx - i;
  1070→  1070→    }
  1071→  1071→  }
  1072→  1072→
  1073→  1073→  const avgVol = lookback > 0 ? totalVol / lookback : 0;
  1074→  1074→  const volumeRatio = (touchCount > 0 && avgVol > 0) ? (totalVolAtTouches / touchCount) / avgVol : 1;
  1075→  1075→  const freshness = Math.max(0, Math.min(1, 1 - barsSinceLastTouch / 30));
  1076→  1076→  const distancePercent = Math.min(1, Math.abs(levelPrice - currentPrice) / currentPrice);
  1077→  1077→
  1078→  1078→  // Features: [touch_count, volume_ratio, overlap_count, freshness, distance_percent]
  1079→  1079→  return [touchCount, volumeRatio, overlapCount - 1, freshness, distancePercent];
  1080→  1080→}

  1082→  1082→// ─── Helper: Train S/R Scoring Weights (Ridge OLS) ───────────────────────
  1083→  1083→function trainSRWeights(data: OHLCV[]): number[] {
  1084→  1084→  const defaultWeights = [0.25, 0.20, 0.25, 0.15, 0.15];
  1085→  1085→
  1086→  1086→  if (data.length < 50) return defaultWeights;
  1087→  1087→
  1088→  1088→  const X: number[][] = [];
  1089→  1089→  const y: number[] = [];
  1090→  1090→
  1091→  1091→  for (let i = 40; i <= data.length - 11; i += 5) {
  1092→  1092→    const barPrice = data[i].close;
  1093→  1093→    const rawLevels = generateAllLevels(data, i);
  1094→  1094→    if (rawLevels.length === 0) continue;
  1095→  1095→    const merged = mergeLevels(rawLevels, barPrice);
  1096→  1096→
  1097→  1097→    for (const level of merged) {
  1098→  1098→      const features = computeLevelFeatures(level.price, data, i, barPrice, level.overlapCount);
  1099→  1099→      X.push(features);
  1100→  1100→
  1101→  1101→      // Label: did price return within 10 bars?
  1102→  1102→      let returned = false;
  1103→  1103→      const retTol = barPrice * 0.005;
  1104→  1104→      for (let j = i + 1; j <= Math.min(i + 10, data.length - 1); j++) {
  1105→  1105→        if (data[j].low <= level.price + retTol && data[j].high >= level.price - retTol) {
  1106→  1106→          returned = true;
  1107→  1107→          break;
  1108→  1108→        }
  1109→  1109→      }
  1110→  1110→      y.push(returned ? 1 : 0);
  1111→  1111→    }
  1112→  1112→  }
  1113→  1113→
  1114→  1114→  if (X.length < 20) return defaultWeights;
  1115→  1115→
  1116→  1116→  // Ridge OLS: w = (X^T X + λI)^(-1) X^T y
  1117→  1117→  const n = X.length;
  1118→  1118→  const p = 5;
  1119→  1119→  const lambda = 0.01;
  1120→  1120→
  1121→  1121→  const XtX: number[][] = Array.from({ length: p }, () => new Array(p).fill(0));
  1122→  1122→  for (let i = 0; i < n; i++) {
  1123→  1123→    for (let a = 0; a < p; a++) {
  1124→  1124→      for (let b = 0; b < p; b++) {
  1125→  1125→        XtX[a][b] += X[i][a] * X[i][b];
  1126→  1126→      }
  1127→  1127→    }
  1128→  1128→  }
  1129→  1129→  for (let a = 0; a < p; a++) XtX[a][a] += lambda;
  1130→  1130→
  1131→  1131→  const Xty: number[] = new Array(p).fill(0);
  1132→  1132→  for (let i = 0; i < n; i++) {
  1133→  1133→    for (let a = 0; a < p; a++) {
  1134→  1134→      Xty[a] += X[i][a] * y[i];
  1135→  1135→    }
  1136→  1136→  }
  1137→  1137→
  1138→  1138→  const weights = solveLinearSystem(XtX, Xty);
  1139→  1139→  if (!weights) return defaultWeights;
  1140→  1140→  if (weights.some(w => isNaN(w) || !isFinite(w))) return defaultWeights;
  1141→  1141→
  1142→  1142→  for (let i = 0; i < p; i++) {
  1143→  1143→    if (weights[i] < 0) weights[i] = 0;
  1144→  1144→  }
  1145→  1145→
  1146→  1146→  const sum = weights.reduce((a, b) => a + b, 0);
  1147→  1147→  if (sum <= 0) return defaultWeights;
  1148→  1148→
  1149→  1149→  return weights.map(w => w / sum);
  1150→  1150→}

  1152→  1152→// ─── Support / Resistance (7-Method ML-Based System) ─────────────────────
  1153→  1153→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
  1154→  1154→  resistances: number[];
  1155→  1155→  supports: number[];
  1156→  1156→  supportStrengths: LevelStrength[];
  1157→  1157→  resistanceStrengths: LevelStrength[];
  1158→  1158→  priceTargets: LevelStrength[];
  1159→  1159→} {
  1160→  1160→  if (data.length < 10) {
  1161→  1161→    return {
  1162→  1162→      resistances: [], supports: [],
  1163→  1163→      supportStrengths: [], resistanceStrengths: [],
  1164→  1164→      priceTargets: [],
  1165→  1165→    };
  1166→  1166→  }
  1167→  1167→
  1168→  1168→  // ── Step 1: Generate all levels from 7 methods ────────────────────────
  1169→  1169→  const rawLevels = generateAllLevels(data, data.length - 1);
  1170→  1170→
  1171→  1171→  // ── Step 2: Merge nearby levels (confluence within 1%) ────────────────
  1172→  1172→  let merged = mergeLevels(rawLevels, currentPrice);
  1173→  1173→
  1174→  1174→  // ── Step 3: Train ML weights on historical data ───────────────────────
  1175→  1175→  const weights = trainSRWeights(data);
  1176→  1176→
  1177→  1177→  // ── Step 4: Compute features & score each merged level ────────────────
  1178→  1178→  for (const level of merged) {
  1179→  1179→    const features = computeLevelFeatures(level.price, data, data.length - 1, currentPrice, level.overlapCount);
  1180→  1180→
  1181→  1181→    const touchScore = Math.min(features[0] / 5, 1);
  1182→  1182→    const volumeScore = Math.min(features[1] / 2, 1);
  1183→  1183→    const overlapScore = Math.min(features[2] * 2, 10) / 10;
  1184→  1184→    const freshnessScore = features[3];
  1185→  1185→    const distanceScore = 1 - features[4];
  1186→  1186→
  1187→  1187→    const powerScore = (
  1188→  1188→      weights[0] * touchScore +
  1189→  1189→      weights[1] * volumeScore +
  1190→  1190→      weights[2] * overlapScore +
  1191→  1191→      weights[3] * freshnessScore +
  1192→  1192→      weights[4] * distanceScore
  1193→  1193→    ) * 10;
  1194→  1194→
  1195→  1195→    level.score = Math.max(0, Math.min(10, powerScore));
  1196→  1196→    level.strength = Math.max(1, Math.min(10, Math.round(level.score)));
  1197→  1197→    level.touchCount = features[0];
  1198→  1198→    level.volumeRatio = features[1];
  1199→  1199→    level.freshness = features[3];
  1200→  1200→    level.distancePercent = features[4];
  1201→  1201→  }
  1202→  1202→
  1203→  1203→  // ── Step 5: Psychological rounding of all level prices ─────────────────
  1204→  1204→  for (const level of merged) {
  1205→  1205→    level.price = psychRound(level.price);
  1206→  1206→  }
  1207→  1207→
  1208→  1208→  // Re-classify types after rounding
  1209→  1209→  for (const level of merged) {
  1210→  1210→    level.type = level.price >= currentPrice ? 'resistance' : 'support';
  1211→  1211→  }
  1212→  1212→
  1213→  1213→  // Deduplicate after rounding (keep strongest at same price)
  1214→  1214→  {
  1215→  1215→    const deduped: MergedSRLevel[] = [];
  1216→  1216→    for (const level of merged) {
  1217→  1217→      const existing = deduped.find(d => Math.abs(d.price - level.price) < currentPrice * 0.001);
  1218→  1218→      if (existing) {
  1219→  1219→        if (level.score > existing.score) {
  1220→  1220→          Object.assign(existing, level);
  1221→  1221→        }
  1222→  1222→      } else {
  1223→  1223→        deduped.push(level);
  1224→  1224→      }
  1225→  1225→    }
  1226→  1226→    merged = deduped;
  1227→  1227→  }
  1228→  1228→
  1229→  1229→  // ── Step 6: Separate into supports and resistances ─────────────────────
  1230→  1230→  let supports = merged.filter(l => l.type === 'support');
  1231→  1231→  let resistances = merged.filter(l => l.type === 'resistance');
  1232→  1232→
  1233→  1233→  // ── Step 7: Distance filtering (strict 5-10% gap rules) ─────────────
  1234→  1234→  // Rules:
  1235→  1235→  //   1. Consecutive lines of same type: 5% ≤ gap ≤ 10%
  1236→  1236→  //   2. Nearest support ≤ 10% below price (can be < 5%)
  1237→  1237→  //   3. Nearest resistance ≤ 10% above price (can be < 5%)
  1238→  1238→
  1239→  1239→  const filterAndSelectLevels = (
  1240→  1240→    levels: MergedSRLevel[],
  1241→  1241→    type: 'support' | 'resistance'
  1242→  1242→  ): MergedSRLevel[] => {
  1243→  1243→    if (levels.length === 0) return [];
  1244→  1244→
  1245→  1245→    // Sort: supports descending (nearest first), resistances ascending (nearest first)
  1246→  1246→    const sorted = [...levels].sort((a, b) =>
  1247→  1247→      type === 'support' ? b.price - a.price : a.price - b.price
  1248→  1248→    );
  1249→  1249→
  1250→  1250→    // ── Rule 2/3: Nearest level must be within 10% of price ──
  1251→  1251→    const nearest = sorted[0];
  1252→  1252→    const distFromPrice = Math.abs(nearest.price - currentPrice) / currentPrice;
  1253→  1253→    if (distFromPrice > 0.10) {
  1254→  1254→      // Nearest level is too far — generate a synthetic level at ~5% from price
  1255→  1255→      const syntheticPrice = type === 'support'
  1256→  1256→        ? Math.round(currentPrice * 0.95 / psychStep(currentPrice)) * psychStep(currentPrice)
  1257→  1257→        : Math.round(currentPrice * 1.05 / psychStep(currentPrice)) * psychStep(currentPrice);
  1258→  1258→      const synthetic: MergedSRLevel = {
  1259→  1259→        price: syntheticPrice,
  1260→  1260→        type,
  1261→  1261→        score: 3,
  1262→  1262→        strength: 3,
  1263→  1263→        overlapCount: 1,
  1264→  1264→        methods: ['Psychological'],
  1265→  1265→        fibRatio: '—',
  1266→  1266→        fibLabel: '',
  1267→  1267→      };
  1268→  1268→      sorted.unshift(synthetic);
  1269→  1269→    }
  1270→  1270→
  1271→  1271→    // ── Rule 1: Select levels with 5-10% gaps between consecutive ──
  1272→  1272→    const result: MergedSRLevel[] = [sorted[0]];
  1273→  1273→
  1274→  1274→    for (let i = 1; i < sorted.length && result.length < 6; i++) {
  1275→  1275→      const prevPrice = result[result.length - 1].price;
  1276→  1276→      const currPrice = sorted[i].price;
  1277→  1277→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
  1278→  1278→
  1279→  1279→      if (gap < 0.05) {
  1280→  1280→        // Too close — keep the stronger one
  1281→  1281→        if (sorted[i].score > result[result.length - 1].score) {
  1282→  1282→          result[result.length - 1] = sorted[i];
  1283→  1283→        }
  1284→  1284→      } else if (gap <= 0.10) {
  1285→  1285→        // Perfect gap (5-10%) — accept
  1286→  1286→        result.push(sorted[i]);
  1287→  1287→      } else {
  1288→  1288→        // Gap > 10% — try to find an intermediate level
  1289→  1289→        let found = false;
  1290→  1290→        for (let j = i + 1; j < sorted.length; j++) {
  1291→  1291→          const midPrice = sorted[j].price;
  1292→  1292→          const gapFromPrev = Math.abs(midPrice - prevPrice) / prevPrice;
  1293→  1293→          const gapToNext = Math.abs(currPrice - midPrice) / midPrice;
  1294→  1294→          if (gapFromPrev >= 0.05 && gapFromPrev <= 0.10) {
  1295→  1295→            result.push(sorted[j]);
  1296→  1296→            found = true;
  1297→  1297→            // Skip levels that are too close to the inserted one
  1298→  1298→            i = j;
  1299→  1299→            break;
  1300→  1300→          }
  1301→  1301→        }
  1302→  1302→        if (!found) {
  1303→  1303→          // No intermediate found — generate a synthetic level at ~7.5% from prev
  1304→  1304→          const synthGap = 0.075;
  1305→  1305→          const syntheticPrice = type === 'support'
  1306→  1306→            ? Math.round((prevPrice * (1 - synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice)
  1307→  1307→            : Math.round((prevPrice * (1 + synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice);
  1308→  1308→          // Make sure it's not too close to next level
  1309→  1309→          const gapToNext = Math.abs(currPrice - syntheticPrice) / syntheticPrice;
  1310→  1310→          if (gapToNext >= 0.05) {
  1311→  1311→            const synthetic: MergedSRLevel = {
  1312→  1312→              price: syntheticPrice,
  1313→  1313→              type,
  1314→  1314→              score: 2,
  1315→  1315→              strength: 2,
  1316→  1316→              overlapCount: 1,
  1317→  1317→              methods: ['Psychological'],
  1318→  1318→              fibRatio: '—',
  1319→  1319→              fibLabel: '',
  1320→  1320→            };
  1321→  1321→            result.push(synthetic);
  1322→  1322→          }
  1323→  1323→        }
  1324→  1324→      }
  1325→  1325→    }
  1326→  1326→
  1327→  1327→    return result;
  1328→  1328→  };
  1329→  1329→
  1330→  1330→  const finalSupports = filterAndSelectLevels(supports, 'support');
  1331→  1331→  const finalResistances = filterAndSelectLevels(resistances, 'resistance');
  1332→  1332→
  1333→  1333→  // ── Step 9: Build LevelStrength objects ────────────────────────────────
  1334→  1334→  const toLevelStrength = (m: MergedSRLevel): LevelStrength => {
  1335→  1335→    let grade: string;
  1336→  1336→    if (m.score >= 8.5) grade = 'Very Strong';
  1337→  1337→    else if (m.score >= 7) grade = 'Strong';
  1338→  1338→    else if (m.score >= 5) grade = 'Moderate';
  1339→  1339→    else grade = 'Weak';
  1340→  1340→
  1341→  1341→    return {
  1342→  1342→      price: m.price,
  1343→  1343→      strength: m.strength,
  1344→  1344→      score: m.score,
  1345→  1345→      grade,
  1346→  1346→      isTarget: m.score >= 7,
  1347→  1347→      overlapCount: m.overlapCount,
  1348→  1348→      methods: m.methods,
  1349→  1349→      fibRatio: m.fibRatio,
  1350→  1350→      fibLabel: m.fibLabel,
  1351→  1351→    };
  1352→  1352→  };
  1353→  1353→
  1354→  1354→  const supportStrengths = finalSupports.map(toLevelStrength);
  1355→  1355→  const resistanceStrengths = finalResistances.map(toLevelStrength);
  1356→  1356→
  1357→  1357→  // Price targets: all levels with isTarget=true
  1358→  1358→  const priceTargets = [...supportStrengths, ...resistanceStrengths].filter(l => l.isTarget);
  1359→  1359→
  1360→  1360→  return {
  1361→  1361→    resistances: resistanceStrengths.map(l => l.price),
  1362→  1362→    supports: supportStrengths.map(l => l.price),
  1363→  1363→    supportStrengths,
  1364→  1364→    resistanceStrengths,
  1365→  1365→    priceTargets,
  1366→  1366→  };
  1367→  1367→}

  1371→  1371→function calcTrend(closes: number[], period: number): TrendResult {
  1372→  1372→  const slice = closes.slice(-Math.min(period, closes.length));
  1373→  1373→  const { slope, r2 } = linearRegression(slice);
  1374→  1374→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
  1375→  1375→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
  1376→  1376→  const priceThreshold = avgPrice * 0.001; // 0.1%
  1377→  1377→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
  1378→  1378→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
  1379→  1379→}

  1386→  1386→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
  1387→  1387→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
  1388→  1388→  const slice = data.slice(0, barIdx + 1);
  1389→  1389→  const price = slice[slice.length - 1].close;
  1390→  1390→  let nearestR = price * 1.05;
  1391→  1391→  let nearestS = price * 0.95;
  1392→  1392→
  1393→  1393→  // Collect all swing highs (resistance) and swing lows (support)
  1394→  1394→  const resistances: number[] = [];
  1395→  1395→  const supports: number[] = [];
  1396→  1396→  for (const lb of [3, 5, 7]) {
  1397→  1397→    const swings = findSwingLevels(slice, lb);
  1398→  1398→    resistances.push(...swings.highs);
  1399→  1399→    supports.push(...swings.lows);
  1400→  1400→  }
  1401→  1401→
  1402→  1402→  // Find nearest resistance above price
  1403→  1403→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
  1404→  1404→  if (abovePrice.length > 0) nearestR = abovePrice[0];
  1405→  1405→
  1406→  1406→  // Find nearest support below price
  1407→  1407→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
  1408→  1408→  if (belowPrice.length > 0) nearestS = belowPrice[0];
  1409→  1409→
  1410→  1410→  return { R1: nearestR, S1: nearestS };
  1411→  1411→}

  1413→  1413→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
  1414→  1414→// Returns true if bullish divergence detected, false if bearish, null if none
  1415→  1415→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
  1416→  1416→  const lookback = 20;
  1417→  1417→  const start = Math.max(0, barIdx - lookback);
  1418→  1418→  const slice = data.slice(start, barIdx + 1);
  1419→  1419→  if (slice.length < 10) return null;
  1420→  1420→
  1421→  1421→  const mid = Math.floor(slice.length / 2);
  1422→  1422→  const firstHalf = slice.slice(0, mid);
  1423→  1423→  const secondHalf = slice.slice(mid);
  1424→  1424→
  1425→  1425→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
  1426→  1426→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
  1427→  1427→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
  1428→  1428→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
  1429→  1429→
  1430→  1430→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
  1431→  1431→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
  1432→  1432→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
  1433→  1433→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
  1434→  1434→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
  1435→  1435→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
  1436→  1436→
  1437→  1437→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
  1438→  1438→    return false; // bearish divergence
  1439→  1439→  }
  1440→  1440→
  1441→  1441→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
  1442→  1442→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
  1443→  1443→
  1444→  1444→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
  1445→  1445→    return true; // bullish divergence
  1446→  1446→  }
  1447→  1447→
  1448→  1448→  return null;
  1449→  1449→}

  1586→  1586→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
  1587→  1587→    : cciCurrent;
  1588→  1588→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
  1589→  1589→
  1590→  1590→  // f_macd — Same as s_macd (no additional correction)
  1591→  1591→  const f_macd = s_macd;
  1592→  1592→
  1593→  1593→  // f_stoch — Same as s_stoch
  1594→  1594→  const f_stoch = s_stoch;
  1595→  1595→
  1596→  1596→  // f_stochCross — Stochastic K/D crossover detection
  1597→  1597→  let f_stochCross = 0.1;
  1598→  1598→  if (barIdx >= 1) {
  1599→  1599→    const prevSlice = data.slice(0, barIdx);
  1600→  1600→    const prevStoch = calcStochastic(prevSlice);
  1601→  1601→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
  1602→  1602→      f_stochCross = 0.9; // K crossed above D (bullish)
  1603→  1603→    }
  1604→  1604→  }
  1605→  1605→
  1606→  1606→  // f_macdCross — MACD/Signal crossover detection
  1607→  1607→  let f_macdCross = 0.1;
  1608→  1608→  if (barIdx >= 1) {
  1609→  1609→    const prevCloses = closes.slice(0, -1);
  1610→  1610→    if (prevCloses.length >= 35) {
  1611→  1611→      const prevMacd = calcMACD(prevCloses);
  1612→  1612→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
  1613→  1613→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
  1614→  1614→      }
  1615→  1615→    }
  1616→  1616→  }
  1617→  1617→
  1618→  1618→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
  1619→  1619→  const divResult = detectDivergenceSimple(data, barIdx);
  1620→  1620→  const f_div = divResult === true ? 0.85 : 0.15;
  1621→  1621→
  1622→  1622→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
  1623→  1623→  return {
  1624→  1624→    rsi: f_rsi,
  1625→  1625→    mfi: f_mfi,
  1626→  1626→    cci: f_cci,
  1627→  1627→    adx: s_adx,
  1628→  1628→    macd: f_macd,
  1629→  1629→    stoch: f_stoch,
  1630→  1630→    bb: s_bb,
  1631→  1631→    ma21: s_ma21,
  1632→  1632→    ma100: s_ma100,
  1633→  1633→    ema: s_ema,
  1634→  1634→    atr: s_atr,
  1635→  1635→    trend: s_trend,
  1636→  1636→    sr: s_sr,
  1637→  1637→    stochCross: f_stochCross,
  1638→  1638→    macdCross: f_macdCross,
  1639→  1639→    div: f_div,
  1640→  1640→  };
  1641→  1641→}
  1642→  1642→
  1643→  1643→// ═══════════════════════════════════════════════════════════════════════════════
  1644→  1644→// TRAINING DATA CREATION (Layer 0 — Data Preparation)
  1645→  1645→// ═══════════════════════════════════════════════════════════════════════════════
  1646→  1646→
  1647→  1647→/**
  1648→  1648→ * Creates supervised training data from historical OHLCV data.
  1649→  1649→ * For each bar, computes features and labels based on future price movement.
  1650→  1650→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
  1651→  1651→ * Samples every `sampleStep` bars to reduce computation.
  1652→  1652→ */
  1653→  1653→function createTrainingData(
  1654→  1654→  data: OHLCV[],
  1655→  1655→  horizon: number = 5,
  1656→  1656→  sampleStep: number = 5
  1657→  1657→): { X: number[][]; y: number[] } {
  1658→  1658→  const X: number[][] = [];
  1659→  1659→  const y: number[] = [];
  1660→  1660→
  1661→  1661→  // Start from horizon+30 to ensure enough data for indicators
  1662→  1662→  // End at data.length - horizon - 1 to have future bars for labeling
  1663→  1663→  // Sample every sampleStep bars for performance
  1664→  1664→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
  1665→  1665→    const features = computeFeaturesAtBar(data, i);
  1666→  1666→
  1667→  1667→    // Convert to array following FEATURES_ORDER
  1668→  1668→    const featureVec: number[] = [];
  1669→  1669→    let valid = true;
  1670→  1670→    for (const key of FEATURES_ORDER) {
  1671→  1671→      const val = features[key];
  1672→  1672→      if (val === undefined || val === null || Number.isNaN(val)) {
  1673→  1673→        valid = false;
  1674→  1674→        break;
  1675→  1675→      }
  1676→  1676→      featureVec.push(val);
  1677→  1677→    }
  1678→  1678→    if (!valid) continue;
  1679→  1679→
  1680→  1680→    // Label: 1 if future close > 1% above current close
  1681→  1681→    const currentClose = data[i].close;
  1682→  1682→    const futureClose = data[i + horizon].close;
  1683→  1683→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
  1684→  1684→
  1685→  1685→    X.push(featureVec);
  1686→  1686→    y.push(label);
  1687→  1687→  }
  1688→  1688→
  1689→  1689→  return { X, y };
  1690→  1690→}
  1691→  1691→
  1692→  1692→// ═══════════════════════════════════════════════════════════════════════════════
  1693→  1693→// VDss GRAPH STRUCTURE (Layer 5 — State Machine)
  1694→  1694→// ═══════════════════════════════════════════════════════════════════════════════
  1695→  1695→
  1696→  1696→/**
  1697→  1697→ * VDss Graph: Each node represents a market state.
  1698→  1698→ * Edges represent transitions with types: 'up', 'down', 'pullback', 'risk', 'terminal'.
  1699→  1699→ * Terminal nodes map to scenario results (R1=strong bullish, R5=deep correction).
  1700→  1700→ */
  1701→  1701→const VDSS_GRAPH = {
  1702→  1702→  // outgoing edges from each node: [target, type]
  1703→  1703→  outgoing: {
  1704→  1704→    A: [['B', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1705→  1705→    B: [['C', 'up'], ['G', 'pullback'], ['A', 'risk']] as [string, string][],
  1706→  1706→    C: [['D', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1707→  1707→    D: [['E', 'up'], ['B', 'pullback'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1708→  1708→    E: [['F', 'up'], ['D', 'pullback'], ['G', 'pullback'], ['H', 'down'], ['R1', 'terminal']] as [string, string][],
  1709→  1709→    F: [['R1', 'terminal'], ['E', 'pullback'], ['D', 'down'], ['G', 'risk']] as [string, string][],
  1710→  1710→    G: [['B', 'up'], ['A', 'pullback'], ['H', 'down'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1711→  1711→    H: [['G', 'pullback'], ['I', 'down'], ['B', 'up'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1712→  1712→    I: [['H', 'pullback'], ['J', 'down'], ['G', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1713→  1713→    J: [['I', 'pullback'], ['K', 'down'], ['H', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1714→  1714→    K: [['J', 'pullback'], ['I', 'up'], ['L', 'risk'], ['R4', 'terminal'], ['R5', 'terminal']] as [string, string][],
  1715→  1715→    L: [['C', 'up'], ['H', 'pullback'], ['J', 'down'], ['K', 'risk'], ['R5', 'terminal']] as [string, string][],
  1716→  1716→  },
  1717→  1717→  // which result scenario each terminal node maps to
  1718→  1718→  terminalMap: { R1: 'R1', R2: 'R2', R3: 'R3', R4: 'R4', R5: 'R5' } as Record<string, string>,
  1719→  1719→  startNode: 'A' as string,
  1720→  1720→};
  1721→  1721→
  1722→  1722→// ═══════════════════════════════════════════════════════════════════════════════
  1723→  1723→// LAYER 5 — Edge Weight Calculation
  1724→  1724→// ═══════════════════════════════════════════════════════════════════════════════
  1725→  1725→
  1726→  1726→/**
  1727→  1727→ * Calculates transition edge weights for the VDss graph.
  1728→  1728→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
  1729→  1729→ * to determine the relative probability of up/down/pullback/risk transitions.
  1730→  1730→ */
  1731→  1731→function calculateEdgeWeights(
  1732→  1732→  bullConsensus: number,
  1733→  1733→  adx: number,
  1734→  1734→  mlModel: AdaptiveWeightModel
  1735→  1735→): { up: number; down: number; pullback: number; risk: number } {
  1736→  1736→  // Extract ML coefficient insights if model is trained
  1737→  1737→  let trend_coef = 1 / 3;
  1738→  1738→  let momentum_coef = 1 / 3;
  1739→  1739→  let volatility_coef = 1 / 3;
  1740→  1740→
  1741→  1741→  if (mlModel.isTrained && mlModel.weights) {
  1742→  1742→    const coefs = mlModel.getCoefficients();
  1743→  1743→    // Map feature indices to coefficient categories
  1744→  1744→    // FEATURES_ORDER: ['rsi','mfi','cci','adx','macd','stoch','bb','ma21','ma100','ema','atr','trend','sr','stochCross','macdCross','div']
  1745→  1745→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER
  1746→  1746→    const rsiIdx = 0;     // 'rsi' in FEATURES_ORDER
  1747→  1747→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
  1748→  1748→
  1749→  1749→    const absTrend = Math.abs(coefs[trendIdx] ?? 0);
  1750→  1750→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
  1751→  1751→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
  1752→  1752→
  1753→  1753→    const coefSum = absTrend + absRsi + absAtr;
  1754→  1754→    if (coefSum > 1e-10) {
  1755→  1755→      trend_coef = absTrend / coefSum;
  1756→  1756→      momentum_coef = absRsi / coefSum;
  1757→  1757→      volatility_coef = absAtr / coefSum;
  1758→  1758→    }
  1759→  1759→  }
  1760→  1760→
  1761→  1761→  // Base weights driven by bull consensus
  1762→  1762→  const up_base = bullConsensus * 0.7 + 0.15;
  1763→  1763→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
  1764→  1764→  const pullback_base = 0.25;
  1765→  1765→  const risk_base = 0.12 * (1.2 - adx / 100);
  1766→  1766→
  1767→  1767→  // Apply ML coefficient adjustments
  1768→  1768→  const up = clamp(
  1769→  1769→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
  1770→  1770→    0.05, 0.95
  1771→  1771→  );
  1772→  1772→  const down = clamp(
  1773→  1773→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
  1774→  1774→    0.05, 0.95
  1775→  1775→  );
  1776→  1776→  const pullback = clamp(
  1777→  1777→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
  1778→  1778→    0.05, 0.50
  1779→  1779→  );
  1780→  1780→  const risk = clamp(
  1781→  1781→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
  1782→  1782→    0.02, 0.30
  1783→  1783→  );
  1784→  1784→
  1785→  1785→  return { up, down, pullback, risk };
  1786→  1786→}
  1787→  1787→
  1788→  1788→// ═══════════════════════════════════════════════════════════════════════════════
  1789→  1789→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)
  1790→  1790→// ═══════════════════════════════════════════════════════════════════════════════
  1791→  1791→
  1792→  1792→/**
  1793→  1793→ * Performs iterative DFS from the start node through the VDss graph,
  1794→  1794→ * computing raw path probabilities, then calibrating them against
  1795→  1795→ * Layer 4 scenario probabilities.
  1796→  1796→ *
  1797→  1797→ * Returns:
  1798→  1798→ * - calibratedPaths: Map<pathString, probability>
  1799→  1799→ * - calibrationFactors: per-scenario multiplier used
  1800→  1800→ * - scenarioSums: raw probability sums per scenario before calibration
  1801→  1801→ */
  1802→  1802→function calculatePathProbabilities(
  1803→  1803→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
  1804→  1804→  scenarioProbs: Record<string, number>
  1805→  1805→): {
  1806→  1806→  calibratedPaths: Map<string, number>;
  1807→  1807→  calibrationFactors: Record<string, number>;
  1808→  1808→  scenarioSums: Record<string, number>;
  1809→  1809→} {
  1810→  1810→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
  1811→  1811→  const MAX_PATHS = 200;
  1812→  1812→
  1813→  1813→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
  1814→  1814→  type StackEntry = [string, string, number];
  1815→  1815→
  1816→  1816→  // Collect all terminal paths with raw probabilities
  1817→  1817→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
  1818→  1818→
  1819→  1819→  // Iterative DFS
  1820→  1820→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
  1821→  1821→
  1822→  1822→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
  1823→  1823→    const [node, pathStr, pathProb] = stack.pop()!;
  1824→  1824→
  1825→  1825→    const edges = outgoing[node as keyof typeof outgoing];
  1826→  1826→    if (!edges) continue;
  1827→  1827→
  1828→  1828→    // Normalize outgoing edge weights to probabilities for this node
  1829→  1829→    // Map edge types to their weights
  1830→  1830→    let totalWeight = 0;
  1831→  1831→    const edgeWeightsList: number[] = [];
  1832→  1832→    for (const [, edgeType] of edges) {
  1833→  1833→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
  1834→  1834→      edgeWeightsList.push(w);
  1835→  1835→      totalWeight += w;
  1836→  1836→    }
  1837→  1837→
  1838→  1838→    if (totalWeight < 1e-10) continue;
  1839→  1839→
  1840→  1840→    // Expand edges
  1841→  1841→    for (let i = 0; i < edges.length; i++) {
  1842→  1842→      const [target, edgeType] = edges[i];
  1843→  1843→      const edgeProb = edgeWeightsList[i] / totalWeight;
  1844→  1844→      const newProb = pathProb * edgeProb;
  1845→  1845→      const newPath = pathStr + '→' + target;
  1846→  1846→
  1847→  1847→      if (edgeType === 'terminal') {
  1848→  1848→        // Reached a terminal node
  1849→  1849→        const scenario = terminalMap[target] ?? 'R3';
  1850→  1850→        rawPaths.push({ path: newPath, scenario, prob: newProb });
  1851→  1851→      } else {
  1852→  1852→        // Continue DFS
  1853→  1853→        stack.push([target, newPath, newProb]);
  1854→  1854→      }
  1855→  1855→    }
  1856→  1856→  }
  1857→  1857→
  1858→  1858→  // Group raw probabilities by scenario
  1859→  1859→  const rawSums: Record<string, number> = { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0 };
  1860→  1860→  for (const rp of rawPaths) {
  1861→  1861→    rawSums[rp.scenario] = (rawSums[rp.scenario] ?? 0) + rp.prob;
  1862→  1862→  }
  1863→  1863→
  1864→  1864→  // Compute calibration factors: scenarioProbs[scenario] / rawSum[scenario]
  1865→  1865→  const calibrationFactors: Record<string, number> = {};
  1866→  1866→  for (const scenario of ['R1', 'R2', 'R3', 'R4', 'R5']) {
  1867→  1867→    const rawSum = rawSums[scenario] ?? 0;
  1868→  1868→    const targetProb = (scenarioProbs[scenario] ?? 20) / 100; // convert from percentage
  1869→  1869→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
  1870→  1870→  }
  1871→  1871→
  1872→  1872→  // Apply calibration to each path
  1873→  1873→  const calibratedPaths = new Map<string, number>();
  1874→  1874→  for (const rp of rawPaths) {
  1875→  1875→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
  1876→  1876→    calibratedPaths.set(rp.path, rp.prob * factor);
  1877→  1877→  }
  1878→  1878→
  1879→  1879→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
  1880→  1880→}
  1881→  1881→
  1882→  1882→// ═══════════════════════════════════════════════════════════════════════════════
  1883→  1883→// LAYER 4 — Scenario Probability Calculation (Adaptive with ML)
  1884→  1884→// ═══════════════════════════════════════════════════════════════════════════════
  1885→  1885→
  1886→  1886→/**
  1887→  1887→ * Calculates the 5 VDss scenario probabilities (R1-R5) that sum to 100.
  1888→  1888→ * Uses bull/bear consensus, key indicator values, S/R levels, and
  1889→  1889→ * ML-derived adaptive factors for fine-tuning.
  1890→  1890→ */
  1891→  1891→function calculateScenarioProbabilities(
  1892→  1892→  bullConsensus: number,
  1893→  1893→  price: number,
  1894→  1894→  R1: number,
  1895→  1895→  S1: number,
  1896→  1896→  MA100: number,
  1897→  1897→  rsi: number,
  1898→  1898→  mfi: number,
  1899→  1899→  stochK: number,
  1900→  1900→  mlModel: AdaptiveWeightModel
  1901→  1901→): { R1: number; R2: number; R3: number; R4: number; R5: number; factors: { momentum: number; volatility: number; trend: number } } {
  1902→  1902→  // Distance metrics to key levels (spec uses exp(-3 * ...))
  1903→  1903→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0;
  1904→  1904→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0;
  1905→  1905→
  1906→  1906→  // Binary risk flags (per spec) — MFI only used when volume is available
  1907→  1907→  const overboughtRisk = (rsi > 70 || (mfi > 80 && mfi !== 50) || stochK > 80) ? 1 : 0;
  1908→  1908→  const oversoldBounce = (rsi < 30 || (mfi < 20 && mfi !== 50) || stochK < 20) ? 1 : 0;
  1909→  1909→
  1910→  1910→  // Extract adaptive factors from ML model if trained
  1911→  1911→  let momentum = 0.7;
  1912→  1912→  let volatility = 0.5;
  1913→  1913→  let trend = 0.6;
  1914→  1914→
  1915→  1915→  if (mlModel.isTrained && mlModel.weights) {
  1916→  1916→    const coefs = mlModel.getCoefficients();
  1917→  1917→    momentum = sigmoid(coefs[0] ?? 0);   // rsi coefficient → momentum factor
  1918→  1918→    volatility = sigmoid(coefs[10] ?? 0); // atr coefficient → volatility factor
  1919→  1919→    trend = sigmoid(coefs[11] ?? 0);     // trend coefficient → trend factor
  1920→  1920→  }
  1921→  1921→
  1922→  1922→  // Below MA100 flag
  1923→  1923→  const belowMA100 = price < MA100 ? 1 : 0;
  1924→  1924→
  1925→  1925→  // ── Raw scenario probabilities (matching user spec) ─────────────────────
  1926→  1926→
  1927→  1927→  // R1: Strong Bullish — aggressive breakout
  1928→  1928→  const raw_R1 =
  1929→  1929→    bullConsensus * distR1 * (1 - overboughtRisk * momentum * 0.7) +
  1930→  1930→    oversoldBounce * 0.3 * distR1 * momentum;
  1931→  1931→
  1932→  1932→  // R2: Gradual Uptrend — steady rise
  1933→  1933→  const raw_R2 =
  1934→  1934→    ((1 - Math.abs(bullConsensus - 0.65)) * 0.8 +
  1935→  1935→    (bullConsensus > 0.4 ? 0.2 : 0)) * trend;
  1936→  1936→
  1937→  1937→  // R3: Range-bound
  1938→  1938→  const raw_R3 =
  1939→  1939→    (1 - Math.abs(bullConsensus - 0.5)) * 0.6 * (1 + (1 - volatility) * 0.2);
  1940→  1940→
  1941→  1941→  // R4: Moderate Correction
  1942→  1942→  const raw_R4 =
  1943→  1943→    ((1 - bullConsensus) * distS1 * 0.7 +
  1944→  1944→    overboughtRisk * 0.5 * momentum) *
  1945→  1945→    (0.8 + 0.2 * (1 - trend));
  1946→  1946→
  1947→  1947→  // R5: Deep Correction
  1948→  1948→  const raw_R5 =
  1949→  1949→    ((1 - bullConsensus) ** 2 * 0.5 + belowMA100 * 0.2) *
  1950→  1950→    (1 + (1 - trend) * 0.3);
  1951→  1951→
  1952→  1952→  // ── Normalize to sum = 100 ───────────────────────────────────────────
  1953→  1953→  const sum = raw_R1 + raw_R2 + raw_R3 + raw_R4 + raw_R5;
  1954→  1954→  const pR1 = sum > 0 ? Math.round(raw_R1 / sum * 100) : 20;
  1955→  1955→  const pR2 = sum > 0 ? Math.round(raw_R2 / sum * 100) : 20;
  1956→  1956→  const pR3 = sum > 0 ? Math.round(raw_R3 / sum * 100) : 20;
  1957→  1957→  const pR4 = sum > 0 ? Math.round(raw_R4 / sum * 100) : 20;
  1958→  1958→  const pR5 = 100 - (pR1 + pR2 + pR3 + pR4);
  1959→  1959→
