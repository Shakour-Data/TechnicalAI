   169→  if (closes.length < period) return 0;
   170→  const slice = closes.slice(-period);

   174→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   175→function emaCalc(closes: number[], period: number): number {
   176→  if (closes.length < period) return 0;
   177→  const k = 2 / (period + 1);
   178→  let emaVal = sma(closes.slice(0, period), period);
   179→  for (let i = period; i < closes.length; i++) {
   180→    emaVal = closes[i] * k + emaVal * (1 - k);
   181→  }
   182→  return emaVal;
   183→}

   185→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   186→function calcRSI(closes: number[], period: number = 14): number {
   187→  if (closes.length < period + 1) return 50;
   188→  let avgGain = 0;
   189→  let avgLoss = 0;
   190→  for (let i = 1; i <= period; i++) {
   191→    const diff = closes[i] - closes[i - 1];
   192→    if (diff > 0) avgGain += diff;
   193→    else avgLoss += Math.abs(diff);
   194→  }
   195→  avgGain /= period;
   196→  avgLoss /= period;
   197→  for (let i = period + 1; i < closes.length; i++) {
   198→    const diff = closes[i] - closes[i - 1];
   199→    const gain = diff > 0 ? diff : 0;
   200→    const loss = diff < 0 ? Math.abs(diff) : 0;
   201→    avgGain = (avgGain * (period - 1) + gain) / period;
   202→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   203→  }
   204→  if (avgLoss === 0) return 100;
   205→  const rs = avgGain / avgLoss;
   206→  return 100 - 100 / (1 + rs);
   207→}

   209→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   210→function calcMFI(data: OHLCV[], period: number = 14): number {
   211→  if (data.length < period + 1) return 50;
   212→  let posFlow = 0;
   213→  let negFlow = 0;
   214→  for (let i = data.length - period; i < data.length; i++) {
   215→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   216→    const mf = tp * data[i].volume;
   217→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   218→    if (tp > prevTp) posFlow += mf;
   219→    else negFlow += mf;
   220→  }
   221→  if (negFlow === 0) return 100;
   222→  const mfr = posFlow / negFlow;
   223→  return 100 - 100 / (1 + mfr);
   224→}

   227→function calcCCI(data: OHLCV[], period: number = 20): number {
   228→  if (data.length < period) return 0;
   229→  const tps: number[] = [];
   230→  for (let i = data.length - period; i < data.length; i++) {
   231→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   232→  }
   233→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   234→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   235→  if (meanDev === 0) return 0;
   236→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   237→}

   239→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   240→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   241→  if (data.length < kPeriod) return { k: 50, d: 50 };
   242→  const rawKs: number[] = [];
   243→  for (let i = kPeriod - 1; i < data.length; i++) {
   244→    let lowest = Infinity;
   245→    let highest = -Infinity;
   246→    for (let j = i - kPeriod + 1; j <= i; j++) {
   247→      if (data[j].low < lowest) lowest = data[j].low;
   248→      if (data[j].high > highest) highest = data[j].high;
   249→    }
   250→    const range = highest - lowest;
   251→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   252→  }
   253→  const smoothedK: number[] = [];
   254→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   255→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   256→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   257→  }
   258→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   259→  const dSlice = smoothedK.slice(-smoothD);
   260→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   261→  return { k, d };
   262→}

   265→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   266→  if (data.length < period) return -50;

   268→  let highest = -Infinity;
   269→  let lowest = Infinity;
   270→  for (const c of slice) {

   274→  const range = highest - lowest;
   275→  if (range === 0) return -50;
   276→  return ((highest - data[data.length - 1].close) / range) * -100;

   279→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   280→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   281→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   282→  const emaFast = emaCalc(closes, fast);
   283→  const emaSlow = emaCalc(closes, slow);
   284→  const macdLine = emaFast - emaSlow;
   285→  // Calculate signal as EMA of MACD values
   286→  const macdValues: number[] = [];
   287→  const kFast = 2 / (fast + 1);
   288→  const kSlow = 2 / (slow + 1);
   289→  let eFast = sma(closes.slice(0, fast), fast);
   290→  let eSlow = sma(closes.slice(0, slow), slow);
   291→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   292→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   293→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   294→    macdValues.push(eFast - eSlow);
   295→  }
   296→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   297→  const kSig = 2 / (sig + 1);
   298→  let signalVal = sma(macdValues.slice(0, sig), sig);
   299→  for (let i = sig; i < macdValues.length; i++) {
   300→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   301→  }
   302→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   303→}

   305→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   306→function calcATR(data: OHLCV[], period: number = 14): number {
   307→  if (data.length < 2) return 0;
   308→  const trs: number[] = [];
   309→  for (let i = 1; i < data.length; i++) {
   310→    const tr = Math.max(
   311→      data[i].high - data[i].low,
   312→      Math.abs(data[i].high - data[i - 1].close),
   313→      Math.abs(data[i].low - data[i - 1].close)
   314→    );
   315→    trs.push(tr);
   316→  }
   317→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   318→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   319→  for (let i = period; i < trs.length; i++) {
   320→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   321→  }
   322→  return atrVal;
   323→}

   325→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   326→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   327→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   328→  const plusDMs: number[] = [];
   329→  const minusDMs: number[] = [];
   330→  const trs: number[] = [];
   331→  for (let i = 1; i < data.length; i++) {
   332→    const upMove = data[i].high - data[i - 1].high;
   333→    const downMove = data[i - 1].low - data[i].low;
   334→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   335→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   336→    trs.push(Math.max(
   337→      data[i].high - data[i].low,
   338→      Math.abs(data[i].high - data[i - 1].close),
   339→      Math.abs(data[i].low - data[i - 1].close)
   340→    ));
   341→  }
   342→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   343→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   344→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   345→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   346→  const dxValues: number[] = [];
   347→  for (let i = period; i < trs.length; i++) {
   348→    smoothTR = smoothTR - smoothTR / period + trs[i];
   349→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   350→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   351→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   352→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   353→    const diSum = diPlus + diMinus;
   354→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   355→    dxValues.push(dx);
   356→  }
   357→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   358→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   359→  for (let i = period; i < dxValues.length; i++) {
   360→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   361→  }
   362→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   363→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   364→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   365→}

   367→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   368→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   369→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   370→  let isLong = true;
   371→  let af = step;
   372→  let ep = data[0].high;
   373→  let sar = data[0].low;
   374→  for (let i = 1; i < data.length; i++) {
   375→    const prevSar = sar;
   376→    sar = prevSar + af * (ep - prevSar);
   377→    // Ensure SAR is within bounds
   378→    if (isLong) {
   379→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   380→      if (data[i].low < sar) {
   381→        isLong = false;
   382→        sar = ep;
   383→        ep = data[i].low;
   384→        af = step;
   385→      } else {
   386→        if (data[i].high > ep) {
   387→          ep = data[i].high;
   388→          af = Math.min(af + step, max);
   389→        }
   390→      }
   391→    } else {
   392→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   393→      if (data[i].high > sar) {
   394→        isLong = true;
   395→        sar = ep;
   396→        ep = data[i].high;
   397→        af = step;
   398→      } else {
   399→        if (data[i].low < ep) {
   400→          ep = data[i].low;
   401→          af = Math.min(af + step, max);
   402→        }
   403→      }
   404→    }
   405→  }
   406→  return sar;
   407→}

   458→// ─── Helper: Ichimoku Cloud (9, 26, 52) ──────────────────────────────────────
   459→function calcIchimoku(data: OHLCV[]): {
   460→  tenkan: number; kijun: number; senkouA: number; senkouB: number; chikou: number;
   461→} {
   462→  if (data.length < 52) {
   463→    const c = data[data.length - 1]?.close ?? 0;
   464→    return { tenkan: c, kijun: c, senkouA: c, senkouB: c, chikou: c };
   465→  }
   466→  const highs = data.map(d => d.high);
   467→  const lows = data.map(d => d.low);
   468→  const closes = data.map(d => d.close);
   469→
   470→  function donchian(h: number[], l: number[], end: number, period: number): number {
   471→    const start = end - period + 1;
   472→    if (start < 0) return (h[end] + l[end]) / 2;
   473→    let hi = -Infinity, lo = Infinity;
   474→    for (let i = start; i <= end; i++) {
   475→      if (h[i] > hi) hi = h[i];
   476→      if (l[i] < lo) lo = l[i];
   477→    }
   478→    return (hi + lo) / 2;
   479→  }
   480→
   481→  const n = data.length;
   482→  const tenkan = donchian(highs, lows, n - 1, 9);
   483→  const kijun = donchian(highs, lows, n - 1, 26);
   484→  // Senkou A = (tenkan + kijun) / 2, shifted 26 periods ahead
   485→  // We compute it at bar n-27 (shifted to current)
   486→  const senkouAIdx = Math.max(n - 1 - 26, 0);
   487→  const tA = donchian(highs, lows, senkouAIdx, 9);
   488→  const kA = donchian(highs, lows, senkouAIdx, 26);
   489→  const senkouA = (tA + kA) / 2;
   490→  // Senkou B = donchian(52), shifted 26 periods ahead
   491→  const senkouBIdx = Math.max(n - 1 - 26, 0);
   492→  const senkouB = donchian(highs, lows, senkouBIdx, 52);
   493→  // Chikou = current close plotted 26 periods back
   494→  const chikou = closes[n - 1];
   495→
   496→  return { tenkan, kijun, senkouA, senkouB, chikou };
   497→}

   500→  tenkan: number[]; kijun: number[]; senkouA: number[]; senkouB: number[];
   501→} {

   506→  function donchian(end: number, period: number): number {

   509→    let hi = -Infinity, lo = Infinity;
   510→    for (let i = start; i <= end; i++) {
   511→      if (highs[i] > hi) hi = highs[i];
   512→      if (lows[i] < lo) lo = lows[i];
   513→    }
   514→    return (hi + lo) / 2;
   515→  }
   516→
   517→  const tenkan: number[] = [];
   518→  const kijun: number[] = [];
   519→  const senkouA: number[] = [];
   520→  const senkouB: number[] = [];
   521→
   522→  for (let i = 0; i < n; i++) {
   523→    tenkan.push(donchian(i, 9));
   524→    kijun.push(donchian(i, 26));
   525→    // Senkou A shifted +26: at bar i, we need tenkan/kijun at bar i-26
   526→    const sIdx = i - 26;
   527→    if (sIdx >= 0) {
   528→      const t = donchian(sIdx, 9);
   529→      const k = donchian(sIdx, 26);
   530→      senkouA.push((t + k) / 2);
   531→    } else {
   532→      senkouA.push(donchian(i, 9)); // fallback
   533→    }
   534→    // Senkou B shifted +26: at bar i, we need donchian(52) at bar i-26
   535→    if (sIdx >= 0) {
   536→      senkouB.push(donchian(sIdx, 52));
   537→    } else {
   538→      senkouB.push(donchian(i, 52)); // fallback
   539→    }
   540→  }
   541→
   542→  return { tenkan, kijun, senkouA, senkouB };
   543→}

   559→  const result: number[] = [];
   560→  if (closes.length < period) return closes.map(() => 0);
   561→  const k = 2 / (period + 1);
   562→  let emaVal = 0;

   564→  emaVal /= period;

   566→    if (i < period) { result.push(0); continue; }

   568→    emaVal = closes[i] * k + emaVal * (1 - k);

   574→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   575→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   576→  const n = values.length;
   577→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   578→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   579→  for (let i = 0; i < n; i++) {
   580→    sumX += i;
   581→    sumY += values[i];
   582→    sumXY += i * values[i];
   583→    sumX2 += i * i;
   584→    sumY2 += values[i] * values[i];
   585→  }
   586→  const denom = n * sumX2 - sumX * sumX;
   587→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   588→  const slope = (n * sumXY - sumX * sumY) / denom;
   589→  const intercept = (sumY - slope * sumX) / n;
   590→  const meanY = sumY / n;
   591→  let ssRes = 0, ssTot = 0;
   592→  for (let i = 0; i < n; i++) {
   593→    const predicted = slope * i + intercept;
   594→    ssRes += (values[i] - predicted) ** 2;
   595→    ssTot += (values[i] - meanY) ** 2;
   596→  }
   597→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   598→  return { slope, intercept, r2 };
   599→}

   601→// ─── Helper: Round to nice number (per spec price-range rules) ─────────────
   602→function roundToNice(price: number): number {
   603→  const abs = Math.abs(price);
   604→  let step: number;
   605→  if (abs >= 100000) step = 100;
   606→  else if (abs >= 10000) step = 10;
   607→  else if (abs >= 1000) step = 5;
   608→  else if (abs >= 100) step = 1;
   609→  else step = 0.5;
   610→  return Math.round(price / step) * step;
   611→}

   614→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: { price: number; idx: number }[]; lows: { price: number; idx: number }[] } {
   615→  const highs: { price: number; idx: number }[] = [];
   616→  const lows: { price: number; idx: number }[] = [];
   617→  for (let i = lookback; i < data.length - lookback; i++) {
   618→    let isHigh = true;
   619→    let isLow = true;
   620→    for (let j = i - lookback; j <= i + lookback; j++) {
   621→      if (j === i) continue;
   622→      if (data[j].high >= data[i].high) isHigh = false;
   623→      if (data[j].low <= data[i].low) isLow = false;
   624→    }
   625→    if (isHigh) highs.push({ price: data[i].high, idx: i });
   626→    if (isLow) lows.push({ price: data[i].low, idx: i });
   627→  }
   628→  return { highs, lows };
   629→}

   679→// ─── Helper: Psychological Rounding ────────────────────────────────────────
   680→function psychStep(price: number): number {
   681→  if (price >= 100000) return 5000;
   682→  if (price >= 10000) return 1000;
   683→  if (price >= 1000) return 100;
   684→  if (price >= 100) return 10;
   685→  if (price >= 10) return 1;
   686→  return 0.01;
   687→}

   689→function psychRound(price: number): number {
   690→  const step = psychStep(price);
   691→  return Math.round(price / step) * step;

   694→// ─── Helper: Linear System Solver (Gaussian Elimination) ─────────────────
   695→function solveLinearSystem(A: number[][], b: number[]): number[] | null {
   696→  const n = b.length;
   697→  const aug = A.map((row, i) => [...row, b[i]]);
   698→
   699→  for (let col = 0; col < n; col++) {
   700→    let maxVal = Math.abs(aug[col][col]);
   701→    let maxRow = col;
   702→    for (let row = col + 1; row < n; row++) {
   703→      if (Math.abs(aug[row][col]) > maxVal) {
   704→        maxVal = Math.abs(aug[row][col]);
   705→        maxRow = row;
   706→      }
   707→    }
   708→    if (maxVal < 1e-10) return null;
   709→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   710→
   711→    for (let row = col + 1; row < n; row++) {
   712→      const factor = aug[row][col] / aug[col][col];
   713→      for (let j = col; j <= n; j++) {
   714→        aug[row][j] -= factor * aug[col][j];
   715→      }
   716→    }
   717→  }
   718→
   719→  const x = new Array(n).fill(0);
   720→  for (let i = n - 1; i >= 0; i--) {
   721→    x[i] = aug[i][n];
   722→    for (let j = i + 1; j < n; j++) {
   723→      x[i] -= aug[i][j] * x[j];
   724→    }
   725→    if (Math.abs(aug[i][i]) < 1e-10) return null;
   726→    x[i] /= aug[i][i];
   727→  }
   728→
   729→  return x;
   730→}

   732→// ─── Helper: Generate All 7-Method S/R Levels ─────────────────────────────
   733→function generateAllLevels(data: OHLCV[], endIdx: number): RawSRLevel[] {
   734→  const levels: RawSRLevel[] = [];
   735→  if (endIdx < 10) return levels;
   736→
   737→  const slice = data.slice(0, endIdx + 1);
   738→  const currentPrice = slice[slice.length - 1].close;
   739→  const closes = slice.map(d => d.close);
   740→  const noFib: [string, string] = ['—', ''];
   741→
   742→  // ── Method 1: Swing High/Low (Price Action) ──────────────────────────
   743→  for (const lookback of [3, 5, 7]) {
   744→    const swings = findSwingLevels(slice, lookback);
   745→    const tol = currentPrice * 0.005; // 0.5% tolerance
   746→
   747→    // Connect swing highs → resistance
   748→    const usedHigh = new Set<number>();
   749→    for (const h of swings.highs) {
   750→      if (usedHigh.has(h.idx)) continue;
   751→      let sum = h.price;
   752→      let count = 1;
   753→      for (const h2 of swings.highs) {
   754→        if (h2 === h || usedHigh.has(h2.idx)) continue;
   755→        if (Math.abs(h2.price - h.price) < tol) {
   756→          sum += h2.price;
   757→          count++;
   758→          usedHigh.add(h2.idx);
   759→        }
   760→      }
   761→      usedHigh.add(h.idx);
   762→      if (sum / count > currentPrice * 1.005) {
   763→        levels.push({ price: sum / count, method: 'Swing_High', type: 'resistance', ...noFib });
   764→      }
   765→    }
   766→
   767→    // Connect swing lows → support
   768→    const usedLow = new Set<number>();
   769→    for (const l of swings.lows) {
   770→      if (usedLow.has(l.idx)) continue;
   771→      let sum = l.price;
   772→      let count = 1;
   773→      for (const l2 of swings.lows) {
   774→        if (l2 === l || usedLow.has(l2.idx)) continue;
   775→        if (Math.abs(l2.price - l.price) < tol) {
   776→          sum += l2.price;
   777→          count++;
   778→          usedLow.add(l2.idx);
   779→        }
   780→      }
   781→      usedLow.add(l.idx);
   782→      if (sum / count < currentPrice * 0.995) {
   783→        levels.push({ price: sum / count, method: 'Swing_Low', type: 'support', ...noFib });
   784→      }
   785→    }
   786→  }
   787→
   788→  // ── Method 2: SMA Dynamic S/R ──────────────────────────────────────
   789→  for (const period of [20, 50, 100, 200]) {
   790→    const smaVal = sma(closes, period);
   791→    if (smaVal <= 0) continue;
   792→    const type: 'support' | 'resistance' = currentPrice > smaVal ? 'support' : 'resistance';
   793→    levels.push({ price: smaVal, method: `SMA_${period}`, type, ...noFib });
   794→  }
   795→
   796→  // ── Method 3: Bollinger Bands ───────────────────────────────────────
   797→  const bb = calcBollingerBands(closes);
   798→  if (bb.upper > 0 && bb.lower > 0) {
   799→    levels.push({ price: bb.upper, method: 'BB_Upper', type: 'resistance', ...noFib });
   800→    levels.push({ price: bb.lower, method: 'BB_Lower', type: 'support', ...noFib });
   801→  }
   802→
   803→  // ── Method 4: Fibonacci Retracement + Extension ─────────────────────
   804→  interface SwingPair { high: number; highIdx: number; low: number; lowIdx: number; range: number }
   805→  let bestSwing: SwingPair | null = null;
   806→
   807→  for (const lb of [3, 5, 7, 10, 15, 20]) {
   808→    const sw = findSwingLevels(slice, lb);
   809→    if (sw.highs.length === 0 || sw.lows.length === 0) continue;
   810→    for (const h of sw.highs) {
   811→      for (const l of sw.lows) {
   812→        if (Math.abs(h.idx - l.idx) < 3) continue;
   813→        const range = Math.abs(h.price - l.price);
   814→        if (range / currentPrice < 0.03) continue;
   815→        if (!bestSwing || range > bestSwing.range) {
   816→          bestSwing = {
   817→            high: Math.max(h.price, l.price),
   818→            highIdx: h.price > l.price ? h.idx : l.idx,
   819→            low: Math.min(h.price, l.price),
   820→            lowIdx: h.price < l.price ? l.idx : h.idx,
   821→            range,
   822→          };
   823→        }
   824→      }
   825→    }
   826→  }
   827→
   828→  if (bestSwing) {
   829→    const H = bestSwing.high;
   830→    const L = bestSwing.low;
   831→    const range = H - L;
   832→
   833→    for (const ratio of FIB_RETRACEMENTS) {
   834→      const price = H - ratio * range;
   835→      if (price <= 0) continue;
   836→      const rounded = roundToNice(price);
   837→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   838→      levels.push({
   839→        price: rounded, method: `Fibonacci_${ratio}`, type,
   840→        fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   841→      });
   842→    }
   843→
   844→    for (const extRatio of FIB_EXTENSIONS) {
   845→      const price = H + (extRatio - 1) * range;
   846→      if (price <= 0) continue;
   847→      const rounded = roundToNice(price);
   848→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   849→      levels.push({
   850→        price: rounded, method: `Fibonacci_${extRatio}`, type,
   851→        fibRatio: String(extRatio), fibLabel: FIB_LABELS[String(extRatio)] || '',
   852→      });
   853→    }
   854→
   855→    // Inverse extensions (below L)
   856→    for (const extRatio of FIB_EXTENSIONS) {
   857→      const price = L - (extRatio - 1) * range;
   858→      if (price <= 0) continue;
   859→      const rounded = roundToNice(price);
   860→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   861→      levels.push({
   862→        price: rounded, method: `Fibonacci_${-extRatio}`, type,
   863→        fibRatio: String(-extRatio), fibLabel: '',
   864→      });
   865→    }
   866→
   867→    // Secondary swing confluence (check overlap with secondary swings)
   868→    const secondarySwings: SwingPair[] = [];
   869→    for (const lb of [3, 5, 7]) {
   870→      const sw = findSwingLevels(slice, lb);
   871→      for (const h of sw.highs) {
   872→        for (const l of sw.lows) {
   873→          if (Math.abs(h.idx - l.idx) < 3) continue;
   874→          const r = Math.abs(h.price - l.price);
   875→          if (r / currentPrice < 0.05) continue;
   876→          if (bestSwing && Math.abs(r - bestSwing.range) / bestSwing.range < 0.1) continue;
   877→          secondarySwings.push({
   878→            high: Math.max(h.price, l.price), highIdx: h.price > l.price ? h.idx : l.idx,
   879→            low: Math.min(h.price, l.price), lowIdx: h.price < l.price ? l.idx : h.idx,
   880→            range: r,
   881→          });
   882→        }
   883→      }
   884→    }
   885→    // Secondary swings' fib levels are merged via confluence (1% rule in mergeLevels)
   886→    const confluenceTol = currentPrice * 0.005;
   887→    for (const sec of secondarySwings.slice(0, 3)) {
   888→      const sH = sec.high, sL = sec.low, sR = sH - sL;
   889→      for (const ratio of FIB_RETRACEMENTS) {
   890→        const price = roundToNice(sH - ratio * sR);
   891→        if (price <= 0) continue;
   892→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   893→        levels.push({
   894→          price, method: `Fibonacci_${ratio}`, type,
   895→          fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   896→        });
   897→      }
   898→      for (const extR of FIB_EXTENSIONS) {
   899→        const price = roundToNice(sH + (extR - 1) * sR);
   900→        if (price <= 0) continue;
   901→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   902→        levels.push({
   903→          price, method: `Fibonacci_${extR}`, type,
   904→          fibRatio: String(extR), fibLabel: FIB_LABELS[String(extR)] || '',
   905→        });
   906→      }
   907→    }
   908→  }
   909→
   910→  // ── Method 5: Volume Profile (VAP) ──────────────────────────────────
   911→  const dataHasVolume = slice.some(d => d.volume > 0);
   912→  if (dataHasVolume) {
   913→    let minP = Infinity, maxP = -Infinity;
   914→    for (const d of slice) {
   915→      if (d.low < minP) minP = d.low;
   916→      if (d.high > maxP) maxP = d.high;
   917→    }
   918→    const numBins = 40;
   919→    const binSize = Math.max((maxP - minP) / numBins, currentPrice * 0.001);
   920→    const vap = new Array(numBins).fill(0);
   921→    const binCenters: number[] = [];
   922→
   923→    for (let b = 0; b < numBins; b++) {
   924→      binCenters.push(minP + (b + 0.5) * binSize);
   925→    }
   926→
   927→    for (const d of slice) {
   928→      for (let b = 0; b < numBins; b++) {
   929→        if (d.low <= binCenters[b] && binCenters[b] <= d.high) {
   930→          vap[b] += d.volume;
   931→        }
   932→      }
   933→    }
   934→
   935→    const maxVAP = Math.max(...vap);
   936→    if (maxVAP > 0) {
   937→      // POC (Point of Control)
   938→      let pocIdx = 0;
   939→      for (let b = 1; b < numBins; b++) {
   940→        if (vap[b] > vap[pocIdx]) pocIdx = b;
   941→      }
   942→      const pocPrice = binCenters[pocIdx];
   943→      const pocType: 'support' | 'resistance' = pocPrice > currentPrice ? 'resistance' : 'support';
   944→      levels.push({ price: pocPrice, method: 'VAP_POC', type: pocType, ...noFib });
   945→
   946→      // HVN (High Volume Nodes)
   947→      for (let b = 0; b < numBins; b++) {
   948→        if (b === pocIdx) continue;
   949→        if (vap[b] > 0.7 * maxVAP) {
   950→          const hvnPrice = binCenters[b];
   951→          const hvnType: 'support' | 'resistance' = hvnPrice > currentPrice ? 'resistance' : 'support';
   952→          levels.push({ price: hvnPrice, method: 'VAP_HVN', type: hvnType, ...noFib });
   953→        }
   954→      }
   955→    }
   956→  }
   957→
   958→  // ── Method 6: Pivot Points (4 algorithms) ───────────────────────────
   959→  if (endIdx >= 1) {
   960→    const prev = slice[endIdx - 1]; // Previous completed bar
   961→    const pH = prev.high, pL = prev.low, pC = prev.close, pO = prev.open;
   962→
   963→    // Standard Pivot
   964→    {
   965→      const PP = (pH + pL + pC) / 3;
   966→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   967→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   968→      const R3 = pH + 2 * (PP - pL), S3 = pL - 2 * (pH - PP);
   969→ if (R1 > 0) levels.push({ price: R1, method: 'Pivot_R1', type: 'resistance', ...noFib });
   970→      if (R2 > 0) levels.push({ price: R2, method: 'Pivot_R2', type: 'resistance', ...noFib });
   971→      if (R3 > 0) levels.push({ price: R3, method: 'Pivot_R3', type: 'resistance', ...noFib });
   972→      if (S1 > 0) levels.push({ price: S1, method: 'Pivot_S1', type: 'support', ...noFib });
   973→      if (S2 > 0) levels.push({ price: S2, method: 'Pivot_S2', type: 'support', ...noFib });
   974→      if (S3 > 0) levels.push({ price: S3, method: 'Pivot_S3', type: 'support', ...noFib });
   975→    }
   976→
   977→    // Fibonacci Pivot
   978→    {
   979→      const PP = (pH + pL + pC) / 3;
   980→      const range = pH - pL;
   981→      for (const [mult, suffix] of [[0.382, '1'], [0.618, '2'], [1.0, '3']] as [number, string][]) {
   982→        const R = PP + mult * range;
   983→        const S = PP - mult * range;
   984→        if (R > 0) levels.push({ price: R, method: `Fib_Pivot_R${suffix}`, type: 'resistance', ...noFib });
   985→        if (S > 0) levels.push({ price: S, method: `Fib_Pivot_S${suffix}`, type: 'support', ...noFib });
   986→      }
   987→    }
   988→
   989→    // Woodie's Pivot
   990→    {
   991→      const PP = (pH + pL + 2 * pC) / 4;
   992→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   993→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   994→      if (R1 > 0) levels.push({ price: R1, method: 'Woodie_R1', type: 'resistance', ...noFib });
   995→      if (R2 > 0) levels.push({ price: R2, method: 'Woodie_R2', type: 'resistance', ...noFib });
   996→      if (S1 > 0) levels.push({ price: S1, method: 'Woodie_S1', type: 'support', ...noFib });
   997→      if (S2 > 0) levels.push({ price: S2, method: 'Woodie_S2', type: 'support', ...noFib });
   998→    }
   999→
  1000→    // DeMark Pivot
  1001→    {
  1002→      const X = pC < pO ? pH + 2 * pL + pC : pC > pO ? 2 * pH + pL + pC : pH + pL + 2 * pC;
  1003→      const PP = X / 4;
  1004→      const R1 = X / 2 - pL;
  1005→      const S1 = X / 2 - pH;
  1006→      if (R1 > 0) levels.push({ price: R1, method: 'DeMark_R1', type: 'resistance', ...noFib });
  1007→      if (S1 > 0) levels.push({ price: S1, method: 'DeMark_S1', type: 'support', ...noFib });
  1008→    }
  1009→  }
  1010→
  1011→  // ── Method 7: Psychological Levels ──────────────────────────────────
  1012→  {
  1013→    const step = psychStep(currentPrice);
  1014→    const base = psychRound(currentPrice);
  1015→    for (const mult of [-2, -1, 1, 2]) {
  1016→      const price = base + mult * step;
  1017→      if (price <= 0) continue;
  1018→      const type: 'support' | 'resistance' = price > currentPrice ? 'resistance' : 'support';
  1019→      levels.push({ price, method: 'Psychological', type, ...noFib });
  1020→    }
  1021→  }
  1022→
  1023→  return levels;
  1024→}

  1026→// ─── Helper: Merge Nearby Levels (1% Confluence) ──────────────────────────
  1027→function mergeLevels(rawLevels: RawSRLevel[], currentPrice: number): MergedSRLevel[] {
  1028→  if (rawLevels.length === 0) return [];
  1029→
  1030→  const tol = currentPrice * 0.01; // 1% tolerance
  1031→  const merged: MergedSRLevel[] = [];
  1032→  const used = new Set<number>();
  1033→
  1034→  for (let i = 0; i < rawLevels.length; i++) {
  1035→    if (used.has(i)) continue;
  1036→
  1037→    let sumPrice = rawLevels[i].price;
  1038→    let count = 1;
  1039→    const methods = new Set<string>();
  1040→    methods.add(rawLevels[i].method);
  1041→    let bestFibRatio = rawLevels[i].fibRatio;
  1042→    let bestFibLabel = rawLevels[i].fibLabel;
  1043→
  1044→    for (let j = i + 1; j < rawLevels.length; j++) {
  1045→      if (used.has(j)) continue;
  1046→      if (Math.abs(rawLevels[j].price - rawLevels[i].price) < tol) {
  1047→        sumPrice += rawLevels[j].price;
  1048→        count++;
  1049→        methods.add(rawLevels[j].method);
  1050→        used.add(j);
  1051→        // Prefer Fibonacci label if available
  1052→        if (rawLevels[j].fibRatio !== '—' && bestFibRatio === '—') {
  1053→          bestFibRatio = rawLevels[j].fibRatio;
  1054→          bestFibLabel = rawLevels[j].fibLabel;
  1055→        }
  1056→      }
  1057→    }
  1058→    used.add(i);
  1059→
  1060→    const avgPrice = sumPrice / count;
  1061→    const type: 'support' | 'resistance' = avgPrice >= currentPrice ? 'resistance' : 'support';
  1062→
  1063→    merged.push({
  1064→      price: avgPrice,
  1065→      type,
  1066→      methods: [...methods],
  1067→      overlapCount: methods.size,
  1068→      fibRatio: bestFibRatio,
  1069→      fibLabel: bestFibLabel,
  1070→      score: 0,
  1071→      strength: 1,
  1072→      touchCount: 0,
  1073→      volumeRatio: 0,
  1074→      freshness: 0,
  1075→      distancePercent: 0,
  1076→    });
  1077→  }
  1078→
  1079→  return merged;
  1080→}

  1087→  currentPrice: number,

  1089→): number[] {
  1090→  const lookback = Math.min(30, endIdx);
  1091→  const startIdx = endIdx - lookback;
  1092→  const touchTol = currentPrice * 0.002; // 0.2%
  1093→
  1094→  let touchCount = 0;
  1095→  let totalVolAtTouches = 0;
  1096→  let totalVol = 0;
  1097→  let barsSinceLastTouch = lookback;
  1098→
  1099→  for (let i = startIdx; i <= endIdx; i++) {
  1100→    const bar = data[i];
  1101→    totalVol += bar.volume;
  1102→
  1103→    if (bar.low <= levelPrice + touchTol && bar.high >= levelPrice - touchTol) {
  1104→      touchCount++;
  1105→      totalVolAtTouches += bar.volume;
  1106→      barsSinceLastTouch = endIdx - i;
  1107→    }
  1108→  }
  1109→
  1110→  const avgVol = lookback > 0 ? totalVol / lookback : 0;
  1111→  const volumeRatio = (touchCount > 0 && avgVol > 0) ? (totalVolAtTouches / touchCount) / avgVol : 1;
  1112→  const freshness = Math.max(0, Math.min(1, 1 - barsSinceLastTouch / 30));
  1113→  const distancePercent = Math.min(1, Math.abs(levelPrice - currentPrice) / currentPrice);
  1114→
  1115→  // Features: [touch_count, volume_ratio, overlap_count, freshness, distance_percent]
  1116→  return [touchCount, volumeRatio, overlapCount - 1, freshness, distancePercent];
  1117→}

  1119→// ─── Helper: Train S/R Scoring Weights (Ridge OLS) ───────────────────────
  1120→function trainSRWeights(data: OHLCV[]): number[] {
  1121→  const defaultWeights = [0.25, 0.20, 0.25, 0.15, 0.15];
  1122→
  1123→  if (data.length < 50) return defaultWeights;
  1124→
  1125→  const X: number[][] = [];
  1126→  const y: number[] = [];
  1127→
  1128→  for (let i = 40; i <= data.length - 11; i += 5) {
  1129→    const barPrice = data[i].close;
  1130→    const rawLevels = generateAllLevels(data, i);
  1131→    if (rawLevels.length === 0) continue;
  1132→    const merged = mergeLevels(rawLevels, barPrice);
  1133→
  1134→    for (const level of merged) {
  1135→      const features = computeLevelFeatures(level.price, data, i, barPrice, level.overlapCount);
  1136→      X.push(features);
  1137→
  1138→      // Label: did price return within 10 bars?
  1139→      let returned = false;
  1140→      const retTol = barPrice * 0.005;
  1141→      for (let j = i + 1; j <= Math.min(i + 10, data.length - 1); j++) {
  1142→        if (data[j].low <= level.price + retTol && data[j].high >= level.price - retTol) {
  1143→          returned = true;
  1144→          break;
  1145→        }
  1146→      }
  1147→      y.push(returned ? 1 : 0);
  1148→    }
  1149→  }
  1150→
  1151→  if (X.length < 20) return defaultWeights;
  1152→
  1153→  // Ridge OLS: w = (X^T X + λI)^(-1) X^T y
  1154→  const n = X.length;
  1155→  const p = 5;
  1156→  const lambda = 0.01;
  1157→
  1158→  const XtX: number[][] = Array.from({ length: p }, () => new Array(p).fill(0));
  1159→  for (let i = 0; i < n; i++) {
  1160→    for (let a = 0; a < p; a++) {
  1161→      for (let b = 0; b < p; b++) {
  1162→        XtX[a][b] += X[i][a] * X[i][b];
  1163→      }
  1164→    }
  1165→  }
  1166→  for (let a = 0; a < p; a++) XtX[a][a] += lambda;
  1167→
  1168→  const Xty: number[] = new Array(p).fill(0);
  1169→  for (let i = 0; i < n; i++) {
  1170→    for (let a = 0; a < p; a++) {
  1171→      Xty[a] += X[i][a] * y[i];
  1172→    }
  1173→  }
  1174→
  1175→  const weights = solveLinearSystem(XtX, Xty);
  1176→  if (!weights) return defaultWeights;
  1177→  if (weights.some(w => isNaN(w) || !isFinite(w))) return defaultWeights;
  1178→
  1179→  for (let i = 0; i < p; i++) {
  1180→    if (weights[i] < 0) weights[i] = 0;
  1181→  }
  1182→
  1183→  const sum = weights.reduce((a, b) => a + b, 0);
  1184→  if (sum <= 0) return defaultWeights;
  1185→
  1186→  return weights.map(w => w / sum);
  1187→}

  1190→function calcWMAArray(closes: number[], period: number): number[] {

  1200→      denom += weight;

  1484→// ─── Support / Resistance (7-Method ML-Based System) ─────────────────────
  1485→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
  1486→  resistances: number[];
  1487→  supports: number[];
  1488→  supportStrengths: LevelStrength[];
  1489→  resistanceStrengths: LevelStrength[];
  1490→  priceTargets: LevelStrength[];
  1491→} {
  1492→  if (data.length < 10) {
  1493→    return {
  1494→      resistances: [], supports: [],
  1495→      supportStrengths: [], resistanceStrengths: [],
  1496→      priceTargets: [],
  1497→    };
  1498→  }
  1499→
  1500→  // ── Step 1: Generate all levels from 7 methods ────────────────────────
  1501→  const rawLevels = generateAllLevels(data, data.length - 1);
  1502→
  1503→  // ── Step 2: Merge nearby levels (confluence within 1%) ────────────────
  1504→  let merged = mergeLevels(rawLevels, currentPrice);
  1505→
  1506→  // ── Step 3: Train ML weights on historical data ───────────────────────
  1507→  const weights = trainSRWeights(data);
  1508→
  1509→  // ── Step 4: Compute features & score each merged level ────────────────
  1510→  for (const level of merged) {
  1511→    const features = computeLevelFeatures(level.price, data, data.length - 1, currentPrice, level.overlapCount);
  1512→
  1513→    const touchScore = Math.min(features[0] / 5, 1);
  1514→    const volumeScore = Math.min(features[1] / 2, 1);
  1515→    const overlapScore = Math.min(features[2] * 2, 10) / 10;
  1516→    const freshnessScore = features[3];
  1517→    const distanceScore = 1 - features[4];
  1518→
  1519→    const powerScore = (
  1520→      weights[0] * touchScore +
  1521→      weights[1] * volumeScore +
  1522→      weights[2] * overlapScore +
  1523→      weights[3] * freshnessScore +
  1524→      weights[4] * distanceScore
  1525→    ) * 10;
  1526→
  1527→    level.score = Math.max(0, Math.min(10, powerScore));
  1528→    level.strength = Math.max(1, Math.min(10, Math.round(level.score)));
  1529→    level.touchCount = features[0];
  1530→    level.volumeRatio = features[1];
  1531→    level.freshness = features[3];
  1532→    level.distancePercent = features[4];
  1533→  }
  1534→
  1535→  // ── Step 5: Psychological rounding of all level prices ─────────────────
  1536→  for (const level of merged) {
  1537→    level.price = psychRound(level.price);
  1538→  }
  1539→
  1540→  // Re-classify types after rounding
  1541→  for (const level of merged) {
  1542→    level.type = level.price >= currentPrice ? 'resistance' : 'support';
  1543→  }
  1544→
  1545→  // Deduplicate after rounding (keep strongest at same price)
  1546→  {
  1547→    const deduped: MergedSRLevel[] = [];
  1548→    for (const level of merged) {
  1549→      const existing = deduped.find(d => Math.abs(d.price - level.price) < currentPrice * 0.001);
  1550→      if (existing) {
  1551→        if (level.score > existing.score) {
  1552→          Object.assign(existing, level);
  1553→        }
  1554→      } else {
  1555→        deduped.push(level);
  1556→      }
  1557→    }
  1558→    merged = deduped;
  1559→  }
  1560→
  1561→  // ── Step 6: Separate into supports and resistances ─────────────────────
  1562→  let supports = merged.filter(l => l.type === 'support');
  1563→  let resistances = merged.filter(l => l.type === 'resistance');
  1564→
  1565→  // ── Step 7: Distance filtering (strict 5-10% gap rules) ─────────────
  1566→  // Rules:
  1567→  //   1. Consecutive lines of same type: 5% ≤ gap ≤ 10%
  1568→  //   2. Nearest support ≤ 10% below price (can be < 5%)
  1569→  //   3. Nearest resistance ≤ 10% above price (can be < 5%)
  1570→
  1571→  const filterAndSelectLevels = (
  1572→    levels: MergedSRLevel[],
  1573→    type: 'support' | 'resistance'
  1574→  ): MergedSRLevel[] => {
  1575→    if (levels.length === 0) return [];
  1576→
  1577→    // Sort: supports descending (nearest first), resistances ascending (nearest first)
  1578→    const sorted = [...levels].sort((a, b) =>
  1579→      type === 'support' ? b.price - a.price : a.price - b.price
  1580→    );
  1581→
  1582→    // ── Rule 2/3: Nearest level must be within 10% of price ──
  1583→    const nearest = sorted[0];
  1584→    const distFromPrice = Math.abs(nearest.price - currentPrice) / currentPrice;
  1585→    if (distFromPrice > 0.10) {
  1586→      // Nearest level is too far — generate a synthetic level at ~5% from price
  1587→      const syntheticPrice = type === 'support'
  1588→        ? Math.round(currentPrice * 0.95 / psychStep(currentPrice)) * psychStep(currentPrice)
  1589→        : Math.round(currentPrice * 1.05 / psychStep(currentPrice)) * psychStep(currentPrice);
  1590→      const synthetic: MergedSRLevel = {
  1591→        price: syntheticPrice,
  1592→        type,
  1593→        score: 3,
  1594→        strength: 3,
  1595→        overlapCount: 1,
  1596→        methods: ['Psychological'],
  1597→        fibRatio: '—',
  1598→        fibLabel: '',
  1599→      };
  1600→      sorted.unshift(synthetic);
  1601→    }
  1602→
  1603→    // ── Rule 1: Select levels with 5-10% gaps between consecutive ──
  1604→    const result: MergedSRLevel[] = [sorted[0]];
  1605→
  1606→    for (let i = 1; i < sorted.length && result.length < 6; i++) {
  1607→      const prevPrice = result[result.length - 1].price;
  1608→      const currPrice = sorted[i].price;
  1609→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
  1610→
  1611→      if (gap < 0.05) {
  1612→        // Too close — keep the stronger one
  1613→        if (sorted[i].score > result[result.length - 1].score) {
  1614→          result[result.length - 1] = sorted[i];
  1615→        }
  1616→      } else if (gap <= 0.10) {
  1617→        // Perfect gap (5-10%) — accept
  1618→        result.push(sorted[i]);
  1619→      } else {
  1620→        // Gap > 10% — try to find an intermediate level
  1621→        let found = false;
  1622→        for (let j = i + 1; j < sorted.length; j++) {
  1623→          const midPrice = sorted[j].price;
  1624→          const gapFromPrev = Math.abs(midPrice - prevPrice) / prevPrice;
  1625→          const gapToNext = Math.abs(currPrice - midPrice) / midPrice;
  1626→          if (gapFromPrev >= 0.05 && gapFromPrev <= 0.10) {
  1627→            result.push(sorted[j]);
  1628→            found = true;
  1629→            // Skip levels that are too close to the inserted one
  1630→            i = j;
  1631→            break;
  1632→          }
  1633→        }
  1634→        if (!found) {
  1635→          // No intermediate found — generate a synthetic level at ~7.5% from prev
  1636→          const synthGap = 0.075;
  1637→          const syntheticPrice = type === 'support'
  1638→            ? Math.round((prevPrice * (1 - synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice)
  1639→            : Math.round((prevPrice * (1 + synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice);
  1640→          // Make sure it's not too close to next level
  1641→          const gapToNext = Math.abs(currPrice - syntheticPrice) / syntheticPrice;
  1642→          if (gapToNext >= 0.05) {
  1643→            const synthetic: MergedSRLevel = {
  1644→              price: syntheticPrice,
  1645→              type,
  1646→              score: 2,
  1647→              strength: 2,
  1648→              overlapCount: 1,
  1649→              methods: ['Psychological'],
  1650→              fibRatio: '—',
  1651→              fibLabel: '',
  1652→            };
  1653→            result.push(synthetic);
  1654→          }
  1655→        }
  1656→      }
  1657→    }
  1658→
  1659→    return result;
  1660→  };
  1661→
  1662→  const finalSupports = filterAndSelectLevels(supports, 'support');
  1663→  const finalResistances = filterAndSelectLevels(resistances, 'resistance');
  1664→
  1665→  // ── Step 9: Build LevelStrength objects ────────────────────────────────
  1666→  const toLevelStrength = (m: MergedSRLevel): LevelStrength => {
  1667→    let grade: string;
  1668→    if (m.score >= 8.5) grade = 'Very Strong';
  1669→    else if (m.score >= 7) grade = 'Strong';
  1670→    else if (m.score >= 5) grade = 'Moderate';
  1671→    else grade = 'Weak';
  1672→
  1673→    return {
  1674→      price: m.price,
  1675→      strength: m.strength,
  1676→      score: m.score,
  1677→      grade,
  1678→      isTarget: m.score >= 7,
  1679→      overlapCount: m.overlapCount,
  1680→      methods: m.methods,
  1681→      fibRatio: m.fibRatio,
  1682→      fibLabel: m.fibLabel,
  1683→    };
  1684→  };
  1685→
  1686→  const supportStrengths = finalSupports.map(toLevelStrength);
  1687→  const resistanceStrengths = finalResistances.map(toLevelStrength);
  1688→
  1689→  // Price targets: all levels with isTarget=true
  1690→  const priceTargets = [...supportStrengths, ...resistanceStrengths].filter(l => l.isTarget);
  1691→
  1692→  return {
  1693→    resistances: resistanceStrengths.map(l => l.price),
  1694→    supports: supportStrengths.map(l => l.price),
  1695→    supportStrengths,
  1696→    resistanceStrengths,
  1697→    priceTargets,
  1698→  };
  1699→}

  1703→function calcTrend(closes: number[], period: number): TrendResult {
  1704→  const slice = closes.slice(-Math.min(period, closes.length));
  1705→  const { slope, r2 } = linearRegression(slice);
  1706→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
  1707→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
  1708→  const priceThreshold = avgPrice * 0.001; // 0.1%
  1709→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
  1710→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
  1711→}

  1718→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
  1719→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
  1720→  const slice = data.slice(0, barIdx + 1);
  1721→  const price = slice[slice.length - 1].close;
  1722→  let nearestR = price * 1.05;
  1723→  let nearestS = price * 0.95;
  1724→
  1725→  // Collect all swing highs (resistance) and swing lows (support)
  1726→  const resistances: number[] = [];
  1727→  const supports: number[] = [];
  1728→  for (const lb of [3, 5, 7]) {
  1729→    const swings = findSwingLevels(slice, lb);
  1730→    resistances.push(...swings.highs);
  1731→    supports.push(...swings.lows);
  1732→  }
  1733→
  1734→  // Find nearest resistance above price
  1735→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
  1736→  if (abovePrice.length > 0) nearestR = abovePrice[0];
  1737→
  1738→  // Find nearest support below price
  1739→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
  1740→  if (belowPrice.length > 0) nearestS = belowPrice[0];
  1741→
  1742→  return { R1: nearestR, S1: nearestS };
  1743→}

  1745→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
  1746→// Returns true if bullish divergence detected, false if bearish, null if none
  1747→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
  1748→  const lookback = 20;
  1749→  const start = Math.max(0, barIdx - lookback);
  1750→  const slice = data.slice(start, barIdx + 1);
  1751→  if (slice.length < 10) return null;
  1752→
  1753→  const mid = Math.floor(slice.length / 2);
  1754→  const firstHalf = slice.slice(0, mid);
  1755→  const secondHalf = slice.slice(mid);
  1756→
  1757→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
  1758→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
  1759→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
  1760→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
  1761→
  1762→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
  1763→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
  1764→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
  1765→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
  1766→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
  1767→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
  1768→
  1769→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
  1770→    return false; // bearish divergence
  1771→  }
  1772→
  1773→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
  1774→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
  1775→
  1776→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
  1777→    return true; // bullish divergence
  1778→  }
  1779→
  1780→  return null;
  1781→}

  1787→/**
  1788→ * Computes all 16 ML features at a specific historical bar.
  1789→ * Uses data[0..barIdx] (inclusive) for all indicator calculations.
  1790→ * Returns an object keyed by feature name matching FEATURES_ORDER.
  1791→ */
  1792→function computeFeaturesAtBar(data: OHLCV[], barIdx: number): Record<string, number> {
  1793→  // Need at least 30 bars for meaningful indicators
  1794→  if (barIdx < 30) {
  1795→    const empty: Record<string, number> = {};
  1796→    for (const key of FEATURES_ORDER) empty[key] = 0.5;
  1797→    return empty;
  1798→  }
  1799→
  1800→  const slice = data.slice(0, barIdx + 1);
  1801→  const closes = slice.map(d => d.close);
  1802→  const price = closes[closes.length - 1];
  1803→  const eps = 1e-10;
  1804→
  1805→  // ── Compute all base indicators ───────────────────────────────────────────
  1806→  const rsiVal = calcRSI(closes);
  1807→  const sliceHasVolume = slice.some(d => d.volume > 0);
  1808→  const mfiVal = sliceHasVolume ? calcMFI(slice) : 50; // neutral when no volume
  1809→  const cciVal = calcCCI(slice);
  1810→  const stoch = calcStochastic(slice);
  1811→  const macd = calcMACD(closes);
  1812→  const adxResult = calcADX(slice);
  1813→  const atrVal = calcATR(slice);
  1814→  const bb = calcBollingerBands(closes);
  1815→  const ma21 = sma(closes, 21);
  1816→  const ma100 = sma(closes, 100);
  1817→  const ema12 = emaCalc(closes, 12);
  1818→  const ema26 = emaCalc(closes, 26);
  1819→  const trendResult = calcTrend(closes, 21);
  1820→  const sr = simplifiedSR(data, barIdx);
  1821→
  1822→  // ── Layer 1: Raw Scores (s_*) ────────────────────────────────────────────
  1823→
  1824→  // s_rsi — RSI mapped to 0-1
  1825→  const s_rsi = rsiVal > 80 ? 0.95
  1826→    : rsiVal > 70 ? 0.85
  1827→    : rsiVal > 60 ? 0.70
  1828→    : rsiVal > 50 ? 0.55
  1829→    : rsiVal > 40 ? 0.45
  1830→    : rsiVal > 30 ? 0.30
  1831→    : rsiVal > 20 ? 0.15
  1832→    : 0.05;
  1833→
  1834→  // s_mfi — MFI mapped to 0-1 (same formula as RSI)
  1835→  const s_mfi = mfiVal > 80 ? 0.95
  1836→    : mfiVal > 70 ? 0.85
  1837→    : mfiVal > 60 ? 0.70
  1838→    : mfiVal > 50 ? 0.55
  1839→    : mfiVal > 40 ? 0.45
  1840→    : mfiVal > 30 ? 0.30
  1841→    : mfiVal > 20 ? 0.15
  1842→    : 0.05;
  1843→
  1844→  // s_cci — CCI mapped to 0-1
  1845→  const s_cci = cciVal > 200 ? 0.90
  1846→    : cciVal > 100 ? 0.75
  1847→    : cciVal > 0 ? 0.60
  1848→    : cciVal > -100 ? 0.40
  1849→    : cciVal > -200 ? 0.25
  1850→    : 0.10;
  1851→
  1852→  // s_adx — Directional bias
  1853→  const s_adx = (adxResult.diPlus - adxResult.diMinus + 100) / 200;
  1854→
  1855→  // s_macd — MACD histogram + signal combined
  1856→  const absMacdLine = Math.abs(macd.line) + 1;
  1857→  const s_macd = macd.line > macd.signal
  1858→    ? 0.5 + Math.min(0.5, macd.histogram / absMacdLine)
  1859→    : 0.5 - Math.min(0.5, Math.abs(macd.histogram) / absMacdLine);
  1860→
  1861→  // s_stoch — Stochastic %K mapped to 0-1
  1862→  const s_stoch = stoch.k / 100;
  1863→
  1864→  // s_bb — Bollinger Band position
  1865→  const bbRange = bb.upper - bb.lower;
  1866→  const s_bb = bbRange > eps
  1867→    ? clamp((price - bb.lower) / bbRange, 0, 1)
  1868→    : 0.5;
  1869→
  1870→  // s_ma21 — Price relative to 21-period SMA
  1871→  const s_ma21 = ma21 > eps
  1872→    ? clamp(0.5 + (price - ma21) / ma21, 0, 1)
  1873→    : 0.5;
  1874→
  1875→  // s_ma100 — Price relative to 100-period SMA
  1876→  const s_ma100 = ma100 > eps
  1877→    ? clamp(0.5 + (price - ma100) / ma100, 0, 1)
  1878→    : 0.5;
  1879→
  1880→  // s_ema — EMA12 vs EMA26 spread
  1881→  const s_ema = ema26 > eps
  1882→    ? clamp(0.5 + (ema12 - ema26) / ema26 * 5, 0, 1)
  1883→    : 0.5;
  1884→
  1885→  // s_atr — Inverse volatility (lower ATR = higher score)
  1886→  const s_atr = price > eps ? 1 - Math.min(atrVal / price * 10, 1) : 0.5;
  1887→
  1888→  // s_trend — R² + angle combination
  1889→  const s_trend = (trendResult.r2 * 0.7) +
  1890→    (Math.max(0, Math.min(1, (trendResult.angle + 45) / 90)) * 0.3);
  1891→
  1892→  // s_sr — Support/Resistance proximity
  1893→  const dR1 = (sr.R1 - price) / (price > eps ? price : 1);
  1894→  const dS1 = (price - sr.S1) / (price > eps ? price : 1);
  1895→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
  1896→
  1897→  // ── Layer 2: Corrections (f_*) ───────────────────────────────────────────
  1898→
  1899→  // f_rsi — RSI with momentum correction
  1900→  const rsiCurrent = rsiVal;
  1901→  const rsi5ago = barIdx >= 5
  1902→    ? calcRSI(closes.slice(0, barIdx - 5 + 1))
  1903→    : rsiCurrent;
  1904→  const f_rsi = clamp(s_rsi + (rsiCurrent - rsi5ago) / 100 * 0.08, 0, 1);
  1905→
  1906→  // f_mfi — MFI with momentum correction (skip when no volume)
  1907→  const mfiCurrent = mfiVal;
  1908→  const mfi5ago = (barIdx >= 5 && sliceHasVolume)
  1909→    ? calcMFI(data.slice(0, barIdx - 5 + 1))
  1910→    : mfiCurrent;
  1911→  const f_mfi = sliceHasVolume
  1912→    ? clamp(s_mfi + (mfiCurrent - mfi5ago) / 100 * 0.08, 0, 1)
  1913→    : 0.5; // neutral when no volume
  1914→
  1915→  // f_cci — CCI with momentum correction (scale: /200 instead of /100)
  1916→  const cciCurrent = cciVal;
  1917→  const cci5ago = barIdx >= 5
  1918→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
  1919→    : cciCurrent;
  1920→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
  1921→
  1922→  // f_macd — Same as s_macd (no additional correction)
  1923→  const f_macd = s_macd;
  1924→
  1925→  // f_stoch — Same as s_stoch
  1926→  const f_stoch = s_stoch;
  1927→
  1928→  // f_stochCross — Stochastic K/D crossover detection
  1929→  let f_stochCross = 0.1;
  1930→  if (barIdx >= 1) {
  1931→    const prevSlice = data.slice(0, barIdx);
  1932→    const prevStoch = calcStochastic(prevSlice);
  1933→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
  1934→      f_stochCross = 0.9; // K crossed above D (bullish)
  1935→    }
  1936→  }
  1937→
  1938→  // f_macdCross — MACD/Signal crossover detection
  1939→  let f_macdCross = 0.1;
  1940→  if (barIdx >= 1) {
  1941→    const prevCloses = closes.slice(0, -1);
  1942→    if (prevCloses.length >= 35) {
  1943→      const prevMacd = calcMACD(prevCloses);
  1944→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
  1945→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
  1946→      }
  1947→    }
  1948→  }
  1949→
  1950→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
  1951→  const divResult = detectDivergenceSimple(data, barIdx);
  1952→  const f_div = divResult === true ? 0.85 : 0.15;
  1953→
  1954→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
  1955→  return {
  1956→    rsi: f_rsi,
  1957→    mfi: f_mfi,
  1958→    cci: f_cci,
  1959→    adx: s_adx,
  1960→    macd: f_macd,
  1961→    stoch: f_stoch,
  1962→    bb: s_bb,
  1963→    ma21: s_ma21,
  1964→    ma100: s_ma100,
  1965→    ema: s_ema,
  1966→    atr: s_atr,
  1967→    trend: s_trend,
  1968→    sr: s_sr,
  1969→    stochCross: f_stochCross,
  1970→    macdCross: f_macdCross,
  1971→    div: f_div,
  1972→  };
  1973→}

  1979→/**
  1980→ * Creates supervised training data from historical OHLCV data.
  1981→ * For each bar, computes features and labels based on future price movement.
  1982→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
  1983→ * Samples every `sampleStep` bars to reduce computation.
  1984→ */
  1985→function createTrainingData(
  1986→  data: OHLCV[],
  1987→  horizon: number = 5,
  1988→  sampleStep: number = 5
  1989→): { X: number[][]; y: number[] } {
  1990→  const X: number[][] = [];
  1991→  const y: number[] = [];
  1992→
  1993→  // Start from horizon+30 to ensure enough data for indicators
  1994→  // End at data.length - horizon - 1 to have future bars for labeling
  1995→  // Sample every sampleStep bars for performance
  1996→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
  1997→    const features = computeFeaturesAtBar(data, i);
  1998→
  1999→    // Convert to array following FEATURES_ORDER
  2000→    const featureVec: number[] = [];
  2001→    let valid = true;
  2002→    for (const key of FEATURES_ORDER) {
  2003→      const val = features[key];
  2004→      if (val === undefined || val === null || Number.isNaN(val)) {
  2005→        valid = false;
  2006→        break;
  2007→      }
  2008→      featureVec.push(val);
  2009→    }
  2010→    if (!valid) continue;
  2011→
  2012→    // Label: 1 if future close > 1% above current close
  2013→    const currentClose = data[i].close;
  2014→    const futureClose = data[i + horizon].close;
  2015→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
  2016→
  2017→    X.push(featureVec);
  2018→    y.push(label);
  2019→  }
  2020→
  2021→  return { X, y };
  2022→}
