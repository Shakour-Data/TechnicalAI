   353→/**
   354→ * Parse a number that might contain Persian digits, commas, or other formatting.
   355→ * Handles null/undefined/empty strings gracefully.
   356→ */
   357→function parsePersianNum(str: string | null | undefined | number): number {
   358→  if (str === null || str === undefined) return 0;
   359→  let s = String(str).trim();
   360→  if (!s || s === '-' || s === '') return 0;
   361→  // Strip trailing % sign (e.g. "0.05%" → "0.05")
   362→  if (s.endsWith('%')) s = s.slice(0, -1).trim();
   363→  // Replace commas and Persian digits
   364→  const cleaned = s.replace(/[,۰-۹]/g, (c: string) => {
   365→    const persianDigits = '۰۱۲۳۴۵۶۷۸۹';
   366→    const idx = persianDigits.indexOf(c);
   367→    return idx >= 0 ? String(idx) : c === ',' ? '' : c;
   368→  });
   369→  const n = Number(cleaned);
   370→  return isFinite(n) ? n : 0;
   371→}

   373→/**
   374→ * Check if a value looks like a date string (contains / or - and is not a number).
   375→ */
   376→function looksLikeDate(val: string | null | undefined | number): boolean {
   377→  if (!val) return false;
   378→  const s = String(val).trim();
   379→  return /[\/\-]/.test(s) && isNaN(Number(s.replace(/[,۰-۹\/\-]/g, '')));
   380→}
   381→
   382→/**
   383→ * Check if a value looks like a percentage (ends with % or is small like 0.5 to 50).
   384→ */
   385→function looksLikePercentage(val: string | null | undefined | number): boolean {
   386→  if (!val) return false;
   387→  const s = String(val).trim();
   388→  if (s.endsWith('%')) return true;
   389→  const n = parsePersianNum(val);
   390→  return !isNaN(n) && Math.abs(n) < 100 && n !== 0;
   391→}

   393→/**
   394→ * Auto-detect column mapping for TGJU API response.
   395→ * Expected format: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   396→ * But the API may change column order.
   397→ */
   398→function detectColumnMap(row: (string | number)[]): {
   399→  openIdx: number; highIdx: number; lowIdx: number; closeIdx: number;
   400→  jalaliDateIdx: number; gregorianDateIdx: number;
   401→} | null {
   402→  if (!row || row.length < 6) return null;
   403→
   404→  let jalaliDateIdx = -1;
   405→  let gregorianDateIdx = -1;
   406→  const numericIndices: number[] = [];
   407→  const percentageIndices: number[] = [];
   408→
   409→  for (let i = 0; i < row.length; i++) {
   410→    const val = row[i];
   411→    if (looksLikeDate(val)) {
   412→      // Jalali dates have Persian digits or format like 1404/05/25
   413→      const s = String(val);
   414→      if (/[۰-۹]/.test(s) || /^14\d{2}/.test(s.replace(/[۰-۹]/g, (d: string) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d))))) {
   415→        if (jalaliDateIdx === -1) jalaliDateIdx = i;
   416→      } else {
   417→        if (gregorianDateIdx === -1) gregorianDateIdx = i;
   418→      }
   419→    } else if (looksLikePercentage(val)) {
   420→      percentageIndices.push(i);
   421→    } else {
   422→      const n = parsePersianNum(val);
   423→      if (isFinite(n) && n !== 0) {
   424→        numericIndices.push(i);
   425→      }
   426→    }
   427→  }
   428→
   429→  // If we couldn't auto-detect dates, fall back to assumed positions
   430→  if (jalaliDateIdx === -1 && row.length >= 8) {
   431→    // Default assumption: row[6] = gregorian, row[7] = jalali
   432→    if (looksLikeDate(row[7])) jalaliDateIdx = 7;
   433→    else if (looksLikeDate(row[6])) jalaliDateIdx = 6;
   434→    if (gregorianDateIdx === -1 && looksLikeDate(row[6]) && jalaliDateIdx !== 6) gregorianDateIdx = 6;
   435→    else if (gregorianDateIdx === -1 && looksLikeDate(row[7]) && jalaliDateIdx !== 7) gregorianDateIdx = 7;
   436→  }
   437→
   438→  // Numeric columns (excluding percentages) should be OHLC + change
   439→  // OHLC: typically 4 largest absolute values among non-percentage numerics
   440→  // change: the remaining one
   441→  const ohlcCandidates = numericIndices.filter(i => !percentageIndices.includes(i));
   442→
   443→  if (ohlcCandidates.length < 4) {
   444→    // Fallback: assume first 4 numeric columns are OHLC
   445→    if (numericIndices.length >= 4) {
   446→      return {
   447→        openIdx: numericIndices[0],
   448→        highIdx: numericIndices[2],
   449→        lowIdx: numericIndices[1],
   450→        closeIdx: numericIndices[3],
   451→        jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : 7,
   452→        gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : 6,
   453→      };
   454→    }
   455→    return null;
   456→  }
   457→
   458→  // Sort by absolute value descending to identify OHLC (largest) vs change (smallest)
   459→  const sorted = ohlcCandidates
   460→    .map(i => ({ idx: i, val: Math.abs(parsePersianNum(row[i])) }))
   461→    .sort((a, b) => b.val - a.val);
   462→
   463→  // The 4 largest should be OHLC. Among them, identify high (max) and low (min).
   464→  // open and close are the other two.
   465→  const top4 = sorted.slice(0, 4);
   466→  const maxIdx = top4.reduce((a, b) => a.val > b.val ? a : b).idx;
   467→  const minIdx = top4.reduce((a, b) => a.val < b.val ? a : b).idx;
   468→  const others = top4.filter(x => x.idx !== maxIdx && x.idx !== minIdx);
   469→
   470→  // Among the two remaining, the one closer to high is open, the other is close
   471→  // (heuristic: in an uptrend open < close, but we can't assume direction)
   472→  // Default: first = open, second = close
   473→  const openIdx = others[0]?.idx ?? ohlcCandidates[0];
   474→  const closeIdx = others[1]?.idx ?? ohlcCandidates[3];
   475→
   476→  return {
   477→    openIdx,
   478→    highIdx: maxIdx,
   479→    lowIdx: minIdx,
   480→    closeIdx,
   481→    jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : row.length - 1,
   482→    gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : row.length - 2,
   483→  };
   484→}

   486→/**
   487→ * Validate a candle's OHLC values for sanity.
   488→ * Returns true if the candle is valid.
   489→ */
   490→function isValidCandle(o: number, h: number, l: number, c: number): boolean {
   491→  if (!isFinite(o) || !isFinite(h) || !isFinite(l) || !isFinite(c)) return false;
   492→  if (o <= 0 || h <= 0 || l <= 0 || c <= 0) return false;
   493→  // High must be >= Low, Open, Close
   494→  if (h < l || h < o || h < c) return false;
   495→  // Low must be <= High, Open, Close
   496→  if (l > o || l > c) return false;
   497→  // No infinite or extreme values
   498→  if (o > 1e15 || h > 1e15 || l > 1e15 || c > 1e15) return false;
   499→  return true;
   500→}

   543→/* ─── Fetch live instruments (currency + gold from tgju-api) + static list ── */

   545→export async function fetchTgjuInstruments(): Promise<TgjuInstrument[]> {
   546→  const now = Date.now();
   547→  if (liveInstrumentsCache.data && now - liveInstrumentsCache.ts < INSTRUMENTS_TTL) {
   548→    return liveInstrumentsCache.data;
   549→  }
   550→
   551→  const controller = new AbortController();
   552→  const timeout = setTimeout(() => controller.abort(), 10000);
   553→
   554→  try {
   555→    const [currencyRes, goldRes] = await Promise.all([
   556→      fetch(`${TGJU_API_BASE}/api/price/currency`, { signal: controller.signal }),
   557→      fetch(`${TGJU_API_BASE}/api/price/gold`, { signal: controller.signal }),
   558→    ]);
   559→
   560→    clearTimeout(timeout);
   561→
   562→    const instruments: TgjuInstrument[] = [];
   563→
   564→    if (currencyRes.ok) {
   565→      const currencies: TgjuPriceItem[] = await currencyRes.json();
   566→      for (const c of currencies) {
   567→        const price = parsePersianNum(c.price);
   568→        const low = parsePersianNum(c.low_price);
   569→        const high = parsePersianNum(c.high_price);
   570→        instruments.push({
   571→          title: c.title,
   572→          key: c.key,
   573→          price,
   574→          highPrice: high,
   575→          lowPrice: low,
   576→          change: 0,
   577→          changePercent: 0,
   578→          category: 'currency',
   579→        });
   580→      }
   581→    }
   582→
   583→    if (goldRes.ok) {
   584→      const goldCategories: TgjuGoldCategory[] = await goldRes.json();
   585→      for (const cat of goldCategories) {
   586→        for (const item of cat.prices) {
   587→          const price = parsePersianNum(item.price);
   588→          const low = parsePersianNum(item.low_price);
   589→          const high = parsePersianNum(item.high_price);
   590→
   591→          let category: TgjuCategoryType = 'gold';
   592→          if (item.key.startsWith('ime_fund_')) category = 'gold_etf';
   593→          else if (item.key.startsWith('silver_')) category = 'silver';
   594→
   595→          instruments.push({
   596→            title: item.title,
   597→            key: item.key,
   598→            price,
   599→            highPrice: high,
   600→            lowPrice: low,
   601→            change: 0,
   602→            changePercent: 0,
   603→            category,
   604→            groupTitle: cat.title,
   605→          });
   606→        }
   607→      }
   608→    }
   609→
   610→    // Add static instruments (crypto, world indices, forex, energy, metals, commodities)
   611→    for (const s of STATIC_INSTRUMENTS) {
   612→      const cached = staticPriceCache.get(s.key);
   613→      const price = cached?.close ?? 0;
   614→      instruments.push({
   615→        title: s.title,
   616→        key: s.key,
   617→        price,
   618→        highPrice: cached?.high ?? 0,
   619→        lowPrice: cached?.low ?? 0,
   620→        change: cached ? (cached.close - cached.prevClose) : 0,
   621→        changePercent: cached && cached.prevClose > 0 ? ((cached.close - cached.prevClose) / cached.prevClose) * 100 : 0,
   622→        category: s.category,
   623→        groupTitle: s.groupTitle,
   624→      });
   625→    }
   626→
   627→    liveInstrumentsCache.data = instruments;
   628→    liveInstrumentsCache.ts = now;
   629→
   630→    // ── Background: fetch chart data for currency/gold to get accurate daily change ──
   631→    backgroundFetchCurrencyGoldChange(instruments).catch(() => {});
   632→
   633→    // ── Background: fetch Yahoo prices for static instruments with price=0 (TGJU API 403) ──
   634→    backgroundFetchYahooFallbackPrices(instruments).catch(() => {});
   635→
   636→    return instruments;
   637→  } catch {
   638→    clearTimeout(timeout);
   639→    return liveInstrumentsCache.data || [];
   640→  }
   641→}

   643→/**
   644→ * Background task: fetch last 2 candles for currency/gold items to compute accurate daily change.
   645→ * Uses direct fetch to avoid queue contention.
   646→ * Only updates change/changePercent — keeps the live price from tgju-api.
   647→ */
   648→async function backgroundFetchCurrencyGoldChange(instruments: TgjuInstrument[]) {
   649→  const currencyGoldItems = instruments.filter(
   650→    (i) => i.category === 'currency' || i.category === 'gold' || i.category === 'silver' || i.category === 'gold_etf'
   651→  );
   652→
   653→  // Limit to first 10 to avoid rate-limiting (most important: USD, EUR, gold coin, etc.)
   654→  const priorityKeys = currencyGoldItems.slice(0, 10).map((i) => i.key);
   655→
   656→  for (const key of priorityKeys) {
   657→    try {
   658→      const url = `${TGJU_CHART_API}/${key}?lang=fa&order_dir=desc&start=0&length=2`;
   659→      const res = await fetch(url, { signal: AbortSignal.timeout(10_000) });
   660→      if (!res.ok) continue;
   661→      const jsonStr = await res.text();
   662→      if (!jsonStr.startsWith('{')) continue;
   663→
   664→      const data = JSON.parse(jsonStr);
   665→      const rows: string[][] = data.data;
   666→      if (!rows || rows.length < 2) continue;
   667→
   668→      const newest = rows[0];
   669→      const prev = rows[1];
   670→      const latestClose = parsePersianNum(newest[3]);
   671→      const prevClose = parsePersianNum(prev[3]);
   672→
   673→      if (latestClose > 0 && prevClose > 0) {
   674→        const inst = liveInstrumentsCache.data?.find((i) => i.key === key);
   675→        if (inst) {
   676→          inst.change = latestClose - prevClose;
   677→          inst.changePercent = ((latestClose - prevClose) / prevClose) * 100;
   678→        }
   679→      }
   680→    } catch {
   681→      // skip individual errors
   682→    }
   683→  }
   684→}

   694→/**
   695→ * Background task: fetch Yahoo Finance prices for static instruments that have price=0
   696→ * (happens when api.tgju.org returns 403). Updates the live cache in-place.
   697→ */
   698→async function backgroundFetchYahooFallbackPrices(instruments: TgjuInstrument[]) {
   699→  // Collect static instruments with price=0 that have a Yahoo mapping
   700→  const zeroPriceItems = instruments.filter(
   701→    (i) => i.price === 0 && i.key in TGJU_TO_YAHOO_MAP
   702→  );
   703→
   704→  if (zeroPriceItems.length === 0) return;
   705→
   706→  console.log(`[TGJU] ${zeroPriceItems.length} static instruments have price=0, fetching from Yahoo Finance as fallback...`);
   707→
   708→  // Collect unique Yahoo symbols
   709→  const keyToSymbol = new Map<string, string>();
   710→  const symbolToKeys = new Map<string, string[]>();
   711→  for (const item of zeroPriceItems) {
   712→    const yahooSym = TGJU_TO_YAHOO_MAP[item.key];
   713→    if (!yahooSym) continue;
   714→    keyToSymbol.set(item.key, yahooSym);
   715→    if (!symbolToKeys.has(yahooSym)) symbolToKeys.set(yahooSym, []);
   716→    symbolToKeys.get(yahooSym)!.push(item.key);
   717→  }
   718→
   719→  const uniqueSymbols = [...symbolToKeys.keys()];
   720→
   721→  try {
   722→    // Dynamic import to avoid loading yahoo-finance2 for TSE-only users
   723→    const YahooFinance = (await import('yahoo-finance2')).default;
   724→    const yahooFinance = new YahooFinance({ suppressNotices: ['yahooSurvey'], validation: { logErrors: false } });
   725→
   726→    // Fetch in batches of 15 (smaller batches reduce Yahoo validation errors)
   727→    const BATCH_SIZE = 15;
   728→    for (let i = 0; i < uniqueSymbols.length; i += BATCH_SIZE) {
   729→      const batch = uniqueSymbols.slice(i, i + BATCH_SIZE);
   730→      try {
   731→        const results = await yahooFinance.quote(batch, {
   732→          fields: ['symbol', 'regularMarketPrice', 'regularMarketChange', 'regularMarketChangePercent', 'regularMarketPreviousClose', 'regularMarketDayHigh', 'regularMarketDayLow'],
   733→        });
   734→
   735→        for (const r of results) {
   736→          if (r.regularMarketPrice == null) continue;
   737→          const tgjuKeys = symbolToKeys.get(r.symbol);
   738→          if (!tgjuKeys) continue;
   739→
   740→          for (const tgjuKey of tgjuKeys) {
   741→            const inst = liveInstrumentsCache.data?.find((x) => x.key === tgjuKey);
   742→            if (inst && inst.price === 0) {
   743→              inst.price = r.regularMarketPrice;
   744→              inst.highPrice = r.regularMarketDayHigh ?? r.regularMarketPrice;
   745→              inst.lowPrice = r.regularMarketDayLow ?? r.regularMarketPrice;
   746→              inst.change = r.regularMarketChange ?? 0;
   747→              inst.changePercent = r.regularMarketChangePercent ?? 0;
   748→
   749→              // Also update static price cache
   750→              staticPriceCache.set(tgjuKey, {
   751→                close: r.regularMarketPrice,
   752→                high: r.regularMarketDayHigh ?? r.regularMarketPrice,
   753→                low: r.regularMarketDayLow ?? r.regularMarketPrice,

   812→// Track which keys were last fetched from Yahoo (for route to detect source)

   814→
   815→let pageReaderQueue: Promise<void> = Promise.resolve();
   816→let lastPageReaderCall = 0;
   817→const PAGE_READER_MIN_INTERVAL = 10_000; // 10 seconds between calls
   818→const PAGE_READER_MAX_RETRIES = 3;
   819→const PAGE_READER_RETRY_BASE_DELAY = 3_000; // 3 seconds base retry delay

   821→/**
   822→ * Fetch historical OHLC data for a TGJU instrument.
   823→ *
   824→ * Strategy (permanent fix):
   825→ * 1. Check in-memory cache (4h TTL)
   826→ * 2. For global instruments with Yahoo mapping → use Yahoo Finance as PRIMARY
   827→ * 3. For Iranian instruments → try direct fetch, then queued page_reader with retry
   828→ * 4. If all fail → return stale cache (up to 24h) or file-based cache
   829→ */
   830→export async function fetchTgjuHistory(tgjuKey: string): Promise<TgjuOHLC[]> {
   831→  const now = Date.now();
   832→
   833→  // ── Step 0: Check in-memory cache ──
   834→  const cached = historyCache.get(tgjuKey);
   835→  if (cached && now - cached.ts < HISTORY_TTL) {
   836→    return cached.data;
   837→  }
   838→
   839→  // ── Step 1: For global instruments, use Yahoo Finance as PRIMARY source ──
   840→  const yahooSymbol = TGJU_TO_YAHOO_MAP[tgjuKey];
   841→  if (yahooSymbol) {
   842→    try {
   843→      const YahooFinance = (await import('yahoo-finance2')).default;
   844→      const yf = new YahooFinance({ suppressNotices: ['yahooSurvey'], validation: { logErrors: false } });
   845→      const result = await yf.chart(yahooSymbol, {
   846→        period1: new Date(Date.now() - 400 * 24 * 60 * 60 * 1000),
   847→        period2: new Date(),
   848→        interval: '1d',
   849→      });
   850→      if (result.quotes && result.quotes.length >= 30) {
   851→        const candles: TgjuOHLC[] = result.quotes
   852→          .filter((q: any) => q.close != null && q.open != null)
   853→          .map((q: any) => ({
   854→            date: q.date?.toISOString().split('T')[0] || '',
   855→            open: q.open!,
   856→            high: q.high || q.close!,
   857→            low: q.low || q.close!,
   858→            close: q.close!,
   859→          }));
   860→        if (candles.length >= 30) {
   861→          console.log(`[TGJU] ${tgjuKey}: Got ${candles.length} candles from Yahoo (${yahooSymbol})`);
   862→          yahooFetchedKeys.add(tgjuKey);
   863→          cacheHistory(tgjuKey, candles, now);
   864→          return candles;
   865→        }
   866→      }
   867→    } catch (err) {
   868→      console.warn(`[TGJU] Yahoo historical fetch failed for ${tgjuKey} (${yahooSymbol}):`, err instanceof Error ? err.message : err);
   869→    }
   870→  }
   871→
   872→  // ── Step 2: For Iranian instruments (or Yahoo fallback failed), try TGJU sources ──
   873→  // 2a: Try direct fetch (fast path, works when Cloudflare not blocking)
   874→  const directResult = await tryDirectFetch(tgjuKey);
   875→  if (directResult && directResult.length >= 30) {
   876→    cacheHistory(tgjuKey, directResult, now);
   877→    return directResult;
   878→  }
   879→
   880→  // 2b: Queued page_reader with retry + rate limiting
   881→  console.log(`[TGJU] Direct fetch returned ${directResult?.length ?? 0} candles for ${tgjuKey}, using queued page_reader...`);
   882→  const pageReaderResult = await fetchViaQueuedPageReader(tgjuKey);
   883→  if (pageReaderResult && pageReaderResult.length >= 30) {
   884→    cacheHistory(tgjuKey, pageReaderResult, now);
   885→    return pageReaderResult;
   886→  }
   887→
   888→  // ── Step 3: All methods failed — return stale cache ──
   889→  if (cached && cached.data.length >= 30) {
   890→    const age = ((now - cached.ts) / 60_000).toFixed(0);
   891→    console.warn(`[TGJU] All methods failed for ${tgjuKey}, returning stale cache (${age}min old, ${cached.data.length} candles)`);
   892→    return cached.data;
   893→  }
   894→
   895→  // ── Step 4: Try file-based cache ──
   896→  const fileCached = await readFromFileCache(tgjuKey);
   897→  if (fileCached && fileCached.length >= 30) {
   898→    console.warn(`[TGJU] Returning file-cached data for ${tgjuKey} (${fileCached.length} candles)`);
   899→    historyCache.set(tgjuKey, { data: fileCached, ts: Date.now() - HISTORY_TTL + 60_000 }); // Mark as almost-expired
   900→    return fileCached;
   901→  }
   902→
   903→  console.error(`[TGJU] All methods and caches failed for ${tgjuKey}`);
   904→  return [];
   905→}

   907→/**
   908→ * Try direct HTTP fetch to TGJU chart API.
   909→ * Returns parsed candles or null on failure.
   910→ */
   911→async function tryDirectFetch(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   912→  try {
   913→    const url = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=365`;
   914→    const controller = new AbortController();
   915→    const timer = setTimeout(() => controller.abort(), 12_000);
   916→
   917→    const res = await fetch(url, { signal: controller.signal, headers: { 'Accept': 'application/json' } });
   918→    clearTimeout(timer);
   919→
   920→    if (!res.ok) {
   921→      console.warn(`[TGJU] Direct fetch HTTP ${res.status} for ${tgjuKey}`);
   922→      return null;
   923→    }
   924→
   925→    const jsonStr = await res.text();
   926→    if (!jsonStr.startsWith('{')) {
   927→      console.warn(`[TGJU] Direct fetch non-JSON for ${tgjuKey}`);
   928→      return null;
   929→    }
   930→
   931→    return parseTgjuChartData(tgjuKey, jsonStr);
   932→  } catch (err) {
   933→    console.warn(`[TGJU] Direct fetch error for ${tgjuKey}:`, err instanceof Error ? err.message : err);
   934→    return null;
   935→  }
   936→}

   938→/**
   939→ * Fetch TGJU historical data via queued page_reader with retry and rate limiting.
   940→ * This prevents 429 errors by serializing requests and adding delays between calls.
   941→ */
   942→async function fetchViaQueuedPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   943→  // Enqueue this request — ensures only one page_reader call at a time
   944→  let result: TgjuOHLC[] | null = null;
   945→  const taskPromise = (async () => {
   946→    // Rate limit: wait if last call was too recent
   947→    const timeSinceLastCall = Date.now() - lastPageReaderCall;
   948→    if (timeSinceLastCall < PAGE_READER_MIN_INTERVAL) {
   949→      const waitTime = PAGE_READER_MIN_INTERVAL - timeSinceLastCall;
   950→      console.log(`[TGJU] Rate limiting page_reader: waiting ${waitTime}ms...`);
   951→      await new Promise(r => setTimeout(r, waitTime));
   952→    }
   953→
   954→    // Retry with exponential backoff
   955→    for (let attempt = 1; attempt <= PAGE_READER_MAX_RETRIES; attempt++) {
   956→      try {
   957→        lastPageReaderCall = Date.now();
   958→        const fetchResult = await fetchTgjuHistoryViaPageReader(tgjuKey);
   959→        if (fetchResult && fetchResult.length >= 30) {
   960→          result = fetchResult;
   961→          return; // success
   962→        }
   963→        // Got data but not enough candles — don't retry
   964→        if (fetchResult && fetchResult.length > 0) {
   965→          console.warn(`[TGJU] page_reader returned only ${fetchResult.length} candles for ${tgjuKey}`);
   966→          return;
   967→        }
   968→      } catch (err) {
   969→        const errMsg = err instanceof Error ? err.message : String(err);
   970→        const isRateLimit = errMsg.includes('429') || errMsg.includes('Too many requests');
   971→        if (isRateLimit && attempt < PAGE_READER_MAX_RETRIES) {
   972→          const delay = PAGE_READER_RETRY_BASE_DELAY * Math.pow(3, attempt - 1); // 3s, 9s, 27s
   973→          console.warn(`[TGJU] page_reader 429 for ${tgjuKey}, retry ${attempt}/${PAGE_READER_MAX_RETRIES} in ${delay}ms...`);
   974→          await new Promise(r => setTimeout(r, delay));
   975→          continue;
   976→        }
   977→        console.error(`[TGJU] page_reader error for ${tgjuKey} (attempt ${attempt}):`, errMsg);
   978→        return;
   979→      }
   980→    }
   981→  })();
   982→
   983→  // Chain onto the queue
   984→  pageReaderQueue = pageReaderQueue.then(() => taskPromise, () => taskPromise);
   985→  await pageReaderQueue;
   986→  return result;
   987→}

   989→/**
   990→ * Single page_reader call to fetch TGJU historical data.
   991→ * Uses z-ai SDK to bypass Cloudflare challenges.
   992→ * Throws on failure so the caller (fetchViaQueuedPageReader) can retry.
   993→ */
   994→async function fetchTgjuHistoryViaPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   995→  const apiUrl = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=365`;
   996→
   997→  const ZAI = (await import('z-ai-web-dev-sdk')).default;
   998→  const zai = await ZAI.create();
   999→  console.log(`[TGJU] Using z-ai SDK page_reader for ${tgjuKey}...`);
  1000→  const invokeResult = await zai.functions.invoke('page_reader', { url: apiUrl });
  1001→
  1002→  if (invokeResult.code !== 200 || !invokeResult.data?.html) {
  1003→    const errMsg = invokeResult.error || `code=${invokeResult.code}`;
  1004→    throw new Error(errMsg);
  1005→  }
  1006→
  1007→  // Extract JSON from <pre> tag in the HTML
  1008→  const html = invokeResult.data.html;
  1009→  const preMatch = html.match(/<pre[^>]*>([\s\S]*?)<\/pre>/);
  1010→  if (!preMatch) {
  1011→    throw new Error('no <pre> block found in page_reader response');
  1012→  }
  1013→
  1014→  // Unescape HTML entities and strip HTML tags from values
  1015→  const jsonStr = preMatch[1]
  1016→    .replace(/<[^>]*>/g, '')        // Strip ALL HTML tags first
  1017→    .replace(/&quot;/g, '"')
  1018→    .replace(/&amp;/g, '&')
  1019→    .replace(/&lt;/g, '<')
  1020→    .replace(/&gt;/g, '>')
  1021→    .replace(/\s+/g, ' ')           // Normalize whitespace
  1022→    .trim();
  1023→
  1024→  if (!jsonStr.startsWith('{')) {
  1025→    throw new Error('extracted content is not JSON');
  1026→  }
  1027→
  1028→  return parseTgjuChartData(tgjuKey, jsonStr);
  1029→}

  1031→/**
  1032→ * Parse the JSON data from TGJU chart API response.
  1033→ * Used by both direct fetch and page_reader paths.
  1034→ */
  1035→function parseTgjuChartData(tgjuKey: string, jsonStr: string): TgjuOHLC[] | null {
  1036→  try {
  1037→    const parsed = JSON.parse(jsonStr);
  1038→    const rawRows: (string | number)[][] = parsed.data;
  1039→
  1040→    if (!rawRows || rawRows.length === 0) {
  1041→      console.warn(`[TGJU] Empty data array for ${tgjuKey}`);
  1042→      return null;
  1043→    }
  1044→
  1045→    // Sanitize row values: strip any residual HTML tags and trim
  1046→    const rows = rawRows.map(row =>
  1047→      row.map(cell => {
  1048→        const s = String(cell).trim();
  1049→        // Strip HTML tags if any remain (safety for direct fetch path)
  1050→        return s.replace(/<[^>]*>/g, '').replace(/\s+/g, ' ');
  1051→      })
  1052→    );
  1053→
  1054→    // ── Auto-detect column mapping from first valid row ──
  1055→    let colMap = detectColumnMap(rows[0]);
  1056→    if (!colMap) {
  1057→      for (let r = 1; r < Math.min(5, rows.length); r++) {
  1058→        colMap = detectColumnMap(rows[r]);
  1059→        if (colMap) break;
  1060→      }
  1061→    }
  1062→    if (!colMap) {
  1063→      console.warn(`[TGJU] Could not auto-detect columns for ${tgjuKey}, using default order`);
  1064→      colMap = { openIdx: 0, highIdx: 2, lowIdx: 1, closeIdx: 3, jalaliDateIdx: 7, gregorianDateIdx: 6 };
  1065→    } else {
  1066→      console.log(`[TGJU] Column map for ${tgjuKey}: O=${colMap.openIdx} H=${colMap.highIdx} L=${colMap.lowIdx} C=${colMap.closeIdx} Date=${colMap.jalaliDateIdx}`);
  1067→    }
  1068→
  1069→    const { openIdx, highIdx, lowIdx, closeIdx, jalaliDateIdx, gregorianDateIdx } = colMap;
  1070→
  1071→    // ── Parse rows ──
  1072→    const candles: TgjuOHLC[] = [];
  1073→    let invalidCount = 0;
  1074→
  1075→    for (const row of rows) {
  1076→      const rawDate = String(row[jalaliDateIdx] || row[gregorianDateIdx] || '');
  1077→      const date = rawDate.replace(/[\/]/g, '-');
  1078→      const o = parsePersianNum(row[openIdx]);
  1079→      const h = parsePersianNum(row[highIdx]);
  1080→      const l = parsePersianNum(row[lowIdx]);
  1081→      const c = parsePersianNum(row[closeIdx]);
  1082→
  1083→      if (!isValidCandle(o, h, l, c) || date.length === 0) {
  1084→        invalidCount++;
  1085→        continue;
  1086→      }
  1087→
  1088→      candles.push({ date, open: o, high: h, low: l, close: c });
  1089→    }
  1090→
  1091→    if (invalidCount > 0) {
  1092→      console.warn(`[TGJU] ${tgjuKey}: ${invalidCount}/${rows.length} invalid candles skipped`);
  1093→    }
  1094→
  1095→    if (candles.length === 0) {
  1096→      return null;
  1097→    }
  1098→
  1099→    // ── Detect and fix data drift ──
  1100→    let driftCount = 0;
  1101→    for (let i = 1; i < candles.length; i++) {
  1102→      const prevClose = candles[i - 1].close;
  1103→      const currClose = candles[i].close;
  1104→      if (prevClose > 0) {
  1105→        const pctChange = Math.abs(currClose - prevClose) / prevClose;
  1106→        if (pctChange > 0.5) {
  1107→          driftCount++;
  1108→        }
  1109→      }
  1110→    }
  1111→    if (driftCount > candles.length * 0.1) {
  1112→      console.warn(`[TGJU] ${tgjuKey}: Possible data drift detected (${driftCount} suspicious jumps out of ${candles.length} candles)`);
  1113→    }
  1114→
  1115→    // Reverse: API returns oldest-first (asc), we want newest-last for charting
  1116→    const firstDate = candles[0].date;
  1117→    const lastDate = candles[candles.length - 1].date;
  1118→    if (firstDate > lastDate) {
  1119→      candles.reverse();
  1120→    }
  1121→
  1122→    console.log(`[TGJU] ${tgjuKey}: Parsed ${candles.length} candles (${candles[0]?.date} to ${candles[candles.length - 1]?.date})`);
  1123→
  1124→    return candles;
  1125→  } catch (err) {
  1126→    console.error(`[TGJU] Failed to parse chart data for ${tgjuKey}:`, err);
  1127→    return null;
  1128→  }
  1129→}

  1131→/**
  1132→ * Cache parsed history data and update related caches.
  1133→ */
  1134→function cacheHistory(tgjuKey: string, candles: TgjuOHLC[], now: number): void {
  1135→  historyCache.set(tgjuKey, { data: candles, ts: now });
  1136→  // Also persist to file for server restart resilience
  1137→  writeToFileCache(tgjuKey, candles).catch(() => {});
  1138→
  1139→  // Also update static price cache for this instrument
  1140→  if (candles.length >= 2) {
  1141→    const last = candles[candles.length - 1];
  1142→    const prev = candles[candles.length - 2];
  1143→    staticPriceCache.set(tgjuKey, {
  1144→      close: last.close,
  1145→      high: last.high,
  1146→      low: last.low,
  1147→      prevClose: prev.close,
  1148→      ts: now,
  1149→    });
  1150→
  1151→    // Update live instruments cache if this is a static instrument
  1152→    if (liveInstrumentsCache.data) {
  1153→      const inst = liveInstrumentsCache.data.find((i) => i.key === tgjuKey);
  1154→      if (inst && inst.price === 0) {
  1155→        inst.price = last.close;
  1156→        inst.highPrice = last.high;
  1157→        inst.lowPrice = last.low;
  1158→        inst.change = last.close - prev.close;
  1159→        inst.changePercent = prev.close > 0 ? ((last.close - prev.close) / prev.close) * 100 : 0;
  1160→      }
  1161→    }
  1162→  }
  1163→}
