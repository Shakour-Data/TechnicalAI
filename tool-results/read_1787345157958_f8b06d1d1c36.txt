   585→/** Simple deterministic hash from string to number */
   586→function djb2Hash(str: string): number {
   587→  let hash = 5381;
   588→  for (let i = 0; i < str.length; i++) {
   589→    hash = ((hash << 5) + hash + str.charCodeAt(i)) & 0x7fffffff;
   590→  }
   591→  return hash;
   592→}

   594→/**
   595→ * Select a V9 persona and template based on deterministic hash rotation.
   596→ * Uses hash of (date + instrument) to pick persona matching the school,
   597→ * then applies day_hash % 3/5/7 logic for style and tone variation.
   598→ *
   599→ * @param instrument - Symbol name (e.g. "خودرو")
   600→ * @param date - Date string in YYYY-MM-DD format
   601→ * @param schoolId - The analysis school (e.g. "classical", "oscillator")
   602→ * @returns Selected persona, template, and variation seed
   603→ */
   604→export function selectV9Persona(
   605→  instrument: string,
   606→  date: string,
   607→  schoolId: string,
   608→): { persona: V9Persona; template: V9TemplateFormat; variationSeed: number } {
   609→  const compositeKey = `${date}:${instrument}`;
   610→  const dayHash = djb2Hash(compositeKey);
   611→
   612→  // Filter personas matching the school affinity (fall back to all if none match)
   613→  const matching = PERSONAS.filter(p => p.schoolAffinity === schoolId);
   614→  const pool = matching.length > 0 ? matching : PERSONAS;
   615→
   616→  // Select persona deterministically from pool
   617→  const personaIndex = dayHash % pool.length;
   618→  let persona = pool[personaIndex];
   619→
   620→  // Apply day_hash % 3/5/7 for subtle variation
   621→  const styleVariation = dayHash % 3;   // 0=normal, 1=more formal, 2=more casual
   622→  const toneVariation = dayHash % 5;    // 0-4 tone modifiers
   623→  const phraseOffset = dayHash % 7;     // phrase bank rotation offset
   624→
   625→  // Adjust persona based on variation (rotate within same school group)
   626→  if (styleVariation === 1 && pool.length > 1) {
   627→    persona = pool[(personaIndex + 1) % pool.length];
   628→  } else if (styleVariation === 2 && pool.length > 2) {
   629→    persona = pool[(personaIndex + 2) % pool.length];
   630→  }
   631→
   632→  // Select template based on tone variation
   633→  const templateIndex = toneVariation % TEMPLATE_FORMATS.length;
   634→  const template = TEMPLATE_FORMATS[templateIndex];
   635→
   636→  // Variation seed for phrase selection
   637→  const variationSeed = dayHash % 1000;
   638→
   639→  return { persona, template, variationSeed };
   640→}

   646→/**
   647→ * Select phrases from the phrase bank, rotated by variationSeed.
   648→ * Returns 2-4 phrases per requested section.
   649→ */
   650→function selectPhrases(sections: string[], variationSeed: number): Record<string, string[]> {
   651→  const result: Record<string, string[]> = {};
   652→  for (const section of sections) {
   653→    const bank = (PHRASE_BANK as Record<string, readonly string[]>)[section];
   654→    if (!bank) continue;
   655→    const offset = variationSeed % bank.length;
   656→    const count = Math.min(2 + (variationSeed % 3), bank.length);
   657→    const selected: string[] = [];
   658→    for (let i = 0; i < count; i++) {
   659→      selected.push(bank[(offset + i) % bank.length]);
   660→    }
   661→    result[section] = selected;
   662→  }
   663→  return result;
   664→}

   853→type ScoreFn<T> = (item: T, input: NarrativeInput) => number;
   854→interface ScoredCandidate<T> { item: T; score: number; }

   856→function scoreAll<T>(items: T[], input: NarrativeInput, scoreFn: ScoreFn<T>): ScoredCandidate<T>[] {
   857→  return items.map(item => ({ item, score: scoreFn(item, input) })).sort((a, b) => b.score - a.score);
   858→}

   860→function scoreSchool(item: SchoolDef, input: NarrativeInput): number {
   861→  let s = 0;
   862→  const id = item.id;
   863→  const { adx, rsi, stochK, macdHist, obv, hasVolume, cci, bbPosition, trendDirection, nearResistance, nearSupport, trendR2, diPlus, diMinus } = input;
   864→  if (id === 'trend') {
   865→    if (adx > 25 && trendR2 > 0.7) s += (trendDirection === 'up' ? 8 : 6);
   866→    if (adx > 35) s += 3;
   867→    if (diPlus > diMinus && macdHist > 0) s += 2;
   868→    if (trendDirection !== 'neutral') s += 2;
   869→  }
   870→  if (id === 'classical') {
   871→    if (nearResistance || nearSupport) s += 4;
   872→    if ((rsi > 60 && trendDirection === 'down') || (rsi < 40 && trendDirection === 'up')) s += 5;
   873→    if (cci > 100 || cci < -100) s += 3;
   874→    if (bbPosition > 85 || bbPosition < 15) s += 2;
   875→  }
   876→  if (id === 'candlestick') {
   877→    if (nearResistance && stochK > 75) s += 5;
   878→    if (nearSupport && stochK < 25) s += 5;
   879→    if (nearResistance || nearSupport) s += 2;
   880→    if (rsi > 70 || rsi < 30) s += 2;
   881→  }
   882→  if (id === 'fibonacci') {
   883→    if (input.priceVsMa21 !== input.priceVsMa100) s += 4;
   884→    if (bbPosition > 80 || bbPosition < 20) s += 3;
   885→    if (nearResistance || nearSupport) s += 2;
   886→  }
   887→  if (id === 'volatility') {
   888→    if (bbPosition > 20 && bbPosition < 80 && adx < 20) s += 6;
   889→    if (bbPosition > 90 || bbPosition < 10) s += 4;
   890→    if (hasVolume) s += 2;

   973→function scoreTone(item: ToneDef, input: NarrativeInput): number {
   974→  let s = 0;
   975→  const id = item.id;
   976→  const { adx, rsi, stochK, macdHist, obv, hasVolume, bbPosition, trendDirection, nearResistance, nearSupport, dominantScenarioKey, trendR2 } = input;
   977→  if (id === 'formal') { if (adx > 25) s += 3; if (trendR2 > 0.7) s += 2; }
   978→  if (id === 'fast') { if (adx > 30 && macdHist > 0 && trendDirection === 'up') s += 5; if (stochK > 60) s += 2; }
   979→  if (id === 'philosophical') { if (adx > 15 && adx < 35) s += 4; if (rsi > 40 && rsi < 60) s += 2; }
   980→  if (id === 'warning') {
   981→    if ((rsi > 70 || stochK > 80) && nearResistance) s += 7;
   982→    if (macdHist > 0 && rsi < 50) s += 3;
   983→    if ((rsi < 30 || stochK < 20) && nearSupport && trendDirection === 'down') s += 4;
   984→  }
   985→  if (id === 'storytelling') { if (adx < 25) s += 4; if (bbPosition > 30 && bbPosition < 70) s += 2; }
   986→  if (id === 'stepByStep') { if (nearResistance || nearSupport) s += 4; if (adx > 25) s += 2; }
   987→  if (id === 'skeptical') {
   988→    if ((macdHist > 0 && rsi < 50) || (macdHist < 0 && rsi > 50)) s += 6;
   989→    if (rsi > 60 && trendDirection === 'down') s += 3;
   990→    if (rsi < 40 && trendDirection === 'up') s += 3;
   991→  }
   992→  if (id === 'optimistic') { if (trendDirection === 'up' && adx > 25 && rsi > 50 && rsi < 70) s += 6; if (macdHist > 0 && obv > 0) s += 2; }
   993→  if (id === 'simple') { if (adx > 20 && adx < 40 && rsi > 40 && rsi < 60) s += 4; if (!nearResistance && !nearSupport) s += 2; }
   994→  if (id === 'quantitative') { if (hasVolume && adx > 25) s += 3; if (rsi > 70 || rsi < 30) s += 2; }
   995→  if (id === 'mysterious') { if (adx < 20) s += 4; if ((macdHist > 0 && rsi < 50) || (macdHist < 0 && rsi > 50)) s += 3; if (dominantScenarioKey === 'R3') s += 2; }
   996→  if (id === 'educational') { if (nearResistance || nearSupport) s += 3; if (rsi > 65 || rsi < 35) s += 2; }
   997→  if (id === 'comparative') {
   998→    const probs = Object.values(input.scenarioProbabilities);
   999→    const maxProb = Math.max(...probs);
  1000→    if (maxProb < 35) s += 5;
  1001→    if (probs.filter(p => p > 20).length >= 3) s += 3;
  1002→  }
  1003→  if (id === 'balanced') { if (adx < 20) s += 5; if (rsi > 40 && rsi < 60) s += 3; }
  1004→  if (id === 'conclusive') { if (adx > 25 && (rsi > 60 || rsi < 40)) s += 4; if (dominantScenarioKey === 'R1' || dominantScenarioKey === 'R2') s += 2; }
  1005→  return s + 1;
  1006→}

  1031→export function buildNarrativeInput(body: {

  1033→  trendDirection: string;
  1034→  rsi: number;
  1035→  adx: number;
  1036→  stochK: number;
  1037→  stochD: number;
  1038→  macdLine: number;
  1039→  macdSignal: number;
  1040→  macdHist: number;
  1041→  diPlus: number;
  1042→  diMinus: number;
  1043→  obv: number;
  1044→  hasVolume: boolean;
  1045→  cci: number;
  1046→  mfi: number;
  1047→  atr: number;
  1048→  bollingerUpper: number;
  1049→  bollingerMiddle: number;
  1050→  bollingerLower: number;
  1051→  trendAngle: number;
  1052→  trendR2: number;
  1053→  overallSignal: string;
  1054→  scenarios: Record<string, { probability: number }>;
  1055→  resistances: number[];
  1056→  supports: number[];
  1057→  ma21: number;
  1058→  ma100: number;
  1059→}): NarrativeInput {
  1060→  const bbRange = body.bollingerUpper - body.bollingerLower;
  1061→  const bbPosition = bbRange > 0
  1062→    ? Math.max(0, Math.min(100, ((body.currentPrice - body.bollingerLower) / bbRange) * 100))
  1063→    : 50;
  1064→  const R1 = body.resistances[0] || body.currentPrice * 1.05;
  1065→  const S1 = body.supports[0] || body.currentPrice * 0.95;
  1066→  let dominantKey = 'R3';
  1067→  let dominantProb = 0;
  1068→  for (const k of ['R1', 'R2', 'R3', 'R4', 'R5'] as const) {
  1069→    const p = body.scenarios[k]?.probability ?? 0;
  1070→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
  1071→  }
  1072→  return {
