   179→function generateAnalysisText(ctx: AnalysisContext) {
   180→  const {
   181→    symbolName, currentPrice, ma21, ma100, rsi, mfi, cci, adx,
   182→    stochK, stochD, macdLine, macdSignal, macdHist,
   183→    diPlus, diMinus, sar, atr, obv,
   184→    bollingerUpper, bollingerMiddle, bollingerLower,
   185→    trendDirection, trendAngle, trendR2, overallSignal,
   186→    highestKey, highestProb, scenarios, S1, R1, R2,
   187→    hasVolume, resistanceStrengths, supportStrengths,
   188→  } = ctx;
   189→
   190→  const R1_info = resistanceStrengths[0];
   191→  const S1_info = supportStrengths[0];
   192→  const R1_grade = R1_info?.grade ? (GRADE_MAP[R1_info.grade]?.label ?? R1_info.grade) : '';
   193→  const S1_grade = S1_info?.grade ? (GRADE_MAP[S1_info.grade]?.label ?? S1_info.grade) : '';
   194→  const R1_methods = R1_info?.methods?.length ? ` با ${toPersianDigits(String(R1_info.methods.length))} روش تأیید شده` : '';
   195→  const S1_methods = S1_info?.methods?.length ? ` با ${toPersianDigits(String(S1_info.methods.length))} روش تأیید شده` : '';
   196→
   197→  const r1r2 = scenarios.R1.probability + scenarios.R2.probability;
   198→  const r4r5 = scenarios.R4.probability + scenarios.R5.probability;
   199→  const r3 = scenarios.R3.probability;
   200→  const dominant = SCENARIO_META[highestKey].label;
   201→
   202→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   203→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   204→  const macdBullish = macdLine > macdSignal;
   205→  const bbRange = bollingerUpper - bollingerLower;

   208→    : currentPrice < bollingerLower ? 'زیر باند پایینی (اشباع فروش)'
   209→    : `داخل باندها (${bbPos}٪ از بازه)`;
   210→  const diSignal = diPlus > diMinus
   211→    ? `DI+ (${toFa(diPlus)}) بالاتر از DI- (${toFa(diMinus)}) — فشار خرید غالب`
   212→    : `DI- (${toFa(diMinus)}) بالاتر از DI+ (${toFa(diPlus)}) — فشار فروش غالب`;

   217→    : 'خنثی و بدون جهت مشخص';
