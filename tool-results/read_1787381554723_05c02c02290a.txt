   129→  } = props;
   130→
   131→  const unit = currencyUnit || 'ریال';
   132→
   133→  // ── Instrument Context (for intelligent naming) ─────────────────

   136→    if (isTgju) return { name: symbolName, type: 'tgju' as const, verb: 'قرار دارد' };

   142→  const containerRef = useRef<HTMLDivElement>(null);

   148→  const [aiLoading, setAiLoading] = useState(false);
   149→  const [aiError, setAiError] = useState<string | null>(null);

   221→          const errData = await res.json();
   222→          errorMsg = errData.error || errorMsg;

   227→        throw new Error(errorMsg);

   234→      }

   237→        if (err instanceof Error && err.name === 'AbortError') {

   277→  const sortedScenarios = useMemo(() => {

   279→      .map(k => ({ key: k, ...scenarios[k] }))

   838→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   839→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   840→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   841→  grade?: string; overlapCount?: number; touchCount?: number;
   842→}) {
   843→  const isSupport = type === 'support';
   844→  const isTargetType = type === 'target';
   845→
   846→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   847→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   848→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   849→
   850→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   851→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   852→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   853→
   854→  return (
   855→    <div
   856→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   857→      style={{
   858→        background: bgAccent,
   859→        border: `1px solid ${borderAccent}`,
   860→        borderRight: `3px solid ${accentColor}`,
   861→      }}
   862→    >
   863→      {/* Label + Price */}
   864→      <div className="flex-1 min-w-0">
   865→        <div className="flex items-center gap-2">
   866→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   867→          {isTarget && <span className="text-[9px]">🎯</span>}
   868→          {overlapCount !== undefined && overlapCount > 0 && (
   869→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   870→              {toPersianDigits(overlapCount)} هم‌پوشانی
   871→            </span>
   872→          )}
   873→        </div>
   874→        <div className="flex items-baseline gap-1">
   875→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   876→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   877→        </div>
   878→      </div>
   879→
   880→      {/* Strength + Grade */}
   881→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   882→        <div className="flex items-center gap-1.5">
   883→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   884→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   885→          <span className="text-[9px] text-gray-500">/۱۰</span>
   886→        </div>
   887→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   888→          <div
   889→            className="h-full rounded-full transition-all duration-500"
   890→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   891→          />
   892→        </div>
   893→        {touchCount !== undefined && touchCount > 0 && (
   894→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   895→        )}
   896→      </div>
   897→    </div>
   898→  );
   899→}
