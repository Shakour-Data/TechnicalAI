   172→// ------------------------------------------------------------------
   173→// 2. موتور محاسباتی (دقیقاً طبق فرمول‌های Master Prompt)
   174→// ------------------------------------------------------------------
   175→function calculateProbabilities(data) {
   176→  const {price, levels, indicators} = data;
   177→  const {R1, S1, MA100, R2, S2, S3, S4} = levels;
   178→  const {RSI, MFI, CCI, ADX} = indicators;
   179→
   180→  const adxScore = ADX / 100;
   181→  const rsiBull = RSI > 70 ? 1 : RSI > 50 ? 0.5 : 0;
   182→  const mfiBull = MFI > 80 ? 1 : MFI > 50 ? 0.5 : 0;
   183→  const cciBull = CCI > 100 ? 1 : CCI > 0 ? 0.5 : 0;
   184→  const consensus = (rsiBull + mfiBull + cciBull) / 3;
   185→  const trendBull = Math.min(1, 0.4 * adxScore + 0.6 * consensus);
   186→
   187→  const dR1 = Math.exp(-5 * Math.abs(price - R1) / R1);
   188→
   189→  const raw_R1 = (0.45 * dR1 + 0.55 * trendBull);
   190→  const raw_R2 = (1 - raw_R1) * 0.6 + 0.4 * (price > S1 ? 1 : 0) * (RSI < 70 || MFI < 80 ? 0.5 : 1);
   191→  
   192→  const dFromR1S1 = (Math.abs(price - R1)/R1 + Math.abs(price - S1)/S1);
   193→  const corrFact = 1 / (1 + dFromR1S1 * 10);
   194→  const raw_R3 = 0.3 * (price > MA100 ? 1 : 0) * corrFact * trendBull;
   195→  
   196→  const raw_R4 = 0.05 * (price > MA100 ? 0.2 : 1);
   197→  const raw_R5 = 0.02 * (price > MA100 ? 0 : 1) + 0.01 * (RSI < 30 || MFI < 30 ? 1 : 0);
   198→
   199→  const rawValues = {R1: raw_R1, R2: raw_R2, R3: raw_R3, R4: raw_R4, R5: raw_R5};
   200→  const sum = Object.values(rawValues).reduce((a,b) => a + b, 0);
   201→  const norm = (k) => Math.round((rawValues[k] / sum) * 100);
   202→
   203→  return { R1: norm('R1'), R2: norm('R2'), R3: norm('R3'), R4: norm('R4'), R5: norm('R5') };
   204→}

   293→function center(id){
   294→  const e = document.getElementById(id);
   295→  return { x:e.offsetLeft + e.offsetWidth/2, y:e.offsetTop + e.offsetHeight/2 };
   296→}

   298→function drawEdges(){
   299→  svg.innerHTML = `<defs>${Object.entries(colors).map(([k,c]) => `<marker id="arrow-${k}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"></path></marker>`).join("")}</defs>`;
   300→  edgesList.forEach((edge)=>{
   301→    const [from,to,label,type] = edge;
   302→    const a=center(from), b=center(to);
   303→    const dx=b.x-a.x, dy=b.y-a.y;
   304→    const dist=Math.sqrt(dx*dx+dy*dy);
   305→    const bend=Math.min(52, Math.max(18, dist*.11));
   306→    const mx=(a.x+b.x)/2, my=(a.y+b.y)/2;
   307→    const nx=-dy/dist, ny=dx/dist;
   308→    const cx=mx+nx*bend, cy=my+ny*bend;
   309→    const d=`M ${a.x} ${a.y} Q ${cx} ${cy} ${b.x} ${b.y}`;
   310→    const path=document.createElementNS("http://www.w3.org/2000/svg","path");
   311→    path.setAttribute("d",d); path.setAttribute("stroke",colors[type]); path.setAttribute("marker-end",`url(#arrow-${type})`);
   312→    path.setAttribute("class",`edge ${type}`); path.dataset.type=type; path.dataset.from=from; path.dataset.to=to;
   313→    svg.appendChild(path);
   314→    const text=document.createElementNS("http://www.w3.org/2000/svg","text");
   315→    text.setAttribute("x",cx); text.setAttribute("y",cy-5); text.setAttribute("class","edge-label"); text.dataset.type=type; text.textContent=label;
   316→    svg.appendChild(text);
   317→  });
   318→}
