    39→function renderAIText(text: string): string {
    40→  // Normalize zero-width and look-alike characters that may interfere with regex
    41→  const normalized = text.replace(/[\u200c\u200d\u200b\ufeff]/g, '');
    42→
    43→  // Color syntax: {color:hex}text{/color} or {color:named}text{/color} → <span>
    44→  const colorMap: Record<string, string> = {
    45→    'red': '#dc2626', 'red-600': '#dc2626', 'red-700': '#b91c1c',
    46→    'green': '#16a34a', 'emerald': '#059669', 'emerald-600': '#059669', 'emerald-700': '#047857',
    47→    'amber': '#d97706', 'amber-600': '#d97706', 'amber-700': '#b45309', 'amber-800': '#92400e',
    48→    'blue': '#2563eb', 'blue-600': '#2563eb', 'blue-700': '#1d4ed8',
    49→    'orange': '#ea580c',
    50→    'purple': '#9333ea', 'purple-600': '#9333ea', 'purple-700': '#7e22ce',
    51→  };
    52→
    53→  const withColors = normalized.replace(/\{color:([^}]+)\}([\s\S]*?)\{\/color\}/g, (_match, colorName: string, inner: string) => {
    54→    const hex = colorMap[colorName.trim()] || (colorName.startsWith('#') ? colorName : null);
    55→    if (!hex) return inner; // Unknown color: just return text without color
    56→    return `<span style="color:${hex}">${inner}</span>`;
    57→  });
    58→
    59→  return withColors
    60→    .split('\n\n')
    61→    .map(p => {
    62→      // Convert **bold** to <strong>
    63→      const html = p.replace(/\*\*(.+?)\*\*/g, '<strong class="font-bold text-gray-900">$1</strong>');
    64→      // Convert newlines within paragraph to <br>
    65→      return `<p class="mb-4">${html.replace(/\n/g, '<br>')}</p>`;
    66→    })
    67→    .join('');
    68→}
