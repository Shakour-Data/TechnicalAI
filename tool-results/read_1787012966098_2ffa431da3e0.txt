   803→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   804→function LevelBox({ index, label, price, strength, isTarget, type, unit }: {
   805→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   806→}) {
   807→  const isSupport = type === 'support';
   808→  const isTargetType = type === 'target';
   809→
   810→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   811→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   812→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   813→
   814→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   815→  const strengthLabel = strength >= 8 ? 'بسیار قوی' : strength >= 6 ? 'قوی' : strength >= 4 ? 'متوسط' : strength >= 2 ? 'ضعیف' : 'بسیار ضعیف';
   816→  const strengthColor = strength >= 8 ? '#34c98b' : strength >= 6 ? '#3ad5db' : strength >= 4 ? '#ffb11b' : strength >= 2 ? '#ff7b32' : '#ef4d62';
   817→
   818→  // Distance from current price (if available from context)
   819→  return (
   820→    <div
   821→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   822→      style={{
   823→        background: bgAccent,
   824→        border: `1px solid ${borderAccent}`,
   825→        borderRight: `3px solid ${accentColor}`,
   826→      }}
   827→    >
   828→      {/* Label + Price */}
   829→      <div className="flex-1 min-w-0">
   830→        <div className="flex items-center gap-2">
   831→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   832→          {isTarget && <span className="text-[9px]">🎯</span>}
   833→        </div>
   834→        <span className="text-sm font-black text-gray-100 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   835→        <span className="text-[10px] text-gray-500 mr-1">{unit || 'ریال'}</span>
   836→      </div>
   837→
   838→      {/* Strength bar */}
   839→      <div className="flex flex-col items-end gap-1 min-w-[80px]">
   840→        <div className="flex items-center gap-1.5">
   841→          <span className="text-[9px] font-medium" style={{ color: strengthColor }}>{strengthLabel}</span>
   842→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(strength)}</span>
   843→          <span className="text-[9px] text-gray-500">/۱۰</span>
   844→        </div>
   845→        <div className="w-full h-1.5 rounded-full bg-white/5 overflow-hidden">
   846→          <div
   847→            className="h-full rounded-full transition-all duration-500"
   848→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   849→          />
   850→        </div>
   851→      </div>
   852→    </div>
   853→  );
   854→}
