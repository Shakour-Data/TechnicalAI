   365→
   366→  // Reversal: Event C → R1-R9 (indices 24-32)
   367→  for (let i = 0; i < 9; i++) {
   368→    probs[String(24 + i)] = reversalCond[i];
   369→  }
   370→
   371→  return probs;
   372→}

   374→// Find all 27 paths from ROOT to terminal nodes
   375→function findAllPaths(edgeProbs: Record<string, number>): PathInfo[] {
   376→  const paths: PathInfo[] = [];
   377→  const branches = ['BR1', 'BR2', 'BR3'];
   378→  const events = ['EA', 'EB', 'EC'];
   379→  const branchNames = ['trend', 'breakout', 'reversal'];
   380→
   381→  for (let b = 0; b < 3; b++) {
   382→    const branchEdgeIdx = b;       // 0, 1, 2
   383→    const eventEdgeIdx = 3 + b;    // 3, 4, 5
   384→    const baseEdgeIdx = 6 + b * 9; // 6, 15, 24
   385→
   386→    const branchProb = edgeProbs[String(branchEdgeIdx)] ?? 0.2;
   387→    const eventProb = edgeProbs[String(eventEdgeIdx)] ?? 1;
   388→
   389→    for (let s = 0; s < 9; s++) {
   390→      const scenarioEdgeIdx = baseEdgeIdx + s;
   391→      const scenarioProb = edgeProbs[String(scenarioEdgeIdx)] ?? 0.1;
   392→      const scenarioKey = SCENARIO_KEYS[s];
   393→
   394→      paths.push({
   395→        nodes: ['ROOT', branches[b], events[b], scenarioKey],
   396→        edges: [branchEdgeIdx, eventEdgeIdx, scenarioEdgeIdx],
   397→        prob: branchProb * eventProb * scenarioProb,
   398→        target: scenarioKey,
   399→        branch: branchNames[b],
   400→      });
   401→    }
   402→  }
   403→
   404→  return paths.sort((a, b) => b.prob - a.prob);
   405→}
