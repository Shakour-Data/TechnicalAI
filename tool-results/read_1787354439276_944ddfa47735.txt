   802→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   803→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   804→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   805→  grade?: string; overlapCount?: number; touchCount?: number;
   806→}) {
   807→  const isSupport = type === 'support';
   808→  const isTargetType = type === 'target';
   809→
   810→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   811→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   812→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   813→
   814→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   815→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   816→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   817→
   818→  return (
   819→    <div
   820→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   821→      style={{
   822→        background: bgAccent,
   823→        border: `1px solid ${borderAccent}`,
   824→        borderRight: `3px solid ${accentColor}`,
   825→      }}
   826→    >
   827→      {/* Label + Price */}
   828→      <div className="flex-1 min-w-0">
   829→        <div className="flex items-center gap-2">
   830→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   831→          {isTarget && <span className="text-[9px]">🎯</span>}
   832→          {overlapCount !== undefined && overlapCount > 0 && (
   833→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   834→              {toPersianDigits(overlapCount)} هم‌پوشانی
   835→            </span>
   836→          )}
   837→        </div>
   838→        <div className="flex items-baseline gap-1">
   839→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   840→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   841→        </div>
   842→      </div>
   843→
   844→      {/* Strength + Grade */}
   845→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   846→        <div className="flex items-center gap-1.5">
   847→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   848→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   849→          <span className="text-[9px] text-gray-500">/۱۰</span>
   850→        </div>
   851→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   852→          <div
   853→            className="h-full rounded-full transition-all duration-500"
   854→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   855→          />
   856→        </div>
   857→        {touchCount !== undefined && touchCount > 0 && (
   858→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   859→        )}
   860→      </div>
   861→    </div>
   862→  );
   863→}
