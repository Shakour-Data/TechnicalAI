   133→// ═══════════════════════════════════════════════════════════════════════════════
   134→
   135→export default function VdssGraph(props: VdssGraphProps) {
   136→  const { colors: C } = useTheme();
   137→  const {
   138→    symbolName,
   139→    currentPrice,
   140→    scenarios,
   141→    decisionGraph,
   142→  } = props;
   143→
   144→  const [activeFilter, setActiveFilter] = useState('all');
   145→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   146→  const graphRef = useRef<HTMLDivElement>(null);
   147→  const svgRef = useRef<SVGSVGElement>(null);
   148→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   149→  const shellRef = useRef<HTMLDivElement>(null);
   150→
   151→  // ── Derived data from decisionGraph ──────────────────────────────
   152→  const edges = decisionGraph?.edges ?? [];
   153→  const nodes = decisionGraph?.nodes ?? [];
   154→  const nodePositions = decisionGraph?.nodePositions ?? {};
   155→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   156→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   157→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   158→  const pathContributions = decisionGraph?.pathContributions ?? {};
   159→  const nodeValues = decisionGraph?.nodeValues ?? {};
   160→
   161→  // Build node lookup map
   162→  const nodeMap = useMemo(() => {
   163→    const map: Record<string, (typeof nodes)[number]> = {};
   164→    for (const n of nodes) map[n.id] = n;
   165→    return map;
   166→  }, [nodes]);
   167→
   168→  // ── Visible edges/nodes based on filter ──────────────────────────
   169→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   170→    if (activeFilter === 'all') {
   171→      const allEdgeIdx = edges.map((_, i) => i);
   172→      const allNodes = new Set(nodes.map(n => n.id));
   173→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   174→    }
   175→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   176→      const edgeSet = new Set<number>();
   177→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   178→      edges.forEach((e, i) => {
   179→        if (e.to === activeFilter) {
   180→          edgeSet.add(i);
   181→          nodeSet.add(e.from);
   182→        }
   183→      });
   184→      edges.forEach((e, i) => {
   185→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   186→          edgeSet.add(i);
   187→          nodeSet.add(e.from);
   188→        }
   189→      });
   190→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   191→    }
   192→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   193→      const branchType = `branch-${activeFilter}` as const;
   194→      const edgeSet = new Set<number>();
   195→      const nodeSet = new Set<string>(['ROOT']);
   196→      edges.forEach((e, i) => {
   197→        if (e.type === branchType) {
   198→          edgeSet.add(i);
   199→          nodeSet.add(e.from);
   200→          nodeSet.add(e.to);
   201→        }
   202→      });
   203→      const expanded = new Set<number>();
   204→      const expandedNodes = new Set<string>(nodeSet);
   205→      edges.forEach((e, i) => {
   206→        if (expandedNodes.has(e.from)) {
   207→          expanded.add(i);
   208→          expandedNodes.add(e.to);
   209→        }
   210→      });
   211→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   212→    }
   213→    // Type filter (up/pullback/down/risk)
   214→    const edgeSet = new Set<number>();
   215→    const nodeSet = new Set<string>(['ROOT']);
   216→    edges.forEach((e, i) => {
   217→      if (e.type === activeFilter) {
   218→        edgeSet.add(i);
   219→        nodeSet.add(e.from);
   220→        nodeSet.add(e.to);
   221→      }
   222→    });
   223→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   224→  }, [activeFilter, edges, nodes]);
   225→
   226→  // ── Draw SVG edges ───────────────────────────────────────────────
   227→  const drawEdges = useCallback(() => {
   228→    const svg = svgRef.current;
   229→    const graph = graphRef.current;
   230→    if (!svg || !graph || edges.length === 0) return;
   231→
   232→    let markersSvg = '';
   233→    const seenColors = new Set<string>();
   234→    for (const e of edges) {
   235→      const c = EDGE_COLORS[e.type];
   236→      if (c && !seenColors.has(e.type)) {
   237→        seenColors.add(e.type);
   238→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   239→      }
   240→    }
   241→
   242→    let pathsSvg = '';
   243→    const graphRect = graph.getBoundingClientRect();
   244→
   245→    for (let ei = 0; ei < edges.length; ei++) {
   246→      const { from: fromId, to: toId, label, type } = edges[ei];
   247→      const isVisible = visibleEdgeIndices.includes(ei);
   248→      const fromEl = nodeRefs.current[fromId];
   249→      const toEl = nodeRefs.current[toId];
   250→      if (!fromEl || !toEl) continue;
   251→
   252→      const fRect = fromEl.getBoundingClientRect();
   253→      const tRect = toEl.getBoundingClientRect();
   254→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   255→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   256→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   257→      const by = tRect.top - graphRect.top + tRect.height / 2;
   258→      const dx = bx - ax;
   259→      const dy = by - ay;
   260→      const dist = Math.sqrt(dx * dx + dy * dy);
   261→
   262→      if (dist === 0) continue;
   263→
   264→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   265→      const mx = (ax + bx) / 2;
   266→      const my = (ay + by) / 2;
   267→      const nx = -dy / dist;
   268→      const ny = dx / dist;
   269→      const cx = mx + nx * bend;
   270→      const cy = my + ny * bend;
   271→
   272→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   273→      const prob = edgeProbabilities[ei] ?? 0;
   274→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   275→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   276→
   277→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.72 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   278→
   279→      if (isVisible && ei >= 6) {
   280→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#ffffff" font-size="10" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="4" stroke-linejoin="round" opacity="0.95" data-type="${type}" class="edge-label">${label}</text>`;
   281→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   282→      }
   283→    }
   284→
   285→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   286→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   287→
   288→  useEffect(() => { drawEdges(); }, [drawEdges]);
   289→
   290→  useEffect(() => {
   291→    const el = shellRef.current;
   292→    if (!el) return;
   293→    const observer = new ResizeObserver(() => drawEdges());
   294→    observer.observe(el);
   295→    return () => observer.disconnect();
   296→  }, [drawEdges]);
   297→
   298→
   299→  // ── Detail panel content ─────────────────────────────────────────
   300→  const detailContent = useMemo(() => {
   301→    if (!selectedNode) return null;
   302→    const node = nodeMap[selectedNode];
   303→    if (!node) return null;
   304→
   305→    if (node.isTerminal && SCENARIO_KEYS.includes(selectedNode as typeof SCENARIO_KEYS[number])) {
   306→      const contrib = pathContributions[selectedNode] ?? { trend: 0, breakout: 0, reversal: 0 };
   307→      const meta = SCENARIO_META_LOCAL[selectedNode];
   308→      const s = scenarios[selectedNode as keyof typeof scenarios];
   309→      return (
   310→        <div style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}>
   311→          <h3 style={{ fontSize: '14px', fontWeight: 700, margin: 0, color: meta.color }}>{node.title}</h3>
   312→          <p style={{ fontSize: '12px', color: '#ffffff', lineHeight: 1.95, margin: 0 }}>{node.desc}</p>
   313→          <span style={{
   314→            display: 'inline-block', padding: '4px 8px', margin: '3px 2px', borderRadius: 8,
   315→            color: '#ffffff', fontSize: '10px',
   316→            background: 'rgba(255,255,255,.06)', border: '1px solid rgba(255,255,255,.09)',
   317→          }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   318→          <div style={{ fontSize: '12px', color: '#e0eaf0' }} dir="ltr">{nodeValues[selectedNode] ?? '--'}</div>
   319→
   320→          <div style={{ marginTop: 12, paddingTop: 12, borderTop: `1px solid ${D.line}` }}>
   321→            <p style={{ fontSize: '12px', fontWeight: 700, marginBottom: 8, color: '#ffffff' }}>سهم هر استراتژی:</p>
   322→            {Object.entries(BRANCH_META).map(([bKey, bMeta]) => {
   323→              const val = contrib[bKey as 'trend' | 'breakout' | 'reversal'];
   324→              const pct = (val * 100).toFixed(1);
   325→              return (
   326→                <div key={bKey} style={{
   327→                  display: 'flex', alignItems: 'center', justifyContent: 'space-between',
   328→                  padding: '9px 0', borderBottom: '1px dashed rgba(255,255,255,.12)',
   329→                }}>
   330→                  <div style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
   331→                    <span style={{ display: 'inline-block', width: 8, height: 8, borderRadius: '50%', background: bMeta.color }} />
   332→                    <span style={{ fontSize: '11px', color: '#ffffff' }}>{bMeta.label}</span>
   333→                  </div>
   334→                  <span style={{ fontSize: '11px', fontWeight: 700, color: bMeta.color }}>{toPersianDigits(pct)}٪</span>
   335→                </div>
   336→              );
   337→            })}
   338→            <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '9px 0' }}>
   339→              <span style={{ fontSize: '11px', fontWeight: 700, color: '#ffffff' }}>مجموع</span>
   340→              <span style={{ fontSize: '11px', fontWeight: 900, color: meta.color }}>{toFa(s?.probability ?? 0)}٪</span>
   341→            </div>
   342→          </div>
   343→        </div>
   344→      );
   345→    }
   346→
   347→    const inputs = edges.filter(e => e.to === selectedNode);
   348→    const outputs = edges.filter(e => e.from === selectedNode);
   349→
   350→    return (
   351→      <div style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}>
   352→        <h3 style={{ fontSize: '14px', fontWeight: 700, margin: 0, color: '#ffffff' }}>{node.title}</h3>
   353→        <p style={{ fontSize: '12px', color: '#ffffff', lineHeight: 1.95, margin: 0 }}><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   354→        <span style={{
   355→          display: 'inline-block', padding: '4px 8px', margin: '3px 2px', borderRadius: 8,
   356→          color: '#ffffff', fontSize: '10px',
   357→          background: 'rgba(255,255,255,.06)', border: '1px solid rgba(255,255,255,.09)',
   358→        }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   359→        <p style={{ fontSize: '12px', color: '#e0eaf0', lineHeight: 1.95, margin: 0 }}>{node.desc}</p>
   360→        {inputs.length > 0 && (
   361→          <div>
   362→            <p style={{ fontSize: '12px', fontWeight: 600, marginBottom: 4, color: '#ffffff' }}>مسیرهای ورودی ({toFa(inputs.length)}):</p>
   363→            <ul style={{ margin: '12px 0 0', padding: 0, listStyle: 'none' }}>
   364→              {inputs.map((e, i) => {
   365→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   366→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   367→                return (
   368→                  <li key={i} style={{
   369→                    padding: '9px 0', borderTop: '1px dashed rgba(255,255,255,.12)',
   370→                    fontSize: '11px', lineHeight: 1.8, color: '#e0eaf0',
   371→                  }}>
   372→                    <b style={{ color: '#ffffff' }}>{SCENARIO_DISPLAY[e.from] || e.from} ← {SCENARIO_DISPLAY[e.to] || e.to}</b>
   373→                    <span style={{
   374→                      marginRight: 8, padding: '2px 6px', borderRadius: 4,
   375→                      fontSize: '9px', fontWeight: 700,
   376→                      background: `${edgeColor}22`, color: edgeColor,
   377→                    }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   378→                    <br />{e.label}
   379→                  </li>
   380→                );
   381→              })}
   382→            </ul>
   383→          </div>
   384→        )}
   385→        {outputs.length > 0 && (
   386→          <div>
   387→            <p style={{ fontSize: '12px', fontWeight: 600, marginBottom: 4, color: '#ffffff' }}>مسیرهای خروجی ({toFa(outputs.length)}):</p>
   388→            <ul style={{ margin: '12px 0 0', padding: 0, listStyle: 'none' }}>
   389→              {outputs.map((e, i) => {
   390→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   391→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   392→                return (
   393→                  <li key={i} style={{
   394→                    padding: '9px 0', borderTop: '1px dashed rgba(255,255,255,.12)',
   395→                    fontSize: '11px', lineHeight: 1.8, color: '#e0eaf0',
   396→                  }}>
   397→                    <b style={{ color: '#ffffff' }}>{SCENARIO_DISPLAY[e.from] || e.from} → {SCENARIO_DISPLAY[e.to] || e.to}</b>
   398→                    <span style={{
   399→                      marginRight: 8, padding: '2px 6px', borderRadius: 4,
   400→                      fontSize: '9px', fontWeight: 700,
   401→                      background: `${edgeColor}22`, color: edgeColor,
   402→                    }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   403→                    <br />{e.label}
   404→                  </li>
   405→                );
   406→              })}
   407→            </ul>
   408→          </div>
   409→        )}
   410→      </div>
   411→    );
   412→  }, [selectedNode, nodeMap, nodeValues, edges, edgeProbabilities, pathContributions, scenarios]);
   413→
   414→  // ── All filter buttons ────────────────────────────────────────────
   415→  const allFilters = [
   416→    ...TYPE_FILTERS,
   417→    { key: 'sep1', label: '│', isSep: true as const },
   418→    ...Object.entries(BRANCH_META).map(([k, v]) => ({ key: k, label: v.label, branchKey: k, branchColor: v.color })),
   419→    { key: 'sep2', label: '│', isSep: true as const },
   420→    ...SCENARIO_KEYS.map(k => ({ key: k, label: SCENARIO_META_LOCAL[k].label, scenarioKey: k })),
   421→  ];
   422→
   423→  // ── Corrected probabilityTrend: day 1 overridden with final scenario probabilities ──
   424→  const correctedTrend = useMemo(() => {
   425→    const raw = props.probabilityTrend;
   426→    if (!raw || !raw.scenarios || raw.scenarios.length === 0) return raw;
   427→
   428→    // Build final individual probabilities (matching boxes) for day 1
   429→    const day1Individuals: Record<string, number> = {};
   430→    for (const key of SCENARIO_KEYS) {
   431→      const s = scenarios[key as keyof typeof scenarios];
   432→      day1Individuals[key] = (s?.probability ?? 0) / 100;
   433→    }
   434→
   435→    // Corrected scenarios array
   436→    const correctedScenarios: ScenarioTrend[] = raw.scenarios.map(sc => {
   437→      const newInd = day1Individuals[sc.scenarioKey] ?? sc.trend[0]?.individualProb ?? 0;

  1001→function MultiSelectTrendChart({ data, selectedKeys, onToggleKey, mode, title }: {
  1002→  data: ProbabilityTrendResult;
  1003→  selectedKeys: Set<string>;
  1004→  onToggleKey: (k: string) => void;
  1005→  mode: 'individual' | 'cumulative';
  1006→  title: string;
  1007→}) {
  1008→  const svgRef = useRef<SVGSVGElement>(null);
  1009→  const tooltipRef = useRef<HTMLDivElement>(null);
  1010→  const [hoveredDay, setHoveredDay] = useState<number | null>(null);
  1011→
  1012→  const scenarios = data.scenarios.filter(s => selectedKeys.has(s.scenarioKey));
  1013→  const maxDays = data.scenarios[0]?.trend?.length ?? 0;
  1014→
  1015→  // Compute Y range — dynamic based on selected scenarios
  1016→  let yMin = 0;
  1017→  let yMax = 0;
  1018→  for (const sc of scenarios) {
  1019→    for (const d of sc.trend) {
  1020→      const v = mode === 'individual' ? d.individualProb : d.cumulativeProb;
  1021→      if (v > yMax) yMax = v;
  1022→    }
  1023→  }
  1024→  // For cumulative mode, ensure minimum range
  1025→  if (yMax <= 0) yMax = 0.1;
  1026→
  1027→  // Compute nice tick interval for Y-axis
  1028→  const range = yMax - yMin;
  1029→  const padding = range * 0.15;
  1030→  const paddedMax = yMax + padding;
  1031→  const paddedMin = Math.max(0, yMin - padding * 0.3);
  1032→
  1033→  // Find a nice step: aim for ~5 grid lines
  1034→  const niceSteps = [0.01, 0.02, 0.025, 0.05, 0.1, 0.15, 0.2, 0.25, 0.5, 1.0];
  1035→  const rawStep = (paddedMax - paddedMin) / 5;
  1036→  let step = niceSteps[0];
  1037→  for (const ns of niceSteps) {
  1038→    if (ns >= rawStep) { step = ns; break; }
  1039→  }
  1040→  // Round yMin down and yMax up to step boundaries
  1041→  const gridYMin = Math.floor(paddedMin / step) * step;
  1042→  const gridYMax = Math.ceil(paddedMax / step) * step;
  1043→  const gridRange = gridYMax - gridYMin;
  1044→
  1045→  const chartW = 1100;
  1046→  const chartH = 300;
  1047→  const padL = 55;
  1048→  const padR = 20;
  1049→  const padT = 20;
  1050→  const padB = 40;
  1051→  const plotW = chartW - padL - padR;
  1052→  const plotH = chartH - padT - padB;
  1053→
  1054→  const xOf = (day: number) => padL + plotW - ((day - 1) / Math.max(1, maxDays - 1)) * plotW;
  1055→  const yOf = (v: number) => padT + plotH - ((v - gridYMin) / Math.max(0.001, gridRange)) * plotH;
  1056→
  1057→  // Dynamic grid lines based on computed range
  1058→  const gridEls: React.JSX.Element[] = [];
  1059→  for (let v = gridYMin; v <= gridYMax + step * 0.01; v += step) {
  1060→    const y = yOf(v);
  1061→    if (y < padT - 2 || y > chartH - padB + 2) continue;
  1062→    const pctVal = v * 100;
  1063→    const pctLabel = Number.isInteger(pctVal) ? String(pctVal) : pctVal.toFixed(step < 0.05 ? 2 : 1);
  1064→    gridEls.push(
  1065→      <line key={`g${v}`} x1={padL} y1={y} x2={chartW - padR} y2={y} stroke="#f3f4f6" strokeWidth={0.7} />,
  1066→      <text key={`gl${v}`} x={padL - 6} y={y + 3.5} textAnchor="end" fill="#9ca3af" fontSize={9} fontFamily="inherit">
  1067→        {toPersianDigits(pctLabel)}%
  1068→      </text>,
  1069→    );
  1070→  }
  1071→  for (let d = 1; d <= maxDays; d += 5) {
  1072→    gridEls.push(
  1073→      <text key={`xl${d}`} x={xOf(d)} y={chartH - 8} textAnchor="middle" fill="#9ca3af" fontSize={8} fontFamily="inherit">
  1074→        {toPersianDigits(String(d))}
  1075→      </text>,
  1076→    );
  1077→  }
  1078→
  1079→  // Vertical hover line
  1080→  const hoverEl = hoveredDay ? (
  1081→    <line x1={xOf(hoveredDay)} y1={padT} x2={xOf(hoveredDay)} y2={chartH - padB} stroke="#94a3b8" strokeWidth={0.8} strokeDasharray="3,3" opacity={0.6} />
  1082→  ) : null;
  1083→
  1084→  // Tooltip
  1085→  const tooltipEl = hoveredDay ? (
  1086→    <div
  1087→      ref={tooltipRef}
  1088→      className="absolute z-10 bg-white border border-gray-200 rounded-lg shadow-lg p-3 text-[10px] pointer-events-none"
  1089→      style={{ left: xOf(hoveredDay) + 10, top: padT }}
  1090→      dir="rtl"
  1091→    >
  1092→      <div className="font-bold text-[#374151] mb-1">روز {toPersianDigits(String(hoveredDay))}</div>
  1093→      {scenarios.map(sc => {
  1094→        const dp = sc.trend.find(d => d.day === hoveredDay);
  1095→        if (!dp) return null;
  1096→        const val = mode === 'individual' ? dp.individualProb : dp.cumulativeProb;
  1097→        const color = SCENARIO_LINE_COLORS[sc.scenarioKey] || '#9ca3af';
  1098→        return (
  1099→          <div key={sc.scenarioKey} className="flex items-center justify-between gap-3">
  1100→            <span className="flex items-center gap-1">
  1101→              <span className="inline-block w-2 h-2 rounded-full" style={{ background: color }} />
  1102→              <span>{sc.label}</span>
  1103→            </span>
  1104→            <span className="font-bold" style={{ color }}>{toPersianDigits((val * 100).toFixed(1))}%</span>
  1105→          </div>
  1106→        );
  1107→      })}
  1108→    </div>
  1109→  ) : null;
  1110→
  1111→  return (
  1112→    <div className="rounded-2xl border border-[#e5e7eb] bg-[#ffffff] overflow-hidden"
  1113→      style={{ boxShadow: '0 2px 12px rgba(0,0,0,.05)' }}>
  1114→      {/* Header + Legend row */}
  1115→      <div className="px-4 pt-3 pb-2">
  1116→        <h3 className="text-xs font-bold text-[#111827] mb-2">{title}</h3>
  1117→        <div className="flex flex-wrap gap-x-4 gap-y-1.5">
  1118→          {data.scenarios.map(sc => {
  1119→            const color = SCENARIO_LINE_COLORS[sc.scenarioKey] || '#9ca3af';
  1120→            const isActive = selectedKeys.has(sc.scenarioKey);
  1121→            const trendIcon = sc.trendDirection === 'rising' ? '↑' : sc.trendDirection === 'falling' ? '↓' : '→';
  1122→            const curVal = sc.trend[0] ? (mode === 'individual' ? sc.trend[0].individualProb : sc.trend[0].cumulativeProb) : 0;
  1123→            return (
  1124→              <button
  1125→                key={sc.scenarioKey}
  1126→                onClick={() => onToggleKey(sc.scenarioKey)}
  1127→                className={`flex items-center gap-1.5 px-2 py-1 rounded-lg border text-[10px] font-medium transition-all cursor-pointer ${
  1128→                  isActive ? 'border-gray-300 bg-gray-50' : 'border-transparent bg-gray-100 opacity-40'
  1129→                }`}
  1130→              >
  1131→                <span className="inline-block w-2.5 h-2.5 rounded-sm" style={{ background: color }} />
  1132→                <span className="text-[#374151]">{sc.scenarioKey}</span>
  1133→                <span style={{ color }}>{trendIcon}</span>
  1134→                {isActive && <span className="text-[#6b7280]">{toPersianDigits((curVal * 100).toFixed(1))}%</span>}
  1135→              </button>
  1136→            );
  1137→          })}
  1138→        </div>
  1139→      </div>
  1140→
  1141→      {/* Chart area */}
  1142→      <div className="relative" style={{ overflowX: 'auto' }}>
  1143→        <svg
  1144→          ref={svgRef}
  1145→          width="100%"
  1146→          viewBox={`0 0 ${chartW} ${chartH}`}
  1147→          style={{ minWidth: 500, display: 'block' }}
  1148→          onMouseMove={(e) => {
  1149→            if (!svgRef.current) return;
  1150→            const rect = svgRef.current.getBoundingClientRect();
  1151→            const scaleX = chartW / rect.width;
  1152→            const svgX = (e.clientX - rect.left) * scaleX;
  1153→            // Find closest day
  1154→            let closestDay = 1;
  1155→            let closestDist = Infinity;
  1156→            for (let d = 1; d <= maxDays; d++) {
  1157→              const dist = Math.abs(xOf(d) - svgX);
  1158→              if (dist < closestDist) { closestDist = dist; closestDay = d; }
  1159→            }
  1160→            if (closestDist < 30) setHoveredDay(closestDay); else setHoveredDay(null);
  1161→          }}
  1162→          onMouseLeave={() => setHoveredDay(null)}
  1163→        >
  1164→          <rect x={0} y={0} width={chartW} height={chartH} fill="#fafafa" />
  1165→          {gridEls}
  1166→          <line x1={padL} y1={padT} x2={padL} y2={chartH - padB} stroke="#d1d5db" strokeWidth={0.7} />
  1167→          <line x1={padL} y1={chartH - padB} x2={chartW - padR} y2={chartH - padB} stroke="#d1d5db" strokeWidth={0.7} />
  1168→          {hoverEl}
  1169→          {scenarios.map(sc => {
  1170→            const color = SCENARIO_LINE_COLORS[sc.scenarioKey] || '#9ca3af';
  1171→            const points = sc.trend.map(d => {
  1172→              const v = mode === 'individual' ? d.individualProb : d.cumulativeProb;
  1173→              return `${xOf(d.day).toFixed(1)},${yOf(v).toFixed(1)}`;
  1174→            }).join(' ');
  1175→            const lastDp = sc.trend[0];
  1176→            const lastV = lastDp ? (mode === 'individual' ? lastDp.individualProb : lastDp.cumulativeProb) : 0;
  1177→            return (
  1178→              <g key={sc.scenarioKey}>
  1179→                <polyline
  1180→                  points={points}
  1181→                  fill="none"
  1182→                  stroke={color}
  1183→                  strokeWidth={2.2}
  1184→                  strokeLinejoin="round"
  1185→                  strokeLinecap="round"
  1186→                  opacity={0.9}
  1187→                />
  1188→                {lastDp && (
  1189→                  <circle cx={xOf(1)} cy={yOf(lastV)} r={3.5} fill={color} stroke="#ffffff" strokeWidth={1.5} />
  1190→                )}
