   143→// Graph Algorithms
   144→// ═══════════════════════════════════════════════════════════════════════════════
   145→
   146→interface EdgeInfo {
   147→  from: string;
   148→  to: string;
   149→  label: string;
   150→  type: string;
   151→  prob: number;
   152→}
   153→
   154→interface PathInfo {
   155→  nodes: string[];
   156→  edges: number[];
   157→  prob: number;
   158→  target: string;
   159→}
   160→
   161→// Calculate edge probabilities based on market conditions
   162→function calcEdgeProbabilities(
   163→  bullScore: number, adxVal: number
   164→): Record<string, number> {
   165→  const adxW = Math.min(adxVal / 100, 1);
   166→  const typeWeights: Record<string, number> = {
   167→    up: bullScore * 0.6 + adxW * 0.4,
   168→    pullback: 0.3,
   169→    down: (1 - bullScore) * 0.6 + adxW * 0.4,
   170→    risk: 0.15 * (1.2 - adxW),
   171→  };
   172→
   173→  // Build adjacency: for each node, group outgoing edges by type
   174→  const adj: Record<string, { idx: number; type: string }[]> = {};
   175→  EDGES.forEach((e, i) => {
   176→    if (!adj[e[0]]) adj[e[0]] = [];
   177→    adj[e[0]].push({ idx: i, type: e[3] });
   178→  });
   179→
   180→  const edgeProbs: Record<string, number> = {};
   181→
   182→  for (const [node, outEdges] of Object.entries(adj)) {
   183→    // Sum weights of outgoing edges
   184→    const totalWeight = outEdges.reduce((sum, e) => sum + (typeWeights[e.type] ?? 0.1), 0);
   185→    if (totalWeight === 0) {
   186→      outEdges.forEach(e => { edgeProbs[String(e.idx)] = 1 / outEdges.length; });
   187→      continue;
   188→    }
   189→    outEdges.forEach(e => {
   190→      edgeProbs[String(e.idx)] = (typeWeights[e.type] ?? 0.1) / totalWeight;
   191→    });
   192→  }
   193→
   194→  return edgeProbs;
   195→}
   196→
   197→// Find all paths from 'A' to terminal nodes using DFS
   198→function findAllPaths(edgeProbs: Record<string, number>): PathInfo[] {
   199→  const adj: Record<string, { to: string; edgeIdx: number }[]> = {};
   200→  EDGES.forEach((e, i) => {
   201→    if (!adj[e[0]]) adj[e[0]] = [];
   202→    adj[e[0]].push({ to: e[1], edgeIdx: i });
   203→  });
   204→
   205→  const paths: PathInfo[] = [];
   206→  const MAX_PATHS = 200;
   207→
   208→  function dfs(node: string, visited: Set<string>, currentPath: string[], edgeIndices: number[], currentProb: number) {
   209→    if (SCENARIO_KEYS.includes(node as any)) {
   210→      paths.push({ nodes: [...currentPath, node], edges: [...edgeIndices], prob: currentProb, target: node });
   211→      return;
   212→    }
   213→    if (paths.length >= MAX_PATHS) return;
   214→    if (visited.has(node)) return;
   215→    visited.add(node);
   216→
   217→    const outs = adj[node] ?? [];
   218→    for (const { to, edgeIdx } of outs) {
   219→      const ep = edgeProbs[String(edgeIdx)] ?? 0.1;
   220→      dfs(to, visited, [...currentPath, node], [...edgeIndices, edgeIdx], currentProb * ep);
   221→    }
   222→    visited.delete(node);
   223→  }
   224→
   225→  dfs('A', new Set(), [], [], 1);
   226→  return paths.sort((a, b) => b.prob - a.prob);
   227→}
