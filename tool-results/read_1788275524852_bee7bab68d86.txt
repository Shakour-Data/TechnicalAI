    20→function toPersianNum(n: number): string {
    21→  if (!isFinite(n) || isNaN(n)) return '\u06f0';
    22→  return Math.round(n).toLocaleString('fa-IR');
    23→}
    24→function srGrade(strength: number): string {
    25→  if (strength >= 8.5) return '\u0628\u0633\u06cc\u0627\u0631 \u0642\u0648\u06cc';
    26→  if (strength >= 7) return '\u0642\u0648\u06cc';
    27→  if (strength >= 5) return '\u0645\u062a\u0648\u0633\u0637';
    28→  if (strength >= 3) return '\u0636\u0639\u06cc\u0641';
    29→  return '\u0628\u0633\u06cc\u0627\u0631 \u0636\u0639\u06cc\u0641';
    30→}

    39→// ─── Build ML Selector Input ──────────────────────────────────────
    40→function buildMLInput(body: Record<string, unknown>): MLSelectorInput {
    41→  const price = (body.currentPrice as number) || 0;
    42→  const trendDir = (body.trendDirection as string) || 'range';
    43→  const adx = (body.adx as number) || 0;
    44→  const rsi = (body.rsi as number) || 50;
    45→  const stochK = (body.stochK as number) || 50;
    46→  const macdHist = (body.macdHist as number) || 0;
    47→  const obv = (body.obv as number) || 0;
    48→  const bollingerUpper = (body.bollingerUpper as number) || 0;
    49→  const bollingerLower = (body.bollingerLower as number) || 0;
    50→  const resistance = (body.resistanceStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
    51→  const support = (body.supportStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
    52→  const atr = (body.atr as number) || 0;
    53→  const bbRange = bollingerUpper - bollingerLower;
    54→  const bbPosition = bbRange > 0 ? Math.min(100, Math.max(0, Math.round((price - bollingerLower) / bbRange * 100))) : 50;
    55→
    56→  // Find dominant scenario across ALL SC1-SC9
    57→  const scenarios = body.scenarios as Record<string, { probability: number }> | undefined;
    58→  let dominantKey = 'SC5';
    59→  let dominantProb = 0;
    60→  for (const k of ['SC1', 'SC2', 'SC3', 'SC4', 'SC5', 'SC6', 'SC7', 'SC8', 'SC9'] as const) {
    61→    const p = scenarios?.[k]?.probability ?? 0;
    62→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
    63→  }
    64→
    65→  return {
    66→    price,
    67→    trend: trendDir === 'up' ? 'up' : trendDir === 'down' ? 'down' : 'range',
    68→    adx, diPlus: (body.diPlus as number) || 0, diMinus: (body.diMinus as number) || 0,
    69→    rsi, stochK, macdHist, obv, bbPosition, resistance, support, atr,
    70→    scenarioDominant: dominantKey,
    71→    hasVolume: (body.hasVolume as boolean) ?? false,
    72→  };
    73→}

    75→// ─── Build MSL v4 Context ────────────────────────────────────────
    76→function buildMSLV4Context(body: Record<string, unknown>): MSLV4Context {
    77→  const price = (body.currentPrice as number) || 0;
    78→  const atr = (body.atr as number) || 0;
    79→  const atrRatio = price > 0 ? atr / price : 0;
    80→
    81→  const v11Probs = body.v11Probabilities as {
    82→    scenarios: Array<{ key: string; name: string; rawProbability: number }>;
    83→  } | undefined;
    84→  let dominantScenario = '';
    85→  if (v11Probs?.scenarios) {
    86→    const sorted = [...v11Probs.scenarios].sort((a, b) => b.rawProbability - a.rawProbability);
    87→    dominantScenario = sorted[0]?.name ?? '';
    88→  } else {
    89→    const scenarios = body.scenarios as Record<string, { name?: string; probability: number }> | undefined;
    90→    let maxProb = 0;
    91→    for (const k of ['SC1', 'SC2', 'SC3', 'SC4', 'SC5', 'SC6', 'SC7', 'SC8', 'SC9']) {
    92→      const p = scenarios?.[k]?.probability ?? 0;
    93→      if (p > maxProb) { maxProb = p; dominantScenario = scenarios?.[k]?.name ?? k; }
    94→    }
    95→  }
    96→
    97→  const trendDir = (body.trendDirection as string) || 'range';
    98→  const isBull = trendDir === 'up';
    99→  const isBear = trendDir === 'down';
   100→  const adx = (body.adx as number) || 0;
   101→  const trendStrength = adx > 40 ? 0.9 : adx > 25 ? 0.7 : adx > 15 ? 0.4 : 0.15;
   102→
   103→  let regime: MSLV4Context['regime'] = 'Range';
   104→  if (isBull && trendStrength > 0.6) regime = 'Strong Bull';
   105→  else if (isBull) regime = 'Weak Bull';
   106→  else if (isBear && trendStrength > 0.6) regime = 'Strong Bear';
   107→  else if (isBear) regime = 'Weak Bear';
   108→
   109→  const instrumentType = (body.instrumentType as string) || (body.symbolName as string) || 'stock';
   110→  const ta = body.technicalAnalysis as {
   111→    classicPatternCount?: number; harmonicPatternCount?: number;
   112→    elliottWaveCount?: number; divergenceCount?: number;
   113→  } | undefined;
   114→  const overallConfidence = (body.overallConfidence as number) ?? (body.confidence as number) ?? 0.6;
   115→
   116→  // Extract probability trend signals for MSL selection
   117→  const probTrend = extractProbTrendSignals(body);
   118→
   119→  return {
   120→    asset: instrumentType, timeframe: (body.timeframe as string) || 'daily',
   121→    regime, confidence: Math.max(0, Math.min(1, overallConfidence)),
   122→    volatility: atrRatio, trend_strength: trendStrength, dominantScenario,
   123→    pattern_counts: { classical: ta?.classicPatternCount ?? 0, harmonic: ta?.harmonicPatternCount ?? 0, elliott: ta?.elliottWaveCount ?? 0 },
   124→    divergence_present: (ta?.divergenceCount ?? 0) > 0,
   125→    hasVolume: (body.hasVolume as boolean) ?? false,
   126→    rsi: (body.rsi as number) || 50, adx,
   127→    audience_level: (body.audienceLevel as string) || 'pro',
   128→    probTrend,
   129→  };
   130→}

   132→// ─── Extract probability trend signals from body ──────────────────
   133→function extractProbTrendSignals(body: Record<string, unknown>): MSLV4Context['probTrend'] {
   134→  const pt = body.probabilityTrend as {
   135→    scenarios?: { scenarioKey: string; group: string; trendDirection: string; trend: { individualProb: number; cumulativeProb: number }[] }[];
   136→    groups?: { group: string; trendDirection: string; trend: { cumulativeProb: number }[] }[];
   137→  } | undefined;
   138→  if (!pt?.groups || pt.groups.length === 0) return undefined;
   139→
   140→  const findGroup = (g: string) => pt.groups!.find(x => x.group === g);
   141→  const bullG = findGroup('bullish');
   142→  const bearG = findGroup('bearish');
   143→  const neutralG = findGroup('neutral');
   144→
   145→  // Find dominant scenario trend
   146→  const scenarios = body.scenarios as Record<string, { probability: number }> | undefined;
   147→  let domKey = 'SC5'; let domProb = 0;
   148→  for (const k of ['SC1','SC2','SC3','SC4','SC5','SC6','SC7','SC8','SC9']) {
   149→    const p = scenarios?.[k]?.probability ?? 0;
   150→    if (p > domProb) { domProb = p; domKey = k; }
   151→  }
   152→  const domScenario = pt.scenarios?.find(s => s.scenarioKey === domKey);
   153→
   154→  // 7-day change (day 1 = today, day 7 = 7 days ago)
   155→  const get7dChange = (trend: { cumulativeProb: number }[]) => {
   156→    if (!trend || trend.length < 7) return 0;
   157→    return trend[0].cumulativeProb - trend[6].cumulativeProb;
   158→  };
   159→
   160→  return {
   161→    bullGroupDir: bullG?.trendDirection as MSLV4Context['probTrend']['bullGroupDir'],
   162→    bearGroupDir: bearG?.trendDirection as MSLV4Context['probTrend']['bearGroupDir'],
   163→    neutralGroupDir: neutralG?.trendDirection as MSLV4Context['probTrend']['neutralGroupDir'],
   164→    bullGroupCum: bullG?.trend?.[0]?.cumulativeProb,
   165→    bearGroupCum: bearG?.trend?.[0]?.cumulativeProb,
   166→    neutralGroupCum: neutralG?.trend?.[0]?.cumulativeProb,
   167→    dominantCumDir: domScenario?.trendDirection as MSLV4Context['probTrend']['dominantCumDir'],
   168→    dominantIndivDir: domScenario?.trendDirection as MSLV4Context['probTrend']['dominantIndivDir'],
   169→    bullGroupChange7d: get7dChange(bullG?.trend ?? []),
   170→    bearGroupChange7d: get7dChange(bearG?.trend ?? []),
   171→    neutralGroupChange7d: get7dChange(neutralG?.trend ?? []),
   172→  };
   173→}

   384→export async function POST(req: NextRequest) {
   385→  const startTime = Date.now();
   386→  try {
   387→    const body = await req.json();
   388→    if (!body.currentPrice) {
   389→      return NextResponse.json({ error: 'currentPrice is required' }, { status: 400 });
   390→    }
   391→
   392→    const symbolName = String(body.symbolName || 'unknown');
   393→    const today = getTodayDateStr();
   394→
   395→    // 1. Check daily persistent cache (Prisma/SQLite)
   396→    try {
   397→      const cached = await db.aiAnalysisCache.findUnique({
   398→        where: { symbol_date: { symbol: symbolName, date: today } },
   399→      });
   400→      if (cached && cached.text) {
   401→        console.log(`[AI] Daily cache HIT for ${symbolName} (${today}) [${Date.now() - startTime}ms]`);
   402→        return NextResponse.json({
   403→          text: cached.text,
   404→          ml: cached.ml ? JSON.parse(cached.ml) : undefined,
   405→          cached: true,
   406→          cachedDate: cached.date,
   407→        });
   408→      }
   409→    } catch (dbErr) {
   410→      console.warn('[AI] DB cache read failed, continuing:', dbErr instanceof Error ? dbErr.message : dbErr);
   411→    }
   412→
   413→    // 1b. Check for previous day's cache (fallback only if price is within 2%)
   414→    try {
   415→      const currentPrice = Number(body.currentPrice) || 0;
   416→      const prevCached = await db.aiAnalysisCache.findFirst({
   417→        where: { symbol: symbolName, date: { not: today } },
   418→        orderBy: { createdAt: 'desc' },
   419→      });
   420→      if (prevCached && prevCached.text && prevCached.price > 0 && currentPrice > 0) {
   421→        const priceDiff = Math.abs(prevCached.price - currentPrice) / currentPrice;
   422→        if (priceDiff < 0.02) { // Less than 2% price difference
   423→          console.log(`[AI] Previous cache fallback HIT for ${symbolName} (${prevCached.date}, price diff ${((priceDiff)*100).toFixed(1)}%) [${Date.now() - startTime}ms]`);
   424→          return NextResponse.json({
   425→            text: prevCached.text,
   426→            ml: prevCached.ml ? JSON.parse(prevCached.ml) : undefined,
   427→            cached: true,
   428→            cachedDate: prevCached.date,
   429→            isFallback: true,
   430→          });
   431→        } else {
   432→          console.log(`[AI] Previous cache SKIP for ${symbolName} (price diff ${((priceDiff)*100).toFixed(1)}% > 2%)`);
   433→        }
   434→      }
   435→    } catch (dbErr) {
   436→      console.warn('[AI] Previous cache fallback read failed, continuing:', dbErr instanceof Error ? dbErr.message : dbErr);
   437→    }
   438→
   439→    // 2. ML selection
   440→    const mlInput = buildMLInput(body);
   441→    const mlSelection = selectMLCombination(mlInput);
   442→    const methods = selectMethods(mlInput);
   443→
   444→    // 3. MSL v4 selection
   445→    const mslCtx = buildMSLV4Context(body);
   446→    const mslResult = selectMSLV4(mslCtx);
   447→    mlSelection.school = mslResult.school_of_analysis.primary.id as typeof mlSelection.school;
   448→    mlSelection.style = mslResult.analysis_style.primary.id as typeof mlSelection.style;
   449→    mlSelection.tone = mslResult.analysis_tone.primary.id as typeof mlSelection.tone;
   450→    mlSelection.reasoning = `MSLv4: ${mslResult.school_of_analysis.primary.id}/${mslResult.analysis_style.primary.id}/${mslResult.analysis_tone.primary.id}`;
   451→
   452→    // 4. Build prompt
   453→    const mslSystemPrompt = buildMSLV4PromptSection(mslResult);
   454→    const userMessage = buildPrompt(body, mlSelection, methods);
   455→    const dynamicSystemPrompt = `${SYSTEM_PROMPT}\n\n${mslSystemPrompt}`;
   456→
   457→    console.log(`[AI] Generating for ${symbolName} (cache miss) [${Date.now() - startTime}ms]`);
   458→
   459→    // 5. Call AI through dedicated channel — fail fast on 429
   460→    //    Only retry 2 times max (3 total attempts) to avoid wasting time.
   461→    //    If rate-limited, the previous-day fallback above already returned.
   462→    const content = await dedicatedAIChatCompletion(
   463→      [
   464→        { role: 'system', content: dynamicSystemPrompt },
   465→        { role: 'user', content: userMessage },
   466→      ],
   467→      {
   468→        timeoutMs: 90_000,  // 90s max (1 initial + 2 retries)
   469→        maxRetries: 2,
   470→      }
   471→    );
   472→
   473→    // Post-process: strip ALL leaked technical codes
   474→    const cleaned = content
   475→      .replace(/\bSC\d+\b/g, '')
   476→      .replace(/\bR\d+\b(?=[\s,.;:!?\)\-\u0627-\u06cc]|$)/g, '')
   477→      .replace(/\bMA\d+\b/g, '')
   478→      .replace(/\bRSI\b/g, 'شاخص قدرت نسبی')
   479→      .replace(/\bMACD\b/g, 'واگراف هیستوگرام')
   480→      .replace(/\bMFI\b/g, 'شاخص جریان نقدي')
   481→      .replace(/\bCCI\b/g, 'شاخص کانال کالا')
   482→      .replace(/\bADX\b/g, 'شاخص شدت روند')
   483→      .replace(/\bATR\b/g, 'دامنه تلواتي')
   484→      .replace(/\bSAR\b/g, 'حمایت پویا')
   485→      .replace(/\bOBV\b/g, 'جریان تجمعی حجم')
   486→      .replace(/\bDI[+\-]/g, '')
   487→      .replace(/\bR\s*\u00b2\s*=\s*[\d.]+/g, '')
   488→      .replace(/\s{2,}/g, ' ')
   489→      .trim();
   490→    const mlData = { school: mlSelection.school, style: mlSelection.style, tone: mlSelection.tone, reasoning: mlSelection.reasoning, methods };
   491→
   492→    // 7. Save to daily persistent cache (save CLEANED text, not raw)
   493→    try {
   494→      await db.aiAnalysisCache.upsert({
   495→        where: { symbol_date: { symbol: symbolName, date: today } },
   496→        create: {
   497→          symbol: symbolName,
   498→          date: today,
   499→          text: cleaned,
   500→          ml: JSON.stringify(mlData),
   501→          price: Number(body.currentPrice) || 0,
   502→        },
   503→        update: {
   504→          text: cleaned,
   505→          ml: JSON.stringify(mlData),
   506→          price: Number(body.currentPrice) || 0,
   507→        },
   508→      });
   509→      console.log(`[AI] Saved to daily cache: ${symbolName} (${today}) [${Date.now() - startTime}ms]`);
   510→    } catch (dbErr) {
   511→      console.warn('[AI] DB cache write failed:', dbErr instanceof Error ? dbErr.message : dbErr);
   512→    }
   513→
   514→    console.log(`[AI] Complete for ${symbolName} [${Date.now() - startTime}ms]`);
   515→    return NextResponse.json({ text: cleaned, ml: mlData });
   516→
   517→  } catch (err) {
   518→    console.error(`[AI] Error [${Date.now() - startTime}ms]:`, err);
   519→    return NextResponse.json({ error: userFriendlyError(err), text: '' }, { status: 500 });
   520→  }
   521→}
   522→
   523→function userFriendlyError(err: unknown): string {
   524→  const msg = err instanceof Error ? err.message : String(err);
   525→  if (msg.includes('Rate limited') || msg.includes('429')) {
   526→    return 'سرور هوشمند محدودیت سرعت دارد.';
   527→  }
   528→  if (msg.includes('Timed out') || msg.includes('timeout') || msg.includes('Timeout')) {
   529→    return 'زمان پاسخدهی هوشمند به پایان رسید.';
   530→  }
   531→  if (msg.includes('concurrent')) {
   532→    return 'درخواست تحلیل قبلی هنوز در حال اجراست.';
   533→  }
   534→  return 'خطایی در تولید تحلیل رخ داد.';
   535→}
