   142→type ToolType = 'cursor' | 'trendline' | 'hline' | 'vline' | 'ray' | 'fibonacci' | 'fibext' | 'rectangle' | 'ellipse' | 'path' | 'text' | 'callout' | 'brush' | 'measure' | 'arrow' | 'pitchfork' | 'channel' | 'regression';

   144→// — S/R strength-based line styling (solid only, thickness = strength) ————
   145→function srLineStyle(strength: number, isTarget: boolean): { lineWidth: number; lineStyle: 0; color: string } {
   146→  if (isTarget) {
   147→    return { lineWidth: 4, lineStyle: 0, color: '' };
   148→  }
   149→  if (strength >= 9) return { lineWidth: 5, lineStyle: 0, color: '' };
   150→  if (strength >= 7) return { lineWidth: 4, lineStyle: 0, color: '' };
   151→  if (strength >= 5) return { lineWidth: 3, lineStyle: 0, color: '' };
   152→  if (strength >= 3) return { lineWidth: 2, lineStyle: 0, color: '' };
   153→  return { lineWidth: 1, lineStyle: 0, color: '' };
   154→}

   195→/* ----------------------------- COMPONENT --- */
   196→
   197→export default function CandlestickChart({ data, ta, height = 520, priceDecimals = 0 }: CandlestickChartProps) {
   198→  const { colors: C, isDark } = useTheme();
   199→  const chartContainerRef = useRef<HTMLDivElement>(null);
   200→  const svgRef = useRef<SVGSVGElement>(null);
   201→  const chartRef = useRef<IChartApi | null>(null);
   202→  const csRef = useRef<any>(null);
   203→  const subscriberRef = useRef<{ dispose: () => void } | null>(null);
   204→  const crosshairSubRef = useRef<{ dispose: () => void } | null>(null);
   205→
   206→  // Crosshair legend state
   207→  const [hoverInfo, setHoverInfo] = useState<{ date: string; o: number; h: number; l: number; c: number; v: number; chg: number } | null>(null);
   208→
   209→  // Build Jalali date strings for all candles (memoized)
   210→  const jalaliDates = useMemo(() => {
   211→    return data.map((d) =>
   212→      isGregorianDate(d.date) ? candleDateToJalali(d.date, 'full') : formatJalaliString(d.date, 'full')
   213→    );
   214→  }, [data]);
   215→
   216→  const [drawings, setDrawings] = useState<Drawing[]>([]);
   217→  const [activeTool, setActiveTool] = useState<ToolType>('cursor');
   218→  const [activeColor, setActiveColor] = useState(PALETTE[0]);
   219→  const [isDrawing, setIsDrawing] = useState(false);
   220→  const [currentDrawing, setCurrentDrawing] = useState<Drawing | null>(null);
   221→  const [textInput, setTextInput] = useState<{ x: number; y: number; drawingId: string } | null>(null);
   222→  const [textValue, setTextValue] = useState('');
   223→  const [renderKey, setRenderKey] = useState(0);
   224→  const [chartSize, setChartSize] = useState({ w: 0, h: 0 });
   225→
   226→  /* - Helper: coordinate conversion -------------- */
   227→  const getChartPoint = useCallback((clientX: number, clientY: number): DrawingPoint | null => {
   228→    const chart = chartRef.current;
   229→    const container = chartContainerRef.current;
   230→    if (!chart || !container) return null;
   231→    const rect = container.getBoundingClientRect();
   232→    const x = clientX - rect.left;
   233→    const y = clientY - rect.top;
   234→    const price = csRef.current?.coordinateToPrice(y) ?? null;
   235→    const time = chart.timeScale().coordinateToTime(x);
   236→    if (price === null || time === null) return null;
   237→    return { time: time as number, price };
   238→  }, []);
   239→
   240→  /* - SVG Rendering --------------------- */
   241→  const renderDrawings = useCallback(() => {
   242→    const chart = chartRef.current;
   243→    const svg = svgRef.current;
   244→    const container = chartContainerRef.current;
   245→    if (!chart || !svg || !container || drawings.length === 0) {
   246→      if (svg) svg.innerHTML = '';
   247→      return;
   248→    }
   249→
   250→    const rect = container.getBoundingClientRect();
   251→    const w = rect.width;
   252→    const h = rect.height;
   253→    if (w === 0 || h === 0) return;
   254→
   255→    let html = '';
   256→
   257→    // Defs for arrow markers
   258→    html += `<defs>`;
   259→    PALETTE.forEach((c) => {
   260→      html += `<marker id="arrow-${c.replace('#','')}" markerWidth="8" markerHeight="6" refX="7" refY="3" orient="auto"><polygon points="0 0, 8 3, 0 6" fill="${c}" /></marker>`;
   261→    });
   262→    html += `</defs>`;
   263→
   264→    const allDrawings = [...drawings];
   265→    if (currentDrawing) allDrawings.push(currentDrawing);
   266→
   267→    for (const d of allDrawings) {
   268→      if (!d.completed && d.points.length === 0) continue;
   269→      const col = d.color;
   270→      const lw = d.lineWidth;
   271→
   272→      switch (d.type) {
   273→        case 'trendline': {
   274→          if (d.points.length < 1) break;
   275→          const p0 = d.points[0];
   276→          const p1 = d.points.length > 1 ? d.points[d.points.length - 1] : p0;
   277→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   278→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   279→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   280→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   281→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   282→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" />`;
   283→          break;
   284→        }
   285→        case 'hline': {
   286→          if (d.points.length < 1) break;
   287→          const p = d.points[0];
   288→          const y = csRef.current?.priceToCoordinate(p.price);
   289→          if (y == null) break;
   290→          html += `<line x1="0" y1="${y}" x2="${w}" y2="${y}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   291→          html += `<text x="4" y="${y - 4}" fill="${col}" font-size="11" font-family="Vazirmatn, sans-serif">${toPersianDigits(p.price.toFixed(priceDecimals))}</text>`;
   292→          break;
   293→        }
   294→        case 'vline': {
   295→          if (d.points.length < 1) break;
   296→          const p = d.points[0];
   297→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   298→          if (x == null) break;
   299→          html += `<line x1="${x}" y1="0" x2="${x}" y2="${h}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   300→          break;
   301→        }
   302→        case 'fibonacci': {
   303→          if (d.points.length < 2) break;
   304→          const p0 = d.points[0];
   305→          const p1 = d.points[1];
   306→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   307→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   308→          if (x0 == null || x1 == null) break;
   309→          const priceHigh = Math.max(p0.price, p1.price);
   310→          const priceLow = Math.min(p0.price, p1.price);
   311→          const priceRange = priceHigh - priceLow;
   312→          if (priceRange === 0) break;
   313→
   314→          const yTop = csRef.current?.priceToCoordinate(priceHigh);
   315→          const yBottom = csRef.current?.priceToCoordinate(priceLow);
   316→          if (yTop == null || yBottom == null) break;
   317→          html += `<rect x="${Math.min(x0,x1)}" y="${yTop}" width="${Math.abs(x1-x0)}" height="${yBottom - yTop}" fill="${col}" opacity="0.04" />`;
   318→
   319→          for (const level of FIB_LEVELS) {
   320→            const price = priceHigh - level.pct * priceRange;
   321→            const y = csRef.current?.priceToCoordinate(price);
   322→            if (y == null) continue;
   323→            html += `<line x1="${Math.min(x0,x1) - 10}" y1="${y}" x2="${Math.max(x0,x1) + 10}" y2="${y}" stroke="${col}" stroke-width="1" opacity="0.6" />`;
   324→            html += `<rect x="${Math.max(x0,x1) + 12}" y="${y - 8}" width="56" height="16" fill="${C.cardBg}" stroke="${C.cardBorder}" stroke-width="1" rx="2" />`;
   325→            html += `<text x="${Math.max(x0,x1) + 14}" y="${y + 4}" fill="${col}" font-size="10" font-family="Vazirmatn, sans-serif">${toPersianDigits(level.label)} ${toPersianDigits(price.toFixed(priceDecimals))}</text>`;
   326→          }
   327→          break;
   328→        }
   329→        case 'rectangle': {
   330→          if (d.points.length < 2) break;
   331→          const p0 = d.points[0];
   332→          const p1 = d.points[d.points.length - 1];
   333→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   334→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   335→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   336→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   337→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   338→          const rx = Math.min(x0, x1);
   339→          const ry = Math.min(y0, y1);
   340→          const rw = Math.abs(x1 - x0);
   341→          const rh = Math.abs(y1 - y0);
   342→          html += `<rect x="${rx}" y="${ry}" width="${rw}" height="${rh}" fill="${col}" fill-opacity="0.08" stroke="${col}" stroke-width="${lw}" rx="2" />`;
   343→          break;
   344→        }
   345→        case 'text': {
   346→          if (d.points.length < 1 || !d.text) break;
   347→          const p = d.points[0];
   348→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   349→          const y = csRef.current?.priceToCoordinate(p.price);
   350→          if (x == null || y == null) break;
   351→          html += `<rect x="${x + 4}" y="${y - 16}" width="${d.text.length * 8 + 12}" height="22" fill="${C.cardBg}" stroke="${C.cardBorder}" stroke-width="1" rx="3" />`;
   352→          html += `<text x="${x + 10}" y="${y + 1}" fill="${col}" font-size="12" font-family="Vazirmatn, sans-serif">${d.text}</text>`;
   353→          break;
   354→        }
   355→        case 'brush': {
   356→          if (d.points.length < 2) break;
   357→          let pts = '';
   358→          let valid = true;
   359→          for (const p of d.points) {
   360→            const x = chart.timeScale().timeToCoordinate(p.time as Time);
   361→            const y = csRef.current?.priceToCoordinate(p.price);
   362→            if (x == null || y == null) { valid = false; break; }
   363→            pts += `${x},${y} `;
   364→          }
   365→          if (!valid || pts.length === 0) break;
   366→          html += `<polyline points="${pts.trim()}" fill="none" stroke="${col}" stroke-width="${lw}" stroke-linecap="round" stroke-linejoin="round" />`;
   367→          break;
   368→        }
   369→        case 'measure': {
   370→          if (d.points.length < 2) break;
   371→          const p0 = d.points[0];
   372→          const p1 = d.points[d.points.length - 1];
   373→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   374→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   375→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   376→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   377→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   378→
   379→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="4,2" />`;
   380→          const diff = p1.price - p0.price;
   381→          const pctChg = p0.price !== 0 ? ((diff / p0.price) * 100) : 0;
   382→          const bars = Math.abs(Math.round(p1.time - p0.time));
   383→          const midX = (x0 + x1) / 2;
   384→          const midY = (y0 + y1) / 2;
   385→          const label = `${diff >= 0 ? '+' : ''}${toPersianDigits(diff.toFixed(priceDecimals))} (${pctChg >= 0 ? '+' : ''}${toPersianDigits(pctChg.toFixed(1))}%) ${toPersianDigits(String(bars))}bar`;
   386→          html += `<rect x="${midX - 40}" y="${midY - 24}" width="80" height="20" fill="${C.cardBg}" stroke="${C.cardBorder}" stroke-width="1" rx="3" />`;
   387→          html += `<text x="${midX - 36}" y="${midY - 10}" fill="${col}" font-size="10" font-family="Vazirmatn, sans-serif">${label}</text>`;
   388→          break;
   389→        }
   390→        case 'arrow': {
   391→          if (d.points.length < 2) break;
   392→          const p0 = d.points[0];
   393→          const p1 = d.points[d.points.length - 1];
   394→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   395→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   396→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   397→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   398→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   399→          const markerId = `arrow-${col.replace('#','')}`;
   400→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" marker-end="url(#${markerId})" />`;
   401→          break;
   402→        }
   403→        case 'ray': {
   404→          if (d.points.length < 2) break;
   405→          const p0 = d.points[0];
   406→          const p1 = d.points[d.points.length - 1];
   407→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   408→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   409→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   410→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   411→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   412→          // Extend the line from p1 to the edge of the chart
   413→          const dx = x1 - x0;
   414→          const dy = y1 - y0;
   415→          const extendFactor = 10;
   416→          const xEnd = x1 + dx * extendFactor;
   417→          const yEnd = y1 + dy * extendFactor;
   418→          html += `<line x1="${x0}" y1="${y0}" x2="${xEnd}" y2="${yEnd}" stroke="${col}" stroke-width="${lw}" />`;
