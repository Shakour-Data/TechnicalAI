  1114→/**
  1115→ * Deterministic seeded shuffle using Fisher-Yates.
  1116→ */
  1117→function seededShuffle<T>(array: T[], seed: number): T[] {
  1118→  const result = [...array];
  1119→  let s = seed;
  1120→  for (let i = result.length - 1; i > 0; i--) {
  1121→    s = (s * 1664525 + 1013904223) & 0xffffffff;
  1122→    const j = s % (i + 1);
  1123→    [result[i], result[j]] = [result[j], result[i]];
  1124→  }
  1125→  return result;
  1126→}

  1128→/**
  1129→ * Get `count` phrases from a section/subcategory, deterministically shuffled by seed.
  1130→ * Returns fewer phrases if the subcategory doesn't have enough.
  1131→ */
  1132→export function getPhrases(section: string, subcategory: string, seed: number, count: number): string[] {
  1133→  const sec = PHRASE_BANK[section];
  1134→  if (!sec) return [];
  1135→  const phrases = sec.phrases[subcategory];
  1136→  if (!phrases || phrases.length === 0) return [];
  1137→  const shuffled = seededShuffle(phrases, seed);
  1138→  return shuffled.slice(0, Math.min(count, shuffled.length));
  1139→}
