    62→function hexToRgba(hex: string, alpha: number): string {
    63→  const r = parseInt(hex.slice(1, 3), 16);
    64→  const g = parseInt(hex.slice(3, 5), 16);
    65→  const b = parseInt(hex.slice(5, 7), 16);
    66→  return `rgba(${r},${g},${b},${alpha})`;
    67→}
    68→
    69→const toFa = (n: number) => Math.round(n).toLocaleString('fa-IR');
    70→
    71→function rsiSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    72→  if (v > 70) return 'bearish';
    73→  if (v < 30) return 'bullish';
    74→  return 'neutral';
    75→}
    76→
    77→function mfiSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    78→  if (v > 80) return 'bearish';
    79→  if (v < 20) return 'bullish';
    80→  return 'neutral';
    81→}
    82→
    83→function stochSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    84→  if (v > 80) return 'bearish';
    85→  if (v < 20) return 'bullish';
    86→  return 'neutral';
    87→}
    88→
    89→function macdSignal(v: { line: number; signal: number; histogram: number }): 'bullish' | 'bearish' | 'neutral' {
    90→  if (v.histogram > 0 && v.line > v.signal) return 'bullish';
    91→  if (v.histogram < 0 && v.line < v.signal) return 'bearish';
    92→  return 'neutral';
    93→}
    94→
    95→function adxSignal(adx: number, diP: number, diM: number): 'bullish' | 'bearish' | 'neutral' {
    96→  if (adx < 20) return 'neutral';
    97→  return diP > diM ? 'bullish' : 'bearish';
    98→}
    99→
   100→function trendDirSignal(dir: string): 'bullish' | 'bearish' | 'neutral' {
   101→  if (dir === 'up') return 'bullish';
   102→  if (dir === 'down') return 'bearish';
   103→  return 'neutral';
   104→}
   105→
   106→function gradeStyle(grade: string, cardSubFg: string): React.CSSProperties {
   107→  switch (grade) {
   108→    case 'Very Strong': return { background: 'rgba(239,68,68,0.2)', color: '#f87171' };
   109→    case 'Strong': return { background: 'rgba(245,158,11,0.2)', color: '#fbbf24' };
   110→    case 'Moderate': return { background: 'rgba(14,165,233,0.2)', color: '#38bdf8' };
   111→    default: return { background: 'rgba(255,255,255,0.1)', color: cardSubFg };
   112→  }
   113→}

   202→// ─── Trend Analysis ────────────────────────────────────────────────────────────
   203→
   204→interface TrendInfo {
   205→  label: string;
   206→  direction: 'rising' | 'falling' | 'stable';
   207→  signal: 'bullish' | 'bearish' | 'neutral';
   208→}
   209→
   210→function analyzeIndicatorTrends(ta: NonNullable<IndicatorsPanelProps['ta']>): TrendInfo[] {
   211→  const trends: TrendInfo[] = [];
   212→
   213→  trends.push({
   214→    label: 'RSI',
   215→    direction: ta.rsi > 60 ? 'rising' : ta.rsi < 40 ? 'falling' : 'stable',
   216→    signal: rsiSignal(ta.rsi),
   217→  });
   218→
   219→  trends.push({
   220→    label: 'استوکاستیک',
   221→    direction: ta.stochK > 60 ? 'rising' : ta.stochK < 40 ? 'falling' : 'stable',
   222→    signal: stochSignal(ta.stochK),
   223→  });
   224→
   225→  trends.push({
   226→    label: 'MACD',
   227→    direction: ta.macd.histogram > 0 ? 'rising' : ta.macd.histogram < 0 ? 'falling' : 'stable',
   228→    signal: macdSignal(ta.macd),
   229→  });
   230→
   231→  trends.push({
   232→    label: 'ADX',
   233→    direction: ta.adx > 25 ? 'rising' : ta.adx < 15 ? 'falling' : 'stable',
   234→    signal: adxSignal(ta.adx, ta.diPlus, ta.diMinus),
   235→  });
   236→
   237→  trends.push({
   238→    label: 'CCI',
   239→    direction: ta.cci > 100 ? 'rising' : ta.cci < -100 ? 'falling' : 'stable',
   240→    signal: ta.cci > 100 ? 'bearish' : ta.cci < -100 ? 'bullish' : 'neutral',
   241→  });
   242→
   243→  if (ta.hasVolume !== false) {
   244→    trends.push({
   245→      label: 'MFI',
   246→      direction: ta.mfi > 60 ? 'rising' : ta.mfi < 40 ? 'falling' : 'stable',
   247→      signal: mfiSignal(ta.mfi),
   248→    });
   249→  }
   250→
   251→  trends.push({
   252→    label: 'ویلیامز %R',
   253→    direction: ta.williamsR > -40 ? 'rising' : ta.williamsR < -60 ? 'falling' : 'stable',
   254→    signal: ta.williamsR > -20 ? 'bearish' : ta.williamsR < -80 ? 'bullish' : 'neutral',
   255→  });
   256→
   257→  const bbWidth = ta.bollingerBands.upper - ta.bollingerBands.lower;
   258→  trends.push({
   259→    label: 'بولینگر',
   260→    direction: bbWidth > ta.bollingerBands.middle * 0.04 ? 'rising' : bbWidth < ta.bollingerBands.middle * 0.02 ? 'falling' : 'stable',
   261→    signal: 'neutral',
   262→  });
   263→
   264→  return trends;
   265→}
