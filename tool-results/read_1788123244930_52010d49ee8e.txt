    40→function hexToRgba(hex: string, alpha: number): string {
    41→  const r = parseInt(hex.slice(1, 3), 16);
    42→  const g = parseInt(hex.slice(3, 5), 16);
    43→  const b = parseInt(hex.slice(5, 7), 16);
    44→  return `rgba(${r},${g},${b},${alpha})`;
    45→}
    46→
    47→const safeNum = (n: number | null | undefined, fallback = 0) => (n ?? fallback);
    48→const toFa = (n: number | null | undefined) => Math.round(safeNum(n)).toLocaleString('fa-IR');
    49→const toFa1 = (n: number | null | undefined) => toPersianDigits(safeNum(n).toFixed(1)).replace(/\./g, '٫').replace(/-/g, '−');
    50→const toFa2 = (n: number | null | undefined) => toPersianDigits(safeNum(n).toFixed(2)).replace(/\./g, '٫').replace(/-/g, '−');
    51→
    52→function rsiSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    53→  if (v > 70) return 'bearish';
    54→  if (v < 30) return 'bullish';
    55→  return 'neutral';
    56→}
    57→
    58→function macdSignal(h: number): 'bullish' | 'bearish' | 'neutral' {
    59→  if (h > 0) return 'bullish';
    60→  if (h < 0) return 'bearish';
    61→  return 'neutral';
    62→}
    63→
    64→function adxSignal(v: number): 'strong' | 'neutral' {
    65→  if (v > 25) return 'strong';
    66→  return 'neutral';
    67→}
    68→
    69→function stochSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    70→  if (v > 80) return 'bearish';
    71→  if (v < 20) return 'bullish';
    72→  return 'neutral';
    73→}

    75→function signalDotBg(signal: 'bullish' | 'bearish' | 'neutral' | 'strong', bullColor: string, bearColor: string, neutralColor: string, cardSubFg: string): string {
    76→  switch (signal) {
    77→    case 'bullish': return bullColor;
    78→    case 'bearish': return bearColor;
    79→    case 'strong': return neutralColor;
    80→    default: return cardSubFg;
    81→  }
    82→}
    83→
    84→function gradeLabel(grade: string): string {
    85→  switch (grade) {
    86→    case 'Very Strong': return 'بسیار قوی';
    87→    case 'Strong': return 'قوی';
    88→    case 'Moderate': return 'متوسط';
    89→    default: return 'ضعیف';
    90→  }
    91→}
    92→
    93→function gradeStyle(grade: string, bearColor: string, neutralColor: string, primary: string, cardBg: string, cardSubFg: string): React.CSSProperties {
    94→  switch (grade) {
    95→    case 'Very Strong': return { background: hexToRgba(bearColor, 0.15), color: bearColor };
    96→    case 'Strong': return { background: hexToRgba(neutralColor, 0.15), color: neutralColor };
    97→    case 'Moderate': return { background: hexToRgba(primary, 0.15), color: primary };
    98→    default: return { background: hexToRgba(cardSubFg, 0.1), color: cardSubFg };
    99→  }
   100→}
   101→
   102→function formatNumber(n: number | null | undefined): string {
   103→  const v = safeNum(n);
   104→  if (Math.abs(v) >= 1e9) return toPersianDigits((v / 1e9).toFixed(1)) + 'B';
   105→  if (Math.abs(v) >= 1e6) return toPersianDigits((v / 1e6).toFixed(1)) + 'M';
   106→  if (Math.abs(v) >= 1e3) return toPersianDigits((v / 1e3).toFixed(1)) + 'K';
   107→  return toFa(v);
   108→}
   109→
   110→function pctDistance(from: number, to: number): number {
   111→  if (from === 0) return 0;
   112→  return ((to - from) / from) * 100;
   113→}
   114→
   115→function trendDirectionLabel(dir: string, bullColor: string, bearColor: string, neutralColor: string): { label: string; color: string } {
   116→  const d = dir?.toLowerCase() ?? '';
   117→  if (d.includes('up') || d === 'صعودی') return { label: 'صعودی', color: bullColor };
   118→  if (d.includes('down') || d === 'نزولی') return { label: 'نزولی', color: bearColor };
   119→  return { label: 'خنثی', color: neutralColor };
   120→}

   145→function ProgressBar({ value, max, color, height = 4 }: { value: number; max: number; color: string; height?: number }) {
   146→  const { colors: C } = useTheme();
   147→  const pct = Math.min(100, Math.max(0, (value / max) * 100));
   148→  return (
   149→    <div className="w-full rounded-full overflow-hidden" style={{ height, background: C.cardBorder }}>
   150→      <div className="h-full rounded-full transition-all duration-500" style={{ width: `${pct}%`, background: color }} />
   151→    </div>
   152→  );
   153→}
