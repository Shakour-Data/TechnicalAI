   248→   248→function addRecentSearch(name: string) {
   249→   249→  if (typeof window === 'undefined' || !name) return;
   250→   250→  try {
   251→   251→    let list = getRecentSearches();
   252→   252→    list = list.filter((r) => r !== name);
   253→   253→    list.unshift(name);
   254→   254→    if (list.length > MAX_RECENT) list = list.slice(0, MAX_RECENT);
   255→   255→    localStorage.setItem(RECENT_KEY, JSON.stringify(list));
   256→   256→  } catch { /* ignore */ }
   257→   257→}
