    34→            print(f'Please enter valid {key_word} date in YYYY-MM-DD format')
    35→    except:
    36→        if(len(date)==10):
    37→            print(f'Please enter valid {key_word} date')

    44→def get_tse_webid(stock:str = 'پترول') -> pd.DataFrame:
    45→    """
    46→    Takes ticker or firm's full name, does a live search in TSE new website and returns a multi-index Pandas dataframe that contains the following columns:
    47→    
    48→    Ticker: Symbol in the Tehran Stock Exchange.
    49→    Active: 1 shows the market in which the stock is currently trading. 
    50→    Name: firm's full name in the relevant market.
    51→    WebID: A numeric code that can be used for building request links and crawling the financial data of the given stock.
    52→    Market: Market name in Tehran Stock Exchange, markets the stock was traded in and is trading now (بورس، فرابورس، پایه زرد، پایه نارنجی، پایه قرمز).
    53→    
    54→    :param stock: (str) Ticker or firm's full name. 
    55→    :return: (pd.DataFrame) A dataframe that contains Ticker, Active, WebID, Name and Market columns for the requested stock.
    56→    """
    57→    
    58→    # basic search function: searches for and cleans the search results
    59→    def srch_req(srch_key):
    60→        srch_page = requests.get(f'http://cdn.tsetmc.com/api/Instrument/GetInstrumentSearch/{srch_key}', headers=headers)
    61→        srch_res = pd.DataFrame(srch_page.json()['instrumentSearch'])
    62→        srch_res = srch_res[['lVal18AFC','lVal30','insCode','lastDate','cgrValCot']]
    63→        srch_res.columns = ['Ticker','Name','WebID','Active','Market']
    64→        srch_res['Name'] = srch_res['Name'].apply(lambda x : characters.ar_to_fa(' '.join([i.strip() for i in x.split('\u200c')]).strip()))
    65→        srch_res['Ticker'] = srch_res['Ticker'].apply(lambda x : characters.ar_to_fa(''.join(x.split('\u200c')).strip()))
    66→        srch_res['NameSplit'] = srch_res['Name'].apply(lambda x : ''.join(x.split()).strip())
    67→        srch_res['SymbolSplit'] = srch_res['Ticker'].apply(lambda x : ''.join(x.split()).strip())
    68→        srch_res['Active'] = pd.to_numeric(srch_res['Active'])
    69→        srch_res = srch_res.sort_values('Ticker')
    70→        srch_res = pd.DataFrame(srch_res[['Name','WebID','NameSplit','SymbolSplit','Market']].values, columns=['Name','WebID',
    71→                                'NameSplit','SymbolSplit','Market'], index=pd.MultiIndex.from_frame(srch_res[['Ticker','Active']]))
    72→        return srch_res
    73→    
    74→    # checking function inputs
    75→    if type(stock) != str:
    76→        print('Please Enetr a Valid Ticker or Name!')
    77→        return False
    78→    
    79→    # special case that can not be found using ticker: convert ticker to full name!
    80→    if(stock=='آ س پ'):
    81→        stock = 'آ.س.پ'
    82→        
    83→    # generating search keys
    84→    stock = characters.ar_to_fa(''.join(stock.split('\u200c')).strip())
    85→    first_name = stock.split()[0]
    86→    stock = ''.join(stock.split())
    87→    
    88→    # start searching using keys, cleaning data, checking search results and handling special cases (ticker or full name)
    89→    data = srch_req(first_name)
    90→    df_symbol = data[data['SymbolSplit'] == stock]
    91→    df_name = data[data['NameSplit'] == stock]
    92→    
    93→    # matching search results with search key, cleaning the data and adding market data  
    94→    if len(df_symbol) > 0 :
    95→        df_symbol = df_symbol.sort_index(level=1,ascending=False).drop(['NameSplit','SymbolSplit'], axis=1)
    96→        df_symbol['Market'] = df_symbol['Market'].apply(lambda x: re.sub('[0-9]', '', x))
    97→        df_symbol['Market'] = df_symbol['Market'].map({'N':'بورس', 'Z':'فرابورس', 'D':'فرابورس', 'A':'پایه زرد', 'P':'پایه زرد', 'C':'پایه نارنجی', 'L':'پایه قرمز',
    98→                                                       'W':'کوچک و متوسط فرابورس', 'V':'کوچک و متوسط فرابورس',})
    99→        df_symbol['Market'] = df_symbol['Market'].fillna('نامعلوم')
   100→        return df_symbol
   101→    elif len(df_name) > 0 :
   102→        symbol = df_name.index[0][0]
   103→        data = srch_req(symbol)
   104→        symbol = characters.ar_to_fa(''.join(symbol.split('\u200c')).strip())
   105→        df_symbol = data[data.index.get_level_values('Ticker') == symbol]
   106→        if len(df_symbol) > 0 :
   107→            df_symbol = df_symbol.sort_index(level=1, ascending=False).drop(['NameSplit','SymbolSplit'], axis=1)
   108→            df_symbol['Market'] = df_symbol['Market'].apply(lambda x: re.sub('[0-9]', '', x))
   109→            df_symbol['Market'] = df_symbol['Market'].map({'N':'بورس', 'Z':'فرابورس', 'D':'فرابورس', 'A':'پایه زرد', 'P':'پایه زرد', 'C':'پایه نارنجی', 'L':'پایه قرمز',
   110→                                                           'W':'کوچک و متوسط فرابورس', 'V':'کوچک و متوسط فرابورس',})
   111→            df_symbol['Market'] = df_symbol['Market'].fillna('نامعلوم')
   112→            return df_symbol
   113→    
   114→    # invalid entry
   115→    print('Please Enetr a Valid Ticker or Name!')
   116→    
   117→    return False

   228→def get_price_history(stock:str = 'خودرو', 
   229→                      start_date:str = '1400-01-01', 
   230→                      end_date:str = '1401-01-01', 
   231→                      ignore_date:bool = False, 
   232→                      adjust_price:bool = False, 
   233→                      show_weekday:bool = False, 
   234→                      double_date:bool = False) -> pd.DataFrame:
   235→    
   236→    """
   237→    Takes ticker or firm's full name and returns a Pandas dataframe that contains the following columns:
   238→    
   239→    J-Date: Jalali date, as index
   240→    Date: Gregorian date
   241→    Weekday: Name of weekdays
   242→    Open: Opening price of day
   243→    High: Maximum price of day 
   244→    Low: Minimum price of day
   245→    Close: Closing price of day (آخرین قیمت)
   246→    Final: Weighted closing price of day (قیمت پایانی)
   247→    Volume: Traded volume of day
   248→    Value: Traded value of day in IRAN's Rial
   249→    No: Number of trades in a day
   250→    Ticker: Ticker/Symbol
   251→    Name: Firm's full name
   252→    Market: Market, the stock is traded in
   253→    Adj Open: Adjusted opening price of day for stock-splits and dividends
   254→    Adj High: Adjusted maximum price of day for stock-splits and dividends
   255→    Adj Low: Adjusted minimum price of day for stock-splits and dividends
   256→    Adj Close: Adjusted closing price of day for stock-splits and dividends
   257→    Adj Final: Adjusted weighted closing price of day for stock-splits and dividends
   258→    
   259→     * All data are taken from the new website of Tehran Stock Exchange.
   260→
   261→    :param stock: (str) Ticker or firm's full name. 
   262→    :param start_date: (str) Jalali date for starting day of historical price data in YYYY-MM-DD format.
   263→    :param end_date: (str) Jalali date for ending day of historical price data in YYYY-MM-DD format.
   264→    :param ignore_date: (bool) Ignores start_date and end_date and returns all available price history, if set to True.
   265→    :param adjust_price: (bool) Adjusts price for stock-splits and dividends, if set to True.
   266→    :param show_weekday: (bool) Shows weekdays in the output, if set to True.
   267→    :param double_date: (bool) Shows Gregorian date in the output, if set to True.
   268→    :return: (pd.DataFrame) A dataframe that contains J-Date as index and Date, Weekday, Open, High, Low, Close, Final, Volume, Value, No, Ticker, Name, Market, Adj Open, Adj High, Adj Low, Adj Close, Adj Final.
   269→    """
   270→
   271→    # basic request and data cleaning function for historical price data of a ticker for a given market 
   272→    def get_price_data(ticker_no, ticker, name, market):
   273→        r = requests.get(f'http://cdn.tsetmc.com/api/ClosingPrice/GetClosingPriceDailyList/{ticker_no}/0', headers=headers)
   274→        df_history = pd.DataFrame(r.json()['closingPriceDaily'])
   275→        columns=['Date','High','Low','Final','Close','Open','Y-Final','Value','Volume','No']
   276→        df_history = df_history[['dEven','priceMax','priceMin','pClosing','pDrCotVal','priceFirst','priceYesterday','qTotCap','qTotTran5J','zTotTran']]
   277→        df_history.columns = ['Date','High','Low','Final','Close','Open','Y-Final','Value','Volume','No']
   278→        df_history['Date'] = df_history['Date'].apply(lambda x: str(x))
   279→        df_history['Date'] = df_history['Date'].apply(lambda x: f'{x[:4]}-{x[4:6]}-{x[-2:]}')
   280→        df_history['Date']=pd.to_datetime(df_history['Date'])
   281→        df_history = df_history[df_history['No']!=0]
   282→        df_history['Ticker'] = ticker
   283→        df_history['Name'] = name
   284→        df_history['Market'] = market
   285→        df_history = df_history.set_index('Date')
   286→        return df_history
   287→    
   288→    # check to see if the entry start and end dates are valid or not
   289→    if(not ignore_date):
   290→        start_date = __Check_JDate_Validity__(start_date,key_word="'START'")
   291→        if(start_date==None):
   292→            return
   293→        end_date = __Check_JDate_Validity__(end_date,key_word="'END'")
   294→        if(end_date==None):
   295→            return
   296→        start = jdatetime.date(year=int(start_date.split('-')[0]), month=int(start_date.split('-')[1]), day=int(start_date.split('-')[2]))
   297→        end = jdatetime.date(year=int(end_date.split('-')[0]), month=int(end_date.split('-')[1]), day=int(end_date.split('-')[2]))
   298→        if(start>end):
   299→            print('Start date must be a day before end date!')
   300→            return
   301→    
   302→    # search for WebIDs
   303→    ticker_no_df = get_tse_webid(stock)
   304→    if(type(ticker_no_df)==bool):
   305→        return
   306→    
   307→    # create an empty dataframe:
   308→    df_history = pd.DataFrame({},columns=['Date','High','Low','Final','Close','Open','Y-Final','Value','Volume','No','Ticker','Name','Market']).set_index('Date')
   309→    
   310→    # loop to get data from different pages of a ticker:
   311→    for index, row in (ticker_no_df.reset_index()).iterrows():
   312→        try:
   313→            df_temp = get_price_data(ticker_no = row['WebID'],ticker = row['Ticker'],name = row['Name'],market = row['Market'])
   314→            df_history = pd.concat([df_history,df_temp])
   315→        except:
   316→            pass
   317→        
   318→    # sort based on dated index:
   319→    df_history = df_history.sort_index(ascending=True)
   320→    df_history = df_history.reset_index()
   321→    
   322→    # add weekdays and j-date columns:
   323→    df_history['Weekday']=df_history['Date'].dt.weekday
   324→    df_history['Weekday'] = df_history['Weekday'].apply(lambda x: calendar.day_name[x])
   325→    df_history['J-Date']=df_history['Date'].apply(lambda x: str(jdatetime.date.fromgregorian(date=x.date())))
   326→    df_history = df_history.set_index('J-Date')
   327→    
   328→    # rearrange columns and convert some columns to numeric
   329→    df_history=df_history[['Date','Weekday','Y-Final','Open','High','Low','Close','Final','Volume','Value','No','Ticker','Name','Market']]
   330→    cols = ['Y-Final','Open','High','Low','Close','Final','Volume','No','Value']
   331→    df_history[cols] = df_history[cols].apply(pd.to_numeric, axis=1)
   332→    
   333→
   334→    # find stock moves between markets and adjust for nominal price in the new market, if necessary
   335→    df_history['Final(+1)'] = df_history['Final'].shift(+1)          
   336→    df_history['Market(+1)'] = df_history['Market'].shift(+1)        
   337→    df_history['temp'] = df_history.apply(lambda x: x['Y-Final'] if((x['Y-Final']!=0)and(x['Y-Final']!=1000)) 
   338→                                          else (x['Y-Final'] if((x['Market(+1)']==x['Market'])or(pd.isnull(x['Final(+1)']))) 
   339→                                          else x['Final(+1)']),axis = 1)
   340→    df_history['Y-Final'] = df_history['temp']
   341→    df_history.drop(columns=['Final(+1)','temp','Market(+1)'],inplace=True)
   342→    
   343→    # convert numbers to int because we do not have less than Rial, just for clean outputs!
   344→    for col in cols:
   345→        df_history[col] = df_history[col].apply(lambda x: int(x))
   346→
   347→    # Adjust price data, if requested:

   807→def __Get_Day_IntradayTrades__(ticker_no, j_date):
   808→    #convert to desired Cristian data format
   809→    year, month, day = j_date.split('-')
   810→    date = jdatetime.date(int(year), int(month), int(day)).togregorian()
   811→    date = f'{date.year:04}{date.month:02}{date.day:02}'
   812→    # request and process
   813→    headers = {'User-Agent': 'Mozilla/5.0 (Macintosh; Intel Mac OS X 10_10_1) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/39.0.2171.95 Safari/537.36'}
   814→    page = requests.get(f'http://cdn.tsetmc.com/api/Trade/GetTradeHistory/{ticker_no}/{date}/false', headers=headers)
   815→    df_intraday = (pd.DataFrame(page.json()['tradeHistory'])).iloc[:,2:6]
   816→    df_intraday = df_intraday.sort_values(by='nTran')
   817→    df_intraday.drop(columns=['nTran'],inplace=True)
   818→    df_intraday.columns = ['Time','Volume','Price']
   819→    df_intraday['Time'] = df_intraday['Time'].astype('str').apply(lambda x: '0'+x[0]+':'+x[1:3]+':'+x[3:]  if len(x)==5 else x[:2]+':'+x[2:4]+':'+x[4:])
   820→    df_intraday['J-Date'] = j_date
   821→    df_intraday = df_intraday.set_index(['J-Date','Time'])
   822→    return df_intraday
