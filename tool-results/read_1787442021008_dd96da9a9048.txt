   181→// Component
   182→// ═══════════════════════════════════════════════════════════════════════════════
   183→
   184→export default function VdssGraph(props: VdssGraphProps) {
   185→  const { symbolName, currentPrice, resistances, supports, ma100, scenarios, rsi, adx, bullScore, edgeWeights, currencyUnit, overallSignal, ma21, atr, bollingerUpper, bollingerMiddle, bollingerLower } = props;
   186→
   187→  // ── Dynamic scenario labels (context-aware) ──────────────────
   188→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal || 'neutral'), [overallSignal]);
   189→
   190→  // ── V11 Scenario Engine ───────────────────────────────────────
   191→  const v11Result = useMemo(() => {
   192→    if (!atr || !currentPrice) return null;
   193→    return buildV11Scenarios({
   194→      currentPrice,
   195→      atr,
   196→      overallSignal: (overallSignal as 'bullish' | 'bearish' | 'neutral') || 'neutral',
   197→      v9Scenarios: scenarios,
   198→      supports,
   199→      resistances,
   200→      bollingerUpper,
   201→      bollingerLower,
   202→      bollingerMiddle,
   203→      ma21,
   204→      ma100,
   205→    });
   206→  }, [currentPrice, atr, overallSignal, scenarios, supports, resistances, bollingerUpper, bollingerLower, bollingerMiddle, ma21, ma100]);
   207→
   208→  // Helper: map R-key → S-key then get V11 scenario
   209→  const getV11ForRKey = useCallback((rKey: string) => {
   210→    if (!v11Result) return null;
   211→    const signal = (overallSignal || 'neutral') as 'bullish' | 'bearish' | 'neutral';
   212→    const idx = SCENARIO_KEYS.indexOf(rKey as any); // R1→0, R2→1, ... R9→8
   213→    const sKey = 'S' + (idx + 1);
   214→    return v11Result.scenarios[sKey] || null;
   215→  }, [v11Result, overallSignal]);
   216→
   217→  // ── Dynamic NODE_DEFS: R1-R5 titles/descriptions adapt to signal ──
   218→  const nodeDefs = useMemo(() => {
   219→    const base = { ...NODE_DEFS };
   220→    for (const key of SCENARIO_KEYS) {
   221→      if (base[key]) {
   222→        base[key] = { ...base[key], title: dynamicMeta[key].nodeTitle, desc: dynamicMeta[key].nodeDesc };
   223→      }
   224→    }
   225→    return base;
   226→  }, [dynamicMeta]);
   227→  const unit = currencyUnit || 'ریال';
   228→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   229→  const S1_level = supports[0] ?? currentPrice * 0.95;
   230→
   231→  const [activeFilter, setActiveFilter] = useState('all');
   232→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   233→  const [selectedPathIdx, setSelectedPathIdx] = useState<number | null>(null);
   234→  const graphRef = useRef<HTMLDivElement>(null);
   235→  const svgRef = useRef<SVGSVGElement>(null);
   236→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   237→  const shellRef = useRef<HTMLDivElement>(null);
   238→
   239→  // ── Edge probabilities (using TA engine's bullScore) ──────────────
   240→  const edgeProbs = useMemo(() =>
   241→    edgeWeights ? calcEdgeProbabilities(edgeWeights) : {},
   242→  [edgeWeights]
   243→  );
   244→
   245→  // ── All paths with raw probabilities ─────────────────────────────
   246→  const allPathsRaw = useMemo(() => {
   247→    if (Object.keys(edgeProbs).length === 0) return [];
   248→    return findAllPaths(edgeProbs);
   249→  }, [edgeProbs]);
   250→
   251→  // ── Calibrate path probabilities to match TA engine scenario probabilities ──
   252→  const allPaths = useMemo(() => {
   253→    const rawSums: Record<string, number> = {};
   254→    for (const key of SCENARIO_KEYS) rawSums[key] = 0;
   255→    for (const p of allPathsRaw) rawSums[p.target] += p.prob;
   256→
   257→    return allPathsRaw.map(p => {
   258→      const rawSum = rawSums[p.target] || 0.0001;
   259→      const targetProb = scenarios[p.target as keyof typeof scenarios].probability / 100;
   260→      const scale = targetProb / rawSum;
   261→      return { ...p, prob: p.prob * scale };
   262→    });
   263→  }, [allPathsRaw, scenarios]);
   264→
   265→  // ── Filtered paths (sorted by probability descending) ─────────────────
   266→  const filteredPaths = useMemo(() => {
   267→    if (activeFilter === 'all') return [...allPaths].sort((a, b) => b.prob - a.prob);
   268→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   269→      return allPaths.filter(p => p.target === activeFilter).sort((a, b) => b.prob - a.prob);
   270→    }
   271→    return allPaths.filter(p =>
   272→      p.edges.every(ei => EDGES[ei][3] === activeFilter)
   273→    ).sort((a, b) => b.prob - a.prob);
   274→  }, [allPaths, activeFilter]);
   275→
   276→  // ── Visible edges/nodes based on filter (or selected path) ──────────────
   277→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   278→    if (selectedPathIdx !== null && filteredPaths[selectedPathIdx]) {
   279→      const p = filteredPaths[selectedPathIdx];
   280→      return {
   281→        visibleEdgeIndices: p.edges,
   282→        visibleNodes: new Set([...p.nodes, 'A']),
   283→      };
   284→    }
   285→    if (activeFilter === 'all') {
   286→      const allEdgeIdx = EDGES.map((_, i) => i);
   287→      const allNodes = new Set(Object.keys(EXPANDED_POSITIONS));
   288→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   289→    }
   290→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   291→      const edgeSet = new Set<number>();
   292→      const nodeSet = new Set<string>();
   293→      for (const p of filteredPaths) {
   294→        p.edges.forEach(ei => edgeSet.add(ei));
   295→        p.nodes.forEach(n => nodeSet.add(n));
   296→      }
   297→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   298→    }
   299→    const edgeSet = new Set<number>();
   300→    const nodeSet = new Set<string>();
   301→    EDGES.forEach((e, i) => {
   302→      if (e[3] === activeFilter) {
   303→        edgeSet.add(i);
   304→        nodeSet.add(e[0]);
   305→        nodeSet.add(e[1]);
   306→      }
   307→    });
   308→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   309→  }, [activeFilter, filteredPaths, selectedPathIdx]);
   310→
   311→  // ── Node values ──────────────────────────────────────────────────
   312→  const nodeValues: Record<string, string> = {
   313→    A: toFa(currentPrice),
   314→    B: toFa(R1_level),
   315→    D: toFa(resistances[1] ?? currentPrice * 1.10),
   316→    E: `${toFa(Math.round(R1_level + (resistances[1] ?? R1_level * 1.05 - R1_level) * 0.5))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 0.8))}`,
   317→    F: `${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 1.2))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 2.0))}`,
   318→    G: toFa(S1_level),
   319→    H: toFa(supports[1] ?? currentPrice * 0.90),
   320→    I: toFa(supports[2] ?? currentPrice * 0.85),
   321→    J: toFa(supports[3] ?? currentPrice * 0.80),
   322→    K: toFa(ma100),
   323→    L: props.trendDirection === 'up' ? 'صعودی' : props.trendDirection === 'down' ? 'نزولی' : 'خنثی',
   324→  };
   325→  for (const key of SCENARIO_KEYS) {
   326→    const s = scenarios[key];
   327→    nodeValues[key] = `${toFa(s.targetMin)} — ${toFa(s.targetMax)} ${unit}`;
   328→  }
   329→
   330→  // ── Draw SVG edges ───────────────────────────────────────────────
   331→  const drawEdges = useCallback(() => {
   332→    const svg = svgRef.current;
   333→    const graph = graphRef.current;
   334→    if (!svg || !graph) return;
   335→
   336→    let markersSvg = '';
   337→    for (const [k, c] of Object.entries(EDGE_COLORS)) {
   338→      markersSvg += `<marker id="arrow-${k}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   339→    }
   340→
   341→    let pathsSvg = '';
   342→    const graphRect = graph.getBoundingClientRect();
   343→
   344→    for (let ei = 0; ei < EDGES.length; ei++) {
   345→      const [fromId, toId, label, type] = EDGES[ei];
   346→      const isVisible = visibleEdgeIndices.includes(ei);
   347→      const fromEl = nodeRefs.current[fromId];
   348→      const toEl = nodeRefs.current[toId];
   349→      if (!fromEl || !toEl) continue;
   350→
   351→      const fRect = fromEl.getBoundingClientRect();
   352→      const tRect = toEl.getBoundingClientRect();
   353→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   354→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   355→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   356→      const by = tRect.top - graphRect.top + tRect.height / 2;
   357→      const dx = bx - ax;
   358→      const dy = by - ay;
   359→      const dist = Math.sqrt(dx * dx + dy * dy);
   360→      if (dist === 0) continue;
   361→
   362→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   363→      const mx = (ax + bx) / 2;
   364→      const my = (ay + by) / 2;
   365→      const nx = -dy / dist;
   366→      const ny = dx / dist;
   367→      const cx = mx + nx * bend;
   368→      const cy = my + ny * bend;
   369→
   370→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   371→      const prob = edgeProbs[String(ei)] ?? 0;
   372→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   373→
   374→      pathsSvg += `<path d="${d}" stroke="${EDGE_COLORS[type]}" stroke-width="${isVisible ? 2 : 1}" opacity="${isVisible ? 0.72 : 0.06}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   375→
   376→      if (isVisible) {
   377→        pathsSvg += `<text x="${cx}" y="${cy - 6}" fill="#374151" font-size="9" text-anchor="middle" paint-order="stroke" stroke="#f3f4f6" stroke-width="3.5" stroke-linejoin="round" opacity="0.85" data-type="${type}" class="edge-label">${label}</text>`;
   378→        pathsSvg += `<text x="${cx}" y="${cy + 8}" fill="${EDGE_COLORS[type]}" font-size="10" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#f3f4f6" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   379→      }

   768→                    const meta = dynamicMeta[p.target];
   769→                    const isSelected = selectedPathIdx === i;
   770→                    return (
   771→                      <div
   772→                        key={i}
   773→                        onClick={() => setSelectedPathIdx(isSelected ? null : i)}
   774→                        className={`flex items-center gap-2 text-[10px] px-2 py-1.5 rounded-lg cursor-pointer transition-all ${isSelected ? 'ring-1 shadow-sm' : 'hover:bg-gray-50'}`}
   775→                        style={isSelected ? { background: meta.color + '08', outline: `1px solid ${meta.color}60` } : undefined}
   776→                      >
   777→                        <span className="w-4 text-center font-bold" style={{ color: meta.color }}>{toPersianDigits(i + 1)}</span>
   778→                        <span className="flex-1 text-gray-500 truncate" dir="ltr">
   779→                          {p.nodes.join(' → ')}
   780→                        </span>
   781→                        <span className="font-bold min-w-[40px] text-left" style={{ color: meta.color }}>
   782→                          {toPersianDigits((p.prob * 100).toFixed(1))}٪
   783→                        </span>
   784→                      </div>
   785→                    );
   786→                  })}
   787→                  {filteredPaths.length > 10 && (
   788→                    <div className="text-[10px] text-gray-400 text-center pt-1">
   789→                      و {toPersianDigits(filteredPaths.length - 10)} مسیر دیگر...
   790→                    </div>
   791→                  )}
   792→                </div>
   793→              </div>
   794→
   795→              <div className="border-t border-gray-300 pt-3 mt-3">
   796→                <p className="text-[10px] text-gray-500 leading-relaxed">
   797→                  <b>قاعده:</b> عبور یا شکست هر سطح فقط با تثبیت معتبر تلقی می‌شود.<br />
   798→                  <b>احتمال یال:</b> حاصل توزیع شرطی بر اساس سیگنال‌های تکنیکال.<br />
   799→                  <b>احتمال مسیر:</b> حاصل‌ضرب احتمال یال‌های مسیر.
   800→                </p>
   801→              </div>
   802→            </div>
   803→          )}
   804→        </div>
   805→      </div>
   806→
   807→      {/* ═══ Scenario Result Cards (Enhanced with V11) ═══ */}
   808→      <div className="mt-4 p-4 rounded-2xl border border-gray-300 bg-gray-50">
   809→        <h2 className="text-sm font-bold text-gray-900 mb-3">گره‌های نتیجه و احتمال تجمیعی مسیرهای ورودی</h2>
   810→        <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-3 gap-3">
   811→          {SCENARIO_KEYS.map((key, idx) => {
   812→            const s = scenarios[key];
   813→            const meta = dynamicMeta[key];
   814→            const sKey = rToS(key);
   815→            const v11Sc = v11Result?.scenarios[sKey];
   816→            const directionInfo = v11Sc ? DIRECTION_LABELS[v11Sc.direction] : null;
   817→            const isDominant = v11Result?.dominantKey === sKey;
   818→            const isHighestCum = v11Result?.highestCumulativeKey === sKey;
   819→            const sColor = v11Result ? (V11_S_COLORS[sKey] || meta.color) : meta.color;
   820→
   821→            return (
   822→              <div
   823→                key={key}
   824→                className="rounded-xl p-3"
   825→                style={{
   826→                  '--scolor': sColor,
   827→                  border: `1px solid color-mix(in srgb, ${sColor} 40%, transparent)`,
   828→                  background: `linear-gradient(160deg, color-mix(in srgb, ${sColor} 12%, transparent), rgba(249,250,251,.65))`,
   829→                } as React.CSSProperties}
   830→              >
   831→                <div className="flex items-center justify-between mb-1">
   832→                  <div className="flex items-center gap-1.5">
   833→                    <strong className="block text-xl font-black" style={{ color: sColor }}>{toFa(s.probability)}٪</strong>
   834→                    {directionInfo && (
   835→                      <span className="inline-block px-1.5 py-0.5 rounded text-[10px] font-bold" style={{ color: directionInfo.color }}>{directionInfo.icon}</span>
   836→                    )}
   837→                  </div>
   838→                  <span className="inline-flex items-center justify-center w-5 h-5 rounded-full text-[10px] font-bold border" style={{
   839→                    borderColor: sColor + '60',
   840→                    color: sColor,
   841→                    background: sColor + '12',
   842→                  }}>{toPersianDigits(idx + 1)}</span>
   843→                </div>
   844→                <span className="text-xs font-bold text-gray-600">{"سناریوی " + toPersianDigits(idx + 1)} | {meta.label}</span>
   845→
   846→                {/* Individual + Cumulative probability */}
   847→                {v11Sc ? (
   848→                  <div className="text-[10px] text-gray-500 mt-1.5">
   849→                    فردی: {toPersianDigits(v11Sc.probability.toFixed(1))}٪ | تجمعی: {toPersianDigits(v11Sc.cumulativeProbability.toFixed(1))}٪
   850→                  </div>
   851→                ) : null}
   852→
   853→                {/* Dominant / Highest Cumulative markers */}
   854→                {(isDominant || isHighestCum) && (
   855→                  <div className="flex items-center gap-1 mt-1.5">
   856→                    {isDominant && (
   857→                      <span className="inline-block px-1.5 py-0.5 rounded text-[8px] font-bold bg-purple-100 text-purple-700 border border-purple-300">غالب</span>
   858→                    )}
   859→                    {isHighestCum && (
   860→                      <span className="inline-block px-1.5 py-0.5 rounded text-[8px] font-bold bg-purple-100 text-purple-700 border border-purple-300">بیشترین تجمعی</span>
   861→                    )}
   862→                  </div>
   863→                )}
   864→
   865→                <small className="block text-[10px] text-gray-500 leading-relaxed mt-2" dir="ltr">
   866→                  {toFa(s.targetMin)} — {toFa(s.targetMax)} {unit}
   867→                </small>
   868→              </div>
   869→            );
   870→          })}
   871→        </div>
   872→        <div className="mt-3 px-4 py-2.5 rounded-lg border-r-3 border-amber-200 bg-amber-50 text-[11px] text-gray-500 leading-relaxed">
   873→          <b>محدودیت مدل:</b> احتمال‌های سناریو توسط موتور محاسباتی بر اساس فرمول‌های تعریف‌شده (ADX, RSI, MFI, CCI و فاصله از سطوح) محاسبه شده‌اند. احتمال مسیرها از حاصل‌ضرب احتمال شرطی یال‌ها به دست می‌آید.
   874→        </div>
   875→      </div>
   876→    </div>
   877→  );
   878→}
