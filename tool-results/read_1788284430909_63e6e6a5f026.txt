    40→function renderAIText(text: string): string {
    41→  // Remove CJK (Chinese/Japanese/Korean) characters — prohibited in output
    42→  const noCjk = text.replace(/[\u4e00-\u9fff\u3400-\u4dbf\u3040-\u309f\u30a0-\u30ff\uac00-\ud7af\uff00-\uffef]/g, '');
    43→
    44→  // Normalize zero-width and look-alike characters that may interfere with regex
    45→  const normalized = noCjk.replace(/[\u200c\u200d\u200b\ufeff]/g, '');
    46→
    47→  // Color syntax: {color:hex}text{/color} or {color:named}text{/color} → <span>
    48→  const colorMap: Record<string, string> = {
    49→    'red': '#dc2626', 'red-600': '#dc2626', 'red-700': '#b91c1c',
    50→    'green': '#16a34a', 'emerald': '#059669', 'emerald-600': '#059669', 'emerald-700': '#047857',
    51→    'amber': '#d97706', 'amber-600': '#d97706', 'amber-700': '#b45309', 'amber-800': '#92400e',
    52→    'blue': '#2563eb', 'blue-600': '#2563eb', 'blue-700': '#1d4ed8',
    53→    'orange': '#ea580c',
    54→    'purple': '#9333ea', 'purple-600': '#9333ea', 'purple-700': '#7e22ce',
    55→  };
    56→
    57→  const withColors = normalized.replace(/\{color:([^}]+)\}([\s\S]*?)\{\/color\}/g, (_match, colorName: string, inner: string) => {
    58→    const hex = colorMap[colorName.trim()] || (colorName.startsWith('#') ? colorName : null);
    59→    if (!hex) return inner; // Unknown color: just return text without color
    60→    return `<span style="color:${hex}">${inner}</span>`;
    61→  });
    62→
    63→  return withColors
    64→    .split('\n\n')
    65→    .map(p => {
    66→      // Convert **bold** to <strong>
    67→      const html = p.replace(/\*\*(.+?)\*\*/g, '<strong class="font-bold text-gray-900">$1</strong>');
    68→      // Convert newlines within paragraph to <br>
    69→      return `<p class="mb-4">${html.replace(/\n/g, '<br>')}</p>`;
    70→    })
    71→    .join('');
    72→}

   817→      localStorage.setItem(`ai-${sym}-latest`, JSON.stringify({ text, date: new Date().toISOString().slice(0, 10), ts: Date.now() }));
   818→    } catch { /* quota exceeded — ignore */ }
   819→  }
