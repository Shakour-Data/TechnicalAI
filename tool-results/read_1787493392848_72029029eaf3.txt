   167→   167→// Graph Algorithms
   168→   168→// ═══════════════════════════════════════════════════════════════════════════════
   169→   169→
   170→   170→interface EdgeInfo {
   171→   171→  from: string;
   172→   172→  to: string;
   173→   173→  label: string;
   174→   174→  type: string;
   175→   175→  prob: number;
   176→   176→}
   177→   177→
   178→   178→interface PathInfo {
   179→   179→  nodes: string[];
   180→   180→  edges: number[];
   181→   181→  prob: number;
   182→   182→  target: string;
   183→   183→}
   184→   184→
   185→   185→// Calculate edge probabilities based on market conditions
   186→   186→function calcEdgeProbabilities(
   187→   187→  bullScore: number, adxVal: number
   188→   188→): Record<string, number> {
   189→   189→  const adxW = Math.min(adxVal / 100, 1);
   190→   190→  const typeWeights: Record<string, number> = {
   191→   191→    up: bullScore * 0.6 + adxW * 0.4,
   192→   192→    pullback: 0.3,
   193→   193→    down: (1 - bullScore) * 0.6 + adxW * 0.4,
   194→   194→    risk: 0.15 * (1.2 - adxW),
   195→   195→  };
   196→   196→
   197→   197→  // Build adjacency: for each node, group outgoing edges by type
   198→   198→  const adj: Record<string, { idx: number; type: string }[]> = {};
   199→   199→  EDGES.forEach((e, i) => {
   200→   200→    if (!adj[e[0]]) adj[e[0]] = [];
   201→   201→    adj[e[0]].push({ idx: i, type: e[3] });
   202→   202→  });
   203→   203→
   204→   204→  const edgeProbs: Record<string, number> = {};
   205→   205→
   206→   206→  for (const [node, outEdges] of Object.entries(adj)) {
   207→   207→    // Sum weights of outgoing edges
   208→   208→    const totalWeight = outEdges.reduce((sum, e) => sum + (typeWeights[e.type] ?? 0.1), 0);
   209→   209→    if (totalWeight === 0) {
   210→   210→      outEdges.forEach(e => { edgeProbs[String(e.idx)] = 1 / outEdges.length; });
   211→   211→      continue;
   212→   212→    }
   213→   213→    outEdges.forEach(e => {
   214→   214→      edgeProbs[String(e.idx)] = (typeWeights[e.type] ?? 0.1) / totalWeight;
   215→   215→    });
   216→   216→  }
   217→   217→
   218→   218→  return edgeProbs;
   219→   219→}
   220→   220→
   221→   221→// Find all paths from 'A' to terminal nodes using DFS
   222→   222→function findAllPaths(edgeProbs: Record<string, number>): PathInfo[] {
   223→   223→  const adj: Record<string, { to: string; edgeIdx: number }[]> = {};
   224→   224→  EDGES.forEach((e, i) => {
   225→   225→    if (!adj[e[0]]) adj[e[0]] = [];
   226→   226→    adj[e[0]].push({ to: e[1], edgeIdx: i });
   227→   227→  });
   228→   228→
   229→   229→  const paths: PathInfo[] = [];
   230→   230→  const MAX_PATHS = 200;
   231→   231→
   232→   232→  function dfs(node: string, visited: Set<string>, currentPath: string[], edgeIndices: number[], currentProb: number) {
   233→   233→    if (SCENARIO_KEYS.includes(node as any)) {
   234→   234→      paths.push({ nodes: [...currentPath, node], edges: [...edgeIndices], prob: currentProb, target: node });
   235→   235→      return;
   236→   236→    }
   237→   237→    if (paths.length >= MAX_PATHS) return;
   238→   238→    if (visited.has(node)) return;
   239→   239→    visited.add(node);
   240→   240→
   241→   241→    const outs = adj[node] ?? [];
   242→   242→    for (const { to, edgeIdx } of outs) {
   243→   243→      const ep = edgeProbs[String(edgeIdx)] ?? 0.1;
   244→   244→      dfs(to, visited, [...currentPath, node], [...edgeIndices, edgeIdx], currentProb * ep);
   245→   245→    }
   246→   246→    visited.delete(node);
   247→   247→  }
   248→   248→
   249→   249→  dfs('A', new Set(), [], [], 1);
   250→   250→  return paths.sort((a, b) => b.prob - a.prob);
   251→   251→}

   259→   259→  const R1_level = resistances[0] ?? currentPrice * 1.05;

   261→   261→

   269→   269→
   270→   270→  // ── Edge probabilities (using TA engine's bullScore) ──────────────
   271→   271→  const edgeProbs = useMemo(() =>
   272→   272→    calcEdgeProbabilities(bullScore, adx),
   273→   273→  [bullScore, adx]
   274→   274→  );
   275→   275→
   276→   276→  // ── All paths with raw probabilities ─────────────────────────────
   277→   277→  const allPathsRaw = useMemo(() => findAllPaths(edgeProbs), [edgeProbs]);
   278→   278→
   279→   279→  // ── Calibrate path probabilities to match TA engine scenario probabilities ──
   280→   280→  const allPaths = useMemo(() => {
   281→   281→    // Compute raw sum per target
   282→   282→    const rawSums: Record<string, number> = {};
   283→   283→    for (const key of SCENARIO_KEYS) rawSums[key] = 0;
   284→   284→    for (const p of allPathsRaw) rawSums[p.target] += p.prob;
   285→   285→
   286→   286→    // Scale each path so the sum per target matches scenario probability
   287→   287→    return allPathsRaw.map(p => {
   288→   288→      const rawSum = rawSums[p.target] || 0.0001;
   289→   289→      const s = scenarios[p.target as keyof typeof scenarios];
   290→   290→      const targetProb = (s?.probability ?? 0) / 100;
   291→   291→      const scale = targetProb / rawSum;
   292→   292→      return { ...p, prob: p.prob * scale };
   293→   293→    });
   294→   294→  }, [allPathsRaw, scenarios]);
   295→   295→
   296→   296→  // ── Filtered paths ────────────────────────────────────────────────
   297→   297→  const filteredPaths = useMemo(() => {
   298→   298→    if (activeFilter === 'all') return allPaths;
   299→   299→    // Check if filter is a scenario key
   300→   300→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   301→   301→      return allPaths.filter(p => p.target === activeFilter);
   302→   302→    }
   303→   303→    // Type filter: paths must use only edges of this type
   304→   304→    return allPaths.filter(p =>
   305→   305→      p.edges.every(ei => EDGES[ei][3] === activeFilter)
   306→   306→    );
   307→   307→  }, [allPaths, activeFilter]);
   308→   308→
   309→   309→  // ── Top 10 paths (sorted by probability, from allPaths not filtered) ──
