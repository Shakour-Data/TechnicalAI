   942→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   943→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   944→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   945→  grade?: string; overlapCount?: number; touchCount?: number;
   946→}) {
   947→  const isSupport = type === 'support';
   948→  const isTargetType = type === 'target';
   949→
   950→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   951→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   952→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   953→
   954→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   955→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   956→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   957→
   958→  return (
   959→    <div
   960→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   961→      style={{
   962→        background: bgAccent,
   963→        border: `1px solid ${borderAccent}`,
   964→        borderRight: `3px solid ${accentColor}`,
   965→      }}
   966→    >
   967→      {/* Label + Price */}
   968→      <div className="flex-1 min-w-0">
   969→        <div className="flex items-center gap-2">
   970→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   971→          {isTarget && <span className="text-[9px]">🎯</span>}
   972→          {overlapCount !== undefined && overlapCount > 0 && (
   973→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   974→              {toPersianDigits(overlapCount)} هم‌پوشانی
   975→            </span>
   976→          )}
   977→        </div>
   978→        <div className="flex items-baseline gap-1">
   979→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   980→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   981→        </div>
   982→      </div>
   983→
   984→      {/* Strength + Grade */}
   985→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   986→        <div className="flex items-center gap-1.5">
   987→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   988→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   989→          <span className="text-[9px] text-gray-500">/۱۰</span>
   990→        </div>
   991→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   992→          <div
   993→            className="h-full rounded-full transition-all duration-500"
   994→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   995→          />
   996→        </div>
   997→        {touchCount !== undefined && touchCount > 0 && (
   998→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   999→        )}
  1000→      </div>
  1001→    </div>
  1002→  );
  1003→}
