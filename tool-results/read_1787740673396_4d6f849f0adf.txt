    61→// ─── Helpers ───────────────────────────────────────────────────────────────────
    62→
    63→const toFa = (n: number) => Math.round(n).toLocaleString('fa-IR');
    64→const toFa1 = (n: number) => n.toFixed(1).replace(/\./g, '/').replace(/-/g, '−');
    65→const toFa2 = (n: number) => n.toFixed(2).replace(/\./g, '/').replace(/-/g, '−');
    66→
    67→function rsiSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    68→  if (v > 70) return 'bearish';
    69→  if (v < 30) return 'bullish';
    70→  return 'neutral';
    71→}
    72→
    73→function macdSignal(h: number): 'bullish' | 'bearish' | 'neutral' {
    74→  if (h > 0) return 'bullish';
    75→  if (h < 0) return 'bearish';
    76→  return 'neutral';
    77→}
    78→
    79→function adxSignal(v: number): 'strong' | 'neutral' {
    80→  if (v > 25) return 'strong';
    81→  return 'neutral';
    82→}
    83→
    84→function stochSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    85→  if (v > 80) return 'bearish';
    86→  if (v < 20) return 'bullish';
    87→  return 'neutral';
    88→}
    89→
    90→function signalDotColor(signal: 'bullish' | 'bearish' | 'neutral' | 'strong'): string {
    91→  switch (signal) {
    92→    case 'bullish': return 'bg-emerald-500';
    93→    case 'bearish': return 'bg-red-500';
    94→    case 'strong': return 'bg-amber-500';
    95→    default: return 'bg-[#3d5575]';
    96→  }
    97→}
    98→
    99→function gradeLabel(grade: string): string {
   100→  switch (grade) {
   101→    case 'Very Strong': return 'بسیار قوی';
   102→    case 'Strong': return 'قوی';
   103→    case 'Moderate': return 'متوسط';
   104→    default: return 'ضعیف';
   105→  }
   106→}
   107→
   108→function gradeColorClass(grade: string): string {
   109→  switch (grade) {
   110→    case 'Very Strong': return 'bg-red-500/20 text-red-400';
   111→    case 'Strong': return 'bg-amber-500/20 text-amber-400';
   112→    case 'Moderate': return 'bg-sky-500/20 text-sky-400';
   113→    default: return 'bg-white/10 text-[#5a7395]';
   114→  }
   115→}
   116→
   117→function formatNumber(n: number): string {
   118→  if (Math.abs(n) >= 1e9) return (n / 1e9).toFixed(1) + 'B';
   119→  if (Math.abs(n) >= 1e6) return (n / 1e6).toFixed(1) + 'M';
   120→  if (Math.abs(n) >= 1e3) return (n / 1e3).toFixed(1) + 'K';
   121→  return toFa(n);
   122→}
   123→
   124→function pctDistance(from: number, to: number): number {
   125→  if (from === 0) return 0;
   126→  return ((to - from) / from) * 100;
   127→}
   128→
   129→function trendDirectionLabel(dir: string): { label: string; color: string } {
   130→  const d = dir?.toLowerCase() ?? '';
   131→  if (d.includes('up') || d === 'صعودی') return { label: 'صعودی', color: C.bull };
   132→  if (d.includes('down') || d === 'نزولی') return { label: 'نزولی', color: C.bear };
   133→  return { label: 'خنثی', color: C.neutral };
   134→}

   146→function GlassCard({ children, className = '', style }: { children: React.ReactNode; className?: string; style?: React.CSSProperties }) {

   153→    </div>
   154→  );
   155→}

   157→function ProgressBar({ value, max, color, height = 4 }: { value: number; max: number; color: string; height?: number }) {
   158→  const pct = Math.min(100, Math.max(0, (value / max) * 100));
   159→  return (
   160→    <div className="w-full rounded-full overflow-hidden" style={{ height, background: 'rgba(255,255,255,0.06)' }}>
   161→      <div className="h-full rounded-full transition-all duration-500" style={{ width: `${pct}%`, background: color }} />
   162→    </div>
   163→  );
   164→}
