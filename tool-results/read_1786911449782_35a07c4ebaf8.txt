   142→// Graph Algorithms
   143→// ═══════════════════════════════════════════════════════════════════════════════
   144→
   145→interface EdgeInfo {
   146→  from: string;
   147→  to: string;
   148→  label: string;
   149→  type: string;
   150→  prob: number;
   151→}
   152→
   153→interface PathInfo {
   154→  nodes: string[];
   155→  edges: number[];
   156→  prob: number;
   157→  target: string;
   158→}
   159→
   160→// Calculate edge probabilities based on market conditions
   161→function calcEdgeProbabilities(
   162→  bullScore: number, adxVal: number
   163→): Record<string, number> {
   164→  const adxW = Math.min(adxVal / 100, 1);
   165→  const typeWeights: Record<string, number> = {
   166→    up: bullScore * 0.6 + adxW * 0.4,
   167→    pullback: 0.3,
   168→    down: (1 - bullScore) * 0.6 + adxW * 0.4,
   169→    risk: 0.15 * (1.2 - adxW),
   170→  };
   171→
   172→  // Build adjacency: for each node, group outgoing edges by type
   173→  const adj: Record<string, { idx: number; type: string }[]> = {};
   174→  EDGES.forEach((e, i) => {
   175→    if (!adj[e[0]]) adj[e[0]] = [];
   176→    adj[e[0]].push({ idx: i, type: e[3] });
   177→  });
   178→
   179→  const edgeProbs: Record<string, number> = {};
   180→
   181→  for (const [node, outEdges] of Object.entries(adj)) {
   182→    // Sum weights of outgoing edges
   183→    const totalWeight = outEdges.reduce((sum, e) => sum + (typeWeights[e.type] ?? 0.1), 0);
   184→    if (totalWeight === 0) {
   185→      outEdges.forEach(e => { edgeProbs[String(e.idx)] = 1 / outEdges.length; });
   186→      continue;
   187→    }
   188→    outEdges.forEach(e => {
   189→      edgeProbs[String(e.idx)] = (typeWeights[e.type] ?? 0.1) / totalWeight;
   190→    });
   191→  }
   192→
   193→  return edgeProbs;
   194→}
   195→
   196→// Find all paths from 'A' to terminal nodes using DFS
   197→function findAllPaths(edgeProbs: Record<string, number>): PathInfo[] {
   198→  const adj: Record<string, { to: string; edgeIdx: number }[]> = {};
   199→  EDGES.forEach((e, i) => {
   200→    if (!adj[e[0]]) adj[e[0]] = [];
   201→    adj[e[0]].push({ to: e[1], edgeIdx: i });
   202→  });
   203→
   204→  const paths: PathInfo[] = [];
   205→  const MAX_PATHS = 200;
   206→
   207→  function dfs(node: string, visited: Set<string>, currentPath: string[], edgeIndices: number[], currentProb: number) {
   208→    if (SCENARIO_KEYS.includes(node as any)) {
   209→      paths.push({ nodes: [...currentPath, node], edges: [...edgeIndices], prob: currentProb, target: node });
   210→      return;
   211→    }
   212→    if (paths.length >= MAX_PATHS) return;
   213→    if (visited.has(node)) return;
   214→    visited.add(node);
   215→
   216→    const outs = adj[node] ?? [];
   217→    for (const { to, edgeIdx } of outs) {
   218→      const ep = edgeProbs[String(edgeIdx)] ?? 0.1;
   219→      dfs(to, visited, [...currentPath, node], [...edgeIndices, edgeIdx], currentProb * ep);
   220→    }
   221→    visited.delete(node);
   222→  }
   223→
   224→  dfs('A', new Set(), [], [], 1);
   225→  return paths.sort((a, b) => b.prob - a.prob);
   226→}

   362→      const ny = dx / dist;
