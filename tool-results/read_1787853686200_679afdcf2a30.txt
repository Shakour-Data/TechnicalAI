    40→function renderAIText(text: string): string {
    41→  // Normalize zero-width and look-alike characters that may interfere with regex
    42→  const normalized = text.replace(/[\u200c\u200d\u200b\ufeff]/g, '');
    43→
    44→  // Color syntax: {color:hex}text{/color} or {color:named}text{/color} → <span>
    45→  const colorMap: Record<string, string> = {
    46→    'red': '#dc2626', 'red-600': '#dc2626', 'red-700': '#b91c1c',
    47→    'green': '#16a34a', 'emerald': '#059669', 'emerald-600': '#059669', 'emerald-700': '#047857',
    48→    'amber': '#d97706', 'amber-600': '#d97706', 'amber-700': '#b45309', 'amber-800': '#92400e',
    49→    'blue': '#2563eb', 'blue-600': '#2563eb', 'blue-700': '#1d4ed8',
    50→    'orange': '#ea580c',
    51→    'purple': '#9333ea', 'purple-600': '#9333ea', 'purple-700': '#7e22ce',
    52→  };
    53→
    54→  const withColors = normalized.replace(/\{color:([^}]+)\}([\s\S]*?)\{\/color\}/g, (_match, colorName: string, inner: string) => {
    55→    const hex = colorMap[colorName.trim()] || (colorName.startsWith('#') ? colorName : null);
    56→    if (!hex) return inner; // Unknown color: just return text without color
    57→    return `<span style="color:${hex}">${inner}</span>`;
    58→  });
    59→
    60→  return withColors
    61→    .split('\n\n')
    62→    .map(p => {
    63→      // Convert **bold** to <strong>
    64→      const html = p.replace(/\*\*(.+?)\*\*/g, '<strong class="font-bold text-gray-900">$1</strong>');
    65→      // Convert newlines within paragraph to <br>
    66→      return `<p class="mb-4">${html.replace(/\n/g, '<br>')}</p>`;
    67→    })
    68→    .join('');
    69→}
