   762→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   763→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   764→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   765→  grade?: string; overlapCount?: number; touchCount?: number;
   766→}) {
   767→  const isSupport = type === 'support';
   768→  const isTargetType = type === 'target';
   769→
   770→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   771→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   772→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   773→
   774→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   775→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   776→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   777→
   778→  return (
   779→    <div
   780→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   781→      style={{
   782→        background: bgAccent,
   783→        border: `1px solid ${borderAccent}`,
   784→        borderRight: `3px solid ${accentColor}`,
   785→      }}
   786→    >
   787→      {/* Label + Price */}
   788→      <div className="flex-1 min-w-0">
   789→        <div className="flex items-center gap-2">
   790→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   791→          {isTarget && <span className="text-[9px]">🎯</span>}
   792→          {overlapCount !== undefined && overlapCount > 0 && (
   793→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   794→              {toPersianDigits(overlapCount)} هم‌پوشانی
   795→            </span>
   796→          )}
   797→        </div>
   798→        <div className="flex items-baseline gap-1">
   799→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   800→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   801→        </div>
   802→      </div>
   803→
   804→      {/* Strength + Grade */}
   805→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   806→        <div className="flex items-center gap-1.5">
   807→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   808→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   809→          <span className="text-[9px] text-gray-500">/۱۰</span>
   810→        </div>
   811→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   812→          <div
   813→            className="h-full rounded-full transition-all duration-500"
   814→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   815→          />
   816→        </div>
   817→        {touchCount !== undefined && touchCount > 0 && (
   818→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   819→        )}
   820→      </div>
   821→    </div>
   822→  );
   823→}
