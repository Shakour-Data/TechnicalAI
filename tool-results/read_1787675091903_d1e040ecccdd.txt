    39→    39→def gregorian_to_jalali(gy, gm, gd):
    40→    40→    """Convert Gregorian to Jalali date string using jdatetime."""
    41→    41→    try:
    42→    42→        jdate = jdatetime.date.fromgregorian(year=gy, month=gm, day=gd)
    43→    43→        return f"{jdate.year}/{str(jdate.month).zfill(2)}/{str(jdate.day).zfill(2)}"
    44→    44→    except:
    45→    45→        return f"{gy}/{str(gm).zfill(2)}/{str(gd).zfill(2)}"

    48→    48→# ── z-ai CLI page_reader wrapper ─────────────────────────────────
    49→    49→def zai_page_reader(url: str, timeout: int = 120) -> str:
    50→    50→    """Fetch URL content using z-ai CLI page_reader. Retries up to 3 times."""
    51→    51→    tmp = f'/tmp/zai_{hash(url) % 100000}_{int(time.time())}.json'
    52→    52→    last_error = None
    53→    53→    for attempt in range(3):
    54→    54→        try:
    55→    55→            result = subprocess.run(
    56→    56→                ['z-ai', 'function', '-n', 'page_reader',
    57→    57→                 '-a', json.dumps({'url': url}),
    58→    58→                 '-o', tmp],
    59→    59→                capture_output=True, text=True, timeout=timeout
    60→    60→            )
    61→    61→            if result.returncode != 0:
    62→    62→                last_error = f"z-ai CLI failed (exit {result.returncode}): {result.stderr[:300]}"
    63→    63→                print(f'[z-ai] Attempt {attempt+1} failed: {last_error[:100]}', flush=True)
    64→    64→                time.sleep(3 * (attempt + 1))  # Backoff: 3s, 6s, 9s
    65→    65→                continue
    66→    66→            with open(tmp) as f:
    67→    67→                data = json.load(f)
    68→    68→            html = data.get('data', {}).get('html', '')
    69→    69→            if not html:
    70→    70→                last_error = 'Empty response from page_reader'
    71→    71→                print(f'[z-ai] Attempt {attempt+1}: empty response', flush=True)
    72→    72→                time.sleep(3 * (attempt + 1))
    73→    73→                continue
    74→    74→            return html
    75→    75→        except subprocess.TimeoutExpired:
    76→    76→            last_error = f'z-ai CLI timed out after {timeout}s'
    77→    77→            print(f'[z-ai] Attempt {attempt+1}: {last_error}', flush=True)
    78→    78→            time.sleep(3)
    79→    79→        except Exception as e:
    80→    80→            last_error = str(e)
    81→    81→            print(f'[z-ai] Attempt {attempt+1} error: {last_error[:100]}', flush=True)
    82→    82→            time.sleep(3)
    83→    83→        finally:
    84→    84→            if os.path.exists(tmp):
    85→    85→                try:
    86→    86→                    os.unlink(tmp)
    87→    87→                except:
    88→    88→                    pass
    89→    89→            # New temp file for next attempt
    90→    90→            tmp = f'/tmp/zai_{hash(url) % 100000}_{int(time.time())}.json'
    91→    91→    raise RuntimeError(f'z-ai page_reader failed after 3 attempts: {last_error}')

    94→    94→# ── TSETMC B2 API fetcher ────────────────────────────────────────
    95→    95→def fetch_b2_data(web_id: str, timeout: int = 120) -> list:
    96→    96→    """Fetch index B2 history from TSETMC CDN via z-ai CLI."""
    97→    97→    url = f'http://cdn.tsetmc.com/api/Index/GetIndexB2History/{web_id}'
    98→    98→    html = zai_page_reader(url, timeout)
    99→    99→
   100→   100→    # Extract JSON from <pre> tag
   101→   101→    pre = re.search(r'<pre[^>]*>([\s\S]*?)</pre>', html, re.IGNORECASE)
   102→   102→    json_str = pre.group(1) if pre else re.sub(r'<[^>]+>', '', html).strip()
   103→   103→
   104→   104→    # Unescape HTML entities
   105→   105→    json_str = json_str.replace('&amp;', '&').replace('&lt;', '<').replace('&gt;', '>').replace('&quot;', '"')
   106→   106→
   107→   107→    # Try direct parse first (full JSON: {"indexB2":[...]}
   108→   108→    try:
   109→   109→        data = json.loads(json_str)
   110→   110→        return data.get('indexB2', [])
   111→   111→    except json.JSONDecodeError:
   112→   112→        pass
   113→   113→
   114→   114→    # Fallback: handle truncated JSON — find last complete object
   115→   115→    last_brace = json_str.rfind('}')
   116→   116→    if last_brace > 0:
   117→   117→        # Try to find the complete outer object: {"indexB2":[...]}
   118→   118→        for end_pos in range(last_brace, max(last_brace - 10, 0), -1):
   119→   119→            try:
   120→   120→                candidate = json_str[:end_pos + 1]
   121→   121→                data = json.loads(candidate)
   122→   122→                items = data.get('indexB2', [])
   123→   123→                if items:
   124→   124→                    return items
   125→   125→            except json.JSONDecodeError:
   126→   126→                continue
   127→   127→
   128→   128→    raise RuntimeError(f'Failed to parse B2 JSON ({len(json_str)} chars)')

   131→   131→def b2_to_candles(b2_items: list) -> list:
   132→   132→    """Convert B2 items to candle list."""
   133→   133→    candles = []
   134→   134→    for item in b2_items:
   135→   135→        close = item.get('xNivInuClMresIbs', 0)
   136→   136→        if close <= 0:
   137→   137→            continue
   138→   138→        deven = str(item.get('dEven', 0))
   139→   139→        if len(deven) < 8:
   140→   140→            continue
   141→   141→        date_str = gregorian_to_jalali(int(deven[:4]), int(deven[4:6]), int(deven[6:8]))
   142→   142→        first = item.get('xNivInuPhMresIbs', 0)
   143→   143→        prev = item.get('xNivInuPbMresIbs', 0)
   144→   144→        open_price = first if first > 0 else prev
   145→   145→        candles.append({
   146→   146→            'date': date_str,
   147→   147→            'open': round(open_price, 1),
   148→   148→            'high': round(max(open_price, close), 1),
   149→   149→            'low': round(min(open_price, close), 1),
   150→   150→            'close': round(close, 1),
   151→   151→            'volume': 0,
   152→   152→        })
   153→   153→    candles.sort(key=lambda x: x['date'])
   154→   154→    return candles

   157→   157→# ── File cache ───────────────────────────────────────────────────
   158→   158→def file_cache_path(key: str) -> str:
   159→   159→    """Generate cache file path from key. Uses key prefix for filename prefix."""
   160→   160→    safe = re.sub(r'[^a-zA-Z0-9\u0600-\u06FF_-]', '_', key)
   161→   161→    # Determine file prefix from key: stock_xxx → stock-, idx_xxx → index-, etc.
   162→   162→    prefix = 'index'
   163→   163→    for p in ('stock', 'symbols', 'sec', 'idx'):
   164→   164→        if key.startswith(p + '_'):
   165→   165→            prefix = p
   166→   166→            break
   167→   167→    return os.path.join(FILE_CACHE_DIR, f'{prefix}-{safe}.json')
   168→   168→

   170→   170→def load_file_cache(key: str) -> list | None:
   171→   171→    path = file_cache_path(key)
   172→   172→    if not os.path.exists(path):
   173→   173→        return None
   174→   174→    try:
   175→   175→        with open(path) as f:
   176→   176→            entry = json.load(f)
   177→   177→        if time.time() * 1000 - entry.get('time', 0) > CACHE_TTL * 1000:
   178→   178→            return None
   179→   179→        return entry.get('data', [])
   180→   180→    except:
   181→   181→        return None

   193→   193→# ── finpy-tse stock data helpers ─────────────────────────────────
   194→   194→def _fetch_price_history(symbol: str, adjust: bool) -> list:
   195→   195→    """Run finpy_tse.Get_Price_History in a thread-safe function.
   196→   196→    Returns list of candle dicts on success, or raises on failure."""
   197→   197→    import finpy_tse
   198→   198→
   199→   199→    df = finpy_tse.Get_Price_History(
   200→   200→        stock=symbol,
   201→   201→        ignore_date=True,
   202→   202→        adjust_price=adjust,
   203→   203→        show_weekday=False,
   204→   204→        double_date=False,
   205→   205→    )
   206→   206→    if df is None or (hasattr(df, 'empty') and df.empty):
   207→   207→        raise RuntimeError(f'No data returned for symbol "{symbol}"')
   208→   208→
   209→   209→    # Reset index to get J-Date as a column
   210→   210→    df = df.reset_index()
   211→   211→    # J-Date is the former index, named 'J-Date' or the first column
   212→   212→    date_col = 'J-Date' if 'J-Date' in df.columns else df.columns[0]
   213→   213→
   214→   214→    # When adjust_price=True, use Adj columns; otherwise use raw columns
   215→   215→    if adjust and 'Adj Close' in df.columns:
   216→   216→        open_col, high_col, low_col, close_col = 'Adj Open', 'Adj High', 'Adj Low', 'Adj Close'
   217→   217→    else:
   218→   218→        open_col, high_col, low_col, close_col = 'Open', 'High', 'Low', 'Close'
   219→   219→
   220→   220→    candles = []
   221→   221→    for _, row in df.iterrows():
   222→   222→        # Convert date from 1404-01-15 to 1404/01/15
   223→   223→        raw_date = str(row[date_col])
   224→   224→        date_str = raw_date.replace('-', '/')
   225→   225→        try:
   226→   226→            candles.append({
   227→   227→                'date': date_str,
   228→   228→                'open': int(row[open_col]),
   229→   229→                'high': int(row[high_col]),
   230→   230→                'low': int(row[low_col]),
   231→   231→                'close': int(row[close_col]),
   232→   232→                'volume': int(row['Volume']),
   233→   233→            })
   234→   234→        except (ValueError, TypeError, KeyError):
   235→   235→            # Skip rows with bad data
   236→   236→            continue
   237→   237→
   238→   238→    # Sort by date ascending
   239→   239→    candles.sort(key=lambda x: x['date'])
   240→   240→    return candles

   422→   422→    def _handle_stock_history(self, params):
   423→   423→        """Handle /api/stock-history?symbol=خودرو&adjust=1
   424→   424→
   425→   425→        Tries finpy_tse.Get_Price_History via ThreadPoolExecutor with 30s timeout.
   426→   426→        If it fails/times out, returns JSON error (caller falls back to file cache).
   427→   427→        """
   428→   428→        symbol = (params.get('symbol') or [''])[0].strip()
   429→   429→        if not symbol:
   430→   430→            return self._send_json({'error': 'پارامتر symbol الزامی است', 'candles': []}, 400)
   431→   431→
   432→   432→        adjust_str = (params.get('adjust') or ['1'])[0].strip()
   433→   433→        adjust = adjust_str != '0'
   434→   434→
   435→   435→        cache_key = f'stock_{symbol}_{"adj" if adjust else "raw"}'
   436→   436→
   437→   437→        # Check file cache first
   438→   438→        cached = load_file_cache(cache_key)
   439→   439→        if cached:
   440→   440→            print(f'[stock] Cache hit: {symbol} (adjust={adjust}, {len(cached)} candles)', flush=True)
   441→   441→            return self._send_json({
   442→   442→                'symbol': symbol,
   443→   443→                'adjust': adjust,
   444→   444→                'count': len(cached),
   445→   445→                'source': 'cache',
   446→   446→                'candles': cached,
   447→   447→            })
   448→   448→
   449→   449→        # Run finpy-tse in a thread with 30s timeout
   450→   450→        print(f'[stock] Fetching price history for "{symbol}" (adjust={adjust}) via finpy-tse...', flush=True)
   451→   451→        try:
   452→   452→            with ThreadPoolExecutor(max_workers=1) as executor:
   453→   453→                future = executor.submit(_fetch_price_history, symbol, adjust)
   454→   454→                try:
   455→   455→                    candles = future.result(timeout=30)
   456→   456→                except FuturesTimeoutError:
   457→   457→                    future.cancel()
   458→   458→                    print(f'[stock] Timeout after 30s for "{symbol}"', flush=True)
   459→   459→                    return self._send_json({
   460→   460→                        'error': 'finpy-tse timed out (30s) — likely CDN unreachable from sandbox',
   461→   461→                        'symbol': symbol,
   462→   462→                        'candles': [],
   463→   463→                    }, 504)
   464→   464→        except Exception as e:
   465→   465→            err_msg = str(e)
   466→   466→            print(f'[stock] Error for "{symbol}": {err_msg[:200]}', flush=True)
   467→   467→            return self._send_json({
   468→   468→                'error': err_msg,
   469→   469→                'symbol': symbol,
   470→   470→                'candles': [],
   471→   471→            }, 500)
   472→   472→
   473→   473→        if not candles:
   474→   474→            return self._send_json({
   475→   475→                'error': f'داده‌ای برای نماد "{symbol}" یافت نشد',
   476→   476→                'symbol': symbol,
   477→   477→                'candles': [],
   478→   478→            }, 404)
   479→   479→
   480→   480→        # Cache and return
   481→   481→        save_file_cache(cache_key, candles)
   482→   482→        print(f'[stock] Saved {len(candles)} candles for "{symbol}" (adjust={adjust})', flush=True)
   483→   483→        return self._send_json({
   484→   484→            'symbol': symbol,
   485→   485→            'adjust': adjust,
   486→   486→            'count': len(candles),
   487→   487→            'source': 'finpy-tse',
   488→   488→            'candles': candles,

   561→   561→    def _filter_symbols(self, stock_list: list, query: str, query_normalized: str) -> list:
   562→   562→        """Filter stock list by query, matching against ticker and name."""
   563→   563→        matches = []
   564→   564→        for stock in stock_list:
   565→   565→            ticker = stock.get('ticker', '')
   566→   566→            name = stock.get('name', '')
   567→   567→            # Normalize for comparison
   568→   568→            ticker_n = re.sub(r'[\u200c\u200d]', '', ticker).strip()
   569→   569→            name_n = re.sub(r'[\u200c\u200d]', '', name).strip()
   570→   570→
   571→   571→            if (query in ticker or query in name or
   572→   572→                query_normalized in ticker_n or query_normalized in name_n):
   573→   573→                matches.append(stock)
   574→   574→
   575→   575→        # Sort: exact ticker prefix matches first, then name matches
   576→   576→        def sort_key(s):
   577→   577→            t = s.get('ticker', '')
   578→   578→            n = s.get('name', '')
   579→   579→            exact_ticker = 0 if t == query else 1
   580→   580→            starts_ticker = 0 if t.startswith(query) else 1
   581→   581→            starts_name = 0 if n.startswith(query) else 1
   582→   582→            return (exact_ticker, starts_ticker, starts_name, t)
   583→   583→
   584→   584→        matches.sort(key=sort_key)
   585→   585→        return matches[:50]  # Limit to 50 results
