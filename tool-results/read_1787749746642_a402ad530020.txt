    16→interface InstrumentItem {
    17→  l18: string;
    18→  l30: string;
    19→  pl: number;
    20→  pcp: number;
    21→  tno: number;
    22→  tvol: number;

    27→    | 'crypto' | 'world_index' | 'foreign_stock' | 'forex' | 'energy' | 'metal' | 'commodity'
    28→    | 'yahoo_stock';
    29→  insCode?: string;
    30→  tgjuKey?: string;

    33→  webId?: string | number;  // TSETMC web ID (string to preserve precision)
    34→  isMainIndex?: boolean; // True for main market indices
    35→  index?: number;
    36→  indexChange?: number;
    37→  indexChangePercent?: number;
    38→  indexMin?: number;
    39→  indexMax?: number;
    40→}

   145→
   146→function formatNum(num: number): string {
   147→  if (num == null) return '—';
   148→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 0 }).format(num);
   149→}
   150→
   151→function formatIdx(num: number): string {
   152→  if (num == null) return '—';
   153→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 2 }).format(num);
   154→}

   165→/* --─ Component ------------------------------------------ */
   166→
   167→export default function SymbolSearch({
   168→  onSelect,
   169→  placeholder = 'جستجوی ابزار مالی ...',
   170→  className,
   171→}: SymbolSearchProps) {
   172→  const [query, setQuery] = React.useState('');
   173→  const [open, setOpen] = React.useState(false);
   174→  const [results, setResults] = React.useState<InstrumentItem[]>([]);
   175→  const [loading, setLoading] = React.useState(false);
   176→  const [activeIndex, setActiveIndex] = React.useState(-1);
   177→  const [totalMatched, setTotalMatched] = React.useState(0);
   178→  const [activeCategory, setActiveCategory] = React.useState<CategoryKey>('all');
   179→  const [activeIndustry, setActiveIndustry] = React.useState<string | null>(null);
   180→  const [showIndustryPicker, setShowIndustryPicker] = React.useState(false);
   181→
   182→  const cacheRef = React.useRef<InstrumentsData | null>(null);
   183→  const tgjuCacheRef = React.useRef<TgjuData | null>(null);
   184→  const yahooCacheRef = React.useRef<InstrumentItem[] | null>(null);
   185→  const fetchRef = React.useRef<Promise<InstrumentsData> | null>(null);
   186→  const tgjuFetchRef = React.useRef<Promise<TgjuData | null> | null>(null);
   187→  const yahooFetchRef = React.useRef<Promise<InstrumentItem[] | null> | null>(null);
   188→  const inputRef = React.useRef<HTMLInputElement>(null);
   189→  const dropdownRef = React.useRef<HTMLDivElement>(null);
   190→  const itemRefs = React.useRef<(HTMLDivElement | null)[]>([]);
   191→  const industryScrollRef = React.useRef<HTMLDivElement>(null);
   192→
   193→  /* -- Data fetching -- */
   194→  const fetchData = React.useCallback(async (): Promise<InstrumentsData> => {
   195→    if (cacheRef.current) return cacheRef.current;
   196→    if (fetchRef.current) return fetchRef.current;
   197→    const p = async () => {
   198→      try {
   199→        const r = await fetch('/api/instruments');
   200→        if (!r.ok) throw new Error(`${r.status}`);
   201→        const d: InstrumentsData = await r.json();
   202→        cacheRef.current = d;
   203→        return d;
   204→      } finally {
   205→        fetchRef.current = null;
   206→      }
   207→    };
   208→    fetchRef.current = p();
   209→    return fetchRef.current;
   210→  }, []);
   211→
   212→  const fetchTgjuData = React.useCallback(async (): Promise<TgjuData | null> => {
   213→    if (tgjuCacheRef.current) return tgjuCacheRef.current;
   214→    if (tgjuFetchRef.current) return tgjuFetchRef.current;
   215→    const p = async () => {
   216→      try {
   217→        const r = await fetch('/api/tgju-instruments');
   218→        if (!r.ok) return null;
   219→        const d: TgjuData = await r.json();
   220→        tgjuCacheRef.current = d;
   221→        return d;
   222→      } catch {
   223→        return null;
   224→      } finally {
   225→        tgjuFetchRef.current = null;
   226→      }
   227→    };
   228→    tgjuFetchRef.current = p();
   229→    return p();
   230→  }, []);
   231→
   232→  const fetchYahooData = React.useCallback(async (): Promise<InstrumentItem[] | null> => {
   233→    if (yahooCacheRef.current) return yahooCacheRef.current;
   234→    if (yahooFetchRef.current) return yahooFetchRef.current;
   235→    const p = async () => {
   236→      try {
   237→        const r = await fetch('/api/yahoo-instruments');
   238→        if (!r.ok) return null;
   239→        const d = await r.json() as { items: InstrumentItem[] };
   240→        yahooCacheRef.current = d.items || [];
   241→        return yahooCacheRef.current;
   242→      } catch {
   243→        return null;
   244→      } finally {
   245→        yahooFetchRef.current = null;
   246→      }
   247→    };
   248→    yahooFetchRef.current = p();
   249→    return p();
   250→  }, []);
   251→
   252→  /* -- Check if category is TGJU-based -- */
   253→  const isTgjuCategory = (cat: CategoryKey) => TGJU_CATEGORIES.has(cat);
   254→  const isYahooCategory = (cat: CategoryKey) => cat === 'yahoo_stock';
   255→
   256→  /* -- Get all items for current category -- */
   257→  const getItemsForCategory = React.useCallback(async (cat: CategoryKey): Promise<InstrumentItem[]> => {
   258→    // Yahoo Finance stocks
   259→    if (isYahooCategory(cat)) {
   260→      const yahoo = yahooCacheRef.current || await fetchYahooData();
   261→      return yahoo || [];
   262→    }
   263→
   264→    if (isTgjuCategory(cat)) {
   265→      const tgju = tgjuCacheRef.current || await fetchTgjuData();
   266→      if (!tgju) return [];
   267→      switch (cat) {
   268→        case 'currency': return tgju.currencies;
   269→        case 'gold': return [...tgju.gold, ...tgju.silver];
   270→        case 'gold_etf': return tgju.goldEtfs;
   271→        case 'silver': return tgju.silver;
   272→        case 'crypto': return tgju.crypto;
   273→        case 'world_index': return [...tgju.worldIndices, ...tgju.foreignStocks];
   274→        case 'foreign_stock': return tgju.foreignStocks;
   275→        case 'forex': return tgju.forex;
   276→        case 'energy': return tgju.energy;
   277→        case 'metal': return tgju.metals;
   278→        case 'commodity': return tgju.commodities;
   279→        default: return [];
   280→      }
   281→    }
   282→
   283→    // TSE categories
   284→    const data = cacheRef.current || await fetchData();
   285→    switch (cat) {
   286→      case 'indices': return data.indices;
   287→      case 'stocks':
   288→        return activeIndustry
   289→          ? data.stocks.filter((s) => s.cs === activeIndustry)
   290→          : data.stocks;
   291→      case 'etf': return data.etfs;
   292→      case 'bond': return [...data.bonds, ...data.mortgage];
   293→      case 'derivative': return [...data.futures, ...data.salaf];
   294→      case 'all':
   295→      default: {
   296→        const tgju = tgjuCacheRef.current || await fetchTgjuData();
   297→        const yahoo = yahooCacheRef.current || await fetchYahooData();
   298→        return [
   299→          ...data.indices,
   300→          ...data.stocks,
   301→          ...data.etfs,
   302→          ...(tgju?.currencies || []),
   303→          ...(tgju?.forex || []),
   304→          ...(tgju?.crypto || []),
   305→          ...(tgju?.gold || []),
   306→          ...(tgju?.silver || []),
   307→          ...(tgju?.goldEtfs || []),
   308→          ...(tgju?.worldIndices || []),
   309→          ...(tgju?.energy || []),
   310→          ...(tgju?.metals || []),
   311→          ...(tgju?.commodities || []),
   312→          ...(yahoo || []),
   313→          ...data.bonds,
   314→          ...data.futures,
   315→          ...data.salaf,
   316→          ...data.mortgage,
   317→        ];
   318→      }
   319→    }
   320→  }, [activeIndustry, fetchData, fetchTgjuData, fetchYahooData]);
   321→
   322→  /* -- Popular items -- */
   323→  const getPopular = React.useCallback(async () => {
   324→    const data = cacheRef.current || await fetchData();
   325→    const tgju = tgjuCacheRef.current;
   326→    const all = [
   327→      ...(data.indices.map((i) => ({ ...i, tval: i.index || 0 }))),
   328→      ...data.stocks,
   329→      ...data.etfs,
   330→      ...(tgju?.currencies || []),
   331→      ...(tgju?.crypto?.slice(0, 5) || []),
   332→      ...(tgju?.gold?.slice(0, 3) || []),
   333→    ];
   334→    const sorted = [...all].sort((a, b) => (b.tval || 0) - (a.tval || 0)).slice(0, 15);
   335→    setResults(sorted);
   336→    setTotalMatched(all.length);
   337→  }, [fetchData]);
   338→
   339→  /* -- Filter by search query -- */
   340→  const doFilter = React.useCallback((items: InstrumentItem[], q: string) => {
   341→    const n = q.trim().toLowerCase();
   342→    const f = items.filter(
   343→      (s) =>
   344→        s.l18?.toLowerCase().includes(n) ||
   345→        s.l30?.toLowerCase().includes(n) ||
   346→        s.cs?.toLowerCase().includes(n),
   347→    );
   348→    setTotalMatched(f.length);
   349→    setResults(f.slice(0, MAX_RESULTS));
   350→    setActiveIndex(-1);
   351→  }, []);
   352→
   353→  /* -- Show dropdown with data -- */
   354→  const showDropdown = React.useCallback(async (q: string) => {
   355→    setOpen(true);
   356→    await Promise.all([fetchData(), fetchTgjuData(), fetchYahooData()]);
   357→    const items = await getItemsForCategory(activeCategory);
   358→    if (q.trim().length === 0) {
   359→      if (activeCategory === 'all') {
   360→        await getPopular();
   361→      } else {
   362→        setResults(items.slice(0, MAX_RESULTS));
   363→        setTotalMatched(items.length);
   364→      }
   365→    } else {
   366→      doFilter(items, q);
   367→    }
   368→    setLoading(false);
   369→  }, [fetchData, fetchTgjuData, getItemsForCategory, activeCategory, getPopular, doFilter]);
   370→
   371→  /* -- Category change -- */
   372→  const handleCategoryChange = React.useCallback(async (cat: CategoryKey) => {
   373→    setActiveCategory(cat);
   374→    if (cat !== 'stocks') setActiveIndustry(null);
   375→    setShowIndustryPicker(false);
   376→
   377→    const items = await getItemsForCategory(cat);
   378→    if (query.trim().length > 0) {
   379→      doFilter(items, query);
   380→    } else if (cat === 'all') {
   381→      await getPopular();
   382→    } else {
   383→      setResults(items.slice(0, MAX_RESULTS));
   384→      setTotalMatched(items.length);
   385→    }
   386→    setActiveIndex(-1);
   387→  }, [getItemsForCategory, query, doFilter, getPopular]);
   388→
   389→  /* -- Industry change -- */
   390→  const handleIndustryChange = React.useCallback((industry: string | null) => {
   391→    setActiveIndustry(industry);
   392→    setShowIndustryPicker(false);
   393→    if (cacheRef.current) {
   394→      const items = industry
   395→        ? cacheRef.current.stocks.filter((s) => s.cs === industry)
   396→        : cacheRef.current.stocks;
   397→      if (query.trim().length > 0) {
   398→        doFilter(items, query);
   399→      } else {
   400→        setResults(items.slice(0, MAX_RESULTS));
   401→        setTotalMatched(items.length);
   402→      }
   403→      setActiveIndex(-1);
   404→    }
   405→  }, [query, doFilter]);
   406→
   407→  const handleChange = React.useCallback((value: string) => {
   408→    setQuery(value);
   409→    getItemsForCategory(activeCategory).then((items) => {
   410→      if (value.trim().length === 0) {
   411→        if (activeCategory === 'all') getPopular();
   412→        else {
   413→          setResults(items.slice(0, MAX_RESULTS));
   414→          setTotalMatched(items.length);
   415→        }
   416→      } else {
   417→        doFilter(items, value);
   418→      }
   419→    });
   420→  }, [activeCategory, getItemsForCategory, getPopular, doFilter]);
   421→
   422→  const handleFocus = React.useCallback(() => {
   423→    setLoading(true);
   424→    showDropdown(query);
   425→  }, [query, showDropdown]);
   426→
   427→  const selectSymbol = React.useCallback((s: InstrumentItem) => {
   428→    setQuery(s.l18);
   429→    setOpen(false);
   430→    onSelect?.(s.l18, s.category, s.insCode, s.tgjuKey, s.finpySector, s.finpyIndex, s.webId, (s as Record<string, unknown>).yahooSymbol as string | undefined);
   431→    inputRef.current?.blur();
   432→  }, [onSelect]);
   433→
   434→  const handleKeyDown = React.useCallback((e: React.KeyboardEvent) => {
   435→    if (!open || results.length === 0) {
   436→      if (e.key === 'ArrowDown' || e.key === 'Enter') { e.preventDefault(); handleFocus(); }
   437→      return;
   438→    }
   439→    if (e.key === 'ArrowDown') { e.preventDefault(); setActiveIndex((p) => p < results.length - 1 ? p + 1 : 0); }
   440→    else if (e.key === 'ArrowUp') { e.preventDefault(); setActiveIndex((p) => p > 0 ? p - 1 : results.length - 1); }
   441→    else if (e.key === 'Enter') { e.preventDefault(); if (activeIndex >= 0) selectSymbol(results[activeIndex]); }
   442→    else if (e.key === 'Escape') { e.preventDefault(); setOpen(false); inputRef.current?.blur(); }
   443→  }, [open, results, activeIndex, selectSymbol, handleFocus]);
   444→
   445→  React.useEffect(() => {
   446→    if (activeIndex >= 0) itemRefs.current[activeIndex]?.scrollIntoView({ block: 'nearest' });
   447→  }, [activeIndex]);
   448→
   449→  React.useEffect(() => {
   450→    if (!open) return;
   451→    const handler = (e: MouseEvent) => {
   452→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node) &&
   453→          inputRef.current && !inputRef.current.contains(e.target as Node)) {
   454→        setOpen(false);
   455→        setShowIndustryPicker(false);
   456→      }
   457→    };
   458→    document.addEventListener('mousedown', handler);
   459→    return () => document.removeEventListener('mousedown', handler);
   460→  }, [open]);
   461→
   462→  const clear = () => { setQuery(''); setActiveIndustry(null); setActiveCategory('all'); setResults([]); inputRef.current?.focus(); };
   463→  const isPopular = query.trim().length === 0 && activeCategory === 'all';
   464→  const isSearch = query.trim().length > 0;
   465→
   466→  /* -- Render instrument price -- */
   467→  const renderPrice = (item: InstrumentItem) => {
   468→    const isTgju = isTgjuCategory(item.category as CategoryKey);
   469→    if (item.category === 'index') {
   470→      return (
   471→        <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   472→          <span className='text-xs font-semibold text-[#111827]'>{formatIdx(item.pl)}</span>
   473→          <span className={cn(
   474→            'text-[11px] font-bold px-1.5 py-0.5 rounded',
   475→            item.pcp > 0 ? 'bg-emerald-50 text-emerald-700' :
   476→            item.pcp < 0 ? 'bg-red-50 text-red-700' : 'text-[#6b7280]',
   477→          )}>
   478→            {item.pcp > 0 ? '+' : ''}{item.pcp?.toFixed(2)}%
   479→          </span>
   480→        </div>
   481→      );
   482→    }
   483→    // For static TGJU items with no live price, show dash
   484→    // Yahoo stock: show dollar-formatted price
   485→    if (item.category === 'yahoo_stock') {
   486→      if (item.pl === 0) {
   487→        return (
   488→          <div className='flex shrink-0 flex-col items-end gap-0.5'>
   489→            <span className='text-xs text-[#6b7280]'>—</span>
   490→          </div>
   491→        );
   492→      }
   493→      return (
   494→        <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   495→          <span className='text-xs font-semibold text-[#111827]'>{formatIdx(item.pl)}</span>
   496→          <span className={cn(
   497→            'text-[11px] font-bold px-1.5 py-0.5 rounded',
   498→            item.pcp > 0 ? 'bg-emerald-50 text-emerald-700' :
   499→            item.pcp < 0 ? 'bg-red-50 text-red-700' : 'text-[#6b7280]',
