   148→// Graph Algorithms
   149→// ═══════════════════════════════════════════════════════════════════════════════
   150→
   151→interface EdgeInfo {
   152→  from: string;
   153→  to: string;
   154→  label: string;
   155→  type: string;
   156→  prob: number;
   157→}
   158→
   159→interface PathInfo {
   160→  nodes: string[];
   161→  edges: number[];
   162→  prob: number;
   163→  target: string;
   164→}
   165→
   166→// Calculate edge probabilities using engine-provided edge weights (Layer 5)
   167→function calcEdgeProbabilities(
   168→  ew: { up: number; down: number; pullback: number; risk: number }
   169→): Record<string, number> {
   170→  const typeWeights: Record<string, number> = {
   171→    up: ew.up,
   172→    pullback: ew.pullback,
   173→    down: ew.down,
   174→    risk: ew.risk,
   175→  };
   176→
   177→  // Build adjacency: for each node, group outgoing edges by type
   178→  const adj: Record<string, { idx: number; type: string }[]> = {};
   179→  EDGES.forEach((e, i) => {
   180→    if (!adj[e[0]]) adj[e[0]] = [];
   181→    adj[e[0]].push({ idx: i, type: e[3] });
   182→  });
   183→
   184→  const edgeProbs: Record<string, number> = {};
   185→
   186→  for (const [node, outEdges] of Object.entries(adj)) {
   187→    // Sum weights of outgoing edges
   188→    const totalWeight = outEdges.reduce((sum, e) => sum + (typeWeights[e.type] ?? 0.1), 0);
   189→    if (totalWeight === 0) {
   190→      outEdges.forEach(e => { edgeProbs[String(e.idx)] = 1 / outEdges.length; });
   191→      continue;
   192→    }
   193→    outEdges.forEach(e => {
   194→      edgeProbs[String(e.idx)] = (typeWeights[e.type] ?? 0.1) / totalWeight;
   195→    });
   196→  }
   197→
   198→  return edgeProbs;
   199→}
   200→
   201→// Find all paths from 'A' to terminal nodes using DFS
   202→function findAllPaths(edgeProbs: Record<string, number>): PathInfo[] {
   203→  const adj: Record<string, { to: string; edgeIdx: number }[]> = {};
   204→  EDGES.forEach((e, i) => {
   205→    if (!adj[e[0]]) adj[e[0]] = [];
   206→    adj[e[0]].push({ to: e[1], edgeIdx: i });
   207→  });
   208→
   209→  const paths: PathInfo[] = [];
   210→  const MAX_PATHS = 200;
   211→
   212→  function dfs(node: string, visited: Set<string>, currentPath: string[], edgeIndices: number[], currentProb: number) {
   213→    if (SCENARIO_KEYS.includes(node as any)) {
   214→      paths.push({ nodes: [...currentPath, node], edges: [...edgeIndices], prob: currentProb, target: node });
   215→      return;
   216→    }
   217→    if (paths.length >= MAX_PATHS) return;
   218→    if (visited.has(node)) return;
   219→    visited.add(node);
   220→
   221→    const outs = adj[node] ?? [];
   222→    for (const { to, edgeIdx } of outs) {
   223→      const ep = edgeProbs[String(edgeIdx)] ?? 0.1;
   224→      dfs(to, visited, [...currentPath, node], [...edgeIndices, edgeIdx], currentProb * ep);
   225→    }
   226→    visited.delete(node);
   227→  }
   228→
   229→  dfs('A', new Set(), [], [], 1);
   230→  return paths.sort((a, b) => b.prob - a.prob);
   231→}
