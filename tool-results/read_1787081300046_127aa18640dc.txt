   139→/* --─ Helpers -------------------------------------------- */
   140→
   141→function formatNum(num: number): string {
   142→  if (num == null) return '—';
   143→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 0 }).format(num);
   144→}
   145→
   146→function formatIdx(num: number): string {
   147→  if (num == null) return '—';
   148→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 2 }).format(num);
   149→}

   156→    </svg>
   157→  );
   158→}

   160→/* --─ Component ------------------------------------------ */
   161→
   162→export default function SymbolSearch({
   163→  onSelect,
   164→  placeholder = 'جستجوی ابزار مالی ...',
   165→  className,
   166→}: SymbolSearchProps) {
   167→  const [query, setQuery] = React.useState('');
   168→  const [open, setOpen] = React.useState(false);
   169→  const [results, setResults] = React.useState<InstrumentItem[]>([]);
   170→  const [loading, setLoading] = React.useState(false);
   171→  const [activeIndex, setActiveIndex] = React.useState(-1);
   172→  const [totalMatched, setTotalMatched] = React.useState(0);
   173→  const [activeCategory, setActiveCategory] = React.useState<CategoryKey>('all');
   174→  const [activeIndustry, setActiveIndustry] = React.useState<string | null>(null);
   175→  const [showIndustryPicker, setShowIndustryPicker] = React.useState(false);
   176→
   177→  const cacheRef = React.useRef<InstrumentsData | null>(null);
   178→  const tgjuCacheRef = React.useRef<TgjuData | null>(null);
   179→  const fetchRef = React.useRef<Promise<InstrumentsData> | null>(null);
   180→  const tgjuFetchRef = React.useRef<Promise<TgjuData | null> | null>(null);
   181→  const inputRef = React.useRef<HTMLInputElement>(null);
   182→  const dropdownRef = React.useRef<HTMLDivElement>(null);
   183→  const itemRefs = React.useRef<(HTMLDivElement | null)[]>([]);
   184→  const industryScrollRef = React.useRef<HTMLDivElement>(null);
   185→
   186→  /* -- Data fetching -- */
   187→  const fetchData = React.useCallback(async (): Promise<InstrumentsData> => {
   188→    if (cacheRef.current) return cacheRef.current;
   189→    if (fetchRef.current) return fetchRef.current;
   190→    const p = async () => {
   191→      try {
   192→        const r = await fetch('/api/instruments');
   193→        if (!r.ok) throw new Error(`${r.status}`);
   194→        const d: InstrumentsData = await r.json();
   195→        cacheRef.current = d;
   196→        return d;
   197→      } finally {
   198→        fetchRef.current = null;
   199→      }
   200→    };
   201→    fetchRef.current = p();
   202→    return fetchRef.current;
   203→  }, []);
   204→
   205→  const fetchTgjuData = React.useCallback(async (): Promise<TgjuData | null> => {
   206→    if (tgjuCacheRef.current) return tgjuCacheRef.current;
   207→    if (tgjuFetchRef.current) return tgjuFetchRef.current;
   208→    const p = async () => {
   209→      try {
   210→        const r = await fetch('/api/tgju-instruments');
   211→        if (!r.ok) return null;
   212→        const d: TgjuData = await r.json();
   213→        tgjuCacheRef.current = d;
   214→        return d;
   215→      } catch {
   216→        return null;
   217→      } finally {
   218→        tgjuFetchRef.current = null;
   219→      }
   220→    };
   221→    tgjuFetchRef.current = p();
   222→    return p();
   223→  }, []);
   224→
   225→  /* -- Check if category is TGJU-based -- */
   226→  const isTgjuCategory = (cat: CategoryKey) => TGJU_CATEGORIES.has(cat);
   227→
   228→  /* -- Get all items for current category -- */
   229→  const getItemsForCategory = React.useCallback(async (cat: CategoryKey): Promise<InstrumentItem[]> => {
   230→    if (isTgjuCategory(cat)) {
   231→      const tgju = tgjuCacheRef.current || await fetchTgjuData();
   232→      if (!tgju) return [];
   233→      switch (cat) {
   234→        case 'currency': return tgju.currencies;
   235→        case 'gold': return [...tgju.gold, ...tgju.silver];
   236→        case 'gold_etf': return tgju.goldEtfs;
   237→        case 'silver': return tgju.silver;
   238→        case 'crypto': return tgju.crypto;
   239→        case 'world_index': return tgju.worldIndices;
   240→        case 'forex': return tgju.forex;
   241→        case 'energy': return tgju.energy;
   242→        case 'metal': return tgju.metals;
   243→        case 'commodity': return tgju.commodities;
   244→        default: return [];
   245→      }
   246→    }
   247→
   248→    // TSE categories
   249→    const data = cacheRef.current || await fetchData();
   250→    switch (cat) {
   251→      case 'indices': return data.indices;
   252→      case 'stocks':
   253→        return activeIndustry
   254→          ? data.stocks.filter((s) => s.cs === activeIndustry)
   255→          : data.stocks;
   256→      case 'etf': return data.etfs;
   257→      case 'bond': return [...data.bonds, ...data.mortgage];
   258→      case 'derivative': return [...data.futures, ...data.salaf];
   259→      case 'all':
   260→      default: {
   261→        const tgju = tgjuCacheRef.current;
   262→        return [
   263→          ...data.indices,
   264→          ...data.stocks,
   265→          ...data.etfs,
   266→          ...(tgju?.currencies || []),
   267→          ...(tgju?.forex || []),
   268→          ...(tgju?.crypto || []),
   269→          ...(tgju?.gold || []),
   270→          ...(tgju?.silver || []),
   271→          ...(tgju?.goldEtfs || []),
   272→          ...(tgju?.worldIndices || []),
   273→          ...(tgju?.energy || []),
   274→          ...(tgju?.metals || []),
   275→          ...(tgju?.commodities || []),
   276→          ...data.bonds,
   277→          ...data.futures,
   278→          ...data.salaf,
   279→          ...data.mortgage,
   280→        ];
   281→      }
   282→    }
   283→  }, [activeIndustry, fetchData, fetchTgjuData]);
   284→
   285→  /* -- Popular items -- */
   286→  const getPopular = React.useCallback(async () => {
   287→    const data = cacheRef.current || await fetchData();
   288→    const tgju = tgjuCacheRef.current;
   289→    const all = [
   290→      ...(data.indices.map((i) => ({ ...i, tval: i.index || 0 }))),
   291→      ...data.stocks,
   292→      ...data.etfs,
   293→      ...(tgju?.currencies || []),
   294→      ...(tgju?.crypto?.slice(0, 5) || []),
   295→      ...(tgju?.gold?.slice(0, 3) || []),
   296→    ];
   297→    const sorted = [...all].sort((a, b) => (b.tval || 0) - (a.tval || 0)).slice(0, 15);
   298→    setResults(sorted);
   299→    setTotalMatched(all.length);
   300→  }, [fetchData]);
   301→
   302→  /* -- Filter by search query -- */
   303→  const doFilter = React.useCallback((items: InstrumentItem[], q: string) => {
   304→    const n = q.trim().toLowerCase();
   305→    const f = items.filter(
   306→      (s) =>
   307→        s.l18?.toLowerCase().includes(n) ||
   308→        s.l30?.toLowerCase().includes(n) ||
   309→        s.cs?.toLowerCase().includes(n),
   310→    );
   311→    setTotalMatched(f.length);
   312→    setResults(f.slice(0, MAX_RESULTS));
   313→    setActiveIndex(-1);
   314→  }, []);
   315→
   316→  /* -- Show dropdown with data -- */
   317→  const showDropdown = React.useCallback(async (q: string) => {
   318→    setOpen(true);
   319→    await Promise.all([fetchData(), fetchTgjuData()]);
   320→    const items = await getItemsForCategory(activeCategory);
   321→    if (q.trim().length === 0) {
   322→      if (activeCategory === 'all') {
   323→        await getPopular();
   324→      } else {
   325→        setResults(items.slice(0, MAX_RESULTS));
   326→        setTotalMatched(items.length);
   327→      }
   328→    } else {
   329→      doFilter(items, q);
   330→    }
   331→    setLoading(false);
   332→  }, [fetchData, fetchTgjuData, getItemsForCategory, activeCategory, getPopular, doFilter]);
   333→
   334→  /* -- Category change -- */
   335→  const handleCategoryChange = React.useCallback(async (cat: CategoryKey) => {
   336→    setActiveCategory(cat);
   337→    if (cat !== 'stocks') setActiveIndustry(null);
   338→    setShowIndustryPicker(false);
   339→
   340→    const items = await getItemsForCategory(cat);
   341→    if (query.trim().length > 0) {
   342→      doFilter(items, query);
   343→    } else if (cat === 'all') {
   344→      await getPopular();
   345→    } else {
   346→      setResults(items.slice(0, MAX_RESULTS));
   347→      setTotalMatched(items.length);
   348→    }
   349→    setActiveIndex(-1);
   350→  }, [getItemsForCategory, query, doFilter, getPopular]);
   351→
   352→  /* -- Industry change -- */
   353→  const handleIndustryChange = React.useCallback((industry: string | null) => {
   354→    setActiveIndustry(industry);
   355→    setShowIndustryPicker(false);
   356→    if (cacheRef.current) {
   357→      const items = industry
   358→        ? cacheRef.current.stocks.filter((s) => s.cs === industry)
   359→        : cacheRef.current.stocks;
   360→      if (query.trim().length > 0) {
   361→        doFilter(items, query);
   362→      } else {
   363→        setResults(items.slice(0, MAX_RESULTS));
   364→        setTotalMatched(items.length);
   365→      }
   366→      setActiveIndex(-1);
   367→    }
   368→  }, [query, doFilter]);
   369→
   370→  const handleChange = React.useCallback((value: string) => {
   371→    setQuery(value);
   372→    getItemsForCategory(activeCategory).then((items) => {
   373→      if (value.trim().length === 0) {
   374→        if (activeCategory === 'all') getPopular();
   375→        else {
   376→          setResults(items.slice(0, MAX_RESULTS));
   377→          setTotalMatched(items.length);
   378→        }
   379→      } else {
   380→        doFilter(items, value);
   381→      }
   382→    });
   383→  }, [activeCategory, getItemsForCategory, getPopular, doFilter]);
   384→
   385→  const handleFocus = React.useCallback(() => {
   386→    setLoading(true);
   387→    showDropdown(query);
   388→  }, [query, showDropdown]);
   389→
   390→  const selectSymbol = React.useCallback((s: InstrumentItem) => {
   391→    setQuery(s.l18);
   392→    setOpen(false);
   393→    onSelect?.(s.l18, s.category, s.insCode, s.tgjuKey, s.finpySector);
   394→    inputRef.current?.blur();
   395→  }, [onSelect]);
   396→
   397→  const handleKeyDown = React.useCallback((e: React.KeyboardEvent) => {
   398→    if (!open || results.length === 0) {
   399→      if (e.key === 'ArrowDown' || e.key === 'Enter') { e.preventDefault(); handleFocus(); }
   400→      return;
   401→    }
   402→    if (e.key === 'ArrowDown') { e.preventDefault(); setActiveIndex((p) => p < results.length - 1 ? p + 1 : 0); }
   403→    else if (e.key === 'ArrowUp') { e.preventDefault(); setActiveIndex((p) => p > 0 ? p - 1 : results.length - 1); }
   404→    else if (e.key === 'Enter') { e.preventDefault(); if (activeIndex >= 0) selectSymbol(results[activeIndex]); }
   405→    else if (e.key === 'Escape') { e.preventDefault(); setOpen(false); inputRef.current?.blur(); }
   406→  }, [open, results, activeIndex, selectSymbol, handleFocus]);
   407→
   408→  React.useEffect(() => {
   409→    if (activeIndex >= 0) itemRefs.current[activeIndex]?.scrollIntoView({ block: 'nearest' });
   410→  }, [activeIndex]);
   411→
   412→  React.useEffect(() => {
   413→    if (!open) return;
   414→    const handler = (e: MouseEvent) => {
   415→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node) &&
   416→          inputRef.current && !inputRef.current.contains(e.target as Node)) {
   417→        setOpen(false);
   418→        setShowIndustryPicker(false);
   419→      }
   420→    };
   421→    document.addEventListener('mousedown', handler);
   422→    return () => document.removeEventListener('mousedown', handler);
   423→  }, [open]);
   424→
   425→  const clear = () => { setQuery(''); setActiveIndustry(null); setActiveCategory('all'); setResults([]); inputRef.current?.focus(); };
   426→  const isPopular = query.trim().length === 0 && activeCategory === 'all';
   427→  const isSearch = query.trim().length > 0;
   428→
   429→  /* -- Render instrument price -- */
   430→  const renderPrice = (item: InstrumentItem) => {
   431→    const isTgju = isTgjuCategory(item.category as CategoryKey);
   432→    if (item.category === 'index') {
   433→      return (
   434→        <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   435→          <span className='text-xs font-semibold text-[#111827]'>{formatIdx(item.pl)}</span>
   436→          <span className={cn(
   437→            'text-[11px] font-bold px-1.5 py-0.5 rounded',
   438→            item.pcp > 0 ? 'bg-emerald-50 text-emerald-700' :
   439→            item.pcp < 0 ? 'bg-red-50 text-red-700' : 'text-[#6b7280]',
   440→          )}>
   441→            {item.pcp > 0 ? '+' : ''}{item.pcp?.toFixed(2)}%
   442→          </span>
   443→        </div>
   444→      );
   445→    }
   446→    // For static TGJU items with no live price, show dash
   447→    if (isTgju && item.pl === 0) {
   448→      return (
   449→        <div className='flex shrink-0 flex-col items-end gap-0.5'>
   450→          <span className='text-xs text-[#6b7280]'>—</span>
   451→        </div>
   452→      );
   453→    }
   454→    const isUp = item.pcp > 0;
   455→    const isDown = item.pcp < 0;
   456→    return (
   457→      <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   458→        <span className='text-xs font-semibold text-[#111827]'>{formatNum(item.pl)}</span>
   459→        <span className={cn(
   460→          'text-[11px] font-bold px-1.5 py-0.5 rounded',
   461→          isUp ? 'bg-emerald-50 text-emerald-700' :
   462→          isDown ? 'bg-red-50 text-red-700' : 'text-[#6b7280]',
   463→        )}>
   464→          {isUp ? '+' : ''}{item.pcp?.toFixed(2)}%
   465→        </span>
   466→      </div>
   467→    );
   468→  };
   469→
   470→  const industries = cacheRef.current?.industries || [];
   471→
   472→  return (
   473→    <div dir='rtl' className={cn('relative', className)}>
   474→
   475→      <div className='relative'>
   476→        <Search className='absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-[#6b7280] pointer-events-none' />
   477→        <Input
   478→          ref={inputRef}
   479→          type='text'
   480→          value={query}
   481→          onChange={(e) => handleChange(e.target.value)}
   482→          onFocus={handleFocus}
   483→          onKeyDown={handleKeyDown}
   484→          placeholder={placeholder}
   485→          suppressHydrationWarning
   486→          className={cn(
   487→            'h-10 w-full rounded-lg border border-[#e5e7eb] bg-white',
   488→            'text-sm text-[#111827] placeholder:text-[#6b7280]',
   489→            'transition-all',
   490→            'focus-visible:border-amber-500/60 focus-visible:ring-amber-500/20 focus-visible:ring-[3px]',
   491→            'pr-9',
   492→          )}
   493→          autoComplete='off' spellCheck={false}
   494→        />
   495→        {query && (
   496→          <button onClick={clear} className='absolute left-3 top-1/2 -translate-y-1/2 text-[#6b7280] hover:text-[#111827] transition-colors' type='button'>
   497→            <X className='w-4 h-4' />
   498→          </button>
   499→        )}
   500→      </div>
   501→
   502→
   503→      {open && (
   504→        <div
   505→          ref={dropdownRef}
   506→          className='absolute top-full left-0 right-0 mt-1.5 z-50 rounded-xl border border-[#e5e7eb] bg-white shadow-xl shadow-black/10 overflow-hidden'
   507→          style={{ width: 'min(660px, 94vw)' }}
