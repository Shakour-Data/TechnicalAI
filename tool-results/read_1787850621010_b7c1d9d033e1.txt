   116→   116→// Component
   117→   117→// ═══════════════════════════════════════════════════════════════════════════════
   118→   118→
   119→   119→export default function VdssGraph(props: VdssGraphProps) {
   120→   120→  const { colors: C } = useTheme();
   121→   121→  const {
   122→   122→    symbolName,
   123→   123→    currentPrice,
   124→   124→    scenarios,
   125→   125→    decisionGraph,
   126→   126→  } = props;
   127→   127→
   128→   128→  const [activeFilter, setActiveFilter] = useState('all');
   129→   129→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   130→   130→  const graphRef = useRef<HTMLDivElement>(null);
   131→   131→  const svgRef = useRef<SVGSVGElement>(null);
   132→   132→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   133→   133→  const shellRef = useRef<HTMLDivElement>(null);
   134→   134→
   135→   135→  // ── Derived data from decisionGraph ──────────────────────────────
   136→   136→  const edges = decisionGraph?.edges ?? [];
   137→   137→  const nodes = decisionGraph?.nodes ?? [];
   138→   138→  const nodePositions = decisionGraph?.nodePositions ?? {};
   139→   139→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   140→   140→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   141→   141→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   142→   142→  const pathContributions = decisionGraph?.pathContributions ?? {};
   143→   143→  const nodeValues = decisionGraph?.nodeValues ?? {};
   144→   144→
   145→   145→  // Build node lookup map
   146→   146→  const nodeMap = useMemo(() => {
   147→   147→    const map: Record<string, (typeof nodes)[number]> = {};
   148→   148→    for (const n of nodes) map[n.id] = n;
   149→   149→    return map;
   150→   150→  }, [nodes]);
   151→   151→
   152→   152→  // ── Visible edges/nodes based on filter ──────────────────────────
   153→   153→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   154→   154→    if (activeFilter === 'all') {
   155→   155→      const allEdgeIdx = edges.map((_, i) => i);
   156→   156→      const allNodes = new Set(nodes.map(n => n.id));
   157→   157→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   158→   158→    }
   159→   159→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   160→   160→      // Show edges going TO this scenario node, plus their source nodes
   161→   161→      const edgeSet = new Set<number>();
   162→   162→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   163→   163→      edges.forEach((e, i) => {
   164→   164→        if (e.to === activeFilter) {
   165→   165→          edgeSet.add(i);
   166→   166→          nodeSet.add(e.from);
   167→   167→        }
   168→   168→      });
   169→   169→      // Also show intermediate nodes connected to those sources
   170→   170→      edges.forEach((e, i) => {
   171→   171→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   172→   172→          edgeSet.add(i);
   173→   173→          nodeSet.add(e.from);
   174→   174→        }
   175→   175→      });
   176→   176→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   177→   177→    }
   178→   178→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   179→   179→      const branchType = `branch-${activeFilter}` as const;
   180→   180→      const edgeSet = new Set<number>();
   181→   181→      const nodeSet = new Set<string>(['ROOT']);
   182→   182→      edges.forEach((e, i) => {
   183→   183→        if (e.type === branchType || (e.type !== `branch-trend` && e.type !== `branch-breakout` && e.type !== `branch-reversal` && nodeSet.has(e.from))) {
   184→   184→          // Include branch-type edges
   185→   185→          if (e.type === branchType) {
   186→   186→            edgeSet.add(i);
   187→   187→            nodeSet.add(e.from);
   188→   188→            nodeSet.add(e.to);
   189→   189→          }
   190→   190→        }
   191→   191→      });
   192→   192→      // Now add all edges whose source is already in nodeSet
   193→   193→      const expanded = new Set<number>();
   194→   194→      const expandedNodes = new Set<string>(nodeSet);
   195→   195→      edges.forEach((e, i) => {
   196→   196→        if (expandedNodes.has(e.from)) {
   197→   197→          expanded.add(i);
   198→   198→          expandedNodes.add(e.to);
   199→   199→        }
   200→   200→      });
   201→   201→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   202→   202→    }
   203→   203→    // Type filter (up/pullback/down/risk)
   204→   204→    const edgeSet = new Set<number>();
   205→   205→    const nodeSet = new Set<string>(['ROOT']);
   206→   206→    edges.forEach((e, i) => {
   207→   207→      if (e.type === activeFilter) {
   208→   208→        edgeSet.add(i);
   209→   209→        nodeSet.add(e.from);
   210→   210→        nodeSet.add(e.to);
   211→   211→      }
   212→   212→    });
   213→   213→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   214→   214→  }, [activeFilter, edges, nodes]);
   215→   215→
   216→   216→  // ── Draw SVG edges ───────────────────────────────────────────────
   217→   217→  const drawEdges = useCallback(() => {
   218→   218→    const svg = svgRef.current;
   219→   219→    const graph = graphRef.current;
   220→   220→    if (!svg || !graph || edges.length === 0) return;
   221→   221→
   222→   222→    let markersSvg = '';
   223→   223→    const seenColors = new Set<string>();
   224→   224→    for (const e of edges) {
   225→   225→      const c = EDGE_COLORS[e.type];
   226→   226→      if (c && !seenColors.has(e.type)) {
   227→   227→        seenColors.add(e.type);
   228→   228→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="8" markerHeight="8" refX="7" refY="4" orient="auto"><path d="M0,0 L8,4 L0,8 z" fill="${c}"/></marker>`;
   229→   229→      }
   230→   230→    }
   231→   231→
   232→   232→    let pathsSvg = '';
   233→   233→    const graphRect = graph.getBoundingClientRect();
   234→   234→
   235→   235→    for (let ei = 0; ei < edges.length; ei++) {
   236→   236→      const { from: fromId, to: toId, label, type } = edges[ei];
   237→   237→      const isVisible = visibleEdgeIndices.includes(ei);
   238→   238→      const fromEl = nodeRefs.current[fromId];
   239→   239→      const toEl = nodeRefs.current[toId];
   240→   240→      if (!fromEl || !toEl) continue;
   241→   241→
   242→   242→      const fRect = fromEl.getBoundingClientRect();
   243→   243→      const tRect = toEl.getBoundingClientRect();
   244→   244→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   245→   245→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   246→   246→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   247→   247→      const by = tRect.top - graphRect.top + tRect.height / 2;
   248→   248→      const dx = bx - ax;
   249→   249→      const dy = by - ay;
   250→   250→      const dist = Math.sqrt(dx * dx + dy * dy);
   251→   251→      if (dist === 0) continue;
   252→   252→
   253→   253→      // Curved edges with bend proportional to distance
   254→   254→      const bend = Math.min(40, Math.max(12, dist * 0.08));
   255→   255→      const mx = (ax + bx) / 2;
   256→   256→      const my = (ay + by) / 2;
   257→   257→      const nx = -dy / dist;
   258→   258→      const ny = dx / dist;
   259→   259→      const cx = mx + nx * bend;
   260→   260→      const cy = my + ny * bend;
   261→   261→
   262→   262→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   263→   263→      const prob = edgeProbabilities[ei] ?? 0;
   264→   264→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   265→   265→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   266→   266→
   267→   267→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.7 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   268→   268→
   269→   269→      // Edge labels — skip deterministic root→branch and branch→event edges (index < 6)
   270→   270→      if (isVisible && ei >= 6) {
   271→   271→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#374151" font-size="8" text-anchor="middle" paint-order="stroke" stroke="#ffffff" stroke-width="3" stroke-linejoin="round" opacity="0.8" data-type="${type}" class="edge-label">${label}</text>`;
   272→   272→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#ffffff" stroke-width="2.5" stroke-linejoin="round" opacity="0.85" class="edge-prob">${probLabel}</text>`;
   273→   273→      }
   274→   274→    }
   275→   275→
   276→   276→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   277→   277→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   278→   278→
   279→   279→  useEffect(() => { drawEdges(); }, [drawEdges]);
   280→   280→
   281→   281→  useEffect(() => {
   282→   282→    const el = shellRef.current;
   283→   283→    if (!el) return;
   284→   284→    const observer = new ResizeObserver(() => drawEdges());
   285→   285→    observer.observe(el);
   286→   286→    return () => observer.disconnect();
   287→   287→  }, [drawEdges]);
   288→   288→
   289→   289→
   290→   290→  // ── Detail panel content ─────────────────────────────────────────
   291→   291→  const detailContent = useMemo(() => {
   292→   292→    if (!selectedNode) return null;
   293→   293→    const node = nodeMap[selectedNode];
   294→   294→    if (!node) return null;
   295→   295→
   296→   296→    // For terminal scenario nodes, show branch breakdown
   297→   297→    if (node.isTerminal && SCENARIO_KEYS.includes(selectedNode as typeof SCENARIO_KEYS[number])) {
   298→   298→      const contrib = pathContributions[selectedNode] ?? { trend: 0, breakout: 0, reversal: 0 };
   299→   299→      const meta = SCENARIO_META[selectedNode];
   300→   300→      const s = scenarios[selectedNode as keyof typeof scenarios];
   301→   301→      return (
   302→   302→        <div className="space-y-3">
   303→   303→          <h3 className="text-sm font-bold" style={{ color: meta.color }}>{node.title}</h3>
   304→   304→          <p className="text-xs" style={{ color: C.cardSubFg }}>{node.desc}</p>
   305→   305→          <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border bg-gray-50 text-gray-500">{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   306→   306→          <div className="text-xs" style={{ color: C.cardSubFg }} dir="ltr">{nodeValues[selectedNode] ?? '--'}</div>
   307→   307→
   308→   308→          <div className="mt-3 pt-3" style={{ borderTop: `1px solid ${C.cardBorder}` }}>
   309→   309→            <p className="text-xs font-bold mb-2" style={{ color: C.cardFg }}>سهم هر استراتژی:</p>
   310→   310→            {Object.entries(BRANCH_META).map(([bKey, bMeta]) => {
   311→   311→              const val = contrib[bKey as 'trend' | 'breakout' | 'reversal'];
   312→   312→              const pct = (val * 100).toFixed(1);
   313→   313→              return (
   314→   314→                <div key={bKey} className="flex items-center justify-between py-1.5" style={{ borderBottom: `1px dashed ${C.cardBorder}` }}>
   315→   315→                  <div className="flex items-center gap-1.5">
   316→   316→                    <span className="inline-block w-2 h-2 rounded-full" style={{ background: bMeta.color }} />
   317→   317→                    <span className="text-[11px]" style={{ color: C.cardFg }}>{bMeta.label}</span>
   318→   318→                  </div>
   319→   319→                  <span className="text-[11px] font-bold" style={{ color: bMeta.color }}>{toPersianDigits(pct)}٪</span>
   320→   320→                </div>
   321→   321→              );
   322→   322→            })}
   323→   323→            <div className="flex items-center justify-between py-1.5">
   324→   324→              <span className="text-[11px] font-bold" style={{ color: C.cardFg }}>مجموع</span>
   325→   325→              <span className="text-[11px] font-black" style={{ color: meta.color }}>{toFa(s?.probability ?? 0)}٪</span>
   326→   326→            </div>
   327→   327→          </div>
   328→   328→        </div>
   329→   329→      );
   330→   330→    }
   331→   331→
   332→   332→    // For non-terminal nodes, show incoming/outgoing edges
   333→   333→    const inputs = edges.filter(e => e.to === selectedNode);
   334→   334→    const outputs = edges.filter(e => e.from === selectedNode);
   335→   335→
   336→   336→    return (
   337→   337→      <div className="space-y-3">
   338→   338→        <h3 className="text-sm font-bold" style={{ color: C.cardFg }}>{node.title}</h3>
   339→   339→        <p className="text-xs" style={{ color: C.cardSubFg }}><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   340→   340→        <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border" style={{ borderColor: C.cardBorder, background: C.cardBorder, color: C.cardSubFg }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   341→   341→        <p className="text-xs leading-relaxed" style={{ color: C.cardSubFg }}>{node.desc}</p>
   342→   342→        {inputs.length > 0 && (
   343→   343→          <div>
   344→   344→            <p className="text-xs font-medium mb-1" style={{ color: C.cardFg }}>مسیرهای ورودی ({toFa(inputs.length)}):</p>
   345→   345→            <ul className="space-y-1">
   346→   346→              {inputs.map((e, i) => {
   347→   347→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   348→   348→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   349→   349→                return (
   350→   350→                  <li key={i} className="text-[11px] leading-relaxed pt-1.5" style={{ color: C.cardSubFg, borderTop: `1px dashed ${C.cardBorder}` }}>
   351→   351→                    <b style={{ color: C.cardFg }}>{SCENARIO_DISPLAY[e.from] || e.from} ← {SCENARIO_DISPLAY[e.to] || e.to}</b>
   352→   352→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${edgeColor}20`, color: edgeColor }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   353→   353→                    <br />{e.label}
   354→   354→                  </li>
   355→   355→                );
   356→   356→              })}
   357→   357→            </ul>
   358→   358→          </div>
   359→   359→        )}
   360→   360→        {outputs.length > 0 && (
   361→   361→          <div>
   362→   362→            <p className="text-xs font-medium mb-1" style={{ color: C.cardFg }}>مسیرهای خروجی ({toFa(outputs.length)}):</p>
   363→   363→            <ul className="space-y-1">
   364→   364→              {outputs.map((e, i) => {
   365→   365→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   366→   366→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   367→   367→                return (
