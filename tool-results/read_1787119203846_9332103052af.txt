   136→   136→/* ────────────────────────────────────────────────────────── COMPONENT ───── */
   137→   137→
   138→   138→export default function CandlestickChart({ data, ta, height = 520 }: CandlestickChartProps) {
   139→   139→  const chartContainerRef = useRef<HTMLDivElement>(null);
   140→   140→  const chartParentRef = useRef<HTMLDivElement>(null);
   141→   141→  const svgRef = useRef<SVGSVGElement>(null);
   142→   142→  const chartRef = useRef<IChartApi | null>(null);
   143→   143→  const subscriberRef = useRef<{ dispose: () => void } | null>(null);
   144→   144→
   145→   145→  const [drawings, setDrawings] = useState<Drawing[]>([]);
   146→   146→  const [activeTool, setActiveTool] = useState<ToolType>('cursor');
   147→   147→  const [activeColor, setActiveColor] = useState(PALETTE[0]);
   148→   148→  const [isDrawing, setIsDrawing] = useState(false);
   149→   149→  const [currentDrawing, setCurrentDrawing] = useState<Drawing | null>(null);
   150→   150→  const [textInput, setTextInput] = useState<{ x: number; y: number; drawingId: string } | null>(null);
   151→   151→  const [textValue, setTextValue] = useState('');
   152→   152→  const [renderKey, setRenderKey] = useState(0);
   153→   153→  const [chartSize, setChartSize] = useState({ w: 0, h: 0 });
   154→   154→
   155→   155→  /* ── Helper: coordinate conversion ─────────────────────────── */
   156→   156→  const getChartPoint = useCallback((clientX: number, clientY: number): DrawingPoint | null => {
   157→   157→    const chart = chartRef.current;
   158→   158→    const container = chartContainerRef.current;
   159→   159→    if (!chart || !container) return null;
   160→   160→    const rect = container.getBoundingClientRect();
   161→   161→    const x = clientX - rect.left;
   162→   162→    const y = clientY - rect.top;
   163→   163→    const price = chart.priceScale('right').coordinateToPrice(y);
   164→   164→    const time = chart.timeScale().coordinateToTime(x);
   165→   165→    if (price === null || time === null) return null;
   166→   166→    return { time, price };
   167→   167→  }, []);
   168→   168→
   169→   169→  /* ── SVG Rendering ────────────────────────────────────────── */
   170→   170→  const renderDrawings = useCallback(() => {
   171→   171→    const chart = chartRef.current;
   172→   172→    const svg = svgRef.current;
   173→   173→    const container = chartContainerRef.current;
   174→   174→    if (!chart || !svg || !container || drawings.length === 0) {
   175→   175→      if (svg) svg.innerHTML = '';
   176→   176→      return;
   177→   177→    }
   178→   178→
   179→   179→    const rect = container.getBoundingClientRect();
   180→   180→    const w = rect.width;
   181→   181→    const h = rect.height;
   182→   182→    if (w === 0 || h === 0) return;
   183→   183→
   184→   184→    let html = '';
   185→   185→
   186→   186→    // Defs for arrow markers
   187→   187→    html += `<defs>`;
   188→   188→    PALETTE.forEach((c, i) => {
   189→   189→      html += `<marker id="arrow-${c.replace('#','')}" markerWidth="8" markerHeight="6" refX="7" refY="3" orient="auto"><polygon points="0 0, 8 3, 0 6" fill="${c}" /></marker>`;
   190→   190→    });
   191→   191→    html += `</defs>`;
   192→   192→
   193→   193→    const allDrawings = [...drawings];
   194→   194→    if (currentDrawing) allDrawings.push(currentDrawing);
   195→   195→
   196→   196→    for (const d of allDrawings) {
   197→   197→      if (!d.completed && d.points.length === 0) continue;
   198→   198→      const col = d.color;
   199→   199→      const lw = d.lineWidth;
   200→   200→
   201→   201→      switch (d.type) {
   202→   202→        case 'trendline': {
   203→   203→          if (d.points.length < 1) break;
   204→   204→          const p0 = d.points[0];
   205→   205→          const p1 = d.points.length > 1 ? d.points[d.points.length - 1] : p0;
   206→   206→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   207→   207→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   208→   208→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   209→   209→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   210→   210→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   211→   211→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" />`;
   212→   212→          break;
   213→   213→        }
   214→   214→        case 'hline': {
   215→   215→          if (d.points.length < 1) break;
   216→   216→          const p = d.points[0];
   217→   217→          const y = chart.priceScale('right').priceToCoordinate(p.price);
   218→   218→          if (y == null) break;
   219→   219→          html += `<line x1="0" y1="${y}" x2="${w}" y2="${y}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   220→   220→          html += `<text x="4" y="${y - 4}" fill="${col}" font-size="11" font-family="sans-serif">${p.price.toFixed(0)}</text>`;
   221→   221→          break;
   222→   222→        }
   223→   223→        case 'vline': {
   224→   224→          if (d.points.length < 1) break;
   225→   225→          const p = d.points[0];
   226→   226→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   227→   227→          if (x == null) break;
   228→   228→          html += `<line x1="${x}" y1="0" x2="${x}" y2="${h}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   229→   229→          break;
   230→   230→        }
   231→   231→        case 'fibonacci': {
   232→   232→          if (d.points.length < 2) break;
   233→   233→          const p0 = d.points[0];
   234→   234→          const p1 = d.points[1];
   235→   235→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   236→   236→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   237→   237→          if (x0 == null || x1 == null) break;
   238→   238→          const priceHigh = Math.max(p0.price, p1.price);
   239→   239→          const priceLow = Math.min(p0.price, p1.price);
   240→   240→          const priceRange = priceHigh - priceLow;
   241→   241→          if (priceRange === 0) break;
   242→   242→
   243→   243→          // Filled background
   244→   244→          const yTop = chart.priceScale('right').priceToCoordinate(priceHigh);
   245→   245→          const yBottom = chart.priceScale('right').priceToCoordinate(priceLow);
   246→   246→          if (yTop == null || yBottom == null) break;
   247→   247→          html += `<rect x="${Math.min(x0,x1)}" y="${yTop}" width="${Math.abs(x1-x0)}" height="${yBottom - yTop}" fill="${col}" opacity="0.04" />`;
   248→   248→
   249→   249→          for (const level of FIB_LEVELS) {
   250→   250→            const price = priceHigh - level.pct * priceRange;
   251→   251→            const y = chart.priceScale('right').priceToCoordinate(price);
   252→   252→            if (y == null) continue;
   253→   253→            html += `<line x1="${Math.min(x0,x1) - 10}" y1="${y}" x2="${Math.max(x0,x1) + 10}" y2="${y}" stroke="${col}" stroke-width="1" opacity="0.6" />`;
   254→   254→            html += `<rect x="${Math.max(x0,x1) + 12}" y="${y - 8}" width="56" height="16" fill="#ffffff" rx="2" />`;
   255→   255→            html += `<text x="${Math.max(x0,x1) + 14}" y="${y + 4}" fill="${col}" font-size="10" font-family="sans-serif">${level.label} ${price.toFixed(0)}</text>`;
   256→   256→          }
   257→   257→          break;
   258→   258→        }
   259→   259→        case 'rectangle': {
   260→   260→          if (d.points.length < 2) break;
   261→   261→          const p0 = d.points[0];
   262→   262→          const p1 = d.points[d.points.length - 1];
   263→   263→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   264→   264→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   265→   265→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   266→   266→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   267→   267→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   268→   268→          const rx = Math.min(x0, x1);
   269→   269→          const ry = Math.min(y0, y1);
   270→   270→          const rw = Math.abs(x1 - x0);
   271→   271→          const rh = Math.abs(y1 - y0);
   272→   272→          html += `<rect x="${rx}" y="${ry}" width="${rw}" height="${rh}" fill="${col}" fill-opacity="0.08" stroke="${col}" stroke-width="${lw}" rx="2" />`;
   273→   273→          break;
   274→   274→        }
   275→   275→        case 'text': {
   276→   276→          if (d.points.length < 1 || !d.text) break;
   277→   277→          const p = d.points[0];
   278→   278→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   279→   279→          const y = chart.priceScale('right').priceToCoordinate(p.price);
   280→   280→          if (x == null || y == null) break;
   281→   281→          html += `<rect x="${x + 4}" y="${y - 16}" width="${d.text.length * 8 + 12}" height="22" fill="#ffffff" stroke="${col}" stroke-width="1" rx="3" />`;
   282→   282→          html += `<text x="${x + 10}" y="${y + 1}" fill="${col}" font-size="12" font-family="sans-serif">${d.text}</text>`;
   283→   283→          break;
   284→   284→        }
   285→   285→        case 'brush': {
   286→   286→          if (d.points.length < 2) break;
   287→   287→          let pts = '';
   288→   288→          let valid = true;
   289→   289→          for (const p of d.points) {
   290→   290→            const x = chart.timeScale().timeToCoordinate(p.time as Time);
   291→   291→            const y = chart.priceScale('right').priceToCoordinate(p.price);
   292→   292→            if (x == null || y == null) { valid = false; break; }
   293→   293→            pts += `${x},${y} `;
   294→   294→          }
   295→   295→          if (!valid || pts.length === 0) break;
   296→   296→          html += `<polyline points="${pts.trim()}" fill="none" stroke="${col}" stroke-width="${lw}" stroke-linecap="round" stroke-linejoin="round" />`;
   297→   297→          break;
   298→   298→        }
   299→   299→        case 'measure': {
   300→   300→          if (d.points.length < 2) break;
   301→   301→          const p0 = d.points[0];
   302→   302→          const p1 = d.points[d.points.length - 1];
   303→   303→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   304→   304→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   305→   305→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   306→   306→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   307→   307→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   308→   308→
   309→   309→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="4,2" />`;
   310→   310→          // Price difference
   311→   311→          const diff = p1.price - p0.price;
   312→   312→          const pctChg = p0.price !== 0 ? ((diff / p0.price) * 100) : 0;
   313→   313→          const bars = Math.abs(typeof p1.time === 'number' && typeof p0.time === 'number' ? Math.round(p1.time - p0.time) : 0);
   314→   314→          const midX = (x0 + x1) / 2;
   315→   315→          const midY = (y0 + y1) / 2;
   316→   316→          const label = `${diff >= 0 ? '+' : ''}${diff.toFixed(0)} (${pctChg >= 0 ? '+' : ''}${pctChg.toFixed(1)}%) ${bars}bar`;
   317→   317→          html += `<rect x="${midX - 40}" y="${midY - 24}" width="80" height="20" fill="#ffffff" stroke="${col}" stroke-width="1" rx="3" />`;
   318→   318→          html += `<text x="${midX - 36}" y="${midY - 10}" fill="${col}" font-size="10" font-family="sans-serif">${label}</text>`;
   319→   319→          break;
   320→   320→        }
   321→   321→        case 'arrow': {
   322→   322→          if (d.points.length < 2) break;
   323→   323→          const p0 = d.points[0];
   324→   324→          const p1 = d.points[d.points.length - 1];
   325→   325→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   326→   326→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   327→   327→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   328→   328→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   329→   329→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   330→   330→          const markerId = `arrow-${col.replace('#','')}`;
   331→   331→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" marker-end="url(#${markerId})" />`;
   332→   332→          break;
   333→   333→        }
   334→   334→      }
   335→   335→    }
   336→   336→
   337→   337→    svg.innerHTML = html;
   338→   338→  }, [drawings, currentDrawing]);
   339→   339→
   340→   340→  /* ── Chart Build ──────────────────────────────────────────── */
   341→   341→  const build = useCallback(() => {
   342→   342→    const chartEl = chartContainerRef.current;
   343→   343→    if (!chartEl || data.length === 0) return;
   344→   344→
   345→   345→    if (chartRef.current) {
   346→   346→      chartRef.current.remove();
   347→   347→      chartRef.current = null;
   348→   348→    }
   349→   349→
   350→   350→    const chart = createChart(chartEl, {
   351→   351→      layout: { background: { type: ColorType.Solid, color: BG }, textColor: TXT, fontSize: 11 },
   352→   352→      grid: { vertLines: { color: GRID }, horzLines: { color: GRID } },
   353→   353→      crosshair: { mode: CrosshairMode.Normal },
   354→   354→      rightPriceScale: {
   355→   355→        borderColor: '#e5e7eb',
   356→   356→        scaleMargins: { top: 0.05, bottom: 0.3 },
   357→   357→        priceFormatter: (price: number) => toPersianDigits(Math.round(price).toLocaleString('en-US')),
   358→   358→      },
   359→   359→      localization: {
   360→   360→        priceFormatter: (price: number) => toPersianDigits(Math.round(price).toLocaleString('en-US')),
   361→   361→      },
   362→   362→      timeScale: {
   363→   363→        borderColor: '#e5e7eb',
   364→   364→        rightOffset: 5,
   365→   365→        barSpacing: 6,
   366→   366→        timeVisible: false,
   367→   367→        tickMarkFormatter: (time: Time, tickMarkType: TickMarkType, _locale: string) => {
   368→   368→          return shamsiTickFormatter(time, tickMarkType as unknown as number);
   369→   369→        },
   370→   370→      },
   371→   371→      width: chartEl.clientWidth,
   372→   372→      height,
   373→   373→    });
   374→   374→    chartRef.current = chart;
   375→   375→
   376→   376→    // Use real dates when available, fall back to numeric index
   377→   377→    const candles = data.map((d, i) => ({
   378→   378→      time: (d.date && d.date.includes('-') ? d.date : undefined) as Time || (i as Time),
   379→   379→      open: d.open,
   380→   380→      high: d.high,
   381→   381→      low: d.low,
   382→   382→      close: d.close,
   383→   383→    }));
   384→   384→
   385→   385→    const cs = chart.addSeries(CandlestickSeries, {
   386→   386→      upColor: BULL,
   387→   387→      downColor: BEAR,
   388→   388→      borderUpColor: BULL,
   389→   389→      borderDownColor: BEAR,
   390→   390→      wickUpColor: BULL,
   391→   391→      wickDownColor: BEAR,
   392→   392→    });

   408→   408→
   409→   409→    // TA overlays

   426→   426→          cs.createPriceLine({ price: bb.upper, color: 'rgba(161,98,255,0.5)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Upper' });

   429→   429→        if (bb.lower > 0)

   431→   431→      }
   432→   432→      if (ta.sar > 0)
   433→   433→        cs.createPriceLine({ price: ta.sar, color: 'rgba(75,85,99,0.7)', lineWidth: 1, lineStyle: 1, axisLabelVisible: true, title: 'SAR' });
   434→   434→
   435→   435→      // S/R with strength
   436→   436→      const resistanceStrengths = ta.resistanceStrengths || [];
   437→   437→      const supportStrengths = ta.supportStrengths || [];

   439→   439→      resistanceStrengths.forEach((r, i) => {
   440→   440→        if (r.price <= 0) return;
   441→   441→        const lw = r.isTarget ? 3 : Math.max(1, Math.round(r.strength / 3));
   442→   442→        const ls = r.isTarget ? 0 : 2;
   443→   443→        const title = `R${i + 1} (${r.strength}/10)`;
   444→   444→        cs.createPriceLine({
   445→   445→          price: r.price,
   446→   446→          color: r.isTarget ? '#ff6b6b' : BEAR,
   447→   447→          lineWidth: lw,
   448→   448→          lineStyle: ls as 0 | 1 | 2,
   449→   449→          axisLabelVisible: i < 6,
   450→   450→          title,
   451→   451→        });
   452→   452→      });
   453→   453→
   454→   454→      supportStrengths.forEach((s, i) => {
   455→   455→        if (s.price <= 0) return;
   456→   456→        const lw = s.isTarget ? 3 : Math.max(1, Math.round(s.strength / 3));
   457→   457→        const ls = s.isTarget ? 0 : 2;
   458→   458→        const title = `S${i + 1} (${s.strength}/10)`;
   459→   459→        cs.createPriceLine({
   460→   460→          price: s.price,
   461→   461→          color: s.isTarget ? '#6bffb8' : BULL,
   462→   462→          lineWidth: lw,
   463→   463→          lineStyle: ls as 0 | 1 | 2,
   464→   464→          axisLabelVisible: i < 6,
   465→   465→          title,
   466→   466→        });
   467→   467→      });
   468→   468→    }
   469→   469→
   470→   470→    chart.timeScale().fitContent();
   471→   471→
   472→   472→    // Subscribe to visible range changes to re-render SVG
   473→   473→    requestAnimationFrame(() => {
   474→   474→      if (chartRef.current) {
   475→   475→        subscriberRef.current = chartRef.current.timeScale().subscribeVisibleLogicalRangeChange(() => {
   476→   476→          setRenderKey((k) => k + 1);
   477→   477→        });
   478→   478→      }
   479→   479→    });
   480→   480→
   481→   481→    // Set initial chart size (deferred to avoid React batching issue)
   482→   482→    requestAnimationFrame(() => {
   483→   483→      if (chartContainerRef.current) {
   484→   484→        setChartSize({ w: chartContainerRef.current.clientWidth, h: height });
   485→   485→      }
   486→   486→    });
   487→   487→  }, [data, ta, height]);
   488→   488→
   489→   489→  /* ── Re-render SVG when renderKey changes ────────────────── */
   490→   490→  useEffect(() => {
   491→   491→    renderDrawings();
   492→   492→  }, [renderKey, renderDrawings]);
   493→   493→
   494→   494→  /* ── Init chart ───────────────────────────────────────────── */
   495→   495→  useEffect(() => {
   496→   496→    const el = chartContainerRef.current;
   497→   497→    if (!el) return;
   498→   498→    const ro = new ResizeObserver((entries) => {
   499→   499→      for (const e of entries) {
   500→   500→        const { width: w, height: h } = e.contentRect;
   501→   501→        if (w > 0 && h > 0) {
   502→   502→          chartRef.current?.applyOptions({ width: w, height: h });
   503→   503→          setChartSize({ w, h });
   504→   504→        }
   505→   505→      }
   506→   506→    });
   507→   507→    ro.observe(el);
   508→   508→    build();
   509→   509→    return () => {
   510→   510→      ro.disconnect();
   511→   511→      subscriberRef.current?.dispose();
   512→   512→      subscriberRef.current = null;
   513→   513→      if (chartRef.current) {
   514→   514→        chartRef.current.remove();
   515→   515→        chartRef.current = null;
   516→   516→      }
   517→   517→    };
   518→   518→  }, [build]);
   519→   519→
   520→   520→  /* ── Refs for native event handlers (avoid stale closures) ─── */
   521→   521→  const activeToolRef = useRef<ToolType>('cursor');
   522→   522→  const activeColorRef = useRef(PALETTE[0]);
   523→   523→  const isDrawingRef = useRef(false);
   524→   524→  const currentDrawingRef = useRef<Drawing | null>(null);
   525→   525→  const textInputRef = useRef<{ x: number; y: number; drawingId: string } | null>(null);
   526→   526→
   527→   527→  useEffect(() => { activeToolRef.current = activeTool; }, [activeTool]);
   528→   528→  useEffect(() => { activeColorRef.current = activeColor; }, [activeColor]);
   529→   529→  useEffect(() => { isDrawingRef.current = isDrawing; }, [isDrawing]);
   530→   530→  useEffect(() => { currentDrawingRef.current = currentDrawing; }, [currentDrawing]);
   531→   531→  useEffect(() => { textInputRef.current = textInput; }, [textInput]);
   532→   532→
   533→   533→  /* ── Native event handling via parent div + window ────────── */
   534→   534→  useEffect(() => {
   535→   535→    const parent = chartParentRef.current;
   536→   536→    if (!parent) return;
   537→   537→
   538→   538→    const onDown = (e: MouseEvent) => {
   539→   539→      const tool = activeToolRef.current;
   540→   540→      if (tool === 'cursor') return;
   541→   541→      if (textInputRef.current) return;
   542→   542→      const point = getChartPoint(e.clientX, e.clientY);
   543→   543→      if (!point) return;
   544→   544→
   545→   545→      if (tool === 'hline') {
   546→   546→        const drawing: Drawing = { id: Date.now().toString(), type: 'hline', points: [point], color: activeColorRef.current, lineWidth: 1, completed: true };
   547→   547→        setDrawings((prev) => [...prev, drawing]);
   548→   548→        return;
   549→   549→      }
   550→   550→      if (tool === 'text') {
   551→   551→        const container = chartContainerRef.current;
   552→   552→        if (!container) return;
   553→   553→        const rect = container.getBoundingClientRect();
   554→   554→        const tempDrawing: Drawing = { id: Date.now().toString(), type: 'text', points: [point], color: activeColorRef.current, lineWidth: 1, text: '', completed: false };
   555→   555→        setCurrentDrawing(tempDrawing);
   556→   556→        currentDrawingRef.current = tempDrawing;
   557→   557→        isDrawingRef.current = true;
   558→   558→        setIsDrawing(true);
   559→   559→        setTextInput({ x: e.clientX - rect.left, y: e.clientY - rect.top, drawingId: tempDrawing.id });
   560→   560→        setTextValue('');
   561→   561→        return;
   562→   562→      }
   563→   563→      const drawing: Drawing = { id: Date.now().toString(), type: tool, points: [point], color: activeColorRef.current, lineWidth: tool === 'brush' ? 2 : 1, completed: false };
   564→   564→      setCurrentDrawing(drawing);
   565→   565→      currentDrawingRef.current = drawing;
   566→   566→      isDrawingRef.current = true;
   567→   567→      setIsDrawing(true);
   568→   568→    };
   569→   569→
   570→   570→    const onMove = (e: MouseEvent) => {
   571→   571→      if (!isDrawingRef.current || !currentDrawingRef.current) return;
   572→   572→      const point = getChartPoint(e.clientX, e.clientY);
   573→   573→      if (!point) return;
   574→   574→      const cd = currentDrawingRef.current;
   575→   575→      if (cd.type === 'brush') {
   576→   576→        const updated = { ...cd, points: [...cd.points, point] };
   577→   577→        currentDrawingRef.current = updated;
   578→   578→        setCurrentDrawing(updated);
   579→   579→      } else if (cd.type !== 'text' && cd.type !== 'hline') {
   580→   580→        const updated = { ...cd, points: [cd.points[0], point] };
   581→   581→        currentDrawingRef.current = updated;
   582→   582→        setCurrentDrawing(updated);
   583→   583→      }
   584→   584→    };
   585→   585→
   586→   586→    const onUp = () => {
   587→   587→      if (!isDrawingRef.current || !currentDrawingRef.current) return;
   588→   588→      const cd = currentDrawingRef.current;
   589→   589→      if (cd.type === 'text') return;
   590→   590→      const completed: Drawing = { ...cd, completed: true };
   591→   591→      if (completed.points.length >= 2) {
   592→   592→        setDrawings((prev) => [...prev, completed]);
   593→   593→      }
   594→   594→      currentDrawingRef.current = null;
   595→   595→      isDrawingRef.current = false;
   596→   596→      setCurrentDrawing(null);
   597→   597→      setIsDrawing(false);
   598→   598→    };
   599→   599→
   600→   600→    parent.addEventListener('mousedown', onDown);
   601→   601→    window.addEventListener('mousemove', onMove);
   602→   602→    window.addEventListener('mouseup', onUp);
   603→   603→
   604→   604→    return () => {
   605→   605→      parent.removeEventListener('mousedown', onDown);
   606→   606→      window.removeEventListener('mousemove', onMove);
   607→   607→      window.removeEventListener('mouseup', onUp);
   608→   608→    };
   609→   609→  }, [getChartPoint]);
   610→   610→
   611→   611→  /* ── Text input submit ────────────────────────────────────── */
   612→   612→  const handleTextSubmit = useCallback(() => {
   613→   613→    if (!textInput || !textValue.trim() || !currentDrawing) {
   614→   614→      setTextInput(null);
   615→   615→      setCurrentDrawing(null);
   616→   616→      setIsDrawing(false);
   617→   617→      return;
   618→   618→    }
   619→   619→
   620→   620→    const completed: Drawing = {
   621→   621→      ...currentDrawing,
   622→   622→      text: textValue.trim(),
   623→   623→      completed: true,
   624→   624→    };
   625→   625→    setDrawings((prev) => [...prev, completed]);
   626→   626→    setTextInput(null);
   627→   627→    setTextValue('');
   628→   628→    setCurrentDrawing(null);
   629→   629→    setIsDrawing(false);
   630→   630→  }, [textInput, textValue, currentDrawing]);
   631→   631→
   632→   632→  const handleTextCancel = useCallback(() => {
   633→   633→    setTextInput(null);
   634→   634→    setTextValue('');
   635→   635→    setCurrentDrawing(null);
   636→   636→    setIsDrawing(false);
   637→   637→  }, []);
   638→   638→
   639→   639→  /* ── Delete / Clear ──────────────────────────────────────── */
   640→   640→  const handleDeleteLast = useCallback(() => {
   641→   641→    setDrawings((prev) => prev.slice(0, -1));
   642→   642→  }, []);
   643→   643→
   644→   644→  const handleClearAll = useCallback(() => {
   645→   645→    setDrawings([]);
   646→   646→  }, []);
   647→   647→
   648→   648→
   649→   649→  /* ── Render ───────────────────────────────────────────────── */
   650→   650→  return (
   651→   651→    <div className="w-full">
   652→   652→      {/* ── TOOLBAR ────────────────────────────────────────── */}
   653→   653→      <div
   654→   654→        className="flex items-center gap-1 px-2 py-1.5 rounded-t-lg border border-b-0 overflow-x-auto flex-nowrap"
   655→   655→        style={{ backgroundColor: TOOLBAR_BG, borderColor: TOOLBAR_BORDER }}
   656→   656→      >
   657→   657→        {/* Tool buttons */}
   658→   658→        {TOOLS.map((tool) => (
   659→   659→          <button
   660→   660→            key={tool.id}
   661→   661→            onClick={() => {
   662→   662→              setActiveTool(tool.id);
   663→   663→              // Cancel ongoing drawing when switching tools
   664→   664→              if (isDrawing) {
   665→   665→                setIsDrawing(false);
   666→   666→                setCurrentDrawing(null);
   667→   667→                setTextInput(null);
   668→   668→              }
   669→   669→            }}
   670→   670→            title={tool.label}
   671→   671→            className={`shrink-0 flex items-center justify-center w-8 h-8 rounded-md transition-all cursor-pointer ${
   672→   672→              activeTool === tool.id
   673→   673→                ? 'bg-amber-50 text-amber-700 ring-1 ring-amber-200'
   674→   674→                : 'text-gray-500 hover:bg-gray-100 hover:text-gray-900'
   675→   675→            }`}
   676→   676→          >
   677→   677→            {tool.icon}
   678→   678→          </button>
   679→   679→        ))}
   680→   680→
   681→   681→        {/* Separator */}
   682→   682→        <div className="w-px h-6 bg-gray-300 mx-1 shrink-0" />
   683→   683→
   684→   684→        {/* Color picker */}
   685→   685→        {PALETTE.map((c) => (
   686→   686→          <button
   687→   687→            key={c}
   688→   688→            onClick={() => setActiveColor(c)}
   689→   689→            title={c}
   690→   690→            className={`shrink-0 w-6 h-6 rounded-full border-2 transition-all cursor-pointer ${
   691→   691→              activeColor === c ? 'border-gray-800 scale-110' : 'border-transparent hover:scale-105'
   692→   692→            }`}
   693→   693→            style={{ backgroundColor: c }}
   694→   694→          />
   695→   695→        ))}
   696→   696→
   697→   697→        {/* Separator */}
   698→   698→        <div className="w-px h-6 bg-gray-300 mx-1 shrink-0" />
   699→   699→
   700→   700→        {/* Delete last */}
   701→   701→        <button
   702→   702→          onClick={handleDeleteLast}
   703→   703→          title="حذف آخر"
   704→   704→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-500 hover:bg-red-50 hover:text-red-700 transition-all cursor-pointer"
   705→   705→        >
   706→   706→          <Trash2 className="w-3.5 h-3.5" />
   707→   707→        </button>
   708→   708→
   709→   709→        {/* Clear all */}
   710→   710→        <button
   711→   711→          onClick={handleClearAll}
   712→   712→          title="پاک کردن همه"
   713→   713→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-500 hover:bg-red-50 hover:text-red-700 transition-all cursor-pointer"
   714→   714→        >

   772→   772→            />
   773→   773→          </div>
   774→   774→        )}
   775→   775→      </div>
   776→   776→    </div>
   777→   777→  );
   778→   778→}
