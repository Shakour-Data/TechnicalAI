    15→  targetMin: number;
    16→  targetMax: number;

    18→}

    21→  symbolName: string;
    22→  candles: CandleData[];
    23→  currentPrice: number;
    24→  resistances: number[];
    25→  supports: number[];
    26→  ma21: number;
    27→  ma100: number;
    28→  rsi: number;
    29→  mfi: number;
    30→  cci: number;
    31→  adx: number;
    32→  stochK: number;
    33→  stochD: number;
    34→  macdLine: number;
    35→  macdSignal: number;
    36→  macdHist: number;
    37→  diPlus: number;
    38→  diMinus: number;
    39→  sar: number;
    40→  atr: number;
    41→  obv: number;
    42→  hasVolume: boolean;
    43→  bollingerUpper: number;
    44→  bollingerMiddle: number;
    45→  bollingerLower: number;
    46→  trendDirection: string;
    47→  trendAngle: number;
    48→  trendR2: number;
    49→  overallSignal: string;
    50→  scenarios: {
    51→    R1: Scenario;
    52→    R2: Scenario;

    54→    R4: Scenario;
    55→    R5: Scenario;
    56→  };
    57→  supportStrengths: { price: number; strength: number; isTarget: boolean }[];
    58→  resistanceStrengths: { price: number; strength: number; isTarget: boolean }[];
    59→  priceTargets: { price: number; strength: number; isTarget: boolean }[];
    60→}

    65→
    66→const toFa = (n: number) => Math.round(n).toLocaleString('fa-IR');
    67→
    68→const SCENARIO_KEYS = ['R1', 'R2', 'R3', 'R4', 'R5'] as const;
    69→
    70→const SCENARIO_META: Record<string, { label: string; type: string; border: string; badgeBg: string; badgeColor: string }> = {
    71→  R1: { label: 'تداوم صعود هیجانی', type: 'up', border: '#59e39b', badgeBg: 'rgba(89,227,155,0.2)', badgeColor: '#59e39b' },
    72→  R2: { label: 'پولبک سالم', type: 'pullback', border: '#ffb25f', badgeBg: 'rgba(255,178,95,0.2)', badgeColor: '#ffb25f' },
    73→  R3: { label: 'اصلاح کنترل‌شده', type: 'down', border: '#ff758a', badgeBg: 'rgba(255,117,138,0.2)', badgeColor: '#ff758a' },
    74→  R4: { label: 'اصلاح عمیق', type: 'down', border: '#ffb11b', badgeBg: 'rgba(255,177,27,0.2)', badgeColor: '#ffb11b' },
    75→  R5: { label: 'تضعیف ساختار', type: 'down', border: '#ef4d62', badgeBg: 'rgba(239,77,98,0.2)', badgeColor: '#ef4d62' },
    76→};
    77→
    78→const STRATEGY_MAP: Record<string, { text: string; tagCls: string }> = {
    79→  R1: { text: 'صعودی قوی — احتمال بالای عبور از مقاومت‌ها', tagCls: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30' },
    80→  R2: { text: 'صعود تدریجی — ورود در اصلاح توصیه می‌شود', tagCls: 'bg-cyan-500/15 text-cyan-400 border-cyan-500/30' },
    81→  R3: { text: 'بازار رنج — منتظر خروج از محدوده بمانید', tagCls: 'bg-amber-500/15 text-amber-400 border-amber-500/30' },
    82→  R4: { text: 'اصلاحی — احتیاط و کاهش حجم معاملات', tagCls: 'bg-orange-500/15 text-orange-400 border-orange-500/30' },
    83→  R5: { text: 'نزولی قوی — خروج فوری توصیه می‌شود', tagCls: 'bg-red-500/15 text-red-400 border-red-500/30' },
    84→};
    85→
    86→// ═══════════════════════════════════════════════════════════════════════════════
    87→// Component
    88→// ═══════════════════════════════════════════════════════════════════════════════
    89→
    90→export default function VdesAnalysis(props: VdesAnalysisProps) {
    91→  const {
    92→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
    93→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
    94→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
    95→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
    96→    supportStrengths, resistanceStrengths, priceTargets,
    97→  } = props;
    98→
    99→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   100→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   101→  const S1_level = supports[0] ?? currentPrice * 0.95;
   102→  const S2_level = supports[1] ?? currentPrice * 0.90;
   103→  const S3_level = supports[2] ?? currentPrice * 0.85;
   104→
   105→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   106→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   107→
   108→  // ── Dominant scenario ──────────────────────────────────────────
   109→  let highestKey = 'R3';
   110→  let highestProb = 0;
   111→  for (const key of SCENARIO_KEYS) {
   112→    if (scenarios[key].probability > highestProb) {
   113→      highestProb = scenarios[key].probability;
   114→      highestKey = key;
   115→    }
   116→  }
   117→  const strategy = STRATEGY_MAP[highestKey];
   118→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   119→
   120→  // ── RSI signal ─────────────────────────────────────────────────
   121→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   122→  const rsiColor = rsi > 70 ? 'text-red-400' : rsi > 60 ? 'text-amber-400' : rsi > 40 ? 'text-gray-300' : rsi > 30 ? 'text-amber-400' : 'text-emerald-400';
   123→
   124→  // ── Stochastic signal ──────────────────────────────────────────
   125→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   126→
   127→  // ── MACD signal ────────────────────────────────────────────────
   128→  const macdSignalText = macdHist > 0 ? 'مومنتوم مثبت (هیستوگرام بالای صفر)' : 'مومنتوم منفی (هیستوگرام زیر صفر)';
   129→  const macdBullish = macdLine > macdSignal;
   130→
   131→  // ── Trend text ─────────────────────────────────────────────────
   132→  const trendText = trendDirection === 'up'
   133→    ? `صعودی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${(trendR2 * 100).toFixed(1)}%`
   134→    : trendDirection === 'down'
   135→    ? `نزولی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${(trendR2 * 100).toFixed(1)}%`
   136→    : 'خنثی و بدون جهت مشخص';
   137→
   138→  // ── DI signal ──────────────────────────────────────────────────
   139→  const diSignal = diPlus > diMinus
   140→    ? `DI+ (${toFa(diPlus)}) بالاتر از DI- (${toFa(diMinus)}) — فشار خرید غالب`
   141→    : `DI- (${toFa(diMinus)}) بالاتر از DI+ (${toFa(diPlus)}) — فشار فروش غالب`;
   142→
   143→  // ── SAR signal ─────────────────────────────────────────────────
   144→  const sarSignal = sar < currentPrice
   145→    ? `SAR (${toFa(sar)}) زیر قیمت — تأیید روند صعودی`
   146→    : `SAR (${toFa(sar)}) بالای قیمت — تأیید روند نزولی`;
   147→
   148→  // ── Bollinger Band position ────────────────────────────────────
   149→  const bbRange = bollingerUpper - bollingerLower;
   150→  const bbPos = bbRange > 0 ? ((currentPrice - bollingerLower) / bbRange * 100).toFixed(0) : '50';
   151→  const bbSignal = currentPrice > bollingerUpper ? 'بالای باند بالایی (اشباع خرید)'
   152→    : currentPrice < bollingerLower ? 'زیر باند پایینی (اشباع فروش)'
   153→    : `داخل باندها (${bbPos}٪ از بازه)`;
   154→
   155→  // ── TradingView scenario colors ────────────────────────────────
   156→  // Limit candles for chart rendering to prevent freeze on large datasets (e.g. derivatives with 10k+ candles)
   157→  const chartCandles = useMemo(() => {
   158→    if (candles.length <= 2000) return candles;
   159→    return candles.slice(-2000);
   160→  }, [candles]);
   161→
   162→  const tvScenarios = useMemo(() => {
   163→    const colors: Record<string, string> = {
   164→      R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62',
   165→    };
   166→    return Object.fromEntries(
   167→      SCENARIO_KEYS.map(k => [k, {
   168→        targetMin: scenarios[k].targetMin,
   169→        targetMax: scenarios[k].targetMax,
   170→        name: scenarios[k].name,
   171→        probability: scenarios[k].probability,
   172→        color: colors[k],
   173→      }])
   174→    );
   175→  }, [scenarios]);
   176→
   177→  // ── Strategy recommendation text ────────────────────────────────
   178→  const strategyText = rsi > 70 || mfi > 80
   179→    ? `با توجه به هشدار اشباع خرید (RSI: ${toFa(rsi)}, MFI: ${toFa(mfi)}) و فاصله قیمت تا مقاومت ${toFa(R1_level)}، استراتژی محتاطانه، انتظار برای اصلاح قیمت و ورود در محدوده حمایت ${toFa(S1_level)} تا ${toFa(S2_level)} ریال می‌باشد. در این محدوده می‌توان با حد ضرر ${toFa(S2_level)} ریال وارد موقعیت خرید شد.`
   180→    : rsi < 30 || mfi < 20
   181→    ? `با توجه به اشباع فروش (RSI: ${toFa(rsi)}, MFI: ${toFa(mfi)}) و نزدیکی به حمایت ${toFa(S1_level)}، فرصت خرید در محدوده فعلی با حد ضرر ${toFa(S2_level)} ریال قابل بررسی است. هدف اولیه ${toFa(R1_level)} و هدف ثانویه ${toFa(R2_level)} ریال تعیین می‌شود.`
   182→    : `با توجه به وضعیت خنثی اندیکاتورها (RSI: ${toFa(rsi)}, ADX: ${toFa(adx)}، قدرت روند: ${adx > 25 ? 'قوی' : 'ضعیف'})، انتظار برای خروج قیمت از محدوده ${toFa(S1_level)} تا ${toFa(R1_level)} ریال و سپس تصمیم‌گیری توصیه می‌شود. حجم معاملات و مومنتوم MACD را برای تأیید سیگنال پایش کنید.`;
   183→
   184→  return (
   185→    <div className="space-y-5" dir="rtl">
   186→      {/* ═══ HEADER ═══ */}
   187→      <div className="rounded-[32px] px-6 py-5 border border-amber-500/20"
   188→        style={{
   189→          background: 'rgba(18,28,46,0.7)',
   190→          backdropFilter: 'blur(14px)',
   191→          boxShadow: '0 25px 50px -12px rgba(0,0,0,0.6)',
   192→        }}>
   193→        <h1 className="text-xl font-bold mb-1" style={{ color: '#f8e365' }}>
   194→          📈 تحلیل تکنیکال {symbolName}
   195→        </h1>
   196→        <div className="flex flex-wrap gap-3 mt-2">
   197→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   198→            📍 قیمت مرجع: <b className="text-gray-200">{toFa(currentPrice)} ریال</b>
   199→          </span>
   200→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   201→            🎯 هدف کوتاه‌مدت: <b className="text-gray-200">{toFa(targetMin)} — {toFa(targetMax)} ریال</b>
   202→          </span>
   203→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   204→            📊 روند: <b className="text-gray-200">{trendText}</b>
   205→          </span>
   206→          <span className={`px-3.5 py-1 rounded-full border text-xs font-medium ${
   207→            rsi > 70 ? 'bg-red-500/15 text-red-400 border-red-500/30'
   208→            : rsi < 30 ? 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30'
   209→            : 'bg-amber-500/15 text-amber-400 border-amber-500/30'
   210→          }`}>
   211→            RSI: {toFa(rsi)} ({rsiSignal})
   212→          </span>
   213→        </div>
   214→      </div>
   215→
   216→      {/* ═══ TRADINGVIEW CHART ═══ */}
   217→      <div className="rounded-[28px] p-4 border border-white/6"
   218→        style={{
   219→          background: 'rgba(14,22,40,0.7)',
   220→          backdropFilter: 'blur(8px)',
   221→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   222→        }}>
   223→        <h2 className="text-sm font-semibold mb-3 text-gray-200 flex items-center gap-3">
   224→          <span style={{ color: '#f8e365' }}>📊</span>
   225→          نمودار قیمتی سهم (۱ روزه) — به همراه MA21، MA100، سطوح حمایت/مقاومت و اهداف قیمتی
   226→        </h2>
   227→        <TradingViewChart
   228→          symbolName={symbolName}
   229→          candles={chartCandles}
   230→          supports={supports}
   231→          resistances={resistances}
   232→          supportStrengths={supportStrengths}
   233→          resistanceStrengths={resistanceStrengths}
   234→          ma21={ma21}
   235→          ma100={ma100}
   236→          scenarios={tvScenarios}
   237→        />
   238→        <div className="mt-3 flex flex-wrap gap-4 text-[10px] text-gray-500">
   239→          <span>🔴 خطوط قرمز: مقاومت‌ها ({resistances.map(toFa).join(' ، ')})</span>
   240→          <span>🟢 خطوط سبز: حمایت‌ها ({supports.map(toFa).join(' ، ')})</span>
   241→          <span>🟣 خط بنفش: MA100 ({toFa(ma100)})</span>
   242→          <span>🔵 خط آبی: MA21 ({toFa(ma21)})</span>
   243→          <span>🟡 خطوط زرد: اهداف قیمتی سناریوها (رند شده)</span>
   244→        </div>
   245→      </div>
   246→
   247→      {/* ═══ DUAL ANALYSIS LAYOUT ═══ */}
   248→      <div className="rounded-[28px] p-5 border border-white/6"
   249→        style={{
   250→          background: 'rgba(14,22,40,0.7)',
   251→          backdropFilter: 'blur(8px)',
   252→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   253→        }}>
   254→        <h2 className="text-sm font-semibold mb-4 text-gray-200 flex items-center gap-3">
   255→          <span style={{ color: '#f8e365' }}>🧠</span>
   256→          تحلیل جامع روند و سطوح کلیدی
   257→        </h2>
   258→
   259→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-5">
   260→          {/* ── Text Analysis (Right in RTL) ── */}
   261→          <div className="rounded-2xl p-4 border border-white/4 space-y-4 flex flex-col"
   262→            style={{ background: 'rgba(8,20,35,0.5)' }}>
   263→
   264→            {/* ── PARAGRAPH 1: General Trend & Price Position ── */}
   265→            <p className="text-sm text-gray-300 leading-[1.85]">
   266→              <strong style={{ color: '#f8e365' }}>روند کلی و موقعیت قیمت:</strong>{' '}
   267→              سهم {symbolName} در حال حاضر در محدوده{' '}
   268→              <b className="text-gray-100">{toFa(currentPrice)} ریال</b> معامله می‌شود و روند میان‌مدت{' '}
   269→              <b className={trendDirection === 'up' ? 'text-emerald-400' : trendDirection === 'down' ? 'text-red-400' : 'text-amber-400'}>{trendDirection === 'up' ? 'صعودی' : trendDirection === 'down' ? 'نزولی' : 'خنثی'}</b>
   270→              {' '}است (زاویه {toFa(Math.abs(trendAngle))}°، R²={+(trendR2 * 100).toFixed(1)}%).
   271→              قیمت نسبت به MA21 ({toFa(ma21)} ریال){' '}
   272→              {currentPrice > ma21 ? <span className="text-emerald-400">بالاتر</span> : <span className="text-red-400">پایین‌تر</span>}
   273→              {' '}و نسبت به MA100 ({toFa(ma100)} ریال){' '}
   274→              {currentPrice > ma100 ? <span className="text-emerald-400">بالاتر</span> : <span className="text-red-400">پایین‌تر</span>}
   275→              {' '}قرار دارد. اندیکاتور Parabolic SAR ({toFa(sar)}) نیز{' '}
   276→              {sar < currentPrice ? <span>زیر قیمت قرار دارد که <b className="text-emerald-400">تأیید روند صعودی</b> است.</span> : <span>بالای قیمت قرار دارد که <b className="text-red-400">تأیید روند نزولی</b> است.</span>}
   277→              {' '}شاخص ADX ({toFa(adx)}) نشان‌دهنده {' '}
   278→              {adx > 40 ? <b className="text-emerald-400">روند قدرتمند</b> : adx > 25 ? <b className="text-amber-400">روند متوسط</b> : <b className="text-gray-500">روند ضعیف یا رنج</b>}
   279→              {' '}می‌باشد. {' '}{diSignal}.
   280→            </p>
   281→
   282→            {/* ── PARAGRAPH 2: Oscillator & Momentum Analysis ── */}
   283→            <p className="text-sm text-gray-300 leading-[1.85]">
   284→              <strong style={{ color: '#f8e365' }}>تحلیل اسیلاتورها و مومنتوم:</strong>{' '}
   285→              اندیکاتور RSI ({toFa(rsi)}) در ناحیه{' '}
   286→              <b className={rsiColor}>{rsiSignal}</b> قرار دارد.
   287→              {hasVolume && <>
   288→              اندیکاتور MFI ({toFa(mfi)}) نیز{' '}
   289→              {mfi > 80 ? <span className="text-red-400">اشباع خرید را تأیید می‌کند</span> : mfi < 20 ? <span className="text-emerald-400">اشباع فروش را نشان می‌دهد</span> : <span>در محدوده عادی است</span>}
   290→              {' '}</>}
   291→              اندیکاتور CCI ({toFa(cci)}){' '}
   292→              {cci > 100 ? <span className="text-red-400">بالاتر از +100 (قدرت خریداری قوی)</span> : cci < -100 ? <span className="text-emerald-400">پایین‌تر از -100 (قدرت فروشندگان)</span> : <span>در محدوده عادی (-100 تا +100)</span>}
   293→              . استوکاستیک (%K={toFa(stochK)}، %D={toFa(stochD)}) وضعیت{' '}
   294→              <b className={stochK > 80 ? 'text-red-400' : stochK < 20 ? 'text-emerald-400' : 'text-gray-300'}>{stochSignal}</b> را نشان می‌دهد.
   295→              {' '}MACD (خط={toFa(macdLine)}، سیگنال={toFa(macdSignal)}) با {macdBullish ? <span className="text-emerald-400">عبور خط اصلی بالای خط سیگنال — سیگنال صعودی</span> : <span className="text-red-400">خط اصلی زیر خط سیگنال — سیگنال نزولی</span>}
   296→              . هیستوگرام MACD ({toFa(macdHist)}) {' '}
   297→              {macdHist > 0 ? <span className="text-emerald-400">مثبت</span> : <span className="text-red-400">منفی</span>}
   298→              {' '}و {' '}{macdSignalText}.
   299→            </p>
   300→
   301→            {/* ── PARAGRAPH 3: Bollinger Bands & Volatility ── */}
   302→            <p className="text-sm text-gray-300 leading-[1.85]">
   303→              <strong style={{ color: '#f8e365' }}>تحلیل نوسانات و باند بولینگر:</strong>{' '}
   304→              قیمت در باند بولینگر{' '}
   305→              <b className="text-gray-200">{bbSignal}</b>
   306→              {' '}قرار دارد. باند بالایی: {toFa(bollingerUpper)}، باند میانی (MA20): {toFa(bollingerMiddle)}، باند پایینی: {toFa(bollingerLower)} ریال.
   307→              {currentPrice > bollingerUpper
   308→                ? ' عبور از باند بالایی معمولاً نشان‌دهنده ادامه حرکت صعودی کوتاه‌مدت یا واکنش به باند است.'
   309→                : currentPrice < bollingerLower
   310→                ? ' نزدیکی یا عبور از باند پایینی می‌تواند نشانه بازگشت قیمت به سمت باند میانی باشد.'

   487→function StrengthBadge({ strength, isTarget }: { strength: number; isTarget: boolean }) {
   488→  const color = strength >= 8 ? 'text-emerald-400 bg-emerald-500/15 border-emerald-500/30'

   493→      {isTarget && '🎯 '}
   494→      قدرت: {strength}/10
   495→    </span>
   496→  );
