   866→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   867→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   868→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   869→  grade?: string; overlapCount?: number; touchCount?: number;
   870→}) {
   871→  const isSupport = type === 'support';
   872→  const isTargetType = type === 'target';
   873→
   874→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   875→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   876→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   877→
   878→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   879→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   880→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   881→
   882→  return (
   883→    <div
   884→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   885→      style={{
   886→        background: bgAccent,
   887→        border: `1px solid ${borderAccent}`,
   888→        borderRight: `3px solid ${accentColor}`,
   889→      }}
   890→    >
   891→      {/* Label + Price */}
   892→      <div className="flex-1 min-w-0">
   893→        <div className="flex items-center gap-2">
   894→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   895→          {isTarget && <span className="text-[9px]">🎯</span>}
   896→          {overlapCount !== undefined && overlapCount > 0 && (
   897→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   898→              {toPersianDigits(overlapCount)} هم‌پوشانی
   899→            </span>
   900→          )}
   901→        </div>
   902→        <div className="flex items-baseline gap-1">
   903→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   904→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   905→        </div>
   906→      </div>
   907→
   908→      {/* Strength + Grade */}
   909→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   910→        <div className="flex items-center gap-1.5">
   911→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   912→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   913→          <span className="text-[9px] text-gray-500">/۱۰</span>
   914→        </div>
   915→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   916→          <div
   917→            className="h-full rounded-full transition-all duration-500"
   918→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   919→          />
   920→        </div>
   921→        {touchCount !== undefined && touchCount > 0 && (
   922→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   923→        )}
   924→      </div>
   925→    </div>
   926→  );
   927→}
