   108→type ToolType = 'cursor' | 'trendline' | 'hline' | 'vline' | 'fibonacci' | 'rectangle' | 'text' | 'brush' | 'measure' | 'arrow';
   109→
   110→// — S/R strength-based line styling (solid only, thickness = strength) ————
   111→function srLineStyle(strength: number, isTarget: boolean): { lineWidth: number; lineStyle: 0; color: string } {
   112→  if (isTarget) {
   113→    return { lineWidth: 4, lineStyle: 0, color: '' };
   114→  }
   115→  if (strength >= 9) return { lineWidth: 5, lineStyle: 0, color: '' };
   116→  if (strength >= 7) return { lineWidth: 4, lineStyle: 0, color: '' };
   117→  if (strength >= 5) return { lineWidth: 3, lineStyle: 0, color: '' };
   118→  if (strength >= 3) return { lineWidth: 2, lineStyle: 0, color: '' };
   119→  return { lineWidth: 1, lineStyle: 0, color: '' };
   120→}

   152→/* ----------------------------- COMPONENT --- */
   153→
   154→export default function CandlestickChart({ data, ta, height = 520 }: CandlestickChartProps) {
   155→  const chartContainerRef = useRef<HTMLDivElement>(null);
   156→  const svgRef = useRef<SVGSVGElement>(null);
   157→  const chartRef = useRef<IChartApi | null>(null);
   158→  const csRef = useRef<any>(null);
   159→  const subscriberRef = useRef<{ dispose: () => void } | null>(null);
   160→  const crosshairSubRef = useRef<{ dispose: () => void } | null>(null);
   161→
   162→  // Crosshair legend state
   163→  const [hoverInfo, setHoverInfo] = useState<{ date: string; o: number; h: number; l: number; c: number; v: number; chg: number } | null>(null);
   164→
   165→  // Build Jalali date strings for all candles (memoized)
   166→  const jalaliDates = useMemo(() => {
   167→    return data.map((d) =>
   168→      isGregorianDate(d.date) ? candleDateToJalali(d.date, 'full') : formatJalaliString(d.date, 'full')
   169→    );
   170→  }, [data]);
   171→
   172→  const [drawings, setDrawings] = useState<Drawing[]>([]);
   173→  const [activeTool, setActiveTool] = useState<ToolType>('cursor');
   174→  const [activeColor, setActiveColor] = useState(PALETTE[0]);
   175→  const [isDrawing, setIsDrawing] = useState(false);
   176→  const [currentDrawing, setCurrentDrawing] = useState<Drawing | null>(null);
   177→  const [textInput, setTextInput] = useState<{ x: number; y: number; drawingId: string } | null>(null);
   178→  const [textValue, setTextValue] = useState('');
   179→  const [renderKey, setRenderKey] = useState(0);
   180→  const [chartSize, setChartSize] = useState({ w: 0, h: 0 });
   181→
   182→  /* - Helper: coordinate conversion -------------- */
   183→  const getChartPoint = useCallback((clientX: number, clientY: number): DrawingPoint | null => {
   184→    const chart = chartRef.current;
   185→    const container = chartContainerRef.current;
   186→    if (!chart || !container) return null;
   187→    const rect = container.getBoundingClientRect();
   188→    const x = clientX - rect.left;
   189→    const y = clientY - rect.top;
   190→    const price = csRef.current?.coordinateToPrice(y) ?? null;
   191→    const time = chart.timeScale().coordinateToTime(x);
   192→    if (price === null || time === null) return null;
   193→    return { time: time as number, price };
   194→  }, []);
   195→
   196→  /* - SVG Rendering --------------------- */
   197→  const renderDrawings = useCallback(() => {
   198→    const chart = chartRef.current;
   199→    const svg = svgRef.current;
   200→    const container = chartContainerRef.current;
   201→    if (!chart || !svg || !container || drawings.length === 0) {
   202→      if (svg) svg.innerHTML = '';
   203→      return;
   204→    }
   205→
   206→    const rect = container.getBoundingClientRect();
   207→    const w = rect.width;
   208→    const h = rect.height;
   209→    if (w === 0 || h === 0) return;
   210→
   211→    let html = '';
   212→
   213→    // Defs for arrow markers
   214→    html += `<defs>`;
   215→    PALETTE.forEach((c) => {
   216→      html += `<marker id="arrow-${c.replace('#','')}" markerWidth="8" markerHeight="6" refX="7" refY="3" orient="auto"><polygon points="0 0, 8 3, 0 6" fill="${c}" /></marker>`;
   217→    });
   218→    html += `</defs>`;
   219→
   220→    const allDrawings = [...drawings];
   221→    if (currentDrawing) allDrawings.push(currentDrawing);
   222→
   223→    for (const d of allDrawings) {
   224→      if (!d.completed && d.points.length === 0) continue;
   225→      const col = d.color;
   226→      const lw = d.lineWidth;
   227→
   228→      switch (d.type) {
   229→        case 'trendline': {
   230→          if (d.points.length < 1) break;
   231→          const p0 = d.points[0];
   232→          const p1 = d.points.length > 1 ? d.points[d.points.length - 1] : p0;
   233→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   234→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   235→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   236→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   237→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   238→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" />`;
   239→          break;
   240→        }
   241→        case 'hline': {
   242→          if (d.points.length < 1) break;
   243→          const p = d.points[0];
   244→          const y = csRef.current?.priceToCoordinate(p.price);
   245→          if (y == null) break;
   246→          html += `<line x1="0" y1="${y}" x2="${w}" y2="${y}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   247→          html += `<text x="4" y="${y - 4}" fill="${col}" font-size="11" font-family="Vazirmatn, sans-serif">${toPersianDigits(p.price.toFixed(0))}</text>`;
   248→          break;
   249→        }
   250→        case 'vline': {
   251→          if (d.points.length < 1) break;
   252→          const p = d.points[0];
   253→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   254→          if (x == null) break;
   255→          html += `<line x1="${x}" y1="0" x2="${x}" y2="${h}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   256→          break;
   257→        }
   258→        case 'fibonacci': {
   259→          if (d.points.length < 2) break;
   260→          const p0 = d.points[0];
   261→          const p1 = d.points[1];
   262→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   263→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   264→          if (x0 == null || x1 == null) break;
   265→          const priceHigh = Math.max(p0.price, p1.price);
   266→          const priceLow = Math.min(p0.price, p1.price);
   267→          const priceRange = priceHigh - priceLow;
   268→          if (priceRange === 0) break;
   269→
   270→          const yTop = csRef.current?.priceToCoordinate(priceHigh);
   271→          const yBottom = csRef.current?.priceToCoordinate(priceLow);
   272→          if (yTop == null || yBottom == null) break;
   273→          html += `<rect x="${Math.min(x0,x1)}" y="${yTop}" width="${Math.abs(x1-x0)}" height="${yBottom - yTop}" fill="${col}" opacity="0.04" />`;
   274→
   275→          for (const level of FIB_LEVELS) {
   276→            const price = priceHigh - level.pct * priceRange;
   277→            const y = csRef.current?.priceToCoordinate(price);
   278→            if (y == null) continue;
   279→            html += `<line x1="${Math.min(x0,x1) - 10}" y1="${y}" x2="${Math.max(x0,x1) + 10}" y2="${y}" stroke="${col}" stroke-width="1" opacity="0.6" />`;
   280→            html += `<rect x="${Math.max(x0,x1) + 12}" y="${y - 8}" width="56" height="16" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="2" />`;
   281→            html += `<text x="${Math.max(x0,x1) + 14}" y="${y + 4}" fill="${col}" font-size="10" font-family="Vazirmatn, sans-serif">${toPersianDigits(level.label)} ${toPersianDigits(price.toFixed(0))}</text>`;
   282→          }
   283→          break;
   284→        }
   285→        case 'rectangle': {
   286→          if (d.points.length < 2) break;
   287→          const p0 = d.points[0];
   288→          const p1 = d.points[d.points.length - 1];
   289→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   290→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   291→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   292→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   293→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   294→          const rx = Math.min(x0, x1);
   295→          const ry = Math.min(y0, y1);
   296→          const rw = Math.abs(x1 - x0);
   297→          const rh = Math.abs(y1 - y0);
   298→          html += `<rect x="${rx}" y="${ry}" width="${rw}" height="${rh}" fill="${col}" fill-opacity="0.08" stroke="${col}" stroke-width="${lw}" rx="2" />`;
   299→          break;
   300→        }
   301→        case 'text': {
   302→          if (d.points.length < 1 || !d.text) break;
   303→          const p = d.points[0];
   304→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   305→          const y = csRef.current?.priceToCoordinate(p.price);
   306→          if (x == null || y == null) break;
   307→          html += `<rect x="${x + 4}" y="${y - 16}" width="${d.text.length * 8 + 12}" height="22" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="3" />`;
   308→          html += `<text x="${x + 10}" y="${y + 1}" fill="${col}" font-size="12" font-family="Vazirmatn, sans-serif">${d.text}</text>`;
   309→          break;
   310→        }
   311→        case 'brush': {
   312→          if (d.points.length < 2) break;
   313→          let pts = '';
   314→          let valid = true;
   315→          for (const p of d.points) {
   316→            const x = chart.timeScale().timeToCoordinate(p.time as Time);
   317→            const y = csRef.current?.priceToCoordinate(p.price);
   318→            if (x == null || y == null) { valid = false; break; }
   319→            pts += `${x},${y} `;
   320→          }
   321→          if (!valid || pts.length === 0) break;
   322→          html += `<polyline points="${pts.trim()}" fill="none" stroke="${col}" stroke-width="${lw}" stroke-linecap="round" stroke-linejoin="round" />`;
   323→          break;
   324→        }
   325→        case 'measure': {
   326→          if (d.points.length < 2) break;
   327→          const p0 = d.points[0];
   328→          const p1 = d.points[d.points.length - 1];
   329→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   330→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   331→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   332→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   333→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   334→
   335→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="4,2" />`;
   336→          const diff = p1.price - p0.price;
   337→          const pctChg = p0.price !== 0 ? ((diff / p0.price) * 100) : 0;
   338→          const bars = Math.abs(Math.round(p1.time - p0.time));
   339→          const midX = (x0 + x1) / 2;
   340→          const midY = (y0 + y1) / 2;
   341→          const label = `${diff >= 0 ? '+' : ''}${toPersianDigits(diff.toFixed(0))} (${pctChg >= 0 ? '+' : ''}${toPersianDigits(pctChg.toFixed(1))}%) ${toPersianDigits(String(bars))}bar`;
   342→          html += `<rect x="${midX - 40}" y="${midY - 24}" width="80" height="20" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="3" />`;
   343→          html += `<text x="${midX - 36}" y="${midY - 10}" fill="${col}" font-size="10" font-family="Vazirmatn, sans-serif">${label}</text>`;
   344→          break;
   345→        }
   346→        case 'arrow': {
   347→          if (d.points.length < 2) break;
   348→          const p0 = d.points[0];
   349→          const p1 = d.points[d.points.length - 1];
   350→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   351→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   352→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   353→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   354→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   355→          const markerId = `arrow-${col.replace('#','')}`;
   356→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" marker-end="url(#${markerId})" />`;
   357→          break;
   358→        }
   359→      }
   360→    }
   361→
   362→    svg.innerHTML = html;
   363→  }, [drawings, currentDrawing]);
   364→
   365→  // OHLC change color helper
   366→  const chgColor = useCallback((o: number, c: number) => c >= o ? BULL : BEAR, []);
   367→
   368→  /* - Chart Build ---------------------- */
   369→  const build = useCallback(() => {
   370→    const chartEl = chartContainerRef.current;
   371→    if (!chartEl || data.length === 0) return;
   372→
   373→    if (chartRef.current) {
   374→      chartRef.current.remove();
   375→      chartRef.current = null;
   376→    }
   377→
   378→    // — Build Jalali time map — detect Gregorian vs Jalali dates —
   379→    const jalaliMap = new Map<number, string>();
   380→    data.forEach((d, i) => {
   381→      jalaliMap.set(i, isGregorianDate(d.date) ? candleDateToJalali(d.date, 'compact') : formatJalaliString(d.date, 'compact'));
   382→    });
   383→
   384→    // — Persian numeral price formatter —
   385→    const persianPriceFormatter = (price: number) => {
   386→      return toPersianDigits(price.toLocaleString('en', { maximumFractionDigits: 0 }));
   387→    };
   388→
   389→    const chart = createChart(chartEl, {
   390→      layout: { background: { type: ColorType.Solid, color: BG }, textColor: TXT, fontSize: 11 },
   391→      grid: { vertLines: { color: GRID }, horzLines: { color: GRID } },
   392→      crosshair: { mode: CrosshairMode.Normal },
   393→      rightPriceScale: {
   394→        borderColor: '#e5e7eb',
   395→        scaleMargins: { top: 0.05, bottom: 0.3 },
   396→      },
   397→      timeScale: {
   398→        borderColor: '#e5e7eb',
   399→        rightOffset: 5,
   400→        barSpacing: 6,
   401→        tickMarkFormatter: (time: Time) => {
   402→          const idx = time as number;
   403→          return jalaliMap.get(idx) || String(idx);
   404→        },
   405→      },
   406→      localization: {
   407→        priceFormatter: persianPriceFormatter,
   408→      },
   409→      width: chartEl.clientWidth,
   410→      height,
   411→    });
   412→    chartRef.current = chart;
   413→
   414→    // Crosshair move handler — show Shamsi date + OHLC legend
   415→    crosshairSubRef.current?.dispose();
   416→    crosshairSubRef.current = chart.subscribeCrosshairMove((param) => {
   417→      if (!param.time || param.time === undefined) {
   418→        setHoverInfo(null);
   419→        return;
   420→      }
   421→      const idx = param.time as number;
   422→      const candle = data[idx];
   423→      if (!candle) {
   424→        setHoverInfo(null);
   425→        return;
   426→      }
   427→      const prevCandle = idx > 0 ? data[idx - 1] : candle;
   428→      const chg = prevCandle.close > 0 ? ((candle.close - prevCandle.close) / prevCandle.close) * 100 : 0;
   429→      setHoverInfo({
   430→        date: jalaliDates[idx] || candle.date,
   431→        o: candle.open,
   432→        h: candle.high,
   433→        l: candle.low,
   434→        c: candle.close,
   435→        v: candle.volume,
   436→        chg,
   437→      });
   438→    }) as unknown as { dispose: () => void };
   439→
   440→    // Numeric time index
   441→    const candles = data.map((d, i) => ({
   442→      time: i as Time,
   443→      open: d.open,
   444→      high: d.high,
   445→      low: d.low,
   446→      close: d.close,
   447→    }));
   448→
   449→    const cs = chart.addSeries(CandlestickSeries, {
   450→      upColor: BULL,
   451→      downColor: BEAR,
   452→      borderUpColor: BULL,
   453→      borderDownColor: BEAR,
   454→      wickUpColor: BULL,
   455→      wickDownColor: BEAR,
   456→    });
   457→    cs.setData(candles);
   458→    csRef.current = cs;
   459→
   460→    // Volume
   461→    const vol = chart.addSeries(HistogramSeries, {
   462→      priceFormat: { type: 'volume' },
   463→      priceScaleId: 'vol',
   464→    });
   465→    vol.setData(
   466→      data.map((d, i) => ({
   467→        time: i as Time,
   468→        value: d.volume,
   469→        color: d.close >= d.open ? BULL : BEAR,
   470→      }))
   471→    );
   472→    chart.priceScale('vol').applyOptions({ scaleMargins: { top: 0.8, bottom: 0 } });
   473→
   474→    // TA overlays
   475→    if (ta) {
   476→      for (const cfg of SMA_CFG) {
   477→        const v = ta.sma[cfg.key];
   478→        if (v && v > 0)
   479→          cs.createPriceLine({
   480→            price: v,
   481→            color: cfg.color,
   482→            lineWidth: 1,
   483→            lineStyle: 0,
   484→            axisLabelVisible: true,
   485→            title: cfg.title,
   486→          });
   487→      }
   488→      const bb = ta.bollingerBands;
   489→      if (bb) {
   490→        if (bb.upper > 0) cs.createPriceLine({ price: bb.upper, color: 'rgba(124,58,237,0.6)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Upper' });
   491→        if (bb.middle > 0) cs.createPriceLine({ price: bb.middle, color: 'rgba(124,58,237,0.6)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Mid' });
   492→        if (bb.lower > 0) cs.createPriceLine({ price: bb.lower, color: 'rgba(124,58,237,0.6)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Lower' });
   493→      }
   494→      // SAR — use dark color instead of white (white invisible on white bg)
   495→      if (ta.sar > 0)
   496→        cs.createPriceLine({ price: ta.sar, color: 'rgba(217,119,6,0.7)', lineWidth: 1, lineStyle: 1, axisLabelVisible: true, title: 'SAR' });
   497→
   498→      // S/R Fibonacci levels with strength
   499→      const resistanceStrengths = ta.resistanceStrengths || [];
   500→      const supportStrengths = ta.supportStrengths || [];
   501→
   502→      resistanceStrengths.slice(0, 6).forEach((r, i) => {
   503→        if (r.price <= 0) return;
   504→        const style = srLineStyle(r.strength, r.isTarget);
   505→        const toFa = (n: number) => String(n).replace(/[0-9]/g, d => '۰۱۲۳۴۵۶۷۸۹'[+d]);
   506→        const methodCount = toFa(r.methods?.length ?? 0);
   507→        const fibTag = r.fibLabel ? ` فیبو ${r.fibLabel}` : '';
   508→        const title = r.isTarget
   509→          ? `★ R${toFa(i + 1)} [${r.grade}]`
   510→          : `R${toFa(i + 1)} [${r.grade}] (${methodCount} method)${fibTag}`;
   511→        cs.createPriceLine({
   512→          price: r.price,
   513→          color: r.isTarget ? '#dc2626' : BEAR,
   514→          lineWidth: style.lineWidth as 1|2|3|4,
   515→          lineStyle: style.lineStyle,
   516→          axisLabelVisible: true,
   517→          title,
   518→        });
   519→      });
   520→
   521→      supportStrengths.slice(0, 6).forEach((s, i) => {
   522→        if (s.price <= 0) return;
   523→        const style = srLineStyle(s.strength, s.isTarget);
   524→        const toFa = (n: number) => String(n).replace(/[0-9]/g, d => '۰۱۲۳۴۵۶۷۸۹'[+d]);
   525→        const methodCount = toFa(s.methods?.length ?? 0);
   526→        const fibTag = s.fibLabel ? ` فیبو ${s.fibLabel}` : '';
   527→        const title = s.isTarget
   528→          ? `★ S${toFa(i + 1)} [${s.grade}]`
   529→          : `S${toFa(i + 1)} [${s.grade}] (${methodCount} method)${fibTag}`;
   530→        cs.createPriceLine({
   531→          price: s.price,
   532→          color: s.isTarget ? '#16a34a' : BULL,
   533→          lineWidth: style.lineWidth as 1|2|3|4,
   534→          lineStyle: style.lineStyle,
   535→          axisLabelVisible: true,
   536→          title,
   537→        });
   538→      });
   539→    }
   540→
   541→    chart.timeScale().fitContent();
   542→
   543→    // Subscribe to visible range changes to re-render SVG
   544→    requestAnimationFrame(() => {
   545→      if (chartRef.current) {
   546→        const sub = chartRef.current.timeScale().subscribeVisibleLogicalRangeChange(() => {
   547→          setRenderKey((k) => k + 1);
   548→        });
   549→        subscriberRef.current = sub as unknown as { dispose: () => void };
   550→      }
   551→    });
   552→
   553→    // Set initial chart size (deferred to avoid React batching issue)
   554→    requestAnimationFrame(() => {
   555→      if (chartContainerRef.current) {
   556→        setChartSize({ w: chartContainerRef.current.clientWidth, h: height });
   557→      }
   558→    });
   559→  }, [data, ta, height, jalaliDates]);
   560→
   561→  /* - Re-render SVG when renderKey changes --------- */
   562→  useEffect(() => {
   563→    renderDrawings();
   564→  }, [renderKey, renderDrawings]);
   565→
   566→  /* - Init chart ----------------------- */
   567→  useEffect(() => {
   568→    const el = chartContainerRef.current;
   569→    if (!el) return;
   570→    const ro = new ResizeObserver((entries) => {
   571→      for (const e of entries) {
   572→        const { width: w, height: h } = e.contentRect;
   573→        if (w > 0 && h > 0) {
   574→          chartRef.current?.applyOptions({ width: w, height: h });
   575→          setChartSize({ w, h });
   576→        }
   577→      }
   578→    });
   579→    ro.observe(el);
   580→    build();
   581→    return () => {
   582→      ro.disconnect();
   583→      subscriberRef.current?.dispose();
   584→      subscriberRef.current = null;
   585→      crosshairSubRef.current?.dispose();
   586→      crosshairSubRef.current = null;
   587→      if (chartRef.current) {
   588→        chartRef.current.remove();
   589→        chartRef.current = null;
   590→      }
   591→    };
   592→  }, [build]);
   593→
   594→  /* - Mouse event handlers ----------------- */
   595→  const handleMouseDown = useCallback(
   596→    (e: React.MouseEvent) => {
   597→      if (activeTool === 'cursor') return;
   598→      if (textInput) return;
   599→
   600→      const point = getChartPoint(e.clientX, e.clientY);
   601→      if (!point) return;
   602→
   603→      if (activeTool === 'hline') {
   604→        const drawing: Drawing = {
   605→          id: Date.now().toString(),
   606→          type: 'hline',
   607→          points: [point],
   608→          color: activeColor,
   609→          lineWidth: 1,
   610→          completed: true,
   611→        };
   612→        setDrawings((prev) => [...prev, drawing]);
   613→        setIsDrawing(false);
   614→        return;
   615→      }
   616→
   617→      if (activeTool === 'text') {
   618→        const chart = chartRef.current;
   619→        const container = chartContainerRef.current;
   620→        if (!chart || !container) return;
   621→        const rect = container.getBoundingClientRect();
   622→        const x = e.clientX - rect.left;
   623→        const y = e.clientY - rect.top;
   624→
   625→        const tempDrawing: Drawing = {
   626→          id: Date.now().toString(),
   627→          type: 'text',
   628→          points: [point],
   629→          color: activeColor,
   630→          lineWidth: 1,
   631→          text: '',
   632→          completed: false,
   633→        };
   634→        setCurrentDrawing(tempDrawing);
   635→        setTextInput({ x, y, drawingId: tempDrawing.id });
   636→        setTextValue('');
   637→        setIsDrawing(true);
   638→        return;
   639→      }
   640→
   641→      const drawing: Drawing = {
   642→        id: Date.now().toString(),
   643→        type: activeTool,
   644→        points: [point],
   645→        color: activeColor,
   646→        lineWidth: activeTool === 'brush' ? 2 : 1,
   647→        completed: false,
   648→      };
   649→      setCurrentDrawing(drawing);
   650→      setIsDrawing(true);
   651→    },
   652→    [activeTool, activeColor, getChartPoint, textInput]
   653→  );
   654→
   655→  const handleMouseMove = useCallback(
   656→    (e: React.MouseEvent) => {
   657→      if (!isDrawing || !currentDrawing) return;
   658→
   659→      const point = getChartPoint(e.clientX, e.clientY);
   660→      if (!point) return;
   661→
   662→      if (currentDrawing.type === 'brush') {
   663→        setCurrentDrawing({
   664→          ...currentDrawing,
   665→          points: [...currentDrawing.points, point],
   666→        });
   667→      } else if (currentDrawing.type !== 'text' && currentDrawing.type !== 'hline') {
   668→        setCurrentDrawing({
   669→          ...currentDrawing,
   670→          points: [currentDrawing.points[0], point],
   671→        });
   672→      }
   673→    },
   674→    [isDrawing, currentDrawing, getChartPoint]
   675→  );
   676→
   677→  const handleMouseUp = useCallback(() => {
   678→    if (!isDrawing || !currentDrawing) return;
   679→
   680→    if (currentDrawing.type === 'text') return;
   681→
   682→    const completed: Drawing = {
   683→      ...currentDrawing,
   684→      completed: true,
   685→    };
   686→
   687→    const minPoints = currentDrawing.type === 'hline' ? 1 : 2;
   688→    if (completed.points.length >= minPoints) {
   689→      setDrawings((prev) => [...prev, completed]);
   690→    }
   691→
   692→    setCurrentDrawing(null);
   693→    setIsDrawing(false);
   694→  }, [isDrawing, currentDrawing]);
   695→
   696→  /* - Text input submit ------------------- */
   697→  const handleTextSubmit = useCallback(() => {
   698→    if (!textInput || !textValue.trim() || !currentDrawing) {
   699→      setTextInput(null);
   700→      setCurrentDrawing(null);
   701→      setIsDrawing(false);
   702→      return;
   703→    }
   704→
   705→    const completed: Drawing = {
   706→      ...currentDrawing,
   707→      text: textValue.trim(),
   708→      completed: true,
   709→    };
   710→    setDrawings((prev) => [...prev, completed]);
   711→    setTextInput(null);
   712→    setTextValue('');
   713→    setCurrentDrawing(null);
   714→    setIsDrawing(false);
   715→  }, [textInput, textValue, currentDrawing]);
   716→
   717→  const handleTextCancel = useCallback(() => {
   718→    setTextInput(null);
   719→    setTextValue('');
   720→    setCurrentDrawing(null);
   721→    setIsDrawing(false);
   722→  }, []);
   723→
   724→  /* - Delete / Clear -------------------- */
   725→  const handleDeleteLast = useCallback(() => {
   726→    setDrawings((prev) => prev.slice(0, -1));
   727→  }, []);
   728→
   729→  const handleClearAll = useCallback(() => {
   730→    setDrawings([]);
   731→  }, []);
   732→
   733→  /* - Render ------------------------- */
   734→  const fmt = (n: number) => toPersianDigits(n.toLocaleString('en', { maximumFractionDigits: 0 }));
   735→
   736→  return (
   737→    <div className="w-full">
   738→      {/* - CROSSHAIR LEGEND (Shamsi date + OHLC) ---------- */}
   739→      {hoverInfo && (
   740→        <div className="flex items-center gap-3 px-3 py-1.5 border border-[#e5e7eb] rounded-t-lg bg-[#ffffff] text-[11px]" dir="rtl" style={{ fontFamily: 'Vazirmatn, sans-serif' }}>
   741→          <span className="text-gray-500 font-medium">{hoverInfo.date}</span>
   742→          <div className="w-px h-3.5 bg-gray-200" />
   743→          <span className="text-gray-500">باز: <span className={chgColor(hoverInfo.o, hoverInfo.c) === BULL ? 'text-emerald-700 font-bold' : 'text-red-700 font-bold'}>{fmt(hoverInfo.o)}</span></span>
   744→          <span className="text-gray-500">بالا: <span className="text-emerald-700 font-bold">{fmt(hoverInfo.h)}</span></span>
   745→          <span className="text-gray-500">پایین: <span className="text-red-700 font-bold">{fmt(hoverInfo.l)}</span></span>
   746→          <span className="text-gray-500">بسته: <span className={chgColor(hoverInfo.o, hoverInfo.c) === BULL ? 'text-emerald-700 font-bold' : 'text-red-700 font-bold'}>{fmt(hoverInfo.c)}</span></span>
   747→          {hoverInfo.v > 0 && (
   748→            <span className="text-gray-500">حجم: <span className="text-gray-900 font-bold">{toPersianDigits((hoverInfo.v / 1e6).toFixed(1))}M</span></span>
   749→          )}
   750→          <span className={`font-bold ${hoverInfo.chg >= 0 ? 'text-emerald-700' : 'text-red-700'}`}>
   751→            {hoverInfo.chg >= 0 ? '▲' : '▼'} {toPersianDigits(Math.abs(hoverInfo.chg).toFixed(2))}٪
   752→          </span>
   753→        </div>
   754→      )}
   755→      {/* - TOOLBAR --------------------- */}
   756→      <div
   757→        className={`flex items-center gap-1 px-2 py-1.5 overflow-x-auto flex-nowrap border ${hoverInfo ? 'border-t-0 rounded-b-lg' : 'rounded-t-lg border-b-0'}`}
   758→        style={{ backgroundColor: TOOLBAR_BG, borderColor: TOOLBAR_BORDER }}
   759→      >
   760→        {TOOLS.map((tool) => (
   761→          <button
   762→            key={tool.id}
   763→            onClick={() => {
   764→              setActiveTool(tool.id);
   765→              if (isDrawing) {
   766→                setIsDrawing(false);
   767→                setCurrentDrawing(null);
   768→                setTextInput(null);
   769→              }
   770→            }}
   771→            title={tool.label}
   772→            className={`shrink-0 flex items-center justify-center w-8 h-8 rounded-md transition-all cursor-pointer ${
   773→              activeTool === tool.id
   774→                ? 'bg-amber-500/20 text-amber-800 ring-1 ring-amber-500/30'
   775→                : 'text-gray-500 hover:bg-gray-100 hover:text-gray-700'
   776→            }`}
   777→          >
   778→            {tool.icon}
   779→          </button>
   780→        ))}
   781→
   782→        <div className="w-px h-6 bg-gray-200 mx-1 shrink-0" />
   783→
   784→        {PALETTE.map((c) => (
   785→          <button
   786→            key={c}
   787→            onClick={() => setActiveColor(c)}
   788→            title={c}
   789→            className={`shrink-0 w-6 h-6 rounded-full border-2 transition-all cursor-pointer ${
   790→              activeColor === c ? 'border-gray-900 scale-110' : 'border-transparent hover:scale-105'
   791→            }`}
   792→            style={{ backgroundColor: c }}
   793→          />
   794→        ))}
   795→
   796→        <div className="w-px h-6 bg-gray-200 mx-1 shrink-0" />
   797→
   798→        <button
   799→          onClick={handleDeleteLast}
   800→          title="حذف آخر"
   801→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-500 hover:bg-red-50 hover:text-red-700 transition-all cursor-pointer"
   802→        >
   803→          <Trash2 className="w-3.5 h-3.5" />
   804→        </button>
   805→
   806→        <button
   807→          onClick={handleClearAll}
   808→          title="پاک کردن همه"
   809→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-500 hover:bg-red-50 hover:text-red-700 transition-all cursor-pointer"
   810→        >
   811→          <X className="w-3.5 h-3.5" />
   812→        </button>
   813→
   814→        <div className="mr-auto shrink-0 text-xs text-gray-500 font-medium" dir="rtl">
   815→          {TOOLS.find((t) => t.id === activeTool)?.label}
   816→        </div>
   817→      </div>
   818→
   819→      <div
   820→        className="relative rounded-b-lg overflow-hidden border border-gray-200"
   821→        style={{ height }}
   822→      >
   823→        <div
   824→          ref={chartContainerRef}
   825→          className="w-full h-full"
   826→          data-chart
   827→        />
   828→
   829→        <svg
   830→          ref={svgRef}
   831→          className="absolute top-0 left-0"
   832→          style={{
   833→            width: chartSize.w || '100%',
   834→            height: chartSize.h || '100%',
   835→            pointerEvents: activeTool !== 'cursor' ? 'auto' : 'none',
   836→            zIndex: 10,
   837→          }}
   838→          onMouseDown={handleMouseDown}
   839→          onMouseMove={handleMouseMove}
   840→          onMouseUp={handleMouseUp}
   841→          onMouseLeave={handleMouseUp}
   842→        />
   843→
   844→        {textInput && (
   845→          <div
   846→            className="absolute z-20"
   847→            style={{
   848→              left: textInput.x,
   849→              top: textInput.y - 8,
   850→            }}
   851→          >
   852→            <input
   853→              type="text"
   854→              value={textValue}
   855→              onChange={(e) => setTextValue(e.target.value)}
   856→              onKeyDown={(e) => {
   857→                if (e.key === 'Enter') handleTextSubmit();
   858→                if (e.key === 'Escape') handleTextCancel();
   859→              }}
   860→              placeholder="متن..."
   861→              autoFocus
   862→              dir="rtl"
   863→              className="bg-white border border-gray-200 rounded px-2 py-1 text-sm text-gray-900 outline-none focus:border-amber-500 w-32"
   864→              onBlur={handleTextSubmit}
   865→            />
   866→          </div>
   867→        )}
   868→      </div>
   869→    </div>
   870→  );
   871→}
