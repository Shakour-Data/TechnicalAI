   230→export default function Home() {
   231→  const [data, setData] = useState<AnalysisData | null>(null);
   232→  const [loading, setLoading] = useState(false);
   233→  const [loadingMessage, setLoadingMessage] = useState<string | null>(null);
   234→  const [error, setError] = useState<string | null>(null);
   235→  const [activePanel, setActivePanel] = useState<SidebarItem>('visual');
   236→  const [refreshing, setRefreshing] = useState(false);
   237→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   238→  const [currentPage, setCurrentPage] = useState<PageId>('analysis');
   239→
   240→  // Store last fetch params for auto-refresh
   241→  const lastFetchRef = useRef<{ symbol: string; category?: string; insCode?: string; tgjuKey?: string; finpySector?: string; finpyIndex?: string; webId?: number } | null>(null);
   242→  const refreshTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   243→  const fetchControllerRef = useRef<AbortController | null>(null);
   244→
   245→  // Chart container ref for image export
   246→  const chartWrapperRef = useRef<HTMLDivElement>(null);
   247→
   248→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number) => {
   249→    // Abort any previous in-flight fetch
   250→    if (fetchControllerRef.current) {
   251→      fetchControllerRef.current.abort();
   252→    }
   253→    const controller = new AbortController();
   254→    fetchControllerRef.current = controller;
   255→
   256→    // TGJU instrument: fetch historical data via tgju.org chart API
   257→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   258→      setLoading(true);
   259→      setLoadingMessage('در حال دریافت داده‌های تاریخی ... (حدود ۱۵ ثانیه)');
   260→      setError(null);
   261→      setData(null);
   262→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId };
   263→      try {
   264→        const res = await fetch('/api/tgju-analysis?key=' + encodeURIComponent(tgjuKey), { signal: controller.signal });
   265→        if (controller.signal.aborted) return;
   266→        if (!res.ok) {
   267→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   268→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   269→          throw new Error(errMsg);
   270→        }
   271→        const json = await res.json();
   272→        if (controller.signal.aborted) return;
   273→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   274→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   275→        }
   276→        setData({ ...json, isTgju: true });
   277→        setActivePanel('visual');
   278→      } catch (err) {
   279→        if (controller.signal.aborted) return;
   280→        setError(String(err));
   281→      } finally {
   282→        if (!controller.signal.aborted) {
   283→          setLoading(false);
   284→          setLoadingMessage(null);
   285→        }
   286→      }
   287→      return;
   288→    }
   289→
   290→    // Index: fetch historical data via TSETMC CDN (webId), then finpy-tse (finpyIndex/finpySector)
   291→    if (category === 'index') {
   292→      setLoading(true);
   293→      setLoadingMessage('در حال دریافت داده‌های شاخص از TSETMC ... (حدود ۳۰ ثانیه برای بار اول)');
   294→      setError(null);
   295→      setData(null);
   296→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId };
   297→      let gotError = false;
   298→      try {
   299→        // 1. Try webId (TSETMC CDN)
   300→        if (webId) {
   301→          setLoadingMessage('در حال دریافت داده‌های شاخص از TSETMC ... (حدود ۳۰ ثانیه برای بار اول)');
   302→          const res = await fetch('/api/finpy-sector?webId=' + encodeURIComponent(String(webId)));
   303→
   304→          if (res.ok) {
   305→            try {
   306→              const json = await res.json();
   307→              if (controller.signal.aborted) return;
   308→              if (json.candles && json.candles.length > 0 && json.ta) {
   309→                setData(json);
   310→                setActivePanel('visual');
   311→                setLoading(false);
   312→                setLoadingMessage(null);
   313→                return;
   314→              }
   315→              if (json.error) { setError(json.error); gotError = true; }
   316→            } catch (_) { /* invalid JSON */ }
   317→          } else { gotError = true; }
   318→        }
   319→
   320→        // 2. Main indices via finpy-tse
   321→        if (finpyIndex) {
   322→          setLoadingMessage('در حال دریافت داده‌های شاخص اصلی ...');
   323→          const res = await fetch('/api/finpy-sector?indexKey=' + encodeURIComponent(finpyIndex), { signal: controller.signal });
   324→          if (controller.signal.aborted) return;
   325→          if (res.ok) {
   326→            try {
   327→              const json = await res.json();
   328→              if (controller.signal.aborted) return;
   329→              if (json.candles && json.candles.length > 0 && json.ta) {
   330→                setData(json);
   331→                setActivePanel('visual');
   332→                setLoading(false);
   333→                setLoadingMessage(null);
