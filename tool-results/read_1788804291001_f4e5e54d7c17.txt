   236→/**
   237→ * Simple Moving Average — the mean of the last `period` closing prices.
   238→ *
   239→ * @param closes - Array of closing prices (oldest → newest)
   240→ * @param period - Lookback window length
   241→ * @returns The SMA value, or 0 if insufficient data
   242→ */
   243→function sma(closes: number[], period: number): number {
   244→  if (closes.length < period) return 0;
   245→  const slice = closes.slice(-period);
   246→  return slice.reduce((a, b) => a + b, 0) / period;
   247→}
   248→
   249→/**
   250→ * Exponential Moving Average using the standard smoothing factor k = 2/(period+1).
   251→ * Seeded with an SMA of the first `period` values, then iterated forward.
   252→ *
   253→ * @param closes - Array of closing prices (oldest → newest)
   254→ * @param period - EMA period (controls responsiveness; lower = more reactive)
   255→ * @returns The EMA value, or 0 if insufficient data
   256→ */
   257→function emaCalc(closes: number[], period: number): number {
   258→  if (closes.length < period) return 0;
   259→  const k = 2 / (period + 1);
   260→  let emaVal = sma(closes.slice(0, period), period);
   261→  for (let i = period; i < closes.length; i++) {
   262→    emaVal = closes[i] * k + emaVal * (1 - k);
   263→  }
   264→  return emaVal;
   265→}

   267→/**
   268→ * Relative Strength Index (Wilder's smoothed RSI).
   269→ * Measures the magnitude of recent gains vs. losses on a 0–100 scale.
   270→ * Values >70 suggest overbought; <30 suggest oversold.
   271→ *
   272→ * @param closes  - Array of closing prices (oldest → newest)
   273→ * @param period  - RSI period (default 14)
   274→ * @returns RSI value 0–100, or 50 (neutral) if insufficient data
   275→ */
   276→function calcRSI(closes: number[], period: number = 14): number {
   277→  if (closes.length < period + 1) return 50;
   278→  let avgGain = 0;
   279→  let avgLoss = 0;
   280→  for (let i = 1; i <= period; i++) {
   281→    const diff = closes[i] - closes[i - 1];
   282→    if (diff > 0) avgGain += diff;
   283→    else avgLoss += Math.abs(diff);
   284→  }
   285→  avgGain /= period;
   286→  avgLoss /= period;
   287→  for (let i = period + 1; i < closes.length; i++) {
   288→    const diff = closes[i] - closes[i - 1];
   289→    const gain = diff > 0 ? diff : 0;
   290→    const loss = diff < 0 ? Math.abs(diff) : 0;
   291→    avgGain = (avgGain * (period - 1) + gain) / period;
   292→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   293→  }
   294→  if (avgLoss === 0) return 100;
   295→  const rs = avgGain / avgLoss;
   296→  return 100 - 100 / (1 + rs);
   297→}

   299→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   300→function calcMFI(data: OHLCV[], period: number = 14): number {
   301→  if (data.length < period + 1) return 50;
   302→  let posFlow = 0;
   303→  let negFlow = 0;
   304→  for (let i = data.length - period; i < data.length; i++) {
   305→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   306→    const mf = tp * data[i].volume;
   307→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   308→    if (tp > prevTp) posFlow += mf;
   309→    else negFlow += mf;
   310→  }
   311→  if (negFlow === 0) return 100;
   312→  const mfr = posFlow / negFlow;
   313→  return 100 - 100 / (1 + mfr);
   314→}

   317→function calcCCI(data: OHLCV[], period: number = 20): number {
   318→  if (data.length < period) return 0;
   319→  const tps: number[] = [];
   320→  for (let i = data.length - period; i < data.length; i++) {
   321→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   322→  }
   323→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   324→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   325→  if (meanDev === 0) return 0;
   326→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   327→}

   329→/**
   330→ * Stochastic Oscillator (%K and %D).
   331→ * Compares the closing price to the high-low range over `kPeriod` bars,
   332→ * then smooths %K by `smoothK` and computes %D as an SMA of smoothed %K.
   333→ *
   334→ * @param data    - OHLCV array (oldest → newest)
   335→ * @param kPeriod - Lookback for raw %K (default 14)
   336→ * @param smoothK - Smoothing period for %K (default 3)
   337→ * @param smoothD - Smoothing period for %D (default 3)
   338→ * @returns Object `{ k, d }` each 0–100; defaults to `{ k: 50, d: 50 }` if insufficient data
   339→ */
   340→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   341→  if (data.length < kPeriod) return { k: 50, d: 50 };
   342→  const rawKs: number[] = [];
   343→  for (let i = kPeriod - 1; i < data.length; i++) {
   344→    let lowest = Infinity;
   345→    let highest = -Infinity;
   346→    for (let j = i - kPeriod + 1; j <= i; j++) {
   347→      if (data[j].low < lowest) lowest = data[j].low;
   348→      if (data[j].high > highest) highest = data[j].high;
   349→    }
   350→    const range = highest - lowest;
   351→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   352→  }
   353→  const smoothedK: number[] = [];
   354→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   355→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   356→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   357→  }
   358→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   359→  const dSlice = smoothedK.slice(-smoothD);
   360→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   361→  return { k, d };
   362→}

   364→// ─── Helper: Williams %R(14) ──────────────────────────────────────────────────
   365→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   366→  if (data.length < period) return -50;
   367→  const slice = data.slice(-period);
   368→  let highest = -Infinity;
   369→  let lowest = Infinity;
   370→  for (const c of slice) {
   371→    if (c.high > highest) highest = c.high;
   372→    if (c.low < lowest) lowest = c.low;
   373→  }
   374→  const range = highest - lowest;
   375→  if (range === 0) return -50;
   376→  return ((highest - data[data.length - 1].close) / range) * -100;
   377→}

   379→/**
   380→ * Moving Average Convergence/Divergence (MACD).
   381→ * MACD Line = EMA(fast) − EMA(slow); Signal = EMA(sig) of MACD Line;
   382→ * Histogram = MACD Line − Signal.
   383→ * Crossovers of line/signal indicate momentum shifts.
   384→ *
   385→ * @param closes - Array of closing prices (oldest → newest)
   386→ * @param fast   - Fast EMA period (default 12)
   387→ * @param slow   - Slow EMA period (default 26)
   388→ * @param sig    - Signal line EMA period (default 9)
   389→ * @returns `{ line, signal, histogram }`; zeros if insufficient data
   390→ */
   391→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   392→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   393→  const emaFast = emaCalc(closes, fast);
   394→  const emaSlow = emaCalc(closes, slow);
   395→  const macdLine = emaFast - emaSlow;
   396→  // Calculate signal as EMA of MACD values
   397→  const macdValues: number[] = [];
   398→  const kFast = 2 / (fast + 1);
   399→  const kSlow = 2 / (slow + 1);
   400→  let eFast = sma(closes.slice(0, fast), fast);
   401→  let eSlow = sma(closes.slice(0, slow), slow);
   402→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   403→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   404→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   405→    macdValues.push(eFast - eSlow);
   406→  }
   407→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   408→  const kSig = 2 / (sig + 1);
   409→  let signalVal = sma(macdValues.slice(0, sig), sig);
   410→  for (let i = sig; i < macdValues.length; i++) {
   411→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   412→  }
   413→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   414→}

   416→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   417→function calcATR(data: OHLCV[], period: number = 14): number {
   418→  if (data.length < 2) return 0;
   419→  const trs: number[] = [];
   420→  for (let i = 1; i < data.length; i++) {
   421→    const tr = Math.max(
   422→      data[i].high - data[i].low,
   423→      Math.abs(data[i].high - data[i - 1].close),
   424→      Math.abs(data[i].low - data[i - 1].close)
   425→    );
   426→    trs.push(tr);
   427→  }
   428→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   429→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   430→  for (let i = period; i < trs.length; i++) {
   431→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   432→  }
   433→  return atrVal;
   434→}

   436→/**
   437→ * Average Directional Index with +DI and −DI.
   438→ * ADX measures trend strength (0–100); >25 indicates a trending market.
   439→ * +DI > −DI suggests bullish directional pressure; vice versa for bearish.
   440→ *
   441→ * @param data   - OHLCV array (oldest → newest)
   442→ * @param period - ADX/DI period (default 14)
   443→ * @returns `{ adx, diPlus, diMinus }`; zeros if insufficient data
   444→ */
   445→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   446→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   447→  const plusDMs: number[] = [];
   448→  const minusDMs: number[] = [];
   449→  const trs: number[] = [];
   450→  for (let i = 1; i < data.length; i++) {
   451→    const upMove = data[i].high - data[i - 1].high;
   452→    const downMove = data[i - 1].low - data[i].low;
   453→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   454→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   455→    trs.push(Math.max(
   456→      data[i].high - data[i].low,
   457→      Math.abs(data[i].high - data[i - 1].close),
   458→      Math.abs(data[i].low - data[i - 1].close)
   459→    ));
   460→  }
   461→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   462→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   463→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   464→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   465→  const dxValues: number[] = [];
   466→  for (let i = period; i < trs.length; i++) {
   467→    smoothTR = smoothTR - smoothTR / period + trs[i];
   468→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   469→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   470→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   471→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   472→    const diSum = diPlus + diMinus;
   473→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   474→    dxValues.push(dx);
   475→  }
   476→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   477→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   478→  for (let i = period; i < dxValues.length; i++) {
   479→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   480→  }
   481→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   482→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   483→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   484→}

   486→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   487→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   488→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   489→  let isLong = true;
   490→  let af = step;
   491→  let ep = data[0].high;
   492→  let sar = data[0].low;
   493→  for (let i = 1; i < data.length; i++) {
   494→    const prevSar = sar;
   495→    sar = prevSar + af * (ep - prevSar);
   496→    // Ensure SAR is within bounds
   497→    if (isLong) {
   498→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   499→      if (data[i].low < sar) {
   500→        isLong = false;
   501→        sar = ep;
   502→        ep = data[i].low;
   503→        af = step;
   504→      } else {
   505→        if (data[i].high > ep) {
   506→          ep = data[i].high;
   507→          af = Math.min(af + step, max);
   508→        }
   509→      }
   510→    } else {
   511→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   512→      if (data[i].high > sar) {
   513→        isLong = true;
   514→        sar = ep;
   515→        ep = data[i].high;
   516→        af = step;
   517→      } else {
   518→        if (data[i].low < ep) {
   519→          ep = data[i].low;
   520→          af = Math.min(af + step, max);
   521→        }
   522→      }
   523→    }
   524→  }
   525→  return sar;
   526→}

   528→/**
   529→ * Bollinger Bands.
   530→ * Middle = SMA(period); Upper/Lower = Middle ± stdDev × σ.
   531→ * Price touching upper band suggests overbought; lower band suggests oversold.
   532→ * Band width (upper − lower) indicates volatility.
   533→ *
   534→ * @param closes  - Array of closing prices (oldest → newest)
   535→ * @param period  - SMA period for the middle band (default 20)
   536→ * @param stdDev  - Number of standard deviations for the bands (default 2)
   537→ * @returns `{ upper, middle, lower }`; collapses to last close if insufficient data
   538→ */
   539→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   540→  if (closes.length < period) {
   541→    const c = closes[closes.length - 1] ?? 0;
   542→    return { upper: c, middle: c, lower: c };
   543→  }
   544→  const slice = closes.slice(-period);
   545→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   546→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   547→  const sd = Math.sqrt(variance);
   548→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   549→}

   587→// ─── Helper: Ichimoku Cloud (9, 26, 52) ──────────────────────────────────────
   588→function calcIchimoku(data: OHLCV[]): {
   589→  tenkan: number; kijun: number; senkouA: number; senkouB: number; chikou: number;
   590→} {
   591→  if (data.length < 52) {
   592→    const c = data[data.length - 1]?.close ?? 0;
   593→    return { tenkan: c, kijun: c, senkouA: c, senkouB: c, chikou: c };
   594→  }
   595→  const highs = data.map(d => d.high);
   596→  const lows = data.map(d => d.low);
   597→  const closes = data.map(d => d.close);
   598→
   599→  function donchian(h: number[], l: number[], end: number, period: number): number {
   600→    const start = end - period + 1;
   601→    if (start < 0) return (h[end] + l[end]) / 2;
   602→    let hi = -Infinity, lo = Infinity;
   603→    for (let i = start; i <= end; i++) {
   604→      if (h[i] > hi) hi = h[i];
   605→      if (l[i] < lo) lo = l[i];
   606→    }
   607→    return (hi + lo) / 2;
   608→  }
   609→
   610→  const n = data.length;
   611→  const tenkan = donchian(highs, lows, n - 1, 9);
   612→  const kijun = donchian(highs, lows, n - 1, 26);
   613→  // Senkou A = (tenkan + kijun) / 2, shifted 26 periods ahead
   614→  // We compute it at bar n-27 (shifted to current)
   615→  const senkouAIdx = Math.max(n - 1 - 26, 0);
   616→  const tA = donchian(highs, lows, senkouAIdx, 9);
   617→  const kA = donchian(highs, lows, senkouAIdx, 26);
   618→  const senkouA = (tA + kA) / 2;
   619→  // Senkou B = donchian(52), shifted 26 periods ahead
   620→  const senkouBIdx = Math.max(n - 1 - 26, 0);
   621→  const senkouB = donchian(highs, lows, senkouBIdx, 52);
   622→  // Chikou = current close plotted 26 periods back
   623→  const chikou = closes[n - 1];
   624→
   625→  return { tenkan, kijun, senkouA, senkouB, chikou };
   626→}

   628→function calcIchimokuArrays(data: OHLCV[]): {
   629→  tenkan: number[]; kijun: number[]; senkouA: number[]; senkouB: number[];
   630→} {
   631→  const n = data.length;
   632→  const highs = data.map(d => d.high);
   633→  const lows = data.map(d => d.low);
   634→
   635→  function donchian(end: number, period: number): number {
   636→    const start = end - period + 1;
   637→    if (start < 0) return (highs[end] + lows[end]) / 2;
   638→    let hi = -Infinity, lo = Infinity;
   639→    for (let i = start; i <= end; i++) {
   640→      if (highs[i] > hi) hi = highs[i];
   641→      if (lows[i] < lo) lo = lows[i];
   642→    }
   643→    return (hi + lo) / 2;
   644→  }
   645→
   646→  const tenkan: number[] = [];
   647→  const kijun: number[] = [];
   648→  const senkouA: number[] = [];
   649→  const senkouB: number[] = [];
   650→
   651→  for (let i = 0; i < n; i++) {
   652→    tenkan.push(donchian(i, 9));
   653→    kijun.push(donchian(i, 26));
   654→    // Senkou A shifted +26: at bar i, we need tenkan/kijun at bar i-26
   655→    const sIdx = i - 26;
   656→    if (sIdx >= 0) {
   657→      const t = donchian(sIdx, 9);
   658→      const k = donchian(sIdx, 26);
   659→      senkouA.push((t + k) / 2);
   660→    } else {
   661→      senkouA.push(donchian(i, 9)); // fallback
   662→    }
   663→    // Senkou B shifted +26: at bar i, we need donchian(52) at bar i-26
   664→    if (sIdx >= 0) {
   665→      senkouB.push(donchian(sIdx, 52));
   666→    } else {
   667→      senkouB.push(donchian(i, 52)); // fallback
   668→    }
   669→  }
   670→
   671→  return { tenkan, kijun, senkouA, senkouB };
   672→}

   675→function smaArray(closes: number[], period: number): number[] {
   676→  const result: number[] = [];
   677→  for (let i = 0; i < closes.length; i++) {
   678→    if (i < period - 1) { result.push(0); continue; }
   679→    let sum = 0;
   680→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   681→    result.push(sum / period);
   682→  }
   683→  return result;
   684→}

   686→// ─── Helper: EMA Array (full series) ──────────────────────────────────────
   687→function emaArrayCalc(closes: number[], period: number): number[] {
   688→  const result: number[] = [];
   689→  if (closes.length < period) return closes.map(() => 0);
   690→  const k = 2 / (period + 1);
   691→  let emaVal = 0;
   692→  for (let i = 0; i < period; i++) emaVal += closes[i];
   693→  emaVal /= period;
   694→  for (let i = 0; i < closes.length; i++) {
   695→    if (i < period) { result.push(0); continue; }
   696→    if (i === period) { result.push(emaVal); continue; }
   697→    emaVal = closes[i] * k + emaVal * (1 - k);
   698→    result.push(emaVal);
   699→  }
   700→  return result;
   701→}

   703→/**
   704→ * Ordinary Least Squares (OLS) linear regression on a time-series.
   705→ * Fits y = slope·x + intercept where x = 0, 1, 2, … (bar index).
   706→ *
   707→ * @param values - Sequential data points (oldest → newest)
   708→ * @returns `{ slope, intercept, r2 }` where:
   709→ *   - slope:      rate of change per bar
   710→ *   - intercept:  y-value at x=0
   711→ *   - r2:         coefficient of determination (0–1), measures fit quality;
   712→ *                 1 = perfect linear fit, 0 = no linear relationship
   713→ */
   714→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   715→  const n = values.length;
   716→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   717→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   718→  for (let i = 0; i < n; i++) {
   719→    sumX += i;
   720→    sumY += values[i];
   721→    sumXY += i * values[i];
   722→    sumX2 += i * i;
   723→    sumY2 += values[i] * values[i];
   724→  }
   725→  const denom = n * sumX2 - sumX * sumX;
   726→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   727→  const slope = (n * sumXY - sumX * sumY) / denom;
   728→  const intercept = (sumY - slope * sumX) / n;
   729→  const meanY = sumY / n;
   730→  let ssRes = 0, ssTot = 0;
   731→  for (let i = 0; i < n; i++) {
   732→    const predicted = slope * i + intercept;
   733→    ssRes += (values[i] - predicted) ** 2;
   734→    ssTot += (values[i] - meanY) ** 2;
   735→  }
   736→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   737→  return { slope, intercept, r2 };
   738→}

   833→// ─── Helper: Linear System Solver (Gaussian Elimination) ─────────────────
   834→function solveLinearSystem(A: number[][], b: number[]): number[] | null {
   835→  const n = b.length;
   836→  const aug = A.map((row, i) => [...row, b[i]]);
   837→
   838→  for (let col = 0; col < n; col++) {
   839→    let maxVal = Math.abs(aug[col][col]);
   840→    let maxRow = col;
   841→    for (let row = col + 1; row < n; row++) {
   842→      if (Math.abs(aug[row][col]) > maxVal) {
   843→        maxVal = Math.abs(aug[row][col]);
   844→        maxRow = row;
   845→      }
   846→    }
   847→    if (maxVal < 1e-10) return null;
   848→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   849→
   850→    for (let row = col + 1; row < n; row++) {
   851→      const factor = aug[row][col] / aug[col][col];
   852→      for (let j = col; j <= n; j++) {
   853→        aug[row][j] -= factor * aug[col][j];
   854→      }
   855→    }
   856→  }
   857→
   858→  const x = new Array(n).fill(0);
   859→  for (let i = n - 1; i >= 0; i--) {
   860→    x[i] = aug[i][n];
   861→    for (let j = i + 1; j < n; j++) {
   862→      x[i] -= aug[i][j] * x[j];
   863→    }
   864→    if (Math.abs(aug[i][i]) < 1e-10) return null;
   865→    x[i] /= aug[i][i];
   866→  }
   867→
   868→  return x;
   869→}

   871→// ─── Helper: Generate All 7-Method S/R Levels ─────────────────────────────
   872→function generateAllLevels(data: OHLCV[], endIdx: number): RawSRLevel[] {
   873→  const levels: RawSRLevel[] = [];
   874→  if (endIdx < 10) return levels;
   875→
   876→  const slice = data.slice(0, endIdx + 1);
   877→  const currentPrice = slice[slice.length - 1].close;
   878→  const closes = slice.map(d => d.close);
   879→  const noFib: [string, string] = ['—', ''];
   880→
   881→  // ── Method 1: Swing High/Low (Price Action) ──────────────────────────
   882→  for (const lookback of [3, 5, 7]) {
   883→    const swings = findSwingLevels(slice, lookback);
   884→    const tol = currentPrice * 0.005; // 0.5% tolerance
   885→
   886→    // Connect swing highs → resistance
   887→    const usedHigh = new Set<number>();
   888→    for (const h of swings.highs) {
   889→      if (usedHigh.has(h.idx)) continue;
   890→      let sum = h.price;
   891→      let count = 1;
   892→      for (const h2 of swings.highs) {
   893→        if (h2 === h || usedHigh.has(h2.idx)) continue;
   894→        if (Math.abs(h2.price - h.price) < tol) {
   895→          sum += h2.price;
   896→          count++;
   897→          usedHigh.add(h2.idx);
   898→        }
   899→      }
   900→      usedHigh.add(h.idx);
   901→      if (sum / count > currentPrice * 1.005) {
   902→        levels.push({ price: sum / count, method: 'Swing_High', type: 'resistance', ...noFib });
   903→      }
   904→    }
   905→
   906→    // Connect swing lows → support
   907→    const usedLow = new Set<number>();
   908→    for (const l of swings.lows) {
   909→      if (usedLow.has(l.idx)) continue;
   910→      let sum = l.price;
   911→      let count = 1;
   912→      for (const l2 of swings.lows) {
   913→        if (l2 === l || usedLow.has(l2.idx)) continue;
   914→        if (Math.abs(l2.price - l.price) < tol) {
   915→          sum += l2.price;
   916→          count++;
   917→          usedLow.add(l2.idx);
   918→        }
   919→      }
   920→      usedLow.add(l.idx);
   921→      if (sum / count < currentPrice * 0.995) {
   922→        levels.push({ price: sum / count, method: 'Swing_Low', type: 'support', ...noFib });
   923→      }
   924→    }
   925→  }
   926→
   927→  // ── Method 2: SMA Dynamic S/R ──────────────────────────────────────
   928→  for (const period of [20, 50, 100, 200]) {
   929→    const smaVal = sma(closes, period);
   930→    if (smaVal <= 0) continue;
   931→    const type: 'support' | 'resistance' = currentPrice > smaVal ? 'support' : 'resistance';
   932→    levels.push({ price: smaVal, method: `SMA_${period}`, type, ...noFib });
   933→  }
   934→
   935→  // ── Method 3: Bollinger Bands ───────────────────────────────────────
   936→  const bb = calcBollingerBands(closes);
   937→  if (bb.upper > 0 && bb.lower > 0) {
   938→    levels.push({ price: bb.upper, method: 'BB_Upper', type: 'resistance', ...noFib });
   939→    levels.push({ price: bb.lower, method: 'BB_Lower', type: 'support', ...noFib });
   940→  }
   941→
   942→  // ── Method 4: Fibonacci Retracement + Extension ─────────────────────
   943→  interface SwingPair { high: number; highIdx: number; low: number; lowIdx: number; range: number }
   944→  let bestSwing: SwingPair | null = null;
   945→
   946→  for (const lb of [3, 5, 7, 10, 15, 20]) {
   947→    const sw = findSwingLevels(slice, lb);
   948→    if (sw.highs.length === 0 || sw.lows.length === 0) continue;
   949→    for (const h of sw.highs) {
   950→      for (const l of sw.lows) {
   951→        if (Math.abs(h.idx - l.idx) < 3) continue;
   952→        const range = Math.abs(h.price - l.price);
   953→        if (range / currentPrice < 0.03) continue;
   954→        if (!bestSwing || range > bestSwing.range) {
   955→          bestSwing = {
   956→            high: Math.max(h.price, l.price),
   957→            highIdx: h.price > l.price ? h.idx : l.idx,
   958→            low: Math.min(h.price, l.price),
   959→            lowIdx: h.price < l.price ? l.idx : h.idx,
   960→            range,
   961→          };
   962→        }
   963→      }
   964→    }
   965→  }
   966→
   967→  if (bestSwing) {
   968→    const H = bestSwing.high;
   969→    const L = bestSwing.low;
   970→    const range = H - L;
   971→
   972→    for (const ratio of FIB_RETRACEMENTS) {
   973→      const price = H - ratio * range;
   974→      if (price <= 0) continue;
   975→      const rounded = roundToNice(price);
   976→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   977→      levels.push({
   978→        price: rounded, method: `Fibonacci_${ratio}`, type,
   979→        fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   980→      });
   981→    }
   982→
   983→    for (const extRatio of FIB_EXTENSIONS) {
   984→      const price = H + (extRatio - 1) * range;
   985→      if (price <= 0) continue;
   986→      const rounded = roundToNice(price);
   987→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   988→      levels.push({
   989→        price: rounded, method: `Fibonacci_${extRatio}`, type,
   990→        fibRatio: String(extRatio), fibLabel: FIB_LABELS[String(extRatio)] || '',
   991→      });
   992→    }
   993→
   994→    // Inverse extensions (below L)
   995→    for (const extRatio of FIB_EXTENSIONS) {
   996→      const price = L - (extRatio - 1) * range;
   997→      if (price <= 0) continue;
   998→      const rounded = roundToNice(price);
   999→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
  1000→      levels.push({
  1001→        price: rounded, method: `Fibonacci_${-extRatio}`, type,
  1002→        fibRatio: String(-extRatio), fibLabel: '',
  1003→      });
  1004→    }
  1005→
  1006→    // Secondary swing confluence (check overlap with secondary swings)
  1007→    const secondarySwings: SwingPair[] = [];
  1008→    for (const lb of [3, 5, 7]) {
  1009→      const sw = findSwingLevels(slice, lb);
  1010→      for (const h of sw.highs) {
  1011→        for (const l of sw.lows) {
  1012→          if (Math.abs(h.idx - l.idx) < 3) continue;
  1013→          const r = Math.abs(h.price - l.price);
  1014→          if (r / currentPrice < 0.05) continue;
  1015→          if (bestSwing && Math.abs(r - bestSwing.range) / bestSwing.range < 0.1) continue;
  1016→          secondarySwings.push({
  1017→            high: Math.max(h.price, l.price), highIdx: h.price > l.price ? h.idx : l.idx,
  1018→            low: Math.min(h.price, l.price), lowIdx: h.price < l.price ? l.idx : h.idx,
  1019→            range: r,
  1020→          });
  1021→        }
  1022→      }
  1023→    }
  1024→    // Secondary swings' fib levels are merged via confluence (1% rule in mergeLevels)
  1025→    const confluenceTol = currentPrice * 0.005;
  1026→    for (const sec of secondarySwings.slice(0, 3)) {
  1027→      const sH = sec.high, sL = sec.low, sR = sH - sL;
  1028→      for (const ratio of FIB_RETRACEMENTS) {
  1029→        const price = roundToNice(sH - ratio * sR);
  1030→        if (price <= 0) continue;
  1031→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
  1032→        levels.push({
  1033→          price, method: `Fibonacci_${ratio}`, type,
  1034→          fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
  1035→        });
  1036→      }
  1037→      for (const extR of FIB_EXTENSIONS) {
  1038→        const price = roundToNice(sH + (extR - 1) * sR);
  1039→        if (price <= 0) continue;
  1040→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
  1041→        levels.push({
  1042→          price, method: `Fibonacci_${extR}`, type,
  1043→          fibRatio: String(extR), fibLabel: FIB_LABELS[String(extR)] || '',
  1044→        });
  1045→      }
  1046→    }
  1047→  }
  1048→
  1049→  // ── Method 5: Volume Profile (VAP) ──────────────────────────────────
  1050→  const dataHasVolume = slice.some(d => d.volume > 0);
  1051→  if (dataHasVolume) {
  1052→    let minP = Infinity, maxP = -Infinity;
  1053→    for (const d of slice) {
  1054→      if (d.low < minP) minP = d.low;
  1055→      if (d.high > maxP) maxP = d.high;
  1056→    }
  1057→    const numBins = 40;
  1058→    const binSize = Math.max((maxP - minP) / numBins, currentPrice * 0.001);
  1059→    const vap = new Array(numBins).fill(0);
  1060→    const binCenters: number[] = [];
  1061→
  1062→    for (let b = 0; b < numBins; b++) {
  1063→      binCenters.push(minP + (b + 0.5) * binSize);
  1064→    }
  1065→
  1066→    for (const d of slice) {
  1067→      for (let b = 0; b < numBins; b++) {
  1068→        if (d.low <= binCenters[b] && binCenters[b] <= d.high) {
  1069→          vap[b] += d.volume;
  1070→        }
  1071→      }
  1072→    }
  1073→
  1074→    const maxVAP = Math.max(...vap);
  1075→    if (maxVAP > 0) {
  1076→      // POC (Point of Control)
  1077→      let pocIdx = 0;
  1078→      for (let b = 1; b < numBins; b++) {
  1079→        if (vap[b] > vap[pocIdx]) pocIdx = b;
  1080→      }
  1081→      const pocPrice = binCenters[pocIdx];
  1082→      const pocType: 'support' | 'resistance' = pocPrice > currentPrice ? 'resistance' : 'support';
  1083→      levels.push({ price: pocPrice, method: 'VAP_POC', type: pocType, ...noFib });
  1084→
  1085→      // HVN (High Volume Nodes)
  1086→      for (let b = 0; b < numBins; b++) {
  1087→        if (b === pocIdx) continue;
  1088→        if (vap[b] > 0.7 * maxVAP) {
  1089→          const hvnPrice = binCenters[b];
  1090→          const hvnType: 'support' | 'resistance' = hvnPrice > currentPrice ? 'resistance' : 'support';
  1091→          levels.push({ price: hvnPrice, method: 'VAP_HVN', type: hvnType, ...noFib });
  1092→        }
  1093→      }
  1094→    }
  1095→  }
  1096→
  1097→  // ── Method 6: Pivot Points (4 algorithms) ───────────────────────────
  1098→  if (endIdx >= 1) {
  1099→    const prev = slice[endIdx - 1]; // Previous completed bar
  1100→    const pH = prev.high, pL = prev.low, pC = prev.close, pO = prev.open;
  1101→
  1102→    // Standard Pivot
  1103→    {
  1104→      const PP = (pH + pL + pC) / 3;
  1105→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
  1106→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
  1107→      const R3 = pH + 2 * (PP - pL), S3 = pL - 2 * (pH - PP);
  1108→ if (R1 > 0) levels.push({ price: R1, method: 'Pivot_R1', type: 'resistance', ...noFib });
  1109→      if (R2 > 0) levels.push({ price: R2, method: 'Pivot_R2', type: 'resistance', ...noFib });
  1110→      if (R3 > 0) levels.push({ price: R3, method: 'Pivot_R3', type: 'resistance', ...noFib });
  1111→      if (S1 > 0) levels.push({ price: S1, method: 'Pivot_S1', type: 'support', ...noFib });
  1112→      if (S2 > 0) levels.push({ price: S2, method: 'Pivot_S2', type: 'support', ...noFib });
  1113→      if (S3 > 0) levels.push({ price: S3, method: 'Pivot_S3', type: 'support', ...noFib });
  1114→    }
  1115→
  1116→    // Fibonacci Pivot
  1117→    {
  1118→      const PP = (pH + pL + pC) / 3;
  1119→      const range = pH - pL;
  1120→      for (const [mult, suffix] of [[0.382, '1'], [0.618, '2'], [1.0, '3']] as [number, string][]) {
  1121→        const R = PP + mult * range;
  1122→        const S = PP - mult * range;
  1123→        if (R > 0) levels.push({ price: R, method: `Fib_Pivot_R${suffix}`, type: 'resistance', ...noFib });
  1124→        if (S > 0) levels.push({ price: S, method: `Fib_Pivot_S${suffix}`, type: 'support', ...noFib });
  1125→      }
  1126→    }
  1127→
  1128→    // Woodie's Pivot
  1129→    {
  1130→      const PP = (pH + pL + 2 * pC) / 4;
  1131→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
  1132→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
  1133→      if (R1 > 0) levels.push({ price: R1, method: 'Woodie_R1', type: 'resistance', ...noFib });
  1134→      if (R2 > 0) levels.push({ price: R2, method: 'Woodie_R2', type: 'resistance', ...noFib });
  1135→      if (S1 > 0) levels.push({ price: S1, method: 'Woodie_S1', type: 'support', ...noFib });
  1136→      if (S2 > 0) levels.push({ price: S2, method: 'Woodie_S2', type: 'support', ...noFib });
  1137→    }
  1138→
  1139→    // DeMark Pivot
  1140→    {
  1141→      const X = pC < pO ? pH + 2 * pL + pC : pC > pO ? 2 * pH + pL + pC : pH + pL + 2 * pC;
  1142→      const PP = X / 4;
  1143→      const R1 = X / 2 - pL;
  1144→      const S1 = X / 2 - pH;
  1145→      if (R1 > 0) levels.push({ price: R1, method: 'DeMark_R1', type: 'resistance', ...noFib });
  1146→      if (S1 > 0) levels.push({ price: S1, method: 'DeMark_S1', type: 'support', ...noFib });
  1147→    }
  1148→  }
  1149→
  1150→  // ── Method 7: Psychological Levels ──────────────────────────────────
  1151→  {
  1152→    const step = psychStep(currentPrice);
  1153→    const base = psychRound(currentPrice);
  1154→    for (const mult of [-2, -1, 1, 2]) {
  1155→      const price = base + mult * step;
  1156→      if (price <= 0) continue;
  1157→      const type: 'support' | 'resistance' = price > currentPrice ? 'resistance' : 'support';
  1158→      levels.push({ price, method: 'Psychological', type, ...noFib });
  1159→    }
  1160→  }
  1161→
  1162→  return levels;
  1163→}

  1165→// ─── Helper: Merge Nearby Levels (1% Confluence) ──────────────────────────
  1166→function mergeLevels(rawLevels: RawSRLevel[], currentPrice: number): MergedSRLevel[] {
  1167→  if (rawLevels.length === 0) return [];
  1168→
  1169→  const tol = currentPrice * 0.01; // 1% tolerance
  1170→  const merged: MergedSRLevel[] = [];
  1171→  const used = new Set<number>();
  1172→
  1173→  for (let i = 0; i < rawLevels.length; i++) {
  1174→    if (used.has(i)) continue;
  1175→
  1176→    let sumPrice = rawLevels[i].price;
  1177→    let count = 1;
  1178→    const methods = new Set<string>();
  1179→    methods.add(rawLevels[i].method);
  1180→    let bestFibRatio = rawLevels[i].fibRatio;
  1181→    let bestFibLabel = rawLevels[i].fibLabel;
  1182→
  1183→    for (let j = i + 1; j < rawLevels.length; j++) {
  1184→      if (used.has(j)) continue;
  1185→      if (Math.abs(rawLevels[j].price - rawLevels[i].price) < tol) {
  1186→        sumPrice += rawLevels[j].price;
  1187→        count++;
  1188→        methods.add(rawLevels[j].method);
  1189→        used.add(j);
  1190→        // Prefer Fibonacci label if available
  1191→        if (rawLevels[j].fibRatio !== '—' && bestFibRatio === '—') {
  1192→          bestFibRatio = rawLevels[j].fibRatio;
  1193→          bestFibLabel = rawLevels[j].fibLabel;
  1194→        }
  1195→      }
  1196→    }
  1197→    used.add(i);
  1198→
  1199→    const avgPrice = sumPrice / count;
  1200→    const type: 'support' | 'resistance' = avgPrice >= currentPrice ? 'resistance' : 'support';
  1201→
  1202→    merged.push({
  1203→      price: avgPrice,
  1204→      type,
  1205→      methods: [...methods],
  1206→      overlapCount: methods.size,
  1207→      fibRatio: bestFibRatio,
  1208→      fibLabel: bestFibLabel,
  1209→      score: 0,
  1210→      strength: 1,
  1211→      touchCount: 0,
  1212→      volumeRatio: 0,
  1213→      freshness: 0,
  1214→      distancePercent: 0,
  1215→    });
  1216→  }
  1217→
  1218→  return merged;
  1219→}

  1221→// ─── Helper: Compute Level Features (5 ML features) ──────────────────────
  1222→function computeLevelFeatures(
  1223→  levelPrice: number,
  1224→  data: OHLCV[],
  1225→  endIdx: number,
  1226→  currentPrice: number,
  1227→  overlapCount: number,
  1228→): number[] {
  1229→  const lookback = Math.min(30, endIdx);
  1230→  const startIdx = endIdx - lookback;
  1231→  const touchTol = currentPrice * 0.002; // 0.2%
  1232→
  1233→  let touchCount = 0;
  1234→  let totalVolAtTouches = 0;
  1235→  let totalVol = 0;
  1236→  let barsSinceLastTouch = lookback;
  1237→
  1238→  for (let i = startIdx; i <= endIdx; i++) {
  1239→    const bar = data[i];
  1240→    totalVol += bar.volume;
  1241→
  1242→    if (bar.low <= levelPrice + touchTol && bar.high >= levelPrice - touchTol) {
  1243→      touchCount++;
  1244→      totalVolAtTouches += bar.volume;
  1245→      barsSinceLastTouch = endIdx - i;
  1246→    }
  1247→  }
  1248→
  1249→  const avgVol = lookback > 0 ? totalVol / lookback : 0;
  1250→  const volumeRatio = (touchCount > 0 && avgVol > 0) ? (totalVolAtTouches / touchCount) / avgVol : 1;
  1251→  const freshness = Math.max(0, Math.min(1, 1 - barsSinceLastTouch / 30));
  1252→  const distancePercent = Math.min(1, Math.abs(levelPrice - currentPrice) / currentPrice);
  1253→
  1254→  // Features: [touch_count, volume_ratio, overlap_count, freshness, distance_percent]
  1255→  return [touchCount, volumeRatio, overlapCount - 1, freshness, distancePercent];
  1256→}

  1258→// ─── Helper: Train S/R Scoring Weights (Ridge OLS) ───────────────────────
  1259→function trainSRWeights(data: OHLCV[]): number[] {
  1260→  const defaultWeights = [0.25, 0.20, 0.25, 0.15, 0.15];
  1261→
  1262→  if (data.length < 50) return defaultWeights;
  1263→
  1264→  const X: number[][] = [];
  1265→  const y: number[] = [];
  1266→
  1267→  for (let i = 40; i <= data.length - 11; i += 5) {
  1268→    const barPrice = data[i].close;
  1269→    const rawLevels = generateAllLevels(data, i);
  1270→    if (rawLevels.length === 0) continue;
  1271→    const merged = mergeLevels(rawLevels, barPrice);
  1272→
  1273→    for (const level of merged) {
  1274→      const features = computeLevelFeatures(level.price, data, i, barPrice, level.overlapCount);
  1275→      X.push(features);
  1276→
  1277→      // Label: did price return within 10 bars?
  1278→      let returned = false;
  1279→      const retTol = barPrice * 0.005;
  1280→      for (let j = i + 1; j <= Math.min(i + 10, data.length - 1); j++) {
  1281→        if (data[j].low <= level.price + retTol && data[j].high >= level.price - retTol) {
  1282→          returned = true;
  1283→          break;
  1284→        }
  1285→      }
  1286→      y.push(returned ? 1 : 0);
  1287→    }
  1288→  }
  1289→
  1290→  if (X.length < 20) return defaultWeights;
  1291→
  1292→  // Ridge OLS: w = (X^T X + λI)^(-1) X^T y
  1293→  const n = X.length;
  1294→  const p = 5;
  1295→  const lambda = 0.01;
  1296→
  1297→  const XtX: number[][] = Array.from({ length: p }, () => new Array(p).fill(0));
  1298→  for (let i = 0; i < n; i++) {
  1299→    for (let a = 0; a < p; a++) {
  1300→      for (let b = 0; b < p; b++) {
  1301→        XtX[a][b] += X[i][a] * X[i][b];
  1302→      }
  1303→    }
  1304→  }
  1305→  for (let a = 0; a < p; a++) XtX[a][a] += lambda;
  1306→
  1307→  const Xty: number[] = new Array(p).fill(0);
  1308→  for (let i = 0; i < n; i++) {
  1309→    for (let a = 0; a < p; a++) {
  1310→      Xty[a] += X[i][a] * y[i];
  1311→    }
  1312→  }
  1313→
  1314→  const weights = solveLinearSystem(XtX, Xty);
  1315→  if (!weights) return defaultWeights;
  1316→  if (weights.some(w => isNaN(w) || !isFinite(w))) return defaultWeights;
  1317→
  1318→  for (let i = 0; i < p; i++) {
  1319→    if (weights[i] < 0) weights[i] = 0;
  1320→  }
  1321→
  1322→  const sum = weights.reduce((a, b) => a + b, 0);
  1323→  if (sum <= 0) return defaultWeights;
  1324→
  1325→  return weights.map(w => w / sum);
  1326→}

  1328→// ─── Helper: WMA Array (full series) ─────────────────────────────────────
  1329→function calcWMAArray(closes: number[], period: number): number[] {
  1330→  const result: number[] = [];
  1331→  if (closes.length < period) return closes.map(() => 0);
  1332→  for (let i = 0; i < closes.length; i++) {
  1333→    if (i < period - 1) { result.push(0); continue; }
  1334→    let numer = 0;
  1335→    let denom = 0;
  1336→    for (let j = 0; j < period; j++) {
  1337→      const weight = period - j;
  1338→      numer += closes[i - period + 1 + j] * weight;
  1339→      denom += weight;
  1340→    }
  1341→    result.push(numer / denom);
  1342→  }
  1343→  return result;
  1344→}

  1346→// ─── Helper: WMA (latest value) ────────────────────────────────────
  1347→function calcWMA(closes: number[], period: number): number {
  1348→  if (closes.length < period) return 0;
  1349→  const slice = closes.slice(-period);
  1350→  let numer = 0;
  1351→  let denom = 0;
  1352→  for (let i = 0; i < period; i++) {
  1353→    const weight = period - i;
  1354→    numer += slice[i] * weight;
  1355→    denom += weight;
  1356→  }
  1357→  return numer / denom;
  1358→}

  1360→// ─── Helper: HMA (Hull Moving Average) ─────────────────────────────
  1361→function calcHMA(closes: number[], period: number = 20): number {
  1362→  if (closes.length < period) return 0;
  1363→  const halfPeriod = Math.max(1, Math.floor(period / 2));
  1364→  const sqrtPeriod = Math.max(1, Math.floor(Math.sqrt(period)));
  1365→  const wmaHalf = calcWMAArray(closes, halfPeriod);
  1366→  const wmaFull = calcWMAArray(closes, period);
  1367→  // Build difference series: 2*WMA(half) - WMA(full)
  1368→  const diff: number[] = [];
  1369→  for (let i = 0; i < closes.length; i++) {
  1370→    diff.push(2 * wmaHalf[i] - wmaFull[i]);
  1371→  }
  1372→  return calcWMA(diff, sqrtPeriod);
  1373→}

  1375→// ─── Helper: TMA (Triangular Moving Average) ───────────────────────
  1376→function calcTMA(closes: number[], period: number = 20): number {
  1377→  if (closes.length < period) return 0;
  1378→  const halfPeriod = Math.max(1, Math.floor(period / 2) + 1);
  1379→  // SMA of SMA
  1380→  const firstSMA = smaArray(closes, halfPeriod);
  1381→  const nonZero = firstSMA.filter(v => v > 0);
  1382→  return nonZero.length >= halfPeriod ? sma(nonZero, halfPeriod) : (nonZero[nonZero.length - 1] ?? 0);
  1383→}

  1385→// ─── Helper: LMA (Linear Moving Average = (SMA+EMA)/2) ─────────────
  1386→function calcLMA(closes: number[], period: number = 20): number {
  1387→  return (sma(closes, period) + emaCalc(closes, period)) / 2;

  1390→// ─── Helper: MA Ribbon Alignment Score ─────────────────────────────
  1391→function calcMARibbonAlignment(closes: number[]): number {
  1392→  const emaPeriods = [5, 10, 20, 50, 100, 200];
  1393→  const emaVals: number[] = [];
  1394→  for (const p of emaPeriods) {
  1395→    emaVals.push(emaCalc(closes, p));
  1396→  }
  1397→  // Count aligned pairs (short-term above long-term = bullish alignment)
  1398→  let aligned = 0;
  1399→  let total = 0;
  1400→  for (let i = 0; i < emaVals.length - 1; i++) {
  1401→    if (emaVals[i] > 0 && emaVals[i + 1] > 0) {
  1402→      total++;
  1403→      if (emaVals[i] > emaVals[i + 1]) aligned++;
  1404→    }
  1405→  }
  1406→  return total > 0 ? aligned / total : 0.5;
  1407→}

  1409→// ─── Helper: Heiken Ashi (latest bar) ──────────────────────────────
  1410→function calcHeikenAshi(data: OHLCV[]): { open: number; high: number; low: number; close: number } {
  1411→  if (data.length === 0) return { open: 0, high: 0, low: 0, close: 0 };
  1412→  let prevHAOpen = 0;
  1413→  let prevHAClose = 0;
  1414→  for (let i = 0; i < data.length; i++) {
  1415→    const haClose = (data[i].open + data[i].high + data[i].low + data[i].close) / 4;
  1416→    const haOpen = i === 0 ? (data[i].open + data[i].close) / 2 : (prevHAOpen + prevHAClose) / 2;
  1417→    const haHigh = Math.max(data[i].high, haOpen, haClose);
  1418→    const haLow = Math.min(data[i].low, haOpen, haClose);
  1419→    prevHAOpen = haOpen;
  1420→    prevHAClose = haClose;
  1421→    if (i === data.length - 1) {
  1422→      return { open: haOpen, high: haHigh, low: haLow, close: haClose };
  1423→    }
  1424→  }
  1425→  return { open: 0, high: 0, low: 0, close: 0 };
  1426→}

  1442→function calcFastStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3): number {
  1443→  if (data.length < kPeriod) return 50;
  1444→  const rawKs: number[] = [];
  1445→  for (let i = kPeriod - 1; i < data.length; i++) {
  1446→    let lowest = Infinity;
  1447→    let highest = -Infinity;
  1448→    for (let j = i - kPeriod + 1; j <= i; j++) {
  1449→      if (data[j].low < lowest) lowest = data[j].low;
  1450→      if (data[j].high > highest) highest = data[j].high;
  1451→    }
  1452→    const range = highest - lowest;
  1453→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
  1454→  }
  1455→  if (rawKs.length < smoothK) return rawKs[rawKs.length - 1] ?? 50;
  1456→  const slice = rawKs.slice(-smoothK);
  1457→  return slice.reduce((a, b) => a + b, 0) / smoothK;
  1458→}

  1460→// ─── Helper: Fisher Transform ──────────────────────────────────────
  1461→function calcFisherTransform(data: OHLCV[], period: number = 9): number {
  1462→  if (data.length < period) return 0;
  1463→  let fisher = 0;
  1464→  for (let i = period; i < data.length; i++) {
  1465→    let maxH = -Infinity;
  1466→    let minL = Infinity;
  1467→    for (let j = i - period + 1; j <= i; j++) {
  1468→      if (data[j].high > maxH) maxH = data[j].high;
  1469→      if (data[j].low < minL) minL = data[j].low;
  1470→    }
  1471→    const range = maxH - minL;
  1472→    if (range === 0) continue;
  1473→    const raw = 2 * ((data[i].close - minL) / range - 0.5);
  1474→    const val = Math.max(0.0001, Math.min(0.9999, (raw + 1) / 2));
  1475→    fisher = 0.5 * Math.log(val / (1 - val)) + 0.5 * fisher;
  1476→  }
  1477→  return fisher;
  1478→}

  1609→// ─── Helper: Historical Volatility ─────────────────────────────────
  1610→function calcHV(closes: number[], period: number = 20): number {
  1611→  if (closes.length < period + 1) return 0;
  1612→  const logReturns: number[] = [];
  1613→  for (let i = closes.length - period; i < closes.length; i++) {
  1614→    if (i < 1 || closes[i - 1] <= 0 || closes[i] <= 0) continue;
  1615→    logReturns.push(Math.log(closes[i] / closes[i - 1]));
  1616→  }
  1617→  if (logReturns.length < 2) return 0;
  1618→  const mean = logReturns.reduce((a, b) => a + b, 0) / logReturns.length;
  1619→  const variance = logReturns.reduce((a, b) => a + (b - mean) ** 2, 0) / (logReturns.length - 1);
  1620→  return Math.sqrt(variance) * Math.sqrt(252);
  1621→}

  1623→// ─── Support / Resistance (7-Method ML-Based System) ─────────────────────
  1624→/**
  1625→ * Multi-method Support & Resistance detection and scoring engine.
  1626→ *
  1627→ * Pipeline:
  1628→ *   1. **Generate** raw levels from 7 independent methods:
  1629→ *      - Swing High/Low (price action with lookbacks 3, 5, 7)
  1630→ *      - Fibonacci Retracements & Extensions
  1631→ *      - Pivot Points (Classic, Camarilla, Woodie)
  1632→ *      - Psychological Round Numbers
  1633→ *      - Bollinger Band touches
  1634→ *      - Volume Profile nodes
  1635→ *      - Ichimoku Cloud levels
  1636→ *   2. **Merge** nearby levels within 1% tolerance (confluence detection)
  1637→ *   3. **Train** ML weights on historical touch/volume/overlap data
  1638→ *   4. **Score** each level using 5 features: touches, volume ratio, overlap count,
  1639→ *      freshness, distance from current price
  1640→ *   5. **Round** to psychological price levels
  1641→ *   6. **Separate** into supports (below price) and resistances (above price)
  1642→ *   7. **Filter** with 5–10% gap rules and select up to 6 of each type
  1643→ *   8. **Assign** grades (Very Strong / Strong / Moderate / Weak) from ML score
  1644→ *
  1645→ * @param data         - OHLCV array (oldest → newest)
  1646→ * @param currentPrice - The current closing price, used as the support/resistance boundary
  1647→ * @returns Up to 6 supports, 6 resistances (as raw prices), plus scored {@link LevelStrength} arrays
  1648→ */
  1649→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
  1650→  resistances: number[];
  1651→  supports: number[];
  1652→  supportStrengths: LevelStrength[];
  1653→  resistanceStrengths: LevelStrength[];
  1654→  priceTargets: LevelStrength[];
  1655→} {
  1656→  if (data.length < 10) {
  1657→    return {
  1658→      resistances: [], supports: [],
  1659→      supportStrengths: [], resistanceStrengths: [],
  1660→      priceTargets: [],
  1661→    };
  1662→  }
  1663→
  1664→  // ── Step 1: Generate all levels from 7 methods ────────────────────────
  1665→  const rawLevels = generateAllLevels(data, data.length - 1);
  1666→
  1667→  // ── Step 2: Merge nearby levels (confluence within 1%) ────────────────
  1668→  let merged = mergeLevels(rawLevels, currentPrice);
  1669→
  1670→  // ── Step 3: Train ML weights on historical data ───────────────────────
  1671→  const weights = trainSRWeights(data);
  1672→
  1673→  // ── Step 4: Compute features & score each merged level ────────────────
  1674→  for (const level of merged) {
  1675→    const features = computeLevelFeatures(level.price, data, data.length - 1, currentPrice, level.overlapCount);
  1676→
  1677→    const touchScore = Math.min(features[0] / 5, 1);
  1678→    const volumeScore = Math.min(features[1] / 2, 1);
  1679→    const overlapScore = Math.min(features[2] * 2, 10) / 10;
  1680→    const freshnessScore = features[3];
  1681→    const distanceScore = 1 - features[4];
  1682→
  1683→    const powerScore = (
  1684→      weights[0] * touchScore +
  1685→      weights[1] * volumeScore +
  1686→      weights[2] * overlapScore +
  1687→      weights[3] * freshnessScore +
  1688→      weights[4] * distanceScore
  1689→    ) * 10;
  1690→
  1691→    level.score = Math.max(0, Math.min(10, powerScore));
  1692→    level.strength = Math.max(1, Math.min(10, Math.round(level.score)));
  1693→    level.touchCount = features[0];
  1694→    level.volumeRatio = features[1];
  1695→    level.freshness = features[3];
  1696→    level.distancePercent = features[4];
  1697→  }
  1698→
  1699→  // ── Step 5: Psychological rounding of all level prices ─────────────────
  1700→  for (const level of merged) {
  1701→    level.price = psychRound(level.price);
  1702→  }
  1703→
  1704→  // Re-classify types after rounding
  1705→  for (const level of merged) {
  1706→    level.type = level.price >= currentPrice ? 'resistance' : 'support';
  1707→  }
  1708→
  1709→  // Deduplicate after rounding (keep strongest at same price)
  1710→  {
  1711→    const deduped: MergedSRLevel[] = [];
  1712→    for (const level of merged) {
  1713→      const existing = deduped.find(d => Math.abs(d.price - level.price) < currentPrice * 0.001);
  1714→      if (existing) {
  1715→        if (level.score > existing.score) {
  1716→          Object.assign(existing, level);
  1717→        }
  1718→      } else {
  1719→        deduped.push(level);
  1720→      }
  1721→    }
  1722→    merged = deduped;
  1723→  }
  1724→
  1725→  // ── Step 6: Separate into supports and resistances ─────────────────────
  1726→  let supports = merged.filter(l => l.type === 'support');
  1727→  let resistances = merged.filter(l => l.type === 'resistance');
  1728→
  1729→  // ── Step 7: Distance filtering (strict 5-10% gap rules) ─────────────
  1730→  // Rules:
  1731→  //   1. Consecutive lines of same type: 5% ≤ gap ≤ 10%
  1732→  //   2. Nearest support ≤ 10% below price (can be < 5%)
  1733→  //   3. Nearest resistance ≤ 10% above price (can be < 5%)
  1734→
  1735→  const filterAndSelectLevels = (
  1736→    levels: MergedSRLevel[],
  1737→    type: 'support' | 'resistance'
  1738→  ): MergedSRLevel[] => {
  1739→    if (levels.length === 0) return [];
  1740→
  1741→    // Sort: supports descending (nearest first), resistances ascending (nearest first)
  1742→    const sorted = [...levels].sort((a, b) =>
  1743→      type === 'support' ? b.price - a.price : a.price - b.price
  1744→    );
  1745→
  1746→    // ── Rule 2/3: Nearest level must be within 10% of price ──
  1747→    const nearest = sorted[0];
  1748→    const distFromPrice = Math.abs(nearest.price - currentPrice) / currentPrice;
  1749→    if (distFromPrice > 0.10) {
  1750→      // Nearest level is too far — generate a synthetic level at ~5% from price
  1751→      const syntheticPrice = type === 'support'
  1752→        ? Math.round(currentPrice * 0.95 / psychStep(currentPrice)) * psychStep(currentPrice)
  1753→        : Math.round(currentPrice * 1.05 / psychStep(currentPrice)) * psychStep(currentPrice);
  1754→      const synthetic: MergedSRLevel = {
  1755→        price: syntheticPrice,
  1756→        type,
  1757→        score: 3,
  1758→        strength: 3,
  1759→        overlapCount: 1,
  1760→        methods: ['Psychological'],
  1761→        fibRatio: '—',
  1762→        fibLabel: '',
  1763→      };
  1764→      sorted.unshift(synthetic);
  1765→    }
  1766→
  1767→    // ── Rule 1: Select levels with 5-10% gaps between consecutive ──
  1768→    const result: MergedSRLevel[] = [sorted[0]];
  1769→
  1770→    for (let i = 1; i < sorted.length && result.length < 6; i++) {
  1771→      const prevPrice = result[result.length - 1].price;
  1772→      const currPrice = sorted[i].price;
  1773→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
  1774→
  1775→      if (gap < 0.05) {
  1776→        // Too close — keep the stronger one
  1777→        if (sorted[i].score > result[result.length - 1].score) {
  1778→          result[result.length - 1] = sorted[i];
  1779→        }
  1780→      } else if (gap <= 0.10) {
  1781→        // Perfect gap (5-10%) — accept
  1782→        result.push(sorted[i]);
  1783→      } else {
  1784→        // Gap > 10% — try to find an intermediate level
  1785→        let found = false;
  1786→        for (let j = i + 1; j < sorted.length; j++) {
  1787→          const midPrice = sorted[j].price;
  1788→          const gapFromPrev = Math.abs(midPrice - prevPrice) / prevPrice;
  1789→          const gapToNext = Math.abs(currPrice - midPrice) / midPrice;
  1790→          if (gapFromPrev >= 0.05 && gapFromPrev <= 0.10) {
  1791→            result.push(sorted[j]);
  1792→            found = true;
  1793→            // Skip levels that are too close to the inserted one
  1794→            i = j;
  1795→            break;
  1796→          }
  1797→        }
  1798→        if (!found) {
  1799→          // No intermediate found — generate a synthetic level at ~7.5% from prev
  1800→          const synthGap = 0.075;
  1801→          const syntheticPrice = type === 'support'
  1802→            ? Math.round((prevPrice * (1 - synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice)
  1803→            : Math.round((prevPrice * (1 + synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice);
  1804→          // Make sure it's not too close to next level
  1805→          const gapToNext = Math.abs(currPrice - syntheticPrice) / syntheticPrice;
  1806→          if (gapToNext >= 0.05) {
  1807→            const synthetic: MergedSRLevel = {
  1808→              price: syntheticPrice,
  1809→              type,
  1810→              score: 2,
  1811→              strength: 2,
  1812→              overlapCount: 1,
  1813→              methods: ['Psychological'],
  1814→              fibRatio: '—',
  1815→              fibLabel: '',
  1816→            };
  1817→            result.push(synthetic);
  1818→          }
  1819→        }
  1820→      }
  1821→    }
  1822→
  1823→    return result;
  1824→  };
  1825→
  1826→  const finalSupports = filterAndSelectLevels(supports, 'support');
  1827→  const finalResistances = filterAndSelectLevels(resistances, 'resistance');
  1828→
  1829→  // ── Step 9: Build LevelStrength objects ────────────────────────────────
  1830→  const toLevelStrength = (m: MergedSRLevel): LevelStrength => {
  1831→    let grade: string;
  1832→    if (m.score >= 8.5) grade = 'Very Strong';
  1833→    else if (m.score >= 7) grade = 'Strong';
  1834→    else if (m.score >= 5) grade = 'Moderate';
  1835→    else grade = 'Weak';
  1836→
  1837→    return {
  1838→      price: m.price,
  1839→      strength: m.strength,
  1840→      score: m.score,
  1841→      grade,
  1842→      isTarget: m.score >= 7,
  1843→      overlapCount: m.overlapCount,
  1844→      methods: m.methods,
  1845→      fibRatio: m.fibRatio,
  1846→      fibLabel: m.fibLabel,
  1847→    };
  1848→  };
  1849→
  1850→  const supportStrengths = finalSupports.map(toLevelStrength);
  1851→  const resistanceStrengths = finalResistances.map(toLevelStrength);
  1852→
  1853→  // Price targets: all levels with isTarget=true
  1854→  const priceTargets = [...supportStrengths, ...resistanceStrengths].filter(l => l.isTarget);
  1855→
  1856→  return {
  1857→    resistances: resistanceStrengths.map(l => l.price),
  1858→    supports: supportStrengths.map(l => l.price),
  1859→    supportStrengths,
  1860→    resistanceStrengths,
  1861→    priceTargets,
  1862→  };
  1863→}

  1866→// ─── Trend Analysis ───────────────────────────────────────────────────────────
  1867→/**
  1868→ * Detect the price trend direction and angle over the last `period` bars.
  1869→ *
  1870→ * Uses OLS linear regression to compute the slope, then converts it to an
  1871→ * angle in degrees via:
  1872→ *
  1873→ *   angle = atan( (slope / avgPrice) × 100 ) × (180 / π)
  1874→ *
  1875→ * The ×100 scaling factor is critical: it converts `slope/avgPrice` from a
  1876→ * raw decimal (e.g. 0.01 for a 1% daily rise) into a percentage (1.0), so that
  1877→ * `atan` receives values in a human-readable range. Without this factor, large
  1878→ * absolute prices (e.g. an index at 6.7M with 1% daily change) would yield
  1879→ * `atan(0.01) ≈ 0.57°` instead of the expected `atan(1) = 45°`.
  1880→ *
  1881→ * Direction is determined by comparing slope to a 0.1% of avgPrice threshold:
  1882→ *   - slope >  +0.1% × avgPrice → 'up'
  1883→ *   - slope <  −0.1% × avgPrice → 'down'
  1884→ *   - otherwise               → 'flat'
  1885→ *
  1886→ * @param closes - Array of closing prices (oldest → newest)
  1887→ * @param period - Number of recent bars to regress (e.g. 21=short, 50=medium, 100=long)
  1888→ * @returns {@link TrendResult} with direction, slope, angle, and clamped r²
  1889→ */
  1890→function calcTrend(closes: number[], period: number): TrendResult {
  1891→  const slice = closes.slice(-Math.min(period, closes.length));
  1892→  const { slope, r2 } = linearRegression(slice);
  1893→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
  1894→  // Convert slope/avgPrice to percentage before atan so that:
  1895→  //   1% daily rise → atan(1) = 45°, 0.5% → ~26.6°, 2% → ~63.4°
  1896→  // Without the ×100 factor, atan(slope/avgPrice) always yields near-0° for
  1897→  // large prices (e.g. index at 6.7M with 1% daily change → 0.57°).
  1898→  const angle = avgPrice === 0 ? 0 : Math.atan((slope / avgPrice) * 100) * (180 / Math.PI);
  1899→  const priceThreshold = avgPrice * 0.001; // 0.1%
  1900→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
  1901→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
  1902→}
