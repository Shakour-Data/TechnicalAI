   470→  for (const period of [5, 10, 21, 50, 100, 200]) {
   471→    const v = sma(closes, period);
   472→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01)
   473→      allRaw.push({ price: v, source: 'ma' });
   474→  }
   475→  const ema12 = emaCalc(closes, 12);
   476→  const ema26 = emaCalc(closes, 26);
   477→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01)
   478→    allRaw.push({ price: ema12, source: 'ma' });
   479→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01)
   480→    allRaw.push({ price: ema26, source: 'ma' });
   481→
   482→  // 4. Bollinger Band levels (source: 'bb')
   483→  const bbLevels = calcBollingerBands(closes);
   484→  if (bbLevels.upper > 0) allRaw.push({ price: bbLevels.upper, source: 'bb' });
   485→  if (bbLevels.lower > 0) allRaw.push({ price: bbLevels.lower, source: 'bb' });
   486→
   487→  // 5. Round number / psychological levels (source: 'round')
   488→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
   489→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
   490→    const p = Math.round(currentPrice * mult / magnitude) * magnitude;
   491→    allRaw.push({ price: p, source: 'round' });
   492→  }
   493→
   494→  // 6. Recent high/low zones (source: 'hilo')
   495→  for (const w of [5, 22, 66]) {
   496→    const slice = data.slice(-Math.min(w, data.length));
   497→    if (slice.length > 0) {
   498→      allRaw.push({ price: Math.max(...slice.map(d => d.high)), source: 'hilo' });
   499→      allRaw.push({ price: Math.min(...slice.map(d => d.low)), source: 'hilo' });
   500→    }
   501→  }
   502→
   503→  // 7. Round all and filter invalid
   504→  const rounded: TaggedLevel[] = allRaw
   505→    .map(l => ({ price: roundToNice(l.price), source: l.source }))
   506→    .filter(l => l.price > 0);
   507→
   508→  // 8. Cluster nearby levels (within 0.5%) — merge sources, keep most common price
   509→  interface Cluster {
   510→    price: number;
   511→    sources: Set<string>;
   512→    touchCount: number;
   513→    totalVolumeAtTouches: number;
   514→    mostRecentTouchBar: number;
   515→  }
   516→  const clusters: Cluster[] = [];
   517→  for (const lvl of rounded) {
   518→    let merged = false;
   519→    for (const c of clusters) {
   520→      if (Math.abs(c.price - lvl.price) / Math.max(c.price, 1) < 0.005) {
   521→        c.sources.add(lvl.source);
   522→        merged = true;
   523→        break;
   524→      }
   525→    }
   526→    if (!merged) {
   527→      clusters.push({ price: lvl.price, sources: new Set([lvl.source]), touchCount: 0, totalVolumeAtTouches: 0, mostRecentTouchBar: -1 });
   528→    }
   529→  }
   530→
   531→  // 9. Count actual price TOUCHES (rejections) and volume at each cluster level
   532→  // A real touch = bar wick reached the level AND bar body closed away from it
   533→  const touchThreshold = currentPrice * 0.002; // 0.2% tight proximity
   534→  for (let i = 0; i < data.length; i++) {
   535→    const bar = data[i];
   536→    for (const c of clusters) {
   537→      const isResistance = c.price > currentPrice;
   538→      if (isResistance) {
   539→        // Resistance: bar high reached near the level AND closed below it (rejection)
   540→        if (bar.high >= c.price - touchThreshold && bar.close < c.price) {
   541→          c.touchCount++;
   542→          c.totalVolumeAtTouches += bar.volume;
   543→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   544→        }
   545→      } else {
   546→        // Support: bar low reached near the level AND closed above it (rejection)
   547→        if (bar.low <= c.price + touchThreshold && bar.close > c.price) {
   548→          c.touchCount++;
   549→          c.totalVolumeAtTouches += bar.volume;
   550→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   551→        }
   552→      }
   553→    }
   554→  }
   555→
   556→  // 10. Calculate multi-factor strength (0-10)
   557→  const levelStrengths: LevelStrength[] = clusters.map(c => {
   558→    // Factor 1: Source Confluence (0-2.0 pts)
   559→    const sourceScore = Math.min(2.0, c.sources.size * 0.4);
   560→    const hasSwingAndPivot = c.sources.has('swing') && c.sources.has('pivot') ? 0.4 : 0;
   561→    const maConfluence = c.sources.has('ma') ? 0.4 : 0;
   562→    const confluenceScore = Math.min(2.0, sourceScore + hasSwingAndPivot + maConfluence);
   563→
   564→    // Factor 2: Touch Count — CAPPED by recency (0-1.5 pts)
   565→    // Only count touches in the most recent 50% of data to avoid ancient history inflating
   566→    const recentBarStart = Math.floor(data.length * 0.5);
   567→    const recentTouches = c.mostRecentTouchBar >= recentBarStart ? c.touchCount : Math.max(0, c.touchCount - 1);
   568→    let touchScore: number;
   569→    if (recentTouches === 0) touchScore = 0;
   570→    else if (recentTouches === 1) touchScore = 0.3;
   571→    else if (recentTouches <= 2) touchScore = 0.6;
   572→    else if (recentTouches <= 4) touchScore = 0.9;
   573→    else if (recentTouches <= 7) touchScore = 1.2;
   574→    else touchScore = 1.5;
   575→
   576→    // Factor 3: Volume Confirmation (0-1.0 pts) — skip when no volume available
   577→    let volumeScore = 0;
   578→    const dataHasVolume = data.some(d => d.volume > 0);
   579→    if (dataHasVolume && c.touchCount > 0) {
   580→      const avgVolAtTouch = c.totalVolumeAtTouches / c.touchCount;
   581→      const volRatio = avgVolume > 0 ? avgVolAtTouch / avgVolume : 1;
   582→      if (volRatio > 2.0) volumeScore = 1.0;
   583→      else if (volRatio > 1.5) volumeScore = 0.8;
   584→      else if (volRatio > 1.0) volumeScore = 0.5;
   585→      else if (volRatio > 0.7) volumeScore = 0.25;
   586→      else volumeScore = 0.1;
   587→    }
   588→
   589→    // Factor 4: Freshness / Recency (0-1.0 pts)
   590→    let freshnessScore = 0;
   591→    if (c.touchCount > 0 && data.length > 0) {
   592→      const barsAgo = data.length - 1 - c.mostRecentTouchBar;
   593→      if (barsAgo <= 3) freshnessScore = 1.0;
   594→      else if (barsAgo <= 10) freshnessScore = 0.8;
   595→      else if (barsAgo <= 25) freshnessScore = 0.5;
   596→      else if (barsAgo <= 50) freshnessScore = 0.25;
   597→      else freshnessScore = 0.05;
   598→    }
   599→
   600→    // Factor 5: Proximity to current price (0-0.5 pts)
   601→    const distRatio = Math.abs(c.price - currentPrice) / currentPrice;
   602→    let proximityScore = 0;
   603→    if (distRatio < 0.01) proximityScore = 0.5;
   604→    else if (distRatio < 0.02) proximityScore = 0.35;
   605→    else if (distRatio < 0.04) proximityScore = 0.2;
   606→    else if (distRatio < 0.07) proximityScore = 0.1;
   607→
   608→    const total = confluenceScore + touchScore + volumeScore + freshnessScore + proximityScore;
   609→    const strength = Math.min(10, Math.max(1, Math.round(total * 10 / 6.0)));
   610→
   611→    return { price: c.price, strength, isTarget: false };
   612→  });
   613→
   614→  // 11. Separate above/below current price
   615→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   616→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   617→
   618→  // 12. Adaptive gap enforcement — keep highest strength in cluster
   619→  // v2: enforce 5-10% gap between consecutive S/R lines
   620→  function enforceGap(levels: LevelStrength[], targetCount: number): LevelStrength[] {
   621→    const gapSteps = [0.10, 0.09, 0.08, 0.07, 0.06, 0.05];
   622→    for (const gapPct of gapSteps) {
   623→      if (levels.length === 0) return [];
   624→      const result: LevelStrength[] = [levels[0]];
   625→      for (let i = 1; i < levels.length; i++) {
   626→        if (result.length >= targetCount) break;
   627→        const prevPrice = result[result.length - 1].price;
   628→        const currPrice = levels[i].price;
   629→        const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   630→        if (gap >= gapPct) {
   631→          result.push(levels[i]);
   632→        } else {
   633→          // Merge: keep stronger level's strength (no inflation)
   634→          const keep = levels[i].strength >= result[result.length - 1].strength ? levels[i] : result[result.length - 1];
   635→          result[result.length - 1] = { ...keep, strength: keep.strength };
   636→        }
   637→      }
   638→      if (result.length >= targetCount) return result;
   639→    }
   640→    return levels.slice(0, targetCount);
   641→  }
   642→
   643→  const finalResistances = enforceGap(above, 6);
   644→  const finalSupports = enforceGap(below, 6);
   645→
   646→  // 13. Determine price targets: top 2 by strength, or any above 7
   647→  const allLevels = [...finalResistances, ...finalSupports];
   648→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   649→  let targets: LevelStrength[];
   650→  if (sortedByStrength.some(l => l.strength >= 7)) {
   651→    targets = sortedByStrength.filter(l => l.strength >= 7).slice(0, 3);
   652→  } else {
   653→    targets = sortedByStrength.slice(0, 2);
   654→  }
   655→
   656→  // Mark targets
   657→  targets = targets.map(t => ({ ...t, isTarget: true }));
   658→
   659→  // Also mark in final arrays
   660→  finalResistances.forEach(r => {
   661→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   662→  });
   663→  finalSupports.forEach(s => {
   664→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   665→  });
   666→
   667→  return {
   668→    resistances: finalResistances.map(l => l.price),
   669→    supports: finalSupports.map(l => l.price),
   670→    supportStrengths: finalSupports,
   671→    resistanceStrengths: finalResistances,
   672→    priceTargets: targets,
   673→  };
   674→}

   676→// ─── Trend Analysis ───────────────────────────────────────────────────────────
   677→function calcTrend(closes: number[], period: number): TrendResult {
   678→  const slice = closes.slice(-Math.min(period, closes.length));
   679→  const { slope, r2 } = linearRegression(slice);
   680→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   681→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   682→  const priceThreshold = avgPrice * 0.001; // 0.1%
   683→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   684→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   685→}

   692→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
   693→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
   694→  const slice = data.slice(0, barIdx + 1);
   695→  const price = slice[slice.length - 1].close;
   696→  let nearestR = price * 1.05;
   697→  let nearestS = price * 0.95;
   698→
   699→  // Collect all swing highs (resistance) and swing lows (support)
   700→  const resistances: number[] = [];
   701→  const supports: number[] = [];
   702→  for (const lb of [3, 5, 7]) {
   703→    const swings = findSwingLevels(slice, lb);
   704→    resistances.push(...swings.highs);
   705→    supports.push(...swings.lows);
   706→  }
   707→
   708→  // Find nearest resistance above price
   709→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
   710→  if (abovePrice.length > 0) nearestR = abovePrice[0];
   711→
   712→  // Find nearest support below price
   713→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
   714→  if (belowPrice.length > 0) nearestS = belowPrice[0];
   715→
   716→  return { R1: nearestR, S1: nearestS };
   717→}

   719→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
   720→// Returns true if bullish divergence detected, false if bearish, null if none
   721→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
   722→  const lookback = 20;
   723→  const start = Math.max(0, barIdx - lookback);
   724→  const slice = data.slice(start, barIdx + 1);
   725→  if (slice.length < 10) return null;
   726→
   727→  const mid = Math.floor(slice.length / 2);
   728→  const firstHalf = slice.slice(0, mid);
   729→  const secondHalf = slice.slice(mid);
   730→
   731→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
   732→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
   733→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
   734→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
   735→
   736→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
   737→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
   738→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
   739→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
   740→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
   741→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
   742→
   743→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
   744→    return false; // bearish divergence
   745→  }
   746→
   747→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
   748→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
   749→
   750→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
   751→    return true; // bullish divergence
   752→  }
   753→
   754→  return null;
   755→}

   761→/**
   762→ * Computes all 16 ML features at a specific historical bar.
   763→ * Uses data[0..barIdx] (inclusive) for all indicator calculations.
   764→ * Returns an object keyed by feature name matching FEATURES_ORDER.
   765→ */
   766→function computeFeaturesAtBar(data: OHLCV[], barIdx: number): Record<string, number> {
   767→  // Need at least 30 bars for meaningful indicators
   768→  if (barIdx < 30) {
   769→    const empty: Record<string, number> = {};
   770→    for (const key of FEATURES_ORDER) empty[key] = 0.5;
   771→    return empty;
   772→  }
   773→
   774→  const slice = data.slice(0, barIdx + 1);
   775→  const closes = slice.map(d => d.close);
   776→  const price = closes[closes.length - 1];
   777→  const eps = 1e-10;
   778→
   779→  // ── Compute all base indicators ───────────────────────────────────────────
   780→  const rsiVal = calcRSI(closes);
   781→  const sliceHasVolume = slice.some(d => d.volume > 0);
   782→  const mfiVal = sliceHasVolume ? calcMFI(slice) : 50; // neutral when no volume
   783→  const cciVal = calcCCI(slice);
   784→  const stoch = calcStochastic(slice);
   785→  const macd = calcMACD(closes);
   786→  const adxResult = calcADX(slice);
   787→  const atrVal = calcATR(slice);
   788→  const bb = calcBollingerBands(closes);
   789→  const ma21 = sma(closes, 21);
   790→  const ma100 = sma(closes, 100);
   791→  const ema12 = emaCalc(closes, 12);
   792→  const ema26 = emaCalc(closes, 26);
   793→  const trendResult = calcTrend(closes, 21);
   794→  const sr = simplifiedSR(data, barIdx);
   795→
   796→  // ── Layer 1: Raw Scores (s_*) ────────────────────────────────────────────
   797→
   798→  // s_rsi — RSI mapped to 0-1
   799→  const s_rsi = rsiVal > 80 ? 0.95
   800→    : rsiVal > 70 ? 0.85
   801→    : rsiVal > 60 ? 0.70
   802→    : rsiVal > 50 ? 0.55
   803→    : rsiVal > 40 ? 0.45
   804→    : rsiVal > 30 ? 0.30
   805→    : rsiVal > 20 ? 0.15
   806→    : 0.05;
   807→
   808→  // s_mfi — MFI mapped to 0-1 (same formula as RSI)
   809→  const s_mfi = mfiVal > 80 ? 0.95
   810→    : mfiVal > 70 ? 0.85
   811→    : mfiVal > 60 ? 0.70
   812→    : mfiVal > 50 ? 0.55
   813→    : mfiVal > 40 ? 0.45
   814→    : mfiVal > 30 ? 0.30
   815→    : mfiVal > 20 ? 0.15
   816→    : 0.05;
   817→
   818→  // s_cci — CCI mapped to 0-1
   819→  const s_cci = cciVal > 200 ? 0.90
   820→    : cciVal > 100 ? 0.75
   821→    : cciVal > 0 ? 0.60
   822→    : cciVal > -100 ? 0.40
   823→    : cciVal > -200 ? 0.25
   824→    : 0.10;
   825→
   826→  // s_adx — Directional bias
   827→  const s_adx = (adxResult.diPlus - adxResult.diMinus + 100) / 200;
   828→
   829→  // s_macd — MACD histogram + signal combined
   830→  const absMacdLine = Math.abs(macd.line) + 1;
   831→  const s_macd = macd.line > macd.signal
   832→    ? 0.5 + Math.min(0.5, macd.histogram / absMacdLine)
   833→    : 0.5 - Math.min(0.5, Math.abs(macd.histogram) / absMacdLine);
   834→
   835→  // s_stoch — Stochastic %K mapped to 0-1
   836→  const s_stoch = stoch.k / 100;
   837→
   838→  // s_bb — Bollinger Band position
   839→  const bbRange = bb.upper - bb.lower;
   840→  const s_bb = bbRange > eps
   841→    ? clamp((price - bb.lower) / bbRange, 0, 1)
   842→    : 0.5;
   843→
   844→  // s_ma21 — Price relative to 21-period SMA
   845→  const s_ma21 = ma21 > eps
   846→    ? clamp(0.5 + (price - ma21) / ma21, 0, 1)
   847→    : 0.5;
   848→
   849→  // s_ma100 — Price relative to 100-period SMA
   850→  const s_ma100 = ma100 > eps
   851→    ? clamp(0.5 + (price - ma100) / ma100, 0, 1)
   852→    : 0.5;
   853→
   854→  // s_ema — EMA12 vs EMA26 spread
   855→  const s_ema = ema26 > eps
   856→    ? clamp(0.5 + (ema12 - ema26) / ema26 * 5, 0, 1)
   857→    : 0.5;
   858→
   859→  // s_atr — Inverse volatility (lower ATR = higher score)
   860→  const s_atr = price > eps ? 1 - Math.min(atrVal / price * 10, 1) : 0.5;
   861→
   862→  // s_trend — R² + angle combination
   863→  const s_trend = (trendResult.r2 * 0.7) +
   864→    (Math.max(0, Math.min(1, (trendResult.angle + 45) / 90)) * 0.3);
   865→
   866→  // s_sr — Support/Resistance proximity
   867→  const dR1 = (sr.R1 - price) / (price > eps ? price : 1);
   868→  const dS1 = (price - sr.S1) / (price > eps ? price : 1);
   869→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
   870→
   871→  // ── Layer 2: Corrections (f_*) ───────────────────────────────────────────
   872→
   873→  // f_rsi — RSI with momentum correction
   874→  const rsiCurrent = rsiVal;
   875→  const rsi5ago = barIdx >= 5
   876→    ? calcRSI(closes.slice(0, barIdx - 5 + 1))
   877→    : rsiCurrent;
   878→  const f_rsi = clamp(s_rsi + (rsiCurrent - rsi5ago) / 100 * 0.08, 0, 1);
   879→
   880→  // f_mfi — MFI with momentum correction (skip when no volume)
   881→  const mfiCurrent = mfiVal;
   882→  const mfi5ago = (barIdx >= 5 && sliceHasVolume)
   883→    ? calcMFI(data.slice(0, barIdx - 5 + 1))
   884→    : mfiCurrent;
   885→  const f_mfi = sliceHasVolume
   886→    ? clamp(s_mfi + (mfiCurrent - mfi5ago) / 100 * 0.08, 0, 1)
   887→    : 0.5; // neutral when no volume
   888→
   889→  // f_cci — CCI with momentum correction (scale: /200 instead of /100)
   890→  const cciCurrent = cciVal;
   891→  const cci5ago = barIdx >= 5
   892→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
   893→    : cciCurrent;
   894→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
   895→
   896→  // f_macd — Same as s_macd (no additional correction)
   897→  const f_macd = s_macd;
   898→
   899→  // f_stoch — Same as s_stoch
   900→  const f_stoch = s_stoch;
   901→
   902→  // f_stochCross — Stochastic K/D crossover detection
   903→  let f_stochCross = 0.1;
   904→  if (barIdx >= 1) {
   905→    const prevSlice = data.slice(0, barIdx);
   906→    const prevStoch = calcStochastic(prevSlice);
   907→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
   908→      f_stochCross = 0.9; // K crossed above D (bullish)
   909→    }
   910→  }
   911→
   912→  // f_macdCross — MACD/Signal crossover detection
   913→  let f_macdCross = 0.1;
   914→  if (barIdx >= 1) {
   915→    const prevCloses = closes.slice(0, -1);
   916→    if (prevCloses.length >= 35) {
   917→      const prevMacd = calcMACD(prevCloses);
   918→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
   919→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
   920→      }
   921→    }
   922→  }
   923→
   924→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
   925→  const divResult = detectDivergenceSimple(data, barIdx);
   926→  const f_div = divResult === true ? 0.85 : 0.15;
   927→
   928→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
   929→  return {
   930→    rsi: f_rsi,
   931→    mfi: f_mfi,
   932→    cci: f_cci,
   933→    adx: s_adx,
   934→    macd: f_macd,
   935→    stoch: f_stoch,
   936→    bb: s_bb,
   937→    ma21: s_ma21,
   938→    ma100: s_ma100,
   939→    ema: s_ema,
   940→    atr: s_atr,
   941→    trend: s_trend,
   942→    sr: s_sr,
   943→    stochCross: f_stochCross,
   944→    macdCross: f_macdCross,
   945→    div: f_div,
   946→  };
   947→}
   948→
   949→// ═══════════════════════════════════════════════════════════════════════════════
   950→// TRAINING DATA CREATION (Layer 0 — Data Preparation)
   951→// ═══════════════════════════════════════════════════════════════════════════════
   952→
   953→/**
   954→ * Creates supervised training data from historical OHLCV data.
   955→ * For each bar, computes features and labels based on future price movement.
   956→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
   957→ * Samples every `sampleStep` bars to reduce computation.
   958→ */
   959→function createTrainingData(
   960→  data: OHLCV[],
   961→  horizon: number = 5,
   962→  sampleStep: number = 5
   963→): { X: number[][]; y: number[] } {
   964→  const X: number[][] = [];
   965→  const y: number[] = [];
   966→
   967→  // Start from horizon+30 to ensure enough data for indicators
   968→  // End at data.length - horizon - 1 to have future bars for labeling
   969→  // Sample every sampleStep bars for performance
   970→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
   971→    const features = computeFeaturesAtBar(data, i);
   972→
   973→    // Convert to array following FEATURES_ORDER
   974→    const featureVec: number[] = [];
   975→    let valid = true;
   976→    for (const key of FEATURES_ORDER) {
   977→      const val = features[key];
   978→      if (val === undefined || val === null || Number.isNaN(val)) {
   979→        valid = false;
   980→        break;
   981→      }
   982→      featureVec.push(val);
   983→    }
   984→    if (!valid) continue;
   985→
   986→    // Label: 1 if future close > 1% above current close
   987→    const currentClose = data[i].close;
   988→    const futureClose = data[i + horizon].close;
   989→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
   990→
   991→    X.push(featureVec);
   992→    y.push(label);
   993→  }
   994→
   995→  return { X, y };
   996→}
   997→
   998→// ═══════════════════════════════════════════════════════════════════════════════
   999→// VDss GRAPH STRUCTURE (Layer 5 — State Machine)
  1000→// ═══════════════════════════════════════════════════════════════════════════════
  1001→
  1002→/**
  1003→ * VDss Graph: Each node represents a market state.
  1004→ * Edges represent transitions with types: 'up', 'down', 'pullback', 'risk', 'terminal'.
  1005→ * Terminal nodes map to scenario results (R1=strong bullish, R5=deep correction).
  1006→ */
  1007→const VDSS_GRAPH = {
  1008→  // outgoing edges from each node: [target, type]
  1009→  outgoing: {
  1010→    A: [['B', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1011→    B: [['C', 'up'], ['G', 'pullback'], ['A', 'risk']] as [string, string][],
  1012→    C: [['D', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1013→    D: [['E', 'up'], ['B', 'pullback'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1014→    E: [['F', 'up'], ['D', 'pullback'], ['G', 'pullback'], ['H', 'down'], ['R1', 'terminal']] as [string, string][],
  1015→    F: [['R1', 'terminal'], ['E', 'pullback'], ['D', 'down'], ['G', 'risk']] as [string, string][],
  1016→    G: [['B', 'up'], ['A', 'pullback'], ['H', 'down'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1017→    H: [['G', 'pullback'], ['I', 'down'], ['B', 'up'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1018→    I: [['H', 'pullback'], ['J', 'down'], ['G', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1019→    J: [['I', 'pullback'], ['K', 'down'], ['H', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1020→    K: [['J', 'pullback'], ['I', 'up'], ['L', 'risk'], ['R4', 'terminal'], ['R5', 'terminal']] as [string, string][],
  1021→    L: [['C', 'up'], ['H', 'pullback'], ['J', 'down'], ['K', 'risk'], ['R5', 'terminal']] as [string, string][],
  1022→  },
  1023→  // which result scenario each terminal node maps to
  1024→  terminalMap: { R1: 'R1', R2: 'R2', R3: 'R3', R4: 'R4', R5: 'R5' } as Record<string, string>,
  1025→  startNode: 'A' as string,
  1026→};
  1027→
  1028→// ═══════════════════════════════════════════════════════════════════════════════
  1029→// LAYER 5 — Edge Weight Calculation
  1030→// ═══════════════════════════════════════════════════════════════════════════════
  1031→
  1032→/**
  1033→ * Calculates transition edge weights for the VDss graph.
  1034→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
  1035→ * to determine the relative probability of up/down/pullback/risk transitions.
  1036→ */
  1037→function calculateEdgeWeights(
  1038→  bullConsensus: number,
  1039→  adx: number,
  1040→  mlModel: AdaptiveWeightModel
  1041→): { up: number; down: number; pullback: number; risk: number } {
  1042→  // Extract ML coefficient insights if model is trained
  1043→  let trend_coef = 1 / 3;
  1044→  let momentum_coef = 1 / 3;
  1045→  let volatility_coef = 1 / 3;
  1046→
  1047→  if (mlModel.isTrained && mlModel.weights) {
  1048→    const coefs = mlModel.getCoefficients();
  1049→    // Map feature indices to coefficient categories
  1050→    // FEATURES_ORDER: ['rsi','mfi','cci','adx','macd','stoch','bb','ma21','ma100','ema','atr','trend','sr','stochCross','macdCross','div']
  1051→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER
  1052→    const rsiIdx = 0;     // 'rsi' in FEATURES_ORDER
  1053→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
  1054→
  1055→    const absTrend = Math.abs(coefs[trendIdx] ?? 0);
  1056→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
  1057→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
  1058→
  1059→    const coefSum = absTrend + absRsi + absAtr;
  1060→    if (coefSum > 1e-10) {
  1061→      trend_coef = absTrend / coefSum;
  1062→      momentum_coef = absRsi / coefSum;
  1063→      volatility_coef = absAtr / coefSum;
  1064→    }
  1065→  }
  1066→
  1067→  // Base weights driven by bull consensus
  1068→  const up_base = bullConsensus * 0.7 + 0.15;
  1069→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
  1070→  const pullback_base = 0.25;
  1071→  const risk_base = 0.12 * (1.2 - adx / 100);
  1072→
  1073→  // Apply ML coefficient adjustments
  1074→  const up = clamp(
  1075→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
  1076→    0.05, 0.95
  1077→  );
  1078→  const down = clamp(
  1079→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
  1080→    0.05, 0.95
  1081→  );
  1082→  const pullback = clamp(
  1083→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
  1084→    0.05, 0.50
  1085→  );
  1086→  const risk = clamp(
  1087→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
  1088→    0.02, 0.30
  1089→  );
  1090→
  1091→  return { up, down, pullback, risk };
  1092→}
  1093→
  1094→// ═══════════════════════════════════════════════════════════════════════════════
  1095→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)
  1096→// ═══════════════════════════════════════════════════════════════════════════════
  1097→
  1098→/**
  1099→ * Performs iterative DFS from the start node through the VDss graph,
  1100→ * computing raw path probabilities, then calibrating them against
  1101→ * Layer 4 scenario probabilities.
  1102→ *
  1103→ * Returns:
  1104→ * - calibratedPaths: Map<pathString, probability>
  1105→ * - calibrationFactors: per-scenario multiplier used
  1106→ * - scenarioSums: raw probability sums per scenario before calibration
  1107→ */
  1108→function calculatePathProbabilities(
  1109→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
  1110→  scenarioProbs: Record<string, number>
  1111→): {
  1112→  calibratedPaths: Map<string, number>;
  1113→  calibrationFactors: Record<string, number>;
  1114→  scenarioSums: Record<string, number>;
  1115→} {
  1116→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
  1117→  const MAX_PATHS = 200;
  1118→
  1119→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
  1120→  type StackEntry = [string, string, number];
  1121→
  1122→  // Collect all terminal paths with raw probabilities
  1123→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
  1124→
  1125→  // Iterative DFS
  1126→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
  1127→
  1128→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
  1129→    const [node, pathStr, pathProb] = stack.pop()!;
  1130→
  1131→    const edges = outgoing[node as keyof typeof outgoing];
  1132→    if (!edges) continue;
  1133→
  1134→    // Normalize outgoing edge weights to probabilities for this node
  1135→    // Map edge types to their weights
  1136→    let totalWeight = 0;
  1137→    const edgeWeightsList: number[] = [];
  1138→    for (const [, edgeType] of edges) {
  1139→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
  1140→      edgeWeightsList.push(w);
  1141→      totalWeight += w;
  1142→    }
  1143→
  1144→    if (totalWeight < 1e-10) continue;
  1145→
  1146→    // Expand edges
  1147→    for (let i = 0; i < edges.length; i++) {
  1148→      const [target, edgeType] = edges[i];
  1149→      const edgeProb = edgeWeightsList[i] / totalWeight;
  1150→      const newProb = pathProb * edgeProb;
  1151→      const newPath = pathStr + '→' + target;
  1152→
  1153→      if (edgeType === 'terminal') {
  1154→        // Reached a terminal node
  1155→        const scenario = terminalMap[target] ?? 'R3';
  1156→        rawPaths.push({ path: newPath, scenario, prob: newProb });
  1157→      } else {
  1158→        // Continue DFS
  1159→        stack.push([target, newPath, newProb]);
  1160→      }
  1161→    }
  1162→  }
  1163→
  1164→  // Group raw probabilities by scenario
  1165→  const rawSums: Record<string, number> = { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0 };
  1166→  for (const rp of rawPaths) {
  1167→    rawSums[rp.scenario] = (rawSums[rp.scenario] ?? 0) + rp.prob;
  1168→  }
  1169→
  1170→  // Compute calibration factors: scenarioProbs[scenario] / rawSum[scenario]
  1171→  const calibrationFactors: Record<string, number> = {};
  1172→  for (const scenario of ['R1', 'R2', 'R3', 'R4', 'R5']) {
  1173→    const rawSum = rawSums[scenario] ?? 0;
  1174→    const targetProb = (scenarioProbs[scenario] ?? 20) / 100; // convert from percentage
  1175→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
  1176→  }
  1177→
  1178→  // Apply calibration to each path
  1179→  const calibratedPaths = new Map<string, number>();
  1180→  for (const rp of rawPaths) {
  1181→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
  1182→    calibratedPaths.set(rp.path, rp.prob * factor);
  1183→  }
  1184→
  1185→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
  1186→}
  1187→
  1188→// ═══════════════════════════════════════════════════════════════════════════════
  1189→// LAYER 4 — Scenario Probability Calculation (Adaptive with ML)
  1190→// ═══════════════════════════════════════════════════════════════════════════════
  1191→
  1192→/**
  1193→ * Calculates the 5 VDss scenario probabilities (R1-R5) that sum to 100.
  1194→ * Uses bull/bear consensus, key indicator values, S/R levels, and
  1195→ * ML-derived adaptive factors for fine-tuning.
  1196→ */
  1197→function calculateScenarioProbabilities(
  1198→  bullConsensus: number,
  1199→  price: number,
  1200→  R1: number,
  1201→  S1: number,
  1202→  MA100: number,
  1203→  rsi: number,
  1204→  mfi: number,
  1205→  stochK: number,
  1206→  mlModel: AdaptiveWeightModel
  1207→): { R1: number; R2: number; R3: number; R4: number; R5: number; factors: { momentum: number; volatility: number; trend: number } } {
  1208→  // Distance metrics to key levels (spec uses exp(-3 * ...))
  1209→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0;
  1210→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0;
  1211→
  1212→  // Binary risk flags (per spec) — MFI only used when volume is available
  1213→  const overboughtRisk = (rsi > 70 || (mfi > 80 && mfi !== 50) || stochK > 80) ? 1 : 0;
  1214→  const oversoldBounce = (rsi < 30 || (mfi < 20 && mfi !== 50) || stochK < 20) ? 1 : 0;
  1215→
  1216→  // Extract adaptive factors from ML model if trained
  1217→  let momentum = 0.7;
  1218→  let volatility = 0.5;
  1219→  let trend = 0.6;
  1220→
  1221→  if (mlModel.isTrained && mlModel.weights) {
  1222→    const coefs = mlModel.getCoefficients();
  1223→    momentum = sigmoid(coefs[0] ?? 0);   // rsi coefficient → momentum factor
  1224→    volatility = sigmoid(coefs[10] ?? 0); // atr coefficient → volatility factor
  1225→    trend = sigmoid(coefs[11] ?? 0);     // trend coefficient → trend factor
  1226→  }
  1227→
  1228→  // Below MA100 flag
  1229→  const belowMA100 = price < MA100 ? 1 : 0;
  1230→
  1231→  // ── Raw scenario probabilities (matching user spec) ─────────────────────
  1232→
  1233→  // R1: Strong Bullish — aggressive breakout
  1234→  const raw_R1 =
  1235→    bullConsensus * distR1 * (1 - overboughtRisk * momentum * 0.7) +
  1236→    oversoldBounce * 0.3 * distR1 * momentum;
  1237→
  1238→  // R2: Gradual Uptrend — steady rise
  1239→  const raw_R2 =
  1240→    ((1 - Math.abs(bullConsensus - 0.65)) * 0.8 +
  1241→    (bullConsensus > 0.4 ? 0.2 : 0)) * trend;
  1242→
  1243→  // R3: Range-bound
  1244→  const raw_R3 =
  1245→    (1 - Math.abs(bullConsensus - 0.5)) * 0.6 * (1 + (1 - volatility) * 0.2);
  1246→
  1247→  // R4: Moderate Correction
  1248→  const raw_R4 =
  1249→    ((1 - bullConsensus) * distS1 * 0.7 +
  1250→    overboughtRisk * 0.5 * momentum) *
  1251→    (0.8 + 0.2 * (1 - trend));
  1252→
  1253→  // R5: Deep Correction
  1254→  const raw_R5 =
  1255→    ((1 - bullConsensus) ** 2 * 0.5 + belowMA100 * 0.2) *
  1256→    (1 + (1 - trend) * 0.3);
  1257→
  1258→  // ── Normalize to sum = 100 ───────────────────────────────────────────
  1259→  const sum = raw_R1 + raw_R2 + raw_R3 + raw_R4 + raw_R5;
  1260→  const pR1 = sum > 0 ? Math.round(raw_R1 / sum * 100) : 20;
  1261→  const pR2 = sum > 0 ? Math.round(raw_R2 / sum * 100) : 20;
  1262→  const pR3 = sum > 0 ? Math.round(raw_R3 / sum * 100) : 20;
  1263→  const pR4 = sum > 0 ? Math.round(raw_R4 / sum * 100) : 20;
  1264→  const pR5 = 100 - (pR1 + pR2 + pR3 + pR4);
  1265→
  1266→  return {
  1267→    R1: pR1, R2: pR2, R3: pR3, R4: pR4, R5: pR5,
  1268→    factors: { momentum, volatility, trend },
  1269→  };
  1270→}
  1271→
  1272→// ═══════════════════════════════════════════════════════════════════════════════

  1322→  const cci = calcCCI(data);

  1353→  const R1_level = resistances[0] ?? price * 1.05;
