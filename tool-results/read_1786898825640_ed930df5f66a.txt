    75→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
    76→function sma(closes: number[], period: number): number {
    77→  if (closes.length < period) return 0;
    78→  const slice = closes.slice(-period);
    79→  return slice.reduce((a, b) => a + b, 0) / period;
    80→}

    82→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
    83→function emaCalc(closes: number[], period: number): number {
    84→  if (closes.length < period) return 0;
    85→  const k = 2 / (period + 1);
    86→  let emaVal = sma(closes.slice(0, period), period);
    87→  for (let i = period; i < closes.length; i++) {
    88→    emaVal = closes[i] * k + emaVal * (1 - k);
    89→  }
    90→  return emaVal;
    91→}

    93→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
    94→function calcRSI(closes: number[], period: number = 14): number {
    95→  if (closes.length < period + 1) return 50;
    96→  let avgGain = 0;
    97→  let avgLoss = 0;
    98→  for (let i = 1; i <= period; i++) {
    99→    const diff = closes[i] - closes[i - 1];
   100→    if (diff > 0) avgGain += diff;
   101→    else avgLoss += Math.abs(diff);
   102→  }
   103→  avgGain /= period;
   104→  avgLoss /= period;
   105→  for (let i = period + 1; i < closes.length; i++) {
   106→    const diff = closes[i] - closes[i - 1];
   107→    const gain = diff > 0 ? diff : 0;
   108→    const loss = diff < 0 ? Math.abs(diff) : 0;
   109→    avgGain = (avgGain * (period - 1) + gain) / period;
   110→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   111→  }
   112→  if (avgLoss === 0) return 100;
   113→  const rs = avgGain / avgLoss;
   114→  return 100 - 100 / (1 + rs);
   115→}

   117→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   118→function calcMFI(data: OHLCV[], period: number = 14): number {
   119→  if (data.length < period + 1) return 50;
   120→  let posFlow = 0;
   121→  let negFlow = 0;
   122→  for (let i = data.length - period; i < data.length; i++) {
   123→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   124→    const mf = tp * data[i].volume;
   125→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   126→    if (tp > prevTp) posFlow += mf;
   127→    else negFlow += mf;
   128→  }
   129→  if (negFlow === 0) return 100;
   130→  const mfr = posFlow / negFlow;
   131→  return 100 - 100 / (1 + mfr);
   132→}

   134→// ─── Helper: CCI(20) ──────────────────────────────────────────────────────────
   135→function calcCCI(data: OHLCV[], period: number = 20): number {
   136→  if (data.length < period) return 0;
   137→  const tps: number[] = [];
   138→  for (let i = data.length - period; i < data.length; i++) {
   139→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   140→  }
   141→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   142→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   143→  if (meanDev === 0) return 0;
   144→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   145→}

   147→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   148→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   149→  if (data.length < kPeriod) return { k: 50, d: 50 };
   150→  const rawKs: number[] = [];
   151→  for (let i = kPeriod - 1; i < data.length; i++) {
   152→    let lowest = Infinity;
   153→    let highest = -Infinity;
   154→    for (let j = i - kPeriod + 1; j <= i; j++) {
   155→      if (data[j].low < lowest) lowest = data[j].low;
   156→      if (data[j].high > highest) highest = data[j].high;
   157→    }
   158→    const range = highest - lowest;
   159→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   160→  }
   161→  const smoothedK: number[] = [];
   162→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   163→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   164→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   165→  }
   166→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   167→  const dSlice = smoothedK.slice(-smoothD);
   168→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   169→  return { k, d };
   170→}

   172→// ─── Helper: Williams %R(14) ──────────────────────────────────────────────────
   173→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   174→  if (data.length < period) return -50;
   175→  const slice = data.slice(-period);
   176→  let highest = -Infinity;
   177→  let lowest = Infinity;
   178→  for (const c of slice) {
   179→    if (c.high > highest) highest = c.high;
   180→    if (c.low < lowest) lowest = c.low;
   181→  }
   182→  const range = highest - lowest;
   183→  if (range === 0) return -50;
   184→  return ((highest - data[data.length - 1].close) / range) * -100;
   185→}

   187→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   188→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   189→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   190→  const emaFast = emaCalc(closes, fast);
   191→  const emaSlow = emaCalc(closes, slow);
   192→  const macdLine = emaFast - emaSlow;
   193→  // Calculate signal as EMA of MACD values
   194→  const macdValues: number[] = [];
   195→  const kFast = 2 / (fast + 1);
   196→  const kSlow = 2 / (slow + 1);
   197→  let eFast = sma(closes.slice(0, fast), fast);
   198→  let eSlow = sma(closes.slice(0, slow), slow);
   199→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   200→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   201→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   202→    macdValues.push(eFast - eSlow);
   203→  }
   204→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   205→  const kSig = 2 / (sig + 1);
   206→  let signalVal = sma(macdValues.slice(0, sig), sig);
   207→  for (let i = sig; i < macdValues.length; i++) {
   208→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   209→  }
   210→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   211→}

   213→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   214→function calcATR(data: OHLCV[], period: number = 14): number {
   215→  if (data.length < 2) return 0;
   216→  const trs: number[] = [];
   217→  for (let i = 1; i < data.length; i++) {
   218→    const tr = Math.max(
   219→      data[i].high - data[i].low,
   220→      Math.abs(data[i].high - data[i - 1].close),
   221→      Math.abs(data[i].low - data[i - 1].close)
   222→    );
   223→    trs.push(tr);
   224→  }
   225→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   226→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   227→  for (let i = period; i < trs.length; i++) {
   228→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   229→  }
   230→  return atrVal;
   231→}

   233→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   234→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   235→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   236→  const plusDMs: number[] = [];
   237→  const minusDMs: number[] = [];
   238→  const trs: number[] = [];
   239→  for (let i = 1; i < data.length; i++) {
   240→    const upMove = data[i].high - data[i - 1].high;
   241→    const downMove = data[i - 1].low - data[i].low;
   242→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   243→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   244→    trs.push(Math.max(
   245→      data[i].high - data[i].low,
   246→      Math.abs(data[i].high - data[i - 1].close),
   247→      Math.abs(data[i].low - data[i - 1].close)
   248→    ));
   249→  }
   250→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   251→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   252→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   253→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   254→  const dxValues: number[] = [];
   255→  for (let i = period; i < trs.length; i++) {
   256→    smoothTR = smoothTR - smoothTR / period + trs[i];
   257→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   258→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   259→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   260→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   261→    const diSum = diPlus + diMinus;
   262→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   263→    dxValues.push(dx);
   264→  }
   265→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   266→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   267→  for (let i = period; i < dxValues.length; i++) {
   268→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   269→  }
   270→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   271→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   272→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   273→}

   275→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   276→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   277→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   278→  let isLong = true;
   279→  let af = step;
   280→  let ep = data[0].high;
   281→  let sar = data[0].low;
   282→  for (let i = 1; i < data.length; i++) {
   283→    const prevSar = sar;
   284→    sar = prevSar + af * (ep - prevSar);
   285→    // Ensure SAR is within bounds
   286→    if (isLong) {
   287→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   288→      if (data[i].low < sar) {
   289→        isLong = false;
   290→        sar = ep;
   291→        ep = data[i].low;
   292→        af = step;
   293→      } else {
   294→        if (data[i].high > ep) {
   295→          ep = data[i].high;
   296→          af = Math.min(af + step, max);
   297→        }
   298→      }
   299→    } else {
   300→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   301→      if (data[i].high > sar) {
   302→        isLong = true;
   303→        sar = ep;
   304→        ep = data[i].high;
   305→        af = step;
   306→      } else {
   307→        if (data[i].low < ep) {
   308→          ep = data[i].low;
   309→          af = Math.min(af + step, max);
   310→        }
   311→      }
   312→    }
   313→  }
   314→  return sar;
   315→}

   317→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   318→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   319→  if (closes.length < period) {
   320→    const c = closes[closes.length - 1] ?? 0;
   321→    return { upper: c, middle: c, lower: c };
   322→  }
   323→  const slice = closes.slice(-period);
   324→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   325→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   326→  const sd = Math.sqrt(variance);
   327→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   328→}

   330→// ─── Helper: OBV ──────────────────────────────────────────────────────────────
   331→function calcOBV(data: OHLCV[]): number {
   332→  let obv = 0;
   333→  for (let i = 1; i < data.length; i++) {
   334→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   335→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   336→  }
   337→  return obv;
   338→}

   340→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   341→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   342→  const n = values.length;
   343→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   344→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   345→  for (let i = 0; i < n; i++) {
   346→    sumX += i;
   347→    sumY += values[i];
   348→    sumXY += i * values[i];
   349→    sumX2 += i * i;
   350→    sumY2 += values[i] * values[i];
   351→  }
   352→  const denom = n * sumX2 - sumX * sumX;
   353→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   354→  const slope = (n * sumXY - sumX * sumY) / denom;
   355→  const intercept = (sumY - slope * sumX) / n;
   356→  const meanY = sumY / n;
   357→  let ssRes = 0, ssTot = 0;
   358→  for (let i = 0; i < n; i++) {
   359→    const predicted = slope * i + intercept;
   360→    ssRes += (values[i] - predicted) ** 2;
   361→    ssTot += (values[i] - meanY) ** 2;
   362→  }
   363→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   364→  return { slope, intercept, r2 };
   365→}

   367→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   368→function roundToNice(price: number): number {
   369→  const abs = Math.abs(price);
   370→  let step: number;
   371→  if (abs >= 100000) step = 100;
   372→  else if (abs >= 10000) step = 50;
   373→  else if (abs >= 1000) step = 10;
   374→  else if (abs >= 100) step = 5;
   375→  else step = 1;
   376→  return Math.round(price / step) * step;
   377→}

   379→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
   380→function findSwingLevels(data: OHLCV[]): { highs: number[]; lows: number[] } {
   381→  const highs: number[] = [];
   382→  const lows: number[] = [];
   383→  const lookback = 3;
   384→  for (let i = lookback; i < data.length - lookback; i++) {
   385→    let isHigh = true;
   386→    let isLow = true;
   387→    for (let j = i - lookback; j <= i + lookback; j++) {
   388→      if (j === i) continue;
   389→      if (data[j].high >= data[i].high) isHigh = false;
   390→      if (data[j].low <= data[i].low) isLow = false;
   391→    }
   392→    if (isHigh) highs.push(data[i].high);
   393→    if (isLow) lows.push(data[i].low);
   394→  }
   395→  return { highs, lows };
   396→}

   398→// ─── Support / Resistance (5 each, rounded) ──────────────────────────────────
   399→function calcSupportResistance(data: OHLCV[], currentPrice: number): { resistances: number[]; supports: number[] } {
   400→  const allLevels: number[] = [];
   401→
   402→  // 1. Pivot Points from previous day
   403→  const prev = data[data.length - 2] ?? data[data.length - 1];
   404→  if (prev) {
   405→    const pp = (prev.high + prev.low + prev.close) / 3;
   406→    const hl = prev.high - prev.low;
   407→    const pivotR = [
   408→      2 * pp - prev.low,
   409→      pp + hl,
   410→      prev.high + 2 * (pp - prev.low),
   411→      prev.high + 2 * (pp - prev.low) + hl,
   412→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   413→    ];
   414→    const pivotS = [
   415→      2 * pp - prev.high,
   416→      pp - hl,
   417→      prev.low - 2 * (prev.high - pp),
   418→      prev.low - 2 * (prev.high - pp) - hl,
   419→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   420→    ];
   421→    allLevels.push(...pivotR, ...pivotS);
   422→  }
   423→
   424→  // 2. Swing levels
   425→  const swings = findSwingLevels(data);
   426→  allLevels.push(...swings.highs, ...swings.lows);
   427→
   428→  // 3. Round all
   429→  const rounded = allLevels.map(roundToNice);
   430→
   431→  // 4. Remove duplicates (within 0.5%)
   432→  const unique: number[] = [];
   433→  for (const level of rounded) {
   434→    if (level <= 0) continue;
   435→    const isDup = unique.some(u => Math.abs(u - level) / u < 0.005);
   436→    if (!isDup) unique.push(level);
   437→  }
   438→
   439→  // 5. Separate and sort
   440→  const above = unique.filter(l => l > currentPrice).sort((a, b) => a - b);
   441→  const below = unique.filter(l => l < currentPrice).sort((a, b) => b - a);
   442→
   443→  return {
   444→    resistances: above.slice(0, 5),
   445→    supports: below.slice(0, 5),
   446→  };
   447→}

   449→// ─── Trend Analysis ───────────────────────────────────────────────────────────
   450→function calcTrend(closes: number[], period: number): TrendResult {
   451→  const slice = closes.slice(-Math.min(period, closes.length));
   452→  const { slope, r2 } = linearRegression(slice);
   453→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   454→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   455→  const priceThreshold = avgPrice * 0.001; // 0.1%
   456→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   457→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   458→}

   463→export function analyze(data: OHLCV[]): TAResult {
   464→  if (!data || data.length < 2) {
   465→    const empty = () => 0;
   466→    const emptyScenario = (): ScenarioResult => ({ name: '', nameEn: '', probability: 20, targetMin: 0, targetMax: 0, description: '' });
   467→    return {
   468→      sma: {}, ema: {}, rsi: 50, mfi: 50, cci: 0, stochK: 50, stochD: 50,

   505→  const adxResult = calcADX(data);

   519→  const trend = {
   520→    short: calcTrend(closes, 21),

   528→

   569→    ? (data.slice(-obvSlice).reduce((s, d, _, arr) => s + (d.close > arr[Math.max(0, arr.length - obvSlice - 1)]?.close ? 1 : 0), 0) / obvSlice) : 0.5;
   570→  signals.push(obvTrend > 0.6 ? 1 : obvTrend > 0.4 ? 0.5 : 0);

   578→  signals.push((trend.long.direction === 'up' ? 1 : trend.long.direction === 'down' ? 0 : 0.5) * trend.long.r2);

   586→  const dR1 = Math.exp(-5 * Math.abs(price - R1_level) / R1_level);

   607→  const pR3 = sum === 0 ? 20 : Math.round(raw_R3 / sum * 100);
   608→  const pR4 = sum === 0 ? 20 : Math.round(raw_R4 / sum * 100);

   610→
