   112→   112→// ═══════════════════════════════════════════════════════════════════════════════
   113→   113→
   114→   114→export default function VdssGraph(props: VdssGraphProps) {
   115→   115→  const {
   116→   116→    symbolName,
   117→   117→    currentPrice,
   118→   118→    scenarios,
   119→   119→    decisionGraph,
   120→   120→  } = props;
   121→   121→
   122→   122→  const [activeFilter, setActiveFilter] = useState('all');
   123→   123→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   124→   124→  const graphRef = useRef<HTMLDivElement>(null);
   125→   125→  const svgRef = useRef<SVGSVGElement>(null);
   126→   126→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   127→   127→  const shellRef = useRef<HTMLDivElement>(null);
   128→   128→
   129→   129→  // ── Derived data from decisionGraph ──────────────────────────────
   130→   130→  const edges = decisionGraph?.edges ?? [];
   131→   131→  const nodes = decisionGraph?.nodes ?? [];
   132→   132→  const nodePositions = decisionGraph?.nodePositions ?? {};
   133→   133→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   134→   134→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   135→   135→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   136→   136→  const pathContributions = decisionGraph?.pathContributions ?? {};
   137→   137→  const nodeValues = decisionGraph?.nodeValues ?? {};
   138→   138→
   139→   139→  // Build node lookup map
   140→   140→  const nodeMap = useMemo(() => {
   141→   141→    const map: Record<string, (typeof nodes)[number]> = {};
   142→   142→    for (const n of nodes) map[n.id] = n;
   143→   143→    return map;
   144→   144→  }, [nodes]);
   145→   145→
   146→   146→  // ── Visible edges/nodes based on filter ──────────────────────────
   147→   147→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   148→   148→    if (activeFilter === 'all') {
   149→   149→      const allEdgeIdx = edges.map((_, i) => i);
   150→   150→      const allNodes = new Set(nodes.map(n => n.id));
   151→   151→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   152→   152→    }
   153→   153→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   154→   154→      // Show edges going TO this scenario node, plus their source nodes
   155→   155→      const edgeSet = new Set<number>();
   156→   156→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   157→   157→      edges.forEach((e, i) => {
   158→   158→        if (e.to === activeFilter) {
   159→   159→          edgeSet.add(i);
   160→   160→          nodeSet.add(e.from);
   161→   161→        }
   162→   162→      });
   163→   163→      // Also show intermediate nodes connected to those sources
   164→   164→      edges.forEach((e, i) => {
   165→   165→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   166→   166→          edgeSet.add(i);
   167→   167→          nodeSet.add(e.from);
   168→   168→        }
   169→   169→      });
   170→   170→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   171→   171→    }
   172→   172→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   173→   173→      const branchType = `branch-${activeFilter}` as const;
   174→   174→      const edgeSet = new Set<number>();
   175→   175→      const nodeSet = new Set<string>(['ROOT']);
   176→   176→      edges.forEach((e, i) => {
   177→   177→        if (e.type === branchType || (e.type !== `branch-trend` && e.type !== `branch-breakout` && e.type !== `branch-reversal` && nodeSet.has(e.from))) {
   178→   178→          // Include branch-type edges
   179→   179→          if (e.type === branchType) {
   180→   180→            edgeSet.add(i);
   181→   181→            nodeSet.add(e.from);
   182→   182→            nodeSet.add(e.to);
   183→   183→          }
   184→   184→        }
   185→   185→      });
   186→   186→      // Now add all edges whose source is already in nodeSet
   187→   187→      const expanded = new Set<number>();
   188→   188→      const expandedNodes = new Set<string>(nodeSet);
   189→   189→      edges.forEach((e, i) => {
   190→   190→        if (expandedNodes.has(e.from)) {
   191→   191→          expanded.add(i);
   192→   192→          expandedNodes.add(e.to);
   193→   193→        }
   194→   194→      });
   195→   195→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   196→   196→    }
   197→   197→    // Type filter (up/pullback/down/risk)
   198→   198→    const edgeSet = new Set<number>();
   199→   199→    const nodeSet = new Set<string>(['ROOT']);
   200→   200→    edges.forEach((e, i) => {
   201→   201→      if (e.type === activeFilter) {
   202→   202→        edgeSet.add(i);
   203→   203→        nodeSet.add(e.from);
   204→   204→        nodeSet.add(e.to);
   205→   205→      }
   206→   206→    });
   207→   207→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   208→   208→  }, [activeFilter, edges, nodes]);
   209→   209→
   210→   210→  // ── Draw SVG edges ───────────────────────────────────────────────
   211→   211→  const drawEdges = useCallback(() => {
   212→   212→    const svg = svgRef.current;
   213→   213→    const graph = graphRef.current;
   214→   214→    if (!svg || !graph || edges.length === 0) return;
   215→   215→
   216→   216→    let markersSvg = '';
   217→   217→    const seenColors = new Set<string>();
   218→   218→    for (const e of edges) {
   219→   219→      const c = EDGE_COLORS[e.type];
   220→   220→      if (c && !seenColors.has(e.type)) {
   221→   221→        seenColors.add(e.type);
   222→   222→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="8" markerHeight="8" refX="7" refY="4" orient="auto"><path d="M0,0 L8,4 L0,8 z" fill="${c}"/></marker>`;
   223→   223→      }
   224→   224→    }
   225→   225→
   226→   226→    let pathsSvg = '';
   227→   227→    const graphRect = graph.getBoundingClientRect();
   228→   228→
   229→   229→    for (let ei = 0; ei < edges.length; ei++) {
   230→   230→      const { from: fromId, to: toId, label, type } = edges[ei];
   231→   231→      const isVisible = visibleEdgeIndices.includes(ei);
   232→   232→      const fromEl = nodeRefs.current[fromId];
   233→   233→      const toEl = nodeRefs.current[toId];
   234→   234→      if (!fromEl || !toEl) continue;
   235→   235→
   236→   236→      const fRect = fromEl.getBoundingClientRect();
   237→   237→      const tRect = toEl.getBoundingClientRect();
   238→   238→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   239→   239→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   240→   240→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   241→   241→      const by = tRect.top - graphRect.top + tRect.height / 2;
   242→   242→      const dx = bx - ax;
   243→   243→      const dy = by - ay;
   244→   244→      const dist = Math.sqrt(dx * dx + dy * dy);
   245→   245→      if (dist === 0) continue;
   246→   246→
   247→   247→      // Curved edges with bend proportional to distance
   248→   248→      const bend = Math.min(40, Math.max(12, dist * 0.08));
   249→   249→      const mx = (ax + bx) / 2;
   250→   250→      const my = (ay + by) / 2;
   251→   251→      const nx = -dy / dist;
   252→   252→      const ny = dx / dist;
   253→   253→      const cx = mx + nx * bend;
   254→   254→      const cy = my + ny * bend;
   255→   255→
   256→   256→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   257→   257→      const prob = edgeProbabilities[ei] ?? 0;
   258→   258→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   259→   259→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   260→   260→
   261→   261→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.7 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   262→   262→
   263→   263→      // Edge labels — skip deterministic root→branch and branch→event edges (index < 6)
   264→   264→      if (isVisible && ei >= 6) {
   265→   265→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#374151" font-size="8" text-anchor="middle" paint-order="stroke" stroke="#ffffff" stroke-width="3" stroke-linejoin="round" opacity="0.8" data-type="${type}" class="edge-label">${label}</text>`;
   266→   266→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#ffffff" stroke-width="2.5" stroke-linejoin="round" opacity="0.85" class="edge-prob">${probLabel}</text>`;
   267→   267→      }
   268→   268→    }
   269→   269→
   270→   270→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   271→   271→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   272→   272→
   273→   273→  useEffect(() => { drawEdges(); }, [drawEdges]);
   274→   274→
   275→   275→  useEffect(() => {
   276→   276→    const el = shellRef.current;
   277→   277→    if (!el) return;
   278→   278→    const observer = new ResizeObserver(() => drawEdges());
   279→   279→    observer.observe(el);
   280→   280→    return () => observer.disconnect();
   281→   281→  }, [drawEdges]);
   282→   282→
   283→   283→
   284→   284→  // ── Detail panel content ─────────────────────────────────────────
   285→   285→  const detailContent = useMemo(() => {
   286→   286→    if (!selectedNode) return null;
   287→   287→    const node = nodeMap[selectedNode];
   288→   288→    if (!node) return null;
   289→   289→
   290→   290→    // For terminal scenario nodes, show branch breakdown
   291→   291→    if (node.isTerminal && SCENARIO_KEYS.includes(selectedNode as typeof SCENARIO_KEYS[number])) {
   292→   292→      const contrib = pathContributions[selectedNode] ?? { trend: 0, breakout: 0, reversal: 0 };
   293→   293→      const meta = SCENARIO_META[selectedNode];
   294→   294→      const s = scenarios[selectedNode as keyof typeof scenarios];
   295→   295→      return (
   296→   296→        <div className="space-y-3">
   297→   297→          <h3 className="text-sm font-bold" style={{ color: meta.color }}>{node.title}</h3>
   298→   298→          <p className="text-xs text-[#374151]">{node.desc}</p>
   299→   299→          <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border border-[#e5e7eb] bg-[#f3f4f6] text-[#374151]">{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   300→   300→          <div className="text-xs text-[#6b7280]" dir="ltr">{nodeValues[selectedNode] ?? '--'}</div>
   301→   301→
   302→   302→          <div className="mt-3 pt-3 border-t border-[#e5e7eb]">
   303→   303→            <p className="text-xs font-bold text-[#374151] mb-2">سهم هر استراتژی:</p>
   304→   304→            {Object.entries(BRANCH_META).map(([bKey, bMeta]) => {
   305→   305→              const val = contrib[bKey as 'trend' | 'breakout' | 'reversal'];
   306→   306→              const pct = (val * 100).toFixed(1);
   307→   307→              return (
   308→   308→                <div key={bKey} className="flex items-center justify-between py-1.5 border-b border-dashed border-[#e5e7eb]">
   309→   309→                  <div className="flex items-center gap-1.5">
   310→   310→                    <span className="inline-block w-2 h-2 rounded-full" style={{ background: bMeta.color }} />
   311→   311→                    <span className="text-[11px] text-[#374151]">{bMeta.label}</span>
   312→   312→                  </div>
   313→   313→                  <span className="text-[11px] font-bold" style={{ color: bMeta.color }}>{toPersianDigits(pct)}٪</span>
   314→   314→                </div>
   315→   315→              );
   316→   316→            })}
   317→   317→            <div className="flex items-center justify-between py-1.5">
   318→   318→              <span className="text-[11px] font-bold text-[#111827]">مجموع</span>
   319→   319→              <span className="text-[11px] font-black" style={{ color: meta.color }}>{toFa(s?.probability ?? 0)}٪</span>
   320→   320→            </div>
   321→   321→          </div>
   322→   322→        </div>
   323→   323→      );
   324→   324→    }
   325→   325→
   326→   326→    // For non-terminal nodes, show incoming/outgoing edges
   327→   327→    const inputs = edges.filter(e => e.to === selectedNode);
   328→   328→    const outputs = edges.filter(e => e.from === selectedNode);
   329→   329→
   330→   330→    return (
   331→   331→      <div className="space-y-3">
   332→   332→        <h3 className="text-sm font-bold text-[#111827]">{node.title}</h3>
   333→   333→        <p className="text-xs text-[#374151]"><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   334→   334→        <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border border-[#e5e7eb] bg-[#f3f4f6] text-[#374151]">{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   335→   335→        <p className="text-xs text-[#6b7280] leading-relaxed">{node.desc}</p>
   336→   336→        {inputs.length > 0 && (
   337→   337→          <div>
   338→   338→            <p className="text-xs font-medium text-[#374151] mb-1">مسیرهای ورودی ({toFa(inputs.length)}):</p>
   339→   339→            <ul className="space-y-1">
   340→   340→              {inputs.map((e, i) => {
   341→   341→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   342→   342→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   343→   343→                return (
   344→   344→                  <li key={i} className="text-[11px] text-[#6b7280] leading-relaxed border-t border-dashed border-[#e5e7eb] pt-1.5">
   345→   345→                    <b className="text-[#374151]">{SCENARIO_DISPLAY[e.from] || e.from} ← {SCENARIO_DISPLAY[e.to] || e.to}</b>
   346→   346→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${edgeColor}20`, color: edgeColor }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   347→   347→                    <br />{e.label}
   348→   348→                  </li>
   349→   349→                );
   350→   350→              })}
   351→   351→            </ul>
   352→   352→          </div>
   353→   353→        )}
   354→   354→        {outputs.length > 0 && (
   355→   355→          <div>
   356→   356→            <p className="text-xs font-medium text-[#374151] mb-1">مسیرهای خروجی ({toFa(outputs.length)}):</p>
   357→   357→            <ul className="space-y-1">
   358→   358→              {outputs.map((e, i) => {
   359→   359→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   360→   360→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   361→   361→                return (
   362→   362→                  <li key={i} className="text-[11px] text-[#6b7280] leading-relaxed border-t border-dashed border-[#e5e7eb] pt-1.5">
   363→   363→                    <b className="text-[#374151]">{SCENARIO_DISPLAY[e.from] || e.from} → {SCENARIO_DISPLAY[e.to] || e.to}</b>
   364→   364→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${edgeColor}20`, color: edgeColor }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   365→   365→                    <br />{e.label}
   366→   366→                  </li>
   367→   367→                );
   368→   368→              })}
   369→   369→            </ul>
   370→   370→          </div>
   371→   371→        )}
   372→   372→      </div>
   373→   373→    );
   374→   374→  }, [selectedNode, nodeMap, nodeValues, edges, edgeProbabilities, pathContributions, scenarios]);
