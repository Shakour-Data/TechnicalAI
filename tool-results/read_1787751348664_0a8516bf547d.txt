   255→  // Store last fetch params for auto-refresh
   256→  const lastFetchRef = useRef<{ symbol: string; category?: string; insCode?: string; tgjuKey?: string; finpySector?: string; finpyIndex?: string; webId?: number; yahooSymbol?: string } | null>(null);
   257→  const refreshTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   258→  const fetchControllerRef = useRef<AbortController | null>(null);
   259→
   260→  // Chart container ref for image export
   261→  const chartWrapperRef = useRef<HTMLDivElement>(null);
   262→
   263→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number, yahooSymbol?: string) => {
   264→    // Abort any previous in-flight fetch
   265→    if (fetchControllerRef.current) {
   266→      fetchControllerRef.current.abort();
   267→    }
   268→    const controller = new AbortController();
   269→    fetchControllerRef.current = controller;
   270→
   271→    // Yahoo Finance stock: fetch historical data via Yahoo Finance API
   272→    if (category === 'yahoo_stock' && yahooSymbol) {
   273→      setLoading(true);
   274→      setLoadingMessage('در حال دریافت داده‌های تاریخی از یاهو فایننس ...');
   275→      setError(null);
   276→      setData(null);
   277→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId, yahooSymbol };
   278→      try {
   279→        const res = await fetch('/api/yahoo-analysis?symbol=' + encodeURIComponent(yahooSymbol), { signal: controller.signal });
   280→        if (controller.signal.aborted) return;
   281→        if (!res.ok) {
   282→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   283→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   284→          throw new Error(errMsg);
   285→        }
   286→        const json = await res.json();
   287→        if (controller.signal.aborted) return;
   288→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   289→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   290→        }
   291→        setData({ ...json, isYahoo: true });
   292→        setActivePanel('visual');
   293→      } catch (err) {
   294→        if (controller.signal.aborted) return;
   295→        setError(String(err));
   296→      } finally {
   297→        if (!controller.signal.aborted) {
   298→          setLoading(false);
   299→          setLoadingMessage(null);
   300→        }
   301→      }
   302→      return;
   303→    }
   304→
   305→    // TGJU instrument: fetch historical data via tgju.org chart API
   306→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   307→      setLoading(true);
   308→      setLoadingMessage('در حال دریافت داده‌های تاریخی ... (حدود ۱۵ ثانیه)');
   309→      setError(null);
   310→      setData(null);
   311→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId, yahooSymbol };
   312→      try {
   313→        const res = await fetch('/api/tgju-analysis?key=' + encodeURIComponent(tgjuKey), { signal: controller.signal });
   314→        if (controller.signal.aborted) return;
   315→        if (!res.ok) {
   316→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   317→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
