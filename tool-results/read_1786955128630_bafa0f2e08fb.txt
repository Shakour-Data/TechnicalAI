    87→// ═══════════════════════════════════════════════════════════════════════════════
    88→
    89→export default function VdesAnalysis(props: VdesAnalysisProps) {
    90→  const {
    91→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
    92→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
    93→    diPlus, diMinus, sar, atr, obv, bollingerUpper, bollingerMiddle, bollingerLower,
    94→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
    95→    supportStrengths, resistanceStrengths, priceTargets,
    96→  } = props;
    97→
    98→  const R1_level = resistances[0] ?? currentPrice * 1.05;
    99→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   100→  const S1_level = supports[0] ?? currentPrice * 0.95;
   101→  const S2_level = supports[1] ?? currentPrice * 0.90;
   102→  const S3_level = supports[2] ?? currentPrice * 0.85;
   103→
   104→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   105→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   106→
   107→  // ── Dominant scenario ──────────────────────────────────────────
   108→  let highestKey = 'R3';
   109→  let highestProb = 0;
   110→  for (const key of SCENARIO_KEYS) {
   111→    if (scenarios[key].probability > highestProb) {
   112→      highestProb = scenarios[key].probability;
   113→      highestKey = key;
   114→    }
   115→  }
   116→  const strategy = STRATEGY_MAP[highestKey];
   117→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   118→
   119→  // ── RSI signal ─────────────────────────────────────────────────
   120→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   121→  const rsiColor = rsi > 70 ? 'text-red-400' : rsi > 60 ? 'text-amber-400' : rsi > 40 ? 'text-gray-300' : rsi > 30 ? 'text-amber-400' : 'text-emerald-400';
   122→
   123→  // ── Stochastic signal ──────────────────────────────────────────
   124→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   125→
   126→  // ── MACD signal ────────────────────────────────────────────────
   127→  const macdSignalText = macdHist > 0 ? 'مومنتوم مثبت (هیستوگرام بالای صفر)' : 'مومنتوم منفی (هیستوگرام زیر صفر)';
   128→  const macdBullish = macdLine > macdSignal;
   129→
   130→  // ── Trend text ─────────────────────────────────────────────────
   131→  const trendText = trendDirection === 'up'
   132→    ? `صعودی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${(trendR2 * 100).toFixed(1)}%`
   133→    : trendDirection === 'down'
   134→    ? `نزولی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${(trendR2 * 100).toFixed(1)}%`
   135→    : 'خنثی و بدون جهت مشخص';
   136→
   137→  // ── DI signal ──────────────────────────────────────────────────
   138→  const diSignal = diPlus > diMinus
   139→    ? `DI+ (${toFa(diPlus)}) بالاتر از DI- (${toFa(diMinus)}) — فشار خرید غالب`
   140→    : `DI- (${toFa(diMinus)}) بالاتر از DI+ (${toFa(diPlus)}) — فشار فروش غالب`;
   141→
   142→  // ── SAR signal ─────────────────────────────────────────────────
   143→  const sarSignal = sar < currentPrice
   144→    ? `SAR (${toFa(sar)}) زیر قیمت — تأیید روند صعودی`
   145→    : `SAR (${toFa(sar)}) بالای قیمت — تأیید روند نزولی`;
   146→
   147→  // ── Bollinger Band position ────────────────────────────────────
   148→  const bbRange = bollingerUpper - bollingerLower;
   149→  const bbPos = bbRange > 0 ? ((currentPrice - bollingerLower) / bbRange * 100).toFixed(0) : '50';
   150→  const bbSignal = currentPrice > bollingerUpper ? 'بالای باند بالایی (اشباع خرید)'
   151→    : currentPrice < bollingerLower ? 'زیر باند پایینی (اشباع فروش)'
   152→    : `داخل باندها (${bbPos}٪ از بازه)`;
   153→
   154→  // ── TradingView scenario colors ────────────────────────────────
   155→  const tvScenarios = useMemo(() => {
   156→    const colors: Record<string, string> = {
   157→      R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62',
   158→    };
   159→    return Object.fromEntries(
   160→      SCENARIO_KEYS.map(k => [k, {
   161→        targetMin: scenarios[k].targetMin,
   162→        targetMax: scenarios[k].targetMax,
   163→        name: scenarios[k].name,
   164→        probability: scenarios[k].probability,
   165→        color: colors[k],
   166→      }])
   167→    );
   168→  }, [scenarios]);
   169→
   170→  // ── Strategy recommendation text ────────────────────────────────
   171→  const strategyText = rsi > 70 || mfi > 80
   172→    ? `با توجه به هشدار اشباع خرید (RSI: ${toFa(rsi)}, MFI: ${toFa(mfi)}) و فاصله قیمت تا مقاومت ${toFa(R1_level)}، استراتژی محتاطانه، انتظار برای اصلاح قیمت و ورود در محدوده حمایت ${toFa(S1_level)} تا ${toFa(S2_level)} ریال می‌باشد. در این محدوده می‌توان با حد ضرر ${toFa(S2_level)} ریال وارد موقعیت خرید شد.`
   173→    : rsi < 30 || mfi < 20
   174→    ? `با توجه به اشباع فروش (RSI: ${toFa(rsi)}, MFI: ${toFa(mfi)}) و نزدیکی به حمایت ${toFa(S1_level)}، فرصت خرید در محدوده فعلی با حد ضرر ${toFa(S2_level)} ریال قابل بررسی است. هدف اولیه ${toFa(R1_level)} و هدف ثانویه ${toFa(R2_level)} ریال تعیین می‌شود.`
   175→    : `با توجه به وضعیت خنثی اندیکاتورها (RSI: ${toFa(rsi)}, ADX: ${toFa(adx)}، قدرت روند: ${adx > 25 ? 'قوی' : 'ضعیف'})، انتظار برای خروج قیمت از محدوده ${toFa(S1_level)} تا ${toFa(R1_level)} ریال و سپس تصمیم‌گیری توصیه می‌شود. حجم معاملات و مومنتوم MACD را برای تأیید سیگنال پایش کنید.`;
   176→
   177→  return (
   178→    <div className="space-y-5" dir="rtl">
   179→      {/* ═══ HEADER ═══ */}
   180→      <div className="rounded-[32px] px-6 py-5 border border-amber-500/20"
   181→        style={{
   182→          background: 'rgba(18,28,46,0.7)',
   183→          backdropFilter: 'blur(14px)',
   184→          boxShadow: '0 25px 50px -12px rgba(0,0,0,0.6)',
   185→        }}>
   186→        <h1 className="text-xl font-bold mb-1" style={{ color: '#f8e365' }}>
   187→          📈 تحلیل تکنیکال {symbolName}
   188→        </h1>
   189→        <div className="flex flex-wrap gap-3 mt-2">
   190→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   191→            📍 قیمت مرجع: <b className="text-gray-200">{toFa(currentPrice)} ریال</b>
   192→          </span>
   193→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   194→            🎯 هدف کوتاه‌مدت: <b className="text-gray-200">{toFa(targetMin)} — {toFa(targetMax)} ریال</b>
   195→          </span>
   196→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   197→            📊 روند: <b className="text-gray-200">{trendText}</b>
   198→          </span>
   199→          <span className={`px-3.5 py-1 rounded-full border text-xs font-medium ${
   200→            rsi > 70 ? 'bg-red-500/15 text-red-400 border-red-500/30'
   201→            : rsi < 30 ? 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30'
   202→            : 'bg-amber-500/15 text-amber-400 border-amber-500/30'
   203→          }`}>
   204→            RSI: {toFa(rsi)} ({rsiSignal})
   205→          </span>
   206→        </div>
   207→      </div>
   208→
   209→      {/* ═══ TRADINGVIEW CHART ═══ */}
   210→      <div className="rounded-[28px] p-4 border border-white/6"
   211→        style={{
   212→          background: 'rgba(14,22,40,0.7)',
   213→          backdropFilter: 'blur(8px)',
   214→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   215→        }}>
   216→        <h2 className="text-sm font-semibold mb-3 text-gray-200 flex items-center gap-3">
   217→          <span style={{ color: '#f8e365' }}>📊</span>
   218→          نمودار قیمتی سهم (۱ روزه) — به همراه MA21، MA100، سطوح حمایت/مقاومت و اهداف قیمتی
   219→        </h2>
   220→        <TradingViewChart
   221→          symbolName={symbolName}
   222→          candles={candles}
   223→          supports={supports}
   224→          resistances={resistances}
   225→          supportStrengths={supportStrengths}
   226→          resistanceStrengths={resistanceStrengths}
   227→          ma21={ma21}
   228→          ma100={ma100}
   229→          scenarios={tvScenarios}
   230→        />
   231→        <div className="mt-3 flex flex-wrap gap-4 text-[10px] text-gray-500">
   232→          <span>🔴 خطوط قرمز: مقاومت‌ها ({resistances.map(toFa).join(' ، ')})</span>
   233→          <span>🟢 خطوط سبز: حمایت‌ها ({supports.map(toFa).join(' ، ')})</span>
   234→          <span>🟣 خط بنفش: MA100 ({toFa(ma100)})</span>
   235→          <span>🔵 خط آبی: MA21 ({toFa(ma21)})</span>
   236→          <span>🟡 خطوط زرد: اهداف قیمتی سناریوها (رند شده)</span>
   237→        </div>
   238→      </div>
   239→
   240→      {/* ═══ DUAL ANALYSIS LAYOUT ═══ */}
   241→      <div className="rounded-[28px] p-5 border border-white/6"
   242→        style={{
   243→          background: 'rgba(14,22,40,0.7)',
   244→          backdropFilter: 'blur(8px)',
   245→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   246→        }}>
   247→        <h2 className="text-sm font-semibold mb-4 text-gray-200 flex items-center gap-3">
   248→          <span style={{ color: '#f8e365' }}>🧠</span>
   249→          تحلیل جامع روند و سطوح کلیدی
   250→        </h2>
   251→
   252→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-5">
   253→          {/* ── Text Analysis (Right in RTL) ── */}
   254→          <div className="rounded-2xl p-4 border border-white/4 space-y-4 flex flex-col"
   255→            style={{ background: 'rgba(8,20,35,0.5)' }}>
   256→
   257→            {/* ── PARAGRAPH 1: General Trend & Price Position ── */}
   258→            <p className="text-sm text-gray-300 leading-[1.85]">
   259→              <strong style={{ color: '#f8e365' }}>روند کلی و موقعیت قیمت:</strong>{' '}
   260→              سهم {symbolName} در حال حاضر در محدوده{' '}
   261→              <b className="text-gray-100">{toFa(currentPrice)} ریال</b> معامله می‌شود و روند میان‌مدت{' '}
   262→              <b className={trendDirection === 'up' ? 'text-emerald-400' : trendDirection === 'down' ? 'text-red-400' : 'text-amber-400'}>{trendDirection === 'up' ? 'صعودی' : trendDirection === 'down' ? 'نزولی' : 'خنثی'}</b>
   263→              {' '}است (زاویه {toFa(Math.abs(trendAngle))}°، R²={+(trendR2 * 100).toFixed(1)}%).
   264→              قیمت نسبت به MA21 ({toFa(ma21)} ریال){' '}
   265→              {currentPrice > ma21 ? <span className="text-emerald-400">بالاتر</span> : <span className="text-red-400">پایین‌تر</span>}
   266→              {' '}و نسبت به MA100 ({toFa(ma100)} ریال){' '}
   267→              {currentPrice > ma100 ? <span className="text-emerald-400">بالاتر</span> : <span className="text-red-400">پایین‌تر</span>}
   268→              {' '}قرار دارد. اندیکاتور Parabolic SAR ({toFa(sar)}) نیز{' '}
   269→              {sar < currentPrice ? <span>زیر قیمت قرار دارد که <b className="text-emerald-400">تأیید روند صعودی</b> است.</span> : <span>بالای قیمت قرار دارد که <b className="text-red-400">تأیید روند نزولی</b> است.</span>}
   270→              {' '}شاخص ADX ({toFa(adx)}) نشان‌دهنده {' '}
   271→              {adx > 40 ? <b className="text-emerald-400">روند قدرتمند</b> : adx > 25 ? <b className="text-amber-400">روند متوسط</b> : <b className="text-gray-500">روند ضعیف یا رنج</b>}
   272→              {' '}می‌باشد. {' '}{diSignal}.
   273→            </p>
   274→
   275→            {/* ── PARAGRAPH 2: Oscillator & Momentum Analysis ── */}
   276→            <p className="text-sm text-gray-300 leading-[1.85]">
   277→              <strong style={{ color: '#f8e365' }}>تحلیل اسیلاتورها و مومنتوم:</strong>{' '}
   278→              اندیکاتور RSI ({toFa(rsi)}) در ناحیه{' '}
   279→              <b className={rsiColor}>{rsiSignal}</b> قرار دارد.
   280→              اندیکاتور MFI ({toFa(mfi)}) نیز{' '}
   281→              {mfi > 80 ? <span className="text-red-400">اشباع خرید را تأیید می‌کند</span> : mfi < 20 ? <span className="text-emerald-400">اشباع فروش را نشان می‌دهد</span> : <span>در محدوده عادی است</span>}
   282→              . اندیکاتور CCI ({toFa(cci)}){' '}
   283→              {cci > 100 ? <span className="text-red-400">بالاتر از +100 (قدرت خریداری قوی)</span> : cci < -100 ? <span className="text-emerald-400">پایین‌تر از -100 (قدرت فروشندگان)</span> : <span>در محدوده عادی (-100 تا +100)</span>}
   284→              . استوکاستیک (%K={toFa(stochK)}، %D={toFa(stochD)}) وضعیت{' '}
   285→              <b className={stochK > 80 ? 'text-red-400' : stochK < 20 ? 'text-emerald-400' : 'text-gray-300'}>{stochSignal}</b> را نشان می‌دهد.
   286→              {' '}MACD (خط={toFa(macdLine)}، سیگنال={toFa(macdSignal)}) با {macdBullish ? <span className="text-emerald-400">عبور خط اصلی بالای خط سیگنال — سیگنال صعودی</span> : <span className="text-red-400">خط اصلی زیر خط سیگنال — سیگنال نزولی</span>}
   287→              . هیستوگرام MACD ({toFa(macdHist)}) {' '}
   288→              {macdHist > 0 ? <span className="text-emerald-400">مثبت</span> : <span className="text-red-400">منفی</span>}
   289→              {' '}و {' '}{macdSignalText}.
   290→            </p>
   291→
   292→            {/* ── PARAGRAPH 3: Bollinger Bands & Volatility ── */}
   293→            <p className="text-sm text-gray-300 leading-[1.85]">
   294→              <strong style={{ color: '#f8e365' }}>تحلیل نوسانات و باند بولینگر:</strong>{' '}
   295→              قیمت در باند بولینگر{' '}
   296→              <b className="text-gray-200">{bbSignal}</b>
   297→              {' '}قرار دارد. باند بالایی: {toFa(bollingerUpper)}، باند میانی (MA20): {toFa(bollingerMiddle)}، باند پایینی: {toFa(bollingerLower)} ریال.
   298→              {currentPrice > bollingerUpper
   299→                ? ' عبور از باند بالایی معمولاً نشان‌دهنده ادامه حرکت صعودی کوتاه‌مدت یا واکنش به باند است.'
   300→                : currentPrice < bollingerLower
   301→                ? ' نزدیکی یا عبور از باند پایینی می‌تواند نشانه بازگشت قیمت به سمت باند میانی باشد.'
   302→                : ' موقعیت قیمت در داخل باندها نشان‌دهنده عدم وجود سیگنال شدید از باند بولینگر است.'}
   303→            </p>
   304→

   471→function StrengthBadge({ strength, isTarget }: { strength: number; isTarget: boolean }) {

   477→      {isTarget && '🎯 '}
   478→      قدرت: {strength}/10
   479→    </span>
   480→  );
