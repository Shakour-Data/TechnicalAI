   134→   134→
   135→   135→export default function VdssGraph(props: VdssGraphProps) {
   136→   136→  const { colors: C } = useTheme();
   137→   137→  const {
   138→   138→    symbolName,
   139→   139→    currentPrice,
   140→   140→    scenarios,
   141→   141→    decisionGraph,
   142→   142→  } = props;
   143→   143→
   144→   144→  const [activeFilter, setActiveFilter] = useState('all');
   145→   145→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   146→   146→  const graphRef = useRef<HTMLDivElement>(null);
   147→   147→  const svgRef = useRef<SVGSVGElement>(null);
   148→   148→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   149→   149→  const shellRef = useRef<HTMLDivElement>(null);
   150→   150→
   151→   151→  // ── Derived data from decisionGraph ──────────────────────────────
   152→   152→  const edges = decisionGraph?.edges ?? [];
   153→   153→  const nodes = decisionGraph?.nodes ?? [];
   154→   154→  const nodePositions = decisionGraph?.nodePositions ?? {};
   155→   155→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   156→   156→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   157→   157→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   158→   158→  const pathContributions = decisionGraph?.pathContributions ?? {};
   159→   159→  const nodeValues = decisionGraph?.nodeValues ?? {};
   160→   160→
   161→   161→  // Build node lookup map
   162→   162→  const nodeMap = useMemo(() => {
   163→   163→    const map: Record<string, (typeof nodes)[number]> = {};
   164→   164→    for (const n of nodes) map[n.id] = n;
   165→   165→    return map;
   166→   166→  }, [nodes]);
   167→   167→
   168→   168→  // ── Visible edges/nodes based on filter ──────────────────────────
   169→   169→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   170→   170→    if (activeFilter === 'all') {
   171→   171→      const allEdgeIdx = edges.map((_, i) => i);
   172→   172→      const allNodes = new Set(nodes.map(n => n.id));
   173→   173→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   174→   174→    }
   175→   175→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   176→   176→      const edgeSet = new Set<number>();
   177→   177→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   178→   178→      edges.forEach((e, i) => {
   179→   179→        if (e.to === activeFilter) {
   180→   180→          edgeSet.add(i);
   181→   181→          nodeSet.add(e.from);
   182→   182→        }
   183→   183→      });
   184→   184→      edges.forEach((e, i) => {
   185→   185→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   186→   186→          edgeSet.add(i);
   187→   187→          nodeSet.add(e.from);
   188→   188→        }
   189→   189→      });
   190→   190→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   191→   191→    }
   192→   192→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   193→   193→      const branchType = `branch-${activeFilter}` as const;
   194→   194→      const edgeSet = new Set<number>();
   195→   195→      const nodeSet = new Set<string>(['ROOT']);
   196→   196→      edges.forEach((e, i) => {
   197→   197→        if (e.type === branchType) {
   198→   198→          edgeSet.add(i);
   199→   199→          nodeSet.add(e.from);
   200→   200→          nodeSet.add(e.to);
   201→   201→        }
   202→   202→      });
   203→   203→      const expanded = new Set<number>();
   204→   204→      const expandedNodes = new Set<string>(nodeSet);
   205→   205→      edges.forEach((e, i) => {
   206→   206→        if (expandedNodes.has(e.from)) {
   207→   207→          expanded.add(i);
   208→   208→          expandedNodes.add(e.to);
   209→   209→        }
   210→   210→      });
   211→   211→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   212→   212→    }
   213→   213→    // Type filter (up/pullback/down/risk)
   214→   214→    const edgeSet = new Set<number>();
   215→   215→    const nodeSet = new Set<string>(['ROOT']);
   216→   216→    edges.forEach((e, i) => {
   217→   217→      if (e.type === activeFilter) {
   218→   218→        edgeSet.add(i);
   219→   219→        nodeSet.add(e.from);
   220→   220→        nodeSet.add(e.to);
   221→   221→      }
   222→   222→    });
   223→   223→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   224→   224→  }, [activeFilter, edges, nodes]);
   225→   225→
   226→   226→  // ── Draw SVG edges ───────────────────────────────────────────────
   227→   227→  const drawEdges = useCallback(() => {
   228→   228→    const svg = svgRef.current;
   229→   229→    const graph = graphRef.current;
   230→   230→    if (!svg || !graph || edges.length === 0) return;
   231→   231→
   232→   232→    let markersSvg = '';
   233→   233→    const seenColors = new Set<string>();
   234→   234→    for (const e of edges) {
   235→   235→      const c = EDGE_COLORS[e.type];
   236→   236→      if (c && !seenColors.has(e.type)) {
   237→   237→        seenColors.add(e.type);
   238→   238→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   239→   239→      }
   240→   240→    }
   241→   241→
   242→   242→    let pathsSvg = '';
   243→   243→    const graphRect = graph.getBoundingClientRect();
   244→   244→
   245→   245→    for (let ei = 0; ei < edges.length; ei++) {
   246→   246→      const { from: fromId, to: toId, label, type } = edges[ei];
   247→   247→      const isVisible = visibleEdgeIndices.includes(ei);
   248→   248→      const fromEl = nodeRefs.current[fromId];
   249→   249→      const toEl = nodeRefs.current[toId];
   250→   250→      if (!fromEl || !toEl) continue;
   251→   251→
   252→   252→      const fRect = fromEl.getBoundingClientRect();
   253→   253→      const tRect = toEl.getBoundingClientRect();
   254→   254→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   255→   255→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   256→   256→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   257→   257→      const by = tRect.top - graphRect.top + tRect.height / 2;
   258→   258→      const dx = bx - ax;
   259→   259→      const dy = by - ay;
   260→   260→      const dist = Math.sqrt(dx * dx + dy * dy);
   261→   261→
   262→   262→      if (dist === 0) continue;
   263→   263→
   264→   264→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   265→   265→      const mx = (ax + bx) / 2;
   266→   266→      const my = (ay + by) / 2;
   267→   267→      const nx = -dy / dist;
   268→   268→      const ny = dx / dist;
   269→   269→      const cx = mx + nx * bend;
   270→   270→      const cy = my + ny * bend;
   271→   271→
   272→   272→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   273→   273→      const prob = edgeProbabilities[ei] ?? 0;
   274→   274→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   275→   275→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   276→   276→
   277→   277→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.72 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   278→   278→
   279→   279→      if (isVisible && ei >= 6) {
   280→   280→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#ffffff" font-size="10" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="4" stroke-linejoin="round" opacity="0.95" data-type="${type}" class="edge-label">${label}</text>`;
   281→   281→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   282→   282→      }
   283→   283→    }
   284→   284→
   285→   285→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   286→   286→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   287→   287→
   288→   288→  useEffect(() => { drawEdges(); }, [drawEdges]);
   289→   289→
   290→   290→  useEffect(() => {
   291→   291→    const el = shellRef.current;
   292→   292→    if (!el) return;
   293→   293→    const observer = new ResizeObserver(() => drawEdges());
   294→   294→    observer.observe(el);
   295→   295→    return () => observer.disconnect();
   296→   296→  }, [drawEdges]);
   297→   297→
   298→   298→
   299→   299→  // ── Detail panel content ─────────────────────────────────────────
   300→   300→  const detailContent = useMemo(() => {
   301→   301→    if (!selectedNode) return null;
   302→   302→    const node = nodeMap[selectedNode];
   303→   303→    if (!node) return null;
   304→   304→
   305→   305→    if (node.isTerminal && SCENARIO_KEYS.includes(selectedNode as typeof SCENARIO_KEYS[number])) {
   306→   306→      const contrib = pathContributions[selectedNode] ?? { trend: 0, breakout: 0, reversal: 0 };
   307→   307→      const meta = SCENARIO_META_LOCAL[selectedNode];
   308→   308→      const s = scenarios[selectedNode as keyof typeof scenarios];
   309→   309→      return (
   310→   310→        <div style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}>
   311→   311→          <h3 style={{ fontSize: '14px', fontWeight: 700, margin: 0, color: meta.color }}>{node.title}</h3>
   312→   312→          <p style={{ fontSize: '12px', color: '#ffffff', lineHeight: 1.95, margin: 0 }}>{node.desc}</p>
   313→   313→          <span style={{
   314→   314→            display: 'inline-block', padding: '4px 8px', margin: '3px 2px', borderRadius: 8,
   315→   315→            color: '#ffffff', fontSize: '10px',
   316→   316→            background: 'rgba(255,255,255,.06)', border: '1px solid rgba(255,255,255,.09)',
   317→   317→          }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   318→   318→          <div style={{ fontSize: '12px', color: '#e0eaf0' }} dir="ltr">{nodeValues[selectedNode] ?? '--'}</div>
   319→   319→
   320→   320→          <div style={{ marginTop: 12, paddingTop: 12, borderTop: `1px solid ${D.line}` }}>
   321→   321→            <p style={{ fontSize: '12px', fontWeight: 700, marginBottom: 8, color: '#ffffff' }}>سهم هر استراتژی:</p>
   322→   322→            {Object.entries(BRANCH_META).map(([bKey, bMeta]) => {
   323→   323→              const val = contrib[bKey as 'trend' | 'breakout' | 'reversal'];
   324→   324→              const pct = (val * 100).toFixed(1);
   325→   325→              return (
   326→   326→                <div key={bKey} style={{
   327→   327→                  display: 'flex', alignItems: 'center', justifyContent: 'space-between',
   328→   328→                  padding: '9px 0', borderBottom: '1px dashed rgba(255,255,255,.12)',
   329→   329→                }}>
   330→   330→                  <div style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
   331→   331→                    <span style={{ display: 'inline-block', width: 8, height: 8, borderRadius: '50%', background: bMeta.color }} />
   332→   332→                    <span style={{ fontSize: '11px', color: '#ffffff' }}>{bMeta.label}</span>
   333→   333→                  </div>
   334→   334→                  <span style={{ fontSize: '11px', fontWeight: 700, color: bMeta.color }}>{toPersianDigits(pct)}٪</span>
   335→   335→                </div>
   336→   336→              );
   337→   337→            })}
   338→   338→            <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '9px 0' }}>
   339→   339→              <span style={{ fontSize: '11px', fontWeight: 700, color: '#ffffff' }}>مجموع</span>
   340→   340→              <span style={{ fontSize: '11px', fontWeight: 900, color: meta.color }}>{toFa(s?.probability ?? 0)}٪</span>
   341→   341→            </div>
   342→   342→          </div>
   343→   343→        </div>
   344→   344→      );
   345→   345→    }
   346→   346→
   347→   347→    const inputs = edges.filter(e => e.to === selectedNode);
   348→   348→    const outputs = edges.filter(e => e.from === selectedNode);
   349→   349→
   350→   350→    return (
   351→   351→      <div style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}>
   352→   352→        <h3 style={{ fontSize: '14px', fontWeight: 700, margin: 0, color: '#ffffff' }}>{node.title}</h3>
   353→   353→        <p style={{ fontSize: '12px', color: '#ffffff', lineHeight: 1.95, margin: 0 }}><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   354→   354→        <span style={{
   355→   355→          display: 'inline-block', padding: '4px 8px', margin: '3px 2px', borderRadius: 8,
   356→   356→          color: '#ffffff', fontSize: '10px',
   357→   357→          background: 'rgba(255,255,255,.06)', border: '1px solid rgba(255,255,255,.09)',
   358→   358→        }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   359→   359→        <p style={{ fontSize: '12px', color: '#e0eaf0', lineHeight: 1.95, margin: 0 }}>{node.desc}</p>
   360→   360→        {inputs.length > 0 && (
   361→   361→          <div>
   362→   362→            <p style={{ fontSize: '12px', fontWeight: 600, marginBottom: 4, color: '#ffffff' }}>مسیرهای ورودی ({toFa(inputs.length)}):</p>
   363→   363→            <ul style={{ margin: '12px 0 0', padding: 0, listStyle: 'none' }}>
   364→   364→              {inputs.map((e, i) => {
   365→   365→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   366→   366→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   367→   367→                return (
   368→   368→                  <li key={i} style={{
   369→   369→                    padding: '9px 0', borderTop: '1px dashed rgba(255,255,255,.12)',
   370→   370→                    fontSize: '11px', lineHeight: 1.8, color: '#e0eaf0',
   371→   371→                  }}>
   372→   372→                    <b style={{ color: '#ffffff' }}>{SCENARIO_DISPLAY[e.from] || e.from} ← {SCENARIO_DISPLAY[e.to] || e.to}</b>
   373→   373→                    <span style={{
   374→   374→                      marginRight: 8, padding: '2px 6px', borderRadius: 4,
   375→   375→                      fontSize: '9px', fontWeight: 700,
