    42→function renderAIText(text: string): string {
    43→  // Remove CJK (Chinese/Japanese/Korean) characters — prohibited in output
    44→  const noCjk = text.replace(/[\u4e00-\u9fff\u3400-\u4dbf\u3040-\u309f\u30a0-\u30ff\uac00-\ud7af\uff00-\uffef]/g, '');
    45→
    46→  // Normalize zero-width and look-alike characters that may interfere with regex
    47→  const normalized = noCjk.replace(/[\u200c\u200d\u200b\ufeff]/g, '');
    48→
    49→  // Color syntax: {color:hex}text{/color} or {color:named}text{/color} → <span>
    50→  const colorMap: Record<string, string> = {
    51→    'red': '#dc2626', 'red-600': '#dc2626', 'red-700': '#b91c1c',
    52→    'green': '#16a34a', 'emerald': '#059669', 'emerald-600': '#059669', 'emerald-700': '#047857',
    53→    'amber': '#d97706', 'amber-600': '#d97706', 'amber-700': '#b45309', 'amber-800': '#92400e',
    54→    'blue': '#2563eb', 'blue-600': '#2563eb', 'blue-700': '#1d4ed8',
    55→    'orange': '#ea580c',
    56→    'purple': '#9333ea', 'purple-600': '#9333ea', 'purple-700': '#7e22ce',
    57→  };
    58→
    59→  const withColors = normalized.replace(/\{color:([^}]+)\}([\s\S]*?)\{\/color\}/g, (_match, colorName: string, inner: string) => {
    60→    const hex = colorMap[colorName.trim()] || (colorName.startsWith('#') ? colorName : null);
    61→    if (!hex) return inner; // Unknown color: just return text without color
    62→    return `<span style="color:${hex}">${inner}</span>`;
    63→  });
    64→
    65→  return withColors
    66→    .split('\n\n')
    67→    .map(p => {
    68→      // Convert **bold** to <strong>
    69→      const html = p.replace(/\*\*(.+?)\*\*/g, '<strong class="font-bold text-gray-900">$1</strong>');
    70→      // Convert newlines within paragraph to <br>
    71→      return `<p class="mb-4">${html.replace(/\n/g, '<br>')}</p>`;
    72→    })
    73→    .join('');
    74→}
