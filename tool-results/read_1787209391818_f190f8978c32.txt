   142→type ToolType = 'cursor' | 'trendline' | 'hline' | 'vline' | 'ray' | 'fibonacci' | 'fibext' | 'rectangle' | 'ellipse' | 'path' | 'text' | 'callout' | 'brush' | 'measure' | 'arrow' | 'pitchfork' | 'channel' | 'regression';

   144→// — S/R strength-based line styling (solid only, thickness = strength) ————
   145→function srLineStyle(strength: number, isTarget: boolean): { lineWidth: number; lineStyle: 0; color: string } {
   146→  if (isTarget) {
   147→    return { lineWidth: 4, lineStyle: 0, color: '' };
   148→  }
   149→  if (strength >= 9) return { lineWidth: 5, lineStyle: 0, color: '' };
   150→  if (strength >= 7) return { lineWidth: 4, lineStyle: 0, color: '' };
   151→  if (strength >= 5) return { lineWidth: 3, lineStyle: 0, color: '' };
   152→  if (strength >= 3) return { lineWidth: 2, lineStyle: 0, color: '' };
   153→  return { lineWidth: 1, lineStyle: 0, color: '' };
   154→}

   195→
   196→export default function CandlestickChart({ data, ta, height = 520 }: CandlestickChartProps) {
   197→  const chartContainerRef = useRef<HTMLDivElement>(null);
   198→  const svgRef = useRef<SVGSVGElement>(null);
   199→  const chartRef = useRef<IChartApi | null>(null);
   200→  const csRef = useRef<any>(null);
   201→  const subscriberRef = useRef<{ dispose: () => void } | null>(null);
   202→  const crosshairSubRef = useRef<{ dispose: () => void } | null>(null);
   203→
   204→  // Crosshair legend state
   205→  const [hoverInfo, setHoverInfo] = useState<{ date: string; o: number; h: number; l: number; c: number; v: number; chg: number } | null>(null);
   206→
   207→  // Build Jalali date strings for all candles (memoized)
   208→  const jalaliDates = useMemo(() => {
   209→    return data.map((d) =>
   210→      isGregorianDate(d.date) ? candleDateToJalali(d.date, 'full') : formatJalaliString(d.date, 'full')
   211→    );
   212→  }, [data]);
   213→
   214→  const [drawings, setDrawings] = useState<Drawing[]>([]);
   215→  const [activeTool, setActiveTool] = useState<ToolType>('cursor');
   216→  const [activeColor, setActiveColor] = useState(PALETTE[0]);
   217→  const [isDrawing, setIsDrawing] = useState(false);
   218→  const [currentDrawing, setCurrentDrawing] = useState<Drawing | null>(null);
   219→  const [textInput, setTextInput] = useState<{ x: number; y: number; drawingId: string } | null>(null);
   220→  const [textValue, setTextValue] = useState('');
   221→  const [renderKey, setRenderKey] = useState(0);
   222→  const [chartSize, setChartSize] = useState({ w: 0, h: 0 });
   223→
   224→  /* - Helper: coordinate conversion -------------- */
   225→  const getChartPoint = useCallback((clientX: number, clientY: number): DrawingPoint | null => {
   226→    const chart = chartRef.current;
   227→    const container = chartContainerRef.current;
   228→    if (!chart || !container) return null;
   229→    const rect = container.getBoundingClientRect();
   230→    const x = clientX - rect.left;
   231→    const y = clientY - rect.top;
   232→    const price = csRef.current?.coordinateToPrice(y) ?? null;
   233→    const time = chart.timeScale().coordinateToTime(x);
   234→    if (price === null || time === null) return null;
   235→    return { time: time as number, price };
   236→  }, []);
   237→
   238→  /* - SVG Rendering --------------------- */
   239→  const renderDrawings = useCallback(() => {
   240→    const chart = chartRef.current;
   241→    const svg = svgRef.current;
   242→    const container = chartContainerRef.current;
   243→    if (!chart || !svg || !container || drawings.length === 0) {
   244→      if (svg) svg.innerHTML = '';
   245→      return;
   246→    }
   247→
   248→    const rect = container.getBoundingClientRect();
   249→    const w = rect.width;
   250→    const h = rect.height;
   251→    if (w === 0 || h === 0) return;
   252→
   253→    let html = '';
   254→
   255→    // Defs for arrow markers
   256→    html += `<defs>`;
   257→    PALETTE.forEach((c) => {
   258→      html += `<marker id="arrow-${c.replace('#','')}" markerWidth="8" markerHeight="6" refX="7" refY="3" orient="auto"><polygon points="0 0, 8 3, 0 6" fill="${c}" /></marker>`;
   259→    });
   260→    html += `</defs>`;
   261→
   262→    const allDrawings = [...drawings];
   263→    if (currentDrawing) allDrawings.push(currentDrawing);
   264→
   265→    for (const d of allDrawings) {
   266→      if (!d.completed && d.points.length === 0) continue;
   267→      const col = d.color;
   268→      const lw = d.lineWidth;
   269→
   270→      switch (d.type) {
   271→        case 'trendline': {
   272→          if (d.points.length < 1) break;
   273→          const p0 = d.points[0];
   274→          const p1 = d.points.length > 1 ? d.points[d.points.length - 1] : p0;
   275→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   276→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   277→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   278→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   279→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   280→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" />`;
   281→          break;
   282→        }
   283→        case 'hline': {
   284→          if (d.points.length < 1) break;
   285→          const p = d.points[0];
   286→          const y = csRef.current?.priceToCoordinate(p.price);
   287→          if (y == null) break;
   288→          html += `<line x1="0" y1="${y}" x2="${w}" y2="${y}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   289→          html += `<text x="4" y="${y - 4}" fill="${col}" font-size="11" font-family="Vazirmatn, sans-serif">${toPersianDigits(p.price.toFixed(0))}</text>`;
   290→          break;
   291→        }
   292→        case 'vline': {
   293→          if (d.points.length < 1) break;
   294→          const p = d.points[0];
   295→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   296→          if (x == null) break;
   297→          html += `<line x1="${x}" y1="0" x2="${x}" y2="${h}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   298→          break;
   299→        }
   300→        case 'fibonacci': {
   301→          if (d.points.length < 2) break;
   302→          const p0 = d.points[0];
   303→          const p1 = d.points[1];
   304→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   305→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   306→          if (x0 == null || x1 == null) break;
   307→          const priceHigh = Math.max(p0.price, p1.price);
   308→          const priceLow = Math.min(p0.price, p1.price);
   309→          const priceRange = priceHigh - priceLow;
   310→          if (priceRange === 0) break;
   311→
   312→          const yTop = csRef.current?.priceToCoordinate(priceHigh);
   313→          const yBottom = csRef.current?.priceToCoordinate(priceLow);
   314→          if (yTop == null || yBottom == null) break;
   315→          html += `<rect x="${Math.min(x0,x1)}" y="${yTop}" width="${Math.abs(x1-x0)}" height="${yBottom - yTop}" fill="${col}" opacity="0.04" />`;
   316→
   317→          for (const level of FIB_LEVELS) {
   318→            const price = priceHigh - level.pct * priceRange;
   319→            const y = csRef.current?.priceToCoordinate(price);
   320→            if (y == null) continue;
   321→            html += `<line x1="${Math.min(x0,x1) - 10}" y1="${y}" x2="${Math.max(x0,x1) + 10}" y2="${y}" stroke="${col}" stroke-width="1" opacity="0.6" />`;
   322→            html += `<rect x="${Math.max(x0,x1) + 12}" y="${y - 8}" width="56" height="16" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="2" />`;
   323→            html += `<text x="${Math.max(x0,x1) + 14}" y="${y + 4}" fill="${col}" font-size="10" font-family="Vazirmatn, sans-serif">${toPersianDigits(level.label)} ${toPersianDigits(price.toFixed(0))}</text>`;
   324→          }
   325→          break;
   326→        }
   327→        case 'rectangle': {
   328→          if (d.points.length < 2) break;
   329→          const p0 = d.points[0];
   330→          const p1 = d.points[d.points.length - 1];
   331→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   332→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   333→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   334→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   335→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   336→          const rx = Math.min(x0, x1);
   337→          const ry = Math.min(y0, y1);
   338→          const rw = Math.abs(x1 - x0);
   339→          const rh = Math.abs(y1 - y0);
   340→          html += `<rect x="${rx}" y="${ry}" width="${rw}" height="${rh}" fill="${col}" fill-opacity="0.08" stroke="${col}" stroke-width="${lw}" rx="2" />`;
   341→          break;
   342→        }
   343→        case 'text': {
   344→          if (d.points.length < 1 || !d.text) break;
   345→          const p = d.points[0];
   346→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   347→          const y = csRef.current?.priceToCoordinate(p.price);
   348→          if (x == null || y == null) break;
   349→          html += `<rect x="${x + 4}" y="${y - 16}" width="${d.text.length * 8 + 12}" height="22" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="3" />`;
   350→          html += `<text x="${x + 10}" y="${y + 1}" fill="${col}" font-size="12" font-family="Vazirmatn, sans-serif">${d.text}</text>`;
   351→          break;
   352→        }
   353→        case 'brush': {
   354→          if (d.points.length < 2) break;
   355→          let pts = '';
   356→          let valid = true;
   357→          for (const p of d.points) {
   358→            const x = chart.timeScale().timeToCoordinate(p.time as Time);
   359→            const y = csRef.current?.priceToCoordinate(p.price);
   360→            if (x == null || y == null) { valid = false; break; }
   361→            pts += `${x},${y} `;
   362→          }
   363→          if (!valid || pts.length === 0) break;
   364→          html += `<polyline points="${pts.trim()}" fill="none" stroke="${col}" stroke-width="${lw}" stroke-linecap="round" stroke-linejoin="round" />`;
   365→          break;
   366→        }
   367→        case 'measure': {
   368→          if (d.points.length < 2) break;
   369→          const p0 = d.points[0];
   370→          const p1 = d.points[d.points.length - 1];
   371→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   372→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   373→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   374→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   375→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   376→
   377→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="4,2" />`;
   378→          const diff = p1.price - p0.price;
   379→          const pctChg = p0.price !== 0 ? ((diff / p0.price) * 100) : 0;
   380→          const bars = Math.abs(Math.round(p1.time - p0.time));
   381→          const midX = (x0 + x1) / 2;
   382→          const midY = (y0 + y1) / 2;
   383→          const label = `${diff >= 0 ? '+' : ''}${toPersianDigits(diff.toFixed(0))} (${pctChg >= 0 ? '+' : ''}${toPersianDigits(pctChg.toFixed(1))}%) ${toPersianDigits(String(bars))}bar`;
   384→          html += `<rect x="${midX - 40}" y="${midY - 24}" width="80" height="20" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="3" />`;
   385→          html += `<text x="${midX - 36}" y="${midY - 10}" fill="${col}" font-size="10" font-family="Vazirmatn, sans-serif">${label}</text>`;
   386→          break;
   387→        }
   388→        case 'arrow': {
   389→          if (d.points.length < 2) break;
   390→          const p0 = d.points[0];
   391→          const p1 = d.points[d.points.length - 1];
   392→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   393→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   394→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   395→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   396→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   397→          const markerId = `arrow-${col.replace('#','')}`;
   398→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" marker-end="url(#${markerId})" />`;
   399→          break;
   400→        }
   401→        case 'ray': {
   402→          if (d.points.length < 2) break;
   403→          const p0 = d.points[0];
   404→          const p1 = d.points[d.points.length - 1];
   405→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   406→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   407→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   408→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   409→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   410→          // Extend the line from p1 to the edge of the chart
   411→          const dx = x1 - x0;
   412→          const dy = y1 - y0;
   413→          const extendFactor = 10;
   414→          const xEnd = x1 + dx * extendFactor;
   415→          const yEnd = y1 + dy * extendFactor;
   416→          html += `<line x1="${x0}" y1="${y0}" x2="${xEnd}" y2="${yEnd}" stroke="${col}" stroke-width="${lw}" />`;
   417→          break;
   418→        }
   419→        case 'fibext': {
   420→          if (d.points.length < 2) break;
   421→          const p0 = d.points[0];
   422→          const p1 = d.points[1];
   423→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);

   439→          }

   445→            const xStart = isExt ? Math.max(x0, x1) : Math.min(x0, x1) - 10;

   450→          }
   451→          break;
   452→        }
   453→        case 'ellipse': {
   454→          if (d.points.length < 2) break;
   455→          const p0 = d.points[0];
   456→          const p1 = d.points[d.points.length - 1];

   519→          const dx = x1 - x0;
   520→          const dy = y1 - y0;

   529→          break;
   530→        }
   531→        case 'channel': {
   532→          if (d.points.length < 3) break;
   533→          const p0 = d.points[0];
   534→          const p1 = d.points[1];
   535→          const p2 = d.points[2];

   548→          const slope = dx !== 0 ? dy / dx : 0;

   550→          const offset = y2 - lineYatX2;

   553→          const extX = x1 + dx * 5;
   554→          const extY1 = y1 + dy * 5;
   555→          const extY2 = y1 + dy * 5 + offset;
   556→          html += `<polygon points="${x0},${y0} ${extX},${extY1} ${extX},${extY2} ${x0},${y0 + offset}" fill="${col}" fill-opacity="0.05" />`;
   557→          break;
   558→        }
   559→        case 'regression': {
   560→          if (d.points.length < 2) break;
   561→          const p0 = d.points[0];
   562→          const p1 = d.points[d.points.length - 1];

   571→          const diff = p1.price - p0.price;

   574→          const midY = (y0 + y1) / 2 - 14;

   578→          break;
   579→        }
   580→      }
   581→    }

   864→      subscriberRef.current = null;
   865→      crosshairSubRef.current?.dispose();
   866→      crosshairSubRef.current = null;
   867→      if (chartRef.current) {
   868→        chartRef.current.remove();
   869→        chartRef.current = null;
   870→      }
   871→    };
   872→  }, [build]);
   873→
   874→  /* - Mouse event handlers ----------------- */
   875→  const handleMouseDown = useCallback(
   876→    (e: React.MouseEvent) => {
   877→      if (activeTool === 'cursor') return;
   878→      if (textInput) return;
   879→
   880→      const point = getChartPoint(e.clientX, e.clientY);
   881→      if (!point) return;
   882→
   883→      if (activeTool === 'hline') {
   884→        const drawing: Drawing = {
   885→          id: Date.now().toString(),
   886→          type: 'hline',
   887→          points: [point],
   888→          color: activeColor,
   889→          lineWidth: 1,
   890→          completed: true,
   891→        };
   892→        setDrawings((prev) => [...prev, drawing]);
   893→        setIsDrawing(false);
   894→        return;
   895→      }
   896→
   897→      if (activeTool === 'text') {
   898→        const chart = chartRef.current;
   899→        const container = chartContainerRef.current;
   900→        if (!chart || !container) return;
   901→        const rect = container.getBoundingClientRect();
   902→        const x = e.clientX - rect.left;
   903→        const y = e.clientY - rect.top;
   904→
   905→        const tempDrawing: Drawing = {
   906→          id: Date.now().toString(),
   907→          type: 'text',
   908→          points: [point],
   909→          color: activeColor,
   910→          lineWidth: 1,
   911→          text: '',
   912→          completed: false,
   913→        };
   914→        setCurrentDrawing(tempDrawing);
   915→        setTextInput({ x, y, drawingId: tempDrawing.id });
   916→        setTextValue('');
   917→        setIsDrawing(true);
   918→        return;
   919→      }
   920→
   921→      if (activeTool === 'callout') {
   922→        const chart = chartRef.current;
   923→        const container = chartContainerRef.current;
   924→        if (!chart || !container) return;
   925→        const rect = container.getBoundingClientRect();
   926→        const x = e.clientX - rect.left;
   927→        const y = e.clientY - rect.top;
   928→
   929→        const tempDrawing: Drawing = {
   930→          id: Date.now().toString(),
   931→          type: 'callout',
   932→          points: [point],
   933→          color: activeColor,
   934→          lineWidth: 1,
   935→          text: '',
   936→          completed: false,
   937→        };
   938→        setCurrentDrawing(tempDrawing);
   939→        setTextInput({ x, y, drawingId: tempDrawing.id });
   940→        setTextValue('');
   941→        setIsDrawing(true);
   942→        return;
   943→      }
   944→
   945→      const drawing: Drawing = {
   946→        id: Date.now().toString(),
   947→        type: activeTool,
   948→        points: [point],
   949→        color: activeColor,
   950→        lineWidth: activeTool === 'brush' ? 2 : 1,
   951→        completed: false,
   952→      };
   953→      setCurrentDrawing(drawing);
   954→      setIsDrawing(true);
   955→    },
   956→    [activeTool, activeColor, getChartPoint, textInput]
   957→  );
   958→
   959→  const handleMouseMove = useCallback(
   960→    (e: React.MouseEvent) => {
   961→      if (!isDrawing || !currentDrawing) return;
   962→
   963→      const point = getChartPoint(e.clientX, e.clientY);
   964→      if (!point) return;
   965→
   966→      if (currentDrawing.type === 'brush' || currentDrawing.type === 'path') {
   967→        setCurrentDrawing({
   968→          ...currentDrawing,
   969→          points: [...currentDrawing.points, point],
   970→        });
   971→      } else if (currentDrawing.type !== 'text' && currentDrawing.type !== 'hline' && currentDrawing.type !== 'callout') {
   972→        if (currentDrawing.type === 'pitchfork' || currentDrawing.type === 'channel') {
   973→          // 3-point tools: accumulate points
   974→          const pts = [...currentDrawing.points, point];
   975→          if (pts.length > 3) {
   976→            pts[pts.length - 1] = point;
   977→          }
   978→          setCurrentDrawing({
   979→            ...currentDrawing,
   980→            points: pts,
   981→          });
   982→        } else {
   983→          setCurrentDrawing({
   984→            ...currentDrawing,
   985→            points: [currentDrawing.points[0], point],
   986→          });
   987→        }
   988→      }
   989→    },
   990→    [isDrawing, currentDrawing, getChartPoint]
   991→  );
   992→
   993→  const handleMouseUp = useCallback(() => {
   994→    if (!isDrawing || !currentDrawing) return;
   995→
   996→    if (currentDrawing.type === 'text') return;
   997→
   998→    const completed: Drawing = {
   999→      ...currentDrawing,
  1000→      completed: true,
  1001→    };
  1002→
  1003→    const needs3Points = currentDrawing.type === 'pitchfork' || currentDrawing.type === 'channel';
  1004→    const minPoints = currentDrawing.type === 'hline' ? 1 : needs3Points ? 3 : 2;
  1005→    if (needs3Points && completed.points.length < 3) {
  1006→      // Don't finalize yet, keep collecting points
  1007→      setCurrentDrawing(completed);
  1008→      return;
  1009→    }
  1010→    if (completed.points.length >= minPoints) {
  1011→      setDrawings((prev) => [...prev, completed]);
  1012→    }
  1013→
  1014→    setCurrentDrawing(null);
  1015→    setIsDrawing(false);
  1016→  }, [isDrawing, currentDrawing]);
  1017→
  1018→  /* - Text input submit ------------------- */
  1019→  const handleTextSubmit = useCallback(() => {
  1020→    if (!textInput || !textValue.trim() || !currentDrawing) {
  1021→      setTextInput(null);
  1022→      setCurrentDrawing(null);
  1023→      setIsDrawing(false);
  1024→      return;
  1025→    }
  1026→
  1027→    const completed: Drawing = {
  1028→      ...currentDrawing,
  1029→      text: textValue.trim(),
  1030→      completed: true,
  1031→    };
  1032→    setDrawings((prev) => [...prev, completed]);
  1033→    setTextInput(null);
  1034→    setTextValue('');
  1035→    setCurrentDrawing(null);
  1036→    setIsDrawing(false);
  1037→  }, [textInput, textValue, currentDrawing]);
  1038→
  1039→  const handleTextCancel = useCallback(() => {
  1040→    setTextInput(null);
  1041→    setTextValue('');
  1042→    setCurrentDrawing(null);
  1043→    setIsDrawing(false);
  1044→  }, []);
  1045→
  1046→  /* - Delete / Clear -------------------- */
  1047→  const handleDeleteLast = useCallback(() => {
  1048→    setDrawings((prev) => prev.slice(0, -1));
  1049→  }, []);
  1050→
  1051→  const handleClearAll = useCallback(() => {
  1052→    setDrawings([]);
  1053→  }, []);
  1054→
  1055→  /* - Render ------------------------- */
  1056→  const fmt = (n: number) => toPersianDigits(n.toLocaleString('en', { maximumFractionDigits: 0 }));
  1057→
  1058→  return (
  1059→    <div className="w-full">
  1060→      {/* - CROSSHAIR LEGEND (Shamsi date + OHLC) ---------- */}
  1061→      {hoverInfo && (
  1062→        <div className="flex items-center gap-3 px-3 py-1.5 border border-[#e5e7eb] rounded-t-lg bg-[#ffffff] text-[11px]" dir="rtl" style={{ fontFamily: 'Vazirmatn, sans-serif' }}>
  1063→          <span className="text-gray-500 font-medium">{hoverInfo.date}</span>
  1064→          <div className="w-px h-3.5 bg-gray-200" />
  1065→          <span className="text-gray-500">باز: <span className={chgColor(hoverInfo.o, hoverInfo.c) === BULL ? 'text-emerald-700 font-bold' : 'text-red-700 font-bold'}>{fmt(hoverInfo.o)}</span></span>
  1066→          <span className="text-gray-500">بالا: <span className="text-emerald-700 font-bold">{fmt(hoverInfo.h)}</span></span>
  1067→          <span className="text-gray-500">پایین: <span className="text-red-700 font-bold">{fmt(hoverInfo.l)}</span></span>
  1068→          <span className="text-gray-500">بسته: <span className={chgColor(hoverInfo.o, hoverInfo.c) === BULL ? 'text-emerald-700 font-bold' : 'text-red-700 font-bold'}>{fmt(hoverInfo.c)}</span></span>
  1069→          {hoverInfo.v > 0 && (
  1070→            <span className="text-gray-500">حجم: <span className="text-gray-900 font-bold">{toPersianDigits((hoverInfo.v / 1e6).toFixed(1))}M</span></span>
  1071→          )}
  1072→          <span className={`font-bold ${hoverInfo.chg >= 0 ? 'text-emerald-700' : 'text-red-700'}`}>
  1073→            {hoverInfo.chg >= 0 ? '▲' : '▼'} {toPersianDigits(Math.abs(hoverInfo.chg).toFixed(2))}٪
  1074→          </span>
  1075→        </div>
  1076→      )}
  1077→      {/* - TOOLBAR --------------------- */}
  1078→      <div
  1079→        className={`flex items-center gap-1 px-2 py-1.5 overflow-x-auto flex-nowrap border ${hoverInfo ? 'border-t-0 rounded-b-lg' : 'rounded-t-lg border-b-0'}`}
  1080→        style={{ backgroundColor: TOOLBAR_BG, borderColor: TOOLBAR_BORDER }}
  1081→      >
  1082→        {TOOLS.map((tool) => (
  1083→          <button
  1084→            key={tool.id}
  1085→            onClick={() => {
  1086→              setActiveTool(tool.id);
  1087→              if (isDrawing) {
  1088→                setIsDrawing(false);
  1089→                setCurrentDrawing(null);
  1090→                setTextInput(null);
  1091→              }
  1092→            }}
  1093→            title={tool.label}
  1094→            className={`shrink-0 flex items-center justify-center w-8 h-8 rounded-md transition-all cursor-pointer ${
  1095→              activeTool === tool.id
  1096→                ? 'bg-amber-500/20 text-amber-800 ring-1 ring-amber-500/30'
  1097→                : 'text-gray-500 hover:bg-gray-100 hover:text-gray-700'
  1098→            }`}
  1099→          >
  1100→            {tool.icon}
  1101→          </button>
  1102→        ))}
  1103→
  1104→        <div className="w-px h-6 bg-gray-200 mx-1 shrink-0" />
  1105→
  1106→        {PALETTE.map((c) => (
  1107→          <button
  1108→            key={c}
  1109→            onClick={() => setActiveColor(c)}
  1110→            title={c}
  1111→            className={`shrink-0 w-6 h-6 rounded-full border-2 transition-all cursor-pointer ${
  1112→              activeColor === c ? 'border-gray-900 scale-110' : 'border-transparent hover:scale-105'
  1113→            }`}
  1114→            style={{ backgroundColor: c }}
  1115→          />
  1116→        ))}
  1117→
  1118→        <div className="w-px h-6 bg-gray-200 mx-1 shrink-0" />
  1119→
  1120→        <button
  1121→          onClick={handleDeleteLast}
  1122→          title="حذف آخر"
  1123→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-500 hover:bg-red-50 hover:text-red-700 transition-all cursor-pointer"
  1124→        >
  1125→          <Trash2 className="w-3.5 h-3.5" />
  1126→        </button>
  1127→
  1128→        <button
  1129→          onClick={handleClearAll}
  1130→          title="پاک کردن همه"
  1131→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-500 hover:bg-red-50 hover:text-red-700 transition-all cursor-pointer"
  1132→        >
  1133→          <X className="w-3.5 h-3.5" />
  1134→        </button>
  1135→
  1136→        <div className="mr-auto shrink-0 text-xs text-gray-500 font-medium" dir="rtl">
  1137→          {TOOLS.find((t) => t.id === activeTool)?.label}
  1138→        </div>
  1139→      </div>
  1140→
  1141→      <div
  1142→        className="relative rounded-b-lg overflow-hidden border border-gray-200"
  1143→        style={{ height }}
  1144→      >
  1145→        <div
  1146→          ref={chartContainerRef}
  1147→          className="w-full h-full"
  1148→          data-chart
  1149→        />
  1150→
  1151→        <svg
  1152→          ref={svgRef}
  1153→          className="absolute top-0 left-0"
  1154→          style={{
  1155→            width: chartSize.w || '100%',
  1156→            height: chartSize.h || '100%',
  1157→            pointerEvents: activeTool !== 'cursor' ? 'auto' : 'none',
  1158→            zIndex: 10,
  1159→          }}
  1160→          onMouseDown={handleMouseDown}
  1161→          onMouseMove={handleMouseMove}
  1162→          onMouseUp={handleMouseUp}
  1163→          onMouseLeave={handleMouseUp}
  1164→        />
  1165→
  1166→        {textInput && (
  1167→          <div
  1168→            className="absolute z-20"
  1169→            style={{
  1170→              left: textInput.x,
  1171→              top: textInput.y - 8,
  1172→            }}
  1173→          >
  1174→            <input
  1175→              type="text"
  1176→              value={textValue}
  1177→              onChange={(e) => setTextValue(e.target.value)}
  1178→              onKeyDown={(e) => {
  1179→                if (e.key === 'Enter') handleTextSubmit();
  1180→                if (e.key === 'Escape') handleTextCancel();
  1181→              }}
  1182→              placeholder="متن..."
  1183→              autoFocus
  1184→              dir="rtl"
  1185→              className="bg-white border border-gray-200 rounded px-2 py-1 text-sm text-gray-900 outline-none focus:border-amber-500 w-32"
  1186→              onBlur={handleTextSubmit}
  1187→            />
  1188→          </div>
  1189→        )}
  1190→      </div>
  1191→    </div>
  1192→  );
  1193→}
