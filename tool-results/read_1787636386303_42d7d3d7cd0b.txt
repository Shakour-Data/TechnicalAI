   167→   167→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   168→   168→function sma(closes: number[], period: number): number {
   169→   169→  if (closes.length < period) return 0;
   170→   170→  const slice = closes.slice(-period);
   171→   171→  return slice.reduce((a, b) => a + b, 0) / period;
   172→   172→}

   174→   174→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   175→   175→function emaCalc(closes: number[], period: number): number {
   176→   176→  if (closes.length < period) return 0;
   177→   177→  const k = 2 / (period + 1);
   178→   178→  let emaVal = sma(closes.slice(0, period), period);
   179→   179→  for (let i = period; i < closes.length; i++) {
   180→   180→    emaVal = closes[i] * k + emaVal * (1 - k);
   181→   181→  }
   182→   182→  return emaVal;
   183→   183→}

   185→   185→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   186→   186→function calcRSI(closes: number[], period: number = 14): number {
   187→   187→  if (closes.length < period + 1) return 50;
   188→   188→  let avgGain = 0;
   189→   189→  let avgLoss = 0;
   190→   190→  for (let i = 1; i <= period; i++) {
   191→   191→    const diff = closes[i] - closes[i - 1];
   192→   192→    if (diff > 0) avgGain += diff;
   193→   193→    else avgLoss += Math.abs(diff);
   194→   194→  }
   195→   195→  avgGain /= period;
   196→   196→  avgLoss /= period;
   197→   197→  for (let i = period + 1; i < closes.length; i++) {
   198→   198→    const diff = closes[i] - closes[i - 1];
   199→   199→    const gain = diff > 0 ? diff : 0;
   200→   200→    const loss = diff < 0 ? Math.abs(diff) : 0;
   201→   201→    avgGain = (avgGain * (period - 1) + gain) / period;
   202→   202→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   203→   203→  }
   204→   204→  if (avgLoss === 0) return 100;
   205→   205→  const rs = avgGain / avgLoss;
   206→   206→  return 100 - 100 / (1 + rs);
   207→   207→}

   209→   209→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   210→   210→function calcMFI(data: OHLCV[], period: number = 14): number {
   211→   211→  if (data.length < period + 1) return 50;
   212→   212→  let posFlow = 0;
   213→   213→  let negFlow = 0;
   214→   214→  for (let i = data.length - period; i < data.length; i++) {
   215→   215→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   216→   216→    const mf = tp * data[i].volume;
   217→   217→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   218→   218→    if (tp > prevTp) posFlow += mf;
   219→   219→    else negFlow += mf;
   220→   220→  }
   221→   221→  if (negFlow === 0) return 100;
   222→   222→  const mfr = posFlow / negFlow;
   223→   223→  return 100 - 100 / (1 + mfr);
   224→   224→}

   226→   226→// ─── Helper: CCI(20) ──────────────────────────────────────────────────────────
   227→   227→function calcCCI(data: OHLCV[], period: number = 20): number {
   228→   228→  if (data.length < period) return 0;
   229→   229→  const tps: number[] = [];
   230→   230→  for (let i = data.length - period; i < data.length; i++) {
   231→   231→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   232→   232→  }
   233→   233→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   234→   234→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   235→   235→  if (meanDev === 0) return 0;
   236→   236→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   237→   237→}

   240→   240→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   241→   241→  if (data.length < kPeriod) return { k: 50, d: 50 };
   242→   242→  const rawKs: number[] = [];
   243→   243→  for (let i = kPeriod - 1; i < data.length; i++) {
   244→   244→    let lowest = Infinity;
   245→   245→    let highest = -Infinity;
   246→   246→    for (let j = i - kPeriod + 1; j <= i; j++) {
   247→   247→      if (data[j].low < lowest) lowest = data[j].low;
   248→   248→      if (data[j].high > highest) highest = data[j].high;
   249→   249→    }
   250→   250→    const range = highest - lowest;
   251→   251→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   252→   252→  }
   253→   253→  const smoothedK: number[] = [];
   254→   254→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   255→   255→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   256→   256→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   257→   257→  }
   258→   258→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   259→   259→  const dSlice = smoothedK.slice(-smoothD);
   260→   260→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   261→   261→  return { k, d };
   262→   262→}

   265→   265→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   266→   266→  if (data.length < period) return -50;
   267→   267→  const slice = data.slice(-period);
   268→   268→  let highest = -Infinity;
   269→   269→  let lowest = Infinity;
   270→   270→  for (const c of slice) {
   271→   271→    if (c.high > highest) highest = c.high;
   272→   272→    if (c.low < lowest) lowest = c.low;
   273→   273→  }
   274→   274→  const range = highest - lowest;
   275→   275→  if (range === 0) return -50;
   276→   276→  return ((highest - data[data.length - 1].close) / range) * -100;
   277→   277→}

   279→   279→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   280→   280→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   281→   281→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   282→   282→  const emaFast = emaCalc(closes, fast);
   283→   283→  const emaSlow = emaCalc(closes, slow);
   284→   284→  const macdLine = emaFast - emaSlow;
   285→   285→  // Calculate signal as EMA of MACD values
   286→   286→  const macdValues: number[] = [];
   287→   287→  const kFast = 2 / (fast + 1);
   288→   288→  const kSlow = 2 / (slow + 1);
   289→   289→  let eFast = sma(closes.slice(0, fast), fast);
   290→   290→  let eSlow = sma(closes.slice(0, slow), slow);
   291→   291→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   292→   292→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   293→   293→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   294→   294→    macdValues.push(eFast - eSlow);
   295→   295→  }
   296→   296→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   297→   297→  const kSig = 2 / (sig + 1);
   298→   298→  let signalVal = sma(macdValues.slice(0, sig), sig);
   299→   299→  for (let i = sig; i < macdValues.length; i++) {
   300→   300→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   301→   301→  }
   302→   302→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   303→   303→}

   305→   305→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   306→   306→function calcATR(data: OHLCV[], period: number = 14): number {
   307→   307→  if (data.length < 2) return 0;
   308→   308→  const trs: number[] = [];
   309→   309→  for (let i = 1; i < data.length; i++) {
   310→   310→    const tr = Math.max(
   311→   311→      data[i].high - data[i].low,
   312→   312→      Math.abs(data[i].high - data[i - 1].close),
   313→   313→      Math.abs(data[i].low - data[i - 1].close)
   314→   314→    );
   315→   315→    trs.push(tr);
   316→   316→  }
   317→   317→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   318→   318→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   319→   319→  for (let i = period; i < trs.length; i++) {
   320→   320→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   321→   321→  }
   322→   322→  return atrVal;
   323→   323→}

   325→   325→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   326→   326→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   327→   327→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   328→   328→  const plusDMs: number[] = [];
   329→   329→  const minusDMs: number[] = [];
   330→   330→  const trs: number[] = [];
   331→   331→  for (let i = 1; i < data.length; i++) {
   332→   332→    const upMove = data[i].high - data[i - 1].high;
   333→   333→    const downMove = data[i - 1].low - data[i].low;
   334→   334→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   335→   335→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   336→   336→    trs.push(Math.max(
   337→   337→      data[i].high - data[i].low,
   338→   338→      Math.abs(data[i].high - data[i - 1].close),
   339→   339→      Math.abs(data[i].low - data[i - 1].close)
   340→   340→    ));
   341→   341→  }
   342→   342→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   343→   343→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   344→   344→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   345→   345→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   346→   346→  const dxValues: number[] = [];
   347→   347→  for (let i = period; i < trs.length; i++) {
   348→   348→    smoothTR = smoothTR - smoothTR / period + trs[i];
   349→   349→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   350→   350→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   351→   351→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   352→   352→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   353→   353→    const diSum = diPlus + diMinus;
   354→   354→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   355→   355→    dxValues.push(dx);
   356→   356→  }
   357→   357→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   358→   358→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   359→   359→  for (let i = period; i < dxValues.length; i++) {
   360→   360→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   361→   361→  }
   362→   362→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   363→   363→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   364→   364→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   365→   365→}

   367→   367→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   368→   368→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   369→   369→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   370→   370→  let isLong = true;
   371→   371→  let af = step;
   372→   372→  let ep = data[0].high;
   373→   373→  let sar = data[0].low;
   374→   374→  for (let i = 1; i < data.length; i++) {
   375→   375→    const prevSar = sar;
   376→   376→    sar = prevSar + af * (ep - prevSar);
   377→   377→    // Ensure SAR is within bounds
   378→   378→    if (isLong) {
   379→   379→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   380→   380→      if (data[i].low < sar) {
   381→   381→        isLong = false;
   382→   382→        sar = ep;
   383→   383→        ep = data[i].low;
   384→   384→        af = step;
   385→   385→      } else {
   386→   386→        if (data[i].high > ep) {
   387→   387→          ep = data[i].high;
   388→   388→          af = Math.min(af + step, max);
   389→   389→        }
   390→   390→      }
   391→   391→    } else {
   392→   392→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   393→   393→      if (data[i].high > sar) {
   394→   394→        isLong = true;
   395→   395→        sar = ep;
   396→   396→        ep = data[i].high;
   397→   397→        af = step;
   398→   398→      } else {
   399→   399→        if (data[i].low < ep) {
   400→   400→          ep = data[i].low;
   401→   401→          af = Math.min(af + step, max);
   402→   402→        }
   403→   403→      }
   404→   404→    }
   405→   405→  }
   406→   406→  return sar;
   407→   407→}

   409→   409→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   410→   410→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   411→   411→  if (closes.length < period) {
   412→   412→    const c = closes[closes.length - 1] ?? 0;
   413→   413→    return { upper: c, middle: c, lower: c };
   414→   414→  }
   415→   415→  const slice = closes.slice(-period);
   416→   416→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   417→   417→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   418→   418→  const sd = Math.sqrt(variance);
   419→   419→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   420→   420→}

   459→   459→function calcIchimoku(data: OHLCV[]): {
   460→   460→  tenkan: number; kijun: number; senkouA: number; senkouB: number; chikou: number;
   461→   461→} {
   462→   462→  if (data.length < 52) {
   463→   463→    const c = data[data.length - 1]?.close ?? 0;
   464→   464→    return { tenkan: c, kijun: c, senkouA: c, senkouB: c, chikou: c };
   465→   465→  }
   466→   466→  const highs = data.map(d => d.high);
   467→   467→  const lows = data.map(d => d.low);
   468→   468→  const closes = data.map(d => d.close);
   469→   469→
   470→   470→  function donchian(h: number[], l: number[], end: number, period: number): number {
   471→   471→    const start = end - period + 1;
   472→   472→    if (start < 0) return (h[end] + l[end]) / 2;
   473→   473→    let hi = -Infinity, lo = Infinity;
   474→   474→    for (let i = start; i <= end; i++) {
   475→   475→      if (h[i] > hi) hi = h[i];
   476→   476→      if (l[i] < lo) lo = l[i];
   477→   477→    }
   478→   478→    return (hi + lo) / 2;
   479→   479→  }
   480→   480→
   481→   481→  const n = data.length;
   482→   482→  const tenkan = donchian(highs, lows, n - 1, 9);
   483→   483→  const kijun = donchian(highs, lows, n - 1, 26);
   484→   484→  // Senkou A = (tenkan + kijun) / 2, shifted 26 periods ahead
   485→   485→  // We compute it at bar n-27 (shifted to current)
   486→   486→  const senkouAIdx = Math.max(n - 1 - 26, 0);
   487→   487→  const tA = donchian(highs, lows, senkouAIdx, 9);
   488→   488→  const kA = donchian(highs, lows, senkouAIdx, 26);
   489→   489→  const senkouA = (tA + kA) / 2;
   490→   490→  // Senkou B = donchian(52), shifted 26 periods ahead
   491→   491→  const senkouBIdx = Math.max(n - 1 - 26, 0);
   492→   492→  const senkouB = donchian(highs, lows, senkouBIdx, 52);
   493→   493→  // Chikou = current close plotted 26 periods back
   494→   494→  const chikou = closes[n - 1];
   495→   495→
   496→   496→  return { tenkan, kijun, senkouA, senkouB, chikou };
   497→   497→}

   501→   501→} {

   506→   506→  function donchian(end: number, period: number): number {

   510→   510→    for (let i = start; i <= end; i++) {
   511→   511→      if (highs[i] > hi) hi = highs[i];

   526→   526→    const sIdx = i - 26;

   530→   530→      senkouA.push((t + k) / 2);

   532→   532→      senkouA.push(donchian(i, 9)); // fallback
   533→   533→    }
   534→   534→    // Senkou B shifted +26: at bar i, we need donchian(52) at bar i-26
   535→   535→    if (sIdx >= 0) {
   536→   536→      senkouB.push(donchian(sIdx, 52));
   537→   537→    } else {
   538→   538→      senkouB.push(donchian(i, 52)); // fallback
   539→   539→    }
   540→   540→  }
   541→   541→
   542→   542→  return { tenkan, kijun, senkouA, senkouB };

   558→   558→function emaArrayCalc(closes: number[], period: number): number[] {
   559→   559→  const result: number[] = [];
   560→   560→  if (closes.length < period) return closes.map(() => 0);
   561→   561→  const k = 2 / (period + 1);
   562→   562→  let emaVal = 0;
   563→   563→  for (let i = 0; i < period; i++) emaVal += closes[i];
   564→   564→  emaVal /= period;
   565→   565→  for (let i = 0; i < closes.length; i++) {
   566→   566→    if (i < period) { result.push(0); continue; }
   567→   567→    if (i === period) { result.push(emaVal); continue; }
   568→   568→    emaVal = closes[i] * k + emaVal * (1 - k);
   569→   569→    result.push(emaVal);
   570→   570→  }
   571→   571→  return result;
   572→   572→}

   574→   574→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   575→   575→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   576→   576→  const n = values.length;
   577→   577→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   578→   578→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   579→   579→  for (let i = 0; i < n; i++) {
   580→   580→    sumX += i;
   581→   581→    sumY += values[i];
   582→   582→    sumXY += i * values[i];
   583→   583→    sumX2 += i * i;
   584→   584→    sumY2 += values[i] * values[i];
   585→   585→  }
   586→   586→  const denom = n * sumX2 - sumX * sumX;
   587→   587→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   588→   588→  const slope = (n * sumXY - sumX * sumY) / denom;
   589→   589→  const intercept = (sumY - slope * sumX) / n;
   590→   590→  const meanY = sumY / n;
   591→   591→  let ssRes = 0, ssTot = 0;
   592→   592→  for (let i = 0; i < n; i++) {
   593→   593→    const predicted = slope * i + intercept;
   594→   594→    ssRes += (values[i] - predicted) ** 2;
   595→   595→    ssTot += (values[i] - meanY) ** 2;
   596→   596→  }
   597→   597→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   598→   598→  return { slope, intercept, r2 };
   599→   599→}

   694→   694→// ─── Helper: Linear System Solver (Gaussian Elimination) ─────────────────
   695→   695→function solveLinearSystem(A: number[][], b: number[]): number[] | null {
   696→   696→  const n = b.length;
   697→   697→  const aug = A.map((row, i) => [...row, b[i]]);
   698→   698→
   699→   699→  for (let col = 0; col < n; col++) {
   700→   700→    let maxVal = Math.abs(aug[col][col]);
   701→   701→    let maxRow = col;
   702→   702→    for (let row = col + 1; row < n; row++) {
   703→   703→      if (Math.abs(aug[row][col]) > maxVal) {
   704→   704→        maxVal = Math.abs(aug[row][col]);
   705→   705→        maxRow = row;
   706→   706→      }
   707→   707→    }
   708→   708→    if (maxVal < 1e-10) return null;
   709→   709→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   710→   710→
   711→   711→    for (let row = col + 1; row < n; row++) {
   712→   712→      const factor = aug[row][col] / aug[col][col];
   713→   713→      for (let j = col; j <= n; j++) {
   714→   714→        aug[row][j] -= factor * aug[col][j];
   715→   715→      }
   716→   716→    }
   717→   717→  }
   718→   718→
   719→   719→  const x = new Array(n).fill(0);
   720→   720→  for (let i = n - 1; i >= 0; i--) {
   721→   721→    x[i] = aug[i][n];
   722→   722→    for (let j = i + 1; j < n; j++) {
   723→   723→      x[i] -= aug[i][j] * x[j];
   724→   724→    }
   725→   725→    if (Math.abs(aug[i][i]) < 1e-10) return null;
   726→   726→    x[i] /= aug[i][i];
   727→   727→  }
   728→   728→
   729→   729→  return x;
   730→   730→}

   732→   732→// ─── Helper: Generate All 7-Method S/R Levels ─────────────────────────────
   733→   733→function generateAllLevels(data: OHLCV[], endIdx: number): RawSRLevel[] {
   734→   734→  const levels: RawSRLevel[] = [];
   735→   735→  if (endIdx < 10) return levels;
   736→   736→
   737→   737→  const slice = data.slice(0, endIdx + 1);
   738→   738→  const currentPrice = slice[slice.length - 1].close;
   739→   739→  const closes = slice.map(d => d.close);
   740→   740→  const noFib: [string, string] = ['—', ''];
   741→   741→
   742→   742→  // ── Method 1: Swing High/Low (Price Action) ──────────────────────────
   743→   743→  for (const lookback of [3, 5, 7]) {
   744→   744→    const swings = findSwingLevels(slice, lookback);
   745→   745→    const tol = currentPrice * 0.005; // 0.5% tolerance
   746→   746→
   747→   747→    // Connect swing highs → resistance
   748→   748→    const usedHigh = new Set<number>();
   749→   749→    for (const h of swings.highs) {
   750→   750→      if (usedHigh.has(h.idx)) continue;
   751→   751→      let sum = h.price;
   752→   752→      let count = 1;
   753→   753→      for (const h2 of swings.highs) {
   754→   754→        if (h2 === h || usedHigh.has(h2.idx)) continue;
   755→   755→        if (Math.abs(h2.price - h.price) < tol) {
   756→   756→          sum += h2.price;
   757→   757→          count++;
   758→   758→          usedHigh.add(h2.idx);
   759→   759→        }
   760→   760→      }
   761→   761→      usedHigh.add(h.idx);
   762→   762→      if (sum / count > currentPrice * 1.005) {
   763→   763→        levels.push({ price: sum / count, method: 'Swing_High', type: 'resistance', ...noFib });
   764→   764→      }
   765→   765→    }
   766→   766→
   767→   767→    // Connect swing lows → support
   768→   768→    const usedLow = new Set<number>();
   769→   769→    for (const l of swings.lows) {
   770→   770→      if (usedLow.has(l.idx)) continue;
   771→   771→      let sum = l.price;
   772→   772→      let count = 1;
   773→   773→      for (const l2 of swings.lows) {
   774→   774→        if (l2 === l || usedLow.has(l2.idx)) continue;
   775→   775→        if (Math.abs(l2.price - l.price) < tol) {
   776→   776→          sum += l2.price;
   777→   777→          count++;
   778→   778→          usedLow.add(l2.idx);
   779→   779→        }
   780→   780→      }
   781→   781→      usedLow.add(l.idx);
   782→   782→      if (sum / count < currentPrice * 0.995) {
   783→   783→        levels.push({ price: sum / count, method: 'Swing_Low', type: 'support', ...noFib });
   784→   784→      }
   785→   785→    }
   786→   786→  }
   787→   787→
   788→   788→  // ── Method 2: SMA Dynamic S/R ──────────────────────────────────────
   789→   789→  for (const period of [20, 50, 100, 200]) {
   790→   790→    const smaVal = sma(closes, period);
   791→   791→    if (smaVal <= 0) continue;
   792→   792→    const type: 'support' | 'resistance' = currentPrice > smaVal ? 'support' : 'resistance';
   793→   793→    levels.push({ price: smaVal, method: `SMA_${period}`, type, ...noFib });
   794→   794→  }
   795→   795→
   796→   796→  // ── Method 3: Bollinger Bands ───────────────────────────────────────
   797→   797→  const bb = calcBollingerBands(closes);
   798→   798→  if (bb.upper > 0 && bb.lower > 0) {
   799→   799→    levels.push({ price: bb.upper, method: 'BB_Upper', type: 'resistance', ...noFib });
   800→   800→    levels.push({ price: bb.lower, method: 'BB_Lower', type: 'support', ...noFib });
   801→   801→  }
   802→   802→
   803→   803→  // ── Method 4: Fibonacci Retracement + Extension ─────────────────────
   804→   804→  interface SwingPair { high: number; highIdx: number; low: number; lowIdx: number; range: number }
   805→   805→  let bestSwing: SwingPair | null = null;
   806→   806→
   807→   807→  for (const lb of [3, 5, 7, 10, 15, 20]) {
   808→   808→    const sw = findSwingLevels(slice, lb);
   809→   809→    if (sw.highs.length === 0 || sw.lows.length === 0) continue;
   810→   810→    for (const h of sw.highs) {
   811→   811→      for (const l of sw.lows) {
   812→   812→        if (Math.abs(h.idx - l.idx) < 3) continue;
   813→   813→        const range = Math.abs(h.price - l.price);
   814→   814→        if (range / currentPrice < 0.03) continue;
   815→   815→        if (!bestSwing || range > bestSwing.range) {
   816→   816→          bestSwing = {
   817→   817→            high: Math.max(h.price, l.price),
   818→   818→            highIdx: h.price > l.price ? h.idx : l.idx,
   819→   819→            low: Math.min(h.price, l.price),
   820→   820→            lowIdx: h.price < l.price ? l.idx : h.idx,
   821→   821→            range,
   822→   822→          };
   823→   823→        }
   824→   824→      }
   825→   825→    }
   826→   826→  }
   827→   827→
   828→   828→  if (bestSwing) {
   829→   829→    const H = bestSwing.high;
   830→   830→    const L = bestSwing.low;
   831→   831→    const range = H - L;
   832→   832→
   833→   833→    for (const ratio of FIB_RETRACEMENTS) {
   834→   834→      const price = H - ratio * range;
   835→   835→      if (price <= 0) continue;
   836→   836→      const rounded = roundToNice(price);
   837→   837→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   838→   838→      levels.push({
   839→   839→        price: rounded, method: `Fibonacci_${ratio}`, type,
   840→   840→        fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   841→   841→      });
   842→   842→    }
   843→   843→
   844→   844→    for (const extRatio of FIB_EXTENSIONS) {
   845→   845→      const price = H + (extRatio - 1) * range;
   846→   846→      if (price <= 0) continue;
   847→   847→      const rounded = roundToNice(price);
   848→   848→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   849→   849→      levels.push({
   850→   850→        price: rounded, method: `Fibonacci_${extRatio}`, type,
   851→   851→        fibRatio: String(extRatio), fibLabel: FIB_LABELS[String(extRatio)] || '',
   852→   852→      });
   853→   853→    }
   854→   854→
   855→   855→    // Inverse extensions (below L)
   856→   856→    for (const extRatio of FIB_EXTENSIONS) {
   857→   857→      const price = L - (extRatio - 1) * range;
   858→   858→      if (price <= 0) continue;
   859→   859→      const rounded = roundToNice(price);
   860→   860→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   861→   861→      levels.push({
   862→   862→        price: rounded, method: `Fibonacci_${-extRatio}`, type,
   863→   863→        fibRatio: String(-extRatio), fibLabel: '',
   864→   864→      });
   865→   865→    }
   866→   866→
   867→   867→    // Secondary swing confluence (check overlap with secondary swings)
   868→   868→    const secondarySwings: SwingPair[] = [];
   869→   869→    for (const lb of [3, 5, 7]) {
   870→   870→      const sw = findSwingLevels(slice, lb);
   871→   871→      for (const h of sw.highs) {
   872→   872→        for (const l of sw.lows) {
   873→   873→          if (Math.abs(h.idx - l.idx) < 3) continue;
   874→   874→          const r = Math.abs(h.price - l.price);
   875→   875→          if (r / currentPrice < 0.05) continue;
   876→   876→          if (bestSwing && Math.abs(r - bestSwing.range) / bestSwing.range < 0.1) continue;
   877→   877→          secondarySwings.push({
   878→   878→            high: Math.max(h.price, l.price), highIdx: h.price > l.price ? h.idx : l.idx,
   879→   879→            low: Math.min(h.price, l.price), lowIdx: h.price < l.price ? l.idx : h.idx,
   880→   880→            range: r,
   881→   881→          });
   882→   882→        }
   883→   883→      }
   884→   884→    }
   885→   885→    // Secondary swings' fib levels are merged via confluence (1% rule in mergeLevels)
   886→   886→    const confluenceTol = currentPrice * 0.005;
   887→   887→    for (const sec of secondarySwings.slice(0, 3)) {
   888→   888→      const sH = sec.high, sL = sec.low, sR = sH - sL;
   889→   889→      for (const ratio of FIB_RETRACEMENTS) {
   890→   890→        const price = roundToNice(sH - ratio * sR);
   891→   891→        if (price <= 0) continue;
   892→   892→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   893→   893→        levels.push({
   894→   894→          price, method: `Fibonacci_${ratio}`, type,
   895→   895→          fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   896→   896→        });
   897→   897→      }
   898→   898→      for (const extR of FIB_EXTENSIONS) {
   899→   899→        const price = roundToNice(sH + (extR - 1) * sR);
   900→   900→        if (price <= 0) continue;
   901→   901→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   902→   902→        levels.push({
   903→   903→          price, method: `Fibonacci_${extR}`, type,
   904→   904→          fibRatio: String(extR), fibLabel: FIB_LABELS[String(extR)] || '',
   905→   905→        });
   906→   906→      }
   907→   907→    }
   908→   908→  }
   909→   909→
   910→   910→  // ── Method 5: Volume Profile (VAP) ──────────────────────────────────
   911→   911→  const dataHasVolume = slice.some(d => d.volume > 0);
   912→   912→  if (dataHasVolume) {
   913→   913→    let minP = Infinity, maxP = -Infinity;
   914→   914→    for (const d of slice) {
   915→   915→      if (d.low < minP) minP = d.low;
   916→   916→      if (d.high > maxP) maxP = d.high;
   917→   917→    }
   918→   918→    const numBins = 40;
   919→   919→    const binSize = Math.max((maxP - minP) / numBins, currentPrice * 0.001);
   920→   920→    const vap = new Array(numBins).fill(0);
   921→   921→    const binCenters: number[] = [];
   922→   922→
   923→   923→    for (let b = 0; b < numBins; b++) {
   924→   924→      binCenters.push(minP + (b + 0.5) * binSize);
   925→   925→    }
   926→   926→
   927→   927→    for (const d of slice) {
   928→   928→      for (let b = 0; b < numBins; b++) {
   929→   929→        if (d.low <= binCenters[b] && binCenters[b] <= d.high) {
   930→   930→          vap[b] += d.volume;
   931→   931→        }
   932→   932→      }
   933→   933→    }
   934→   934→
   935→   935→    const maxVAP = Math.max(...vap);
   936→   936→    if (maxVAP > 0) {
   937→   937→      // POC (Point of Control)
   938→   938→      let pocIdx = 0;
   939→   939→      for (let b = 1; b < numBins; b++) {
   940→   940→        if (vap[b] > vap[pocIdx]) pocIdx = b;
   941→   941→      }
   942→   942→      const pocPrice = binCenters[pocIdx];
   943→   943→      const pocType: 'support' | 'resistance' = pocPrice > currentPrice ? 'resistance' : 'support';
   944→   944→      levels.push({ price: pocPrice, method: 'VAP_POC', type: pocType, ...noFib });
   945→   945→
   946→   946→      // HVN (High Volume Nodes)
   947→   947→      for (let b = 0; b < numBins; b++) {
   948→   948→        if (b === pocIdx) continue;
   949→   949→        if (vap[b] > 0.7 * maxVAP) {
   950→   950→          const hvnPrice = binCenters[b];
   951→   951→          const hvnType: 'support' | 'resistance' = hvnPrice > currentPrice ? 'resistance' : 'support';
   952→   952→          levels.push({ price: hvnPrice, method: 'VAP_HVN', type: hvnType, ...noFib });
   953→   953→        }
   954→   954→      }
   955→   955→    }
   956→   956→  }
   957→   957→
   958→   958→  // ── Method 6: Pivot Points (4 algorithms) ───────────────────────────
   959→   959→  if (endIdx >= 1) {
   960→   960→    const prev = slice[endIdx - 1]; // Previous completed bar
   961→   961→    const pH = prev.high, pL = prev.low, pC = prev.close, pO = prev.open;
   962→   962→
   963→   963→    // Standard Pivot
   964→   964→    {
   965→   965→      const PP = (pH + pL + pC) / 3;
   966→   966→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   967→   967→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   968→   968→      const R3 = pH + 2 * (PP - pL), S3 = pL - 2 * (pH - PP);
   969→   969→ if (R1 > 0) levels.push({ price: R1, method: 'Pivot_R1', type: 'resistance', ...noFib });
   970→   970→      if (R2 > 0) levels.push({ price: R2, method: 'Pivot_R2', type: 'resistance', ...noFib });
   971→   971→      if (R3 > 0) levels.push({ price: R3, method: 'Pivot_R3', type: 'resistance', ...noFib });
   972→   972→      if (S1 > 0) levels.push({ price: S1, method: 'Pivot_S1', type: 'support', ...noFib });
   973→   973→      if (S2 > 0) levels.push({ price: S2, method: 'Pivot_S2', type: 'support', ...noFib });
   974→   974→      if (S3 > 0) levels.push({ price: S3, method: 'Pivot_S3', type: 'support', ...noFib });
   975→   975→    }
   976→   976→
   977→   977→    // Fibonacci Pivot
   978→   978→    {
   979→   979→      const PP = (pH + pL + pC) / 3;
   980→   980→      const range = pH - pL;
   981→   981→      for (const [mult, suffix] of [[0.382, '1'], [0.618, '2'], [1.0, '3']] as [number, string][]) {
   982→   982→        const R = PP + mult * range;
   983→   983→        const S = PP - mult * range;
   984→   984→        if (R > 0) levels.push({ price: R, method: `Fib_Pivot_R${suffix}`, type: 'resistance', ...noFib });
   985→   985→        if (S > 0) levels.push({ price: S, method: `Fib_Pivot_S${suffix}`, type: 'support', ...noFib });
   986→   986→      }
   987→   987→    }
   988→   988→
   989→   989→    // Woodie's Pivot
   990→   990→    {
   991→   991→      const PP = (pH + pL + 2 * pC) / 4;
   992→   992→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   993→   993→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   994→   994→      if (R1 > 0) levels.push({ price: R1, method: 'Woodie_R1', type: 'resistance', ...noFib });
   995→   995→      if (R2 > 0) levels.push({ price: R2, method: 'Woodie_R2', type: 'resistance', ...noFib });
   996→   996→      if (S1 > 0) levels.push({ price: S1, method: 'Woodie_S1', type: 'support', ...noFib });
   997→   997→      if (S2 > 0) levels.push({ price: S2, method: 'Woodie_S2', type: 'support', ...noFib });
   998→   998→    }
   999→   999→
  1000→  1000→    // DeMark Pivot
  1001→  1001→    {
  1002→  1002→      const X = pC < pO ? pH + 2 * pL + pC : pC > pO ? 2 * pH + pL + pC : pH + pL + 2 * pC;
  1003→  1003→      const PP = X / 4;
  1004→  1004→      const R1 = X / 2 - pL;
  1005→  1005→      const S1 = X / 2 - pH;
  1006→  1006→      if (R1 > 0) levels.push({ price: R1, method: 'DeMark_R1', type: 'resistance', ...noFib });
  1007→  1007→      if (S1 > 0) levels.push({ price: S1, method: 'DeMark_S1', type: 'support', ...noFib });
  1008→  1008→    }
  1009→  1009→  }
  1010→  1010→
  1011→  1011→  // ── Method 7: Psychological Levels ──────────────────────────────────
  1012→  1012→  {
  1013→  1013→    const step = psychStep(currentPrice);
  1014→  1014→    const base = psychRound(currentPrice);
  1015→  1015→    for (const mult of [-2, -1, 1, 2]) {
  1016→  1016→      const price = base + mult * step;
  1017→  1017→      if (price <= 0) continue;
  1018→  1018→      const type: 'support' | 'resistance' = price > currentPrice ? 'resistance' : 'support';
  1019→  1019→      levels.push({ price, method: 'Psychological', type, ...noFib });
  1020→  1020→    }
  1021→  1021→  }
  1022→  1022→
  1023→  1023→  return levels;
  1024→  1024→}

  1026→  1026→// ─── Helper: Merge Nearby Levels (1% Confluence) ──────────────────────────
  1027→  1027→function mergeLevels(rawLevels: RawSRLevel[], currentPrice: number): MergedSRLevel[] {
  1028→  1028→  if (rawLevels.length === 0) return [];
  1029→  1029→
  1030→  1030→  const tol = currentPrice * 0.01; // 1% tolerance
  1031→  1031→  const merged: MergedSRLevel[] = [];
  1032→  1032→  const used = new Set<number>();
  1033→  1033→
  1034→  1034→  for (let i = 0; i < rawLevels.length; i++) {
  1035→  1035→    if (used.has(i)) continue;
  1036→  1036→
  1037→  1037→    let sumPrice = rawLevels[i].price;
  1038→  1038→    let count = 1;
  1039→  1039→    const methods = new Set<string>();
  1040→  1040→    methods.add(rawLevels[i].method);
  1041→  1041→    let bestFibRatio = rawLevels[i].fibRatio;
  1042→  1042→    let bestFibLabel = rawLevels[i].fibLabel;
  1043→  1043→
  1044→  1044→    for (let j = i + 1; j < rawLevels.length; j++) {
  1045→  1045→      if (used.has(j)) continue;
  1046→  1046→      if (Math.abs(rawLevels[j].price - rawLevels[i].price) < tol) {
  1047→  1047→        sumPrice += rawLevels[j].price;
  1048→  1048→        count++;
  1049→  1049→        methods.add(rawLevels[j].method);
  1050→  1050→        used.add(j);
  1051→  1051→        // Prefer Fibonacci label if available
  1052→  1052→        if (rawLevels[j].fibRatio !== '—' && bestFibRatio === '—') {
  1053→  1053→          bestFibRatio = rawLevels[j].fibRatio;
  1054→  1054→          bestFibLabel = rawLevels[j].fibLabel;
  1055→  1055→        }
  1056→  1056→      }
  1057→  1057→    }
  1058→  1058→    used.add(i);
  1059→  1059→
  1060→  1060→    const avgPrice = sumPrice / count;
  1061→  1061→    const type: 'support' | 'resistance' = avgPrice >= currentPrice ? 'resistance' : 'support';
  1062→  1062→
  1063→  1063→    merged.push({
  1064→  1064→      price: avgPrice,
  1065→  1065→      type,
  1066→  1066→      methods: [...methods],
  1067→  1067→      overlapCount: methods.size,
  1068→  1068→      fibRatio: bestFibRatio,
  1069→  1069→      fibLabel: bestFibLabel,
  1070→  1070→      score: 0,
  1071→  1071→      strength: 1,
  1072→  1072→      touchCount: 0,
  1073→  1073→      volumeRatio: 0,
  1074→  1074→      freshness: 0,
  1075→  1075→      distancePercent: 0,
  1076→  1076→    });
  1077→  1077→  }
  1078→  1078→
  1079→  1079→  return merged;
  1080→  1080→}

  1082→  1082→// ─── Helper: Compute Level Features (5 ML features) ──────────────────────
  1083→  1083→function computeLevelFeatures(
  1084→  1084→  levelPrice: number,
  1085→  1085→  data: OHLCV[],
  1086→  1086→  endIdx: number,
  1087→  1087→  currentPrice: number,
  1088→  1088→  overlapCount: number,
  1089→  1089→): number[] {
  1090→  1090→  const lookback = Math.min(30, endIdx);
  1091→  1091→  const startIdx = endIdx - lookback;
  1092→  1092→  const touchTol = currentPrice * 0.002; // 0.2%
  1093→  1093→
  1094→  1094→  let touchCount = 0;
  1095→  1095→  let totalVolAtTouches = 0;
  1096→  1096→  let totalVol = 0;
  1097→  1097→  let barsSinceLastTouch = lookback;
  1098→  1098→
  1099→  1099→  for (let i = startIdx; i <= endIdx; i++) {
  1100→  1100→    const bar = data[i];
  1101→  1101→    totalVol += bar.volume;
  1102→  1102→
  1103→  1103→    if (bar.low <= levelPrice + touchTol && bar.high >= levelPrice - touchTol) {
  1104→  1104→      touchCount++;
  1105→  1105→      totalVolAtTouches += bar.volume;
  1106→  1106→      barsSinceLastTouch = endIdx - i;
  1107→  1107→    }
  1108→  1108→  }
  1109→  1109→
  1110→  1110→  const avgVol = lookback > 0 ? totalVol / lookback : 0;
  1111→  1111→  const volumeRatio = (touchCount > 0 && avgVol > 0) ? (totalVolAtTouches / touchCount) / avgVol : 1;
  1112→  1112→  const freshness = Math.max(0, Math.min(1, 1 - barsSinceLastTouch / 30));
  1113→  1113→  const distancePercent = Math.min(1, Math.abs(levelPrice - currentPrice) / currentPrice);
  1114→  1114→
  1115→  1115→  // Features: [touch_count, volume_ratio, overlap_count, freshness, distance_percent]
  1116→  1116→  return [touchCount, volumeRatio, overlapCount - 1, freshness, distancePercent];
  1117→  1117→}

  1119→  1119→// ─── Helper: Train S/R Scoring Weights (Ridge OLS) ───────────────────────
  1120→  1120→function trainSRWeights(data: OHLCV[]): number[] {
  1121→  1121→  const defaultWeights = [0.25, 0.20, 0.25, 0.15, 0.15];
  1122→  1122→
  1123→  1123→  if (data.length < 50) return defaultWeights;
  1124→  1124→
  1125→  1125→  const X: number[][] = [];
  1126→  1126→  const y: number[] = [];
  1127→  1127→
  1128→  1128→  for (let i = 40; i <= data.length - 11; i += 5) {
  1129→  1129→    const barPrice = data[i].close;
  1130→  1130→    const rawLevels = generateAllLevels(data, i);
  1131→  1131→    if (rawLevels.length === 0) continue;
  1132→  1132→    const merged = mergeLevels(rawLevels, barPrice);
  1133→  1133→
  1134→  1134→    for (const level of merged) {
  1135→  1135→      const features = computeLevelFeatures(level.price, data, i, barPrice, level.overlapCount);
  1136→  1136→      X.push(features);
  1137→  1137→
  1138→  1138→      // Label: did price return within 10 bars?
  1139→  1139→      let returned = false;
  1140→  1140→      const retTol = barPrice * 0.005;
  1141→  1141→      for (let j = i + 1; j <= Math.min(i + 10, data.length - 1); j++) {
  1142→  1142→        if (data[j].low <= level.price + retTol && data[j].high >= level.price - retTol) {
  1143→  1143→          returned = true;
  1144→  1144→          break;
  1145→  1145→        }
  1146→  1146→      }
  1147→  1147→      y.push(returned ? 1 : 0);
  1148→  1148→    }
  1149→  1149→  }
  1150→  1150→
  1151→  1151→  if (X.length < 20) return defaultWeights;
  1152→  1152→
  1153→  1153→  // Ridge OLS: w = (X^T X + λI)^(-1) X^T y
  1154→  1154→  const n = X.length;
  1155→  1155→  const p = 5;
  1156→  1156→  const lambda = 0.01;
  1157→  1157→
  1158→  1158→  const XtX: number[][] = Array.from({ length: p }, () => new Array(p).fill(0));
  1159→  1159→  for (let i = 0; i < n; i++) {
  1160→  1160→    for (let a = 0; a < p; a++) {
  1161→  1161→      for (let b = 0; b < p; b++) {
  1162→  1162→        XtX[a][b] += X[i][a] * X[i][b];
  1163→  1163→      }
  1164→  1164→    }
  1165→  1165→  }
  1166→  1166→  for (let a = 0; a < p; a++) XtX[a][a] += lambda;
  1167→  1167→
  1168→  1168→  const Xty: number[] = new Array(p).fill(0);
  1169→  1169→  for (let i = 0; i < n; i++) {
  1170→  1170→    for (let a = 0; a < p; a++) {
  1171→  1171→      Xty[a] += X[i][a] * y[i];
  1172→  1172→    }
  1173→  1173→  }
  1174→  1174→
  1175→  1175→  const weights = solveLinearSystem(XtX, Xty);
  1176→  1176→  if (!weights) return defaultWeights;
  1177→  1177→  if (weights.some(w => isNaN(w) || !isFinite(w))) return defaultWeights;
  1178→  1178→
  1179→  1179→  for (let i = 0; i < p; i++) {
  1180→  1180→    if (weights[i] < 0) weights[i] = 0;
  1181→  1181→  }
  1182→  1182→
  1183→  1183→  const sum = weights.reduce((a, b) => a + b, 0);
  1184→  1184→  if (sum <= 0) return defaultWeights;
  1185→  1185→
  1186→  1186→  return weights.map(w => w / sum);
  1187→  1187→}

  1189→  1189→// ─── Helper: WMA Array (full series) ─────────────────────────────────────
  1190→  1190→function calcWMAArray(closes: number[], period: number): number[] {
  1191→  1191→  const result: number[] = [];
  1192→  1192→  if (closes.length < period) return closes.map(() => 0);
  1193→  1193→  for (let i = 0; i < closes.length; i++) {
  1194→  1194→    if (i < period - 1) { result.push(0); continue; }
  1195→  1195→    let numer = 0;
  1196→  1196→    let denom = 0;
  1197→  1197→    for (let j = 0; j < period; j++) {
  1198→  1198→      const weight = period - j;
  1199→  1199→      numer += closes[i - period + 1 + j] * weight;
  1200→  1200→      denom += weight;
  1201→  1201→    }
  1202→  1202→    result.push(numer / denom);
  1203→  1203→  }
  1204→  1204→  return result;
  1205→  1205→}

  1207→  1207→// ─── Helper: WMA (latest value) ────────────────────────────────────
  1208→  1208→function calcWMA(closes: number[], period: number): number {
  1209→  1209→  if (closes.length < period) return 0;
  1210→  1210→  const slice = closes.slice(-period);
  1211→  1211→  let numer = 0;

  1578→  1578→    const sorted = [...levels].sort((a, b) =>
  1579→  1579→      type === 'support' ? b.price - a.price : a.price - b.price
  1580→  1580→    );
  1581→  1581→
  1582→  1582→    // ── Rule 2/3: Nearest level must be within 10% of price ──
  1583→  1583→    const nearest = sorted[0];
  1584→  1584→    const distFromPrice = Math.abs(nearest.price - currentPrice) / currentPrice;
  1585→  1585→    if (distFromPrice > 0.10) {
  1586→  1586→      // Nearest level is too far — generate a synthetic level at ~5% from price
  1587→  1587→      const syntheticPrice = type === 'support'
  1588→  1588→        ? Math.round(currentPrice * 0.95 / psychStep(currentPrice)) * psychStep(currentPrice)
  1589→  1589→        : Math.round(currentPrice * 1.05 / psychStep(currentPrice)) * psychStep(currentPrice);
  1590→  1590→      const synthetic: MergedSRLevel = {
  1591→  1591→        price: syntheticPrice,
  1592→  1592→        type,
  1593→  1593→        score: 3,
  1594→  1594→        strength: 3,
  1595→  1595→        overlapCount: 1,
  1596→  1596→        methods: ['Psychological'],
  1597→  1597→        fibRatio: '—',
  1598→  1598→        fibLabel: '',
  1599→  1599→      };
  1600→  1600→      sorted.unshift(synthetic);
  1601→  1601→    }
  1602→  1602→
  1603→  1603→    // ── Rule 1: Select levels with 5-10% gaps between consecutive ──
  1604→  1604→    const result: MergedSRLevel[] = [sorted[0]];
  1605→  1605→
  1606→  1606→    for (let i = 1; i < sorted.length && result.length < 6; i++) {
  1607→  1607→      const prevPrice = result[result.length - 1].price;
  1608→  1608→      const currPrice = sorted[i].price;
  1609→  1609→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
  1610→  1610→
  1611→  1611→      if (gap < 0.05) {
  1612→  1612→        // Too close — keep the stronger one
  1613→  1613→        if (sorted[i].score > result[result.length - 1].score) {
  1614→  1614→          result[result.length - 1] = sorted[i];
  1615→  1615→        }
  1616→  1616→      } else if (gap <= 0.10) {
  1617→  1617→        // Perfect gap (5-10%) — accept
  1618→  1618→        result.push(sorted[i]);
  1619→  1619→      } else {
  1620→  1620→        // Gap > 10% — try to find an intermediate level
  1621→  1621→        let found = false;
  1622→  1622→        for (let j = i + 1; j < sorted.length; j++) {
  1623→  1623→          const midPrice = sorted[j].price;
  1624→  1624→          const gapFromPrev = Math.abs(midPrice - prevPrice) / prevPrice;
  1625→  1625→          const gapToNext = Math.abs(currPrice - midPrice) / midPrice;
  1626→  1626→          if (gapFromPrev >= 0.05 && gapFromPrev <= 0.10) {
  1627→  1627→            result.push(sorted[j]);
  1628→  1628→            found = true;
  1629→  1629→            // Skip levels that are too close to the inserted one
  1630→  1630→            i = j;
  1631→  1631→            break;
  1632→  1632→          }
  1633→  1633→        }
  1634→  1634→        if (!found) {
  1635→  1635→          // No intermediate found — generate a synthetic level at ~7.5% from prev
  1636→  1636→          const synthGap = 0.075;
  1637→  1637→          const syntheticPrice = type === 'support'
  1638→  1638→            ? Math.round((prevPrice * (1 - synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice)
  1639→  1639→            : Math.round((prevPrice * (1 + synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice);
  1640→  1640→          // Make sure it's not too close to next level
  1641→  1641→          const gapToNext = Math.abs(currPrice - syntheticPrice) / syntheticPrice;
  1642→  1642→          if (gapToNext >= 0.05) {
  1643→  1643→            const synthetic: MergedSRLevel = {
  1644→  1644→              price: syntheticPrice,
  1645→  1645→              type,
  1646→  1646→              score: 2,
  1647→  1647→              strength: 2,
  1648→  1648→              overlapCount: 1,
  1649→  1649→              methods: ['Psychological'],
  1650→  1650→              fibRatio: '—',
  1651→  1651→              fibLabel: '',
  1652→  1652→            };
  1653→  1653→            result.push(synthetic);
  1654→  1654→          }
  1655→  1655→        }
  1656→  1656→      }
  1657→  1657→    }
  1658→  1658→
  1659→  1659→    return result;
  1660→  1660→  };
  1661→  1661→
  1662→  1662→  const finalSupports = filterAndSelectLevels(supports, 'support');
  1663→  1663→  const finalResistances = filterAndSelectLevels(resistances, 'resistance');
  1664→  1664→
  1665→  1665→  // ── Step 9: Build LevelStrength objects ────────────────────────────────
  1666→  1666→  const toLevelStrength = (m: MergedSRLevel): LevelStrength => {
  1667→  1667→    let grade: string;
  1668→  1668→    if (m.score >= 8.5) grade = 'Very Strong';
  1669→  1669→    else if (m.score >= 7) grade = 'Strong';
  1670→  1670→    else if (m.score >= 5) grade = 'Moderate';
  1671→  1671→    else grade = 'Weak';
  1672→  1672→
  1673→  1673→    return {
  1674→  1674→      price: m.price,
  1675→  1675→      strength: m.strength,
  1676→  1676→      score: m.score,
  1677→  1677→      grade,
  1678→  1678→      isTarget: m.score >= 7,
  1679→  1679→      overlapCount: m.overlapCount,
  1680→  1680→      methods: m.methods,
  1681→  1681→      fibRatio: m.fibRatio,
  1682→  1682→      fibLabel: m.fibLabel,
  1683→  1683→    };
  1684→  1684→  };
  1685→  1685→
  1686→  1686→  const supportStrengths = finalSupports.map(toLevelStrength);
  1687→  1687→  const resistanceStrengths = finalResistances.map(toLevelStrength);
  1688→  1688→
  1689→  1689→  // Price targets: all levels with isTarget=true
  1690→  1690→  const priceTargets = [...supportStrengths, ...resistanceStrengths].filter(l => l.isTarget);
  1691→  1691→
  1692→  1692→  return {
  1693→  1693→    resistances: resistanceStrengths.map(l => l.price),
  1694→  1694→    supports: supportStrengths.map(l => l.price),
  1695→  1695→    supportStrengths,
  1696→  1696→    resistanceStrengths,
  1697→  1697→    priceTargets,
  1698→  1698→  };
  1699→  1699→}

  1703→  1703→function calcTrend(closes: number[], period: number): TrendResult {
  1704→  1704→  const slice = closes.slice(-Math.min(period, closes.length));
  1705→  1705→  const { slope, r2 } = linearRegression(slice);
  1706→  1706→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
  1707→  1707→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
  1708→  1708→  const priceThreshold = avgPrice * 0.001; // 0.1%
  1709→  1709→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
  1710→  1710→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
  1711→  1711→}

  1718→  1718→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
  1719→  1719→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
  1720→  1720→  const slice = data.slice(0, barIdx + 1);
  1721→  1721→  const price = slice[slice.length - 1].close;
  1722→  1722→  let nearestR = price * 1.05;
  1723→  1723→  let nearestS = price * 0.95;
  1724→  1724→
  1725→  1725→  // Collect all swing highs (resistance) and swing lows (support)
  1726→  1726→  const resistances: number[] = [];
  1727→  1727→  const supports: number[] = [];
  1728→  1728→  for (const lb of [3, 5, 7]) {
  1729→  1729→    const swings = findSwingLevels(slice, lb);
  1730→  1730→    resistances.push(...swings.highs);
  1731→  1731→    supports.push(...swings.lows);
  1732→  1732→  }
  1733→  1733→
  1734→  1734→  // Find nearest resistance above price
  1735→  1735→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
  1736→  1736→  if (abovePrice.length > 0) nearestR = abovePrice[0];
  1737→  1737→
  1738→  1738→  // Find nearest support below price
  1739→  1739→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
  1740→  1740→  if (belowPrice.length > 0) nearestS = belowPrice[0];
  1741→  1741→
  1742→  1742→  return { R1: nearestR, S1: nearestS };
  1743→  1743→}

  1745→  1745→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
  1746→  1746→// Returns true if bullish divergence detected, false if bearish, null if none
  1747→  1747→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
  1748→  1748→  const lookback = 20;
  1749→  1749→  const start = Math.max(0, barIdx - lookback);
  1750→  1750→  const slice = data.slice(start, barIdx + 1);
  1751→  1751→  if (slice.length < 10) return null;
  1752→  1752→
  1753→  1753→  const mid = Math.floor(slice.length / 2);
  1754→  1754→  const firstHalf = slice.slice(0, mid);
  1755→  1755→  const secondHalf = slice.slice(mid);
  1756→  1756→
  1757→  1757→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
  1758→  1758→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
  1759→  1759→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
  1760→  1760→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
  1761→  1761→
  1762→  1762→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
  1763→  1763→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
  1764→  1764→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
  1765→  1765→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
  1766→  1766→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
  1767→  1767→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
  1768→  1768→
  1769→  1769→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
  1770→  1770→    return false; // bearish divergence
  1771→  1771→  }
  1772→  1772→
  1773→  1773→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
  1774→  1774→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
  1775→  1775→
  1776→  1776→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
  1777→  1777→    return true; // bullish divergence
  1778→  1778→  }
  1779→  1779→
  1780→  1780→  return null;
  1781→  1781→}
