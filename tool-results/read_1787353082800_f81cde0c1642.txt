   765→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   766→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   767→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   768→  grade?: string; overlapCount?: number; touchCount?: number;
   769→}) {
   770→  const isSupport = type === 'support';
   771→  const isTargetType = type === 'target';
   772→
   773→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   774→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   775→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   776→
   777→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   778→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   779→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   780→
   781→  return (
   782→    <div
   783→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   784→      style={{
   785→        background: bgAccent,
   786→        border: `1px solid ${borderAccent}`,
   787→        borderRight: `3px solid ${accentColor}`,
   788→      }}
   789→    >
   790→      {/* Label + Price */}
   791→      <div className="flex-1 min-w-0">
   792→        <div className="flex items-center gap-2">
   793→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   794→          {isTarget && <span className="text-[9px]">🎯</span>}
   795→          {overlapCount !== undefined && overlapCount > 0 && (
   796→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   797→              {toPersianDigits(overlapCount)} هم‌پوشانی
   798→            </span>
   799→          )}
   800→        </div>
   801→        <div className="flex items-baseline gap-1">
   802→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   803→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   804→        </div>
   805→      </div>
   806→
   807→      {/* Strength + Grade */}
   808→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   809→        <div className="flex items-center gap-1.5">
   810→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   811→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   812→          <span className="text-[9px] text-gray-500">/۱۰</span>
   813→        </div>
   814→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   815→          <div
   816→            className="h-full rounded-full transition-all duration-500"
   817→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   818→          />
   819→        </div>
   820→        {touchCount !== undefined && touchCount > 0 && (
   821→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   822→        )}
   823→      </div>
   824→    </div>
   825→  );
   826→}
