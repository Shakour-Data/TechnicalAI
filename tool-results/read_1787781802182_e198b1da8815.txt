    16→interface InstrumentItem {
    17→  l18: string;
    18→  l30: string;
    19→  pl: number;
    20→  pcp: number;
    21→  tno: number;
    22→  tvol: number;
    23→  tval: number;
    24→  cs: string;
    25→  category: 'stock' | 'etf' | 'bond' | 'future' | 'salaf' | 'mortgage' | 'index'
    26→    | 'currency' | 'gold' | 'silver' | 'gold_etf'
    27→    | 'crypto' | 'world_index' | 'foreign_stock' | 'forex' | 'energy' | 'metal' | 'commodity'
    28→    | 'yahoo_stock' | 'yahoo_etf';
    29→  insCode?: string;
    30→  tgjuKey?: string;
    31→  finpySector?: string;  // finpy-tse sector name for industry indices
    32→  finpyIndex?: string;   // finpy-tse index function key (CWI, EWI, etc.)
    33→  webId?: string | number;  // TSETMC web ID (string to preserve precision)
    34→  isMainIndex?: boolean; // True for main market indices
    35→  index?: number;
    36→  indexChange?: number;
    37→  indexChangePercent?: number;
    38→  indexMin?: number;
    39→  indexMax?: number;
    40→  yahooSymbol?: string;  // Yahoo Finance symbol for international instruments
    41→  yahooCategory?: string;  // Original Yahoo category (yahoo_stock, yahoo_index, etc.)
    42→  currency?: string;  // Currency unit from Yahoo Finance (USD, EUR, etc.)
    43→  unit?: string;  // Same as currency for convenience
    44→}

   174→function formatNum(num: number): string {
   175→  if (num == null) return '—';
   176→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 0 }).format(num);
   177→}

   179→function formatIdx(num: number): string {
   180→  if (num == null) return '—';
   181→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 2 }).format(num);
   182→}

   194→
   195→export default function SymbolSearch({
   196→  onSelect,
   197→  placeholder = 'جستجوی ابزار مالی ...',
   198→  className,
   199→}: SymbolSearchProps) {
   200→  const [query, setQuery] = React.useState('');
   201→  const [open, setOpen] = React.useState(false);
   202→  const [results, setResults] = React.useState<InstrumentItem[]>([]);
   203→  const [loading, setLoading] = React.useState(false);
   204→  const [activeIndex, setActiveIndex] = React.useState(-1);
   205→  const [totalMatched, setTotalMatched] = React.useState(0);
   206→  const [activeCategory, setActiveCategory] = React.useState<CategoryKey>('all');
   207→  const [activeIndustry, setActiveIndustry] = React.useState<string | null>(null);
   208→  const [showIndustryPicker, setShowIndustryPicker] = React.useState(false);
   209→
   210→  const cacheRef = React.useRef<InstrumentsData | null>(null);
   211→  const tgjuCacheRef = React.useRef<TgjuData | null>(null);
   212→  const yahooCacheRef = React.useRef<YahooData | null>(null);
   213→  const fetchRef = React.useRef<Promise<InstrumentsData> | null>(null);
   214→  const tgjuFetchRef = React.useRef<Promise<TgjuData | null> | null>(null);
   215→  const yahooFetchRef = React.useRef<Promise<YahooData | null> | null>(null);
   216→  const inputRef = React.useRef<HTMLInputElement>(null);
   217→  const dropdownRef = React.useRef<HTMLDivElement>(null);
   218→  const itemRefs = React.useRef<(HTMLDivElement | null)[]>([]);
   219→  const industryScrollRef = React.useRef<HTMLDivElement>(null);
   220→
   221→  /* -- Data fetching -- */
   222→  const fetchData = React.useCallback(async (): Promise<InstrumentsData> => {
   223→    if (cacheRef.current) return cacheRef.current;
   224→    if (fetchRef.current) return fetchRef.current;
   225→    const p = async () => {
   226→      try {
   227→        const r = await fetch('/api/instruments');
   228→        if (!r.ok) throw new Error(`${r.status}`);
   229→        const d: InstrumentsData = await r.json();
   230→        cacheRef.current = d;
   231→        return d;
   232→      } finally {
   233→        fetchRef.current = null;
   234→      }
   235→    };
   236→    fetchRef.current = p();
   237→    return fetchRef.current;
   238→  }, []);
   239→
   240→  const fetchTgjuData = React.useCallback(async (): Promise<TgjuData | null> => {
   241→    if (tgjuCacheRef.current) return tgjuCacheRef.current;
   242→    if (tgjuFetchRef.current) return tgjuFetchRef.current;
   243→    const p = async () => {
   244→      try {
   245→        const r = await fetch('/api/tgju-instruments');
   246→        if (!r.ok) return null;
   247→        const d: TgjuData = await r.json();
   248→        tgjuCacheRef.current = d;
   249→        return d;
   250→      } catch {
   251→        return null;
   252→      } finally {
   253→        tgjuFetchRef.current = null;
   254→      }
   255→    };
   256→    tgjuFetchRef.current = p();
   257→    return p();
   258→  }, []);
   259→
   260→  const fetchYahooData = React.useCallback(async (): Promise<YahooData | null> => {
   261→    if (yahooCacheRef.current) return yahooCacheRef.current;
   262→    if (yahooFetchRef.current) return yahooFetchRef.current;
   263→    const p = async () => {
   264→      try {
   265→        const r = await fetch('/api/yahoo-instruments');
   266→        if (!r.ok) return null;
   267→        const d = await r.json() as YahooData;
   268→        yahooCacheRef.current = d;
   269→        return d;
   270→      } catch {
   271→        return null;
   272→      } finally {
   273→        yahooFetchRef.current = null;
   274→      }
   275→    };
   276→    yahooFetchRef.current = p();
   277→    return p();
   278→  }, []);
   279→
   280→  /* -- Check if category is TGJU-based -- */
   281→  const isTgjuCategory = (cat: CategoryKey) => TGJU_CATEGORIES.has(cat);
   282→  const isYahooCategory = (cat: CategoryKey) => cat === 'yahoo_stock' || cat === 'yahoo_etf';
   283→
   284→  /* -- Get all items for current category -- */
   285→  const getItemsForCategory = React.useCallback(async (cat: CategoryKey): Promise<InstrumentItem[]> => {
   286→    // Yahoo Finance stocks and ETFs
   287→    if (cat === 'yahoo_stock') {
   288→      const yahoo = yahooCacheRef.current || await fetchYahooData();
   289→      return yahoo?.yahooStocks || [];
   290→    }
   291→    if (cat === 'yahoo_etf') {
   292→      const yahoo = yahooCacheRef.current || await fetchYahooData();
   293→      return yahoo?.yahooEtfs || [];
   294→    }
   295→
   296→    if (isTgjuCategory(cat)) {
   297→      const tgju = tgjuCacheRef.current || await fetchTgjuData();
   298→      const yahoo = yahooCacheRef.current || await fetchYahooData();
   299→      if (!tgju && !yahoo) return [];
   300→      switch (cat) {
   301→        case 'currency': return tgju?.currencies || [];
   302→        case 'gold': return [...(tgju?.gold || []), ...(tgju?.silver || [])];
   303→        case 'gold_etf': return tgju?.goldEtfs || [];
   304→        case 'silver': return tgju?.silver || [];
   305→        case 'crypto': return [...(tgju?.crypto || []), ...(yahoo?.yahooCrypto || [])];
   306→        case 'world_index': return [...(tgju?.worldIndices || []), ...(tgju?.foreignStocks || []), ...(yahoo?.yahooIndices || [])];
   307→        case 'foreign_stock': return tgju?.foreignStocks || [];
   308→        case 'forex': return [...(tgju?.forex || []), ...(yahoo?.yahooForex || [])];
   309→        case 'energy': return [...(tgju?.energy || []), ...(yahoo?.yahooEnergy || [])];
   310→        case 'metal': return [...(tgju?.metals || []), ...(yahoo?.yahooMetals || [])];
   311→        case 'commodity': return [...(tgju?.commodities || []), ...(yahoo?.yahooCommodities || [])];
   312→        default: return [];
   313→      }
   314→    }
   315→
   316→    // TSE categories
   317→    const data = cacheRef.current || await fetchData();
   318→    switch (cat) {
   319→      case 'indices': return data.indices;
   320→      case 'stocks':
   321→        return activeIndustry
   322→          ? data.stocks.filter((s) => s.cs === activeIndustry)
   323→          : data.stocks;
   324→      case 'etf': return data.etfs;
   325→      case 'bond': return [...data.bonds, ...data.mortgage];
   326→      case 'derivative': return [...data.futures, ...data.salaf];
   327→      case 'all':
   328→      default: {
   329→        const tgju = tgjuCacheRef.current || await fetchTgjuData();
   330→        const yahoo = yahooCacheRef.current || await fetchYahooData();
   331→        return [
   332→          ...data.indices,
   333→          ...data.stocks,
   334→          ...data.etfs,
   335→          ...(tgju?.currencies || []),
   336→          ...(tgju?.forex || []),
   337→          ...(yahoo?.yahooForex || []),
   338→          ...(tgju?.crypto || []),
   339→          ...(yahoo?.yahooCrypto || []),
   340→          ...(tgju?.gold || []),
   341→          ...(tgju?.silver || []),
   342→          ...(tgju?.goldEtfs || []),
   343→          ...(tgju?.worldIndices || []),
   344→          ...(yahoo?.yahooIndices || []),
   345→          ...(tgju?.energy || []),
   346→          ...(yahoo?.yahooEnergy || []),
   347→          ...(tgju?.metals || []),
   348→          ...(yahoo?.yahooMetals || []),
   349→          ...(tgju?.commodities || []),
   350→          ...(yahoo?.yahooCommodities || []),
   351→          ...(yahoo?.yahooStocks || []),
   352→          ...(yahoo?.yahooEtfs || []),
   353→          ...data.bonds,
   354→          ...data.futures,
   355→          ...data.salaf,
   356→          ...data.mortgage,
   357→        ];
   358→      }
   359→    }
   360→  }, [activeIndustry, fetchData, fetchTgjuData, fetchYahooData]);
   361→
   362→  /* -- Popular items -- */
   363→  const getPopular = React.useCallback(async () => {
   364→    const data = cacheRef.current || await fetchData();
   365→    const tgju = tgjuCacheRef.current;
   366→    const all = [
   367→      ...(data.indices.map((i) => ({ ...i, tval: i.index || 0 }))),
   368→      ...data.stocks,
   369→      ...data.etfs,
   370→      ...(tgju?.currencies || []),
   371→      ...(tgju?.crypto?.slice(0, 5) || []),
   372→      ...(tgju?.gold?.slice(0, 3) || []),
   373→    ];
   374→    const sorted = [...all].sort((a, b) => (b.tval || 0) - (a.tval || 0)).slice(0, 15);
   375→    setResults(sorted);
   376→    setTotalMatched(all.length);
   377→  }, [fetchData]);
   378→
   379→  /* -- Filter by search query -- */
   380→  const doFilter = React.useCallback((items: InstrumentItem[], q: string) => {
   381→    const n = q.trim().toLowerCase();
   382→    const f = items.filter(
   383→      (s) =>
   384→        s.l18?.toLowerCase().includes(n) ||
   385→        s.l30?.toLowerCase().includes(n) ||
   386→        s.cs?.toLowerCase().includes(n),
   387→    );
   388→    setTotalMatched(f.length);
   389→    setResults(f.slice(0, MAX_RESULTS));
   390→    setActiveIndex(-1);
   391→  }, []);
   392→
   393→  /* -- Show dropdown with data -- */
   394→  const showDropdown = React.useCallback(async (q: string) => {
   395→    setOpen(true);
   396→    await Promise.all([fetchData(), fetchTgjuData(), fetchYahooData()]);
   397→    const items = await getItemsForCategory(activeCategory);
   398→    if (q.trim().length === 0) {
   399→      if (activeCategory === 'all') {
   400→        await getPopular();
   401→      } else {
   402→        setResults(items.slice(0, MAX_RESULTS));
   403→        setTotalMatched(items.length);
   404→      }
   405→    } else {
   406→      doFilter(items, q);
   407→    }
   408→    setLoading(false);
   409→  }, [fetchData, fetchTgjuData, getItemsForCategory, activeCategory, getPopular, doFilter]);
   410→
   411→  /* -- Category change -- */
   412→  const handleCategoryChange = React.useCallback(async (cat: CategoryKey) => {
   413→    setActiveCategory(cat);
   414→    if (cat !== 'stocks') setActiveIndustry(null);
   415→    setShowIndustryPicker(false);
   416→
   417→    const items = await getItemsForCategory(cat);
   418→    if (query.trim().length > 0) {
   419→      doFilter(items, query);
   420→    } else if (cat === 'all') {
   421→      await getPopular();
   422→    } else {
   423→      setResults(items.slice(0, MAX_RESULTS));
   424→      setTotalMatched(items.length);
   425→    }
   426→    setActiveIndex(-1);
   427→  }, [getItemsForCategory, query, doFilter, getPopular]);
   428→
   429→  /* -- Industry change -- */
   430→  const handleIndustryChange = React.useCallback((industry: string | null) => {
   431→    setActiveIndustry(industry);
   432→    setShowIndustryPicker(false);
   433→    if (cacheRef.current) {
   434→      const items = industry
   435→        ? cacheRef.current.stocks.filter((s) => s.cs === industry)
   436→        : cacheRef.current.stocks;
   437→      if (query.trim().length > 0) {
   438→        doFilter(items, query);
   439→      } else {
   440→        setResults(items.slice(0, MAX_RESULTS));
   441→        setTotalMatched(items.length);
   442→      }
   443→      setActiveIndex(-1);
   444→    }
   445→  }, [query, doFilter]);
   446→
   447→  const handleChange = React.useCallback((value: string) => {
   448→    setQuery(value);
   449→    getItemsForCategory(activeCategory).then((items) => {
   450→      if (value.trim().length === 0) {
   451→        if (activeCategory === 'all') getPopular();
   452→        else {
   453→          setResults(items.slice(0, MAX_RESULTS));
   454→          setTotalMatched(items.length);
   455→        }
   456→      } else {
   457→        doFilter(items, value);
   458→      }
   459→    });
   460→  }, [activeCategory, getItemsForCategory, getPopular, doFilter]);
   461→
   462→  const handleFocus = React.useCallback(() => {
   463→    setLoading(true);
   464→    showDropdown(query);
   465→  }, [query, showDropdown]);
   466→
   467→  const selectSymbol = React.useCallback((s: InstrumentItem) => {
   468→    setQuery(s.l18);
   469→    setOpen(false);
   470→    onSelect?.(
   471→      s.l18,
   472→      s.category,
   473→      s.insCode,
   474→      s.tgjuKey,
   475→      s.finpySector,
   476→      s.finpyIndex,
   477→      s.webId,
   478→      s.yahooSymbol || (s as Record<string, unknown>).yahooSymbol as string | undefined,
   479→    );
   480→    inputRef.current?.blur();
   481→  }, [onSelect]);
   482→
   483→  const handleKeyDown = React.useCallback((e: React.KeyboardEvent) => {
   484→    if (!open || results.length === 0) {
   485→      if (e.key === 'ArrowDown' || e.key === 'Enter') { e.preventDefault(); handleFocus(); }
   486→      return;
   487→    }
   488→    if (e.key === 'ArrowDown') { e.preventDefault(); setActiveIndex((p) => p < results.length - 1 ? p + 1 : 0); }
   489→    else if (e.key === 'ArrowUp') { e.preventDefault(); setActiveIndex((p) => p > 0 ? p - 1 : results.length - 1); }
   490→    else if (e.key === 'Enter') { e.preventDefault(); if (activeIndex >= 0) selectSymbol(results[activeIndex]); }
   491→    else if (e.key === 'Escape') { e.preventDefault(); setOpen(false); inputRef.current?.blur(); }
   492→  }, [open, results, activeIndex, selectSymbol, handleFocus]);
   493→
   494→  React.useEffect(() => {
   495→    if (activeIndex >= 0) itemRefs.current[activeIndex]?.scrollIntoView({ block: 'nearest' });
   496→  }, [activeIndex]);
   497→
   498→  React.useEffect(() => {
   499→    if (!open) return;
