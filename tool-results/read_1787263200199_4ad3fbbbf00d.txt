    29→function renderAIText(text: string): string {
    30→  // Normalize zero-width and look-alike characters that may interfere with regex
    31→  const normalized = text.replace(/[\u200c\u200d\u200b\ufeff]/g, '');
    32→
    33→  // Color syntax: {color:hex}text{/color} or {color:named}text{/color} → <span>
    34→  const colorMap: Record<string, string> = {
    35→    'red': '#dc2626', 'red-600': '#dc2626', 'red-700': '#b91c1c',
    36→    'green': '#16a34a', 'emerald': '#059669', 'emerald-600': '#059669', 'emerald-700': '#047857',
    37→    'amber': '#d97706', 'amber-600': '#d97706', 'amber-700': '#b45309', 'amber-800': '#92400e',
    38→    'blue': '#2563eb', 'blue-600': '#2563eb', 'blue-700': '#1d4ed8',
    39→    'orange': '#ea580c',
    40→    'purple': '#9333ea', 'purple-600': '#9333ea', 'purple-700': '#7e22ce',
    41→  };
    42→
    43→  const withColors = normalized.replace(/\{color:([^}]+)\}([\s\S]*?)\{\/color\}/g, (_match, colorName: string, inner: string) => {
    44→    const hex = colorMap[colorName.trim()] || (colorName.startsWith('#') ? colorName : null);
    45→    if (!hex) return inner; // Unknown color: just return text without color
    46→    return `<span style="color:${hex}">${inner}</span>`;
    47→  });
    48→
    49→  return withColors
    50→    .split('\n\n')
    51→    .map(p => {
    52→      // Convert **bold** to <strong>
    53→      const html = p.replace(/\*\*(.+?)\*\*/g, '<strong class="font-bold text-gray-900">$1</strong>');
    54→      // Convert newlines within paragraph to <br>
    55→      return `<p class="mb-4">${html.replace(/\n/g, '<br>')}</p>`;
    56→    })
    57→    .join('');
    58→}

   517→      </div>
   518→      <span className="text-[10px] font-bold tabular-nums" style={{ color }}>{toPersianDigits(String(strength))}</span>
   519→    </div>
   520→  );
