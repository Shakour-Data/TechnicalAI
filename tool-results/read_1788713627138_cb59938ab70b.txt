    43→function hexToRgba(hex: string, alpha: number): string {
    44→  const r = parseInt(hex.slice(1, 3), 16);
    45→  const g = parseInt(hex.slice(3, 5), 16);
    46→  const b = parseInt(hex.slice(5, 7), 16);
    47→  return `rgba(${r},${g},${b},${alpha})`;
    48→}
    49→
    50→const safeNum = (n: number | null | undefined, fallback = 0) => (n ?? fallback);
    51→// Dynamic price formatter using data-source-derived decimals
    52→const toFaDyn = (n: number | null | undefined, decimals: number) => formatPriceFa(safeNum(n), decimals);
    53→const toFa1 = (n: number | null | undefined) => toPersianDigits(safeNum(n).toFixed(1)).replace(/\./g, '٫').replace(/-/g, '−');
    54→const toFa2 = (n: number | null | undefined) => toPersianDigits(safeNum(n).toFixed(2)).replace(/\./g, '٫').replace(/-/g, '−');
    55→
    56→function rsiSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    57→  if (v > 70) return 'bearish';
    58→  if (v < 30) return 'bullish';
    59→  return 'neutral';
    60→}
    61→
    62→function macdSignal(h: number): 'bullish' | 'bearish' | 'neutral' {
    63→  if (h > 0) return 'bullish';
    64→  if (h < 0) return 'bearish';
    65→  return 'neutral';
    66→}

    68→function adxSignal(v: number): 'strong' | 'neutral' {

    73→function stochSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    74→  if (v > 80) return 'bearish';

    76→  return 'neutral';

    79→function signalDotBg(signal: 'bullish' | 'bearish' | 'neutral' | 'strong', bullColor: string, bearColor: string, neutralColor: string, cardSubFg: string): string {

    81→    case 'bullish': return bullColor;
    82→    case 'bearish': return bearColor;
    83→    case 'strong': return neutralColor;
    84→    default: return cardSubFg;
    85→  }
    86→}

    88→function gradeLabel(grade: string): string {

    92→    case 'Moderate': return 'متوسط';

    95→}
    96→
    97→function gradeStyle(grade: string, bearColor: string, neutralColor: string, primary: string, cardBg: string, cardSubFg: string): React.CSSProperties {
    98→  switch (grade) {
    99→    case 'Very Strong': return { background: hexToRgba(bearColor, 0.15), color: bearColor };
   100→    case 'Strong': return { background: hexToRgba(neutralColor, 0.15), color: neutralColor };
   101→    case 'Moderate': return { background: hexToRgba(primary, 0.15), color: primary };
   102→    default: return { background: hexToRgba(cardSubFg, 0.1), color: cardSubFg };
   103→  }
   104→}
   105→
   106→function formatNumber(n: number | null | undefined): string {
   107→  const v = safeNum(n);
   108→  if (Math.abs(v) >= 1e9) return toPersianDigits((v / 1e9).toFixed(1)) + 'B';
   109→  if (Math.abs(v) >= 1e6) return toPersianDigits((v / 1e6).toFixed(1)) + 'M';
   110→  if (Math.abs(v) >= 1e3) return toPersianDigits((v / 1e3).toFixed(1)) + 'K';
   111→  return Math.round(v).toLocaleString('fa-IR');
   112→}
   113→
   114→function pctDistance(from: number, to: number): number {
   115→  if (from === 0) return 0;
   116→  return ((to - from) / from) * 100;
   117→}
   118→
   119→function trendDirectionLabel(dir: string, bullColor: string, bearColor: string, neutralColor: string): { label: string; color: string } {
   120→  const d = dir?.toLowerCase() ?? '';
   121→  if (d.includes('up') || d === 'صعودی') return { label: 'صعودی', color: bullColor };
   122→  if (d.includes('down') || d === 'نزولی') return { label: 'نزولی', color: bearColor };
   123→  return { label: 'خنثی', color: neutralColor };
   124→}

   149→function ProgressBar({ value, max, color, height = 4 }: { value: number; max: number; color: string; height?: number }) {
   150→  const { colors: C } = useTheme();
   151→  const pct = Math.min(100, Math.max(0, (value / max) * 100));
   152→  return (
   153→    <div className="w-full rounded-full overflow-hidden" style={{ height, background: C.cardBorder }}>
   154→      <div className="h-full rounded-full transition-all duration-500" style={{ width: `${pct}%`, background: color }} />
   155→    </div>
   156→  );
   157→}
