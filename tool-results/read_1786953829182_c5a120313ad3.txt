   141→/* --─ Helpers -------------------------------------------- */
   142→
   143→function formatNum(num: number): string {
   144→  if (num == null) return '—';
   145→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 0 }).format(num);
   146→}
   147→
   148→function formatIdx(num: number): string {
   149→  if (num == null) return '—';
   150→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 2 }).format(num);
   151→}
   152→

   155→    <svg className='animate-spin text-gray-400' xmlns='http://www.w3.org/2000/svg' fill='none' viewBox='0 0 24 24' width='16' height='16'>

   158→    </svg>
   159→  );
   160→}

   162→/* --─ Component ------------------------------------------ */
   163→
   164→export default function SymbolSearch({
   165→  onSelect,
   166→  placeholder = 'جستجوی ابزار مالی ...',
   167→  className,
   168→}: SymbolSearchProps) {
   169→  const [query, setQuery] = React.useState('');
   170→  const [open, setOpen] = React.useState(false);
   171→  const [results, setResults] = React.useState<InstrumentItem[]>([]);
   172→  const [loading, setLoading] = React.useState(false);
   173→  const [activeIndex, setActiveIndex] = React.useState(-1);
   174→  const [totalMatched, setTotalMatched] = React.useState(0);
   175→  const [activeCategory, setActiveCategory] = React.useState<CategoryKey>('all');
   176→  const [activeIndustry, setActiveIndustry] = React.useState<string | null>(null);
   177→  const [showIndustryPicker, setShowIndustryPicker] = React.useState(false);
   178→
   179→  const cacheRef = React.useRef<InstrumentsData | null>(null);
   180→  const tgjuCacheRef = React.useRef<TgjuData | null>(null);
   181→  const fetchRef = React.useRef<Promise<InstrumentsData> | null>(null);
   182→  const tgjuFetchRef = React.useRef<Promise<TgjuData | null> | null>(null);
   183→  const inputRef = React.useRef<HTMLInputElement>(null);
   184→  const dropdownRef = React.useRef<HTMLDivElement>(null);
   185→  const itemRefs = React.useRef<(HTMLDivElement | null)[]>([]);
   186→  const industryScrollRef = React.useRef<HTMLDivElement>(null);
   187→
   188→  /* -- Data fetching -- */
   189→  const fetchData = React.useCallback(async (): Promise<InstrumentsData> => {
   190→    if (cacheRef.current) return cacheRef.current;
   191→    if (fetchRef.current) return fetchRef.current;
   192→    const p = async () => {
   193→      try {
   194→        const r = await fetch('/api/instruments');
   195→        if (!r.ok) throw new Error(`${r.status}`);
   196→        const d: InstrumentsData = await r.json();
   197→        cacheRef.current = d;
   198→        return d;
   199→      } finally {
   200→        fetchRef.current = null;
   201→      }
   202→    };
   203→    fetchRef.current = p();
   204→    return fetchRef.current;
   205→  }, []);
   206→
   207→  const fetchTgjuData = React.useCallback(async (): Promise<TgjuData | null> => {
   208→    if (tgjuCacheRef.current) return tgjuCacheRef.current;
   209→    if (tgjuFetchRef.current) return tgjuFetchRef.current;
   210→    const p = async () => {
   211→      try {
   212→        const r = await fetch('/api/tgju-instruments');
   213→        if (!r.ok) return null;
   214→        const d: TgjuData = await r.json();
   215→        tgjuCacheRef.current = d;
   216→        return d;
   217→      } catch {
   218→        return null;
   219→      } finally {
   220→        tgjuFetchRef.current = null;
   221→      }
   222→    };
   223→    tgjuFetchRef.current = p();
   224→    return p();
   225→  }, []);
   226→
   227→  /* -- Check if category is TGJU-based -- */
   228→  const isTgjuCategory = (cat: CategoryKey) => TGJU_CATEGORIES.has(cat);
   229→
   230→  /* -- Get all items for current category -- */
   231→  const getItemsForCategory = React.useCallback(async (cat: CategoryKey): Promise<InstrumentItem[]> => {
   232→    if (isTgjuCategory(cat)) {
   233→      const tgju = tgjuCacheRef.current || await fetchTgjuData();
   234→      if (!tgju) return [];
   235→      switch (cat) {
   236→        case 'currency': return tgju.currencies;
   237→        case 'gold': return [...tgju.gold, ...tgju.silver];
   238→        case 'gold_etf': return tgju.goldEtfs;
   239→        case 'silver': return tgju.silver;
   240→        case 'crypto': return tgju.crypto;
   241→        case 'world_index': return tgju.worldIndices;
   242→        case 'foreign_stock': return tgju.foreignStocks;
   243→        case 'forex': return tgju.forex;
   244→        case 'energy': return tgju.energy;
   245→        case 'metal': return tgju.metals;
   246→        case 'commodity': return tgju.commodities;
   247→        default: return [];
   248→      }
   249→    }
   250→
   251→    // TSE categories
   252→    const data = cacheRef.current || await fetchData();
   253→    switch (cat) {
   254→      case 'indices': return data.indices;
   255→      case 'stocks':
   256→        return activeIndustry
   257→          ? data.stocks.filter((s) => s.cs === activeIndustry)
   258→          : data.stocks;
   259→      case 'etf': return data.etfs;
   260→      case 'bond': return [...data.bonds, ...data.mortgage];
   261→      case 'derivative': return [...data.futures, ...data.salaf];
   262→      case 'all':
   263→      default: {
   264→        const tgju = tgjuCacheRef.current;
   265→        return [
   266→          ...data.indices,
   267→          ...data.stocks,
   268→          ...data.etfs,
   269→          ...(tgju?.currencies || []),
   270→          ...(tgju?.forex || []),
   271→          ...(tgju?.crypto || []),
   272→          ...(tgju?.gold || []),
   273→          ...(tgju?.silver || []),
   274→          ...(tgju?.goldEtfs || []),
   275→          ...(tgju?.worldIndices || []),
   276→          ...(tgju?.foreignStocks || []),
   277→          ...(tgju?.energy || []),
   278→          ...(tgju?.metals || []),
   279→          ...(tgju?.commodities || []),
   280→          ...data.bonds,
   281→          ...data.futures,
   282→          ...data.salaf,
   283→          ...data.mortgage,
   284→        ];
   285→      }
   286→    }
   287→  }, [activeIndustry, fetchData, fetchTgjuData]);
   288→
   289→  /* -- Popular items -- */
   290→  const getPopular = React.useCallback(async () => {
   291→    const data = cacheRef.current || await fetchData();
   292→    const tgju = tgjuCacheRef.current;
   293→    const all = [
   294→      ...(data.indices.map((i) => ({ ...i, tval: i.index || 0 }))),
   295→      ...data.stocks,
   296→      ...data.etfs,
   297→      ...(tgju?.currencies || []),
   298→      ...(tgju?.crypto?.slice(0, 5) || []),
   299→      ...(tgju?.foreignStocks?.slice(0, 3) || []),
   300→      ...(tgju?.gold?.slice(0, 3) || []),
   301→    ];
   302→    const sorted = [...all].sort((a, b) => (b.tval || 0) - (a.tval || 0)).slice(0, 15);
   303→    setResults(sorted);
   304→    setTotalMatched(all.length);
   305→  }, [fetchData]);
   306→
   307→  /* -- Filter by search query -- */
   308→  const doFilter = React.useCallback((items: InstrumentItem[], q: string) => {
   309→    const n = q.trim().toLowerCase();
   310→    const f = items.filter(
   311→      (s) =>
   312→        s.l18?.toLowerCase().includes(n) ||
   313→        s.l30?.toLowerCase().includes(n) ||
   314→        s.cs?.toLowerCase().includes(n),
   315→    );
   316→    setTotalMatched(f.length);
   317→    setResults(f.slice(0, MAX_RESULTS));
   318→    setActiveIndex(-1);
   319→  }, []);
   320→
   321→  /* -- Show dropdown with data -- */
   322→  const showDropdown = React.useCallback(async (q: string) => {
   323→    setOpen(true);
   324→    await Promise.all([fetchData(), fetchTgjuData()]);
   325→    const items = await getItemsForCategory(activeCategory);
   326→    if (q.trim().length === 0) {
   327→      if (activeCategory === 'all') {
   328→        await getPopular();
   329→      } else {
   330→        setResults(items.slice(0, MAX_RESULTS));
   331→        setTotalMatched(items.length);
   332→      }
   333→    } else {
   334→      doFilter(items, q);
   335→    }
   336→    setLoading(false);
   337→  }, [fetchData, fetchTgjuData, getItemsForCategory, activeCategory, getPopular, doFilter]);
   338→
   339→  /* -- Category change -- */
   340→  const handleCategoryChange = React.useCallback(async (cat: CategoryKey) => {
   341→    setActiveCategory(cat);
   342→    if (cat !== 'stocks') setActiveIndustry(null);
   343→    setShowIndustryPicker(false);
   344→
   345→    const items = await getItemsForCategory(cat);
   346→    if (query.trim().length > 0) {
   347→      doFilter(items, query);
   348→    } else if (cat === 'all') {
   349→      await getPopular();
   350→    } else {
   351→      setResults(items.slice(0, MAX_RESULTS));
   352→      setTotalMatched(items.length);
   353→    }
   354→    setActiveIndex(-1);
   355→  }, [getItemsForCategory, query, doFilter, getPopular]);
   356→
   357→  /* -- Industry change -- */
   358→  const handleIndustryChange = React.useCallback((industry: string | null) => {
   359→    setActiveIndustry(industry);
   360→    setShowIndustryPicker(false);
   361→    if (cacheRef.current) {
   362→      const items = industry
   363→        ? cacheRef.current.stocks.filter((s) => s.cs === industry)
   364→        : cacheRef.current.stocks;
   365→      if (query.trim().length > 0) {
   366→        doFilter(items, query);
   367→      } else {
   368→        setResults(items.slice(0, MAX_RESULTS));
   369→        setTotalMatched(items.length);
   370→      }
   371→      setActiveIndex(-1);
   372→    }
   373→  }, [query, doFilter]);
   374→
   375→  const handleChange = React.useCallback((value: string) => {
   376→    setQuery(value);
   377→    getItemsForCategory(activeCategory).then((items) => {
   378→      if (value.trim().length === 0) {
   379→        if (activeCategory === 'all') getPopular();
   380→        else {
   381→          setResults(items.slice(0, MAX_RESULTS));
   382→          setTotalMatched(items.length);
   383→        }
   384→      } else {
   385→        doFilter(items, value);
   386→      }
   387→    });
   388→  }, [activeCategory, getItemsForCategory, getPopular, doFilter]);
   389→
   390→  const handleFocus = React.useCallback(() => {
   391→    setLoading(true);
   392→    showDropdown(query);
   393→  }, [query, showDropdown]);
   394→
   395→  const selectSymbol = React.useCallback((s: InstrumentItem) => {
   396→    setQuery(s.l18);
   397→    setOpen(false);
   398→    onSelect?.(s.l18, s.category, s.insCode, s.tgjuKey);
   399→    inputRef.current?.blur();
   400→  }, [onSelect]);
   401→
   402→  const handleKeyDown = React.useCallback((e: React.KeyboardEvent) => {
   403→    if (!open || results.length === 0) {
   404→      if (e.key === 'ArrowDown' || e.key === 'Enter') { e.preventDefault(); handleFocus(); }
   405→      return;
   406→    }
   407→    if (e.key === 'ArrowDown') { e.preventDefault(); setActiveIndex((p) => p < results.length - 1 ? p + 1 : 0); }
   408→    else if (e.key === 'ArrowUp') { e.preventDefault(); setActiveIndex((p) => p > 0 ? p - 1 : results.length - 1); }
   409→    else if (e.key === 'Enter') { e.preventDefault(); if (activeIndex >= 0) selectSymbol(results[activeIndex]); }
   410→    else if (e.key === 'Escape') { e.preventDefault(); setOpen(false); inputRef.current?.blur(); }
   411→  }, [open, results, activeIndex, selectSymbol, handleFocus]);
   412→
   413→  React.useEffect(() => {
   414→    if (activeIndex >= 0) itemRefs.current[activeIndex]?.scrollIntoView({ block: 'nearest' });
   415→  }, [activeIndex]);
   416→
   417→  React.useEffect(() => {
   418→    if (!open) return;
   419→    const handler = (e: MouseEvent) => {
   420→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node) &&
   421→          inputRef.current && !inputRef.current.contains(e.target as Node)) {
   422→        setOpen(false);
   423→        setShowIndustryPicker(false);
   424→      }
   425→    };
   426→    document.addEventListener('mousedown', handler);
   427→    return () => document.removeEventListener('mousedown', handler);
   428→  }, [open]);
   429→
   430→  const clear = () => { setQuery(''); setActiveIndustry(null); setActiveCategory('all'); setResults([]); inputRef.current?.focus(); };
   431→  const isPopular = query.trim().length === 0 && activeCategory === 'all';
   432→  const isSearch = query.trim().length > 0;
   433→
   434→  /* -- Render instrument price -- */
   435→  const renderPrice = (item: InstrumentItem) => {
   436→    const isTgju = isTgjuCategory(item.category as CategoryKey);
   437→    if (item.category === 'index') {
   438→      return (
   439→        <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   440→          <span className='text-xs font-semibold text-gray-200'>{formatIdx(item.pl)}</span>
   441→          <span className={cn(
   442→            'text-[11px] font-bold px-1.5 py-0.5 rounded',
   443→            item.pcp > 0 ? 'bg-emerald-500/15 text-emerald-400' :
   444→            item.pcp < 0 ? 'bg-red-500/15 text-red-400' : 'text-gray-500',
   445→          )}>
   446→            {item.pcp > 0 ? '+' : ''}{item.pcp?.toFixed(2)}%
   447→          </span>
   448→        </div>
   449→      );
   450→    }
   451→    // For static TGJU items with no live price, show dash
   452→    if (isTgju && item.pl === 0) {
   453→      return (
   454→        <div className='flex shrink-0 flex-col items-end gap-0.5'>
   455→          <span className='text-xs text-gray-500'>—</span>
   456→        </div>
   457→      );
   458→    }
   459→    const isUp = item.pcp > 0;
   460→    const isDown = item.pcp < 0;
   461→    return (
   462→      <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   463→        <span className='text-xs font-semibold text-gray-200'>{formatNum(item.pl)}</span>
   464→        <span className={cn(
   465→          'text-[11px] font-bold px-1.5 py-0.5 rounded',
   466→          isUp ? 'bg-emerald-500/15 text-emerald-400' :
   467→          isDown ? 'bg-red-500/15 text-red-400' : 'text-gray-500',
   468→        )}>
   469→          {isUp ? '+' : ''}{item.pcp?.toFixed(2)}%
   470→        </span>
   471→      </div>
   472→    );
   473→  };
   474→
   475→  const industries = cacheRef.current?.industries || [];
   476→
   477→  return (
   478→    <div dir='rtl' className={cn('relative', className)}>
   479→
   480→      <div className='relative'>
   481→        <Search className='absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-500 pointer-events-none' />
   482→        <Input
   483→          ref={inputRef}
   484→          type='text'
   485→          value={query}
   486→          onChange={(e) => handleChange(e.target.value)}
   487→          onFocus={handleFocus}
   488→          onKeyDown={handleKeyDown}
   489→          placeholder={placeholder}
   490→          suppressHydrationWarning
   491→          className={cn(
   492→            'h-10 w-full rounded-lg border border-gray-700/80 bg-gray-900/90',
   493→            'text-sm text-gray-100 placeholder:text-gray-500',
   494→            'transition-all',
   495→            'focus-visible:border-amber-500/60 focus-visible:ring-amber-500/20 focus-visible:ring-[3px]',
   496→            'pr-9',
   497→          )}
   498→          autoComplete='off' spellCheck={false}
   499→        />
   500→        {query && (
   501→          <button onClick={clear} className='absolute left-3 top-1/2 -translate-y-1/2 text-gray-500 hover:text-gray-300 transition-colors' type='button'>
   502→            <X className='w-4 h-4' />
   503→          </button>
   504→        )}
   505→      </div>
   506→
   507→
   508→      {open && (
   509→        <div
