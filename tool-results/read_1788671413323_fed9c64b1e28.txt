   360→/**
   361→ * Parse a number that might contain Persian digits, commas, or other formatting.
   362→ * Handles null/undefined/empty strings gracefully.
   363→ */
   364→function parsePersianNum(str: string | null | undefined | number): number {
   365→  if (str === null || str === undefined) return 0;
   366→  let s = String(str).trim();
   367→  if (!s || s === '-' || s === '') return 0;
   368→  // Strip trailing % sign (e.g. "0.05%" → "0.05")
   369→  if (s.endsWith('%')) s = s.slice(0, -1).trim();
   370→  // Replace commas and Persian digits
   371→  const cleaned = s.replace(/[,۰-۹]/g, (c: string) => {
   372→    const persianDigits = '۰۱۲۳۴۵۶۷۸۹';
   373→    const idx = persianDigits.indexOf(c);
   374→    return idx >= 0 ? String(idx) : c === ',' ? '' : c;
   375→  });
   376→  const n = Number(cleaned);
   377→  return isFinite(n) ? n : 0;
   378→}

   380→/**
   381→ * Check if a value looks like a date string (contains / or - and is not a number).
   382→ */
   383→function looksLikeDate(val: string | null | undefined | number): boolean {
   384→  if (!val) return false;
   385→  const s = String(val).trim();
   386→  return /[\/\-]/.test(s) && isNaN(Number(s.replace(/[,۰-۹\/\-]/g, '')));
   387→}
   388→
   389→/**
   390→ * Check if a value looks like a percentage (ends with % or is small like 0.5 to 50).
   391→ */
   392→function looksLikePercentage(val: string | null | undefined | number): boolean {
   393→  if (!val) return false;
   394→  const s = String(val).trim();
   395→  if (s.endsWith('%')) return true;
   396→  const n = parsePersianNum(val);
   397→  return !isNaN(n) && Math.abs(n) < 100 && n !== 0;
   398→}

   400→/**
   401→ * Auto-detect column mapping for TGJU API response.
   402→ * Expected format: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   403→ * But the API may change column order.
   404→ */
   405→function detectColumnMap(row: (string | number)[]): {
   406→  openIdx: number; highIdx: number; lowIdx: number; closeIdx: number;
   407→  jalaliDateIdx: number; gregorianDateIdx: number;
   408→} | null {
   409→  if (!row || row.length < 6) return null;
   410→
   411→  let jalaliDateIdx = -1;
   412→  let gregorianDateIdx = -1;
   413→  const numericIndices: number[] = [];
   414→  const percentageIndices: number[] = [];
   415→
   416→  for (let i = 0; i < row.length; i++) {
   417→    const val = row[i];
   418→    if (looksLikeDate(val)) {
   419→      // Jalali dates have Persian digits or format like 1404/05/25
   420→      const s = String(val);
   421→      if (/[۰-۹]/.test(s) || /^14\d{2}/.test(s.replace(/[۰-۹]/g, (d: string) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d))))) {
   422→        if (jalaliDateIdx === -1) jalaliDateIdx = i;
   423→      } else {
   424→        if (gregorianDateIdx === -1) gregorianDateIdx = i;
   425→      }
   426→    } else if (looksLikePercentage(val)) {
   427→      percentageIndices.push(i);
   428→    } else {
   429→      const n = parsePersianNum(val);
   430→      if (isFinite(n) && n !== 0) {
   431→        numericIndices.push(i);
   432→      }
   433→    }
   434→  }
   435→
   436→  // If we couldn't auto-detect dates, fall back to assumed positions
   437→  if (jalaliDateIdx === -1 && row.length >= 8) {
   438→    // Default assumption: row[6] = gregorian, row[7] = jalali
   439→    if (looksLikeDate(row[7])) jalaliDateIdx = 7;
   440→    else if (looksLikeDate(row[6])) jalaliDateIdx = 6;
   441→    if (gregorianDateIdx === -1 && looksLikeDate(row[6]) && jalaliDateIdx !== 6) gregorianDateIdx = 6;
   442→    else if (gregorianDateIdx === -1 && looksLikeDate(row[7]) && jalaliDateIdx !== 7) gregorianDateIdx = 7;
   443→  }
   444→
   445→  // Numeric columns (excluding percentages) should be OHLC + change
   446→  // OHLC: typically 4 largest absolute values among non-percentage numerics
   447→  // change: the remaining one
   448→  const ohlcCandidates = numericIndices.filter(i => !percentageIndices.includes(i));
   449→
   450→  if (ohlcCandidates.length < 4) {
   451→    // Fallback: assume first 4 numeric columns are OHLC
   452→    if (numericIndices.length >= 4) {
   453→      return {
   454→        openIdx: numericIndices[0],
   455→        highIdx: numericIndices[2],
   456→        lowIdx: numericIndices[1],
   457→        closeIdx: numericIndices[3],
   458→        jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : 7,
   459→        gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : 6,
   460→      };
   461→    }
   462→    return null;
   463→  }
   464→
   465→  // Sort by absolute value descending to identify OHLC (largest) vs change (smallest)
   466→  const sorted = ohlcCandidates
   467→    .map(i => ({ idx: i, val: Math.abs(parsePersianNum(row[i])) }))
   468→    .sort((a, b) => b.val - a.val);
   469→
   470→  // The 4 largest should be OHLC. Among them, identify high (max) and low (min).
   471→  // open and close are the other two.
   472→  const top4 = sorted.slice(0, 4);
   473→  const maxIdx = top4.reduce((a, b) => a.val > b.val ? a : b).idx;
   474→  const minIdx = top4.reduce((a, b) => a.val < b.val ? a : b).idx;
   475→  const others = top4.filter(x => x.idx !== maxIdx && x.idx !== minIdx);
   476→
   477→  // Among the two remaining, the one closer to high is open, the other is close
   478→  // (heuristic: in an uptrend open < close, but we can't assume direction)
   479→  // Default: first = open, second = close
   480→  const openIdx = others[0]?.idx ?? ohlcCandidates[0];
   481→  const closeIdx = others[1]?.idx ?? ohlcCandidates[3];
   482→
   483→  return {
   484→    openIdx,
   485→    highIdx: maxIdx,
   486→    lowIdx: minIdx,
   487→    closeIdx,
   488→    jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : row.length - 1,
   489→    gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : row.length - 2,
   490→  };
   491→}

   493→/**
   494→ * Validate a candle's OHLC values for sanity.
   495→ * Returns true if the candle is valid.
   496→ */
   497→function isValidCandle(o: number, h: number, l: number, c: number): boolean {
   498→  if (!isFinite(o) || !isFinite(h) || !isFinite(l) || !isFinite(c)) return false;
   499→  if (o <= 0 || h <= 0 || l <= 0 || c <= 0) return false;
   500→  // High must be >= Low, Open, Close
   501→  if (h < l || h < o || h < c) return false;
   502→  // Low must be <= High, Open, Close
   503→  if (l > o || l > c) return false;
   504→  // No infinite or extreme values
   505→  if (o > 1e15 || h > 1e15 || l > 1e15 || c > 1e15) return false;
   506→  return true;
   507→}

   509→/**
   510→ * Check if parsed candle data is reasonably fresh.
   511→ * Rejects data where the most recent candle is older than 2 years.
   512→ * This prevents serving data from e.g. 1392 when current year is 1404.
   513→ */
   514→function isDataFresh(candles: TgjuOHLC[]): boolean {
   515→  if (candles.length === 0) return false;
   516→  const lastDateStr = candles[candles.length - 1].date;
   517→  const yearMatch = lastDateStr.match(/(\d{4})/);
   518→  if (!yearMatch) return true; // Can't determine year, assume fresh
   519→
   520→  const year = parseInt(yearMatch[1], 10);
   521→  const now = new Date();
   522→  const gregorianYear = now.getFullYear();
   523→  // Approximate current Jalali year (Gregorian - 621 or 622)
   524→  const approxJalaliYear = gregorianYear - 621;
   525→
   526→  // Jalali years: 1390-1500 range
   527→  if (year >= 1390 && year <= 1500) {
   528→    return year >= approxJalaliYear - 2; // Allow up to 2 years old
   529→  }
   530→
   531→  // Gregorian years
   532→  if (year >= 2020 && year <= 2035) {
   533→    return year >= gregorianYear - 2;
   534→  }
   535→
   536→  // Unknown format, don't reject
   537→  return true;
   538→}

   587→/* ─── Fetch live instruments (currency + gold from tgju-api) + static list ── */
   588→
   589→export async function fetchTgjuInstruments(): Promise<TgjuInstrument[]> {
   590→  const now = Date.now();
   591→  if (liveInstrumentsCache.data && now - liveInstrumentsCache.ts < INSTRUMENTS_TTL) {
   592→    return liveInstrumentsCache.data;
   593→  }
   594→
   595→  const controller = new AbortController();
   596→  const timeout = setTimeout(() => controller.abort(), 10000);
   597→
   598→  try {
   599→    const [currencyRes, goldRes] = await Promise.all([
   600→      fetch(`${TGJU_API_BASE}/api/price/currency`, { signal: controller.signal }),
   601→      fetch(`${TGJU_API_BASE}/api/price/gold`, { signal: controller.signal }),
   602→    ]);
   603→
   604→    clearTimeout(timeout);
   605→
   606→    const instruments: TgjuInstrument[] = [];
   607→
   608→    if (currencyRes.ok) {
   609→      const currencies: TgjuPriceItem[] = await currencyRes.json();
   610→      for (const c of currencies) {
   611→        const price = parsePersianNum(c.price);
   612→        const low = parsePersianNum(c.low_price);
   613→        const high = parsePersianNum(c.high_price);
   614→        instruments.push({
   615→          title: c.title,
   616→          key: c.key,
   617→          price,
   618→          highPrice: high,
   619→          lowPrice: low,
   620→          change: 0,
   621→          changePercent: 0,
   622→          category: 'currency',
   623→        });
   624→      }
   625→    }
   626→
   627→    if (goldRes.ok) {
   628→      const goldCategories: TgjuGoldCategory[] = await goldRes.json();
   629→      for (const cat of goldCategories) {
   630→        for (const item of cat.prices) {
   631→          const price = parsePersianNum(item.price);
   632→          const low = parsePersianNum(item.low_price);
   633→          const high = parsePersianNum(item.high_price);
   634→
   635→          let category: TgjuCategoryType = 'gold';
   636→          if (item.key.startsWith('ime_fund_')) category = 'gold_etf';
   637→          else if (item.key.startsWith('silver_')) category = 'silver';
   638→
   639→          instruments.push({
   640→            title: item.title,
   641→            key: item.key,
   642→            price,
   643→            highPrice: high,
   644→            lowPrice: low,
   645→            change: 0,
   646→            changePercent: 0,
   647→            category,
   648→            groupTitle: cat.title,
   649→          });
   650→        }
   651→      }
   652→    }
   653→
   654→    // Add static instruments (crypto, world indices, forex, energy, metals, commodities)
   655→    for (const s of STATIC_INSTRUMENTS) {
   656→      const cached = staticPriceCache.get(s.key);
   657→      const price = cached?.close ?? 0;
   658→      instruments.push({
   659→        title: s.title,
   660→        key: s.key,
   661→        price,
   662→        highPrice: cached?.high ?? 0,
   663→        lowPrice: cached?.low ?? 0,
   664→        change: cached ? (cached.close - cached.prevClose) : 0,
   665→        changePercent: cached && cached.prevClose > 0 ? ((cached.close - cached.prevClose) / cached.prevClose) * 100 : 0,
   666→        category: s.category,
   667→        groupTitle: s.groupTitle,
   668→      });
   669→    }
   670→
   671→    liveInstrumentsCache.data = instruments;
   672→    liveInstrumentsCache.ts = now;
   673→
   674→    // ── Background: fetch chart data for currency/gold to get accurate daily change ──
   675→    backgroundFetchCurrencyGoldChange(instruments).catch(() => {});
   676→
   677→    // ── Background: fetch Yahoo prices for static instruments with price=0 (TGJU API 403) ──
   678→    backgroundFetchYahooFallbackPrices(instruments).catch(() => {});
   679→
   680→    return instruments;
   681→  } catch {
   682→    clearTimeout(timeout);
   683→    return liveInstrumentsCache.data || [];
   684→  }
   685→}

   687→/**
   688→ * Background task: fetch last 2 candles for currency/gold items to compute accurate daily change.
   689→ * Uses direct fetch to avoid queue contention.
   690→ * Only updates change/changePercent — keeps the live price from tgju-api.
   691→ */
   692→async function backgroundFetchCurrencyGoldChange(instruments: TgjuInstrument[]) {
   693→  const currencyGoldItems = instruments.filter(
   694→    (i) => i.category === 'currency' || i.category === 'gold' || i.category === 'silver' || i.category === 'gold_etf'
   695→  );
   696→
   697→  // Limit to first 10 to avoid rate-limiting (most important: USD, EUR, gold coin, etc.)
   698→  const priorityKeys = currencyGoldItems.slice(0, 10).map((i) => i.key);
   699→
   700→  for (const key of priorityKeys) {
   701→    try {
   702→      const url = `${TGJU_CHART_API}/${key}?lang=fa&order_dir=desc&start=0&length=2`;
   703→      const res = await fetch(url, { signal: AbortSignal.timeout(10_000) });
   704→      if (!res.ok) continue;
   705→      const jsonStr = await res.text();
   706→      if (!jsonStr.startsWith('{')) continue;
   707→
   708→      const data = JSON.parse(jsonStr);
   709→      const rows: string[][] = data.data;
   710→      if (!rows || rows.length < 2) continue;
   711→
   712→      const newest = rows[0];
   713→      const prev = rows[1];
   714→      const latestClose = parsePersianNum(newest[3]);
   715→      const prevClose = parsePersianNum(prev[3]);
   716→
   717→      if (latestClose > 0 && prevClose > 0) {
   718→        const inst = liveInstrumentsCache.data?.find((i) => i.key === key);
   719→        if (inst) {
   720→          inst.change = latestClose - prevClose;
   721→          inst.changePercent = ((latestClose - prevClose) / prevClose) * 100;
   722→        }
   723→      }
   724→    } catch {
   725→      // skip individual errors
   726→    }
   727→  }
   728→}

   738→/**
   739→ * Background task: fetch Yahoo Finance prices for static instruments that have price=0
   740→ * (happens when api.tgju.org returns 403). Updates the live cache in-place.
   741→ */
   742→async function backgroundFetchYahooFallbackPrices(instruments: TgjuInstrument[]) {
   743→  // Collect static instruments with price=0 that have a Yahoo mapping
   744→  const zeroPriceItems = instruments.filter(
   745→    (i) => i.price === 0 && i.key in TGJU_TO_YAHOO_MAP
   746→  );
   747→
   748→  if (zeroPriceItems.length === 0) return;
   749→
   750→  console.log(`[TGJU] ${zeroPriceItems.length} static instruments have price=0, fetching from Yahoo Finance as fallback...`);
   751→
   752→  // Collect unique Yahoo symbols
   753→  const keyToSymbol = new Map<string, string>();
   754→  const symbolToKeys = new Map<string, string[]>();
   755→  for (const item of zeroPriceItems) {
   756→    const yahooSym = TGJU_TO_YAHOO_MAP[item.key];
   757→    if (!yahooSym) continue;
   758→    keyToSymbol.set(item.key, yahooSym);
   759→    if (!symbolToKeys.has(yahooSym)) symbolToKeys.set(yahooSym, []);
   760→    symbolToKeys.get(yahooSym)!.push(item.key);
   761→  }
   762→

   855→/* ─── Page Reader Request Queue ─────────────────────── */
   856→// Track which keys were last fetched from Yahoo (for route to detect source)
   857→export const yahooFetchedKeys = new Set<string>();
   858→
   859→let pageReaderQueue: Promise<void> = Promise.resolve();
   860→let lastPageReaderCall = 0;
   861→const PAGE_READER_MIN_INTERVAL = 10_000; // 10 seconds between calls
   862→const PAGE_READER_MAX_RETRIES = 3;
   863→const PAGE_READER_RETRY_BASE_DELAY = 3_000; // 3 seconds base retry delay

   865→/**
   866→ * Fetch historical OHLC data for a TGJU instrument.
   867→ *
   868→ * Strategy (permanent fix):
   869→ * 1. Check in-memory cache (4h TTL)
   870→ * 2. For global instruments with Yahoo mapping → use Yahoo Finance as PRIMARY
   871→ * 3. For Iranian instruments → try direct fetch, then queued page_reader with retry
   872→ * 4. If all fail → return stale cache (up to 24h) or file-based cache
   873→ */
   874→export async function fetchTgjuHistory(tgjuKey: string): Promise<TgjuOHLC[]> {
   875→  const now = Date.now();
   876→
   877→  // ── Step 0: Check in-memory cache (with freshness validation) ──
   878→  const cached = historyCache.get(tgjuKey);
   879→  if (cached && now - cached.ts < HISTORY_TTL) {
   880→    // Validate cached data is not stale (e.g. 1392-era data from a previous bug)
   881→    if (isDataFresh(cached.data)) {
   882→      return cached.data;
   883→    } else {
   884→      console.warn(`[TGJU] In-memory cache for ${tgjuKey} is STALE (latest: ${cached.data[cached.data.length - 1]?.date}), invalidating`);
   885→      historyCache.delete(tgjuKey);
   886→    }
   887→  }
   888→
   889→  // ── Step 1: For global instruments, use Yahoo Finance as PRIMARY source ──
   890→  const yahooSymbol = TGJU_TO_YAHOO_MAP[tgjuKey];
   891→  if (yahooSymbol) {
   892→    try {
   893→      const YahooFinance = (await import('yahoo-finance2')).default;
   894→      const yf = new YahooFinance({ suppressNotices: ['yahooSurvey'], validation: { logErrors: false } });
   895→      const result = await yf.chart(yahooSymbol, {
   896→        period1: new Date(Date.now() - 400 * 24 * 60 * 60 * 1000),
   897→        period2: new Date(),
   898→        interval: '1d',
   899→      });
   900→      if (result.quotes && result.quotes.length >= 30) {
   901→        const candles: TgjuOHLC[] = result.quotes
   902→          .filter((q: any) => q.close != null && q.open != null)
   903→          .map((q: any) => ({
   904→            date: q.date?.toISOString().split('T')[0] || '',
   905→            open: q.open!,
   906→            high: q.high || q.close!,
   907→            low: q.low || q.close!,
   908→            close: q.close!,
   909→          }));
   910→        if (candles.length >= 30) {
   911→          console.log(`[TGJU] ${tgjuKey}: Got ${candles.length} candles from Yahoo (${yahooSymbol})`);
   912→          yahooFetchedKeys.add(tgjuKey);
   913→          cacheHistory(tgjuKey, candles, now);
   914→          return candles;
   915→        }
   916→      }
   917→    } catch (err) {
   918→      console.warn(`[TGJU] Yahoo historical fetch failed for ${tgjuKey} (${yahooSymbol}):`, err instanceof Error ? err.message : err);
   919→    }
   920→  }
   921→
   922→  // ── Step 2: For Iranian instruments (or Yahoo fallback failed), try TGJU sources ──
   923→  // 2a: Try direct fetch (fast path, works when Cloudflare not blocking)
   924→  const directResult = await tryDirectFetch(tgjuKey);
   925→  if (directResult && directResult.length >= 30) {
   926→    cacheHistory(tgjuKey, directResult, now);
   927→    return directResult;
   928→  }
   929→
   930→  // 2b: Queued page_reader with retry + rate limiting
   931→  console.log(`[TGJU] Direct fetch returned ${directResult?.length ?? 0} candles for ${tgjuKey}, using queued page_reader...`);
   932→  const pageReaderResult = await fetchViaQueuedPageReader(tgjuKey);
   933→  if (pageReaderResult && pageReaderResult.length >= 30) {
   934→    cacheHistory(tgjuKey, pageReaderResult, now);
   935→    return pageReaderResult;
   936→  }
   937→
   938→  // ── Step 3: All methods failed — return stale cache (only if fresh) ──
   939→  if (cached && cached.data.length >= 30 && isDataFresh(cached.data)) {
   940→    const age = ((now - cached.ts) / 60_000).toFixed(0);
   941→    console.warn(`[TGJU] All methods failed for ${tgjuKey}, returning stale cache (${age}min old, ${cached.data.length} candles)`);
   942→    return cached.data;
   943→  }
   944→
   945→  // ── Step 4: Try file-based cache ──
   946→  const fileCached = await readFromFileCache(tgjuKey);
   947→  if (fileCached && fileCached.length >= 30) {
   948→    console.warn(`[TGJU] Returning file-cached data for ${tgjuKey} (${fileCached.length} candles)`);
   949→    historyCache.set(tgjuKey, { data: fileCached, ts: Date.now() - HISTORY_TTL + 60_000 }); // Mark as almost-expired
   950→    return fileCached;
   951→  }
   952→
   953→  console.error(`[TGJU] All methods and caches failed for ${tgjuKey}`);
   954→  return [];
   955→}

   957→/**
   958→ * Try direct HTTP fetch to TGJU chart API.
   959→ * Returns parsed candles or null on failure.
   960→ */
   961→async function tryDirectFetch(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   962→  try {
   963→    // CRITICAL: order_dir=desc to get the MOST RECENT candles, not the oldest.
   964→    // With asc+start=0, we'd get data from 1392 for long-history instruments!
   965→    const url = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=desc&start=0&length=365`;
   966→    const controller = new AbortController();
   967→    const timer = setTimeout(() => controller.abort(), 12_000);
   968→
   969→    const res = await fetch(url, { signal: controller.signal, headers: { 'Accept': 'application/json' } });
   970→    clearTimeout(timer);
   971→
   972→    if (!res.ok) {
   973→      console.warn(`[TGJU] Direct fetch HTTP ${res.status} for ${tgjuKey}`);
   974→      return null;
   975→    }
   976→
   977→    const jsonStr = await res.text();
   978→    if (!jsonStr.startsWith('{')) {
   979→      console.warn(`[TGJU] Direct fetch non-JSON for ${tgjuKey}`);
   980→      return null;
   981→    }
   982→
   983→    return parseTgjuChartData(tgjuKey, jsonStr);
   984→  } catch (err) {
   985→    console.warn(`[TGJU] Direct fetch error for ${tgjuKey}:`, err instanceof Error ? err.message : err);
   986→    return null;
   987→  }
   988→}

   990→/**
   991→ * Fetch TGJU historical data via queued page_reader with retry and rate limiting.
   992→ * This prevents 429 errors by serializing requests and adding delays between calls.
   993→ */
   994→async function fetchViaQueuedPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   995→  // Enqueue this request — ensures only one page_reader call at a time
   996→  let result: TgjuOHLC[] | null = null;
   997→  const taskPromise = (async () => {
   998→    // Rate limit: wait if last call was too recent
   999→    const timeSinceLastCall = Date.now() - lastPageReaderCall;
  1000→    if (timeSinceLastCall < PAGE_READER_MIN_INTERVAL) {
  1001→      const waitTime = PAGE_READER_MIN_INTERVAL - timeSinceLastCall;
  1002→      console.log(`[TGJU] Rate limiting page_reader: waiting ${waitTime}ms...`);
  1003→      await new Promise(r => setTimeout(r, waitTime));
  1004→    }
  1005→
  1006→    // Retry with exponential backoff
  1007→    for (let attempt = 1; attempt <= PAGE_READER_MAX_RETRIES; attempt++) {
  1008→      try {
  1009→        lastPageReaderCall = Date.now();
  1010→        const fetchResult = await fetchTgjuHistoryViaPageReader(tgjuKey);
  1011→        if (fetchResult && fetchResult.length >= 30) {
  1012→          result = fetchResult;
  1013→          return; // success
  1014→        }
  1015→        // Got data but not enough candles — don't retry
  1016→        if (fetchResult && fetchResult.length > 0) {
  1017→          console.warn(`[TGJU] page_reader returned only ${fetchResult.length} candles for ${tgjuKey}`);
  1018→          return;
  1019→        }
  1020→      } catch (err) {
  1021→        const errMsg = err instanceof Error ? err.message : String(err);
  1022→        const isRateLimit = errMsg.includes('429') || errMsg.includes('Too many requests');
  1023→        if (isRateLimit && attempt < PAGE_READER_MAX_RETRIES) {
  1024→          const delay = PAGE_READER_RETRY_BASE_DELAY * Math.pow(3, attempt - 1); // 3s, 9s, 27s
  1025→          console.warn(`[TGJU] page_reader 429 for ${tgjuKey}, retry ${attempt}/${PAGE_READER_MAX_RETRIES} in ${delay}ms...`);
  1026→          await new Promise(r => setTimeout(r, delay));
  1027→          continue;
  1028→        }
  1029→        console.error(`[TGJU] page_reader error for ${tgjuKey} (attempt ${attempt}):`, errMsg);
  1030→        return;
  1031→      }
  1032→    }
  1033→  })();
  1034→
  1035→  // Chain onto the queue
  1036→  pageReaderQueue = pageReaderQueue.then(() => taskPromise, () => taskPromise);
  1037→  await pageReaderQueue;
  1038→  return result;
  1039→}

  1041→/**
  1042→ * Single page_reader call to fetch TGJU historical data.
  1043→ * Uses z-ai SDK to bypass Cloudflare challenges.
  1044→ * Throws on failure so the caller (fetchViaQueuedPageReader) can retry.
  1045→ */
  1046→async function fetchTgjuHistoryViaPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
  1047→  // CRITICAL: order_dir=desc to get the MOST RECENT candles, not the oldest.
  1048→  const apiUrl = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=desc&start=0&length=365`;
  1049→
  1050→  const ZAI = (await import('z-ai-web-dev-sdk')).default;
  1051→  const zai = await ZAI.create();
  1052→  console.log(`[TGJU] Using z-ai SDK page_reader for ${tgjuKey}...`);
  1053→  const invokeResult = await zai.functions.invoke('page_reader', { url: apiUrl });
  1054→
  1055→  if (invokeResult.code !== 200 || !invokeResult.data?.html) {
  1056→    const errMsg = invokeResult.error || `code=${invokeResult.code}`;
  1057→    throw new Error(errMsg);
  1058→  }
  1059→
  1060→  // Extract JSON from <pre> tag in the HTML
  1061→  const html = invokeResult.data.html;
  1062→  const preMatch = html.match(/<pre[^>]*>([\s\S]*?)<\/pre>/);
  1063→  if (!preMatch) {
  1064→    throw new Error('no <pre> block found in page_reader response');
  1065→  }
  1066→
  1067→  // Unescape HTML entities and strip HTML tags from values
  1068→  const jsonStr = preMatch[1]
  1069→    .replace(/<[^>]*>/g, '')        // Strip ALL HTML tags first
  1070→    .replace(/&quot;/g, '"')
  1071→    .replace(/&amp;/g, '&')
  1072→    .replace(/&lt;/g, '<')
  1073→    .replace(/&gt;/g, '>')
  1074→    .replace(/\s+/g, ' ')           // Normalize whitespace
  1075→    .trim();
  1076→
  1077→  if (!jsonStr.startsWith('{')) {
  1078→    throw new Error('extracted content is not JSON');
  1079→  }
  1080→
  1081→  return parseTgjuChartData(tgjuKey, jsonStr);
  1082→}

  1084→/**
  1085→ * Parse the JSON data from TGJU chart API response.
  1086→ * Used by both direct fetch and page_reader paths.
  1087→ */
  1088→function parseTgjuChartData(tgjuKey: string, jsonStr: string): TgjuOHLC[] | null {
  1089→  try {
  1090→    const parsed = JSON.parse(jsonStr);
  1091→    const rawRows: (string | number)[][] = parsed.data;
  1092→
  1093→    if (!rawRows || rawRows.length === 0) {
  1094→      console.warn(`[TGJU] Empty data array for ${tgjuKey}`);
  1095→      return null;
  1096→    }
  1097→
  1098→    // Sanitize row values: strip any residual HTML tags and trim
  1099→    const rows = rawRows.map(row =>
  1100→      row.map(cell => {
  1101→        const s = String(cell).trim();
  1102→        // Strip HTML tags if any remain (safety for direct fetch path)
  1103→        return s.replace(/<[^>]*>/g, '').replace(/\s+/g, ' ');
  1104→      })
  1105→    );
  1106→
  1107→    // ── Auto-detect column mapping from first valid row ──
  1108→    let colMap = detectColumnMap(rows[0]);
  1109→    if (!colMap) {
  1110→      for (let r = 1; r < Math.min(5, rows.length); r++) {
  1111→        colMap = detectColumnMap(rows[r]);
  1112→        if (colMap) break;
  1113→      }
  1114→    }
  1115→    if (!colMap) {
  1116→      console.warn(`[TGJU] Could not auto-detect columns for ${tgjuKey}, using default order`);
  1117→      colMap = { openIdx: 0, highIdx: 2, lowIdx: 1, closeIdx: 3, jalaliDateIdx: 7, gregorianDateIdx: 6 };
  1118→    } else {
  1119→      console.log(`[TGJU] Column map for ${tgjuKey}: O=${colMap.openIdx} H=${colMap.highIdx} L=${colMap.lowIdx} C=${colMap.closeIdx} Date=${colMap.jalaliDateIdx}`);
  1120→    }
  1121→
  1122→    const { openIdx, highIdx, lowIdx, closeIdx, jalaliDateIdx, gregorianDateIdx } = colMap;
  1123→
  1124→    // ── Parse rows ──
  1125→    const candles: TgjuOHLC[] = [];
  1126→    let invalidCount = 0;
  1127→
  1128→    for (const row of rows) {
  1129→      const rawDate = String(row[jalaliDateIdx] || row[gregorianDateIdx] || '');
  1130→      const date = rawDate.replace(/[\/]/g, '-');
  1131→      const o = parsePersianNum(row[openIdx]);
  1132→      const h = parsePersianNum(row[highIdx]);
  1133→      const l = parsePersianNum(row[lowIdx]);
  1134→      const c = parsePersianNum(row[closeIdx]);
  1135→
  1136→      if (!isValidCandle(o, h, l, c) || date.length === 0) {
  1137→        invalidCount++;
  1138→        continue;
  1139→      }
  1140→
  1141→      candles.push({ date, open: o, high: h, low: l, close: c });
  1142→    }
  1143→
  1144→    if (invalidCount > 0) {
  1145→      console.warn(`[TGJU] ${tgjuKey}: ${invalidCount}/${rows.length} invalid candles skipped`);
  1146→    }
  1147→
  1148→    if (candles.length === 0) {
  1149→      return null;
  1150→    }
  1151→
  1152→    // ── Detect and fix data drift ──
  1153→    let driftCount = 0;
  1154→    for (let i = 1; i < candles.length; i++) {
  1155→      const prevClose = candles[i - 1].close;
  1156→      const currClose = candles[i].close;
  1157→      if (prevClose > 0) {
  1158→        const pctChange = Math.abs(currClose - prevClose) / prevClose;
  1159→        if (pctChange > 0.5) {
  1160→          driftCount++;
  1161→        }
  1162→      }
  1163→    }
  1164→    if (driftCount > candles.length * 0.1) {
  1165→      console.warn(`[TGJU] ${tgjuKey}: Possible data drift detected (${driftCount} suspicious jumps out of ${candles.length} candles)`);
  1166→    }
  1167→
  1168→    // Reverse: API returns oldest-first (asc), we want newest-last for charting
  1169→    const firstDate = candles[0].date;
  1170→    const lastDate = candles[candles.length - 1].date;
  1171→    if (firstDate > lastDate) {
  1172→      candles.reverse();
  1173→    }
  1174→
  1175→    // ── Freshness validation: reject data that is too old (e.g. from 1392) ──
  1176→    if (!isDataFresh(candles)) {
  1177→      console.warn(`[TGJU] ${tgjuKey}: Data is STALE (latest candle: ${candles[candles.length - 1]?.date}), REJECTING`);
  1178→      return null;
  1179→    }
  1180→
  1181→    console.log(`[TGJU] ${tgjuKey}: Parsed ${candles.length} candles (${candles[0]?.date} to ${candles[candles.length - 1]?.date})`);
  1182→
  1183→    return candles;
  1184→  } catch (err) {
  1185→    console.error(`[TGJU] Failed to parse chart data for ${tgjuKey}:`, err);
  1186→    return null;
  1187→  }
  1188→}

  1190→/**
  1191→ * Cache parsed history data and update related caches.
  1192→ */
  1193→function cacheHistory(tgjuKey: string, candles: TgjuOHLC[], now: number): void {
  1194→  historyCache.set(tgjuKey, { data: candles, ts: now });
  1195→  // Also persist to file for server restart resilience
  1196→  writeToFileCache(tgjuKey, candles).catch(() => {});
  1197→
  1198→  // Also update static price cache for this instrument
  1199→  if (candles.length >= 2) {
  1200→    const last = candles[candles.length - 1];
  1201→    const prev = candles[candles.length - 2];
  1202→    staticPriceCache.set(tgjuKey, {
  1203→      close: last.close,
  1204→      high: last.high,
  1205→      low: last.low,
  1206→      prevClose: prev.close,
  1207→      ts: now,
  1208→    });
  1209→
  1210→    // Update live instruments cache if this is a static instrument
  1211→    if (liveInstrumentsCache.data) {
  1212→      const inst = liveInstrumentsCache.data.find((i) => i.key === tgjuKey);
  1213→      if (inst && inst.price === 0) {
  1214→        inst.price = last.close;
  1215→        inst.highPrice = last.high;
  1216→        inst.lowPrice = last.low;
  1217→        inst.change = last.close - prev.close;
  1218→        inst.changePercent = prev.close > 0 ? ((last.close - prev.close) / prev.close) * 100 : 0;
  1219→      }
  1220→    }
  1221→  }
  1222→}
