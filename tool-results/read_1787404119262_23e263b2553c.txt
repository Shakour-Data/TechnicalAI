   978→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   979→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   980→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   981→  grade?: string; overlapCount?: number; touchCount?: number;
   982→}) {
   983→  const isSupport = type === 'support';
   984→  const isTargetType = type === 'target';
   985→
   986→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   987→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   988→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   989→
   990→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   991→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   992→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   993→
   994→  return (
   995→    <div
   996→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   997→      style={{
   998→        background: bgAccent,
   999→        border: `1px solid ${borderAccent}`,
  1000→        borderRight: `3px solid ${accentColor}`,
  1001→      }}
  1002→    >
  1003→      {/* Label + Price */}
  1004→      <div className="flex-1 min-w-0">
  1005→        <div className="flex items-center gap-2">
  1006→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
  1007→          {isTarget && <span className="text-[9px]">🎯</span>}
  1008→          {overlapCount !== undefined && overlapCount > 0 && (
  1009→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
  1010→              {toPersianDigits(overlapCount)} هم‌پوشانی
  1011→            </span>
  1012→          )}
  1013→        </div>
  1014→        <div className="flex items-baseline gap-1">
  1015→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
  1016→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
  1017→        </div>
  1018→      </div>
  1019→
  1020→      {/* Strength + Grade */}
  1021→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
  1022→        <div className="flex items-center gap-1.5">
  1023→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
  1024→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
  1025→          <span className="text-[9px] text-gray-500">/۱۰</span>
  1026→        </div>
  1027→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
  1028→          <div
  1029→            className="h-full rounded-full transition-all duration-500"
  1030→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
  1031→          />
  1032→        </div>
  1033→        {touchCount !== undefined && touchCount > 0 && (
  1034→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
  1035→        )}
  1036→      </div>
  1037→    </div>
  1038→  );
  1039→}
