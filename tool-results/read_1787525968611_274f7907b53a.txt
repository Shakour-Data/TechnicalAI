   164→   164→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   165→   165→function sma(closes: number[], period: number): number {
   166→   166→  if (closes.length < period) return 0;
   167→   167→  const slice = closes.slice(-period);
   168→   168→  return slice.reduce((a, b) => a + b, 0) / period;
   169→   169→}

   171→   171→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   172→   172→function emaCalc(closes: number[], period: number): number {
   173→   173→  if (closes.length < period) return 0;
   174→   174→  const k = 2 / (period + 1);
   175→   175→  let emaVal = sma(closes.slice(0, period), period);
   176→   176→  for (let i = period; i < closes.length; i++) {
   177→   177→    emaVal = closes[i] * k + emaVal * (1 - k);
   178→   178→  }
   179→   179→  return emaVal;
   180→   180→}

   182→   182→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   183→   183→function calcRSI(closes: number[], period: number = 14): number {
   184→   184→  if (closes.length < period + 1) return 50;
   185→   185→  let avgGain = 0;
   186→   186→  let avgLoss = 0;
   187→   187→  for (let i = 1; i <= period; i++) {
   188→   188→    const diff = closes[i] - closes[i - 1];
   189→   189→    if (diff > 0) avgGain += diff;
   190→   190→    else avgLoss += Math.abs(diff);
   191→   191→  }
   192→   192→  avgGain /= period;
   193→   193→  avgLoss /= period;
   194→   194→  for (let i = period + 1; i < closes.length; i++) {
   195→   195→    const diff = closes[i] - closes[i - 1];
   196→   196→    const gain = diff > 0 ? diff : 0;
   197→   197→    const loss = diff < 0 ? Math.abs(diff) : 0;
   198→   198→    avgGain = (avgGain * (period - 1) + gain) / period;
   199→   199→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   200→   200→  }
   201→   201→  if (avgLoss === 0) return 100;
   202→   202→  const rs = avgGain / avgLoss;
   203→   203→  return 100 - 100 / (1 + rs);
   204→   204→}

   206→   206→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   207→   207→function calcMFI(data: OHLCV[], period: number = 14): number {
   208→   208→  if (data.length < period + 1) return 50;
   209→   209→  let posFlow = 0;
   210→   210→  let negFlow = 0;
   211→   211→  for (let i = data.length - period; i < data.length; i++) {
   212→   212→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   213→   213→    const mf = tp * data[i].volume;
   214→   214→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   215→   215→    if (tp > prevTp) posFlow += mf;
   216→   216→    else negFlow += mf;
   217→   217→  }
   218→   218→  if (negFlow === 0) return 100;
   219→   219→  const mfr = posFlow / negFlow;
   220→   220→  return 100 - 100 / (1 + mfr);
   221→   221→}

   224→   224→function calcCCI(data: OHLCV[], period: number = 20): number {
   225→   225→  if (data.length < period) return 0;
   226→   226→  const tps: number[] = [];
   227→   227→  for (let i = data.length - period; i < data.length; i++) {
   228→   228→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   229→   229→  }
   230→   230→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   231→   231→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   232→   232→  if (meanDev === 0) return 0;
   233→   233→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   234→   234→}

   237→   237→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   238→   238→  if (data.length < kPeriod) return { k: 50, d: 50 };
   239→   239→  const rawKs: number[] = [];
   240→   240→  for (let i = kPeriod - 1; i < data.length; i++) {
   241→   241→    let lowest = Infinity;
   242→   242→    let highest = -Infinity;
   243→   243→    for (let j = i - kPeriod + 1; j <= i; j++) {
   244→   244→      if (data[j].low < lowest) lowest = data[j].low;
   245→   245→      if (data[j].high > highest) highest = data[j].high;
   246→   246→    }
   247→   247→    const range = highest - lowest;
   248→   248→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   249→   249→  }
   250→   250→  const smoothedK: number[] = [];
   251→   251→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   252→   252→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   253→   253→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   254→   254→  }
   255→   255→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   256→   256→  const dSlice = smoothedK.slice(-smoothD);
   257→   257→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   258→   258→  return { k, d };
   259→   259→}

   262→   262→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   263→   263→  if (data.length < period) return -50;
   264→   264→  const slice = data.slice(-period);
   265→   265→  let highest = -Infinity;
   266→   266→  let lowest = Infinity;
   267→   267→  for (const c of slice) {
   268→   268→    if (c.high > highest) highest = c.high;
   269→   269→    if (c.low < lowest) lowest = c.low;
   270→   270→  }
   271→   271→  const range = highest - lowest;
   272→   272→  if (range === 0) return -50;
   273→   273→  return ((highest - data[data.length - 1].close) / range) * -100;
   274→   274→}

   276→   276→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   277→   277→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   278→   278→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   279→   279→  const emaFast = emaCalc(closes, fast);
   280→   280→  const emaSlow = emaCalc(closes, slow);
   281→   281→  const macdLine = emaFast - emaSlow;
   282→   282→  // Calculate signal as EMA of MACD values
   283→   283→  const macdValues: number[] = [];
   284→   284→  const kFast = 2 / (fast + 1);
   285→   285→  const kSlow = 2 / (slow + 1);
   286→   286→  let eFast = sma(closes.slice(0, fast), fast);
   287→   287→  let eSlow = sma(closes.slice(0, slow), slow);
   288→   288→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   289→   289→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   290→   290→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   291→   291→    macdValues.push(eFast - eSlow);
   292→   292→  }
   293→   293→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   294→   294→  const kSig = 2 / (sig + 1);
   295→   295→  let signalVal = sma(macdValues.slice(0, sig), sig);
   296→   296→  for (let i = sig; i < macdValues.length; i++) {
   297→   297→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   298→   298→  }
   299→   299→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   300→   300→}

   302→   302→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   303→   303→function calcATR(data: OHLCV[], period: number = 14): number {
   304→   304→  if (data.length < 2) return 0;
   305→   305→  const trs: number[] = [];
   306→   306→  for (let i = 1; i < data.length; i++) {
   307→   307→    const tr = Math.max(
   308→   308→      data[i].high - data[i].low,
   309→   309→      Math.abs(data[i].high - data[i - 1].close),
   310→   310→      Math.abs(data[i].low - data[i - 1].close)
   311→   311→    );
   312→   312→    trs.push(tr);
   313→   313→  }
   314→   314→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   315→   315→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   316→   316→  for (let i = period; i < trs.length; i++) {
   317→   317→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   318→   318→  }
   319→   319→  return atrVal;
   320→   320→}

   322→   322→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   323→   323→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   324→   324→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   325→   325→  const plusDMs: number[] = [];
   326→   326→  const minusDMs: number[] = [];
   327→   327→  const trs: number[] = [];
   328→   328→  for (let i = 1; i < data.length; i++) {
   329→   329→    const upMove = data[i].high - data[i - 1].high;
   330→   330→    const downMove = data[i - 1].low - data[i].low;
   331→   331→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   332→   332→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   333→   333→    trs.push(Math.max(
   334→   334→      data[i].high - data[i].low,
   335→   335→      Math.abs(data[i].high - data[i - 1].close),
   336→   336→      Math.abs(data[i].low - data[i - 1].close)
   337→   337→    ));
   338→   338→  }
   339→   339→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   340→   340→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   341→   341→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   342→   342→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   343→   343→  const dxValues: number[] = [];
   344→   344→  for (let i = period; i < trs.length; i++) {
   345→   345→    smoothTR = smoothTR - smoothTR / period + trs[i];
   346→   346→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   347→   347→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   348→   348→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   349→   349→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   350→   350→    const diSum = diPlus + diMinus;
   351→   351→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   352→   352→    dxValues.push(dx);
   353→   353→  }
   354→   354→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   355→   355→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   356→   356→  for (let i = period; i < dxValues.length; i++) {
   357→   357→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   358→   358→  }
   359→   359→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   360→   360→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   361→   361→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   362→   362→}

   364→   364→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   365→   365→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   366→   366→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   367→   367→  let isLong = true;
   368→   368→  let af = step;
   369→   369→  let ep = data[0].high;
   370→   370→  let sar = data[0].low;
   371→   371→  for (let i = 1; i < data.length; i++) {
   372→   372→    const prevSar = sar;
   373→   373→    sar = prevSar + af * (ep - prevSar);
   374→   374→    // Ensure SAR is within bounds
   375→   375→    if (isLong) {
   376→   376→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   377→   377→      if (data[i].low < sar) {
   378→   378→        isLong = false;
   379→   379→        sar = ep;
   380→   380→        ep = data[i].low;
   381→   381→        af = step;
   382→   382→      } else {
   383→   383→        if (data[i].high > ep) {
   384→   384→          ep = data[i].high;
   385→   385→          af = Math.min(af + step, max);
   386→   386→        }
   387→   387→      }
   388→   388→    } else {
   389→   389→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   390→   390→      if (data[i].high > sar) {
   391→   391→        isLong = true;
   392→   392→        sar = ep;
   393→   393→        ep = data[i].high;
   394→   394→        af = step;
   395→   395→      } else {
   396→   396→        if (data[i].low < ep) {
   397→   397→          ep = data[i].low;
   398→   398→          af = Math.min(af + step, max);
   399→   399→        }
   400→   400→      }
   401→   401→    }
   402→   402→  }
   403→   403→  return sar;
   404→   404→}

   406→   406→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   407→   407→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   408→   408→  if (closes.length < period) {
   409→   409→    const c = closes[closes.length - 1] ?? 0;
   410→   410→    return { upper: c, middle: c, lower: c };
   411→   411→  }
   412→   412→  const slice = closes.slice(-period);
   413→   413→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   414→   414→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   415→   415→  const sd = Math.sqrt(variance);
   416→   416→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   417→   417→}

   456→   456→function calcIchimoku(data: OHLCV[]): {
   457→   457→  tenkan: number; kijun: number; senkouA: number; senkouB: number; chikou: number;
   458→   458→} {
   459→   459→  if (data.length < 52) {
   460→   460→    const c = data[data.length - 1]?.close ?? 0;
   461→   461→    return { tenkan: c, kijun: c, senkouA: c, senkouB: c, chikou: c };
   462→   462→  }
   463→   463→  const highs = data.map(d => d.high);
   464→   464→  const lows = data.map(d => d.low);
   465→   465→  const closes = data.map(d => d.close);
   466→   466→
   467→   467→  function donchian(h: number[], l: number[], end: number, period: number): number {
   468→   468→    const start = end - period + 1;
   469→   469→    if (start < 0) return (h[end] + l[end]) / 2;
   470→   470→    let hi = -Infinity, lo = Infinity;
   471→   471→    for (let i = start; i <= end; i++) {
   472→   472→      if (h[i] > hi) hi = h[i];
   473→   473→      if (l[i] < lo) lo = l[i];
   474→   474→    }
   475→   475→    return (hi + lo) / 2;
   476→   476→  }
   477→   477→
   478→   478→  const n = data.length;
   479→   479→  const tenkan = donchian(highs, lows, n - 1, 9);
   480→   480→  const kijun = donchian(highs, lows, n - 1, 26);
   481→   481→  // Senkou A = (tenkan + kijun) / 2, shifted 26 periods ahead
   482→   482→  // We compute it at bar n-27 (shifted to current)
   483→   483→  const senkouAIdx = Math.max(n - 1 - 26, 0);
   484→   484→  const tA = donchian(highs, lows, senkouAIdx, 9);
   485→   485→  const kA = donchian(highs, lows, senkouAIdx, 26);
   486→   486→  const senkouA = (tA + kA) / 2;
   487→   487→  // Senkou B = donchian(52), shifted 26 periods ahead
   488→   488→  const senkouBIdx = Math.max(n - 1 - 26, 0);
   489→   489→  const senkouB = donchian(highs, lows, senkouBIdx, 52);
   490→   490→  // Chikou = current close plotted 26 periods back
   491→   491→  const chikou = closes[n - 1];
   492→   492→
   493→   493→  return { tenkan, kijun, senkouA, senkouB, chikou };
   494→   494→}

   497→   497→  tenkan: number[]; kijun: number[]; senkouA: number[]; senkouB: number[];
   498→   498→} {

   501→   501→  const lows = data.map(d => d.low);
   502→   502→
   503→   503→  function donchian(end: number, period: number): number {
   504→   504→    const start = end - period + 1;

   507→   507→    for (let i = start; i <= end; i++) {
   508→   508→      if (highs[i] > hi) hi = highs[i];

   511→   511→    return (hi + lo) / 2;

   516→   516→  const senkouA: number[] = [];

   523→   523→    const sIdx = i - 26;

   525→   525→      const t = donchian(sIdx, 9);
   526→   526→      const k = donchian(sIdx, 26);
   527→   527→      senkouA.push((t + k) / 2);
   528→   528→    } else {
   529→   529→      senkouA.push(donchian(i, 9)); // fallback
   530→   530→    }
   531→   531→    // Senkou B shifted +26: at bar i, we need donchian(52) at bar i-26
   532→   532→    if (sIdx >= 0) {
   533→   533→      senkouB.push(donchian(sIdx, 52));
   534→   534→    } else {
   535→   535→      senkouB.push(donchian(i, 52)); // fallback
   536→   536→    }
   537→   537→  }
   538→   538→
   539→   539→  return { tenkan, kijun, senkouA, senkouB };
   540→   540→}

   544→   544→  const result: number[] = [];

   554→   554→// ─── Helper: EMA Array (full series) ──────────────────────────────────────
   555→   555→function emaArrayCalc(closes: number[], period: number): number[] {
   556→   556→  const result: number[] = [];
   557→   557→  if (closes.length < period) return closes.map(() => 0);
   558→   558→  const k = 2 / (period + 1);
   559→   559→  let emaVal = 0;
   560→   560→  for (let i = 0; i < period; i++) emaVal += closes[i];
   561→   561→  emaVal /= period;
   562→   562→  for (let i = 0; i < closes.length; i++) {
   563→   563→    if (i < period) { result.push(0); continue; }
   564→   564→    if (i === period) { result.push(emaVal); continue; }
   565→   565→    emaVal = closes[i] * k + emaVal * (1 - k);
   566→   566→    result.push(emaVal);
   567→   567→  }
   568→   568→  return result;
   569→   569→}

   571→   571→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   572→   572→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   573→   573→  const n = values.length;
   574→   574→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   575→   575→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   576→   576→  for (let i = 0; i < n; i++) {
   577→   577→    sumX += i;
   578→   578→    sumY += values[i];
   579→   579→    sumXY += i * values[i];
   580→   580→    sumX2 += i * i;
   581→   581→    sumY2 += values[i] * values[i];
   582→   582→  }
   583→   583→  const denom = n * sumX2 - sumX * sumX;
   584→   584→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   585→   585→  const slope = (n * sumXY - sumX * sumY) / denom;
   586→   586→  const intercept = (sumY - slope * sumX) / n;
   587→   587→  const meanY = sumY / n;
   588→   588→  let ssRes = 0, ssTot = 0;
   589→   589→  for (let i = 0; i < n; i++) {
   590→   590→    const predicted = slope * i + intercept;
   591→   591→    ssRes += (values[i] - predicted) ** 2;
   592→   592→    ssTot += (values[i] - meanY) ** 2;
   593→   593→  }
   594→   594→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   595→   595→  return { slope, intercept, r2 };
   596→   596→}

   691→   691→// ─── Helper: Linear System Solver (Gaussian Elimination) ─────────────────
   692→   692→function solveLinearSystem(A: number[][], b: number[]): number[] | null {
   693→   693→  const n = b.length;
   694→   694→  const aug = A.map((row, i) => [...row, b[i]]);
   695→   695→
   696→   696→  for (let col = 0; col < n; col++) {
   697→   697→    let maxVal = Math.abs(aug[col][col]);
   698→   698→    let maxRow = col;
   699→   699→    for (let row = col + 1; row < n; row++) {
   700→   700→      if (Math.abs(aug[row][col]) > maxVal) {
   701→   701→        maxVal = Math.abs(aug[row][col]);
   702→   702→        maxRow = row;
   703→   703→      }
   704→   704→    }
   705→   705→    if (maxVal < 1e-10) return null;
   706→   706→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   707→   707→
   708→   708→    for (let row = col + 1; row < n; row++) {
   709→   709→      const factor = aug[row][col] / aug[col][col];
   710→   710→      for (let j = col; j <= n; j++) {
   711→   711→        aug[row][j] -= factor * aug[col][j];
   712→   712→      }
   713→   713→    }
   714→   714→  }
   715→   715→
   716→   716→  const x = new Array(n).fill(0);
   717→   717→  for (let i = n - 1; i >= 0; i--) {
   718→   718→    x[i] = aug[i][n];
   719→   719→    for (let j = i + 1; j < n; j++) {
   720→   720→      x[i] -= aug[i][j] * x[j];
   721→   721→    }
   722→   722→    if (Math.abs(aug[i][i]) < 1e-10) return null;
   723→   723→    x[i] /= aug[i][i];
   724→   724→  }
   725→   725→
   726→   726→  return x;
   727→   727→}

   729→   729→// ─── Helper: Generate All 7-Method S/R Levels ─────────────────────────────
   730→   730→function generateAllLevels(data: OHLCV[], endIdx: number): RawSRLevel[] {
   731→   731→  const levels: RawSRLevel[] = [];
   732→   732→  if (endIdx < 10) return levels;
   733→   733→
   734→   734→  const slice = data.slice(0, endIdx + 1);
   735→   735→  const currentPrice = slice[slice.length - 1].close;
   736→   736→  const closes = slice.map(d => d.close);
   737→   737→  const noFib: [string, string] = ['—', ''];
   738→   738→
   739→   739→  // ── Method 1: Swing High/Low (Price Action) ──────────────────────────
   740→   740→  for (const lookback of [3, 5, 7]) {
   741→   741→    const swings = findSwingLevels(slice, lookback);
   742→   742→    const tol = currentPrice * 0.005; // 0.5% tolerance
   743→   743→
   744→   744→    // Connect swing highs → resistance
   745→   745→    const usedHigh = new Set<number>();
   746→   746→    for (const h of swings.highs) {
   747→   747→      if (usedHigh.has(h.idx)) continue;
   748→   748→      let sum = h.price;
   749→   749→      let count = 1;
   750→   750→      for (const h2 of swings.highs) {
   751→   751→        if (h2 === h || usedHigh.has(h2.idx)) continue;
   752→   752→        if (Math.abs(h2.price - h.price) < tol) {
   753→   753→          sum += h2.price;
   754→   754→          count++;
   755→   755→          usedHigh.add(h2.idx);
   756→   756→        }
   757→   757→      }
   758→   758→      usedHigh.add(h.idx);
   759→   759→      if (sum / count > currentPrice * 1.005) {
   760→   760→        levels.push({ price: sum / count, method: 'Swing_High', type: 'resistance', ...noFib });
   761→   761→      }
   762→   762→    }
   763→   763→
   764→   764→    // Connect swing lows → support
   765→   765→    const usedLow = new Set<number>();
   766→   766→    for (const l of swings.lows) {
   767→   767→      if (usedLow.has(l.idx)) continue;
   768→   768→      let sum = l.price;
   769→   769→      let count = 1;
   770→   770→      for (const l2 of swings.lows) {
   771→   771→        if (l2 === l || usedLow.has(l2.idx)) continue;
   772→   772→        if (Math.abs(l2.price - l.price) < tol) {
   773→   773→          sum += l2.price;
   774→   774→          count++;
   775→   775→          usedLow.add(l2.idx);
   776→   776→        }
   777→   777→      }
   778→   778→      usedLow.add(l.idx);
   779→   779→      if (sum / count < currentPrice * 0.995) {
   780→   780→        levels.push({ price: sum / count, method: 'Swing_Low', type: 'support', ...noFib });
   781→   781→      }
   782→   782→    }
   783→   783→  }
   784→   784→
   785→   785→  // ── Method 2: SMA Dynamic S/R ──────────────────────────────────────
   786→   786→  for (const period of [20, 50, 100, 200]) {
   787→   787→    const smaVal = sma(closes, period);
   788→   788→    if (smaVal <= 0) continue;
   789→   789→    const type: 'support' | 'resistance' = currentPrice > smaVal ? 'support' : 'resistance';
   790→   790→    levels.push({ price: smaVal, method: `SMA_${period}`, type, ...noFib });
   791→   791→  }
   792→   792→
   793→   793→  // ── Method 3: Bollinger Bands ───────────────────────────────────────
   794→   794→  const bb = calcBollingerBands(closes);
   795→   795→  if (bb.upper > 0 && bb.lower > 0) {
   796→   796→    levels.push({ price: bb.upper, method: 'BB_Upper', type: 'resistance', ...noFib });
   797→   797→    levels.push({ price: bb.lower, method: 'BB_Lower', type: 'support', ...noFib });
   798→   798→  }
   799→   799→
   800→   800→  // ── Method 4: Fibonacci Retracement + Extension ─────────────────────
   801→   801→  interface SwingPair { high: number; highIdx: number; low: number; lowIdx: number; range: number }
   802→   802→  let bestSwing: SwingPair | null = null;
   803→   803→
   804→   804→  for (const lb of [3, 5, 7, 10, 15, 20]) {
   805→   805→    const sw = findSwingLevels(slice, lb);
   806→   806→    if (sw.highs.length === 0 || sw.lows.length === 0) continue;
   807→   807→    for (const h of sw.highs) {
   808→   808→      for (const l of sw.lows) {
   809→   809→        if (Math.abs(h.idx - l.idx) < 3) continue;
   810→   810→        const range = Math.abs(h.price - l.price);
   811→   811→        if (range / currentPrice < 0.03) continue;
   812→   812→        if (!bestSwing || range > bestSwing.range) {
   813→   813→          bestSwing = {
   814→   814→            high: Math.max(h.price, l.price),
   815→   815→            highIdx: h.price > l.price ? h.idx : l.idx,
   816→   816→            low: Math.min(h.price, l.price),
   817→   817→            lowIdx: h.price < l.price ? l.idx : h.idx,
   818→   818→            range,
   819→   819→          };
   820→   820→        }
   821→   821→      }
   822→   822→    }
   823→   823→  }
   824→   824→
   825→   825→  if (bestSwing) {
   826→   826→    const H = bestSwing.high;
   827→   827→    const L = bestSwing.low;
   828→   828→    const range = H - L;
   829→   829→
   830→   830→    for (const ratio of FIB_RETRACEMENTS) {
   831→   831→      const price = H - ratio * range;
   832→   832→      if (price <= 0) continue;
   833→   833→      const rounded = roundToNice(price);
   834→   834→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   835→   835→      levels.push({
   836→   836→        price: rounded, method: `Fibonacci_${ratio}`, type,
   837→   837→        fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   838→   838→      });
   839→   839→    }
   840→   840→
   841→   841→    for (const extRatio of FIB_EXTENSIONS) {
   842→   842→      const price = H + (extRatio - 1) * range;
   843→   843→      if (price <= 0) continue;
   844→   844→      const rounded = roundToNice(price);
   845→   845→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   846→   846→      levels.push({
   847→   847→        price: rounded, method: `Fibonacci_${extRatio}`, type,
   848→   848→        fibRatio: String(extRatio), fibLabel: FIB_LABELS[String(extRatio)] || '',
   849→   849→      });
   850→   850→    }
   851→   851→
   852→   852→    // Inverse extensions (below L)
   853→   853→    for (const extRatio of FIB_EXTENSIONS) {
   854→   854→      const price = L - (extRatio - 1) * range;
   855→   855→      if (price <= 0) continue;
   856→   856→      const rounded = roundToNice(price);
   857→   857→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   858→   858→      levels.push({
   859→   859→        price: rounded, method: `Fibonacci_${-extRatio}`, type,
   860→   860→        fibRatio: String(-extRatio), fibLabel: '',
   861→   861→      });
   862→   862→    }
   863→   863→
   864→   864→    // Secondary swing confluence (check overlap with secondary swings)
   865→   865→    const secondarySwings: SwingPair[] = [];
   866→   866→    for (const lb of [3, 5, 7]) {
   867→   867→      const sw = findSwingLevels(slice, lb);
   868→   868→      for (const h of sw.highs) {
   869→   869→        for (const l of sw.lows) {
   870→   870→          if (Math.abs(h.idx - l.idx) < 3) continue;
   871→   871→          const r = Math.abs(h.price - l.price);
   872→   872→          if (r / currentPrice < 0.05) continue;
   873→   873→          if (bestSwing && Math.abs(r - bestSwing.range) / bestSwing.range < 0.1) continue;
   874→   874→          secondarySwings.push({
   875→   875→            high: Math.max(h.price, l.price), highIdx: h.price > l.price ? h.idx : l.idx,
   876→   876→            low: Math.min(h.price, l.price), lowIdx: h.price < l.price ? l.idx : h.idx,
   877→   877→            range: r,
   878→   878→          });
   879→   879→        }
   880→   880→      }
   881→   881→    }
   882→   882→    // Secondary swings' fib levels are merged via confluence (1% rule in mergeLevels)
   883→   883→    const confluenceTol = currentPrice * 0.005;
   884→   884→    for (const sec of secondarySwings.slice(0, 3)) {
   885→   885→      const sH = sec.high, sL = sec.low, sR = sH - sL;
   886→   886→      for (const ratio of FIB_RETRACEMENTS) {
   887→   887→        const price = roundToNice(sH - ratio * sR);
   888→   888→        if (price <= 0) continue;
   889→   889→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   890→   890→        levels.push({
   891→   891→          price, method: `Fibonacci_${ratio}`, type,
   892→   892→          fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   893→   893→        });
   894→   894→      }
   895→   895→      for (const extR of FIB_EXTENSIONS) {
   896→   896→        const price = roundToNice(sH + (extR - 1) * sR);
   897→   897→        if (price <= 0) continue;
   898→   898→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   899→   899→        levels.push({
   900→   900→          price, method: `Fibonacci_${extR}`, type,
   901→   901→          fibRatio: String(extR), fibLabel: FIB_LABELS[String(extR)] || '',
   902→   902→        });
   903→   903→      }
   904→   904→    }
   905→   905→  }
   906→   906→
   907→   907→  // ── Method 5: Volume Profile (VAP) ──────────────────────────────────
   908→   908→  const dataHasVolume = slice.some(d => d.volume > 0);
   909→   909→  if (dataHasVolume) {
   910→   910→    let minP = Infinity, maxP = -Infinity;
   911→   911→    for (const d of slice) {
   912→   912→      if (d.low < minP) minP = d.low;
   913→   913→      if (d.high > maxP) maxP = d.high;
   914→   914→    }
   915→   915→    const numBins = 40;
   916→   916→    const binSize = Math.max((maxP - minP) / numBins, currentPrice * 0.001);
   917→   917→    const vap = new Array(numBins).fill(0);
   918→   918→    const binCenters: number[] = [];
   919→   919→
   920→   920→    for (let b = 0; b < numBins; b++) {
   921→   921→      binCenters.push(minP + (b + 0.5) * binSize);
   922→   922→    }
   923→   923→
   924→   924→    for (const d of slice) {
   925→   925→      for (let b = 0; b < numBins; b++) {
   926→   926→        if (d.low <= binCenters[b] && binCenters[b] <= d.high) {
   927→   927→          vap[b] += d.volume;
   928→   928→        }
   929→   929→      }
   930→   930→    }
   931→   931→
   932→   932→    const maxVAP = Math.max(...vap);
   933→   933→    if (maxVAP > 0) {
   934→   934→      // POC (Point of Control)
   935→   935→      let pocIdx = 0;
   936→   936→      for (let b = 1; b < numBins; b++) {
   937→   937→        if (vap[b] > vap[pocIdx]) pocIdx = b;
   938→   938→      }
   939→   939→      const pocPrice = binCenters[pocIdx];
   940→   940→      const pocType: 'support' | 'resistance' = pocPrice > currentPrice ? 'resistance' : 'support';
   941→   941→      levels.push({ price: pocPrice, method: 'VAP_POC', type: pocType, ...noFib });
   942→   942→
   943→   943→      // HVN (High Volume Nodes)
   944→   944→      for (let b = 0; b < numBins; b++) {
   945→   945→        if (b === pocIdx) continue;
   946→   946→        if (vap[b] > 0.7 * maxVAP) {
   947→   947→          const hvnPrice = binCenters[b];
   948→   948→          const hvnType: 'support' | 'resistance' = hvnPrice > currentPrice ? 'resistance' : 'support';
   949→   949→          levels.push({ price: hvnPrice, method: 'VAP_HVN', type: hvnType, ...noFib });
   950→   950→        }
   951→   951→      }
   952→   952→    }
   953→   953→  }
   954→   954→
   955→   955→  // ── Method 6: Pivot Points (4 algorithms) ───────────────────────────
   956→   956→  if (endIdx >= 1) {
   957→   957→    const prev = slice[endIdx - 1]; // Previous completed bar
   958→   958→    const pH = prev.high, pL = prev.low, pC = prev.close, pO = prev.open;
   959→   959→
   960→   960→    // Standard Pivot
   961→   961→    {
   962→   962→      const PP = (pH + pL + pC) / 3;
   963→   963→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   964→   964→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   965→   965→      const R3 = pH + 2 * (PP - pL), S3 = pL - 2 * (pH - PP);
   966→   966→ if (R1 > 0) levels.push({ price: R1, method: 'Pivot_R1', type: 'resistance', ...noFib });
   967→   967→      if (R2 > 0) levels.push({ price: R2, method: 'Pivot_R2', type: 'resistance', ...noFib });
   968→   968→      if (R3 > 0) levels.push({ price: R3, method: 'Pivot_R3', type: 'resistance', ...noFib });
   969→   969→      if (S1 > 0) levels.push({ price: S1, method: 'Pivot_S1', type: 'support', ...noFib });
   970→   970→      if (S2 > 0) levels.push({ price: S2, method: 'Pivot_S2', type: 'support', ...noFib });
   971→   971→      if (S3 > 0) levels.push({ price: S3, method: 'Pivot_S3', type: 'support', ...noFib });
   972→   972→    }
   973→   973→
   974→   974→    // Fibonacci Pivot
   975→   975→    {
   976→   976→      const PP = (pH + pL + pC) / 3;
   977→   977→      const range = pH - pL;
   978→   978→      for (const [mult, suffix] of [[0.382, '1'], [0.618, '2'], [1.0, '3']] as [number, string][]) {
   979→   979→        const R = PP + mult * range;
   980→   980→        const S = PP - mult * range;
   981→   981→        if (R > 0) levels.push({ price: R, method: `Fib_Pivot_R${suffix}`, type: 'resistance', ...noFib });
   982→   982→        if (S > 0) levels.push({ price: S, method: `Fib_Pivot_S${suffix}`, type: 'support', ...noFib });
   983→   983→      }
   984→   984→    }
   985→   985→
   986→   986→    // Woodie's Pivot
   987→   987→    {
   988→   988→      const PP = (pH + pL + 2 * pC) / 4;
   989→   989→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   990→   990→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   991→   991→      if (R1 > 0) levels.push({ price: R1, method: 'Woodie_R1', type: 'resistance', ...noFib });
   992→   992→      if (R2 > 0) levels.push({ price: R2, method: 'Woodie_R2', type: 'resistance', ...noFib });
   993→   993→      if (S1 > 0) levels.push({ price: S1, method: 'Woodie_S1', type: 'support', ...noFib });
   994→   994→      if (S2 > 0) levels.push({ price: S2, method: 'Woodie_S2', type: 'support', ...noFib });
   995→   995→    }
   996→   996→
   997→   997→    // DeMark Pivot
   998→   998→    {
   999→   999→      const X = pC < pO ? pH + 2 * pL + pC : pC > pO ? 2 * pH + pL + pC : pH + pL + 2 * pC;
  1000→  1000→      const PP = X / 4;
  1001→  1001→      const R1 = X / 2 - pL;
  1002→  1002→      const S1 = X / 2 - pH;
  1003→  1003→      if (R1 > 0) levels.push({ price: R1, method: 'DeMark_R1', type: 'resistance', ...noFib });
  1004→  1004→      if (S1 > 0) levels.push({ price: S1, method: 'DeMark_S1', type: 'support', ...noFib });
  1005→  1005→    }
  1006→  1006→  }
  1007→  1007→
  1008→  1008→  // ── Method 7: Psychological Levels ──────────────────────────────────
  1009→  1009→  {
  1010→  1010→    const step = psychStep(currentPrice);
  1011→  1011→    const base = psychRound(currentPrice);
  1012→  1012→    for (const mult of [-2, -1, 1, 2]) {
  1013→  1013→      const price = base + mult * step;
  1014→  1014→      if (price <= 0) continue;
  1015→  1015→      const type: 'support' | 'resistance' = price > currentPrice ? 'resistance' : 'support';
  1016→  1016→      levels.push({ price, method: 'Psychological', type, ...noFib });
  1017→  1017→    }
  1018→  1018→  }
  1019→  1019→
  1020→  1020→  return levels;
  1021→  1021→}

  1023→  1023→// ─── Helper: Merge Nearby Levels (1% Confluence) ──────────────────────────
  1024→  1024→function mergeLevels(rawLevels: RawSRLevel[], currentPrice: number): MergedSRLevel[] {
  1025→  1025→  if (rawLevels.length === 0) return [];
  1026→  1026→
  1027→  1027→  const tol = currentPrice * 0.01; // 1% tolerance
  1028→  1028→  const merged: MergedSRLevel[] = [];
  1029→  1029→  const used = new Set<number>();
  1030→  1030→
  1031→  1031→  for (let i = 0; i < rawLevels.length; i++) {
  1032→  1032→    if (used.has(i)) continue;
  1033→  1033→
  1034→  1034→    let sumPrice = rawLevels[i].price;
  1035→  1035→    let count = 1;
  1036→  1036→    const methods = new Set<string>();
  1037→  1037→    methods.add(rawLevels[i].method);
  1038→  1038→    let bestFibRatio = rawLevels[i].fibRatio;
  1039→  1039→    let bestFibLabel = rawLevels[i].fibLabel;
  1040→  1040→
  1041→  1041→    for (let j = i + 1; j < rawLevels.length; j++) {
  1042→  1042→      if (used.has(j)) continue;
  1043→  1043→      if (Math.abs(rawLevels[j].price - rawLevels[i].price) < tol) {
  1044→  1044→        sumPrice += rawLevels[j].price;
  1045→  1045→        count++;
  1046→  1046→        methods.add(rawLevels[j].method);
  1047→  1047→        used.add(j);
  1048→  1048→        // Prefer Fibonacci label if available
  1049→  1049→        if (rawLevels[j].fibRatio !== '—' && bestFibRatio === '—') {
  1050→  1050→          bestFibRatio = rawLevels[j].fibRatio;
  1051→  1051→          bestFibLabel = rawLevels[j].fibLabel;
  1052→  1052→        }
  1053→  1053→      }
  1054→  1054→    }
  1055→  1055→    used.add(i);
  1056→  1056→
  1057→  1057→    const avgPrice = sumPrice / count;
  1058→  1058→    const type: 'support' | 'resistance' = avgPrice >= currentPrice ? 'resistance' : 'support';
  1059→  1059→
  1060→  1060→    merged.push({
  1061→  1061→      price: avgPrice,
  1062→  1062→      type,
  1063→  1063→      methods: [...methods],
  1064→  1064→      overlapCount: methods.size,
  1065→  1065→      fibRatio: bestFibRatio,
  1066→  1066→      fibLabel: bestFibLabel,
  1067→  1067→      score: 0,
  1068→  1068→      strength: 1,
  1069→  1069→      touchCount: 0,
  1070→  1070→      volumeRatio: 0,
  1071→  1071→      freshness: 0,
  1072→  1072→      distancePercent: 0,
  1073→  1073→    });
  1074→  1074→  }
  1075→  1075→
  1076→  1076→  return merged;
  1077→  1077→}

  1079→  1079→// ─── Helper: Compute Level Features (5 ML features) ──────────────────────
  1080→  1080→function computeLevelFeatures(
  1081→  1081→  levelPrice: number,
  1082→  1082→  data: OHLCV[],
  1083→  1083→  endIdx: number,
  1084→  1084→  currentPrice: number,
  1085→  1085→  overlapCount: number,
  1086→  1086→): number[] {
  1087→  1087→  const lookback = Math.min(30, endIdx);
  1088→  1088→  const startIdx = endIdx - lookback;
  1089→  1089→  const touchTol = currentPrice * 0.002; // 0.2%
  1090→  1090→
  1091→  1091→  let touchCount = 0;
  1092→  1092→  let totalVolAtTouches = 0;
  1093→  1093→  let totalVol = 0;
  1094→  1094→  let barsSinceLastTouch = lookback;
  1095→  1095→
  1096→  1096→  for (let i = startIdx; i <= endIdx; i++) {
  1097→  1097→    const bar = data[i];
  1098→  1098→    totalVol += bar.volume;
  1099→  1099→
  1100→  1100→    if (bar.low <= levelPrice + touchTol && bar.high >= levelPrice - touchTol) {
  1101→  1101→      touchCount++;
  1102→  1102→      totalVolAtTouches += bar.volume;
  1103→  1103→      barsSinceLastTouch = endIdx - i;
  1104→  1104→    }
  1105→  1105→  }
  1106→  1106→
  1107→  1107→  const avgVol = lookback > 0 ? totalVol / lookback : 0;
  1108→  1108→  const volumeRatio = (touchCount > 0 && avgVol > 0) ? (totalVolAtTouches / touchCount) / avgVol : 1;
  1109→  1109→  const freshness = Math.max(0, Math.min(1, 1 - barsSinceLastTouch / 30));
  1110→  1110→  const distancePercent = Math.min(1, Math.abs(levelPrice - currentPrice) / currentPrice);
  1111→  1111→
  1112→  1112→  // Features: [touch_count, volume_ratio, overlap_count, freshness, distance_percent]
  1113→  1113→  return [touchCount, volumeRatio, overlapCount - 1, freshness, distancePercent];
  1114→  1114→}

  1116→  1116→// ─── Helper: Train S/R Scoring Weights (Ridge OLS) ───────────────────────
  1117→  1117→function trainSRWeights(data: OHLCV[]): number[] {
  1118→  1118→  const defaultWeights = [0.25, 0.20, 0.25, 0.15, 0.15];
  1119→  1119→
  1120→  1120→  if (data.length < 50) return defaultWeights;
  1121→  1121→
  1122→  1122→  const X: number[][] = [];
  1123→  1123→  const y: number[] = [];
  1124→  1124→
  1125→  1125→  for (let i = 40; i <= data.length - 11; i += 5) {
  1126→  1126→    const barPrice = data[i].close;
  1127→  1127→    const rawLevels = generateAllLevels(data, i);
  1128→  1128→    if (rawLevels.length === 0) continue;
  1129→  1129→    const merged = mergeLevels(rawLevels, barPrice);
  1130→  1130→
  1131→  1131→    for (const level of merged) {
  1132→  1132→      const features = computeLevelFeatures(level.price, data, i, barPrice, level.overlapCount);
  1133→  1133→      X.push(features);
  1134→  1134→
  1135→  1135→      // Label: did price return within 10 bars?
  1136→  1136→      let returned = false;
  1137→  1137→      const retTol = barPrice * 0.005;
  1138→  1138→      for (let j = i + 1; j <= Math.min(i + 10, data.length - 1); j++) {
  1139→  1139→        if (data[j].low <= level.price + retTol && data[j].high >= level.price - retTol) {
  1140→  1140→          returned = true;
  1141→  1141→          break;
  1142→  1142→        }
  1143→  1143→      }
  1144→  1144→      y.push(returned ? 1 : 0);
  1145→  1145→    }
  1146→  1146→  }
  1147→  1147→
  1148→  1148→  if (X.length < 20) return defaultWeights;
  1149→  1149→
  1150→  1150→  // Ridge OLS: w = (X^T X + λI)^(-1) X^T y
  1151→  1151→  const n = X.length;
  1152→  1152→  const p = 5;
  1153→  1153→  const lambda = 0.01;
  1154→  1154→
  1155→  1155→  const XtX: number[][] = Array.from({ length: p }, () => new Array(p).fill(0));
  1156→  1156→  for (let i = 0; i < n; i++) {
  1157→  1157→    for (let a = 0; a < p; a++) {
  1158→  1158→      for (let b = 0; b < p; b++) {
  1159→  1159→        XtX[a][b] += X[i][a] * X[i][b];
  1160→  1160→      }
  1161→  1161→    }
  1162→  1162→  }
  1163→  1163→  for (let a = 0; a < p; a++) XtX[a][a] += lambda;
  1164→  1164→
  1165→  1165→  const Xty: number[] = new Array(p).fill(0);
  1166→  1166→  for (let i = 0; i < n; i++) {
  1167→  1167→    for (let a = 0; a < p; a++) {
  1168→  1168→      Xty[a] += X[i][a] * y[i];
  1169→  1169→    }
  1170→  1170→  }
  1171→  1171→
  1172→  1172→  const weights = solveLinearSystem(XtX, Xty);
  1173→  1173→  if (!weights) return defaultWeights;
  1174→  1174→  if (weights.some(w => isNaN(w) || !isFinite(w))) return defaultWeights;
  1175→  1175→
  1176→  1176→  for (let i = 0; i < p; i++) {
  1177→  1177→    if (weights[i] < 0) weights[i] = 0;
  1178→  1178→  }
  1179→  1179→
  1180→  1180→  const sum = weights.reduce((a, b) => a + b, 0);
  1181→  1181→  if (sum <= 0) return defaultWeights;
  1182→  1182→
  1183→  1183→  return weights.map(w => w / sum);
  1184→  1184→}

  1186→  1186→// ─── Helper: WMA Array (full series) ─────────────────────────────────────
  1187→  1187→function calcWMAArray(closes: number[], period: number): number[] {
  1188→  1188→  const result: number[] = [];
  1189→  1189→  if (closes.length < period) return closes.map(() => 0);
  1190→  1190→  for (let i = 0; i < closes.length; i++) {
  1191→  1191→    if (i < period - 1) { result.push(0); continue; }
  1192→  1192→    let numer = 0;
  1193→  1193→    let denom = 0;
  1194→  1194→    for (let j = 0; j < period; j++) {
  1195→  1195→      const weight = period - j;
  1196→  1196→      numer += closes[i - period + 1 + j] * weight;
  1197→  1197→      denom += weight;
  1198→  1198→    }
  1199→  1199→    result.push(numer / denom);
  1200→  1200→  }
  1201→  1201→  return result;
  1202→  1202→}

  1204→  1204→// ─── Helper: WMA (latest value) ────────────────────────────────────
  1205→  1205→function calcWMA(closes: number[], period: number): number {
  1206→  1206→  if (closes.length < period) return 0;
  1207→  1207→  const slice = closes.slice(-period);
  1208→  1208→  let numer = 0;
  1209→  1209→  let denom = 0;
  1210→  1210→  for (let i = 0; i < period; i++) {
  1211→  1211→    const weight = period - i;

  1319→  1319→function calcFisherTransform(data: OHLCV[], period: number = 9): number {
  1320→  1320→  if (data.length < period) return 0;

  1322→  1322→  for (let i = period; i < data.length; i++) {
  1323→  1323→    let maxH = -Infinity;
  1324→  1324→    let minL = Infinity;
  1325→  1325→    for (let j = i - period + 1; j <= i; j++) {
  1326→  1326→      if (data[j].high > maxH) maxH = data[j].high;
  1327→  1327→      if (data[j].low < minL) minL = data[j].low;
  1328→  1328→    }
  1329→  1329→    const range = maxH - minL;

  1332→  1332→    const val = Math.max(0.0001, Math.min(0.9999, (raw + 1) / 2));
  1333→  1333→    fisher = 0.5 * Math.log(val / (1 - val)) + 0.5 * fisher;
  1334→  1334→  }

  1576→  1576→      type === 'support' ? b.price - a.price : a.price - b.price

  1578→  1578→
  1579→  1579→    // ── Rule 2/3: Nearest level must be within 10% of price ──
  1580→  1580→    const nearest = sorted[0];
  1581→  1581→    const distFromPrice = Math.abs(nearest.price - currentPrice) / currentPrice;
  1582→  1582→    if (distFromPrice > 0.10) {
  1583→  1583→      // Nearest level is too far — generate a synthetic level at ~5% from price
  1584→  1584→      const syntheticPrice = type === 'support'
  1585→  1585→        ? Math.round(currentPrice * 0.95 / psychStep(currentPrice)) * psychStep(currentPrice)
  1586→  1586→        : Math.round(currentPrice * 1.05 / psychStep(currentPrice)) * psychStep(currentPrice);
  1587→  1587→      const synthetic: MergedSRLevel = {
  1588→  1588→        price: syntheticPrice,
  1589→  1589→        type,
  1590→  1590→        score: 3,
  1591→  1591→        strength: 3,
  1592→  1592→        overlapCount: 1,
  1593→  1593→        methods: ['Psychological'],
  1594→  1594→        fibRatio: '—',
  1595→  1595→        fibLabel: '',
  1596→  1596→      };
  1597→  1597→      sorted.unshift(synthetic);
  1598→  1598→    }
  1599→  1599→
  1600→  1600→    // ── Rule 1: Select levels with 5-10% gaps between consecutive ──
  1601→  1601→    const result: MergedSRLevel[] = [sorted[0]];
  1602→  1602→
  1603→  1603→    for (let i = 1; i < sorted.length && result.length < 6; i++) {
  1604→  1604→      const prevPrice = result[result.length - 1].price;
  1605→  1605→      const currPrice = sorted[i].price;
  1606→  1606→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
  1607→  1607→
  1608→  1608→      if (gap < 0.05) {
  1609→  1609→        // Too close — keep the stronger one
  1610→  1610→        if (sorted[i].score > result[result.length - 1].score) {
  1611→  1611→          result[result.length - 1] = sorted[i];
  1612→  1612→        }
  1613→  1613→      } else if (gap <= 0.10) {
  1614→  1614→        // Perfect gap (5-10%) — accept
  1615→  1615→        result.push(sorted[i]);
  1616→  1616→      } else {
  1617→  1617→        // Gap > 10% — try to find an intermediate level
  1618→  1618→        let found = false;
  1619→  1619→        for (let j = i + 1; j < sorted.length; j++) {
  1620→  1620→          const midPrice = sorted[j].price;
  1621→  1621→          const gapFromPrev = Math.abs(midPrice - prevPrice) / prevPrice;
  1622→  1622→          const gapToNext = Math.abs(currPrice - midPrice) / midPrice;
  1623→  1623→          if (gapFromPrev >= 0.05 && gapFromPrev <= 0.10) {
  1624→  1624→            result.push(sorted[j]);
  1625→  1625→            found = true;
  1626→  1626→            // Skip levels that are too close to the inserted one
  1627→  1627→            i = j;
  1628→  1628→            break;
  1629→  1629→          }
  1630→  1630→        }
  1631→  1631→        if (!found) {
  1632→  1632→          // No intermediate found — generate a synthetic level at ~7.5% from prev
  1633→  1633→          const synthGap = 0.075;
  1634→  1634→          const syntheticPrice = type === 'support'
  1635→  1635→            ? Math.round((prevPrice * (1 - synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice)
  1636→  1636→            : Math.round((prevPrice * (1 + synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice);
  1637→  1637→          // Make sure it's not too close to next level
  1638→  1638→          const gapToNext = Math.abs(currPrice - syntheticPrice) / syntheticPrice;
  1639→  1639→          if (gapToNext >= 0.05) {
  1640→  1640→            const synthetic: MergedSRLevel = {
  1641→  1641→              price: syntheticPrice,
  1642→  1642→              type,
  1643→  1643→              score: 2,
  1644→  1644→              strength: 2,
  1645→  1645→              overlapCount: 1,
  1646→  1646→              methods: ['Psychological'],
  1647→  1647→              fibRatio: '—',
  1648→  1648→              fibLabel: '',
  1649→  1649→            };
  1650→  1650→            result.push(synthetic);
  1651→  1651→          }
  1652→  1652→        }
  1653→  1653→      }
  1654→  1654→    }
  1655→  1655→
  1656→  1656→    return result;
  1657→  1657→  };
  1658→  1658→
  1659→  1659→  const finalSupports = filterAndSelectLevels(supports, 'support');
  1660→  1660→  const finalResistances = filterAndSelectLevels(resistances, 'resistance');
  1661→  1661→
  1662→  1662→  // ── Step 9: Build LevelStrength objects ────────────────────────────────
  1663→  1663→  const toLevelStrength = (m: MergedSRLevel): LevelStrength => {
  1664→  1664→    let grade: string;
  1665→  1665→    if (m.score >= 8.5) grade = 'Very Strong';
  1666→  1666→    else if (m.score >= 7) grade = 'Strong';
  1667→  1667→    else if (m.score >= 5) grade = 'Moderate';
  1668→  1668→    else grade = 'Weak';
  1669→  1669→
  1670→  1670→    return {
  1671→  1671→      price: m.price,
  1672→  1672→      strength: m.strength,
  1673→  1673→      score: m.score,
  1674→  1674→      grade,
  1675→  1675→      isTarget: m.score >= 7,
  1676→  1676→      overlapCount: m.overlapCount,
  1677→  1677→      methods: m.methods,
  1678→  1678→      fibRatio: m.fibRatio,
  1679→  1679→      fibLabel: m.fibLabel,
  1680→  1680→    };
  1681→  1681→  };
  1682→  1682→
  1683→  1683→  const supportStrengths = finalSupports.map(toLevelStrength);
  1684→  1684→  const resistanceStrengths = finalResistances.map(toLevelStrength);
  1685→  1685→
  1686→  1686→  // Price targets: all levels with isTarget=true
  1687→  1687→  const priceTargets = [...supportStrengths, ...resistanceStrengths].filter(l => l.isTarget);
  1688→  1688→
  1689→  1689→  return {
  1690→  1690→    resistances: resistanceStrengths.map(l => l.price),
  1691→  1691→    supports: supportStrengths.map(l => l.price),
  1692→  1692→    supportStrengths,
  1693→  1693→    resistanceStrengths,
  1694→  1694→    priceTargets,
  1695→  1695→  };
  1696→  1696→}

  1700→  1700→function calcTrend(closes: number[], period: number): TrendResult {
  1701→  1701→  const slice = closes.slice(-Math.min(period, closes.length));
  1702→  1702→  const { slope, r2 } = linearRegression(slice);
  1703→  1703→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
  1704→  1704→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
  1705→  1705→  const priceThreshold = avgPrice * 0.001; // 0.1%
  1706→  1706→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
  1707→  1707→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
  1708→  1708→}

  1715→  1715→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
  1716→  1716→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
  1717→  1717→  const slice = data.slice(0, barIdx + 1);
  1718→  1718→  const price = slice[slice.length - 1].close;
  1719→  1719→  let nearestR = price * 1.05;
  1720→  1720→  let nearestS = price * 0.95;
  1721→  1721→
  1722→  1722→  // Collect all swing highs (resistance) and swing lows (support)
  1723→  1723→  const resistances: number[] = [];
  1724→  1724→  const supports: number[] = [];
  1725→  1725→  for (const lb of [3, 5, 7]) {
  1726→  1726→    const swings = findSwingLevels(slice, lb);
  1727→  1727→    resistances.push(...swings.highs);
  1728→  1728→    supports.push(...swings.lows);
  1729→  1729→  }
  1730→  1730→
  1731→  1731→  // Find nearest resistance above price
  1732→  1732→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
  1733→  1733→  if (abovePrice.length > 0) nearestR = abovePrice[0];
  1734→  1734→
  1735→  1735→  // Find nearest support below price
  1736→  1736→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
  1737→  1737→  if (belowPrice.length > 0) nearestS = belowPrice[0];
  1738→  1738→
  1739→  1739→  return { R1: nearestR, S1: nearestS };
  1740→  1740→}

  1742→  1742→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
  1743→  1743→// Returns true if bullish divergence detected, false if bearish, null if none
  1744→  1744→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
  1745→  1745→  const lookback = 20;
  1746→  1746→  const start = Math.max(0, barIdx - lookback);
  1747→  1747→  const slice = data.slice(start, barIdx + 1);
  1748→  1748→  if (slice.length < 10) return null;
  1749→  1749→
  1750→  1750→  const mid = Math.floor(slice.length / 2);
  1751→  1751→  const firstHalf = slice.slice(0, mid);
  1752→  1752→  const secondHalf = slice.slice(mid);
  1753→  1753→
  1754→  1754→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
  1755→  1755→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
  1756→  1756→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
  1757→  1757→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
  1758→  1758→
  1759→  1759→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
  1760→  1760→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
  1761→  1761→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
  1762→  1762→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
  1763→  1763→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
  1764→  1764→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
  1765→  1765→
  1766→  1766→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
  1767→  1767→    return false; // bearish divergence
  1768→  1768→  }
  1769→  1769→
  1770→  1770→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
  1771→  1771→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
  1772→  1772→
  1773→  1773→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
  1774→  1774→    return true; // bullish divergence
  1775→  1775→  }
  1776→  1776→
  1777→  1777→  return null;
  1778→  1778→}
