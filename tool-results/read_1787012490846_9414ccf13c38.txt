    97→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
    98→function sma(closes: number[], period: number): number {
    99→  if (closes.length < period) return 0;
   100→  const slice = closes.slice(-period);
   101→  return slice.reduce((a, b) => a + b, 0) / period;
   102→}

   104→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   105→function emaCalc(closes: number[], period: number): number {
   106→  if (closes.length < period) return 0;
   107→  const k = 2 / (period + 1);
   108→  let emaVal = sma(closes.slice(0, period), period);
   109→  for (let i = period; i < closes.length; i++) {
   110→    emaVal = closes[i] * k + emaVal * (1 - k);
   111→  }
   112→  return emaVal;
   113→}

   115→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   116→function calcRSI(closes: number[], period: number = 14): number {
   117→  if (closes.length < period + 1) return 50;
   118→  let avgGain = 0;
   119→  let avgLoss = 0;
   120→  for (let i = 1; i <= period; i++) {
   121→    const diff = closes[i] - closes[i - 1];
   122→    if (diff > 0) avgGain += diff;
   123→    else avgLoss += Math.abs(diff);
   124→  }
   125→  avgGain /= period;
   126→  avgLoss /= period;
   127→  for (let i = period + 1; i < closes.length; i++) {
   128→    const diff = closes[i] - closes[i - 1];
   129→    const gain = diff > 0 ? diff : 0;
   130→    const loss = diff < 0 ? Math.abs(diff) : 0;
   131→    avgGain = (avgGain * (period - 1) + gain) / period;
   132→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   133→  }
   134→  if (avgLoss === 0) return 100;
   135→  const rs = avgGain / avgLoss;
   136→  return 100 - 100 / (1 + rs);
   137→}

   139→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   140→function calcMFI(data: OHLCV[], period: number = 14): number {
   141→  if (data.length < period + 1) return 50;
   142→  let posFlow = 0;
   143→  let negFlow = 0;
   144→  for (let i = data.length - period; i < data.length; i++) {
   145→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   146→    const mf = tp * data[i].volume;
   147→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   148→    if (tp > prevTp) posFlow += mf;
   149→    else negFlow += mf;
   150→  }
   151→  if (negFlow === 0) return 100;
   152→  const mfr = posFlow / negFlow;
   153→  return 100 - 100 / (1 + mfr);
   154→}

   156→// ─── Helper: CCI(20) ──────────────────────────────────────────────────────────
   157→function calcCCI(data: OHLCV[], period: number = 20): number {
   158→  if (data.length < period) return 0;
   159→  const tps: number[] = [];
   160→  for (let i = data.length - period; i < data.length; i++) {
   161→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   162→  }
   163→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   164→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   165→  if (meanDev === 0) return 0;
   166→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   167→}

   169→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   170→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   171→  if (data.length < kPeriod) return { k: 50, d: 50 };
   172→  const rawKs: number[] = [];
   173→  for (let i = kPeriod - 1; i < data.length; i++) {
   174→    let lowest = Infinity;
   175→    let highest = -Infinity;
   176→    for (let j = i - kPeriod + 1; j <= i; j++) {
   177→      if (data[j].low < lowest) lowest = data[j].low;
   178→      if (data[j].high > highest) highest = data[j].high;
   179→    }
   180→    const range = highest - lowest;
   181→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   182→  }
   183→  const smoothedK: number[] = [];
   184→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   185→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   186→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   187→  }
   188→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   189→  const dSlice = smoothedK.slice(-smoothD);
   190→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   191→  return { k, d };
   192→}

   194→// ─── Helper: Williams %R(14) ──────────────────────────────────────────────────
   195→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   196→  if (data.length < period) return -50;
   197→  const slice = data.slice(-period);
   198→  let highest = -Infinity;
   199→  let lowest = Infinity;
   200→  for (const c of slice) {
   201→    if (c.high > highest) highest = c.high;
   202→    if (c.low < lowest) lowest = c.low;
   203→  }
   204→  const range = highest - lowest;
   205→  if (range === 0) return -50;
   206→  return ((highest - data[data.length - 1].close) / range) * -100;
   207→}

   209→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   210→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   211→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   212→  const emaFast = emaCalc(closes, fast);
   213→  const emaSlow = emaCalc(closes, slow);
   214→  const macdLine = emaFast - emaSlow;
   215→  // Calculate signal as EMA of MACD values
   216→  const macdValues: number[] = [];
   217→  const kFast = 2 / (fast + 1);
   218→  const kSlow = 2 / (slow + 1);
   219→  let eFast = sma(closes.slice(0, fast), fast);
   220→  let eSlow = sma(closes.slice(0, slow), slow);
   221→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   222→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   223→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   224→    macdValues.push(eFast - eSlow);
   225→  }
   226→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   227→  const kSig = 2 / (sig + 1);
   228→  let signalVal = sma(macdValues.slice(0, sig), sig);
   229→  for (let i = sig; i < macdValues.length; i++) {
   230→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   231→  }
   232→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   233→}

   235→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   236→function calcATR(data: OHLCV[], period: number = 14): number {
   237→  if (data.length < 2) return 0;
   238→  const trs: number[] = [];
   239→  for (let i = 1; i < data.length; i++) {
   240→    const tr = Math.max(
   241→      data[i].high - data[i].low,
   242→      Math.abs(data[i].high - data[i - 1].close),
   243→      Math.abs(data[i].low - data[i - 1].close)
   244→    );
   245→    trs.push(tr);
   246→  }
   247→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   248→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   249→  for (let i = period; i < trs.length; i++) {
   250→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   251→  }
   252→  return atrVal;
   253→}

   255→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   256→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   257→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   258→  const plusDMs: number[] = [];
   259→  const minusDMs: number[] = [];
   260→  const trs: number[] = [];
   261→  for (let i = 1; i < data.length; i++) {
   262→    const upMove = data[i].high - data[i - 1].high;
   263→    const downMove = data[i - 1].low - data[i].low;
   264→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   265→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   266→    trs.push(Math.max(
   267→      data[i].high - data[i].low,
   268→      Math.abs(data[i].high - data[i - 1].close),
   269→      Math.abs(data[i].low - data[i - 1].close)
   270→    ));
   271→  }
   272→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   273→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   274→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   275→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   276→  const dxValues: number[] = [];
   277→  for (let i = period; i < trs.length; i++) {
   278→    smoothTR = smoothTR - smoothTR / period + trs[i];
   279→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   280→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   281→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   282→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   283→    const diSum = diPlus + diMinus;
   284→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   285→    dxValues.push(dx);
   286→  }
   287→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   288→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   289→  for (let i = period; i < dxValues.length; i++) {
   290→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   291→  }
   292→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   293→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   294→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   295→}

   297→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   298→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   299→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   300→  let isLong = true;
   301→  let af = step;
   302→  let ep = data[0].high;
   303→  let sar = data[0].low;
   304→  for (let i = 1; i < data.length; i++) {
   305→    const prevSar = sar;
   306→    sar = prevSar + af * (ep - prevSar);
   307→    // Ensure SAR is within bounds
   308→    if (isLong) {
   309→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   310→      if (data[i].low < sar) {
   311→        isLong = false;
   312→        sar = ep;
   313→        ep = data[i].low;
   314→        af = step;
   315→      } else {
   316→        if (data[i].high > ep) {
   317→          ep = data[i].high;
   318→          af = Math.min(af + step, max);
   319→        }
   320→      }
   321→    } else {
   322→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   323→      if (data[i].high > sar) {
   324→        isLong = true;
   325→        sar = ep;
   326→        ep = data[i].high;
   327→        af = step;
   328→      } else {
   329→        if (data[i].low < ep) {
   330→          ep = data[i].low;
   331→          af = Math.min(af + step, max);
   332→        }
   333→      }
   334→    }
   335→  }
   336→  return sar;
   337→}

   339→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   340→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   341→  if (closes.length < period) {
   342→    const c = closes[closes.length - 1] ?? 0;
   343→    return { upper: c, middle: c, lower: c };
   344→  }
   345→  const slice = closes.slice(-period);
   346→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   347→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   348→  const sd = Math.sqrt(variance);
   349→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   350→}

   352→// ─── Helper: OBV ──────────────────────────────────────────────────────────────
   353→function calcOBV(data: OHLCV[]): number {
   354→  let obv = 0;
   355→  for (let i = 1; i < data.length; i++) {
   356→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   357→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   358→  }
   359→  return obv;
   360→}

   362→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   363→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   364→  const n = values.length;
   365→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   366→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   367→  for (let i = 0; i < n; i++) {
   368→    sumX += i;
   369→    sumY += values[i];
   370→    sumXY += i * values[i];
   371→    sumX2 += i * i;
   372→    sumY2 += values[i] * values[i];
   373→  }
   374→  const denom = n * sumX2 - sumX * sumX;
   375→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   376→  const slope = (n * sumXY - sumX * sumY) / denom;
   377→  const intercept = (sumY - slope * sumX) / n;
   378→  const meanY = sumY / n;
   379→  let ssRes = 0, ssTot = 0;
   380→  for (let i = 0; i < n; i++) {
   381→    const predicted = slope * i + intercept;
   382→    ssRes += (values[i] - predicted) ** 2;
   383→    ssTot += (values[i] - meanY) ** 2;
   384→  }
   385→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   386→  return { slope, intercept, r2 };
   387→}

   389→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   390→function roundToNice(price: number): number {
   391→  const abs = Math.abs(price);
   392→  let step: number;
   393→  if (abs >= 100000) step = 100;
   394→  else if (abs >= 10000) step = 50;
   395→  else if (abs >= 1000) step = 10;
   396→  else if (abs >= 100) step = 5;
   397→  else step = 1;
   398→  return Math.round(price / step) * step;
   399→}

   401→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
   402→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: number[]; lows: number[] } {
   403→  const highs: number[] = [];
   404→  const lows: number[] = [];
   405→  for (let i = lookback; i < data.length - lookback; i++) {
   406→    let isHigh = true;
   407→    let isLow = true;
   408→    for (let j = i - lookback; j <= i + lookback; j++) {
   409→      if (j === i) continue;
   410→      if (data[j].high >= data[i].high) isHigh = false;
   411→      if (data[j].low <= data[i].low) isLow = false;
   412→    }
   413→    if (isHigh) highs.push(data[i].high);
   414→    if (isLow) lows.push(data[i].low);
   415→  }
   416→  return { highs, lows };
   417→}

   419→// ─── Support / Resistance (6 each, adaptive gap, multi-source, 10% window strength) ─
   420→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   421→  resistances: number[];
   422→  supports: number[];
   423→  supportStrengths: LevelStrength[];
   424→  resistanceStrengths: LevelStrength[];
   425→  priceTargets: LevelStrength[];
   426→} {
   427→  const allRawLevels: number[] = [];
   428→  const closes = data.map(d => d.close);
   429→
   430→  // 1. Pivot Points from previous day (5R + 5S)
   431→  const prev = data[data.length - 2] ?? data[data.length - 1];
   432→  if (prev) {
   433→    const pp = (prev.high + prev.low + prev.close) / 3;
   434→    const hl = prev.high - prev.low;
   435→    const pivotR = [
   436→      2 * pp - prev.low,
   437→      pp + hl,
   438→      prev.high + 2 * (pp - prev.low),
   439→      prev.high + 2 * (pp - prev.low) + hl,
   440→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   441→    ];
   442→    const pivotS = [
   443→      2 * pp - prev.high,
   444→      pp - hl,
   445→      prev.low - 2 * (prev.high - pp),
   446→      prev.low - 2 * (prev.high - pp) - hl,
   447→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   448→    ];
   449→    allRawLevels.push(...pivotR, ...pivotS);
   450→  }
   451→
   452→  // 2. Swing levels (multiple lookback periods for better coverage)
   453→  for (const lb of [3, 5, 7, 10]) {
   454→    const swings = findSwingLevels(data, lb);
   455→    allRawLevels.push(...swings.highs, ...swings.lows);
   456→  }
   457→
   458→  // 3. Moving Average levels (potential dynamic S/R)
   459→  for (const period of [5, 10, 21, 50, 100, 200]) {
   460→    const v = sma(closes, period);
   461→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01) allRawLevels.push(v);
   462→  }
   463→  const ema12 = emaCalc(closes, 12);
   464→  const ema26 = emaCalc(closes, 26);
   465→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01) allRawLevels.push(ema12);
   466→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01) allRawLevels.push(ema26);
   467→
   468→  // 4. Bollinger Band levels
   469→  const bbLevels = calcBollingerBands(closes);
   470→  if (bbLevels.upper > 0) allRawLevels.push(bbLevels.upper);
   471→  if (bbLevels.lower > 0) allRawLevels.push(bbLevels.lower);
   472→
   473→  // 5. Round number / psychological levels
   474→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
   475→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
   476→    allRawLevels.push(Math.round(currentPrice * mult / magnitude) * magnitude);
   477→  }
   478→
   479→  // 6. Recent high/low zones (5-day, 22-day, 66-day)
   480→  for (const window of [5, 22, 66]) {
   481→    const slice = data.slice(-Math.min(window, data.length));
   482→    if (slice.length > 0) {
   483→      allRawLevels.push(Math.max(...slice.map(d => d.high)));
   484→      allRawLevels.push(Math.min(...slice.map(d => d.low)));
   485→    }
   486→  }
   487→
   488→  // 7. Round all and filter invalid
   489→  const rounded = allRawLevels.map(roundToNice).filter(l => l > 0);
   490→
   491→  // 8. Remove duplicates (within 0.5%)
   492→  const unique: number[] = [];
   493→  for (const level of rounded) {
   494→    const isDup = unique.some(u => Math.abs(u - level) / Math.max(u, 1) < 0.005);
   495→    if (!isDup) unique.push(level);
   496→  }
   497→
   498→  // 9. Calculate strength for each unique level — LOGICAL MULTI-FACTOR FORMULA
   499→  //
   500→  // Formula: strength = min(10, max(1, round(
   501→  //   touchFactor × convergenceFactor × recencyFactor × proximityFactor
   502→  // )))
   503→  //
   504→  // touchFactor: How many times price touched (bounced off) this level.
   505→  //   Count candles where high/low is within 0.5% of the level.
   506→  //   More touches = stronger level. Scale: 1 + touches * 0.3 (max 3.0)
   507→  //
   508→  // convergenceFactor: How many different sources identified this same level.
   509→  //   Count of raw levels within 1% of this level. More convergence = stronger.
   510→  //   Scale: 0.5 + convergences * 0.15 (max 2.5)
   511→  //
   512→  // recencyFactor: Recent touches are more relevant than old ones.
   513→  //   Weighted average of touch recency. Scale: 0.5 to 1.5
   514→  //
   515→  // proximityFactor: Distance from current price affects relevance.
   516→  //   Levels too far away are less relevant. Scale: 0.5 to 1.2
   517→  //
   518→  // volumeTouchFactor (when has volume): Average volume at touch points
   519→  //   relative to average volume. Higher volume touches = stronger.
   520→  //   Scale: 0.7 to 1.3
   521→
   522→  const hasVol = data.some(d => d.volume > 0);
   523→  const avgVolume = hasVol ? data.slice(-50).reduce((s, d) => s + d.volume, 0) / Math.min(50, data.length) : 0;
   524→
   525→  const levelStrengths: LevelStrength[] = unique.map(level => {
   526→    // ── touchFactor: count price touches within 0.5% ──
   527→    const touchThreshold = level * 0.005;
   528→    let touchCount = 0;
   529→    let touchRecencySum = 0; // weighted by recency
   530→    let touchVolumeSum = 0;
   531→
   532→    for (let i = 0; i < data.length; i++) {
   533→      const d = data[i];
   534→      const nearHigh = Math.abs(d.high - level) <= touchThreshold;
   535→      const nearLow = Math.abs(d.low - level) <= touchThreshold;
   536→      if (nearHigh || nearLow) {
   537→        touchCount++;
   538→        // Recency: more recent touches get higher weight (0 to 1, linear)
   539→        const recencyWeight = (i + 1) / data.length;
   540→        touchRecencySum += recencyWeight;
   541→        // Volume at touch
   542→        if (hasVol && d.volume > 0) {
   543→          touchVolumeSum += d.volume;
   544→        }
   545→      }
   546→    }
   547→
   548→    const touchFactor = Math.min(3.0, 1.0 + touchCount * 0.3);
   549→
   550→    // ── convergenceFactor: how many raw levels cluster near this one ──
   551→    const convergenceCount = unique.filter(other =>
   552→      other !== level && Math.abs(other - level) / Math.max(level, 1) < 0.01
   553→    ).length;
   554→    const convergenceFactor = Math.min(2.5, 0.5 + convergenceCount * 0.15);
   555→
   556→    // ── recencyFactor: weighted average of how recent the touches are ──
   557→    const recencyFactor = touchCount > 0
   558→      ? Math.min(1.5, 0.5 + (touchRecencySum / touchCount) * 1.0)
   559→      : 0.7;
   560→
   561→    // ── proximityFactor: closer to current price = more relevant ──
   562→    const distFromPrice = Math.abs(level - currentPrice) / currentPrice;
   563→    const proximityFactor = Math.min(1.2, Math.max(0.5, 1.2 - distFromPrice * 3));
   564→
   565→    // ── volumeTouchFactor: volume at touches vs average ──
   566→    let volumeTouchFactor = 1.0;
   567→    if (hasVol && touchCount > 0 && avgVolume > 0) {
   568→      const avgTouchVolume = touchVolumeSum / touchCount;
   569→      const volRatio = avgTouchVolume / avgVolume;
   570→      volumeTouchFactor = Math.min(1.3, Math.max(0.7, 0.7 + volRatio * 0.3));
   571→    }
   572→
   573→    // ── Final strength calculation ──
   574→    const rawStrength = touchFactor * convergenceFactor * recencyFactor * proximityFactor * volumeTouchFactor;
   575→    const strength = Math.min(10, Math.max(1, Math.round(rawStrength)));
   576→
   577→    return { price: level, strength, isTarget: false };
   578→  });
   579→
   580→  // 10. Separate above/below current price
   581→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   582→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   583→
   584→  // 11. Enforce 2%-5% gap, always return exactly targetCount levels
   585→  //    Step 1: Merge levels closer than 2%
   586→  //    Step 2: Walk from nearest level, keep only chain within 2-5% gap
   587→  //    Step 3: If fewer than targetCount, extend chain by computing ~3% steps outward
   588→  //    Step 4: If more than targetCount, trim by keeping highest strength
   589→  function enforceGap(
   590→    levels: LevelStrength[],
   591→    targetCount: number,
   592→    _allUnique: LevelStrength[],
   593→  ): LevelStrength[] {
   594→    if (levels.length === 0) return [];
   595→
   596→    const MIN_GAP = 0.02;
   597→    const MAX_GAP = 0.05;
   598→    const EXTEND_GAP = 0.035; // 3.5% — middle of 2-5% range
   599→    const step = currentPrice >= 100000 ? 100 : currentPrice >= 10000 ? 50 : currentPrice >= 1000 ? 10 : currentPrice >= 100 ? 5 : 1;
   600→    const isDescending = levels.length > 0 && levels[0].price > (levels[levels.length - 1]?.price ?? 0);
   601→
   602→    // Step 1: Merge levels closer than MIN_GAP
   603→    const merged: LevelStrength[] = [levels[0]];
   604→    for (let i = 1; i < levels.length; i++) {
   605→      const prevPrice = merged[merged.length - 1].price;
   606→      const currPrice = levels[i].price;
   607→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   608→      if (gap < MIN_GAP) {
   609→        const mergedStrength = Math.min(10, Math.max(merged[merged.length - 1].strength, levels[i].strength) + 1);
   610→        merged[merged.length - 1] = {
   611→          ...(levels[i].strength > merged[merged.length - 1].strength ? levels[i] : merged[merged.length - 1]),
   612→          strength: mergedStrength,
   613→        };
   614→      } else {
   615→        merged.push(levels[i]);
   616→      }
   617→    }
   618→
   619→    // Step 2: Walk from first level (nearest to price), keep only if gap is <= MAX_GAP
   620→    const chain: LevelStrength[] = [merged[0]];
   621→    for (let i = 1; i < merged.length; i++) {
   622→      const prevPrice = chain[chain.length - 1].price;
   623→      const currPrice = merged[i].price;
   624→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   625→      if (gap <= MAX_GAP) {
   626→        chain.push(merged[i]);
   627→      }
   628→    }
   629→
   630→    // Step 3: If fewer than targetCount, extend the chain outward at ~3.5% intervals
   631→    while (chain.length < targetCount) {
   632→      const lastPrice = chain[chain.length - 1].price;
   633→      const nextRaw = isDescending
   634→        ? lastPrice * (1 - EXTEND_GAP)  // supports go down
   635→        : lastPrice * (1 + EXTEND_GAP); // resistances go up
   636→      const nextPrice = Math.round(nextRaw / step) * step;
   637→      if (nextPrice <= 0) break;
   638→      if (nextPrice === lastPrice) break; // rounding made no change (very low price)
   639→      chain.push({ price: nextPrice, strength: 2, isTarget: false });
   640→    }
   641→
   642→    // Step 4: If more than targetCount, trim by keeping highest strength
   643→    if (chain.length > targetCount) {
   644→      chain.sort((a, b) => b.strength - a.strength);
   645→      const kept = chain.slice(0, targetCount);
   646→      kept.sort((a, b) => isDescending ? b.price - a.price : a.price - b.price);
   647→      return kept;
   648→    }
   649→
   650→    return chain;
   651→  }
   652→
   653→  const finalResistances = enforceGap(above, 6, levelStrengths);
   654→  const finalSupports = enforceGap(below, 6, levelStrengths);
   655→
   656→  // 12. Determine price targets: strength > 7 is a target
   657→  const allLevels = [...finalResistances, ...finalSupports];
   658→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   659→  let targets: LevelStrength[];
   660→  if (sortedByStrength.some(l => l.strength > 7)) {
   661→    targets = sortedByStrength.filter(l => l.strength > 7);
   662→  } else {
   663→    targets = sortedByStrength.slice(0, 2);
   664→  }
   665→
   666→  // Mark targets
   667→  targets = targets.map(t => ({ ...t, isTarget: true }));
   668→
   669→  // Also mark in final arrays
   670→  finalResistances.forEach(r => {
   671→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   672→  });
   673→  finalSupports.forEach(s => {
   674→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   675→  });
   676→
   677→  return {
   678→    resistances: finalResistances.map(l => l.price),
   679→    supports: finalSupports.map(l => l.price),
   680→    supportStrengths: finalSupports,
   681→    resistanceStrengths: finalResistances,
   682→    priceTargets: targets,
   683→  };
   684→}

   686→// ─── Trend Analysis ───────────────────────────────────────────────────────────
   687→function calcTrend(closes: number[], period: number): TrendResult {
   688→  const slice = closes.slice(-Math.min(period, closes.length));
   689→  const { slope, r2 } = linearRegression(slice);
   690→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   691→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   692→  const priceThreshold = avgPrice * 0.001; // 0.1%
   693→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   694→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   695→}
