   146→   146→// — S/R strength-based line styling (solid only, thickness = strength) ————
   147→   147→function srLineStyle(strength: number, isTarget: boolean): { lineWidth: number; lineStyle: 0; color: string } {
   148→   148→  if (isTarget) {
   149→   149→    return { lineWidth: 4, lineStyle: 0, color: '' };
   150→   150→  }
   151→   151→  if (strength >= 9) return { lineWidth: 5, lineStyle: 0, color: '' };
   152→   152→  if (strength >= 7) return { lineWidth: 4, lineStyle: 0, color: '' };
   153→   153→  if (strength >= 5) return { lineWidth: 3, lineStyle: 0, color: '' };
   154→   154→  if (strength >= 3) return { lineWidth: 2, lineStyle: 0, color: '' };
   155→   155→  return { lineWidth: 1, lineStyle: 0, color: '' };
   156→   156→}

   198→   198→export default function CandlestickChart({ data, ta, height = 520 }: CandlestickChartProps) {
   199→   199→  const chartContainerRef = useRef<HTMLDivElement>(null);
   200→   200→  const svgRef = useRef<SVGSVGElement>(null);
   201→   201→  const chartRef = useRef<IChartApi | null>(null);
   202→   202→  const csRef = useRef<any>(null);
   203→   203→  const subscriberRef = useRef<{ dispose: () => void } | null>(null);
   204→   204→  const crosshairSubRef = useRef<{ dispose: () => void } | null>(null);
   205→   205→
   206→   206→  // Crosshair legend state
   207→   207→  const [hoverInfo, setHoverInfo] = useState<{ date: string; o: number; h: number; l: number; c: number; v: number; chg: number } | null>(null);
   208→   208→
   209→   209→  // Build Jalali date strings for all candles (memoized)
   210→   210→  const jalaliDates = useMemo(() => {
   211→   211→    return data.map((d) =>
   212→   212→      isGregorianDate(d.date) ? candleDateToJalali(d.date, 'full') : formatJalaliString(d.date, 'full')
   213→   213→    );
   214→   214→  }, [data]);
   215→   215→
   216→   216→  const [drawings, setDrawings] = useState<Drawing[]>([]);
   217→   217→  const [activeTool, setActiveTool] = useState<ToolType>('cursor');
   218→   218→  const [activeColor, setActiveColor] = useState(PALETTE[0]);
   219→   219→  const [isDrawing, setIsDrawing] = useState(false);
   220→   220→  const [currentDrawing, setCurrentDrawing] = useState<Drawing | null>(null);
   221→   221→  const [textInput, setTextInput] = useState<{ x: number; y: number; drawingId: string } | null>(null);
   222→   222→  const [textValue, setTextValue] = useState('');
   223→   223→  const [renderKey, setRenderKey] = useState(0);
   224→   224→  const [chartSize, setChartSize] = useState({ w: 0, h: 0 });
   225→   225→
   226→   226→  /* - Helper: coordinate conversion -------------- */
   227→   227→  const getChartPoint = useCallback((clientX: number, clientY: number): DrawingPoint | null => {
   228→   228→    const chart = chartRef.current;
   229→   229→    const container = chartContainerRef.current;
   230→   230→    if (!chart || !container) return null;
   231→   231→    const rect = container.getBoundingClientRect();
   232→   232→    const x = clientX - rect.left;
   233→   233→    const y = clientY - rect.top;
   234→   234→    const price = csRef.current?.coordinateToPrice(y) ?? null;
   235→   235→    const time = chart.timeScale().coordinateToTime(x);
   236→   236→    if (price === null || time === null) return null;
   237→   237→    return { time: time as number, price };
   238→   238→  }, []);
   239→   239→
   240→   240→  /* - SVG Rendering --------------------- */
   241→   241→  const renderDrawings = useCallback(() => {
   242→   242→    const chart = chartRef.current;
   243→   243→    const svg = svgRef.current;
   244→   244→    const container = chartContainerRef.current;
   245→   245→    if (!chart || !svg || !container || drawings.length === 0) {
   246→   246→      if (svg) svg.innerHTML = '';
   247→   247→      return;
   248→   248→    }
   249→   249→
   250→   250→    const rect = container.getBoundingClientRect();
   251→   251→    const w = rect.width;
   252→   252→    const h = rect.height;
   253→   253→    if (w === 0 || h === 0) return;
   254→   254→
   255→   255→    let html = '';
   256→   256→
   257→   257→    // Defs for arrow markers
   258→   258→    html += `<defs>`;
   259→   259→    PALETTE.forEach((c) => {
   260→   260→      html += `<marker id="arrow-${c.replace('#','')}" markerWidth="8" markerHeight="6" refX="7" refY="3" orient="auto"><polygon points="0 0, 8 3, 0 6" fill="${c}" /></marker>`;
   261→   261→    });
   262→   262→    html += `</defs>`;
   263→   263→
   264→   264→    const allDrawings = [...drawings];
   265→   265→    if (currentDrawing) allDrawings.push(currentDrawing);
   266→   266→
   267→   267→    for (const d of allDrawings) {
   268→   268→      if (!d.completed && d.points.length === 0) continue;
   269→   269→      const col = d.color;
   270→   270→      const lw = d.lineWidth;
   271→   271→
   272→   272→      switch (d.type) {
   273→   273→        case 'trendline': {
   274→   274→          if (d.points.length < 1) break;
   275→   275→          const p0 = d.points[0];
   276→   276→          const p1 = d.points.length > 1 ? d.points[d.points.length - 1] : p0;
   277→   277→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   278→   278→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   279→   279→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   280→   280→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   281→   281→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   282→   282→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" />`;
   283→   283→          break;
   284→   284→        }
   285→   285→        case 'hline': {
   286→   286→          if (d.points.length < 1) break;
   287→   287→          const p = d.points[0];
   288→   288→          const y = csRef.current?.priceToCoordinate(p.price);
   289→   289→          if (y == null) break;
   290→   290→          html += `<line x1="0" y1="${y}" x2="${w}" y2="${y}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   291→   291→          html += `<text x="4" y="${y - 4}" fill="${col}" font-size="11" font-family="Vazirmatn, sans-serif">${toPersianDigits(p.price.toFixed(0))}</text>`;
   292→   292→          break;
   293→   293→        }
   294→   294→        case 'vline': {
   295→   295→          if (d.points.length < 1) break;
   296→   296→          const p = d.points[0];
   297→   297→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   298→   298→          if (x == null) break;
   299→   299→          html += `<line x1="${x}" y1="0" x2="${x}" y2="${h}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   300→   300→          break;
   301→   301→        }
   302→   302→        case 'fibonacci': {
   303→   303→          if (d.points.length < 2) break;
   304→   304→          const p0 = d.points[0];
   305→   305→          const p1 = d.points[1];
   306→   306→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   307→   307→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   308→   308→          if (x0 == null || x1 == null) break;
   309→   309→          const priceHigh = Math.max(p0.price, p1.price);
   310→   310→          const priceLow = Math.min(p0.price, p1.price);
   311→   311→          const priceRange = priceHigh - priceLow;
   312→   312→          if (priceRange === 0) break;
   313→   313→
   314→   314→          const yTop = csRef.current?.priceToCoordinate(priceHigh);
   315→   315→          const yBottom = csRef.current?.priceToCoordinate(priceLow);
   316→   316→          if (yTop == null || yBottom == null) break;
   317→   317→          html += `<rect x="${Math.min(x0,x1)}" y="${yTop}" width="${Math.abs(x1-x0)}" height="${yBottom - yTop}" fill="${col}" opacity="0.04" />`;
   318→   318→
   319→   319→          for (const level of FIB_LEVELS) {
   320→   320→            const price = priceHigh - level.pct * priceRange;
   321→   321→            const y = csRef.current?.priceToCoordinate(price);
   322→   322→            if (y == null) continue;
   323→   323→            html += `<line x1="${Math.min(x0,x1) - 10}" y1="${y}" x2="${Math.max(x0,x1) + 10}" y2="${y}" stroke="${col}" stroke-width="1" opacity="0.6" />`;
   324→   324→            html += `<rect x="${Math.max(x0,x1) + 12}" y="${y - 8}" width="56" height="16" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="2" />`;
   325→   325→            html += `<text x="${Math.max(x0,x1) + 14}" y="${y + 4}" fill="${col}" font-size="10" font-family="Vazirmatn, sans-serif">${toPersianDigits(level.label)} ${toPersianDigits(price.toFixed(0))}</text>`;
   326→   326→          }
   327→   327→          break;
   328→   328→        }
   329→   329→        case 'rectangle': {
   330→   330→          if (d.points.length < 2) break;
   331→   331→          const p0 = d.points[0];
   332→   332→          const p1 = d.points[d.points.length - 1];
   333→   333→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   334→   334→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   335→   335→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   336→   336→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   337→   337→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   338→   338→          const rx = Math.min(x0, x1);
   339→   339→          const ry = Math.min(y0, y1);
   340→   340→          const rw = Math.abs(x1 - x0);
   341→   341→          const rh = Math.abs(y1 - y0);
   342→   342→          html += `<rect x="${rx}" y="${ry}" width="${rw}" height="${rh}" fill="${col}" fill-opacity="0.08" stroke="${col}" stroke-width="${lw}" rx="2" />`;
   343→   343→          break;
   344→   344→        }
   345→   345→        case 'text': {
   346→   346→          if (d.points.length < 1 || !d.text) break;
   347→   347→          const p = d.points[0];
   348→   348→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   349→   349→          const y = csRef.current?.priceToCoordinate(p.price);
   350→   350→          if (x == null || y == null) break;
   351→   351→          html += `<rect x="${x + 4}" y="${y - 16}" width="${d.text.length * 8 + 12}" height="22" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="3" />`;
   352→   352→          html += `<text x="${x + 10}" y="${y + 1}" fill="${col}" font-size="12" font-family="Vazirmatn, sans-serif">${d.text}</text>`;
   353→   353→          break;
   354→   354→        }
   355→   355→        case 'brush': {
   356→   356→          if (d.points.length < 2) break;
   357→   357→          let pts = '';
   358→   358→          let valid = true;
   359→   359→          for (const p of d.points) {
   360→   360→            const x = chart.timeScale().timeToCoordinate(p.time as Time);
   361→   361→            const y = csRef.current?.priceToCoordinate(p.price);
   362→   362→            if (x == null || y == null) { valid = false; break; }
   363→   363→            pts += `${x},${y} `;
   364→   364→          }
   365→   365→          if (!valid || pts.length === 0) break;
   366→   366→          html += `<polyline points="${pts.trim()}" fill="none" stroke="${col}" stroke-width="${lw}" stroke-linecap="round" stroke-linejoin="round" />`;
   367→   367→          break;
   368→   368→        }
   369→   369→        case 'measure': {
   370→   370→          if (d.points.length < 2) break;
   371→   371→          const p0 = d.points[0];
   372→   372→          const p1 = d.points[d.points.length - 1];
