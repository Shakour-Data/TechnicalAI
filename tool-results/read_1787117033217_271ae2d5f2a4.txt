   136→/* ────────────────────────────────────────────────────────── COMPONENT ───── */
   137→
   138→export default function CandlestickChart({ data, ta, height = 520 }: CandlestickChartProps) {
   139→  const chartContainerRef = useRef<HTMLDivElement>(null);
   140→  const chartParentRef = useRef<HTMLDivElement>(null);
   141→  const svgRef = useRef<SVGSVGElement>(null);
   142→  const chartRef = useRef<IChartApi | null>(null);
   143→  const subscriberRef = useRef<{ dispose: () => void } | null>(null);
   144→
   145→  const [drawings, setDrawings] = useState<Drawing[]>([]);
   146→  const [activeTool, setActiveTool] = useState<ToolType>('cursor');
   147→  const [activeColor, setActiveColor] = useState(PALETTE[0]);
   148→  const [isDrawing, setIsDrawing] = useState(false);
   149→  const [currentDrawing, setCurrentDrawing] = useState<Drawing | null>(null);
   150→  const [textInput, setTextInput] = useState<{ x: number; y: number; drawingId: string } | null>(null);
   151→  const [textValue, setTextValue] = useState('');
   152→  const [renderKey, setRenderKey] = useState(0);
   153→  const [chartSize, setChartSize] = useState({ w: 0, h: 0 });
   154→
   155→  /* ── Helper: coordinate conversion ─────────────────────────── */
   156→  const getChartPoint = useCallback((clientX: number, clientY: number): DrawingPoint | null => {
   157→    const chart = chartRef.current;
   158→    const container = chartContainerRef.current;
   159→    if (!chart || !container) return null;
   160→    const rect = container.getBoundingClientRect();
   161→    const x = clientX - rect.left;
   162→    const y = clientY - rect.top;
   163→    const price = chart.priceScale('right').coordinateToPrice(y);
   164→    const time = chart.timeScale().coordinateToTime(x);
   165→    if (price === null || time === null) return null;
   166→    return { time, price };
   167→  }, []);
   168→
   169→  /* ── SVG Rendering ────────────────────────────────────────── */
   170→  const renderDrawings = useCallback(() => {
   171→    const chart = chartRef.current;
   172→    const svg = svgRef.current;
   173→    const container = chartContainerRef.current;
   174→    if (!chart || !svg || !container || drawings.length === 0) {
   175→      if (svg) svg.innerHTML = '';
   176→      return;
   177→    }
   178→
   179→    const rect = container.getBoundingClientRect();
   180→    const w = rect.width;
   181→    const h = rect.height;
   182→    if (w === 0 || h === 0) return;
   183→
   184→    let html = '';
   185→
   186→    // Defs for arrow markers
   187→    html += `<defs>`;
   188→    PALETTE.forEach((c, i) => {
   189→      html += `<marker id="arrow-${c.replace('#','')}" markerWidth="8" markerHeight="6" refX="7" refY="3" orient="auto"><polygon points="0 0, 8 3, 0 6" fill="${c}" /></marker>`;
   190→    });
   191→    html += `</defs>`;
   192→
   193→    const allDrawings = [...drawings];
   194→    if (currentDrawing) allDrawings.push(currentDrawing);
   195→
   196→    for (const d of allDrawings) {
   197→      if (!d.completed && d.points.length === 0) continue;
   198→      const col = d.color;
   199→      const lw = d.lineWidth;
   200→
   201→      switch (d.type) {
   202→        case 'trendline': {
   203→          if (d.points.length < 1) break;
   204→          const p0 = d.points[0];
   205→          const p1 = d.points.length > 1 ? d.points[d.points.length - 1] : p0;
   206→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   207→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   208→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   209→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   210→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   211→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" />`;
   212→          break;
   213→        }
   214→        case 'hline': {
   215→          if (d.points.length < 1) break;
   216→          const p = d.points[0];
   217→          const y = chart.priceScale('right').priceToCoordinate(p.price);
   218→          if (y == null) break;
   219→          html += `<line x1="0" y1="${y}" x2="${w}" y2="${y}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   220→          html += `<text x="4" y="${y - 4}" fill="${col}" font-size="11" font-family="sans-serif">${p.price.toFixed(0)}</text>`;
   221→          break;
   222→        }
   223→        case 'vline': {
   224→          if (d.points.length < 1) break;
   225→          const p = d.points[0];
   226→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   227→          if (x == null) break;
   228→          html += `<line x1="${x}" y1="0" x2="${x}" y2="${h}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   229→          break;
   230→        }
   231→        case 'fibonacci': {
   232→          if (d.points.length < 2) break;
   233→          const p0 = d.points[0];
   234→          const p1 = d.points[1];
   235→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   236→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   237→          if (x0 == null || x1 == null) break;
   238→          const priceHigh = Math.max(p0.price, p1.price);
   239→          const priceLow = Math.min(p0.price, p1.price);
   240→          const priceRange = priceHigh - priceLow;
   241→          if (priceRange === 0) break;
   242→
   243→          // Filled background
   244→          const yTop = chart.priceScale('right').priceToCoordinate(priceHigh);
   245→          const yBottom = chart.priceScale('right').priceToCoordinate(priceLow);
   246→          if (yTop == null || yBottom == null) break;
   247→          html += `<rect x="${Math.min(x0,x1)}" y="${yTop}" width="${Math.abs(x1-x0)}" height="${yBottom - yTop}" fill="${col}" opacity="0.04" />`;
   248→
   249→          for (const level of FIB_LEVELS) {
   250→            const price = priceHigh - level.pct * priceRange;
   251→            const y = chart.priceScale('right').priceToCoordinate(price);
   252→            if (y == null) continue;
   253→            html += `<line x1="${Math.min(x0,x1) - 10}" y1="${y}" x2="${Math.max(x0,x1) + 10}" y2="${y}" stroke="${col}" stroke-width="1" opacity="0.6" />`;
   254→            html += `<rect x="${Math.max(x0,x1) + 12}" y="${y - 8}" width="56" height="16" fill="#0b0f1a" rx="2" />`;
   255→            html += `<text x="${Math.max(x0,x1) + 14}" y="${y + 4}" fill="${col}" font-size="10" font-family="sans-serif">${level.label} ${price.toFixed(0)}</text>`;
   256→          }
   257→          break;
   258→        }
   259→        case 'rectangle': {
   260→          if (d.points.length < 2) break;
   261→          const p0 = d.points[0];
   262→          const p1 = d.points[d.points.length - 1];
   263→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   264→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   265→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   266→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   267→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   268→          const rx = Math.min(x0, x1);
   269→          const ry = Math.min(y0, y1);
   270→          const rw = Math.abs(x1 - x0);
   271→          const rh = Math.abs(y1 - y0);
   272→          html += `<rect x="${rx}" y="${ry}" width="${rw}" height="${rh}" fill="${col}" fill-opacity="0.08" stroke="${col}" stroke-width="${lw}" rx="2" />`;
   273→          break;
   274→        }
   275→        case 'text': {
   276→          if (d.points.length < 1 || !d.text) break;
   277→          const p = d.points[0];
   278→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   279→          const y = chart.priceScale('right').priceToCoordinate(p.price);
   280→          if (x == null || y == null) break;
   281→          html += `<rect x="${x + 4}" y="${y - 16}" width="${d.text.length * 8 + 12}" height="22" fill="#0b0f1a" stroke="${col}" stroke-width="1" rx="3" />`;
   282→          html += `<text x="${x + 10}" y="${y + 1}" fill="${col}" font-size="12" font-family="sans-serif">${d.text}</text>`;
   283→          break;
   284→        }
   285→        case 'brush': {
   286→          if (d.points.length < 2) break;
   287→          let pts = '';
   288→          let valid = true;
   289→          for (const p of d.points) {
   290→            const x = chart.timeScale().timeToCoordinate(p.time as Time);
   291→            const y = chart.priceScale('right').priceToCoordinate(p.price);
   292→            if (x == null || y == null) { valid = false; break; }
   293→            pts += `${x},${y} `;
   294→          }
   295→          if (!valid || pts.length === 0) break;
   296→          html += `<polyline points="${pts.trim()}" fill="none" stroke="${col}" stroke-width="${lw}" stroke-linecap="round" stroke-linejoin="round" />`;
   297→          break;
   298→        }
   299→        case 'measure': {
   300→          if (d.points.length < 2) break;
   301→          const p0 = d.points[0];
   302→          const p1 = d.points[d.points.length - 1];
   303→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   304→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   305→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   306→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   307→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   308→
   309→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="4,2" />`;
   310→          // Price difference
   311→          const diff = p1.price - p0.price;
   312→          const pctChg = p0.price !== 0 ? ((diff / p0.price) * 100) : 0;
   313→          const bars = Math.abs(typeof p1.time === 'number' && typeof p0.time === 'number' ? Math.round(p1.time - p0.time) : 0);
   314→          const midX = (x0 + x1) / 2;
   315→          const midY = (y0 + y1) / 2;
   316→          const label = `${diff >= 0 ? '+' : ''}${diff.toFixed(0)} (${pctChg >= 0 ? '+' : ''}${pctChg.toFixed(1)}%) ${bars}bar`;
   317→          html += `<rect x="${midX - 40}" y="${midY - 24}" width="80" height="20" fill="#0b0f1a" stroke="${col}" stroke-width="1" rx="3" />`;
   318→          html += `<text x="${midX - 36}" y="${midY - 10}" fill="${col}" font-size="10" font-family="sans-serif">${label}</text>`;
   319→          break;
   320→        }
   321→        case 'arrow': {
   322→          if (d.points.length < 2) break;
   323→          const p0 = d.points[0];
   324→          const p1 = d.points[d.points.length - 1];
   325→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   326→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   327→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   328→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   329→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   330→          const markerId = `arrow-${col.replace('#','')}`;
   331→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" marker-end="url(#${markerId})" />`;
   332→          break;
   333→        }
   334→      }
   335→    }
   336→
   337→    svg.innerHTML = html;
   338→  }, [drawings, currentDrawing]);
   339→
   340→  /* ── Chart Build ──────────────────────────────────────────── */
   341→  const build = useCallback(() => {
   342→    const chartEl = chartContainerRef.current;
   343→    if (!chartEl || data.length === 0) return;
   344→
   345→    if (chartRef.current) {
   346→      chartRef.current.remove();
   347→      chartRef.current = null;
   348→    }
   349→
   350→    const chart = createChart(chartEl, {
   351→      layout: { background: { type: ColorType.Solid, color: BG }, textColor: TXT, fontSize: 11 },
   352→      grid: { vertLines: { color: GRID }, horzLines: { color: GRID } },
   353→      crosshair: { mode: CrosshairMode.Normal },
   354→      rightPriceScale: {
   355→        borderColor: 'rgba(255,255,255,0.08)',
   356→        scaleMargins: { top: 0.05, bottom: 0.3 },
   357→        priceFormatter: (price: number) => toPersianDigits(Math.round(price).toLocaleString('en-US')),
   358→      },
   359→      localization: {
   360→        priceFormatter: (price: number) => toPersianDigits(Math.round(price).toLocaleString('en-US')),
   361→      },
   362→      timeScale: {
   363→        borderColor: 'rgba(255,255,255,0.08)',
   364→        rightOffset: 5,
   365→        barSpacing: 6,
   366→        timeVisible: false,
   367→        tickMarkFormatter: (time: Time, tickMarkType: TickMarkType, _locale: string) => {
   368→          return shamsiTickFormatter(time, tickMarkType as unknown as number);
   369→        },
   370→      },
   371→      width: chartEl.clientWidth,
   372→      height,
   373→    });
   374→    chartRef.current = chart;
   375→
   376→    // Use real dates when available, fall back to numeric index
   377→    const candles = data.map((d, i) => ({
   378→      time: (d.date && d.date.includes('-') ? d.date : undefined) as Time || (i as Time),
   379→      open: d.open,
   380→      high: d.high,
   381→      low: d.low,
   382→      close: d.close,
   383→    }));
   384→
   385→    const cs = chart.addSeries(CandlestickSeries, {
   386→      upColor: BULL,
   387→      downColor: BEAR,
   388→      borderUpColor: BULL,
   389→      borderDownColor: BEAR,
   390→      wickUpColor: BULL,
   391→      wickDownColor: BEAR,
   392→    });
   393→    cs.setData(candles);
   394→
   395→    // Volume
   396→    const vol = chart.addSeries(HistogramSeries, {
   397→      priceFormat: { type: 'volume' },
   398→      priceScaleId: 'vol',
   399→    });
   400→    vol.setData(
   401→      data.map((d, i) => ({
   402→        time: (d.date && d.date.includes('-') ? d.date : undefined) as Time || (i as Time),
   403→        value: d.volume,
   404→        color: d.close >= d.open ? BULL : BEAR,
   405→      }))
   406→    );
   407→    chart.priceScale('vol').applyOptions({ scaleMargins: { top: 0.8, bottom: 0 } });
   408→
   409→    // TA overlays
   410→    if (ta) {
   411→      for (const cfg of SMA_CFG) {
   412→        const v = ta.sma[cfg.key];
   413→        if (v && v > 0)
   414→          cs.createPriceLine({
   415→            price: v,
   416→            color: cfg.color,
   417→            lineWidth: 1,
   418→            lineStyle: 0,
   419→            axisLabelVisible: true,
   420→            title: cfg.title,
   421→          });
   422→      }
   423→      const bb = ta.bollingerBands;
   424→      if (bb) {
   425→        if (bb.upper > 0)
   426→          cs.createPriceLine({ price: bb.upper, color: 'rgba(161,98,255,0.5)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Upper' });
   427→        if (bb.middle > 0)
   428→          cs.createPriceLine({ price: bb.middle, color: 'rgba(161,98,255,0.5)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Mid' });
   429→        if (bb.lower > 0)
   430→          cs.createPriceLine({ price: bb.lower, color: 'rgba(161,98,255,0.5)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Lower' });
   431→      }
   432→      if (ta.sar > 0)
   433→        cs.createPriceLine({ price: ta.sar, color: 'rgba(255,255,255,0.35)', lineWidth: 1, lineStyle: 1, axisLabelVisible: true, title: 'SAR' });
   434→
   435→      // S/R with strength
   436→      const resistanceStrengths = ta.resistanceStrengths || [];
   437→      const supportStrengths = ta.supportStrengths || [];
   438→
   439→      resistanceStrengths.forEach((r, i) => {
   440→        if (r.price <= 0) return;
   441→        const lw = r.isTarget ? 3 : Math.max(1, Math.round(r.strength / 3));
   442→        const ls = r.isTarget ? 0 : 2;
   443→        const title = `R${i + 1} (${r.strength}/10)`;
   444→        cs.createPriceLine({
   445→          price: r.price,
   446→          color: r.isTarget ? '#ff6b6b' : BEAR,
   447→          lineWidth: lw,
   448→          lineStyle: ls as 0 | 1 | 2,
   449→          axisLabelVisible: i < 6,
   450→          title,
   451→        });
   452→      });
   453→
   454→      supportStrengths.forEach((s, i) => {
   455→        if (s.price <= 0) return;
   456→        const lw = s.isTarget ? 3 : Math.max(1, Math.round(s.strength / 3));
   457→        const ls = s.isTarget ? 0 : 2;
   458→        const title = `S${i + 1} (${s.strength}/10)`;
   459→        cs.createPriceLine({
   460→          price: s.price,
   461→          color: s.isTarget ? '#6bffb8' : BULL,
   462→          lineWidth: lw,
   463→          lineStyle: ls as 0 | 1 | 2,
   464→          axisLabelVisible: i < 6,
   465→          title,
   466→        });
   467→      });
   468→    }
   469→
   470→    chart.timeScale().fitContent();
   471→
   472→    // Subscribe to visible range changes to re-render SVG
   473→    requestAnimationFrame(() => {
   474→      if (chartRef.current) {
   475→        subscriberRef.current = chartRef.current.timeScale().subscribeVisibleLogicalRangeChange(() => {
   476→          setRenderKey((k) => k + 1);
   477→        });
   478→      }
   479→    });
   480→
   481→    // Set initial chart size (deferred to avoid React batching issue)
   482→    requestAnimationFrame(() => {
   483→      if (chartContainerRef.current) {
   484→        setChartSize({ w: chartContainerRef.current.clientWidth, h: height });
   485→      }
   486→    });
   487→  }, [data, ta, height]);
   488→
   489→  /* ── Re-render SVG when renderKey changes ────────────────── */
   490→  useEffect(() => {
   491→    renderDrawings();
   492→  }, [renderKey, renderDrawings]);
   493→
   494→  /* ── Init chart ───────────────────────────────────────────── */
   495→  useEffect(() => {
   496→    const el = chartContainerRef.current;
   497→    if (!el) return;
   498→    const ro = new ResizeObserver((entries) => {
   499→      for (const e of entries) {
   500→        const { width: w, height: h } = e.contentRect;
   501→        if (w > 0 && h > 0) {
   502→          chartRef.current?.applyOptions({ width: w, height: h });
   503→          setChartSize({ w, h });
   504→        }
   505→      }
   506→    });
   507→    ro.observe(el);
   508→    build();
   509→    return () => {
   510→      ro.disconnect();
   511→      subscriberRef.current?.dispose();
   512→      subscriberRef.current = null;
   513→      if (chartRef.current) {
   514→        chartRef.current.remove();
   515→        chartRef.current = null;
   516→      }
   517→    };
   518→  }, [build]);
   519→
   520→  /* ── Refs for native event handlers (avoid stale closures) ─── */
   521→  const activeToolRef = useRef<ToolType>('cursor');
   522→  const activeColorRef = useRef(PALETTE[0]);
   523→  const isDrawingRef = useRef(false);
   524→  const currentDrawingRef = useRef<Drawing | null>(null);
   525→  const textInputRef = useRef<{ x: number; y: number; drawingId: string } | null>(null);
   526→
   527→  useEffect(() => { activeToolRef.current = activeTool; }, [activeTool]);
   528→  useEffect(() => { activeColorRef.current = activeColor; }, [activeColor]);
   529→  useEffect(() => { isDrawingRef.current = isDrawing; }, [isDrawing]);
   530→  useEffect(() => { currentDrawingRef.current = currentDrawing; }, [currentDrawing]);
   531→  useEffect(() => { textInputRef.current = textInput; }, [textInput]);
   532→
   533→  /* ── Native event handling via parent div + window ────────── */
   534→  useEffect(() => {
   535→    const parent = chartParentRef.current;
   536→    if (!parent) return;
   537→
   538→    const onDown = (e: MouseEvent) => {
   539→      const tool = activeToolRef.current;
   540→      if (tool === 'cursor') return;
   541→      if (textInputRef.current) return;
   542→      const point = getChartPoint(e.clientX, e.clientY);
   543→      if (!point) return;
   544→
   545→      if (tool === 'hline') {
   546→        const drawing: Drawing = { id: Date.now().toString(), type: 'hline', points: [point], color: activeColorRef.current, lineWidth: 1, completed: true };
   547→        setDrawings((prev) => [...prev, drawing]);
   548→        return;
   549→      }
   550→      if (tool === 'text') {
   551→        const container = chartContainerRef.current;
   552→        if (!container) return;
   553→        const rect = container.getBoundingClientRect();
   554→        const tempDrawing: Drawing = { id: Date.now().toString(), type: 'text', points: [point], color: activeColorRef.current, lineWidth: 1, text: '', completed: false };
   555→        setCurrentDrawing(tempDrawing);
   556→        currentDrawingRef.current = tempDrawing;
   557→        isDrawingRef.current = true;
   558→        setIsDrawing(true);
   559→        setTextInput({ x: e.clientX - rect.left, y: e.clientY - rect.top, drawingId: tempDrawing.id });
   560→        setTextValue('');
   561→        return;
   562→      }
   563→      const drawing: Drawing = { id: Date.now().toString(), type: tool, points: [point], color: activeColorRef.current, lineWidth: tool === 'brush' ? 2 : 1, completed: false };
   564→      setCurrentDrawing(drawing);
   565→      currentDrawingRef.current = drawing;
   566→      isDrawingRef.current = true;
   567→      setIsDrawing(true);
   568→    };
   569→
   570→    const onMove = (e: MouseEvent) => {
   571→      if (!isDrawingRef.current || !currentDrawingRef.current) return;
   572→      const point = getChartPoint(e.clientX, e.clientY);
   573→      if (!point) return;
   574→      const cd = currentDrawingRef.current;
   575→      if (cd.type === 'brush') {
   576→        const updated = { ...cd, points: [...cd.points, point] };
   577→        currentDrawingRef.current = updated;
   578→        setCurrentDrawing(updated);
   579→      } else if (cd.type !== 'text' && cd.type !== 'hline') {
   580→        const updated = { ...cd, points: [cd.points[0], point] };
   581→        currentDrawingRef.current = updated;
   582→        setCurrentDrawing(updated);
   583→      }
   584→    };
   585→
   586→    const onUp = () => {
   587→      if (!isDrawingRef.current || !currentDrawingRef.current) return;
   588→      const cd = currentDrawingRef.current;
   589→      if (cd.type === 'text') return;
   590→      const completed: Drawing = { ...cd, completed: true };
   591→      if (completed.points.length >= 2) {
   592→        setDrawings((prev) => [...prev, completed]);
   593→      }
   594→      currentDrawingRef.current = null;
   595→      isDrawingRef.current = false;
   596→      setCurrentDrawing(null);
   597→      setIsDrawing(false);
   598→    };
   599→
   600→    parent.addEventListener('mousedown', onDown);
   601→    window.addEventListener('mousemove', onMove);
   602→    window.addEventListener('mouseup', onUp);
   603→
   604→    return () => {
   605→      parent.removeEventListener('mousedown', onDown);
   606→      window.removeEventListener('mousemove', onMove);
   607→      window.removeEventListener('mouseup', onUp);
   608→    };
   609→  }, [getChartPoint]);
   610→
   611→  /* ── Text input submit ────────────────────────────────────── */
   612→  const handleTextSubmit = useCallback(() => {
   613→    if (!textInput || !textValue.trim() || !currentDrawing) {
   614→      setTextInput(null);
   615→      setCurrentDrawing(null);
   616→      setIsDrawing(false);
   617→      return;
   618→    }
   619→
   620→    const completed: Drawing = {
   621→      ...currentDrawing,
   622→      text: textValue.trim(),
   623→      completed: true,
   624→    };
   625→    setDrawings((prev) => [...prev, completed]);
   626→    setTextInput(null);
   627→    setTextValue('');
   628→    setCurrentDrawing(null);
   629→    setIsDrawing(false);
   630→  }, [textInput, textValue, currentDrawing]);
   631→
   632→  const handleTextCancel = useCallback(() => {
   633→    setTextInput(null);
   634→    setTextValue('');
   635→    setCurrentDrawing(null);
   636→    setIsDrawing(false);
   637→  }, []);
   638→
   639→  /* ── Delete / Clear ──────────────────────────────────────── */
   640→  const handleDeleteLast = useCallback(() => {
   641→    setDrawings((prev) => prev.slice(0, -1));
   642→  }, []);
   643→
   644→  const handleClearAll = useCallback(() => {
   645→    setDrawings([]);
   646→  }, []);
   647→
   648→
   649→  /* ── Render ───────────────────────────────────────────────── */
   650→  return (
   651→    <div className="w-full">
   652→      {/* ── TOOLBAR ────────────────────────────────────────── */}
   653→      <div
   654→        className="flex items-center gap-1 px-2 py-1.5 rounded-t-lg border border-b-0 overflow-x-auto flex-nowrap"
   655→        style={{ backgroundColor: TOOLBAR_BG, borderColor: TOOLBAR_BORDER }}
   656→      >
   657→        {/* Tool buttons */}
   658→        {TOOLS.map((tool) => (
   659→          <button
   660→            key={tool.id}
   661→            onClick={() => {
   662→              setActiveTool(tool.id);
   663→              // Cancel ongoing drawing when switching tools
   664→              if (isDrawing) {
   665→                setIsDrawing(false);
   666→                setCurrentDrawing(null);
   667→                setTextInput(null);
   668→              }
   669→            }}
   670→            title={tool.label}
   671→            className={`shrink-0 flex items-center justify-center w-8 h-8 rounded-md transition-all cursor-pointer ${
   672→              activeTool === tool.id
   673→                ? 'bg-amber-500/20 text-amber-400 ring-1 ring-amber-500/30'
   674→                : 'text-gray-400 hover:bg-white/5 hover:text-gray-200'
   675→            }`}
   676→          >
   677→            {tool.icon}
   678→          </button>
   679→        ))}
   680→
   681→        {/* Separator */}
   682→        <div className="w-px h-6 bg-white/10 mx-1 shrink-0" />
   683→
   684→        {/* Color picker */}
   685→        {PALETTE.map((c) => (
   686→          <button
   687→            key={c}
   688→            onClick={() => setActiveColor(c)}
   689→            title={c}
   690→            className={`shrink-0 w-6 h-6 rounded-full border-2 transition-all cursor-pointer ${
   691→              activeColor === c ? 'border-white scale-110' : 'border-transparent hover:scale-105'
   692→            }`}
   693→            style={{ backgroundColor: c }}
   694→          />
   695→        ))}
   696→
   697→        {/* Separator */}
   698→        <div className="w-px h-6 bg-white/10 mx-1 shrink-0" />
   699→
   700→        {/* Delete last */}
   701→        <button
   702→          onClick={handleDeleteLast}
   703→          title="حذف آخر"
   704→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-400 hover:bg-red-500/15 hover:text-red-400 transition-all cursor-pointer"
   705→        >
   706→          <Trash2 className="w-3.5 h-3.5" />
   707→        </button>
   708→
   709→        {/* Clear all */}
   710→        <button
   711→          onClick={handleClearAll}
   712→          title="پاک کردن همه"
   713→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-400 hover:bg-red-500/15 hover:text-red-400 transition-all cursor-pointer"
   714→        >
   715→          <X className="w-3.5 h-3.5" />
   716→        </button>
   717→
   718→        {/* Active tool label */}
   719→        <div className="mr-auto shrink-0 text-xs text-gray-500 font-medium" dir="rtl">
   720→          {TOOLS.find((t) => t.id === activeTool)?.label}
   721→        </div>
   722→      </div>
   723→
   724→      {/* ── CHART + SVG OVERLAY ─────────────────────────────── */}
   725→      <div
   726→        ref={chartParentRef}
   727→        className="relative rounded-b-lg overflow-hidden border"
   728→        style={{ borderColor: TOOLBAR_BORDER, height }}
   729→      >
   730→        {/* Chart container */}
   731→        <div
   732→          ref={chartContainerRef}
   733→          className="w-full h-full"
   734→          data-chart
   735→        />
   736→
   737→        {/* SVG overlay for drawings */}
   738→        <svg
   739→          ref={svgRef}
   740→          className="absolute top-0 left-0"
   741→          style={{
   742→            width: chartSize.w || '100%',
   743→            height: chartSize.h || '100%',
   744→            pointerEvents: 'none',
   745→            zIndex: 10,
   746→          }}
   747→        />
   748→
   749→        {/* Text input overlay */}
   750→        {textInput && (
   751→          <div
   752→            className="absolute z-20"
   753→            style={{
   754→              left: textInput.x,
   755→              top: textInput.y - 8,
   756→            }}
   757→          >
   758→            <input
   759→              type="text"
   760→              value={textValue}
   761→              onChange={(e) => setTextValue(e.target.value)}
   762→              onKeyDown={(e) => {
   763→                if (e.key === 'Enter') handleTextSubmit();
   764→                if (e.key === 'Escape') handleTextCancel();
   765→              }}
   766→              placeholder="متن..."
   767→              suppressHydrationWarning
   768→              autoFocus
   769→              dir="rtl"
   770→              className="bg-[#0b0f1a] border border-gray-600 rounded px-2 py-1 text-sm text-white outline-none focus:border-amber-500 w-32"
   771→              onBlur={handleTextSubmit}
   772→            />
   773→          </div>
   774→        )}
   775→      </div>
   776→    </div>
   777→  );
   778→}
