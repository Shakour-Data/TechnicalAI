   347→/**
   348→ * Parse a number that might contain Persian digits, commas, or other formatting.
   349→ * Handles null/undefined/empty strings gracefully.
   350→ */
   351→function parsePersianNum(str: string | null | undefined | number): number {
   352→  if (str === null || str === undefined) return 0;
   353→  let s = String(str).trim();
   354→  if (!s || s === '-' || s === '') return 0;
   355→  // Strip trailing % sign (e.g. "0.05%" → "0.05")
   356→  if (s.endsWith('%')) s = s.slice(0, -1).trim();
   357→  // Replace commas and Persian digits
   358→  const cleaned = s.replace(/[,۰-۹]/g, (c: string) => {
   359→    const persianDigits = '۰۱۲۳۴۵۶۷۸۹';
   360→    const idx = persianDigits.indexOf(c);
   361→    return idx >= 0 ? String(idx) : c === ',' ? '' : c;
   362→  });
   363→  const n = Number(cleaned);
   364→  return isFinite(n) ? n : 0;
   365→}

   367→/**
   368→ * Check if a value looks like a date string (contains / or - and is not a number).
   369→ */
   370→function looksLikeDate(val: string | null | undefined | number): boolean {
   371→  if (!val) return false;
   372→  const s = String(val).trim();
   373→  return /[\/\-]/.test(s) && isNaN(Number(s.replace(/[,۰-۹\/\-]/g, '')));
   374→}
   375→
   376→/**
   377→ * Check if a value looks like a percentage (ends with % or is small like 0.5 to 50).
   378→ */
   379→function looksLikePercentage(val: string | null | undefined | number): boolean {
   380→  if (!val) return false;
   381→  const s = String(val).trim();
   382→  if (s.endsWith('%')) return true;
   383→  const n = parsePersianNum(val);
   384→  return !isNaN(n) && Math.abs(n) < 100 && n !== 0;
   385→}

   387→/**
   388→ * Auto-detect column mapping for TGJU API response.
   389→ * Expected format: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   390→ * But the API may change column order.
   391→ */
   392→function detectColumnMap(row: (string | number)[]): {
   393→  openIdx: number; highIdx: number; lowIdx: number; closeIdx: number;
   394→  jalaliDateIdx: number; gregorianDateIdx: number;
   395→} | null {
   396→  if (!row || row.length < 6) return null;
   397→
   398→  let jalaliDateIdx = -1;
   399→  let gregorianDateIdx = -1;
   400→  const numericIndices: number[] = [];
   401→  const percentageIndices: number[] = [];
   402→
   403→  for (let i = 0; i < row.length; i++) {
   404→    const val = row[i];
   405→    if (looksLikeDate(val)) {
   406→      // Jalali dates have Persian digits or format like 1404/05/25
   407→      const s = String(val);
   408→      if (/[۰-۹]/.test(s) || /^14\d{2}/.test(s.replace(/[۰-۹]/g, (d: string) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d))))) {
   409→        if (jalaliDateIdx === -1) jalaliDateIdx = i;
   410→      } else {
   411→        if (gregorianDateIdx === -1) gregorianDateIdx = i;
   412→      }
   413→    } else if (looksLikePercentage(val)) {
   414→      percentageIndices.push(i);
   415→    } else {
   416→      const n = parsePersianNum(val);
   417→      if (isFinite(n) && n !== 0) {
   418→        numericIndices.push(i);
   419→      }
   420→    }
   421→  }
   422→
   423→  // If we couldn't auto-detect dates, fall back to assumed positions
   424→  if (jalaliDateIdx === -1 && row.length >= 8) {
   425→    // Default assumption: row[6] = gregorian, row[7] = jalali
   426→    if (looksLikeDate(row[7])) jalaliDateIdx = 7;
   427→    else if (looksLikeDate(row[6])) jalaliDateIdx = 6;
   428→    if (gregorianDateIdx === -1 && looksLikeDate(row[6]) && jalaliDateIdx !== 6) gregorianDateIdx = 6;
   429→    else if (gregorianDateIdx === -1 && looksLikeDate(row[7]) && jalaliDateIdx !== 7) gregorianDateIdx = 7;
   430→  }
   431→
   432→  // Numeric columns (excluding percentages) should be OHLC + change
   433→  // OHLC: typically 4 largest absolute values among non-percentage numerics
   434→  // change: the remaining one
   435→  const ohlcCandidates = numericIndices.filter(i => !percentageIndices.includes(i));
   436→
   437→  if (ohlcCandidates.length < 4) {
   438→    // Fallback: assume first 4 numeric columns are OHLC
   439→    if (numericIndices.length >= 4) {
   440→      return {
   441→        openIdx: numericIndices[0],
   442→        highIdx: numericIndices[2],
   443→        lowIdx: numericIndices[1],
   444→        closeIdx: numericIndices[3],
   445→        jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : 7,
   446→        gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : 6,
   447→      };
   448→    }
   449→    return null;
   450→  }
   451→
   452→  // Sort by absolute value descending to identify OHLC (largest) vs change (smallest)
   453→  const sorted = ohlcCandidates
   454→    .map(i => ({ idx: i, val: Math.abs(parsePersianNum(row[i])) }))
   455→    .sort((a, b) => b.val - a.val);
   456→
   457→  // The 4 largest should be OHLC. Among them, identify high (max) and low (min).
   458→  // open and close are the other two.
   459→  const top4 = sorted.slice(0, 4);
   460→  const maxIdx = top4.reduce((a, b) => a.val > b.val ? a : b).idx;
   461→  const minIdx = top4.reduce((a, b) => a.val < b.val ? a : b).idx;
   462→  const others = top4.filter(x => x.idx !== maxIdx && x.idx !== minIdx);
   463→
   464→  // Among the two remaining, the one closer to high is open, the other is close
   465→  // (heuristic: in an uptrend open < close, but we can't assume direction)
   466→  // Default: first = open, second = close
   467→  const openIdx = others[0]?.idx ?? ohlcCandidates[0];
   468→  const closeIdx = others[1]?.idx ?? ohlcCandidates[3];
   469→
   470→  return {
   471→    openIdx,
   472→    highIdx: maxIdx,
   473→    lowIdx: minIdx,
   474→    closeIdx,
   475→    jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : row.length - 1,
   476→    gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : row.length - 2,
   477→  };
   478→}

   480→/**
   481→ * Validate a candle's OHLC values for sanity.
   482→ * Returns true if the candle is valid.
   483→ */
   484→function isValidCandle(o: number, h: number, l: number, c: number): boolean {
   485→  if (!isFinite(o) || !isFinite(h) || !isFinite(l) || !isFinite(c)) return false;
   486→  if (o <= 0 || h <= 0 || l <= 0 || c <= 0) return false;
   487→  // High must be >= Low, Open, Close
   488→  if (h < l || h < o || h < c) return false;
   489→  // Low must be <= High, Open, Close
   490→  if (l > o || l > c) return false;
   491→  // No infinite or extreme values
   492→  if (o > 1e15 || h > 1e15 || l > 1e15 || c > 1e15) return false;
   493→  return true;
   494→}

   508→/* ─── Fetch live instruments (currency + gold from tgju-api) + static list ── */
   509→
   510→export async function fetchTgjuInstruments(): Promise<TgjuInstrument[]> {
   511→  const now = Date.now();
   512→  if (liveInstrumentsCache.data && now - liveInstrumentsCache.ts < INSTRUMENTS_TTL) {
   513→    return liveInstrumentsCache.data;
   514→  }
   515→
   516→  const controller = new AbortController();
   517→  const timeout = setTimeout(() => controller.abort(), 10000);
   518→
   519→  try {
   520→    const [currencyRes, goldRes] = await Promise.all([
   521→      fetch(`${TGJU_API_BASE}/api/price/currency`, { signal: controller.signal }),
   522→      fetch(`${TGJU_API_BASE}/api/price/gold`, { signal: controller.signal }),
   523→    ]);
   524→
   525→    clearTimeout(timeout);
   526→
   527→    const instruments: TgjuInstrument[] = [];
   528→
   529→    if (currencyRes.ok) {
   530→      const currencies: TgjuPriceItem[] = await currencyRes.json();
   531→      for (const c of currencies) {
   532→        const price = parsePersianNum(c.price);
   533→        const low = parsePersianNum(c.low_price);
   534→        const high = parsePersianNum(c.high_price);
   535→        instruments.push({
   536→          title: c.title,
   537→          key: c.key,
   538→          price,
   539→          highPrice: high,
   540→          lowPrice: low,
   541→          change: 0,
   542→          changePercent: 0,
   543→          category: 'currency',
   544→        });
   545→      }
   546→    }
   547→
   548→    if (goldRes.ok) {
   549→      const goldCategories: TgjuGoldCategory[] = await goldRes.json();
   550→      for (const cat of goldCategories) {
   551→        for (const item of cat.prices) {
   552→          const price = parsePersianNum(item.price);
   553→          const low = parsePersianNum(item.low_price);
   554→          const high = parsePersianNum(item.high_price);
   555→
   556→          let category: TgjuCategoryType = 'gold';
   557→          if (item.key.startsWith('ime_fund_')) category = 'gold_etf';
   558→          else if (item.key.startsWith('silver_')) category = 'silver';
   559→
   560→          instruments.push({
   561→            title: item.title,
   562→            key: item.key,
   563→            price,
   564→            highPrice: high,
   565→            lowPrice: low,
   566→            change: 0,
   567→            changePercent: 0,
   568→            category,
   569→            groupTitle: cat.title,
   570→          });
   571→        }
   572→      }
   573→    }
   574→
   575→    // Add static instruments (crypto, world indices, forex, energy, metals, commodities)
   576→    for (const s of STATIC_INSTRUMENTS) {
   577→      const cached = staticPriceCache.get(s.key);
   578→      const price = cached?.close ?? 0;
   579→      instruments.push({
   580→        title: s.title,
   581→        key: s.key,
   582→        price,
   583→        highPrice: cached?.high ?? 0,
   584→        lowPrice: cached?.low ?? 0,
   585→        change: cached ? (cached.close - cached.prevClose) : 0,
   586→        changePercent: cached && cached.prevClose > 0 ? ((cached.close - cached.prevClose) / cached.prevClose) * 100 : 0,
   587→        category: s.category,
   588→        groupTitle: s.groupTitle,
   589→      });
   590→    }
   591→
   592→    liveInstrumentsCache.data = instruments;
   593→    liveInstrumentsCache.ts = now;
   594→
   595→    // ── Background: fetch chart data for currency/gold to get accurate daily change ──
   596→    backgroundFetchCurrencyGoldChange(instruments).catch(() => {});
   597→
   598→    // ── Background: fetch Yahoo prices for static instruments with price=0 (TGJU API 403) ──
   599→    backgroundFetchYahooFallbackPrices(instruments).catch(() => {});
   600→
   601→    return instruments;
   602→  } catch {
   603→    clearTimeout(timeout);
   604→    return liveInstrumentsCache.data || [];
   605→  }
   606→}

   608→/**
   609→ * Background task: fetch last 2 candles for currency/gold items to compute accurate daily change.
   610→ * Uses direct fetch to avoid queue contention.
   611→ * Only updates change/changePercent — keeps the live price from tgju-api.
   612→ */
   613→async function backgroundFetchCurrencyGoldChange(instruments: TgjuInstrument[]) {
   614→  const currencyGoldItems = instruments.filter(
   615→    (i) => i.category === 'currency' || i.category === 'gold' || i.category === 'silver' || i.category === 'gold_etf'
   616→  );
   617→
   618→  // Limit to first 10 to avoid rate-limiting (most important: USD, EUR, gold coin, etc.)
   619→  const priorityKeys = currencyGoldItems.slice(0, 10).map((i) => i.key);
   620→
   621→  for (const key of priorityKeys) {
   622→    try {
   623→      const url = `${TGJU_CHART_API}/${key}?lang=fa&order_dir=desc&start=0&length=2`;
   624→      const res = await fetch(url, { signal: AbortSignal.timeout(10_000) });
   625→      if (!res.ok) continue;
   626→      const jsonStr = await res.text();
   627→      if (!jsonStr.startsWith('{')) continue;
   628→
   629→      const data = JSON.parse(jsonStr);
   630→      const rows: string[][] = data.data;
   631→      if (!rows || rows.length < 2) continue;
   632→
   633→      const newest = rows[0];
   634→      const prev = rows[1];
   635→      const latestClose = parsePersianNum(newest[3]);
   636→      const prevClose = parsePersianNum(prev[3]);
   637→
   638→      if (latestClose > 0 && prevClose > 0) {
   639→        const inst = liveInstrumentsCache.data?.find((i) => i.key === key);
   640→        if (inst) {
   641→          inst.change = latestClose - prevClose;
   642→          inst.changePercent = ((latestClose - prevClose) / prevClose) * 100;
   643→        }
   644→      }
   645→    } catch {
   646→      // skip individual errors
   647→    }
   648→  }
   649→}

   659→/**
   660→ * Background task: fetch Yahoo Finance prices for static instruments that have price=0
   661→ * (happens when api.tgju.org returns 403). Updates the live cache in-place.
   662→ */
   663→async function backgroundFetchYahooFallbackPrices(instruments: TgjuInstrument[]) {
   664→  // Collect static instruments with price=0 that have a Yahoo mapping
   665→  const zeroPriceItems = instruments.filter(
   666→    (i) => i.price === 0 && i.key in TGJU_TO_YAHOO_MAP
   667→  );
   668→
   669→  if (zeroPriceItems.length === 0) return;
   670→
   671→  console.log(`[TGJU] ${zeroPriceItems.length} static instruments have price=0, fetching from Yahoo Finance as fallback...`);
   672→
   673→  // Collect unique Yahoo symbols
   674→  const keyToSymbol = new Map<string, string>();
   675→  const symbolToKeys = new Map<string, string[]>();
   676→  for (const item of zeroPriceItems) {
   677→    const yahooSym = TGJU_TO_YAHOO_MAP[item.key];
   678→    if (!yahooSym) continue;
   679→    keyToSymbol.set(item.key, yahooSym);
   680→    if (!symbolToKeys.has(yahooSym)) symbolToKeys.set(yahooSym, []);
   681→    symbolToKeys.get(yahooSym)!.push(item.key);
   682→  }
   683→
   684→  const uniqueSymbols = [...symbolToKeys.keys()];
   685→
   686→  try {
   687→    // Dynamic import to avoid loading yahoo-finance2 for TSE-only users
   688→    const YahooFinance = (await import('yahoo-finance2')).default;
   689→    const yahooFinance = new YahooFinance({ suppressNotices: ['yahooSurvey'], validation: { logErrors: false } });
   690→
   691→    // Fetch in batches of 15 (smaller batches reduce Yahoo validation errors)
   692→    const BATCH_SIZE = 15;
   693→    for (let i = 0; i < uniqueSymbols.length; i += BATCH_SIZE) {
   694→      const batch = uniqueSymbols.slice(i, i + BATCH_SIZE);
   695→      try {
   696→        const results = await yahooFinance.quote(batch, {
   697→          fields: ['symbol', 'regularMarketPrice', 'regularMarketChange', 'regularMarketChangePercent', 'regularMarketPreviousClose', 'regularMarketDayHigh', 'regularMarketDayLow'],
   698→        });
   699→
   700→        for (const r of results) {
   701→          if (r.regularMarketPrice == null) continue;
   702→          const tgjuKeys = symbolToKeys.get(r.symbol);
   703→          if (!tgjuKeys) continue;
   704→
   705→          for (const tgjuKey of tgjuKeys) {
   706→            const inst = liveInstrumentsCache.data?.find((x) => x.key === tgjuKey);
   707→            if (inst && inst.price === 0) {
   708→              inst.price = r.regularMarketPrice;
   709→              inst.highPrice = r.regularMarketDayHigh ?? r.regularMarketPrice;
   710→              inst.lowPrice = r.regularMarketDayLow ?? r.regularMarketPrice;
   711→              inst.change = r.regularMarketChange ?? 0;
   712→              inst.changePercent = r.regularMarketChangePercent ?? 0;
   713→
   714→              // Also update static price cache
   715→              staticPriceCache.set(tgjuKey, {
   716→                close: r.regularMarketPrice,
   717→                high: r.regularMarketDayHigh ?? r.regularMarketPrice,
   718→                low: r.regularMarketDayLow ?? r.regularMarketPrice,
   719→                prevClose: r.regularMarketPreviousClose ?? r.regularMarketPrice,
   720→                ts: Date.now(),
   721→              });
   722→            }
   723→          }
   724→        }
   725→      } catch (err) {
   726→        const errMsg = err instanceof Error ? err.message : String(err);
   727→        if (errMsg.includes('validation') || errMsg.includes('FailedYahooValidationError')) {
   728→          // Batch failed due to schema validation — retry individually
   729→          console.warn(`[TGJU] Yahoo fallback batch ${Math.floor(i / BATCH_SIZE) + 1} validation error, retrying individually...`);
   730→          for (const symbol of batch) {
   731→            try {
   732→              const singleResults = await yahooFinance.quote([symbol], {
   733→                fields: ['symbol', 'regularMarketPrice', 'regularMarketChange', 'regularMarketChangePercent', 'regularMarketPreviousClose', 'regularMarketDayHigh', 'regularMarketDayLow'],
   734→              });
   735→              for (const r of singleResults) {
   736→                if (r.regularMarketPrice == null) continue;
   737→                const tgjuKeys = symbolToKeys.get(r.symbol);
   738→                if (!tgjuKeys) continue;
   739→                for (const tgjuKey of tgjuKeys) {
   740→                  const inst = liveInstrumentsCache.data?.find((x) => x.key === tgjuKey);
   741→                  if (inst && inst.price === 0) {
   742→                    inst.price = r.regularMarketPrice;
   743→                    inst.highPrice = r.regularMarketDayHigh ?? r.regularMarketPrice;
   744→                    inst.lowPrice = r.regularMarketDayLow ?? r.regularMarketPrice;
   745→                    inst.change = r.regularMarketChange ?? 0;
   746→                    inst.changePercent = r.regularMarketChangePercent ?? 0;
   747→                    staticPriceCache.set(tgjuKey, {
   748→                      close: r.regularMarketPrice,
   749→                      high: r.regularMarketDayHigh ?? r.regularMarketPrice,
   750→                      low: r.regularMarketDayLow ?? r.regularMarketPrice,
   751→                      prevClose: r.regularMarketPreviousClose ?? r.regularMarketPrice,
   752→                      ts: Date.now(),
   753→                    });
   754→                  }
   755→                }
   756→              }
   757→            } catch {
   758→              // skip individual symbol errors
   759→            }
   760→          }
   761→        } else {
   762→          console.warn(`[TGJU] Yahoo fallback batch ${Math.floor(i / BATCH_SIZE) + 1} failed:`, errMsg);
   763→        }
   764→      }
   765→    }
   766→
   767→    const filledCount = zeroPriceItems.filter((i) => i.price > 0).length;
   768→    console.log(`[TGJU] Yahoo fallback filled ${filledCount}/${zeroPriceItems.length} instrument prices`);
   769→  } catch (err) {
   770→    console.error(`[TGJU] Yahoo fallback fetch failed:`, err);
   771→  }
   772→}

   774→/* ─── Fetch historical OHLC data ── */
   775→
   776→/**
   777→ * Fetch historical OHLC data from TGJU chart API.
   778→ * Strategy:
   779→ * 1. Try direct fetch (fast, ~100ms) — works when Cloudflare is not blocking
   780→ * 2. If 403/blocked, fall back to z-ai page_reader (bypasses Cloudflare, ~3s)
   781→ * 3. If both fail, return cached data if available
   782→ *
   783→ * All results are cached for HISTORY_TTL (30 min) to minimize page_reader calls.
   784→ */
   785→export async function fetchTgjuHistory(tgjuKey: string): Promise<TgjuOHLC[]> {
   786→  const cached = historyCache.get(tgjuKey);
   787→  const now = Date.now();
   788→  if (cached && now - cached.ts < HISTORY_TTL) {
   789→    return cached.data;
   790→  }
   791→
   792→  // ── Step 1: Try direct fetch (fast path) ──
   793→  const directResult = await tryDirectFetch(tgjuKey);
   794→  if (directResult && directResult.length >= 30) {
   795→    cacheHistory(tgjuKey, directResult, now);
   796→    return directResult;
   797→  }
   798→
   799→  // ── Step 2: Fall back to page_reader (bypasses Cloudflare) ──
   800→  console.log(`[TGJU] Direct fetch returned ${directResult?.length ?? 0} candles for ${tgjuKey}, trying page_reader...`);
   801→  const pageReaderResult = await fetchTgjuHistoryViaPageReader(tgjuKey);
   802→  if (pageReaderResult && pageReaderResult.length >= 30) {
   803→    cacheHistory(tgjuKey, pageReaderResult, now);
   804→    return pageReaderResult;
   805→  }
   806→
   807→  // ── Step 3: Both failed — return stale cache or empty ──
   808→  console.warn(`[TGJU] All methods failed for ${tgjuKey}, returning ${cached?.data?.length ?? 0} cached candles`);
   809→  return cached?.data || [];
   810→}

   812→/**
   813→ * Try direct HTTP fetch to TGJU chart API.
   814→ * Returns parsed candles or null on failure.
   815→ */
   816→async function tryDirectFetch(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   817→  try {
   818→    const url = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=365`;
   819→    const controller = new AbortController();
   820→    const timer = setTimeout(() => controller.abort(), 12_000);
   821→
   822→    const res = await fetch(url, { signal: controller.signal, headers: { 'Accept': 'application/json' } });
   823→    clearTimeout(timer);
   824→
   825→    if (!res.ok) {
   826→      console.warn(`[TGJU] Direct fetch HTTP ${res.status} for ${tgjuKey}`);
   827→      return null;
   828→    }
   829→
   830→    const jsonStr = await res.text();
   831→    if (!jsonStr.startsWith('{')) {
   832→      console.warn(`[TGJU] Direct fetch non-JSON for ${tgjuKey}`);
   833→      return null;
   834→    }
   835→
   836→    return parseTgjuChartData(tgjuKey, jsonStr);
   837→  } catch (err) {
   838→    console.warn(`[TGJU] Direct fetch error for ${tgjuKey}:`, err instanceof Error ? err.message : err);
   839→    return null;
   840→  }
   841→}
   842→
   843→/**
   844→ * Fetch TGJU historical data via z-ai CLI page_reader (bypasses Cloudflare).
   845→ * Uses child_process to call `z-ai function -n page_reader` which runs
   846→ * a headless browser internally and can solve Cloudflare challenges.
   847→ * Slower (~5-15s) but reliable when direct fetch is blocked.
   848→ */
   849→async function fetchTgjuHistoryViaPageReader(tgjuKey: string): Promise<TgjuOHLC[] | null> {
   850→  const apiUrl = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=365`;
   851→
   852→  try {
   853→    // Use z-ai-web-dev-sdk directly (works within Next.js process)
   854→    const ZAI = (await import('z-ai-web-dev-sdk')).default;
   855→    const zai = await ZAI.create();
   856→    console.log(`[TGJU] Using z-ai SDK page_reader for ${tgjuKey}...`);
   857→    const result = await zai.functions.invoke('page_reader', { url: apiUrl });
   858→
   859→    if (result.code !== 200 || !result.data?.html) {
   860→      console.warn(`[TGJU] page_reader SDK failed for ${tgjuKey}: code=${result.code}`);
   861→      return null;
   862→    }
   863→
   864→    // Extract JSON from <pre> tag in the HTML
   865→    const html = result.data.html;
   866→    const preMatch = html.match(/<pre[^>]*>([\s\S]*?)<\/pre>/);
   867→    if (!preMatch) {
   868→      console.warn(`[TGJU] page_reader: no <pre> block found for ${tgjuKey}`);
   869→      return null;
   870→    }
   871→
   872→    // Unescape HTML entities and strip HTML tags from values (e.g. <span class="high" dir="ltr">1000</span> → 1000)
   873→    const jsonStr = preMatch[1]
   874→      .replace(/<[^>]*>/g, '')        // Strip ALL HTML tags first
   875→      .replace(/&quot;/g, '"')
   876→      .replace(/&amp;/g, '&')
   877→      .replace(/&lt;/g, '<')
   878→      .replace(/&gt;/g, '>')
   879→      .replace(/\s+/g, ' ')           // Normalize whitespace
   880→      .trim();
   881→
   882→    if (!jsonStr.startsWith('{')) {
   883→      console.warn(`[TGJU] page_reader: extracted content is not JSON for ${tgjuKey}`);
   884→      return null;
   885→    }
   886→
   887→    return parseTgjuChartData(tgjuKey, jsonStr);
   888→  } catch (err) {
   889→    console.error(`[TGJU] page_reader SDK error for ${tgjuKey}:`, err instanceof Error ? err.message : err);
   890→    return null;
   891→  }
   892→}

   894→/**
   895→ * Parse the JSON data from TGJU chart API response.
   896→ * Used by both direct fetch and page_reader paths.
   897→ */
   898→function parseTgjuChartData(tgjuKey: string, jsonStr: string): TgjuOHLC[] | null {
   899→  try {
   900→    const parsed = JSON.parse(jsonStr);
   901→    const rawRows: (string | number)[][] = parsed.data;
   902→
   903→    if (!rawRows || rawRows.length === 0) {
   904→      console.warn(`[TGJU] Empty data array for ${tgjuKey}`);
   905→      return null;
   906→    }
   907→
   908→    // Sanitize row values: strip any residual HTML tags and trim
   909→    const rows = rawRows.map(row =>
   910→      row.map(cell => {
   911→        const s = String(cell).trim();
   912→        // Strip HTML tags if any remain (safety for direct fetch path)
   913→        return s.replace(/<[^>]*>/g, '').replace(/\s+/g, ' ');
   914→      })
   915→    );
   916→
   917→    // ── Auto-detect column mapping from first valid row ──
   918→    let colMap = detectColumnMap(rows[0]);
   919→    if (!colMap) {
   920→      for (let r = 1; r < Math.min(5, rows.length); r++) {
   921→        colMap = detectColumnMap(rows[r]);
   922→        if (colMap) break;
   923→      }
   924→    }
   925→    if (!colMap) {
   926→      console.warn(`[TGJU] Could not auto-detect columns for ${tgjuKey}, using default order`);
   927→      colMap = { openIdx: 0, highIdx: 2, lowIdx: 1, closeIdx: 3, jalaliDateIdx: 7, gregorianDateIdx: 6 };
   928→    } else {
   929→      console.log(`[TGJU] Column map for ${tgjuKey}: O=${colMap.openIdx} H=${colMap.highIdx} L=${colMap.lowIdx} C=${colMap.closeIdx} Date=${colMap.jalaliDateIdx}`);
   930→    }
   931→
   932→    const { openIdx, highIdx, lowIdx, closeIdx, jalaliDateIdx, gregorianDateIdx } = colMap;
   933→
   934→    // ── Parse rows ──
   935→    const candles: TgjuOHLC[] = [];
   936→    let invalidCount = 0;
   937→
   938→    for (const row of rows) {
   939→      const rawDate = String(row[jalaliDateIdx] || row[gregorianDateIdx] || '');
   940→      const date = rawDate.replace(/[\/]/g, '-');
   941→      const o = parsePersianNum(row[openIdx]);
   942→      const h = parsePersianNum(row[highIdx]);
   943→      const l = parsePersianNum(row[lowIdx]);
   944→      const c = parsePersianNum(row[closeIdx]);
   945→
   946→      if (!isValidCandle(o, h, l, c) || date.length === 0) {
   947→        invalidCount++;
   948→        continue;
   949→      }
   950→
   951→      candles.push({ date, open: o, high: h, low: l, close: c });
   952→    }
   953→
   954→    if (invalidCount > 0) {
   955→      console.warn(`[TGJU] ${tgjuKey}: ${invalidCount}/${rows.length} invalid candles skipped`);
   956→    }
   957→
   958→    if (candles.length === 0) {
   959→      return null;
   960→    }
   961→
   962→    // ── Detect and fix data drift ──
   963→    let driftCount = 0;
   964→    for (let i = 1; i < candles.length; i++) {
   965→      const prevClose = candles[i - 1].close;
   966→      const currClose = candles[i].close;
   967→      if (prevClose > 0) {
   968→        const pctChange = Math.abs(currClose - prevClose) / prevClose;
   969→        if (pctChange > 0.5) {
   970→          driftCount++;
   971→        }
   972→      }
   973→    }
   974→    if (driftCount > candles.length * 0.1) {
   975→      console.warn(`[TGJU] ${tgjuKey}: Possible data drift detected (${driftCount} suspicious jumps out of ${candles.length} candles)`);
   976→    }
   977→
   978→    // Reverse: API returns oldest-first (asc), we want newest-last for charting
   979→    const firstDate = candles[0].date;
   980→    const lastDate = candles[candles.length - 1].date;
   981→    if (firstDate > lastDate) {
   982→      candles.reverse();
   983→    }
   984→
   985→    console.log(`[TGJU] ${tgjuKey}: Parsed ${candles.length} candles (${candles[0]?.date} to ${candles[candles.length - 1]?.date})`);
   986→
   987→    return candles;
   988→  } catch (err) {
   989→    console.error(`[TGJU] Failed to parse chart data for ${tgjuKey}:`, err);
   990→    return null;
   991→  }
   992→}

   994→/**
   995→ * Cache parsed history data and update related caches.
   996→ */
   997→function cacheHistory(tgjuKey: string, candles: TgjuOHLC[], now: number): void {
   998→  historyCache.set(tgjuKey, { data: candles, ts: now });
   999→
  1000→  // Also update static price cache for this instrument
  1001→  if (candles.length >= 2) {
  1002→    const last = candles[candles.length - 1];
  1003→    const prev = candles[candles.length - 2];
  1004→    staticPriceCache.set(tgjuKey, {
  1005→      close: last.close,
  1006→      high: last.high,
  1007→      low: last.low,
  1008→      prevClose: prev.close,
  1009→      ts: now,
  1010→    });
  1011→
  1012→    // Update live instruments cache if this is a static instrument
  1013→    if (liveInstrumentsCache.data) {
  1014→      const inst = liveInstrumentsCache.data.find((i) => i.key === tgjuKey);
  1015→      if (inst && inst.price === 0) {
  1016→        inst.price = last.close;
  1017→        inst.highPrice = last.high;
  1018→        inst.lowPrice = last.low;
  1019→        inst.change = last.close - prev.close;
  1020→        inst.changePercent = prev.close > 0 ? ((last.close - prev.close) / prev.close) * 100 : 0;
  1021→      }
  1022→    }
  1023→  }
  1024→}
