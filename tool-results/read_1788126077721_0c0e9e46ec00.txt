    15→// === Helper Functions =========================================================
    16→
    17→function clamp(v: number, min: number, max: number): number {
    18→  return Math.max(min, Math.min(max, v));
    19→}
    20→
    21→function normalize(arr: number[]): number[] {
    22→  const sum = arr.reduce((a, b) => a + b, 0);
    23→  if (sum === 0) return arr.map(() => 1 / arr.length);
    24→  return arr.map((v) => v / sum);
    25→}
    26→
    27→function sigmoid(x: number): number {
    28→  return 1 / (1 + Math.exp(-x));
    29→}

  1027→    1
  1028→  );
  1029→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1030→
  1031→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1032→  const overboughtRisk =
  1033→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1034→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1035→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1036→
  1037→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1038→  const oversoldBounce =
  1039→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1040→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1041→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1042→
  1043→  return {
  1044→    input,
  1045→    adxNorm,
  1046→    srProximity,
  1047→    oscillatorExtreme,
  1048→    divergenceProxy,
  1049→    overboughtRisk,
  1050→    oversoldBounce,
  1051→  };
  1052→}
  1053→
  1054→// === Node Probability Computation =============================================
  1055→// For each non-terminal node, returns an array of probabilities for its
  1056→// outgoing edges. The order MUST match the order of edges in createEdges().
  1057→// =============================================================================
  1058→
  1059→function computeNodeProbabilities(
  1060→  nodeId: string,
  1061→  ctx: ProbContext,
  1062→  _numChildren: number
  1063→): number[] {
  1064→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1065→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1066→
  1067→  switch (nodeId) {
  1068→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1069→    case 'ROOT': {
  1070→      const trendW = adxNorm * 0.5 + input.mlTrend * 0.18 + 0.1;
  1071→      const breakW = srProximity * 0.3 + (1 - adxNorm) * 0.08 + 0.08;
  1072→      const revW = ctx.divergenceProxy * 0.35 + (1 - adxNorm) * 0.12 + 0.06;
  1073→      return normalize([trendW, breakW, revW]);
  1074→    }
  1075→
  1076→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1077→    case 'N_TREND': {
  1078→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1079→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1080→      const flatUncertainty = (1 - Math.abs(input.bullConsensus - 0.5) * 2);
  1081→      const flat = flatUncertainty * clamp(1 - adxNorm * 1.5, 0, 1) * 0.45;
  1082→      return normalize([bull, bear, flat]);
  1083→    }
  1084→
  1085→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1086→    case 'N_T_BULL': {
  1087→      const momHigh =
  1088→        clamp(
  1089→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1090→            (input.macdHist > 0
  1091→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1092→              : 0),
  1093→          0,
  1094→          1
  1095→        ) * input.mlMomentum;
  1096→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1097→      return normalize([momHigh, momMod, overboughtRisk]);
  1098→    }
  1099→
  1100→    // ── N_T_B_MOM_HIGH → [SC8, SC9, SC7] ────────────────────────────────────
  1101→    // Strong bullish momentum — all three outcomes are bullish (no range)
  1102→    case 'N_T_B_MOM_HIGH': {
  1103→      const pSC8 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1104→      const pSC9 = 0.15 + input.strengthIndex * 0.2;
  1105→      const pSC7 = 0.35 * input.confidenceIndex + 0.15;
  1106→      return normalize([pSC8, pSC9, pSC7]);
  1107→    }
  1108→
  1109→    // ── N_T_B_MOM_MOD → [SC7, SC6, SC5] ─────────────────────────────────────
  1110→    // Moderate bullish — small explicit SC5 based on uncertainty
  1111→    case 'N_T_B_MOM_MOD': {
  1112→      const pSC7 = 0.55 * input.confidenceIndex + 0.2;
  1113→      const pSC6 = 0.55 * (1 - input.confidenceIndex) + 0.15;
  1114→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.02, 0.15);
  1115→      return normalize([pSC7, pSC6, pSC5]);
  1116→    }
  1117→
  1118→    // ── N_T_B_OVERBOUGHT → [SC4, SC1, SC9] ──────────────────────────────────
  1119→    // Overbought in uptrend — correction or last-gasp surge (no range)
  1120→    case 'N_T_B_OVERBOUGHT': {
  1121→      const pSC4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1122→      const pSC1 = 0.15 + input.strengthIndex * 0.15;
  1123→      const pSC9 = 0.1 + input.mlMomentum * 0.2;
  1124→      return normalize([pSC4, pSC1, pSC9]);
  1125→    }
  1126→
  1127→    // ── N_T_BEAR → [N_T_BE_MOM_LOW, N_T_BE_MOM_MOD, N_T_BE_OVERSOLD] ────
  1128→    case 'N_T_BEAR': {
  1129→      const momLow =
  1130→        clamp(
  1131→          (input.momentum < 0 ? Math.abs(input.momentum) / 50 : 0) +
  1132→            (input.macdHist < 0
  1133→              ? (Math.abs(input.macdHist) / (atrSafe * 0.5)) * 0.3
  1134→              : 0),
  1135→          0,
  1136→          1
  1137→        ) * input.mlMomentum;
  1138→      const momMod = (1 - momLow) * 0.6 * input.mlMomentum;
  1139→      return normalize([momLow, momMod, oversoldBounce]);
  1140→    }
  1141→
  1142→    // ── N_T_BE_MOM_LOW → [SC2, SC1, SC3] ────────────────────────────────────
  1143→    // Strong bearish momentum — all three outcomes are bearish (no range)
  1144→    case 'N_T_BE_MOM_LOW': {
  1145→      const pSC2 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1146→      const pSC1 = 0.15 + input.strengthIndex * 0.2;
  1147→      const pSC3 = 0.35 * input.confidenceIndex + 0.15;
  1148→      return normalize([pSC2, pSC1, pSC3]);
  1149→    }
  1150→
  1151→    // ── N_T_BE_MOM_MOD → [SC3, SC4, SC5] ────────────────────────────────────
  1152→    // Moderate bearish — small explicit SC5 based on uncertainty
  1153→    case 'N_T_BE_MOM_MOD': {
  1154→      const pSC3 = 0.55 * input.confidenceIndex + 0.2;
  1155→      const pSC4 = 0.55 * (1 - input.confidenceIndex) + 0.15;
  1156→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.02, 0.15);
  1157→      return normalize([pSC3, pSC4, pSC5]);
  1158→    }
  1159→
  1160→    // ── N_T_BE_OVERSOLD → [SC6, SC9, SC4] ───────────────────────────────────
  1161→    // Oversold in downtrend — bounce or mild continuation (no range)
  1162→    case 'N_T_BE_OVERSOLD': {
  1163→      const pSC6 = 0.55 + (1 - input.mlMomentum) * 0.2;
  1164→      const pSC9 = 0.15 + input.strengthIndex * 0.15;
  1165→      const pSC4 = 0.1 + input.mlMomentum * 0.2;
  1166→      return normalize([pSC6, pSC9, pSC4]);
  1167→    }
  1168→
  1169→    // ── N_T_FLAT → [N_T_F_VOL_LOW, N_T_F_VOL_HIGH, N_T_F_VOL_MOD] ────────
  1170→    case 'N_T_FLAT': {
  1171→      const volLow =
  1172→        clamp(1 - (input.atr / input.price) * 20, 0, 1) *
  1173→        (1 - input.mlVolatility) *
  1174→        0.7;
  1175→      const volHigh =
  1176→        clamp((input.atr / input.price) * 20, 0, 1) * input.mlVolatility * 0.7;
  1177→      const volMod = (1 - volLow - volHigh) * 0.5 + 0.1;
  1178→      return normalize([volLow, volHigh, volMod]);
  1179→    }
  1180→
  1181→    // ── N_T_F_VOL_LOW → [SC5, SC6, SC4] ─────────────────────────────────────
  1182→    case 'N_T_F_VOL_LOW': {
  1183→      const pSC5 = 0.55 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1184→      const pSC6 = 0.2 * input.bullConsensus + 0.08;
  1185→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.08;
  1186→      return normalize([pSC5, pSC6, pSC4]);
  1187→    }
  1188→
  1189→    // ── N_T_F_VOL_HIGH → [SC9, SC1, SC5] ────────────────────────────────────
  1190→    // High volatility shock — directional shock much more likely than range
  1191→    case 'N_T_F_VOL_HIGH': {
  1192→      const pSC9 = 0.4 * input.bullConsensus + 0.15;
  1193→      const pSC1 = 0.4 * (1 - input.bullConsensus) + 0.15;
  1194→      const pSC5 = clamp((1 - input.mlVolatility) * 0.12, 0.02, 0.12);
  1195→      return normalize([pSC9, pSC1, pSC5]);
  1196→    }
  1197→
  1198→    // ── N_T_F_VOL_MOD → [SC6, SC4, SC5] ─────────────────────────────────────
  1199→    // Moderate volatility — directional more likely than range
  1200→    case 'N_T_F_VOL_MOD': {
  1201→      const pSC6 = 0.38 * input.bullConsensus + 0.15;
  1202→      const pSC4 = 0.38 * (1 - input.bullConsensus) + 0.15;
  1203→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.03, 0.15);
  1204→      return normalize([pSC6, pSC4, pSC5]);
  1205→    }
  1206→
  1207→    // ── N_BREAK → [N_B_UP, N_B_DOWN, N_B_NONE] ────────────────────────────
  1208→    case 'N_BREAK': {
  1209→      const upProb = input.bullConsensus * srProximity * 0.75 + 0.08;
  1210→      const downProb = (1 - input.bullConsensus) * srProximity * 0.75 + 0.08;
  1211→      const noneProb = (1 - srProximity) * 0.35 + 0.05;
  1212→      return normalize([upProb, downProb, noneProb]);
  1213→    }
  1214→
  1215→    // ── N_B_UP → [N_B_U_VOL_C, N_B_U_VOL_W] ─────────────────────────────
  1216→    case 'N_B_UP': {
  1217→      let volConfirm: number;
  1218→      let volWeak: number;
  1219→      if (input.hasVolume) {
  1220→        volConfirm = clamp(input.mfi / 60, 0, 1) * 0.6 + 0.2;
  1221→        volWeak = 1 - volConfirm;
  1222→      } else {
  1223→        volConfirm = 0.4;
  1224→        volWeak = 0.6;
  1225→      }
  1226→      return normalize([volConfirm, volWeak]);
  1227→    }
  1228→
  1229→    // ── N_B_U_VOL_C → [SC7, SC8, SC9] ──────────────────────────────────────
  1230→    // Confirmed bullish breakout — all bullish, no range
  1231→    case 'N_B_U_VOL_C': {
  1232→      const pSC7 = 0.45 + input.confidenceIndex * 0.15;
  1233→      const pSC8 = 0.2 + input.strengthIndex * 0.15;
  1234→      const pSC9 = 0.15 + input.strengthIndex * 0.1;
  1235→      return normalize([pSC7, pSC8, pSC9]);
  1236→    }
  1237→
  1238→    // ── N_B_U_VOL_W → [SC6, SC4, SC5] ──────────────────────────────────────
  1239→    // Weak volume breakout — fake breakout risk, capped SC5
  1240→    case 'N_B_U_VOL_W': {
  1241→      const pSC6 = 0.4 + input.bullConsensus * 0.15;
  1242→      const pSC4 = 0.3 + (1 - input.bullConsensus) * 0.15;
  1243→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.12, 0.02, 0.12);
  1244→      return normalize([pSC6, pSC4, pSC5]);
  1245→    }
  1246→
  1247→    // ── N_B_DOWN → [N_B_D_VOL_C, N_B_D_VOL_W] ────────────────────────────
  1248→    case 'N_B_DOWN': {
  1249→      let volConfirm: number;
  1250→      if (input.hasVolume) {
  1251→        volConfirm = clamp((100 - input.mfi) / 60, 0, 1) * 0.6 + 0.2;
  1252→      } else {
  1253→        volConfirm = 0.4;
  1254→      }
  1255→      const volWeak = 1 - volConfirm;
  1256→      return normalize([volConfirm, volWeak]);
  1257→    }
  1258→
  1259→    // ── N_B_D_VOL_C → [SC3, SC2, SC1] ──────────────────────────────────────
  1260→    // Confirmed bearish breakout — all bearish, no range
  1261→    case 'N_B_D_VOL_C': {
  1262→      const pSC3 = 0.45 + input.confidenceIndex * 0.15;
  1263→      const pSC2 = 0.2 + input.strengthIndex * 0.15;
  1264→      const pSC1 = 0.15 + input.strengthIndex * 0.1;
  1265→      return normalize([pSC3, pSC2, pSC1]);
  1266→    }
  1267→
  1268→    // ── N_B_D_VOL_W → [SC4, SC6, SC5] ──────────────────────────────────────
  1269→    // Weak volume bearish breakout — fake breakout risk, capped SC5
  1270→    case 'N_B_D_VOL_W': {
  1271→      const pSC4 = 0.4 + (1 - input.bullConsensus) * 0.15;
  1272→      const pSC6 = 0.3 + input.bullConsensus * 0.15;
  1273→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.12, 0.02, 0.12);
  1274→      return normalize([pSC4, pSC6, pSC5]);
  1275→    }
  1276→
  1277→    // ── N_B_NONE → [SC4, SC5, SC6] ──────────────────────────────────────────
  1278→    case 'N_B_NONE': {
  1279→      const pSC5 = 0.45 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1280→      const pSC6 = 0.2 * input.bullConsensus + 0.08;
  1281→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.08;
  1282→      return normalize([pSC4, pSC5, pSC6]);
  1283→    }
  1284→
  1285→    // ── N_REVERSAL → [N_R_BULL, N_R_BEAR, N_R_NONE] ──────────────────────
  1286→    case 'N_REVERSAL': {
  1287→      const bullRev = overboughtRisk;
  1288→      const bearRev = oversoldBounce;
  1289→      const maxRevSignal = Math.max(overboughtRisk, oversoldBounce);
  1290→      const noneRev = 1 - maxRevSignal * 0.85;
  1291→      return normalize([bullRev, bearRev, noneRev]);
  1292→    }
  1293→
  1294→    // ── N_R_BULL → [N_R_B_DIV, N_R_B_CANDLE, N_R_B_SR] ──────────────────
  1295→    case 'N_R_BULL': {
  1296→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1297→      const candleProb =
  1298→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1299→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1300→      return normalize([divProb, candleProb, srProb]);
  1301→    }
  1302→
  1303→    // ── N_R_B_DIV → [SC6, SC8, SC5] ────────────────────────────────────────
  1304→    // Bullish divergence — explicit small SC5
  1305→    case 'N_R_B_DIV': {
  1306→      const pSC6 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1307→      const pSC8 = 0.2 + input.mlMomentum * 0.15;
  1308→      const pSC5 = clamp((1 - adxNorm) * 0.1, 0.02, 0.1);
  1309→      return normalize([pSC6, pSC8, pSC5]);
  1310→    }
  1311→
  1312→    // ── N_R_B_CANDLE → [SC7, SC6, SC5] ─────────────────────────────────────
  1313→    // Bullish candle pattern — explicit small SC5
  1314→    case 'N_R_B_CANDLE': {
  1315→      const pSC7 = 0.5 * input.confidenceIndex + 0.2;
  1316→      const pSC6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1317→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.1, 0.02, 0.1);
  1318→      return normalize([pSC7, pSC6, pSC5]);
  1319→    }
  1320→
  1321→    // ── N_R_B_SR → [SC6, SC5, SC4] ─────────────────────────────────────────
  1322→    // Support bounce — explicit small SC5
  1323→    // Edge order: SC6, SC5, SC4 — probabilities computed then reordered
  1324→    case 'N_R_B_SR': {
  1325→      const pSC6 = 0.5 + input.srAvgStrength * 0.2;
  1326→      const pSC4 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1327→      const pSC5 = clamp((1 - input.srAvgStrength) * 0.1, 0.02, 0.1);
  1328→      // Return in edge order: [SC6, SC5, SC4]
  1329→      return normalize([pSC6, pSC5, pSC4]);
  1330→    }
  1331→
  1332→    // ── N_R_BEAR → [N_R_BE_DIV, N_R_BE_CANDLE, N_R_BE_SR] ───────────────
  1333→    // Mirror of N_R_BULL
  1334→    case 'N_R_BEAR': {
  1335→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1336→      const candleProb =
  1337→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1338→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1339→      return normalize([divProb, candleProb, srProb]);
  1340→    }
  1341→
  1342→    // ── N_R_BE_DIV → [SC4, SC2, SC5] ───────────────────────────────────────
  1343→    // Bearish divergence — explicit small SC5
  1344→    case 'N_R_BE_DIV': {
  1345→      const pSC4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1346→      const pSC2 = 0.2 + input.mlMomentum * 0.15;
  1347→      const pSC5 = clamp((1 - adxNorm) * 0.1, 0.02, 0.1);
  1348→      return normalize([pSC4, pSC2, pSC5]);
  1349→    }
  1350→
  1351→    // ── N_R_BE_CANDLE → [SC3, SC4, SC5] ────────────────────────────────────
  1352→    // Bearish candle pattern — explicit small SC5
  1353→    case 'N_R_BE_CANDLE': {
  1354→      const pSC3 = 0.5 * input.confidenceIndex + 0.2;
  1355→      const pSC4 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1356→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.1, 0.02, 0.1);
  1357→      return normalize([pSC3, pSC4, pSC5]);
  1358→    }
  1359→
  1360→    // ── N_R_BE_SR → [SC4, SC5, SC6] ─────────────────────────────────────────
  1361→    // Resistance rejection — explicit small SC5
  1362→    // Edge order: SC4, SC5, SC6 — probabilities computed then reordered
  1363→    case 'N_R_BE_SR': {
  1364→      const pSC4 = 0.5 + input.srAvgStrength * 0.2;
  1365→      const pSC6 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1366→      const pSC5 = clamp((1 - input.srAvgStrength) * 0.1, 0.02, 0.1);
  1367→      // Return in edge order: [SC4, SC5, SC6]
  1368→      return normalize([pSC4, pSC5, pSC6]);
  1369→    }
  1370→
  1371→    // ── N_R_NONE → [SC4, SC5, SC6] ──────────────────────────────────────────
  1372→    case 'N_R_NONE': {
  1373→      const pSC5 = 0.4 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1374→      const pSC6 = 0.2 * input.bullConsensus + 0.1;
  1375→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.1;
  1376→      return normalize([pSC4, pSC5, pSC6]);
  1377→    }
  1378→
  1379→    // ── Terminal nodes (SC1–SC9) have no outgoing edges ─────────────────────
  1380→    default:
  1381→      return [];
  1382→  }
  1383→}
  1384→
  1385→// === Graph Traversal ==========================================================
  1386→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1387→// =============================================================================
  1388→
  1389→function traverseGraph(
  1390→  nodeId: string,
  1391→  pathProb: number,
  1392→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1393→  ctx: ProbContext,
  1394→  adj: Map<string, number[]>,
  1395→  edges: GraphEdge[],
  1396→  edgeProbabilities: Record<number, number>,
  1397→  nodeAccum: Record<string, number>,
  1398→  scenarios: Record<string, number>,
  1399→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1400→): void {
  1401→  // Accumulate path probability at this node
  1402→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1403→
  1404→  const outEdgeIndices = adj.get(nodeId);
  1405→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1406→
  1407→  // Compute outgoing edge probabilities for this node
  1408→  const probs = computeNodeProbabilities(
  1409→    nodeId,
  1410→    ctx,
  1411→    outEdgeIndices.length
  1412→  );
  1413→
  1414→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1415→    const edgeIdx = outEdgeIndices[i];
  1416→    const edge = edges[edgeIdx];
  1417→    const prob = probs[i] || 0;
  1418→
  1419→    // Store edge probability
  1420→    edgeProbabilities[edgeIdx] = prob;
  1421→
  1422→    // Determine branch for child nodes
  1423→    let childBranch = currentBranch;
  1424→    if (nodeId === 'ROOT') {
  1425→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1426→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1427→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1428→    }
  1429→
  1430→    const childPathProb = pathProb * prob;
  1431→
  1432→    // Check if target is a terminal node (SC1–SC9)
  1433→    const isTerminal =
  1434→      edge.to.length === 3 &&
  1435→      edge.to[0] === 'S' &&
  1436→      edge.to[1] === 'C' &&
  1437→      edge.to[2] >= '1' &&
  1438→      edge.to[2] <= '9';
  1439→
  1440→    if (isTerminal) {
  1441→      // Accumulate scenario probability
  1442→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1443→      // Accumulate branch contribution
  1444→      if (childBranch) {
  1445→        contributions[edge.to][childBranch] += childPathProb;
  1446→      }
  1447→    }
  1448→
  1449→    // Recurse into child
  1450→    traverseGraph(
  1451→      edge.to,
  1452→      childPathProb,
  1453→      childBranch,
  1454→      ctx,
  1455→      adj,
  1456→      edges,
  1457→      edgeProbabilities,
  1458→      nodeAccum,
  1459→      scenarios,
  1460→      contributions
  1461→    );
  1462→  }
  1463→}
  1464→
  1465→// === Main Export: buildDecisionGraph ===========================================
  1466→
  1467→export function buildDecisionGraph(input: GraphInput): GraphData {
  1468→  // 1. Build static graph structure
  1469→  const nodes = createNodes();
  1470→  const edges = createEdges();
  1471→  const nodePositions = createPositions();
  1472→
  1473→  // 2. Build adjacency map: nodeId → array of edge indices
  1474→  const adj = new Map<string, number[]>();
  1475→  for (let i = 0; i < edges.length; i++) {
  1476→    const from = edges[i].from;
  1477→    if (!adj.has(from)) adj.set(from, []);
  1478→    adj.get(from)!.push(i);
  1479→  }
  1480→
  1481→  // 3. Build probability context
  1482→  const ctx = buildContext(input);
  1483→
  1484→  // 4. Initialize accumulation structures
  1485→  const edgeProbabilities: Record<number, number> = {};
  1486→  const nodeAccum: Record<string, number> = {};
  1487→  const scenarios: Record<string, number> = {};
  1488→  const contributions: Record<
  1489→    string,
  1490→    { trend: number; breakout: number; reversal: number }
  1491→  > = {};
  1492→
  1493→  for (let i = 1; i <= 9; i++) {
  1494→    const key = `SC${i}`;
  1495→    scenarios[key] = 0;
  1496→    contributions[key] = { trend: 0, breakout: 0, reversal: 0 };
  1497→  }
  1498→
  1499→  // 5. Traverse graph from ROOT
  1500→  traverseGraph(
  1501→    'ROOT',
  1502→    1.0,
  1503→    null,
  1504→    ctx,
  1505→    adj,
  1506→    edges,
  1507→    edgeProbabilities,
  1508→    nodeAccum,
  1509→    scenarios,
  1510→    contributions
  1511→  );
  1512→
  1513→  // 6. Post-process scenario probabilities
  1514→  // Convert to percentage, clamp [2, 35], re-normalize to sum=100
  1515→  const rawPcts: Record<string, number> = {};
  1516→  for (let i = 1; i <= 9; i++) {
  1517→    const key = `SC${i}`;
  1518→    rawPcts[key] = Math.round((scenarios[key] || 0) * 100);
  1519→  }
  1520→
  1521→  // Clamp each to [2, 35]
  1522→  const clamped = Object.entries(rawPcts).map(([k, v]) => ({
  1523→    key: k,
  1524→    value: clamp(v, 2, 35),
  1525→  }));
  1526→
  1527→  // Re-normalize to sum=100
  1528→  const clampedSum = clamped.reduce((s, c) => s + c.value, 0);
  1529→  const finalPcts: Record<string, number> = {};
  1530→  if (clampedSum > 0) {
  1531→    // Distribute proportionally and round, then fix rounding errors
  1532→    let runningTotal = 0;
  1533→    const entries = clamped
  1534→      .map((c) => {
  1535→        const raw = (c.value / clampedSum) * 100;
  1536→        return { key: c.key, raw };
  1537→      })
  1538→      .sort((a, b) => b.raw - a.raw);
  1539→
  1540→    for (let i = 0; i < entries.length; i++) {
  1541→      const isLast = i === entries.length - 1;
  1542→      if (isLast) {
  1543→        finalPcts[entries[i].key] = clamp(Math.round(100 - runningTotal), 2, 35);
  1544→      } else {
  1545→        const rounded = clamp(Math.round(entries[i].raw), 2, 35);
  1546→        finalPcts[entries[i].key] = rounded;
  1547→        runningTotal += rounded;
  1548→      }
  1549→    }
  1550→  }
  1551→
  1552→  // Ensure no negative values due to rounding
  1553→  for (const key of Object.keys(finalPcts)) {
  1554→    if (finalPcts[key] < 0) finalPcts[key] = 0;
  1555→  }
  1556→
  1557→  // 7. Compute branch probabilities (from ROOT outgoing edges)
  1558→  const rootEdgeIndices = adj.get('ROOT') || [];
  1559→  const rootProbs = computeNodeProbabilities('ROOT', ctx, rootEdgeIndices.length);
  1560→  const branchProbabilities = {
  1561→    trend: rootProbs[0] || 0,
  1562→    breakout: rootProbs[1] || 0,
  1563→    reversal: rootProbs[2] || 0,
  1564→  };
  1565→
  1566→  // 8. Build node display values (path probability as percentage)
  1567→  const nodeValues: Record<string, string> = {};
  1568→  for (const [nodeId, accum] of Object.entries(nodeAccum)) {
  1569→    nodeValues[nodeId] = `${Math.round(accum * 100)}%`;
  1570→  }
  1571→
  1572→  // Override terminal node values with final post-processed percentages
  1573→  for (const [key, pct] of Object.entries(finalPcts)) {
  1574→    nodeValues[key] = `${pct}%`;
  1575→  }
  1576→
  1577→  // 9. Build path contributions (as percentages)
  1578→  const pathContributions: Record<
  1579→    string,
  1580→    { trend: number; breakout: number; reversal: number }
  1581→  > = {};
  1582→  for (let i = 1; i <= 9; i++) {
  1583→    const key = `SC${i}`;
  1584→    const c = contributions[key];
  1585→    const total = c.trend + c.breakout + c.reversal;
  1586→    if (total > 0) {
  1587→      pathContributions[key] = {
  1588→        trend: Math.round((c.trend / total) * 100),
  1589→        breakout: Math.round((c.breakout / total) * 100),
  1590→        reversal: Math.round((c.reversal / total) * 100),
  1591→      };
  1592→    } else {
  1593→      pathContributions[key] = { trend: 33, breakout: 34, reversal: 33 };
  1594→    }
  1595→  }
  1596→
  1597→  // 10. Compute 30-day probability trend
  1598→  const probFractions: Record<string, number> = {};
  1599→  for (const [k, v] of Object.entries(finalPcts)) {
  1600→    probFractions[k] = v / 100;
  1601→  }
  1602→  const probabilityTrend = calcTrendFromProbs(probFractions, 30);
  1603→
  1604→  // 11. Return complete GraphData
  1605→  return {
  1606→    nodes,
  1607→    edges,
  1608→    nodePositions,
  1609→    edgeProbabilities,
  1610→    nodeValues,
  1611→    branchProbabilities,
  1612→    scenarioProbabilities: finalPcts,
  1613→    pathContributions,
  1614→    probabilityTrend,
  1615→  };
  1616→}
