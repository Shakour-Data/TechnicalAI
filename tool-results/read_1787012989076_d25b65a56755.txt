   102→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
   103→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
   104→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
   105→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
   106→    supportStrengths, resistanceStrengths, priceTargets,
   107→    isIndex, isTgju, currencyUnit,
   108→    edgeWeights,
   109→  } = props;
   110→
   111→  const unit = currencyUnit || 'ریال';
   112→
   113→  // ── Instrument Context (for intelligent naming) ─────────────────
   114→  const instrumentLabel = useMemo(() => {
   115→    if (isIndex) return { name: `شاخص ${symbolName}`, type: 'index' as const, verb: 'قرار دارد' };
   116→    if (isTgju) return { name: symbolName, type: 'tgju' as const, verb: 'قرار دارد' };
   117→    return { name: `سهم ${symbolName}`, type: 'stock' as const, verb: 'قرار دارد' };
   118→  }, [symbolName, isIndex, isTgju]);
   119→
   120→  const IL = instrumentLabel;
   121→
   122→  const containerRef = useRef<HTMLDivElement>(null);
   123→  const textAnalysisRef = useRef<HTMLDivElement>(null);
   124→  const [exporting, setExporting] = useState(false);
   125→
   126→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   127→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   128→  const S1_level = supports[0] ?? currentPrice * 0.95;
   129→  const S2_level = supports[1] ?? currentPrice * 0.90;
   130→  const S3_level = supports[2] ?? currentPrice * 0.85;
   131→
   132→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   133→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   134→
   135→  // ── Dominant scenario ──────────────────────────────────────────
   136→  let highestKey = 'R3';
   137→  let highestProb = 0;
   138→  for (const key of SCENARIO_KEYS) {
   139→    if (scenarios[key].probability > highestProb) {
   140→      highestProb = scenarios[key].probability;
   141→      highestKey = key;
   142→    }
   143→  }
   144→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   145→
   146→  // ── Dynamic scenario labels (context-aware based on overallSignal) ─
   147→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal), [overallSignal]);
   148→  const strategy = dynamicMeta[highestKey];
   149→
   150→  // ── Text variety: hash-based selector for varied language ──────────
   151→  const varietySeed = useMemo(() => {
   152→    const raw = `${rsi}${macdHist}${cci}${adx}${stochK}${trendAngle}${Math.round(currentPrice)}`;
   153→    let hash = 0;
   154→    for (let i = 0; i < raw.length; i++) hash = ((hash << 5) - hash + raw.charCodeAt(i)) | 0;
   155→    return Math.abs(hash);
   156→  }, [rsi, macdHist, cci, adx, stochK, trendAngle, currentPrice]);
   157→
   158→  // ── RSI signal ─────────────────────────────────────────────────
   159→  const rsiZone = rsi > 70 ? 'overbought' : rsi > 60 ? 'weak_ob' : rsi > 40 ? 'neutral' : rsi > 30 ? 'weak_os' : 'oversold';
   160→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   161→  const rsiColor = rsi > 70 ? 'text-red-400' : rsi > 60 ? 'text-amber-400' : rsi > 40 ? 'text-gray-300' : rsi > 30 ? 'text-amber-400' : 'text-emerald-400';
   162→
   163→  // ── Stochastic signal ──────────────────────────────────────────
   164→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   165→
   166→  // ── Trend text ─────────────────────────────────────────────────
   167→  const isUp = trendDirection === 'up';
   168→  const isDown = trendDirection === 'down';
   169→  const trendLabel = isUp ? 'صعودی' : isDown ? 'نزولی' : 'خنثی';
   170→  const trendColorCls = isUp ? 'text-emerald-400' : isDown ? 'text-red-400' : 'text-amber-400';
   171→  const adxStrength = adx > 40 ? 'بسیار قوی' : adx > 25 ? 'قوی' : adx > 15 ? 'متوسط' : 'ضعیف';
   172→  const adxColor = adx > 40 ? 'text-emerald-400' : adx > 25 ? 'text-amber-400' : 'text-gray-500';
   173→  const r2Pct = (trendR2 * 100).toFixed(1);
   174→
   175→  // ── Bollinger position ─────────────────────────────────────────
   176→  const bbRange = bollingerUpper - bollingerLower;
   177→  const bbPos = bbRange > 0 ? toPersianDigits(((currentPrice - bollingerLower) / bbRange * 100).toFixed(0)) : '۵۰';
   178→  const bbAbove = currentPrice > bollingerUpper;
   179→  const bbBelow = currentPrice < bollingerLower;
   180→  const bbSignal = bbAbove ? 'بالای باند بالایی' : bbBelow ? 'زیر باند پایینی' : `داخل باندها (${bbPos}٪)`;
   181→
   182→  // ── Distance to MA ─────────────────────────────────────────────
   183→  const distMa21 = ((currentPrice - ma21) / ma21 * 100).toFixed(1);
   184→  const distMa100 = ((currentPrice - ma100) / ma100 * 100).toFixed(1);
   185→
   186→  // ── TradingView scenario colors ────────────────────────────────
   187→  const chartCandles = useMemo(() => {
   188→    if (candles.length <= 2000) return candles;
   189→    return candles.slice(-2000);
   190→  }, [candles]);
   191→
   192→  const tvScenarios = useMemo(() => {
   193→    const colors: Record<string, string> = { R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62' };
   194→    return Object.fromEntries(
   195→      SCENARIO_KEYS.map(k => [k, { targetMin: scenarios[k].targetMin, targetMax: scenarios[k].targetMax, name: scenarios[k].name, probability: scenarios[k].probability, color: colors[k] }])
   196→    );
   197→  }, [scenarios]);
   198→
   199→  // ── VDss path probabilities (for result nodes section) ──────────
   200→  const VDSS_COLORS: Record<string, string> = {
   201→    R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62',
   202→  };
   203→
   204→  const vdssPathsInfo = useMemo(() => {
   205→    if (!edgeWeights) return null;
   206→
   207→    const edgeProbs = calcEdgeProbabilities(edgeWeights);
   208→    const allPathsRaw = findAllPaths(edgeProbs);
   209→
   210→    // Calibrate path probabilities to match TA engine scenario probabilities
   211→    const rawSums: Record<string, number> = {};
   212→    for (const key of SCENARIO_KEYS) rawSums[key] = 0;
   213→    for (const p of allPathsRaw) rawSums[p.target] += p.prob;
   214→
   215→    const allPaths: PathInfo[] = allPathsRaw.map(p => {
   216→      const rawSum = rawSums[p.target] || 0.0001;
   217→      const targetProb = scenarios[p.target as keyof typeof scenarios].probability / 100;
   218→      const scale = targetProb / rawSum;
   219→      return { ...p, prob: p.prob * scale };
   220→    });
   221→
   222→    // Find top path per scenario
   223→    const topPaths: Record<string, PathInfo | null> = {};
   224→    for (const key of SCENARIO_KEYS) {
   225→      const best = allPaths.filter(p => p.target === key).sort((a, b) => b.prob - a.prob)[0] ?? null;
   226→      topPaths[key] = best;
   227→    }
   228→
   229→    return { topPaths };
   230→  }, [edgeWeights, scenarios]);
   231→
   232→  // ── Strategy recommendation text (intelligent) ────────────────────
   233→  const mfiWarning = hasVolume ? ` (مقدار ${toFa(mfi)})` : '';
   234→  const strategyText = useMemo(() => {
   235→    const v = varietySeed % 3; // 3 variants per signal type
   236→    if (rsi > 70 || (hasVolume && mfi > 80)) {
   237→      const variants = [
   238→        `هشدار اشباع خرید فعال است (RSI: ${toFa(rsi)}${mfiWarning}). فاصله تا مقاومت اول ${toFa(R1_level)} ${unit} است. بهترین استراتژی در این شرایط، صبر برای اصلاح قیمت تا محدوده حمایت ${toFa(S1_level)} تا ${toFa(S2_level)} و ورود در نقطه بهینه‌تر می‌باشد.`,
   239→        `با توجه به ورود به منطقه اشباع خرید (RSI=${toFa(rsi)}${mfiWarning}) و نزدیکی به مقاومت ${toFa(R1_level)}، ریسک اصلاح کوتاه‌مدت بالا ارزیابی می‌شود. پیشنهاد: انتظار برای بازگشت قیمت به ${toFa(S1_level)}-${toFa(S2_level)} و سپس بررسی مجدد سیگنال‌ها.`,
   240→        `${IL.name} در محدوده اشباع خرید (RSI: ${toFa(rsi)}${mfiWarning}) ${IL.verb}. با در نظر گرفتن مقاومت ${toFa(R1_level)} در بالا و حمایت ${toFa(S1_level)} در پایین، ورود جدید در این نقطه توصیه نمی‌شود. بهتر است منتظر عقبنشینی به ${toFa(S2_level)} بمانید.`,
   241→      ];
   242→      return variants[v];
   243→    }
   244→    if (rsi < 30 || (hasVolume && mfi < 20)) {
   245→      const variants = [
   246→        `اشباع فروش قابل توجه (RSI: ${toFa(rsi)}${mfiWarning}) در محدوده حمایت ${toFa(S1_level)} شناسایی شد. این شرایط می‌تواند فرصت مناسبی برای ورود باشد، با حد ضرر ${toFa(S2_level)}. هدف اول ${toFa(R1_level)} و هدف دوم ${toFa(R2_level)} ${unit}.`,
   247→        `${IL.name} به منطقه اشباع فروش رسیده (RSI=${toFa(rsi)}${mfiWarning}). فاصله تا حمایت کلیدی ${toFa(S1_level)} کم است. اگر واکنش مثبت به این حمایت مشاهده شود، بازگشت به ${toFa(R1_level)} محتمل خواهد بود.`,
   248→        `شاخص قدرت نسبی در محدوده اشباع فروش (مقدار ${toFa(rsi)}${mfiWarning}) ${IL.verb}. نزدیکی به حمایت ${toFa(S1_level)} و سیگنال فروش بیش از حد، احتمال بازگشت کوتاه‌مدت را تقویت می‌کند. حد ضرر پیشنهادی: ${toFa(S2_level)}.`,
   249→      ];
   250→      return variants[v];
   251→    }
   252→    const variants = [
   253→      `سیگنال‌ها در محدوده خنثی قرار دارند (RSI: ${toFa(rsi)}، قدرت روند: ${adxStrength}). محدوده رنج فعلی بین ${toFa(S1_level)} و ${toFa(R1_level)} است. تا خروج قیمت از این کانال، بهترین اقدام نظاره و عدم ورود است.`,
   254→      `وضعیت فعلی ${IL.name} نشان‌دهنده عدم قطعیت (RSI=${toFa(rsi)}، ADX=${toFa(adx)}). ${IL.name} در بازه ${toFa(S1_level)} تا ${toFa(R1_level)} در نوسان است. شکست معتبر یکی از این سطوح، مسیر آینده را مشخص خواهد کرد.`,
   255→      `شاخص‌ها هم‌جهت نیستند (RSI: ${toFa(rsi)}) و قدرت روند ${adxStrength} ارزیابی می‌شود. تا زمانی که ${IL.name} از محدوده ${toFa(S1_level)}-${toFa(R1_level)} خارج نشود، ورود با ریسک بالا همراه خواهد بود.`,
   256→    ];
   257→    return variants[v];

   282→    try {

   308→<html lang="fa" dir="rtl">
   309→<head>
   310→<meta charset="UTF-8">

   321→  }

   329→    display: block;

   336→    color: #4b5563;
   337→    margin-top: 16px;
   338→    padding-top: 12px;
   339→    border-top: 1px solid rgba(255,255,255,0.05);
   340→  }
   341→  @media print {
   342→    body { padding: 0; background: white; }
   343→    .footer { color: #999; }
   344→  }
   345→</style>
   346→</head>
   347→<body>
   348→<div class="page-wrapper">
   349→  <img src="data:image/png;base64,${base64}" alt="تحلیل تکنیکال ${IL.name}" />
   350→  <div class="footer">تاریخ تهیه: ${dateStr} — صرفاً جنبه تحلیلی دارد و توصیه سرمایه‌گذاری نیست.</div>
   351→</div>
   352→</body>
   353→</html>`;
   354→      const blob = new Blob([html], { type: 'text/html;charset=utf-8' });
   355→      const url = URL.createObjectURL(blob);
   356→      const a = document.createElement('a'); a.href = url; a.download = `${IL.name}-تحلیل-تکنیکال.html`; a.click();
   357→      URL.revokeObjectURL(url);
   358→    } catch (err) { console.error('HTML export failed:', err); }
   359→    finally { setExporting(false); }
   360→  }, [IL.name, captureContainer]);
   361→
   362→  // ── Export Plain Text ──────────────────────────────────────────
   363→  const handleExportText = useCallback(() => {

   367→    try {
   368→      const now = new Date();
   369→      const dateStr = toShamsi(now, 'long');
   370→      // Extract text content, clean up whitespace
   371→      const rawText = el.innerText || el.textContent || '';
   372→      const lines = rawText.split('\n').map(l => l.trim()).filter(Boolean);
   373→      const text = `تحلیل تکنیکال ${IL.name}
   374→${'═'.repeat(50)}
   375→تاریخ تهیه: ${dateStr}
   376→
   377→${lines.join('\n')}
   378→
   379→${'─'.repeat(50)}

   806→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   807→function LevelBox({ index, label, price, strength, isTarget, type, unit }: {
   808→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   809→}) {
   810→  const isSupport = type === 'support';
   811→  const isTargetType = type === 'target';
   812→
   813→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   814→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   815→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   816→
   817→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   818→  const strengthLabel = strength >= 8 ? 'بسیار قوی' : strength >= 6 ? 'قوی' : strength >= 4 ? 'متوسط' : strength >= 2 ? 'ضعیف' : 'بسیار ضعیف';
   819→  const strengthColor = strength >= 8 ? '#34c98b' : strength >= 6 ? '#3ad5db' : strength >= 4 ? '#ffb11b' : strength >= 2 ? '#ff7b32' : '#ef4d62';
   820→
   821→  // Distance from current price (if available from context)
   822→  return (
   823→    <div
   824→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   825→      style={{
   826→        background: bgAccent,
   827→        border: `1px solid ${borderAccent}`,
   828→        borderRight: `3px solid ${accentColor}`,
   829→      }}
   830→    >
   831→      {/* Label + Price */}
   832→      <div className="flex-1 min-w-0">
   833→        <div className="flex items-center gap-2">
   834→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   835→          {isTarget && <span className="text-[9px]">🎯</span>}
   836→        </div>
   837→        <span className="text-sm font-black text-gray-100 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   838→        <span className="text-[10px] text-gray-500 mr-1">{unit || 'ریال'}</span>
   839→      </div>
   840→
   841→      {/* Strength bar */}
   842→      <div className="flex flex-col items-end gap-1 min-w-[80px]">
   843→        <div className="flex items-center gap-1.5">
   844→          <span className="text-[9px] font-medium" style={{ color: strengthColor }}>{strengthLabel}</span>
   845→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(strength)}</span>
   846→          <span className="text-[9px] text-gray-500">/۱۰</span>
   847→        </div>
   848→        <div className="w-full h-1.5 rounded-full bg-white/5 overflow-hidden">
   849→          <div
   850→            className="h-full rounded-full transition-all duration-500"
   851→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   852→          />
   853→        </div>
   854→      </div>
   855→    </div>
   856→  );
   857→}
