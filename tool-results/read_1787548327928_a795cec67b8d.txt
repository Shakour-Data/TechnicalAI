    17→function sma(closes: number[], period: number): number {
    18→  if (closes.length < period) return 0;
    19→  const slice = closes.slice(-period);
    20→  return slice.reduce((a, b) => a + b, 0) / period;
    21→}
    22→
    23→function emaCalc(closes: number[], period: number): number {
    24→  if (closes.length < period) return 0;
    25→  const k = 2 / (period + 1);
    26→  let v = sma(closes.slice(0, period), period);
    27→  for (let i = period; i < closes.length; i++) {
    28→    v = closes[i] * k + v * (1 - k);
    29→  }
    30→  return v;
    31→}

    33→function calcRSI(closes: number[], period = 14): number {
    34→  if (closes.length < period + 1) return 50;
    35→  let avgGain = 0, avgLoss = 0;
    36→  for (let i = 1; i <= period; i++) {
    37→    const diff = closes[i] - closes[i - 1];
    38→    if (diff > 0) avgGain += diff; else avgLoss += Math.abs(diff);
    39→  }
    40→  avgGain /= period; avgLoss /= period;
    41→  for (let i = period + 1; i < closes.length; i++) {
    42→    const diff = closes[i] - closes[i - 1];
    43→    avgGain = (avgGain * (period - 1) + (diff > 0 ? diff : 0)) / period;
    44→    avgLoss = (avgLoss * (period - 1) + (diff < 0 ? Math.abs(diff) : 0)) / period;
    45→  }
    46→  if (avgLoss === 0) return 100;
    47→  return 100 - 100 / (1 + avgGain / avgLoss);
    48→}

    50→function calcCCI(data: OHLCV[], period = 20): number {
    51→  if (data.length < period) return 0;
    52→  const tps: number[] = [];
    53→  for (let i = data.length - period; i < data.length; i++) {
    54→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
    55→  }
    56→  const mean = tps.reduce((a, b) => a + b, 0) / tps.length;
    57→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - mean), 0) / tps.length;
    58→  return meanDev === 0 ? 0 : (tps[tps.length - 1] - mean) / (0.015 * meanDev);
    59→}
    60→
    61→function calcMFI(data: OHLCV[], period = 14): number {
    62→  if (data.length < period + 1) return 50;
    63→  let posFlow = 0, negFlow = 0;
    64→  for (let i = data.length - period; i < data.length; i++) {
    65→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
    66→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
    67→    const mf = tp * data[i].volume;
    68→    if (tp > prevTp) posFlow += mf; else negFlow += mf;
    69→  }
    70→  return negFlow === 0 ? 100 : 100 - 100 / (1 + posFlow / negFlow);
    71→}

    73→function calcStochastic(data: OHLCV[], kPeriod = 14, smoothK = 3, smoothD = 3): { k: number; d: number } {
    74→  const rawKs: number[] = [];
    75→  for (let i = kPeriod - 1; i < data.length; i++) {
    76→    let lowest = Infinity, highest = -Infinity;
    77→    for (let j = i - kPeriod + 1; j <= i; j++) {
    78→      if (data[j].low < lowest) lowest = data[j].low;
    79→      if (data[j].high > highest) highest = data[j].high;
    80→    }
    81→    const range = highest - lowest;
    82→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
    83→  }
    84→  const smoothedK: number[] = [];
    85→  for (let i = smoothK - 1; i < rawKs.length; i++) {
    86→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
    87→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
    88→  }
    89→  const k = smoothedK[smoothedK.length - 1] ?? 50;
    90→  const dSlice = smoothedK.slice(-smoothD);
    91→  const d = dSlice.length > 0 ? dSlice.reduce((a, b) => a + b, 0) / dSlice.length : 50;
    92→  return { k, d };
    93→}

    95→function calcMACD(closes: number[], fast = 12, slow = 26, sig = 9): { line: number; signal: number; histogram: number } {
    96→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
    97→  const kFast = 2 / (fast + 1), kSlow = 2 / (slow + 1), kSig = 2 / (sig + 1);
    98→  let eFast = sma(closes.slice(0, fast), fast);
    99→  let eSlow = sma(closes.slice(0, slow), slow);
   100→  const macdVals: number[] = [];
   101→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   102→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   103→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   104→    macdVals.push(eFast - eSlow);
   105→  }
   106→  if (macdVals.length < sig) return { line: 0, signal: 0, histogram: 0 };
   107→  let s = sma(macdVals.slice(0, sig), sig);
   108→  for (let i = sig; i < macdVals.length; i++) {
   109→    s = macdVals[i] * kSig + s * (1 - kSig);
   110→  }
   111→  const line = macdVals[macdVals.length - 1];
   112→  return { line, signal: s, histogram: line - s };
   113→}

   115→function calcBollingerBands(closes: number[], period = 20, mult = 2): { upper: number; middle: number; lower: number } {
   116→  if (closes.length < period) return { upper: 0, middle: 0, lower: 0 };
   117→  const slice = closes.slice(-period);
   118→  const mid = slice.reduce((a, b) => a + b, 0) / period;
   119→  const variance = slice.reduce((a, b) => a + (b - mid) ** 2, 0) / period;
   120→  const sd = Math.sqrt(variance);
   121→  return { upper: mid + mult * sd, middle: mid, lower: mid - mult * sd };
   122→}

   124→function calcADX(data: OHLCV[], period = 14): { adx: number; diPlus: number; diMinus: number } {
   125→  if (data.length < period * 2) return { adx: 0, diPlus: 0, diMinus: 0 };
   126→  let sumPDM = 0, sumMDM = 0, sumTR = 0;
   127→  for (let i = data.length - period; i < data.length; i++) {
   128→    const upMove = data[i].high - data[i - 1].high;
   129→    const downMove = data[i - 1].low - data[i].low;
   130→    sumPDM += upMove > downMove && upMove > 0 ? upMove : 0;
   131→    sumMDM += downMove > upMove && downMove > 0 ? downMove : 0;
   132→    sumTR += Math.max(
   133→      data[i].high - data[i].low,
   134→      Math.abs(data[i].high - data[i - 1].close),
   135→      Math.abs(data[i].low - data[i - 1].close)
   136→    );
   137→  }
   138→  const diPlus = sumTR > 0 ? (sumPDM / sumTR) * 100 : 0;
   139→  const diMinus = sumTR > 0 ? (sumMDM / sumTR) * 100 : 0;
   140→  const dx = (diPlus + diMinus) > 0 ? Math.abs(diPlus - diMinus) / (diPlus + diMinus) * 100 : 0;
   141→  return { adx: dx, diPlus, diMinus };
   142→}

   144→function calcATR(data: OHLCV[], period = 14): number {
   145→  if (data.length < period + 1) return 0;
   146→  let sum = 0;
   147→  for (let i = data.length - period; i < data.length; i++) {
   148→    sum += Math.max(
   149→      data[i].high - data[i].low,
   150→      Math.abs(data[i].high - data[i - 1].close),
   151→      Math.abs(data[i].low - data[i - 1].close)
   152→    );
   153→  }
   154→  return sum / period;
   155→}

   159→  // Simplified Parabolic SAR

   161→}

   163→// ─── 16-Feature Extraction for VDss ─────────────────────────────────────────

   165→const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));
   166→
   167→/**
   168→ * Extract the 16 VDSS features at a given point in the data.
   169→ * Feature order must match VDSS_FEATURE_NAMES exactly.
   170→ *
   171→ * @param data - Full OHLCV history (chronological)
   172→ * @param endIdx - Index of the candle to compute features for
   173→ * @param hasVolume - Whether data has valid volume
   174→ * @returns Array of 16 feature values in [0,1] range
   175→ */
   176→export function extractVDSSFeatures(data: OHLCV[], endIdx: number, hasVolume: boolean): number[] {
   177→  const slice = data.slice(0, endIdx + 1);
   178→  const closes = slice.map(d => d.close);
   179→  const price = closes[closes.length - 1];
   180→  if (price <= 0) return new Array(16).fill(0.5);
   181→
   182→  // ── Compute indicators ──
   183→  const rsi = calcRSI(closes);
   184→  const mfi = hasVolume ? calcMFI(slice) : 50;
   185→  const cci = calcCCI(slice);
   186→  const adxResult = calcADX(slice);
   187→  const macd = calcMACD(closes);
   188→  const stoch = calcStochastic(slice);
   189→  const bb = calcBollingerBands(closes);
   190→  const atr = calcATR(slice);
   191→  const ma21 = sma(closes, 21);
   192→  const ma100 = sma(closes, 100);
   193→  const ema12 = emaCalc(closes, 12);
   194→  const ema26 = emaCalc(closes, 26);
   195→
   196→  // ── LAYER 1: Raw scores (s_*) ──
   197→  const s_rsi = rsi > 80 ? 0.95 : rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   198→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   199→  const s_mfi = mfi > 80 ? 0.95 : mfi > 70 ? 0.85 : mfi > 60 ? 0.7 : mfi > 50 ? 0.55
   200→    : mfi > 40 ? 0.45 : mfi > 30 ? 0.3 : mfi > 20 ? 0.15 : 0.05;
   201→  const s_cci = cci > 200 ? 0.9 : cci > 100 ? 0.75 : cci > 0 ? 0.6
   202→    : cci > -100 ? 0.4 : cci > -200 ? 0.25 : 0.1;
   203→  const s_adx = adxResult.diPlus > adxResult.diMinus
   204→    ? clamp(0.5 + (adxResult.diPlus - adxResult.diMinus) / 100, 0.5, 1)
   205→    : clamp(0.5 - (adxResult.diMinus - adxResult.diPlus) / 100, 0, 0.5);
   206→  const s_macd = macd.histogram > 0 && macd.line > macd.signal ? 0.9
   207→    : macd.histogram > 0 ? 0.7 : macd.line > macd.signal ? 0.55
   208→    : macd.histogram < 0 && macd.line < macd.signal ? 0.1 : 0.3;
   209→  const s_stoch = clamp(
   210→    (stoch.k > 80 ? 0.9 : stoch.k > 70 ? 0.8 : stoch.k > 50 ? 0.6
   211→      : stoch.k > 30 ? 0.4 : stoch.k > 20 ? 0.2 : 0.1)
   212→    + (stoch.k > stoch.d ? 0.1 : -0.1), 0, 1);
   213→  const bbRange = bb.upper - bb.lower;
   214→  const s_bb = bbRange === 0 ? 0.5 : clamp((price - bb.lower) / bbRange, 0, 1);
   215→  const s_ma21 = clamp(price > ma21 ? 0.7 + ((price - ma21) / ma21) * 0.5 : 0.3 + ((price - ma21) / ma21) * 0.5, 0, 1);
   216→  const s_ma100 = clamp(price > ma100 ? 0.65 + ((price - ma100) / ma100) * 0.3 : 0.35 + ((price - ma100) / ma100) * 0.3, 0, 1);
   217→  const emaGap = ema12 > 0 && ema26 > 0 ? (ema12 - ema26) / ema26 : 0;
   218→  const s_ema = clamp(0.5 + emaGap * 5, 0, 1);
   219→  const s_atr = clamp(1 - atr / price * 10, 0.2, 1);
   220→
   221→  // Trend (R² + angle)
   222→  const trendPeriod = Math.min(21, closes.length);
   223→  const trendSlice = closes.slice(-trendPeriod);
   224→  const n = trendSlice.length;
   225→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   226→  for (let i = 0; i < n; i++) { sumX += i; sumY += trendSlice[i]; sumXY += i * trendSlice[i]; sumX2 += i * i; sumY2 += trendSlice[i] ** 2; }
   227→  const denom = n * sumX2 - sumX * sumX;
   228→  let r2 = 0, angle = 0;
   229→  if (denom !== 0) {
   230→    const slope = (n * sumXY - sumX * sumY) / denom;
   231→    const meanY = sumY / n;
   232→    let ssRes = 0, ssTot = 0;
   233→    for (let i = 0; i < n; i++) {
   234→      ssRes += (trendSlice[i] - (slope * i + meanY - slope * (n - 1) / 2)) ** 2;
   235→      ssTot += (trendSlice[i] - meanY) ** 2;
   236→    }
   237→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
   238→    angle = meanY > 0 ? Math.atan(slope / meanY) * (180 / Math.PI) : 0;
   239→  }
   240→  const tStr = r2 * Math.abs(angle) / 45;
   241→  const direction = denom !== 0 ? (n * sumXY - sumX * sumY) / denom : 0;
   242→  const s_trend = direction > 0 ? clamp(0.5 + tStr / 2, 0.5, 1)
   243→    : direction < 0 ? clamp(0.5 - tStr / 2, 0, 0.5) : 0.5;
   244→
   245→  // Simplified S/R: distance to nearest local high/low
   246→  let s_sr = 0.5;
   247→  if (slice.length > 20) {
   248→    const lookback = slice.slice(-30);
   249→    let nearestHigh = Infinity, nearestLow = 0;
   250→    for (const d of lookback) {
   251→      if (d.high > price && d.high < nearestHigh) nearestHigh = d.high;
   252→      if (d.low < price && d.low > nearestLow) nearestLow = d.low;
   253→    }
   254→    const dR = nearestHigh < Infinity ? Math.abs(price - nearestHigh) / nearestHigh : 1;
   255→    const dS = nearestLow > 0 ? Math.abs(price - nearestLow) / nearestLow : 1;
   256→    s_sr = clamp(0.5 - dR * 1.5 + dS * 1.0, 0, 1);
   257→  }
   258→
   259→  // ── LAYER 2: Momentum correction ──
   260→  const TB = 5;
   261→  let f_rsi = s_rsi, f_mfi = s_mfi, f_cci = s_cci, f_macd = s_macd, f_stoch = s_stoch;
   262→  if (slice.length > TB + 15) {
   263→    const pastSlice = slice.slice(0, -TB);
   264→    const pastCloses = pastSlice.map(d => d.close);
   265→    const rsiPast = calcRSI(pastCloses);
   266→    const mfiPast = hasVolume ? calcMFI(pastSlice) : 50;
   267→    const cciPast = calcCCI(pastSlice);
   268→    const macdPast = calcMACD(pastCloses);
   269→    const stochPast = calcStochastic(pastSlice);
   270→
   271→    f_rsi = clamp(s_rsi + (rsi - rsiPast) / 100 * 0.08, 0, 1);
   272→    f_mfi = clamp(s_mfi + (mfi - mfiPast) / 100 * 0.08, 0, 1);
   273→    f_cci = clamp(s_cci + (cci - cciPast) / 300 * 0.06, 0, 1);
   274→    f_macd = clamp(s_macd + (macd.histogram - macdPast.histogram) / (Math.abs(macd.histogram) + Math.abs(macdPast.histogram) + 1) * 3 * 0.08, 0, 1);
   275→    f_stoch = clamp(s_stoch + (stoch.k - stochPast.k) / 50 * 0.06, 0, 1);
   276→  }
   277→
   278→  // ── LAYER 2: Cross signals (binary per spec) ──
   279→  let f_stochCross = 0.1;
   280→  let f_macdCross = 0.1;
   281→  let f_div = 0.15;
   282→
   283→  if (slice.length > 20) {
   284→    const prevSlice = slice.slice(0, -1);
   285→    const prevStoch = calcStochastic(prevSlice);
   286→    const prevMacd = calcMACD(prevSlice.map(d => d.close));
   287→
   288→    // Stochastic bullish cross: K_prev < D_prev AND K_curr > D_curr
   289→    f_stochCross = (prevStoch.k < prevStoch.d && stoch.k > stoch.d) ? 0.9 : 0.1;
   290→
   291→    // MACD bullish cross: line_prev < signal_prev AND line_curr > signal_curr
   292→    f_macdCross = (prevMacd.line < prevMacd.signal && macd.line > macd.signal) ? 0.9 : 0.1;
   293→
   294→    // Divergence detection
   295→    if (closes.length > 40) {
   296→      const p20 = closes[closes.length - 21];
   297→      const rsi20 = calcRSI(closes.slice(0, -20));
   298→      const pChange = price - p20;
   299→      const rsiChange = rsi - rsi20;
   300→      if (pChange !== 0 && rsiChange !== 0) {
   301→        f_div = (pChange > 0 && rsiChange < 0) ? 0.85 : (pChange < 0 && rsiChange > 0) ? 0.85 : 0.15;
   302→      }
   303→    }
   304→  }
   305→
   306→  // ── Return 16 features in exact order ──
   307→  return [
   308→    f_rsi, f_mfi, f_cci, s_adx,
   309→    f_macd, f_stoch, s_bb, s_ma21, s_ma100, s_ema,
   310→    s_atr, s_trend, s_sr,
   311→    f_stochCross, f_macdCross, f_div,
   312→  ];
   313→}

   317→interface CachedModel {
   318→  result: AdaptiveModelResult;
   319→  time: number;
   320→}
   321→
   322→const modelCache = new Map<string, CachedModel>();
   323→const CACHE_TTL = 60 * 60 * 1000; // 1 hour
   324→
   325→export function getCachedAdaptiveModel(symbol: string): AdaptiveModelResult | null {
   326→  const cached = modelCache.get(symbol);
   327→  if (cached && Date.now() - cached.time < CACHE_TTL) return cached.result;
   328→  return null;
   329→}

   331→export function setCachedAdaptiveModel(symbol: string, result: AdaptiveModelResult): void {
   332→  modelCache.set(symbol, { result, time: Date.now() });
   333→}

   335→// ─── Main Training Function ─────────────────────────────────────────────────
   336→
   337→/**
   338→ * Train the Adaptive Weight Model for a given symbol.
   339→ * Extracts 16 VDSS features for each historical candle, creates binary labels,
   340→ * trains Logistic Regression with TimeSeriesSplit validation.
   341→ *
   342→ * @param data - Full OHLCV history (chronological, oldest first)
   343→ * @param symbol - Symbol identifier for caching
   344→ * @param minSamples - Minimum samples needed (default 70)
   345→ * @returns AdaptiveModelResult or null
   346→ */
   347→export function trainAdaptiveModel(data: OHLCV[], symbol: string, minSamples = 70): AdaptiveModelResult | null {
   348→  const FORWARD_DAYS = 5;
   349→  const START_IDX = 40; // Need enough history for all indicators to stabilize
   350→
   351→  if (data.length < START_IDX + FORWARD_DAYS + minSamples) return null;
   352→
   353→  const hasVolume = data.some(d => d.volume > 0);
   354→
   355→  const X: number[][] = [];
   356→  const y: number[] = [];
   357→
   358→  for (let i = START_IDX; i < data.length - FORWARD_DAYS; i++) {
   359→    const features = extractVDSSFeatures(data, i, hasVolume);
   360→    const currentPrice = data[i].close;
   361→    const futurePrice = data[i + FORWARD_DAYS].close;
   362→
   363→    if (currentPrice <= 0) continue;
   364→    if (features.some(f => !isFinite(f))) continue;
   365→
   366→    // Label: 1 if 5-day forward return > 1%, else 0
   367→    const forwardReturn = (futurePrice - currentPrice) / currentPrice;
   368→    const label = forwardReturn > 0.01 ? 1 : 0;
   369→
   370→    X.push(features);
   371→    y.push(label);
   372→  }
   373→
   374→  if (X.length < minSamples) return null;
   375→
   376→  const model = new AdaptiveWeightModel(minSamples);
   377→  const success = model.train(X, y);
   378→
   379→  if (!success) return null;
   380→
   381→  // Predict current features
   382→  const currentFeatures = extractVDSSFeatures(data, data.length - 1, hasVolume);
   383→  model.predictScore(currentFeatures);
   384→
   385→  const result = model.getResult();
   386→  setCachedAdaptiveModel(symbol, result);
   387→
   388→  return result;
   389→}

   391→// ─── Calculate bullConsensus with ML Weights (Layer 3) ──────────────────────
   392→
   393→/**
   394→ * Calculate bullConsensus using ML weights.
   395→ * Falls back to default weights if ML model is not trained.
   396→ *
   397→ * @param features - 16 VDSS features for the current candle
   398→ * @param mlResult - Adaptive model result (or null for fallback)
   399→ * @param hasVolume - Whether MFI feature is valid
   400→ */
   401→export function calculateBullConsensus(
   402→  features: number[],
   403→  mlResult: AdaptiveModelResult | null,
   404→  hasVolume: boolean,
   405→): { bullConsensus: number; usedML: boolean } {
   406→  if (mlResult && mlResult.isTrained && mlResult.weights.length === 16) {
   407→    const weights = mlResult.weights;
   408→    let bullConsensus = 0;
   409→    let totalWeight = 0;
   410→
   411→    for (let j = 0; j < 16; j++) {
   412→      // Skip MFI (index 1) if no volume
   413→      if (j === 1 && !hasVolume) continue;
   414→      bullConsensus += weights[j] * features[j];
   415→      totalWeight += weights[j];
   416→    }
   417→
   418→    if (totalWeight > 0) bullConsensus /= totalWeight;
   419→
   420→    // Blend with ML direct prediction (dynamic weight based on accuracy)
   421→    if (mlResult.predictionProb !== null) {
   422→      const mlWeight = Math.max(0, Math.min(mlResult.recentAccuracy * 0.30, 0.30));
   423→      bullConsensus = bullConsensus * (1 - mlWeight) + mlResult.predictionProb * mlWeight;
   424→    }
   425→
   426→    return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: true };
   427→  }
   428→
   429→  // Fallback: default weights (per spec, only when ML is unavailable)
   430→  const DEFAULT_WEIGHTS = [
   431→    0.15, 0.12, 0.10, 0.08,  // rsi, mfi, cci, adx
   432→    0.14, 0.10, 0.08,         // macd, stoch, bb
   433→    0.10, 0.08, 0.10,         // ma21, ma100, ema
   434→    0.05, 0.10, 0.05,         // atr, trend, sr
   435→    0.06, 0.06, 0.08,         // stochCross, macdCross, div
   436→  ];
   437→
   438→  let bullConsensus = 0;
   439→  let totalWeight = 0;
   440→  for (let j = 0; j < 16; j++) {
   441→    if (j === 1 && !hasVolume) continue;
   442→    bullConsensus += DEFAULT_WEIGHTS[j] * features[j];
   443→    totalWeight += DEFAULT_WEIGHTS[j];
   444→  }
   445→  if (totalWeight > 0) bullConsensus /= totalWeight;
   446→
   447→  return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: false };
   448→}

   450→// ─── Scenario Probabilities with Adaptive Parameters (Layer 4) ─────────────
   451→
   452→/**
   453→ * Calculate 9-scenario probabilities with ML-adaptive parameters (V8).
   454→ * 3 Bullish (R1-R3) + 3 Bearish (R4-R6) + 2 Range (R7-R8) + 1 Shock (R9)
   455→ */
   456→export function calculateScenarioProbabilities(
   457→  bullConsensus: number,
   458→  price: number,
   459→  R1: number,
   460→  S1: number,
   461→  MA100: number,
   462→  rsi: number,
   463→  mfi: number,
   464→  stochK: number,
   465→  hasVolume: boolean,
   466→  mlResult: AdaptiveModelResult | null,
   467→): { pR1: number; pR2: number; pR3: number; pR4: number; pR5: number; pR6: number; pR7: number; pR8: number; pR9: number; factors: { momentum: number; volatility: number; trend: number } } {
   468→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0.5;
   469→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0.5;
   470→  const overboughtRisk = rsi > 70 ? (rsi - 70) / 30 : 0;
   471→  const oversoldBounce = rsi < 30 ? (30 - rsi) / 30 : 0;
   472→  const mfiMomentum = hasVolume
   473→    ? (mfi > 80 ? 1 : mfi < 20 ? 0 : 0.5)
   474→    : 0.5;
   475→  const stochMomentum = stochK > 80 ? 1 : stochK < 20 ? 0 : 0.5;
   476→  const momentum = (mfiMomentum + stochMomentum) / 2;
   477→
   478→  // Adaptive parameters from ML coefficients
   479→  const mlParams = mlResult?.adaptiveParams;
   480→  const factors = {
   481→    momentum: mlParams?.momentumFactor ?? 0.7,
   482→    volatility: mlParams?.volatilityFactor ?? 0.5,
   483→    trend: mlParams?.trendFactor ?? 0.6,
   484→  };
   485→
   486→  const bull = bullConsensus;
   487→  const bear = 1 - bullConsensus;
   488→
   489→  // ── Bullish Scenarios ──
   490→  // R1: Strong Bull Trend — high bull consensus, strong trend, near target
   491→  const raw_R1 = bull * bull * factors.trend * 1.2 * (0.7 + 0.3 * distR1);
   492→
   493→  // R2: Accelerating Bull — bull + high momentum, volatility expanding
   494→  const raw_R2 = bull * momentum * factors.momentum * 0.8 * (1 + factors.volatility * 0.3);
   495→
   496→  // R3: Cautious Bull — bull but divergence warning or weak momentum
   497→  const raw_R3 = bull * (1 - momentum * 0.5) * 0.6 * (1 + overboughtRisk * 0.4);
   498→
   499→  // ── Bearish Scenarios (mirror) ──
   500→  // R4: Strong Bear Trend
   501→  const raw_R4 = bear * bear * factors.trend * 1.2 * (0.7 + 0.3 * distS1);
   502→
   503→  // R5: Accelerating Bear
   504→  const raw_R5 = bear * (1 - momentum) * factors.momentum * 0.8 * (1 + factors.volatility * 0.3);
   505→
   506→  // R6: Cautious Bear
   507→  const raw_R6 = bear * momentum * 0.5 * 0.6 * (1 + oversoldBounce * 0.4);
   508→
   509→  // ── Range Scenarios ──
   510→  const neutralStrength = 1 - Math.abs(bull - 0.5) * 2; // 1 at bull=0.5, 0 at extremes
   511→
   512→  // R7: Low Volatility Range — neutral, BB squeeze, low ADX effect
   513→  const raw_R7 = neutralStrength * (1 - factors.volatility) * 0.7 * (1 + (price > MA100 ? 0.1 : -0.1));
   514→
   515→  // R8: High Volatility Range — neutral but high volatility/choppy
   516→  const raw_R8 = neutralStrength * factors.volatility * 0.5 * (1 + Math.abs(momentum - 0.5) * 0.4);
   517→
   518→  // ── Shock Scenario ──
   519→  // R9: Shock/Anomaly — always small, increases with extreme indicators
   520→   let raw_R9 = 0.03;
   521→  if (overboughtRisk > 0.7 || oversoldBounce > 0.7) raw_R9 += 0.05;
   522→  if (factors.volatility > 0.8) raw_R9 += 0.03;
   523→  if (factors.momentum > 0.85) raw_R9 += 0.02;
   524→
   525→  // Normalize to sum = 100
   526→  const rawSum = raw_R1 + raw_R2 + raw_R3 + raw_R4 + raw_R5 + raw_R6 + raw_R7 + raw_R8 + raw_R9;
   527→  if (rawSum <= 0) {
   528→    return {
   529→      pR1: 12, pR2: 10, pR3: 8, pR4: 12, pR5: 10, pR6: 8,
   530→      pR7: 15, pR8: 15, pR9: 10, factors,
   531→    };
   532→  }
   533→
   534→  const pR1 = Math.round(raw_R1 / rawSum * 100);
   535→  const pR2 = Math.round(raw_R2 / rawSum * 100);
   536→  const pR3 = Math.round(raw_R3 / rawSum * 100);
   537→  const pR4 = Math.round(raw_R4 / rawSum * 100);
   538→  const pR5 = Math.round(raw_R5 / rawSum * 100);
   539→  const pR6 = Math.round(raw_R6 / rawSum * 100);
   540→  const pR7 = Math.round(raw_R7 / rawSum * 100);
   541→  const pR8 = Math.round(raw_R8 / rawSum * 100);
   542→  const pR9 = 100 - (pR1 + pR2 + pR3 + pR4 + pR5 + pR6 + pR7 + pR8); // remainder
   543→
   544→  return { pR1, pR2, pR3, pR4, pR5, pR6, pR7, pR8, pR9: Math.max(1, pR9), factors };
   545→}

   547→// ─── Edge Weights with ML-Driven Parameters (Layer 5) ─────────────────────
   548→
   549→/**
   550→ * Calculate graph edge weights with ML-driven adaptive parameters.
   551→ */
   552→export function calculateEdgeWeights(
   553→  bullConsensus: number,
   554→  adx: number,
   555→  mlResult: AdaptiveModelResult | null,
   556→): { up: number; down: number; pullback: number; risk: number } {
   557→  // Default coefficients
   558→  let trendCoef = 1 / 3;
   559→  let momentumCoef = 1 / 3;
   560→  let volatilityCoef = 1 / 3;
   561→
   562→  if (mlResult?.isTrained && mlResult.coefficients.length === 16) {
   563→    const ec = mlResult.coefficients;
   564→    const absTrend = Math.abs(ec[11] ?? 0);
   565→    const absMomentum = Math.abs(ec[0] ?? 0);
   566→    const absVolatility = Math.abs(ec[10] ?? 0);
   567→    const total = absTrend + absMomentum + absVolatility;
   568→    if (total > 0) {
   569→      trendCoef = absTrend / total;
   570→      momentumCoef = absMomentum / total;
   571→      volatilityCoef = absVolatility / total;
   572→    }
   573→  }
   574→
   575→  // Dynamic base calculations
   576→  const up_base = bullConsensus * 0.7 + 0.15;
   577→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
   578→  const pullback_base = 0.25;
   579→  const risk_base = 0.12 * (1.2 - Math.min(adx / 100, 1));
   580→
   581→  // Apply ML-driven adjustments
   582→  const up = clamp(up_base * (1 + (momentumCoef - 1 / 3) * 0.3), 0.05, 0.95);
   583→  const down = clamp(down_base * (1 + (trendCoef - 1 / 3) * 0.3), 0.05, 0.95);
   584→  const pullback = clamp(pullback_base * (1 + (volatilityCoef - 1 / 3) * 0.2), 0.05, 0.50);
   585→  const risk = clamp(risk_base * (1 - (trendCoef - 1 / 3) * 0.5), 0.02, 0.30);
   586→
   587→  return { up, down, pullback, risk };
   588→}

   654→export interface PricePrediction {
   655→  predictedPrices: number[];
   656→  predictedReturn5d: number;
   657→  confidence: number;
   658→  trendDirection: 'up' | 'down' | 'flat';
   659→}
   660→
   661→export function predictPrices(data: OHLCV[]): PricePrediction | null {
   662→  if (data.length < 40) return null;
   663→  const PREDICT_DAYS = 5;
   664→  const closes = data.map(d => d.close);
   665→  const price = closes[closes.length - 1];
   666→  if (price <= 0) return null;
   667→
   668→  const LOOKBACK = 20;
   669→  const returns: number[] = [];
   670→  for (let i = closes.length - LOOKBACK; i < closes.length; i++) {
   671→    returns.push(i > 0 && closes[i - 1] > 0 ? (closes[i] - closes[i - 1]) / closes[i - 1] : 0);
   672→  }
   673→
   674→  const mom3 = returns.slice(-3).reduce((a, b) => a + b, 0) / 3;
   675→  const mom10 = returns.slice(-10).reduce((a, b) => a + b, 0) / 10;
   676→  const ma20 = sma(closes, 20);
   677→  const deviation = price > 0 && ma20 > 0 ? (price - ma20) / ma20 : 0;
   678→
   679→  const trendSlice = closes.slice(-20);
   680→  const tn = trendSlice.length;
   681→  let sx = 0, sy = 0, sxy = 0, sx2 = 0;
   682→  for (let i = 0; i < tn; i++) { sx += i; sy += trendSlice[i]; sxy += i * trendSlice[i]; sx2 += i * i; }
   683→  const d = tn * sx2 - sx * sx;
   684→  let slope = 0, r2 = 0;
   685→  if (d !== 0) {
   686→    slope = (tn * sxy - sx * sy) / d;
   687→    const meanY = sy / tn;
   688→    let ssRes = 0, ssTot = 0;
   689→    for (let i = 0; i < tn; i++) { ssRes += (trendSlice[i] - (slope * i + meanY - slope * (tn - 1) / 2)) ** 2; ssTot += (trendSlice[i] - meanY) ** 2; }
   690→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
   691→  }
   692→
   693→  let atr = 0;
   694→  for (let i = Math.max(1, data.length - 14); i < data.length; i++) {
   695→    atr += Math.max(data[i].high - data[i].low, Math.abs(data[i].high - data[i - 1].close), Math.abs(data[i].low - data[i - 1].close));
   696→  }
   697→  atr /= Math.min(14, data.length - 1);
   698→  const dailyVol = price > 0 ? atr / price : 0.02;
   699→
   700→  const trendDaily = price > 0 ? slope / price : 0;
   701→  const meanRevDaily = -deviation * 0.05;
   702→  const momentumDaily = mom3 * 0.4 + mom10 * 0.1;
   703→  const trendW = r2 * 0.6, mrW = Math.min(Math.abs(deviation) * 5, 0.4), momW = 0.2;
   704→  const predDaily = trendDaily * trendW + meanRevDaily * mrW + momentumDaily * momW;
   705→
   706→  const predictedPrices: number[] = [];
   707→  let cp = price;
   708→  for (let dd = 1; dd <= PREDICT_DAYS; dd++) {
   709→    const decay = Math.exp(-0.1 * (dd - 1));
   710→    cp = cp * (1 + predDaily * decay);
   711→    predictedPrices.push(Math.round(cp));
   712→  }
   713→
   714→  const predictedReturn5d = (predictedPrices[PREDICT_DAYS - 1] - price) / price;
   715→  return {
   716→    predictedPrices,
   717→    predictedReturn5d,
   718→    confidence: Math.min(0.9, Math.max(0.1, r2 * 0.7 + 0.2)),
   719→    trendDirection: predictedReturn5d > 0.005 ? 'up' : predictedReturn5d < -0.005 ? 'down' : 'flat',
   720→  };
   721→}

   723→export function evaluatePredictedIndicators(data: OHLCV[], prediction: PricePrediction): { mlScore: number; predictedBullConsensus: number } {
   724→  const price = data[data.length - 1].close;
   725→  if (price <= 0 || prediction.predictedPrices.length === 0) return { mlScore: 0.5, predictedBullConsensus: 0.5 };
   726→  const avgP = prediction.predictedPrices.reduce((a, b) => a + b, 0) / prediction.predictedPrices.length;
   727→  const closes = data.map(d => d.close);
   728→  const ma21 = sma(closes, 21), ma100 = sma(closes, 100), rsi = calcRSI(closes);
   729→  const rsiP = Math.max(0, Math.min(1, (rsi + (avgP > price ? 5 : -5)) / 100));
   730→  const ma21S = ma21 > 0 ? (avgP > ma21 ? 0.7 : 0.3) : 0.5;
   731→  const ma100S = ma100 > 0 ? (avgP > ma100 ? 0.65 : 0.35) : 0.5;
   732→  const dirS = avgP > price ? 0.65 : 0.35;
   733→  return {
   734→    mlScore: Math.max(0, Math.min(1, 0.5 + prediction.predictedReturn5d * 10 * prediction.confidence)),
   735→    predictedBullConsensus: Math.max(0, Math.min(1, rsiP * 0.3 + ma21S * 0.25 + ma100S * 0.2 + dirS * 0.25)),
   736→  };
   737→}
