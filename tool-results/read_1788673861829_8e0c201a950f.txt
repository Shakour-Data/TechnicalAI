    58→def compute_features(df: pd.DataFrame) -> pd.DataFrame:
    59→    """Compute 30+ technical features from OHLCV data.
    60→
    61→    Args:
    62→        df: DataFrame with columns [open, high, low, close, volume].
    63→
    64→    Returns:
    65→        DataFrame with all feature columns. Rows with NaN (insufficient
    66→        lookback) are NOT dropped here — caller decides.
    67→    """
    68→    df = df.copy()
    69→    c = df["close"].astype(float)
    70→    h = df["high"].astype(float)
    71→    l = df["low"].astype(float)
    72→    o = df["open"].astype(float)
    73→    v = df["volume"].astype(float)
    74→
    75→    # --- Returns (5 features) ---
    76→    for period in [1, 3, 5, 10, 20]:
    77→        df[f"ret_{period}"] = c.pct_change(period)
    78→
    79→    # --- SMA ratios (4 features) ---
    80→    for period in [5, 10, 20, 50]:
    81→        sma = c.rolling(window=period, min_periods=period).mean()
    82→        df[f"close_sma{period}"] = c / sma
    83→
    84→    # --- EMA ratios (2 features) ---
    85→    for period in [12, 26]:
    86→        ema = c.ewm(span=period, adjust=False).mean()
    87→        df[f"close_ema{period}"] = c / ema
    88→
    89→    # --- RSI (2 features) ---
    90→    for period in [7, 14]:
    91→        delta = c.diff()
    92→        gain = delta.clip(lower=0)
    93→        loss = (-delta).clip(lower=0)
    94→        avg_gain = gain.ewm(alpha=1.0 / period, min_periods=period, adjust=False).mean()
    95→        avg_loss = loss.ewm(alpha=1.0 / period, min_periods=period, adjust=False).mean()
    96→        rs = avg_gain / (avg_loss + 1e-10)
    97→        df[f"rsi_{period}"] = 100.0 - (100.0 / (1.0 + rs))
    98→
    99→    # --- MACD (3 features) ---
   100→    ema12 = c.ewm(span=12, adjust=False).mean()
   101→    ema26 = c.ewm(span=26, adjust=False).mean()
   102→    macd_line = ema12 - ema26
   103→    macd_signal = macd_line.ewm(span=9, adjust=False).mean()
   104→    df["macd"] = macd_line / (c + 1e-10)        # normalized
   105→    df["macd_signal"] = macd_signal / (c + 1e-10)
   106→    df["macd_hist"] = (macd_line - macd_signal) / (c + 1e-10)
   107→
   108→    # --- Stochastic K & D (2 features) ---
   109→    lowest_low = l.rolling(window=14, min_periods=14).min()
   110→    highest_high = h.rolling(window=14, min_periods=14).max()
   111→    stoch_k = 100.0 * (c - lowest_low) / (highest_high - lowest_low + 1e-10)
   112→    df["stoch_k"] = stoch_k
   113→    df["stoch_d"] = stoch_k.rolling(window=3, min_periods=3).mean()
   114→
   115→    # --- CCI(20) (1 feature) ---
   116→    tp = (h + l + c) / 3.0
   117→    sma_tp = tp.rolling(window=20, min_periods=20).mean()
   118→    mad = tp.rolling(window=20, min_periods=20).apply(
   119→        lambda x: np.mean(np.abs(x - x.mean())), raw=True
   120→    )
   121→    df["cci"] = (tp - sma_tp) / (0.015 * mad + 1e-10)
   122→
   123→    # --- Williams %R (1 feature) ---
   124→    hh14 = h.rolling(window=14, min_periods=14).max()
   125→    ll14 = l.rolling(window=14, min_periods=14).min()
   126→    df["williams_r"] = -100.0 * (hh14 - c) / (hh14 - ll14 + 1e-10)
   127→
   128→    # --- ADX, DI+, DI- (3 features) ---
   129→    df["adx"], df["di_plus"], df["di_minus"] = _compute_adx(h, l, c, 14)
   130→
   131→    # --- ATR normalized (1 feature) ---
   132→    tr = pd.concat([h - l, (h - c.shift(1)).abs(), (l - c.shift(1)).abs()], axis=1).max(axis=1)
   133→    atr = tr.rolling(window=14, min_periods=14).mean()
   134→    df["atr_norm"] = atr / (c + 1e-10)
   135→
   136→    # --- Bollinger Band position (1 feature) ---
   137→    sma20 = c.rolling(window=20, min_periods=20).mean()
   138→    std20 = c.rolling(window=20, min_periods=20).std()
   139→    bb_upper = sma20 + 2.0 * std20
   140→    bb_lower = sma20 - 2.0 * std20
   141→    df["bb_pos"] = (c - bb_lower) / (bb_upper - bb_lower + 1e-10)
   142→
   143→    # --- Volume features (2 features) ---
   144→    vol_sma20 = v.rolling(window=20, min_periods=20).mean()
   145→    df["vol_ratio"] = v / (vol_sma20 + 1e-10)
   146→
   147→    # OBV
   148→    obv = pd.Series(0.0, index=df.index, dtype=float)
   149→    for i in range(1, len(df)):
   150→        if c.iloc[i] > c.iloc[i - 1]:
   151→            obv.iloc[i] = obv.iloc[i - 1] + v.iloc[i]
   152→        elif c.iloc[i] < c.iloc[i - 1]:
   153→            obv.iloc[i] = obv.iloc[i - 1] - v.iloc[i]
   154→        else:
   155→            obv.iloc[i] = obv.iloc[i - 1]
   156→    df["obv_trend_5"] = obv.pct_change(5)
   157→
   158→    # --- Price structure features (5 features) ---
   159→    df["hl_range"] = (h - l) / (c + 1e-10)
   160→    df["oc_range"] = (o - c).abs() / (c + 1e-10)
   161→    df["upper_shadow"] = (h - pd.concat([o, c], axis=1).max(axis=1)) / (c + 1e-10)
   162→    df["lower_shadow"] = (pd.concat([o, c], axis=1).min(axis=1) - l) / (c + 1e-10)
   163→
   164→    # --- Momentum (1 feature) ---
   165→    df["momentum_10"] = c - c.shift(10)
   166→    df["momentum_10"] = df["momentum_10"] / (c + 1e-10)
   167→
   168→    # --- Rate of Change (1 feature) ---
   169→    df["roc_10"] = (c - c.shift(10)) / (c.shift(10) + 1e-10) * 100.0
   170→
   171→    # --- Price position in 20-session range (1 feature) ---
   172→    low20 = l.rolling(window=20, min_periods=20).min()
   173→    high20 = h.rolling(window=20, min_periods=20).max()
   174→    df["price_pos_20"] = (c - low20) / (high20 - low20 + 1e-10)
   175→
   176→    return df

   179→def _compute_adx(
   180→    high: pd.Series, low: pd.Series, close: pd.Series, period: int = 14
   181→) -> Tuple[pd.Series, pd.Series, pd.Series]:
   182→    """Compute ADX, DI+ and DI-."""
   183→    n = len(close)
   184→    adx = pd.Series(np.nan, index=close.index)
   185→    di_plus = pd.Series(np.nan, index=close.index)
   186→    di_minus = pd.Series(np.nan, index=close.index)
   187→
   188→    if n < period + 1:
   189→        return adx, di_plus, di_minus
   190→
   191→    plus_dm = pd.Series(0.0, index=close.index)
   192→    minus_dm = pd.Series(0.0, index=close.index)
   193→    tr = pd.Series(0.0, index=close.index)
   194→
   195→    for i in range(1, n):
   196→        up_move = high.iloc[i] - high.iloc[i - 1]
   197→        down_move = low.iloc[i - 1] - low.iloc[i]
   198→        plus_dm.iloc[i] = up_move if (up_move > down_move and up_move > 0) else 0.0
   199→        minus_dm.iloc[i] = down_move if (down_move > up_move and down_move > 0) else 0.0
   200→        tr_val = max(
   201→            high.iloc[i] - low.iloc[i],
   202→            abs(high.iloc[i] - close.iloc[i - 1]),
   203→            abs(low.iloc[i] - close.iloc[i - 1]),
   204→        )
   205→        tr.iloc[i] = tr_val
   206→
   207→    # Smooth using Wilder's method
   208→    atr_smooth = tr.copy()
   209→    smooth_plus = plus_dm.copy()
   210→    smooth_minus = minus_dm.copy()
   211→
   212→    for i in range(period, n):
   213→        if i == period:
   214→            atr_smooth.iloc[i] = tr.iloc[1 : i + 1].sum()
   215→            smooth_plus.iloc[i] = plus_dm.iloc[1 : i + 1].sum()
   216→            smooth_minus.iloc[i] = minus_dm.iloc[1 : i + 1].sum()
   217→        else:
   218→            atr_smooth.iloc[i] = (atr_smooth.iloc[i - 1] * (period - 1) + tr.iloc[i]) / period
   219→            smooth_plus.iloc[i] = (smooth_plus.iloc[i - 1] * (period - 1) + plus_dm.iloc[i]) / period
   220→            smooth_minus.iloc[i] = (smooth_minus.iloc[i - 1] * (period - 1) + minus_dm.iloc[i]) / period
   221→
   222→    di_plus = 100.0 * smooth_plus / (atr_smooth + 1e-10)
   223→    di_minus = 100.0 * smooth_minus / (atr_smooth + 1e-10)
   224→    dx = 100.0 * (di_plus - di_minus).abs() / (di_plus + di_minus + 1e-10)
   225→
   226→    adx = pd.Series(np.nan, index=close.index)
   227→    for i in range(2 * period - 1, n):
   228→        if i == 2 * period - 1:
   229→            adx.iloc[i] = dx.iloc[period : i + 1].mean()
   230→        else:
   231→            adx.iloc[i] = (adx.iloc[i - 1] * (period - 1) + dx.iloc[i]) / period
   232→
   233→    return adx, di_plus, di_minus

   278→def build_model(model_key: str) -> Any:
   279→    """Instantiate a model by its registry key."""
   280→    if model_key == "rf":
   281→        return RandomForestRegressor(
   282→            n_estimators=100, max_depth=10, min_samples_split=5,
   283→            min_samples_leaf=2, random_state=42, verbose=0, n_jobs=-1,
   284→        )
   285→    elif model_key == "xgboost":
   286→        return XGBRegressor(
   287→            n_estimators=100, max_depth=6, learning_rate=0.1,
   288→            subsample=0.8, colsample_bytree=0.8, random_state=42,
   289→            verbosity=0, n_jobs=-1,
   290→        )
   291→    elif model_key == "lightgbm":
   292→        if not HAS_LIGHTGBM:
   293→            return None
   294→        return lgb.LGBMRegressor(
   295→            n_estimators=100, max_depth=6, learning_rate=0.1,
   296→            subsample=0.8, colsample_bytree=0.8, random_state=42,
   297→            verbose=-1, n_jobs=-1,
   298→        )
   299→    elif model_key == "svr":
   300→        return SVR(kernel="rbf", C=100.0, gamma="scale", epsilon=0.001)
   301→    elif model_key == "gbr":
   302→        return GradientBoostingRegressor(
   303→            n_estimators=100, max_depth=5, learning_rate=0.1,
   304→            subsample=0.8, min_samples_split=5, random_state=42, verbose=0,
   305→        )
   306→    return None

   313→def prepare_dataset(
   314→    candles: List[Dict[str, Any]]
   315→) -> Tuple[pd.DataFrame, pd.DataFrame, pd.Series, int]:
   316→    """Convert raw candles to feature matrix and target vector.
   317→
   318→    Target = next-session percentage return: (close[t+1] - close[t]) / close[t] * 100
   319→
   320→    Returns:
   321→        (df_with_features, X, y, features_count)
   322→    """
   323→    df = pd.DataFrame(candles)
   324→    for col in ["open", "high", "low", "close", "volume"]:
   325→        df[col] = pd.to_numeric(df[col], errors="coerce")
   326→
   327→    df_feat = compute_features(df)
   328→
   329→    # Target: next-session return in percent
   330→    df_feat["target"] = df_feat["close"].pct_change(1).shift(-1) * 100.0
   331→
   332→    # Drop rows with NaN in features or target
   333→    valid_mask = df_feat[FEATURE_COLUMNS + ["target"]].notna().all(axis=1)
   334→    df_clean = df_feat[valid_mask].copy()
   335→
   336→    # Replace inf with NaN then drop
   337→    df_clean = df_clean.replace([np.inf, -np.inf], np.nan)
   338→    valid_mask2 = df_clean[FEATURE_COLUMNS + ["target"]].notna().all(axis=1)
   339→    df_clean = df_clean[valid_mask2].copy()
   340→
   341→    X = df_clean[FEATURE_COLUMNS]
   342→    y = df_clean["target"]
   343→
   344→    return df_clean, X, y, len(FEATURE_COLUMNS)

   347→def cross_validate_model(
   348→    model: Any,
   349→    X: pd.DataFrame,
   350→    y: pd.Series,
   351→    model_key: str,
   352→    n_splits: int = 5,
   353→) -> Tuple[float, float, float]:
   354→    """Run TimeSeriesSplit CV and return mean R², mean RMSE (in percent), and
   355→    std of CV predictions (for confidence intervals).
   356→
   357→    Returns:
   358→        (mean_r2, mean_rmse_pct, pred_std)
   359→    """
   360→    tscv = TimeSeriesSplit(n_splits=n_splits)
   361→    r2_scores: List[float] = []
   362→    rmse_scores: List[float] = []
   363→    all_test_preds: List[np.ndarray] = []
   364→
   365→    scaler: Optional[StandardScaler] = None
   366→    if model_key == "svr":
   367→        scaler = StandardScaler()
   368→
   369→    for train_idx, test_idx in tscv.split(X):
   370→        X_train, X_test = X.iloc[train_idx], X.iloc[test_idx]
   371→        y_train, y_test = y.iloc[train_idx], y.iloc[test_idx]
   372→
   373→        # Scale for SVR
   374→        if scaler is not None:
   375→            X_train_s = scaler.fit_transform(X_train)
   376→            X_test_s = scaler.transform(X_test)
   377→        else:
   378→            X_train_s = X_train
   379→            X_test_s = X_test
   380→
   381→        m = build_model(model_key)
   382→        if m is None:
   383→            return 0.0, 999.0, 0.0
   384→        m.fit(X_train_s, y_train)
   385→        preds = m.predict(X_test_s)
   386→
   387→        r2_scores.append(m.score(X_test_s, y_test))
   388→        rmse = np.sqrt(np.mean((preds - y_test.values) ** 2))
   389→        rmse_scores.append(rmse)
   390→        all_test_preds.append(preds)
   391→
   392→    mean_r2 = float(np.mean(r2_scores)) if r2_scores else 0.0
   393→    mean_rmse = float(np.mean(rmse_scores)) if rmse_scores else 0.0
   394→    pred_std = float(np.std(np.concatenate(all_test_preds))) if all_test_preds else 0.0
   395→
   396→    return mean_r2, mean_rmse, pred_std

   399→def train_final_model(
   400→    model_key: str,
   401→    X: pd.DataFrame,
   402→    y: pd.Series,
   403→) -> Tuple[Any, Optional[StandardScaler]]:
   404→    """Train the final model on all training data (last 80%).
   405→
   406→    Returns:
   407→        (trained_model, scaler_or_none)
   408→    """
   409→    split_idx = int(len(X) * 0.8)
   410→    X_train = X.iloc[:split_idx]
   411→    y_train = y.iloc[:split_idx]
   412→
   413→    scaler: Optional[StandardScaler] = None
   414→    if model_key == "svr":
   415→        scaler = StandardScaler()
   416→        X_train = scaler.fit_transform(X_train)
   417→
   418→    model = build_model(model_key)
   419→    if model is None:
   420→        return None, None
   421→    model.fit(X_train, y_train)
   422→    return model, scaler

   425→def iterative_predict(
   426→    model: Any,
   427→    model_key: str,
   428→    scaler: Optional[StandardScaler],
   429→    last_features: np.ndarray,
   430→    current_close: float,
   431→    df_all: pd.DataFrame,
   432→    n_sessions: int,
   433→) -> List[Dict[str, float]]:
   434→    """Iteratively predict N sessions ahead.
   435→
   436→    For each step:
   437→      1. Predict next-session return from current features
   438→      2. Compute new price
   439→      3. Append a synthetic candle to the history
   440→      4. Recompute features
   441→
   442→    Returns:
   443→        List of {session, price, change_pct} dicts
   444→    """
   445→    predictions: List[Dict[str, float]] = []
   446→    df_sim = df_all.copy()
   447→    feat = last_features.copy()
   448→    price = current_close
   449→
   450→    for step in range(1, n_sessions + 1):
   451→        # Predict
   452→        if scaler is not None:
   453→            feat_2d = feat.reshape(1, -1)
   454→            feat_scaled = scaler.transform(feat_2d)
   455→            pred_return = float(model.predict(feat_scaled)[0])
   456→        else:
   457→            pred_return = float(model.predict(feat.reshape(1, -1))[0])
   458→
   459→        # Compute new price
   460→        new_price = price * (1.0 + pred_return / 100.0)
   461→        change_pct = (new_price - current_close) / current_close * 100.0
   462→
   463→        predictions.append({
   464→            "session": step,
   465→            "price": round(new_price, 2),
   466→            "change_pct": round(change_pct, 2),
   467→        })
   468→
   469→        if step < n_sessions:
   470→            # Build synthetic candle: use predicted price for all OHLC
   471→            # Keep volume as average of last 5 sessions
   472→            recent_vol = df_sim["volume"].tail(5).mean()
   473→            new_candle = pd.DataFrame({
   474→                "open": [price],
   475→                "high": [max(price, new_price) * 1.001],
   476→                "low": [min(price, new_price) * 0.999],
   477→                "close": [new_price],
   478→                "volume": [recent_vol],
   479→            }, index=[df_sim.index[-1] + 1])
   480→
   481→            df_sim = pd.concat([df_sim, new_candle], ignore_index=True)
   482→            df_sim_feat = compute_features(df_sim)
   483→
   484→            # Get last row of features
   485→            last_row = df_sim_feat.iloc[-1]
   486→            feat_values = []
   487→            for col in FEATURE_COLUMNS:
   488→                val = last_row.get(col, np.nan)
   489→                if pd.isna(val) or np.isinf(val):
   490→                    val = 0.0
   491→                feat_values.append(float(val))
   492→            feat = np.array(feat_values, dtype=np.float64)
   493→
   494→            price = new_price
   495→
   496→    return predictions
   497→
   498→
   499→# ═══════════════════════════════════════════════════════════════════
   500→# Cache Helpers
   501→# ═══════════════════════════════════════════════════════════════════
   502→
   503→def _cache_key(candles: List[Dict], sessions: int, models: List[str]) -> str:
   504→    """Deterministic cache key from request params."""
   505→    raw = json.dumps({"c": candles, "s": sessions, "m": sorted(models)}, sort_keys=True)
   506→    return hashlib.sha256(raw.encode()).hexdigest()
   507→
   508→
   509→def _get_cache(key: str) -> Optional[Any]:
   510→    """Return cached data if still fresh, else None."""
   511→    if key in _cache:
   512→        data, ts = _cache[key]
   513→        if time.time() - ts < CACHE_TTL_SECONDS:
   514→            return data
   515→        del _cache[key]
   516→    return None
   517→
   518→
   519→def _set_cache(key: str, data: Any) -> None:
   520→    _cache[key] = (data, time.time())
   521→

   523→# ═══════════════════════════════════════════════════════════════════
   524→# Main Prediction Logic
   525→# ═══════════════════════════════════════════════════════════════════
   526→
   527→def run_prediction(
   528→    candles: List[Dict[str, Any]],
   529→    sessions: int,
   530→    model_keys: List[str],
   531→) -> Dict[str, Any]:
   532→    """Execute the full prediction pipeline."""
   533→
   534→    # Validate model keys
   535→    available = list(MODEL_REGISTRY.keys())
   536→    if HAS_LIGHTGBM is False:
   537→        available = [k for k in available if k != "lightgbm"]
   538→    for mk in model_keys:
   539→        if mk not in available:
   540→            raise ValueError(f"Unknown model: {mk}. Available: {available}")
   541→
   542→    # Prepare dataset
   543→    df_clean, X, y, feat_count = prepare_dataset(candles)
   544→    training_samples = len(X)
   545→
   546→    if training_samples < 30:
   547→        raise ValueError(
   548→            f"Not enough valid samples after feature computation. "
   549→            f"Need at least 30, got {training_samples}. "
   550→            f"Provide more candle data (minimum 60 recommended)."
   551→        )
   552→
   553→    # Check for flat prices
   554→    close_range = X["close_sma5"].max() - X["close_sma5"].min() if "close_sma5" in X.columns else 0
   555→    # Not a hard error but we should handle gracefully
   556→
   557→    # Get the raw OHLCV dataframe for iterative prediction
   558→    df_raw = pd.DataFrame(candles)
   559→    for col in ["open", "high", "low", "close", "volume"]:
   560→        df_raw[col] = pd.to_numeric(df_raw[col], errors="coerce")
   561→    df_raw = df_raw.dropna(subset=["close"])
   562→
   563→    # Last known close price
   564→    last_close = float(df_raw["close"].iloc[-1])
   565→
   566→    # Get features for the last valid row (the one just before the target)
   567→    last_feat_row = df_clean.iloc[-1]
   568→    last_features = np.array(
   569→        [float(last_feat_row[col]) if not pd.isna(last_feat_row[col]) else 0.0 for col in FEATURE_COLUMNS],
   570→        dtype=np.float64,
   571→    )
   572→
   573→    # Train, CV, and predict for each model
   574→    forecasts: Dict[str, Any] = {}
   575→    cv_results: Dict[str, float] = {}  # model_key -> r2
   576→    model_objects: Dict[str, Tuple[Any, Optional[StandardScaler]]] = {}
   577→
   578→    for mk in model_keys:
   579→        # Cross-validate
   580→        cv_r2, cv_rmse, pred_std = cross_validate_model(build_model(mk), X, y, mk)
   581→        cv_results[mk] = cv_r2
   582→
   583→        # Train final model on 80% of data
   584→        final_model, scaler = train_final_model(mk, X, y)
   585→        model_objects[mk] = (final_model, scaler)
   586→
   587→        # Iterative prediction
   588→        preds = iterative_predict(
   589→            final_model, mk, scaler, last_features, last_close, df_raw, sessions
   590→        )
   591→
   592→        # Add confidence intervals
   593→        z = 1.96
   594→        for p in preds:
   595→            session_num = p["session"]
   596→            # Widen intervals for longer horizons (sqrt scaling)
   597→            interval_width = pred_std * np.sqrt(session_num) * z
   598→            p["lower"] = round(p["price"] * (1 - interval_width / 100), 2)
   599→            p["upper"] = round(p["price"] * (1 + interval_width / 100), 2)
   600→
   601→        forecasts[mk] = {
   602→            "model_name": MODEL_REGISTRY[mk]["name"],
   603→            "cv_r2": round(cv_r2, 4),
   604→            "cv_rmse_pct": round(cv_rmse, 2),
   605→            "predictions": preds,
   606→        }
   607→
   608→    # ── Ensemble ─────────────────────────────────────────────────
   609→    # Weights based on CV R² (clamp negatives to small positive, normalize)
   610→    raw_weights: Dict[str, float] = {}
   611→    for mk in model_keys:
   612→        r2 = cv_results[mk]
   613→        raw_weights[mk] = max(r2, 0.01)  # floor at 0.01
   614→
   615→    total_w = sum(raw_weights.values())
   616→    weights = {mk: round(w / total_w, 4) for mk, w in raw_weights.items()}
   617→
   618→    ensemble_preds: List[Dict[str, Any]] = []
   619→    for step in range(1, sessions + 1):
   620→        w_price = 0.0
   621→        w_lower = 0.0
   622→        w_upper = 0.0
   623→        for mk in model_keys:
   624→            pred_entry = forecasts[mk]["predictions"][step - 1]
   625→            w_price += weights[mk] * pred_entry["price"]
   626→            w_lower += weights[mk] * pred_entry["lower"]
   627→            w_upper += weights[mk] * pred_entry["upper"]
   628→
   629→        change_pct = (w_price - last_close) / last_close * 100.0
   630→        ensemble_preds.append({
   631→            "session": step,
   632→            "price": round(w_price, 2),
   633→            "change_pct": round(change_pct, 2),
   634→            "lower": round(w_lower, 2),
   635→            "upper": round(w_upper, 2),
   636→        })
   637→
   638→    # ── Feature Importance from RandomForest ─────────────────────
   639→    feature_importance: Dict[str, float] = {}
   640→    if "rf" in model_objects:
   641→        rf_model, _ = model_objects["rf"]
   642→        if rf_model is not None and hasattr(rf_model, "feature_importances_"):
   643→            imp = rf_model.feature_importances_
   644→            for i, col in enumerate(FEATURE_COLUMNS):
   645→                feature_importance[col] = round(float(imp[i]), 4)
   646→    # Fallback: use first available tree model
   647→    if not feature_importance:
   648→        for mk in ["xgboost", "gbr", "lightgbm"]:
   649→            if mk in model_objects:
   650→                m, _ = model_objects[mk]
   651→                if m is not None and hasattr(m, "feature_importances_"):
   652→                    imp = m.feature_importances_
   653→                    for i, col in enumerate(FEATURE_COLUMNS):
   654→                        feature_importance[col] = round(float(imp[i]), 4)
   655→                    break
   656→
   657→    return {
   658→        "status": "ok",
   659→        "symbol": None,
   660→        "training_samples": training_samples,
   661→        "features_count": feat_count,
   662→        "forecasts": forecasts,
   663→        "ensemble": {
   664→            "model_name": "Ensemble (Weighted Average)",
   665→            "weights": weights,
   666→            "predictions": ensemble_preds,
   667→        },
   668→        "feature_importance": feature_importance,
   669→    }
   670→

   688→
   689→@app.route("/api/predict", methods=["POST"])
   690→def predict() -> Tuple[Any, int]:
   691→    """ML-based multi-session price prediction.
   692→
   693→    Request body:
   694→        candles: list of OHLCV dicts
   695→        sessions: int (1-30)
   696→        models: list of model keys
   697→    """
   698→    try:
   699→        body = request.get_json(force=True, silent=True)
   700→        if body is None:
   701→            return jsonify({"status": "error", "error": "Invalid JSON body"}), 400
   702→
   703→        candles: List[Dict] = body.get("candles", [])
   704→        sessions: int = int(body.get("sessions", 10))
   705→        model_keys: List[str] = body.get("models", list(MODEL_REGISTRY.keys()))
   706→
   707→        # Validate candles
   708→        if not candles or not isinstance(candles, list):
   709→            return jsonify({"status": "error", "error": "candles must be a non-empty list"}), 400
   710→
   711→        if len(candles) < MIN_CANDLES:
   712→            return jsonify({
   713→                "status": "error",
   714→                "error": f"Minimum {MIN_CANDLES} candles required, got {len(candles)}",
   715→            }), 400
   716→
   717→        # Validate each candle has required fields
   718→        required_fields = {"open", "high", "low", "close", "volume"}
   719→        for i, c in enumerate(candles):
   720→            if not required_fields.issubset(c.keys()):
   721→                missing = required_fields - set(c.keys())
   722→                return jsonify({
   723→                    "status": "error",
   724→                    "error": f"Candle at index {i} missing fields: {missing}",
   725→                }), 400
   726→
   727→        # Validate sessions
   728→        if sessions < 1 or sessions > 30:
   729→            return jsonify({
   730→                "status": "error",
   731→                "error": "sessions must be between 1 and 30",
   732→            }), 400
   733→
   734→        # Validate models
   735→        if not model_keys or not isinstance(model_keys, list):
   736→            model_keys = list(MODEL_REGISTRY.keys())
   737→
   738→        # Check cache
   739→        ck = _cache_key(candles, sessions, model_keys)
   740→        cached = _get_cache(ck)
   741→        if cached is not None:
   742→            return jsonify(cached), 200
   743→
   744→        # Run prediction
   745→        result = run_prediction(candles, sessions, model_keys)
   746→
   747→        # Cache result
   748→        _set_cache(ck, result)
   749→
   750→        return jsonify(result), 200
   751→
   752→    except ValueError as e:
   753→        return jsonify({"status": "error", "error": str(e)}), 400
   754→    except Exception as e:
   755→        traceback.print_exc()
   756→        return jsonify({"status": "error", "error": f"Internal error: {str(e)}"}), 500
