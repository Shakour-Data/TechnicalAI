    70→  R1: { label: 'تداوم صعود هیجانی', type: 'up', border: '#59e39b', badgeBg: 'rgba(89,227,155,0.2)', badgeColor: '#59e39b' },

    75→};
    76→

    78→  R1: { text: 'صعودی قوی — احتمال بالای عبور از مقاومت‌ها', tagCls: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30' },
    79→  R2: { text: 'صعود تدریجی — ورود در اصلاح توصیه می‌شود', tagCls: 'bg-cyan-500/15 text-cyan-400 border-cyan-500/30' },
    80→  R3: { text: 'بازار رنج — منتظر خروج از محدوده بمانید', tagCls: 'bg-amber-500/15 text-amber-400 border-amber-500/30' },
    81→  R4: { text: 'اصلاحی — احتیاط و کاهش حجم معاملات', tagCls: 'bg-orange-500/15 text-orange-400 border-orange-500/30' },

    83→};

    86→// Component
    87→// ═══════════════════════════════════════════════════════════════════════════════
    88→
    89→export default function VdesAnalysis(props: VdesAnalysisProps) {
    90→  const {
    91→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
    92→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
    93→    diPlus, diMinus, sar, atr, obv, bollingerUpper, bollingerMiddle, bollingerLower,
    94→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
    95→    supportStrengths, resistanceStrengths, priceTargets,
    96→  } = props;
    97→
    98→  const R1_level = resistances[0] ?? currentPrice * 1.05;
    99→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   100→  const S1_level = supports[0] ?? currentPrice * 0.95;
   101→  const S2_level = supports[1] ?? currentPrice * 0.90;
   102→  const S3_level = supports[2] ?? currentPrice * 0.85;
   103→
   104→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   105→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   106→
   107→  // ── Dominant scenario ──────────────────────────────────────────
   108→  let highestKey = 'R3';
   109→  let highestProb = 0;
   110→  for (const key of SCENARIO_KEYS) {
   111→    if (scenarios[key].probability > highestProb) {
   112→      highestProb = scenarios[key].probability;
   113→      highestKey = key;
   114→    }
   115→  }
   116→  const strategy = STRATEGY_MAP[highestKey];
   117→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   118→
   119→  // ── RSI signal ─────────────────────────────────────────────────
   120→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   121→  const rsiColor = rsi > 70 ? 'text-red-400' : rsi > 60 ? 'text-amber-400' : rsi > 40 ? 'text-gray-300' : rsi > 30 ? 'text-amber-400' : 'text-emerald-400';
   122→
   123→  // ── Stochastic signal ──────────────────────────────────────────
   124→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   125→
   126→  // ── MACD signal ────────────────────────────────────────────────
   127→  const macdSignalText = macdHist > 0 ? 'مومنتوم مثبت (هیستوگرام بالای صفر)' : 'مومنتوم منفی (هیستوگرام زیر صفر)';
   128→  const macdBullish = macdLine > macdSignal;
   129→
   130→  // ── Trend text ─────────────────────────────────────────────────
   131→  const trendText = trendDirection === 'up'
   132→    ? `صعودی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${(trendR2 * 100).toFixed(1)}%`
   133→    : trendDirection === 'down'
   134→    ? `نزولی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${(trendR2 * 100).toFixed(1)}%`
   135→    : 'خنثی و بدون جهت مشخص';
   136→
   137→  // ── DI signal ──────────────────────────────────────────────────
   138→  const diSignal = diPlus > diMinus
   139→    ? `DI+ (${toFa(diPlus)}) بالاتر از DI- (${toFa(diMinus)}) — فشار خرید غالب`
   140→    : `DI- (${toFa(diMinus)}) بالاتر از DI+ (${toFa(diPlus)}) — فشار فروش غالب`;
   141→
   142→  // ── SAR signal ─────────────────────────────────────────────────
   143→  const sarSignal = sar < currentPrice
   144→    ? `SAR (${toFa(sar)}) زیر قیمت — تأیید روند صعودی`
   145→    : `SAR (${toFa(sar)}) بالای قیمت — تأیید روند نزولی`;
   146→
   147→  // ── Bollinger Band position ────────────────────────────────────
   148→  const bbRange = bollingerUpper - bollingerLower;
   149→  const bbPos = bbRange > 0 ? ((currentPrice - bollingerLower) / bbRange * 100).toFixed(0) : '50';
   150→  const bbSignal = currentPrice > bollingerUpper ? 'بالای باند بالایی (اشباع خرید)'
   151→    : currentPrice < bollingerLower ? 'زیر باند پایینی (اشباع فروش)'
   152→    : `داخل باندها (${bbPos}٪ از بازه)`;
   153→
   154→  // ── TradingView scenario colors ────────────────────────────────
   155→  // Limit candles for chart rendering to prevent freeze on large datasets (e.g. derivatives with 10k+ candles)
   156→  const chartCandles = useMemo(() => {
   157→    if (candles.length <= 2000) return candles;
   158→    return candles.slice(-2000);
   159→  }, [candles]);
   160→
   161→  const tvScenarios = useMemo(() => {
   162→    const colors: Record<string, string> = {
   163→      R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62',
   164→    };
   165→    return Object.fromEntries(
   166→      SCENARIO_KEYS.map(k => [k, {
   167→        targetMin: scenarios[k].targetMin,
   168→        targetMax: scenarios[k].targetMax,
   169→        name: scenarios[k].name,
   170→        probability: scenarios[k].probability,
   171→        color: colors[k],
   172→      }])
   173→    );
   174→  }, [scenarios]);
   175→
   176→  // ── Strategy recommendation text ────────────────────────────────
   177→  const strategyText = rsi > 70 || mfi > 80
   178→    ? `با توجه به هشدار اشباع خرید (RSI: ${toFa(rsi)}, MFI: ${toFa(mfi)}) و فاصله قیمت تا مقاومت ${toFa(R1_level)}، استراتژی محتاطانه، انتظار برای اصلاح قیمت و ورود در محدوده حمایت ${toFa(S1_level)} تا ${toFa(S2_level)} ریال می‌باشد. در این محدوده می‌توان با حد ضرر ${toFa(S2_level)} ریال وارد موقعیت خرید شد.`
   179→    : rsi < 30 || mfi < 20
   180→    ? `با توجه به اشباع فروش (RSI: ${toFa(rsi)}, MFI: ${toFa(mfi)}) و نزدیکی به حمایت ${toFa(S1_level)}، فرصت خرید در محدوده فعلی با حد ضرر ${toFa(S2_level)} ریال قابل بررسی است. هدف اولیه ${toFa(R1_level)} و هدف ثانویه ${toFa(R2_level)} ریال تعیین می‌شود.`
   181→    : `با توجه به وضعیت خنثی اندیکاتورها (RSI: ${toFa(rsi)}, ADX: ${toFa(adx)}، قدرت روند: ${adx > 25 ? 'قوی' : 'ضعیف'})، انتظار برای خروج قیمت از محدوده ${toFa(S1_level)} تا ${toFa(R1_level)} ریال و سپس تصمیم‌گیری توصیه می‌شود. حجم معاملات و مومنتوم MACD را برای تأیید سیگنال پایش کنید.`;
   182→
   183→  return (
   184→    <div className="space-y-5" dir="rtl">
   185→      {/* ═══ HEADER ═══ */}
   186→      <div className="rounded-[32px] px-6 py-5 border border-amber-500/20"
   187→        style={{
   188→          background: 'rgba(18,28,46,0.7)',
   189→          backdropFilter: 'blur(14px)',
   190→          boxShadow: '0 25px 50px -12px rgba(0,0,0,0.6)',
   191→        }}>
   192→        <h1 className="text-xl font-bold mb-1" style={{ color: '#f8e365' }}>
   193→          📈 تحلیل تکنیکال {symbolName}
   194→        </h1>
   195→        <div className="flex flex-wrap gap-3 mt-2">
   196→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   197→            📍 قیمت مرجع: <b className="text-gray-200">{toFa(currentPrice)} ریال</b>
   198→          </span>
   199→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   200→            🎯 هدف کوتاه‌مدت: <b className="text-gray-200">{toFa(targetMin)} — {toFa(targetMax)} ریال</b>
   201→          </span>
   202→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   203→            📊 روند: <b className="text-gray-200">{trendText}</b>
   204→          </span>
   205→          <span className={`px-3.5 py-1 rounded-full border text-xs font-medium ${
   206→            rsi > 70 ? 'bg-red-500/15 text-red-400 border-red-500/30'
   207→            : rsi < 30 ? 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30'
   208→            : 'bg-amber-500/15 text-amber-400 border-amber-500/30'
   209→          }`}>
   210→            RSI: {toFa(rsi)} ({rsiSignal})
   211→          </span>
   212→        </div>
   213→      </div>
   214→
   215→      {/* ═══ TRADINGVIEW CHART ═══ */}
   216→      <div className="rounded-[28px] p-4 border border-white/6"
   217→        style={{
   218→          background: 'rgba(14,22,40,0.7)',
   219→          backdropFilter: 'blur(8px)',
   220→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   221→        }}>
   222→        <h2 className="text-sm font-semibold mb-3 text-gray-200 flex items-center gap-3">
   223→          <span style={{ color: '#f8e365' }}>📊</span>
   224→          نمودار قیمتی سهم (۱ روزه) — به همراه MA21، MA100، سطوح حمایت/مقاومت و اهداف قیمتی
   225→        </h2>
   226→        <TradingViewChart
   227→          symbolName={symbolName}
   228→          candles={chartCandles}
   229→          supports={supports}
   230→          resistances={resistances}
   231→          supportStrengths={supportStrengths}
   232→          resistanceStrengths={resistanceStrengths}
   233→          ma21={ma21}
   234→          ma100={ma100}
   235→          scenarios={tvScenarios}
   236→        />
   237→        <div className="mt-3 flex flex-wrap gap-4 text-[10px] text-gray-500">
   238→          <span>🔴 خطوط قرمز: مقاومت‌ها ({resistances.map(toFa).join(' ، ')})</span>
   239→          <span>🟢 خطوط سبز: حمایت‌ها ({supports.map(toFa).join(' ، ')})</span>
   240→          <span>🟣 خط بنفش: MA100 ({toFa(ma100)})</span>
   241→          <span>🔵 خط آبی: MA21 ({toFa(ma21)})</span>
   242→          <span>🟡 خطوط زرد: اهداف قیمتی سناریوها (رند شده)</span>
   243→        </div>
   244→      </div>
   245→
   246→      {/* ═══ DUAL ANALYSIS LAYOUT ═══ */}
   247→      <div className="rounded-[28px] p-5 border border-white/6"
   248→        style={{
   249→          background: 'rgba(14,22,40,0.7)',
   250→          backdropFilter: 'blur(8px)',
   251→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   252→        }}>
   253→        <h2 className="text-sm font-semibold mb-4 text-gray-200 flex items-center gap-3">
   254→          <span style={{ color: '#f8e365' }}>🧠</span>
   255→          تحلیل جامع روند و سطوح کلیدی
   256→        </h2>
   257→
   258→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-5">
   259→          {/* ── Text Analysis (Right in RTL) ── */}
   260→          <div className="rounded-2xl p-4 border border-white/4 space-y-4 flex flex-col"
   261→            style={{ background: 'rgba(8,20,35,0.5)' }}>
   262→
   263→            {/* ── PARAGRAPH 1: General Trend & Price Position ── */}
   264→            <p className="text-sm text-gray-300 leading-[1.85]">
   265→              <strong style={{ color: '#f8e365' }}>روند کلی و موقعیت قیمت:</strong>{' '}
   266→              سهم {symbolName} در حال حاضر در محدوده{' '}
   267→              <b className="text-gray-100">{toFa(currentPrice)} ریال</b> معامله می‌شود و روند میان‌مدت{' '}
   268→              <b className={trendDirection === 'up' ? 'text-emerald-400' : trendDirection === 'down' ? 'text-red-400' : 'text-amber-400'}>{trendDirection === 'up' ? 'صعودی' : trendDirection === 'down' ? 'نزولی' : 'خنثی'}</b>
   269→              {' '}است (زاویه {toFa(Math.abs(trendAngle))}°، R²={+(trendR2 * 100).toFixed(1)}%).
   270→              قیمت نسبت به MA21 ({toFa(ma21)} ریال){' '}
   271→              {currentPrice > ma21 ? <span className="text-emerald-400">بالاتر</span> : <span className="text-red-400">پایین‌تر</span>}
   272→              {' '}و نسبت به MA100 ({toFa(ma100)} ریال){' '}
   273→              {currentPrice > ma100 ? <span className="text-emerald-400">بالاتر</span> : <span className="text-red-400">پایین‌تر</span>}
   274→              {' '}قرار دارد. اندیکاتور Parabolic SAR ({toFa(sar)}) نیز{' '}
   275→              {sar < currentPrice ? <span>زیر قیمت قرار دارد که <b className="text-emerald-400">تأیید روند صعودی</b> است.</span> : <span>بالای قیمت قرار دارد که <b className="text-red-400">تأیید روند نزولی</b> است.</span>}
   276→              {' '}شاخص ADX ({toFa(adx)}) نشان‌دهنده {' '}
   277→              {adx > 40 ? <b className="text-emerald-400">روند قدرتمند</b> : adx > 25 ? <b className="text-amber-400">روند متوسط</b> : <b className="text-gray-500">روند ضعیف یا رنج</b>}
   278→              {' '}می‌باشد. {' '}{diSignal}.
   279→            </p>
   280→
   281→            {/* ── PARAGRAPH 2: Oscillator & Momentum Analysis ── */}
   282→            <p className="text-sm text-gray-300 leading-[1.85]">
   283→              <strong style={{ color: '#f8e365' }}>تحلیل اسیلاتورها و مومنتوم:</strong>{' '}
   284→              اندیکاتور RSI ({toFa(rsi)}) در ناحیه{' '}
   285→              <b className={rsiColor}>{rsiSignal}</b> قرار دارد.
   286→              اندیکاتور MFI ({toFa(mfi)}) نیز{' '}
   287→              {mfi > 80 ? <span className="text-red-400">اشباع خرید را تأیید می‌کند</span> : mfi < 20 ? <span className="text-emerald-400">اشباع فروش را نشان می‌دهد</span> : <span>در محدوده عادی است</span>}
   288→              . اندیکاتور CCI ({toFa(cci)}){' '}
   289→              {cci > 100 ? <span className="text-red-400">بالاتر از +100 (قدرت خریداری قوی)</span> : cci < -100 ? <span className="text-emerald-400">پایین‌تر از -100 (قدرت فروشندگان)</span> : <span>در محدوده عادی (-100 تا +100)</span>}
   290→              . استوکاستیک (%K={toFa(stochK)}، %D={toFa(stochD)}) وضعیت{' '}
   291→              <b className={stochK > 80 ? 'text-red-400' : stochK < 20 ? 'text-emerald-400' : 'text-gray-300'}>{stochSignal}</b> را نشان می‌دهد.
   292→              {' '}MACD (خط={toFa(macdLine)}، سیگنال={toFa(macdSignal)}) با {macdBullish ? <span className="text-emerald-400">عبور خط اصلی بالای خط سیگنال — سیگنال صعودی</span> : <span className="text-red-400">خط اصلی زیر خط سیگنال — سیگنال نزولی</span>}
   293→              . هیستوگرام MACD ({toFa(macdHist)}) {' '}
   294→              {macdHist > 0 ? <span className="text-emerald-400">مثبت</span> : <span className="text-red-400">منفی</span>}
   295→              {' '}و {' '}{macdSignalText}.
   296→            </p>
   297→
   298→            {/* ── PARAGRAPH 3: Bollinger Bands & Volatility ── */}
   299→            <p className="text-sm text-gray-300 leading-[1.85]">
   300→              <strong style={{ color: '#f8e365' }}>تحلیل نوسانات و باند بولینگر:</strong>{' '}
   301→              قیمت در باند بولینگر{' '}
   302→              <b className="text-gray-200">{bbSignal}</b>
   303→              {' '}قرار دارد. باند بالایی: {toFa(bollingerUpper)}، باند میانی (MA20): {toFa(bollingerMiddle)}، باند پایینی: {toFa(bollingerLower)} ریال.
   304→              {currentPrice > bollingerUpper
   305→                ? ' عبور از باند بالایی معمولاً نشان‌دهنده ادامه حرکت صعودی کوتاه‌مدت یا واکنش به باند است.'
   306→                : currentPrice < bollingerLower
   307→                ? ' نزدیکی یا عبور از باند پایینی می‌تواند نشانه بازگشت قیمت به سمت باند میانی باشد.'
   308→                : ' موقعیت قیمت در داخل باندها نشان‌دهنده عدم وجود سیگنال شدید از باند بولینگر است.'}
   309→            </p>

   477→function StrengthBadge({ strength, isTarget }: { strength: number; isTarget: boolean }) {

   482→    <span className={`inline-flex items-center gap-1 px-1.5 py-0.5 rounded-md text-[9px] font-bold border mr-1 ${color}`}>
   483→      {isTarget && '🎯 '}
   484→      قدرت: {strength}/10
   485→    </span>
   486→  );
