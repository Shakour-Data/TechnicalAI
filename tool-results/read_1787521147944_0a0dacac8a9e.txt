   230→export default function Home() {
   231→  const [data, setData] = useState<AnalysisData | null>(null);
   232→  const [loading, setLoading] = useState(false);
   233→  const [loadingMessage, setLoadingMessage] = useState<string | null>(null);
   234→  const [error, setError] = useState<string | null>(null);
   235→  const [activePanel, setActivePanel] = useState<SidebarItem>('visual');
   236→  const [refreshing, setRefreshing] = useState(false);
   237→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   238→  const [currentPage, setCurrentPage] = useState<PageId>('analysis');
   239→
   240→  // Store last fetch params for auto-refresh
   241→  const lastFetchRef = useRef<{ symbol: string; category?: string; insCode?: string; tgjuKey?: string; finpySector?: string; finpyIndex?: string; webId?: number } | null>(null);
   242→  const refreshTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   243→
   244→  // Chart container ref for image export
   245→  const chartWrapperRef = useRef<HTMLDivElement>(null);
   246→
   247→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number) => {
   248→    // TGJU instrument: fetch historical data via tgju.org chart API
   249→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   250→      setLoading(true);
   251→      setLoadingMessage('در حال دریافت داده‌های تاریخی ... (حدود ۱۵ ثانیه)');
   252→      setError(null);
   253→      setData(null);
   254→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId };
   255→      try {
   256→        const res = await fetch('/api/tgju-analysis?key=' + encodeURIComponent(tgjuKey));
   257→        if (!res.ok) {
   258→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   259→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   260→          throw new Error(errMsg);
   261→        }
   262→        const json = await res.json();
   263→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   264→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   265→        }
   266→        setData({ ...json, isTgju: true });
   267→        setActivePanel('visual');
   268→      } catch (err) {
   269→        setError(String(err));
   270→      } finally {
   271→        setLoading(false);
   272→        setLoadingMessage(null);
   273→      }
   274→      return;
   275→    }
   276→
   277→    // Index: fetch historical data via TSETMC CDN (webId), then finpy-tse (finpyIndex/finpySector)
   278→    if (category === 'index') {
   279→      setLoading(true);
   280→      setLoadingMessage('در حال دریافت داده‌های شاخص از TSETMC ... (حدود ۳۰ ثانیه برای بار اول)');
   281→      setError(null);
   282→      setData(null);
   283→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId };
   284→      let gotError = false;
   285→      try {
   286→        // 1. Try webId (TSETMC CDN)
   287→        if (webId) {
   288→          setLoadingMessage('در حال دریافت داده‌های شاخص از TSETMC ... (حدود ۳۰ ثانیه برای بار اول)');
   289→          const res = await fetch('/api/finpy-sector?webId=' + encodeURIComponent(String(webId)));
   290→          if (res.ok) {
   291→            try {
   292→              const json = await res.json();
   293→              if (json.candles && json.candles.length > 0 && json.ta) {
   294→                setData(json);
   295→                setActivePanel('visual');
   296→                setLoading(false);
   297→                setLoadingMessage(null);
   298→                return;
   299→              }
   300→              if (json.error) { setError(json.error); gotError = true; }
   301→            } catch (_) { /* invalid JSON */ }
   302→          } else { gotError = true; }
   303→        }
   304→
   305→        // 2. Main indices via finpy-tse
   306→        if (finpyIndex) {
   307→          setLoadingMessage('در حال دریافت داده‌های شاخص اصلی ...');
   308→          const res = await fetch('/api/finpy-sector?indexKey=' + encodeURIComponent(finpyIndex));
   309→          if (res.ok) {
   310→            try {
   311→              const json = await res.json();
   312→              if (json.candles && json.candles.length > 0 && json.ta) {
   313→                setData(json);
   314→                setActivePanel('visual');
   315→                setLoading(false);
   316→                setLoadingMessage(null);
   317→                return;
   318→              }
   319→              if (json.error) { setError(json.error); gotError = true; }
   320→            } catch (_) { /* invalid JSON */ }
   321→          } else { gotError = true; }
   322→        }
   323→
   324→        // 3. Sector indices via finpy-tse
   325→        if (finpySector) {
   326→          setLoadingMessage('در حال دریافت داده‌های شاخص گروه ...');
   327→          const res = await fetch('/api/finpy-sector?sector=' + encodeURIComponent(finpySector));
   328→          if (res.ok) {
   329→            try {
   330→              const json = await res.json();
   331→              if (json.candles && json.candles.length > 0 && json.ta) {
