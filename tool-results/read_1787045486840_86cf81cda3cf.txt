     1→   400→  else if (abs >= 1000) step = 10;
     2→   401→  else if (abs >= 100) step = 5;
     3→   402→  else step = 1;
     4→   403→  return Math.round(price / step) * step;
     5→   404→}

     7→   406→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
     8→   407→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: number[]; lows: number[] } {
     9→   408→  const highs: number[] = [];
    10→   409→  const lows: number[] = [];
    11→   410→  for (let i = lookback; i < data.length - lookback; i++) {
    12→   411→    let isHigh = true;
    13→   412→    let isLow = true;
    14→   413→    for (let j = i - lookback; j <= i + lookback; j++) {
    15→   414→      if (j === i) continue;
    16→   415→      if (data[j].high >= data[i].high) isHigh = false;
    17→   416→      if (data[j].low <= data[i].low) isLow = false;
    18→   417→    }
    19→   418→    if (isHigh) highs.push(data[i].high);
    20→   419→    if (isLow) lows.push(data[i].low);
    21→   420→  }
    22→   421→  return { highs, lows };
    23→   422→}

    25→   424→// ─── Support / Resistance (6 each, adaptive gap, multi-factor strength) ───
    26→   425→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
    27→   426→  resistances: number[];
    28→   427→  supports: number[];
    29→   428→  supportStrengths: LevelStrength[];
    30→   429→  resistanceStrengths: LevelStrength[];
    31→   430→  priceTargets: LevelStrength[];
    32→   431→} {
    33→   432→  const closes = data.map(d => d.close);
    34→   433→  const avgVolume = data.reduce((s, d) => s + d.volume, 0) / data.length;
    35→   434→
    36→   435→  // Tag each raw level with its source type for confluence scoring
    37→   436→  interface TaggedLevel { price: number; source: string };
    38→   437→  const allRaw: TaggedLevel[] = [];
    39→   438→
    40→   439→  // 1. Pivot Points (source: 'pivot')
    41→   440→  const prev = data[data.length - 2] ?? data[data.length - 1];
    42→   441→  if (prev) {
    43→   442→    const pp = (prev.high + prev.low + prev.close) / 3;
    44→   443→    const hl = prev.high - prev.low;
    45→   444→    const pivotR = [
    46→   445→      2 * pp - prev.low, pp + hl,
    47→   446→      prev.high + 2 * (pp - prev.low),
    48→   447→      prev.high + 2 * (pp - prev.low) + hl,
    49→   448→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
    50→   449→    ];
    51→   450→    const pivotS = [
    52→   451→      2 * pp - prev.high, pp - hl,
    53→   452→      prev.low - 2 * (prev.high - pp),
    54→   453→      prev.low - 2 * (prev.high - pp) - hl,
    55→   454→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
    56→   455→    ];
    57→   456→    for (const p of pivotR) allRaw.push({ price: p, source: 'pivot' });
    58→   457→    for (const p of pivotS) allRaw.push({ price: p, source: 'pivot' });
    59→   458→  }
    60→   459→
    61→   460→  // 2. Swing levels (source: 'swing')
    62→   461→  for (const lb of [3, 5, 7, 10]) {
    63→   462→    const swings = findSwingLevels(data, lb);
    64→   463→    for (const h of swings.highs) allRaw.push({ price: h, source: 'swing' });
    65→   464→    for (const l of swings.lows) allRaw.push({ price: l, source: 'swing' });
    66→   465→  }
    67→   466→
    68→   467→  // 3. Moving Average levels (source: 'ma')
    69→   468→  for (const period of [5, 10, 21, 50, 100, 200]) {
    70→   469→    const v = sma(closes, period);
    71→   470→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01)
    72→   471→      allRaw.push({ price: v, source: 'ma' });
    73→   472→  }
    74→   473→  const ema12 = emaCalc(closes, 12);
    75→   474→  const ema26 = emaCalc(closes, 26);
    76→   475→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01)
    77→   476→    allRaw.push({ price: ema12, source: 'ma' });
    78→   477→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01)
    79→   478→    allRaw.push({ price: ema26, source: 'ma' });
    80→   479→
    81→   480→  // 4. Bollinger Band levels (source: 'bb')
    82→   481→  const bbLevels = calcBollingerBands(closes);
    83→   482→  if (bbLevels.upper > 0) allRaw.push({ price: bbLevels.upper, source: 'bb' });
    84→   483→  if (bbLevels.lower > 0) allRaw.push({ price: bbLevels.lower, source: 'bb' });
    85→   484→
    86→   485→  // 5. Round number / psychological levels (source: 'round')
    87→   486→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
    88→   487→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
    89→   488→    const p = Math.round(currentPrice * mult / magnitude) * magnitude;
    90→   489→    allRaw.push({ price: p, source: 'round' });
    91→   490→  }
    92→   491→
    93→   492→  // 6. Recent high/low zones (source: 'hilo')
    94→   493→  for (const w of [5, 22, 66]) {
    95→   494→    const slice = data.slice(-Math.min(w, data.length));
    96→   495→    if (slice.length > 0) {
    97→   496→      allRaw.push({ price: Math.max(...slice.map(d => d.high)), source: 'hilo' });
    98→   497→      allRaw.push({ price: Math.min(...slice.map(d => d.low)), source: 'hilo' });
    99→   498→    }
   100→   499→  }
   101→   500→
   102→   501→  // 7. Round all and filter invalid
   103→   502→  const rounded: TaggedLevel[] = allRaw
   104→   503→    .map(l => ({ price: roundToNice(l.price), source: l.source }))
   105→   504→    .filter(l => l.price > 0);
   106→   505→
   107→   506→  // 8. Cluster nearby levels (within 0.5%) — merge sources, keep most common price
   108→   507→  interface Cluster {
   109→   508→    price: number;
   110→   509→    sources: Set<string>;
   111→   510→    touchCount: number;
   112→   511→    totalVolumeAtTouches: number;
   113→   512→    mostRecentTouchBar: number;
   114→   513→  }
   115→   514→  const clusters: Cluster[] = [];
   116→   515→  for (const lvl of rounded) {
   117→   516→    let merged = false;
   118→   517→    for (const c of clusters) {
   119→   518→      if (Math.abs(c.price - lvl.price) / Math.max(c.price, 1) < 0.005) {
   120→   519→        c.sources.add(lvl.source);
   121→   520→        merged = true;
   122→   521→        break;
   123→   522→      }
   124→   523→    }
   125→   524→    if (!merged) {
   126→   525→      clusters.push({ price: lvl.price, sources: new Set([lvl.source]), touchCount: 0, totalVolumeAtTouches: 0, mostRecentTouchBar: -1 });
   127→   526→    }
   128→   527→  }
   129→   528→
   130→   529→  // 9. Count actual price TOUCHES (rejections) and volume at each cluster level
   131→   530→  // A real touch = bar wick reached the level AND bar body closed away from it
   132→   531→  const touchThreshold = currentPrice * 0.004; // 0.4% tight proximity
   133→   532→  for (let i = 0; i < data.length; i++) {
   134→   533→    const bar = data[i];
   135→   534→    for (const c of clusters) {
   136→   535→      const isResistance = c.price > currentPrice;
   137→   536→      if (isResistance) {
   138→   537→        // Resistance: bar high reached near the level AND closed below it (rejection)
   139→   538→        if (bar.high >= c.price - touchThreshold && bar.close < c.price) {
   140→   539→          c.touchCount++;
   141→   540→          c.totalVolumeAtTouches += bar.volume;
   142→   541→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   143→   542→        }
   144→   543→      } else {
   145→   544→        // Support: bar low reached near the level AND closed above it (rejection)
   146→   545→        if (bar.low <= c.price + touchThreshold && bar.close > c.price) {
   147→   546→          c.touchCount++;
   148→   547→          c.totalVolumeAtTouches += bar.volume;
   149→   548→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   150→   549→        }
   151→   550→      }
   152→   551→    }
   153→   552→  }
   154→   553→
   155→   554→  // 10. Calculate multi-factor strength (0-10)
   156→   555→  const levelStrengths: LevelStrength[] = clusters.map(c => {
   157→   556→    // Factor 1: Source Confluence (0-2.5 pts)
   158→   557→    const sourceScore = Math.min(2.5, c.sources.size * 0.5);
   159→   558→    const hasSwingAndPivot = c.sources.has('swing') && c.sources.has('pivot') ? 0.5 : 0;
   160→   559→    const maConfluence = c.sources.has('ma') ? 0.5 : 0;
   161→   560→    const confluenceScore = Math.min(2.5, sourceScore + hasSwingAndPivot + maConfluence);
   162→   561→
   163→   562→    // Factor 2: Touch Count — CAPPED by recency (0-2 pts)
   164→   563→    // Only count touches in the most recent 50% of data to avoid ancient history inflating
   165→   564→    const recentBarStart = Math.floor(data.length * 0.5);
   166→   565→    const recentTouches = c.mostRecentTouchBar >= recentBarStart ? c.touchCount : Math.max(0, c.touchCount - 1);
   167→   566→    let touchScore: number;
   168→   567→    if (recentTouches === 0) touchScore = 0;
   169→   568→    else if (recentTouches === 1) touchScore = 0.4;
   170→   569→    else if (recentTouches <= 2) touchScore = 0.8;
   171→   570→    else if (recentTouches <= 4) touchScore = 1.2;
   172→   571→    else if (recentTouches <= 7) touchScore = 1.6;
   173→   572→    else touchScore = 2.0;
   174→   573→
   175→   574→    // Factor 3: Volume Confirmation (0-1.5 pts)
   176→   575→    let volumeScore = 0;
   177→   576→    if (c.touchCount > 0) {
   178→   577→      const avgVolAtTouch = c.totalVolumeAtTouches / c.touchCount;
   179→   578→      const volRatio = avgVolume > 0 ? avgVolAtTouch / avgVolume : 1;
   180→   579→      if (volRatio > 2.0) volumeScore = 1.5;
   181→   580→      else if (volRatio > 1.5) volumeScore = 1.2;
   182→   581→      else if (volRatio > 1.0) volumeScore = 0.8;
   183→   582→      else if (volRatio > 0.7) volumeScore = 0.4;
   184→   583→      else volumeScore = 0.2;
   185→   584→    }
   186→   585→
   187→   586→    // Factor 4: Freshness / Recency (0-1.5 pts)
   188→   587→    let freshnessScore = 0;
   189→   588→    if (c.touchCount > 0 && data.length > 0) {
   190→   589→      const barsAgo = data.length - 1 - c.mostRecentTouchBar;
   191→   590→      if (barsAgo <= 3) freshnessScore = 1.5;
   192→   591→      else if (barsAgo <= 10) freshnessScore = 1.2;
   193→   592→      else if (barsAgo <= 25) freshnessScore = 0.8;
   194→   593→      else if (barsAgo <= 50) freshnessScore = 0.4;
   195→   594→      else freshnessScore = 0.1;
   196→   595→    }
   197→   596→
   198→   597→    // Factor 5: Proximity to current price (0-1 pt)
   199→   598→    const distRatio = Math.abs(c.price - currentPrice) / currentPrice;
   200→   599→    let proximityScore = 0;
   201→   600→    if (distRatio < 0.015) proximityScore = 1.0;
   202→   601→    else if (distRatio < 0.03) proximityScore = 0.7;
   203→   602→    else if (distRatio < 0.06) proximityScore = 0.4;
   204→   603→    else if (distRatio < 0.10) proximityScore = 0.2;
   205→   604→
   206→   605→    const total = confluenceScore + touchScore + volumeScore + freshnessScore + proximityScore;
   207→   606→    const strength = Math.min(10, Math.max(1, Math.round(total)));
   208→   607→
   209→   608→    return { price: c.price, strength, isTarget: false };
   210→   609→  });
   211→   610→
   212→   611→  // 11. Separate above/below current price
   213→   612→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   214→   613→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   215→   614→
   216→   615→  // 12. Adaptive gap enforcement — keep highest strength in cluster
   217→   616→  function enforceGap(levels: LevelStrength[], targetCount: number): LevelStrength[] {
   218→   617→    const gapSteps = [0.05, 0.04, 0.03, 0.025, 0.02, 0.015, 0.01, 0.008, 0.005];
   219→   618→    for (const gapPct of gapSteps) {
   220→   619→      if (levels.length === 0) return [];
   221→   620→      const result: LevelStrength[] = [levels[0]];
   222→   621→      for (let i = 1; i < levels.length; i++) {
   223→   622→        if (result.length >= targetCount) break;
   224→   623→        const prevPrice = result[result.length - 1].price;
   225→   624→        const currPrice = levels[i].price;
   226→   625→        const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   227→   626→        if (gap >= gapPct) {
   228→   627→          result.push(levels[i]);
   229→   628→        } else {
   230→   629→          // Merge: keep stronger level's price, add 0.5 for confluence
   231→   630→          const keep = levels[i].strength >= result[result.length - 1].strength ? levels[i] : result[result.length - 1];
   232→   631→          const mergedStrength = Math.min(10, Math.max(keep.strength, result[result.length - 1].strength) + 0.5);
   233→   632→          result[result.length - 1] = { ...keep, strength: Math.round(mergedStrength) };
   234→   633→        }
   235→   634→      }
   236→   635→      if (result.length >= targetCount) return result;
   237→   636→    }
   238→   637→    return levels.slice(0, targetCount);
   239→   638→  }
   240→   639→
   241→   640→  const finalResistances = enforceGap(above, 6);
   242→   641→  const finalSupports = enforceGap(below, 6);
   243→   642→
   244→   643→  // 13. Determine price targets: top 2 by strength, or any above 7
   245→   644→  const allLevels = [...finalResistances, ...finalSupports];
   246→   645→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   247→   646→  let targets: LevelStrength[];
   248→   647→  if (sortedByStrength.some(l => l.strength >= 7)) {
   249→   648→    targets = sortedByStrength.filter(l => l.strength >= 7).slice(0, 3);
   250→   649→  } else {
   251→   650→    targets = sortedByStrength.slice(0, 2);
   252→   651→  }
   253→   652→
   254→   653→  // Mark targets
   255→   654→  targets = targets.map(t => ({ ...t, isTarget: true }));
   256→   655→
   257→   656→  // Also mark in final arrays
   258→   657→  finalResistances.forEach(r => {
   259→   658→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   260→   659→  });
   261→   660→  finalSupports.forEach(s => {
   262→   661→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   263→   662→  });
   264→   663→
   265→   664→  return {
   266→   665→    resistances: finalResistances.map(l => l.price),
   267→   666→    supports: finalSupports.map(l => l.price),
   268→   667→    supportStrengths: finalSupports,
   269→   668→    resistanceStrengths: finalResistances,
   270→   669→    priceTargets: targets,
   271→   670→  };
   272→   671→}

   291→   690→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
   292→   691→  const slice = data.slice(0, barIdx + 1);
   293→   692→  const price = slice[slice.length - 1].close;
   294→   693→  let nearestR = price * 1.05;
   295→   694→  let nearestS = price * 0.95;
   296→   695→
   297→   696→  // Collect all swing highs (resistance) and swing lows (support)
   298→   697→  const resistances: number[] = [];
   299→   698→  const supports: number[] = [];
   300→   699→  for (const lb of [3, 5, 7]) {
   301→   700→    const swings = findSwingLevels(slice, lb);
   302→   701→    resistances.push(...swings.highs);
   303→   702→    supports.push(...swings.lows);
   304→   703→  }
   305→   704→
   306→   705→  // Find nearest resistance above price
   307→   706→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
   308→   707→  if (abovePrice.length > 0) nearestR = abovePrice[0];
   309→   708→
   310→   709→  // Find nearest support below price
   311→   710→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
   312→   711→  if (belowPrice.length > 0) nearestS = belowPrice[0];
   313→   712→
   314→   713→  return { R1: nearestR, S1: nearestS };
   315→   714→}

   317→   716→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
   318→   717→// Returns true if bullish divergence detected, false if bearish, null if none
   319→   718→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
   320→   719→  const lookback = 20;
   321→   720→  const start = Math.max(0, barIdx - lookback);
   322→   721→  const slice = data.slice(start, barIdx + 1);
   323→   722→  if (slice.length < 10) return null;
   324→   723→
   325→   724→  const mid = Math.floor(slice.length / 2);
   326→   725→  const firstHalf = slice.slice(0, mid);
   327→   726→  const secondHalf = slice.slice(mid);
   328→   727→
   329→   728→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
   330→   729→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
   331→   730→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
   332→   731→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
   333→   732→
   334→   733→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
   335→   734→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
   336→   735→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
   337→   736→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
   338→   737→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
   339→   738→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
   340→   739→
   341→   740→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
   342→   741→    return false; // bearish divergence
   343→   742→  }
   344→   743→
   345→   744→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
   346→   745→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
   347→   746→
   348→   747→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
   349→   748→    return true; // bullish divergence
   350→   749→  }
   351→   750→
   352→   751→  return null;
   353→   752→}

   389→   788→  const ema26 = emaCalc(closes, 26);
   390→   789→  const trendResult = calcTrend(closes, 21);
   391→   790→  const sr = simplifiedSR(data, barIdx);
   392→   791→
   393→   792→  // ── Layer 1: Raw Scores (s_*) ────────────────────────────────────────────
   394→   793→
   395→   794→  // s_rsi — RSI mapped to 0-1
   396→   795→  const s_rsi = rsiVal > 80 ? 0.95
   397→   796→    : rsiVal > 70 ? 0.85
   398→   797→    : rsiVal > 60 ? 0.70
   399→   798→    : rsiVal > 50 ? 0.55
   400→   799→    : rsiVal > 40 ? 0.45
   401→   800→    : rsiVal > 30 ? 0.30
   402→   801→    : rsiVal > 20 ? 0.15
   403→   802→    : 0.05;
   404→   803→
   405→   804→  // s_mfi — MFI mapped to 0-1 (same formula as RSI)
   406→   805→  const s_mfi = mfiVal > 80 ? 0.95
   407→   806→    : mfiVal > 70 ? 0.85
   408→   807→    : mfiVal > 60 ? 0.70
   409→   808→    : mfiVal > 50 ? 0.55
   410→   809→    : mfiVal > 40 ? 0.45
   411→   810→    : mfiVal > 30 ? 0.30
   412→   811→    : mfiVal > 20 ? 0.15
   413→   812→    : 0.05;
   414→   813→
   415→   814→  // s_cci — CCI mapped to 0-1

   417→   816→    : cciVal > 100 ? 0.75
   418→   817→    : cciVal > 0 ? 0.60
   419→   818→    : cciVal > -100 ? 0.40
   420→   819→    : cciVal > -200 ? 0.25
   421→   820→    : 0.10;
   422→   821→
   423→   822→  // s_adx — Directional bias
   424→   823→  const s_adx = (adxResult.diPlus - adxResult.diMinus + 100) / 200;
   425→   824→
   426→   825→  // s_macd — MACD histogram + signal combined
   427→   826→  const absMacdLine = Math.abs(macd.line) + 1;
   428→   827→  const s_macd = macd.line > macd.signal
   429→   828→    ? 0.5 + Math.min(0.5, macd.histogram / absMacdLine)
   430→   829→    : 0.5 - Math.min(0.5, Math.abs(macd.histogram) / absMacdLine);
   431→   830→

   435→   834→  // s_bb — Bollinger Band position

   439→   838→    : 0.5;

   443→   842→    ? clamp(0.5 + (price - ma21) / ma21, 0, 1)
   444→   843→    : 0.5;
   445→   844→
   446→   845→  // s_ma100 — Price relative to 100-period SMA
   447→   846→  const s_ma100 = ma100 > eps
   448→   847→    ? clamp(0.5 + (price - ma100) / ma100, 0, 1)
   449→   848→    : 0.5;
   450→   849→
   451→   850→  // s_ema — EMA12 vs EMA26 spread
   452→   851→  const s_ema = ema26 > eps
   453→   852→    ? clamp(0.5 + (ema12 - ema26) / ema26 * 5, 0, 1)
   454→   853→    : 0.5;
   455→   854→
   456→   855→  // s_atr — Inverse volatility (lower ATR = higher score)
   457→   856→  const s_atr = price > eps ? 1 - Math.min(atrVal / price * 10, 1) : 0.5;
   458→   857→
   459→   858→  // s_trend — R² + angle combination
   460→   859→  const s_trend = (trendResult.r2 * 0.7) +
   461→   860→    (Math.max(0, Math.min(1, (trendResult.angle + 45) / 90)) * 0.3);
   462→   861→

   464→   863→  const dR1 = (sr.R1 - price) / (price > eps ? price : 1);
   465→   864→  const dS1 = (price - sr.S1) / (price > eps ? price : 1);
   466→   865→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
   467→   866→
   468→   867→  // ── Layer 2: Corrections (f_*) ───────────────────────────────────────────

   474→   873→    : rsiCurrent;

   476→   875→
   477→   876→  // f_mfi — MFI with momentum correction
   478→   877→  const mfiCurrent = mfiVal;
   479→   878→  const mfi5ago = barIdx >= 5
   480→   879→    ? calcMFI(data.slice(0, barIdx - 5 + 1))
   481→   880→    : mfiCurrent;
   482→   881→  const f_mfi = clamp(s_mfi + (mfiCurrent - mfi5ago) / 100 * 0.08, 0, 1);
   483→   882→
   484→   883→  // f_cci — CCI with momentum correction (scale: /200 instead of /100)
   485→   884→  const cciCurrent = cciVal;
   486→   885→  const cci5ago = barIdx >= 5
   487→   886→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
   488→   887→    : cciCurrent;
   489→   888→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
   490→   889→
   491→   890→  // f_macd — Same as s_macd (no additional correction)
   492→   891→  const f_macd = s_macd;
   493→   892→
   494→   893→  // f_stoch — Same as s_stoch
   495→   894→  const f_stoch = s_stoch;
   496→   895→
   497→   896→  // f_stochCross — Stochastic K/D crossover detection
   498→   897→  let f_stochCross = 0.1;
   499→   898→  if (barIdx >= 1) {
   500→   899→    const prevSlice = data.slice(0, barIdx);
   501→   900→    const prevStoch = calcStochastic(prevSlice);
   502→   901→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
   503→   902→      f_stochCross = 0.9; // K crossed above D (bullish)
   504→   903→    }
   505→   904→  }
   506→   905→
   507→   906→  // f_macdCross — MACD/Signal crossover detection
   508→   907→  let f_macdCross = 0.1;
   509→   908→  if (barIdx >= 1) {
   510→   909→    const prevCloses = closes.slice(0, -1);
   511→   910→    if (prevCloses.length >= 35) {
   512→   911→      const prevMacd = calcMACD(prevCloses);
   513→   912→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
   514→   913→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
   515→   914→      }
   516→   915→    }
   517→   916→  }
   518→   917→
   519→   918→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
   520→   919→  const divResult = detectDivergenceSimple(data, barIdx);
   521→   920→  const f_div = divResult === true ? 0.85 : 0.15;
   522→   921→
   523→   922→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
   524→   923→  return {
   525→   924→    rsi: f_rsi,
   526→   925→    mfi: f_mfi,
   527→   926→    cci: f_cci,
   528→   927→    adx: s_adx,
   529→   928→    macd: f_macd,
   530→   929→    stoch: f_stoch,
   531→   930→    bb: s_bb,
   532→   931→    ma21: s_ma21,
   533→   932→    ma100: s_ma100,
   534→   933→    ema: s_ema,
   535→   934→    atr: s_atr,
   536→   935→    trend: s_trend,
   537→   936→    sr: s_sr,
   538→   937→    stochCross: f_stochCross,
   539→   938→    macdCross: f_macdCross,
   540→   939→    div: f_div,
   541→   940→  };
   542→   941→}

   544→   943→// ═══════════════════════════════════════════════════════════════════════════════
   545→   944→// TRAINING DATA CREATION (Layer 0 — Data Preparation)
   546→   945→// ═══════════════════════════════════════════════════════════════════════════════
   547→   946→
   548→   947→/**
   549→   948→ * Creates supervised training data from historical OHLCV data.
   550→   949→ * For each bar, computes features and labels based on future price movement.
   551→   950→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
   552→   951→ * Samples every `sampleStep` bars to reduce computation.
   553→   952→ */
   554→   953→function createTrainingData(
   555→   954→  data: OHLCV[],
   556→   955→  horizon: number = 5,
   557→   956→  sampleStep: number = 5
   558→   957→): { X: number[][]; y: number[] } {
   559→   958→  const X: number[][] = [];
   560→   959→  const y: number[] = [];
   561→   960→
   562→   961→  // Start from horizon+30 to ensure enough data for indicators
   563→   962→  // End at data.length - horizon - 1 to have future bars for labeling
   564→   963→  // Sample every sampleStep bars for performance
   565→   964→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
   566→   965→    const features = computeFeaturesAtBar(data, i);
   567→   966→
   568→   967→    // Convert to array following FEATURES_ORDER
   569→   968→    const featureVec: number[] = [];
   570→   969→    let valid = true;
   571→   970→    for (const key of FEATURES_ORDER) {
   572→   971→      const val = features[key];
   573→   972→      if (val === undefined || val === null || Number.isNaN(val)) {
   574→   973→        valid = false;
   575→   974→        break;
   576→   975→      }
   577→   976→      featureVec.push(val);
   578→   977→    }
   579→   978→    if (!valid) continue;
   580→   979→
   581→   980→    // Label: 1 if future close > 1% above current close
   582→   981→    const currentClose = data[i].close;
   583→   982→    const futureClose = data[i + horizon].close;
   584→   983→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
   585→   984→
   586→   985→    X.push(featureVec);
   587→   986→    y.push(label);
   588→   987→  }
   589→   988→
   590→   989→  return { X, y };
   591→   990→}

   593→   992→// ═══════════════════════════════════════════════════════════════════════════════
   594→   993→// VDss GRAPH STRUCTURE (Layer 5 — State Machine)
   595→   994→// ═══════════════════════════════════════════════════════════════════════════════
   596→   995→
   597→   996→/**
   598→   997→ * VDss Graph: Each node represents a market state.
   599→   998→ * Edges represent transitions with types: 'up', 'down', 'pullback', 'risk', 'terminal'.
   600→   999→ * Terminal nodes map to scenario results (R1=strong bullish, R5=deep correction).
   601→  1000→ */
   602→  1001→const VDSS_GRAPH = {
   603→  1002→  // outgoing edges from each node: [target, type]
   604→  1003→  outgoing: {
   605→  1004→    A: [['B', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
   606→  1005→    B: [['C', 'up'], ['G', 'pullback'], ['A', 'risk']] as [string, string][],
   607→  1006→    C: [['D', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
   608→  1007→    D: [['E', 'up'], ['B', 'pullback'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
   609→  1008→    E: [['F', 'up'], ['D', 'pullback'], ['G', 'pullback'], ['H', 'down'], ['R1', 'terminal']] as [string, string][],
   610→  1009→    F: [['R1', 'terminal'], ['E', 'pullback'], ['D', 'down'], ['G', 'risk']] as [string, string][],
   611→  1010→    G: [['B', 'up'], ['A', 'pullback'], ['H', 'down'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
   612→  1011→    H: [['G', 'pullback'], ['I', 'down'], ['B', 'up'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
   613→  1012→    I: [['H', 'pullback'], ['J', 'down'], ['G', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
   614→  1013→    J: [['I', 'pullback'], ['K', 'down'], ['H', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
   615→  1014→    K: [['J', 'pullback'], ['I', 'up'], ['L', 'risk'], ['R4', 'terminal'], ['R5', 'terminal']] as [string, string][],
   616→  1015→    L: [['C', 'up'], ['H', 'pullback'], ['J', 'down'], ['K', 'risk'], ['R5', 'terminal']] as [string, string][],
   617→  1016→  },
   618→  1017→  // which result scenario each terminal node maps to
   619→  1018→  terminalMap: { R1: 'R1', R2: 'R2', R3: 'R3', R4: 'R4', R5: 'R5' } as Record<string, string>,
   620→  1019→  startNode: 'A' as string,
   621→  1020→};
   622→  1021→
   623→  1022→// ═══════════════════════════════════════════════════════════════════════════════
   624→  1023→// LAYER 5 — Edge Weight Calculation
   625→  1024→// ═══════════════════════════════════════════════════════════════════════════════
   626→  1025→
   627→  1026→/**
   628→  1027→ * Calculates transition edge weights for the VDss graph.
   629→  1028→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
   630→  1029→ * to determine the relative probability of up/down/pullback/risk transitions.
   631→  1030→ */
   632→  1031→function calculateEdgeWeights(
   633→  1032→  bullConsensus: number,
   634→  1033→  adx: number,
   635→  1034→  mlModel: AdaptiveWeightModel
   636→  1035→): { up: number; down: number; pullback: number; risk: number } {
   637→  1036→  // Extract ML coefficient insights if model is trained
   638→  1037→  let trend_coef = 1 / 3;
   639→  1038→  let momentum_coef = 1 / 3;
   640→  1039→  let volatility_coef = 1 / 3;
   641→  1040→
   642→  1041→  if (mlModel.isTrained && mlModel.weights) {
   643→  1042→    const coefs = mlModel.getCoefficients();
   644→  1043→    // Map feature indices to coefficient categories
   645→  1044→    // FEATURES_ORDER: ['rsi','mfi','cci','adx','macd','stoch','bb','ma21','ma100','ema','atr','trend','sr','stochCross','macdCross','div']
   646→  1045→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER
   647→  1046→    const rsiIdx = 0;     // 'rsi' in FEATURES_ORDER
   648→  1047→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
   649→  1048→
   650→  1049→    const absTrend = Math.abs(coefs[trendIdx] ?? 0);
   651→  1050→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
   652→  1051→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
   653→  1052→
   654→  1053→    const coefSum = absTrend + absRsi + absAtr;
   655→  1054→    if (coefSum > 1e-10) {
   656→  1055→      trend_coef = absTrend / coefSum;
   657→  1056→      momentum_coef = absRsi / coefSum;
   658→  1057→      volatility_coef = absAtr / coefSum;
   659→  1058→    }
   660→  1059→  }
   661→  1060→
   662→  1061→  // Base weights driven by bull consensus
   663→  1062→  const up_base = bullConsensus * 0.7 + 0.15;
   664→  1063→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
   665→  1064→  const pullback_base = 0.25;
   666→  1065→  const risk_base = 0.12 * (1.2 - adx / 100);
   667→  1066→
   668→  1067→  // Apply ML coefficient adjustments
   669→  1068→  const up = clamp(
   670→  1069→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
   671→  1070→    0.05, 0.95
   672→  1071→  );
   673→  1072→  const down = clamp(
   674→  1073→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
   675→  1074→    0.05, 0.95
   676→  1075→  );
   677→  1076→  const pullback = clamp(
   678→  1077→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
   679→  1078→    0.05, 0.50
   680→  1079→  );
   681→  1080→  const risk = clamp(
   682→  1081→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
   683→  1082→    0.02, 0.30
   684→  1083→  );
   685→  1084→
   686→  1085→  return { up, down, pullback, risk };
   687→  1086→}
   688→  1087→
   689→  1088→// ═══════════════════════════════════════════════════════════════════════════════
   690→  1089→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)
   691→  1090→// ═══════════════════════════════════════════════════════════════════════════════
   692→  1091→
   693→  1092→/**
   694→  1093→ * Performs iterative DFS from the start node through the VDss graph,
   695→  1094→ * computing raw path probabilities, then calibrating them against
   696→  1095→ * Layer 4 scenario probabilities.
   697→  1096→ *
   698→  1097→ * Returns:
   699→  1098→ * - calibratedPaths: Map<pathString, probability>
   700→  1099→ * - calibrationFactors: per-scenario multiplier used
   701→  1100→ * - scenarioSums: raw probability sums per scenario before calibration
   702→  1101→ */
   703→  1102→function calculatePathProbabilities(
   704→  1103→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
   705→  1104→  scenarioProbs: Record<string, number>
   706→  1105→): {
   707→  1106→  calibratedPaths: Map<string, number>;
   708→  1107→  calibrationFactors: Record<string, number>;
   709→  1108→  scenarioSums: Record<string, number>;
   710→  1109→} {
   711→  1110→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
   712→  1111→  const MAX_PATHS = 200;
   713→  1112→
   714→  1113→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
   715→  1114→  type StackEntry = [string, string, number];
   716→  1115→
   717→  1116→  // Collect all terminal paths with raw probabilities
   718→  1117→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
   719→  1118→
   720→  1119→  // Iterative DFS
   721→  1120→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
   722→  1121→
   723→  1122→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
   724→  1123→    const [node, pathStr, pathProb] = stack.pop()!;
   725→  1124→
   726→  1125→    const edges = outgoing[node as keyof typeof outgoing];
   727→  1126→    if (!edges) continue;
   728→  1127→
   729→  1128→    // Normalize outgoing edge weights to probabilities for this node
   730→  1129→    // Map edge types to their weights
   731→  1130→    let totalWeight = 0;
   732→  1131→    const edgeWeightsList: number[] = [];
   733→  1132→    for (const [, edgeType] of edges) {
   734→  1133→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
   735→  1134→      edgeWeightsList.push(w);
   736→  1135→      totalWeight += w;
   737→  1136→    }
   738→  1137→
   739→  1138→    if (totalWeight < 1e-10) continue;
   740→  1139→
   741→  1140→    // Expand edges
   742→  1141→    for (let i = 0; i < edges.length; i++) {
   743→  1142→      const [target, edgeType] = edges[i];
   744→  1143→      const edgeProb = edgeWeightsList[i] / totalWeight;
   745→  1144→      const newProb = pathProb * edgeProb;
   746→  1145→      const newPath = pathStr + '→' + target;
   747→  1146→
   748→  1147→      if (edgeType === 'terminal') {
   749→  1148→        // Reached a terminal node
   750→  1149→        const scenario = terminalMap[target] ?? 'R3';
   751→  1150→        rawPaths.push({ path: newPath, scenario, prob: newProb });
   752→  1151→      } else {
   753→  1152→        // Continue DFS
   754→  1153→        stack.push([target, newPath, newProb]);
   755→  1154→      }
   756→  1155→    }
   757→  1156→  }
   758→  1157→
   759→  1158→  // Group raw probabilities by scenario
   760→  1159→  const rawSums: Record<string, number> = { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0 };
   761→  1160→  for (const rp of rawPaths) {
   762→  1161→    rawSums[rp.scenario] = (rawSums[rp.scenario] ?? 0) + rp.prob;
   763→  1162→  }
   764→  1163→
   765→  1164→  // Compute calibration factors: scenarioProbs[scenario] / rawSum[scenario]
   766→  1165→  const calibrationFactors: Record<string, number> = {};
   767→  1166→  for (const scenario of ['R1', 'R2', 'R3', 'R4', 'R5']) {
   768→  1167→    const rawSum = rawSums[scenario] ?? 0;
   769→  1168→    const targetProb = (scenarioProbs[scenario] ?? 20) / 100; // convert from percentage
   770→  1169→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
   771→  1170→  }
   772→  1171→
   773→  1172→  // Apply calibration to each path
   774→  1173→  const calibratedPaths = new Map<string, number>();
   775→  1174→  for (const rp of rawPaths) {
   776→  1175→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
   777→  1176→    calibratedPaths.set(rp.path, rp.prob * factor);
   778→  1177→  }
   779→  1178→
   780→  1179→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
   781→  1180→}
   782→  1181→
   783→  1182→// ═══════════════════════════════════════════════════════════════════════════════
   784→  1183→// LAYER 4 — Scenario Probability Calculation (Adaptive with ML)
   785→  1184→// ═══════════════════════════════════════════════════════════════════════════════
   786→  1185→
   787→  1186→/**
   788→  1187→ * Calculates the 5 VDss scenario probabilities (R1-R5) that sum to 100.
   789→  1188→ * Uses bull/bear consensus, key indicator values, S/R levels, and
   790→  1189→ * ML-derived adaptive factors for fine-tuning.
   791→  1190→ */
   792→  1191→function calculateScenarioProbabilities(
   793→  1192→  bullConsensus: number,
   794→  1193→  price: number,
   795→  1194→  R1: number,
   796→  1195→  S1: number,
   797→  1196→  MA100: number,
   798→  1197→  rsi: number,
   799→  1198→  mfi: number,
   800→  1199→  stochK: number,
   801→  1200→  mlModel: AdaptiveWeightModel
   802→  1201→): { R1: number; R2: number; R3: number; R4: number; R5: number; factors: { momentum: number; volatility: number; trend: number } } {
   803→  1202→  // Distance metrics to key levels (spec uses exp(-3 * ...))
   804→  1203→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0;
   805→  1204→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0;
   806→  1205→
   807→  1206→  // Binary risk flags (per spec)
   808→  1207→  const overboughtRisk = (rsi > 70 || mfi > 80 || stochK > 80) ? 1 : 0;
   809→  1208→  const oversoldBounce = (rsi < 30 || mfi < 20 || stochK < 20) ? 1 : 0;
   810→  1209→
   811→  1210→  // Extract adaptive factors from ML model if trained
   812→  1211→  let momentum = 0.7;
   813→  1212→  let volatility = 0.5;
   814→  1213→  let trend = 0.6;
   815→  1214→
   816→  1215→  if (mlModel.isTrained && mlModel.weights) {
   817→  1216→    const coefs = mlModel.getCoefficients();
   818→  1217→    momentum = sigmoid(coefs[0] ?? 0);   // rsi coefficient → momentum factor
   819→  1218→    volatility = sigmoid(coefs[10] ?? 0); // atr coefficient → volatility factor
   820→  1219→    trend = sigmoid(coefs[11] ?? 0);     // trend coefficient → trend factor
   821→  1220→  }
   822→  1221→
   823→  1222→  // Below MA100 flag
   824→  1223→  const belowMA100 = price < MA100 ? 1 : 0;
   825→  1224→
   826→  1225→  // ── Raw scenario probabilities (matching user spec) ─────────────────────
   827→  1226→
   828→  1227→  // R1: Strong Bullish — aggressive breakout
   829→  1228→  const raw_R1 =
   830→  1229→    bullConsensus * distR1 * (1 - overboughtRisk * momentum * 0.7) +
   831→  1230→    oversoldBounce * 0.3 * distR1 * momentum;
   832→  1231→
   833→  1232→  // R2: Gradual Uptrend — steady rise
   834→  1233→  const raw_R2 =
   835→  1234→    ((1 - Math.abs(bullConsensus - 0.65)) * 0.8 +
   836→  1235→    (bullConsensus > 0.4 ? 0.2 : 0)) * trend;
   837→  1236→
   838→  1237→  // R3: Range-bound
   839→  1238→  const raw_R3 =
   840→  1239→    (1 - Math.abs(bullConsensus - 0.5)) * 0.6 * (1 + (1 - volatility) * 0.2);
   841→  1240→
   842→  1241→  // R4: Moderate Correction
   843→  1242→  const raw_R4 =
   844→  1243→    ((1 - bullConsensus) * distS1 * 0.7 +
   845→  1244→    overboughtRisk * 0.5 * momentum) *
   846→  1245→    (0.8 + 0.2 * (1 - trend));
   847→  1246→
   848→  1247→  // R5: Deep Correction
   849→  1248→  const raw_R5 =
   850→  1249→    ((1 - bullConsensus) ** 2 * 0.5 + belowMA100 * 0.2) *
   851→  1250→    (1 + (1 - trend) * 0.3);
   852→  1251→
   853→  1252→  // ── Normalize to sum = 100 ───────────────────────────────────────────
   854→  1253→  const sum = raw_R1 + raw_R2 + raw_R3 + raw_R4 + raw_R5;
   855→  1254→  const pR1 = sum > 0 ? Math.round(raw_R1 / sum * 100) : 20;
   856→  1255→  const pR2 = sum > 0 ? Math.round(raw_R2 / sum * 100) : 20;
   857→  1256→  const pR3 = sum > 0 ? Math.round(raw_R3 / sum * 100) : 20;
   858→  1257→  const pR4 = sum > 0 ? Math.round(raw_R4 / sum * 100) : 20;
   859→  1258→  const pR5 = 100 - (pR1 + pR2 + pR3 + pR4);
   860→  1259→
   861→  1260→  return {
   862→  1261→    R1: pR1, R2: pR2, R3: pR3, R4: pR4, R5: pR5,
   863→  1262→    factors: { momentum, volatility, trend },
   864→  1263→  };
   865→  1264→}
