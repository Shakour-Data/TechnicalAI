   137→// Component
   138→// ═══════════════════════════════════════════════════════════════════════════════
   139→
   140→export default function VdssGraph(props: VdssGraphProps) {
   141→  const { colors: C } = useTheme();
   142→  const {
   143→    symbolName,
   144→    currentPrice,
   145→    scenarios,
   146→    decisionGraph,
   147→  } = props;
   148→
   149→  const [activeFilter, setActiveFilter] = useState('all');
   150→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   151→  const graphRef = useRef<HTMLDivElement>(null);
   152→  const svgRef = useRef<SVGSVGElement>(null);
   153→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   154→  const shellRef = useRef<HTMLDivElement>(null);
   155→
   156→  // ── Derived data from decisionGraph ──────────────────────────────
   157→  const edges = decisionGraph?.edges ?? [];
   158→  const nodes = decisionGraph?.nodes ?? [];
   159→  const nodePositions = decisionGraph?.nodePositions ?? {};
   160→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   161→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   162→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   163→  const pathContributions = decisionGraph?.pathContributions ?? {};
   164→
   165→  // Build node lookup map
   166→  const nodeMap = useMemo(() => {
   167→    const map: Record<string, (typeof nodes)[number]> = {};
   168→    for (const n of nodes) map[n.id] = n;
   169→    return map;
   170→  }, [nodes]);
   171→
   172→  // ── Visible edges/nodes based on filter ──────────────────────────
   173→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   174→    if (activeFilter === 'all') {
   175→      const allEdgeIdx = edges.map((_, i) => i);
   176→      const allNodes = new Set(nodes.map(n => n.id));
   177→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   178→    }
   179→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   180→      const edgeSet = new Set<number>();
   181→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   182→      edges.forEach((e, i) => {
   183→        if (e.to === activeFilter) {
   184→          edgeSet.add(i);
   185→          nodeSet.add(e.from);
   186→        }
   187→      });
   188→      edges.forEach((e, i) => {
   189→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   190→          edgeSet.add(i);
   191→          nodeSet.add(e.from);
   192→        }
   193→      });
   194→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   195→    }
   196→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   197→      const branchType = `branch-${activeFilter}` as const;
   198→      const edgeSet = new Set<number>();
   199→      const nodeSet = new Set<string>(['ROOT']);
   200→      edges.forEach((e, i) => {
   201→        if (e.type === branchType) {
   202→          edgeSet.add(i);
   203→          nodeSet.add(e.from);
   204→          nodeSet.add(e.to);
   205→        }
   206→      });
   207→      const expanded = new Set<number>();
   208→      const expandedNodes = new Set<string>(nodeSet);
   209→      edges.forEach((e, i) => {
   210→        if (expandedNodes.has(e.from)) {
   211→          expanded.add(i);
   212→          expandedNodes.add(e.to);
   213→        }
   214→      });
   215→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   216→    }
   217→    // Type filter (up/pullback/down/risk)
   218→    const edgeSet = new Set<number>();
   219→    const nodeSet = new Set<string>(['ROOT']);
   220→    edges.forEach((e, i) => {
   221→      if (e.type === activeFilter) {
   222→        edgeSet.add(i);
   223→        nodeSet.add(e.from);
   224→        nodeSet.add(e.to);
   225→      }
   226→    });
   227→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   228→  }, [activeFilter, edges, nodes]);
   229→
   230→  // ── Draw SVG edges ───────────────────────────────────────────────
   231→  const drawEdges = useCallback(() => {
   232→    const svg = svgRef.current;
   233→    const graph = graphRef.current;
   234→    if (!svg || !graph || edges.length === 0) return;
   235→
   236→    let markersSvg = '';
   237→    const seenColors = new Set<string>();
   238→    for (const e of edges) {
   239→      const c = EDGE_COLORS[e.type];
   240→      if (c && !seenColors.has(e.type)) {
   241→        seenColors.add(e.type);
   242→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   243→      }
   244→    }
   245→
   246→    let pathsSvg = '';
   247→    const graphRect = graph.getBoundingClientRect();
   248→
   249→    for (let ei = 0; ei < edges.length; ei++) {
   250→      const { from: fromId, to: toId, label, type } = edges[ei];
   251→      const isVisible = visibleEdgeIndices.includes(ei);
   252→      const fromEl = nodeRefs.current[fromId];
   253→      const toEl = nodeRefs.current[toId];
   254→      if (!fromEl || !toEl) continue;
   255→
   256→      const fRect = fromEl.getBoundingClientRect();
   257→      const tRect = toEl.getBoundingClientRect();
   258→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   259→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   260→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   261→      const by = tRect.top - graphRect.top + tRect.height / 2;
   262→      const dx = bx - ax;
   263→      const dy = by - ay;
   264→      const dist = Math.sqrt(dx * dx + dy * dy);
   265→
   266→      if (dist === 0) continue;
   267→
   268→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   269→      const mx = (ax + bx) / 2;
   270→      const my = (ay + by) / 2;
   271→      const nx = -dy / dist;
   272→      const ny = dx / dist;
   273→      const cx = mx + nx * bend;
   274→      const cy = my + ny * bend;
   275→
   276→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   277→      const prob = edgeProbabilities[ei] ?? 0;
   278→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   279→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   280→
   281→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.72 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   282→
   283→      if (isVisible && ei >= 6) {
   284→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#ffffff" font-size="10" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="4" stroke-linejoin="round" opacity="0.95" data-type="${type}" class="edge-label">${label}</text>`;
   285→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   286→      }
   287→    }
   288→
   289→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   290→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   291→
   292→  useEffect(() => { drawEdges(); }, [drawEdges]);
   293→
   294→  useEffect(() => {
   295→    const el = shellRef.current;
   296→    if (!el) return;
   297→    let observer: ResizeObserver | null = null;
   298→    try {
   299→      observer = new ResizeObserver(() => { try { drawEdges(); } catch {} });
   300→      observer.observe(el);
   301→    } catch {}
   302→    return () => { try { observer?.disconnect(); } catch {} };
   303→  }, [drawEdges]);
   304→
   305→
   306→  // ── All filter buttons ────────────────────────────────────────────
   307→  const allFilters = [
   308→    ...TYPE_FILTERS,
   309→    { key: 'sep1', label: '│', isSep: true as const },
   310→    ...Object.entries(BRANCH_META).map(([k, v]) => ({ key: k, label: v.label, branchKey: k, branchColor: v.color })),
   311→    { key: 'sep2', label: '│', isSep: true as const },
   312→    ...SCENARIO_KEYS.map(k => ({ key: k, label: SCENARIO_META_LOCAL[k].label, scenarioKey: k })),
   313→  ];
   314→
   315→  // ── Corrected probabilityTrend: day 1 overridden with final scenario probabilities ──
   316→  const correctedTrend = useMemo(() => {
   317→    const raw = props.probabilityTrend;
   318→    if (!raw || !raw.scenarios || raw.scenarios.length === 0) return raw;
   319→
   320→    // Build final individual probabilities (matching boxes) for day 1
   321→    const day1Individuals: Record<string, number> = {};
   322→    for (const key of SCENARIO_KEYS) {
   323→      const s = scenarios[key as keyof typeof scenarios];
   324→      day1Individuals[key] = (s?.probability ?? 0) / 100;
   325→    }
   326→
   327→    // Corrected scenarios array
   328→    const correctedScenarios: ScenarioTrend[] = raw.scenarios.map(sc => {
   329→      const newInd = day1Individuals[sc.scenarioKey] ?? sc.trend[0]?.individualProb ?? 0;
   330→      const newCum = calculateCDF(sc.scenarioKey, day1Individuals);
   331→      const newTrend: DayPoint[] = sc.trend.map((dp, i) => {
   332→        if (i === 0) {
   333→          return { ...dp, individualProb: newInd, cumulativeProb: newCum };
   334→        }
   335→        return dp;
   336→      });
   337→      return {
   338→        ...sc,
   339→        currentProbability: newInd,
   340→        trend: newTrend,
   341→        peakProbability: Math.max(newInd, sc.peakProbability),
   342→      };
   343→    });
   344→
   345→    // Corrected groups
   346→    const correctedGroups = raw.groups.map(g => {
   347→      // Group cumulative for day 1
   348→      const day1AllInds: Record<string, number> = {};
   349→      for (const sc of raw.scenarios) {
   350→        day1AllInds[sc.scenarioKey] = day1Individuals[sc.scenarioKey] ?? sc.trend[0]?.individualProb ?? 0;
   351→      }
   352→      let groupCumDay1 = 0;
   353→      if (g.group === 'bullish') {
   354→        groupCumDay1 = calculateCDF('SC6', day1AllInds);
   355→      } else if (g.group === 'bearish') {
   356→        groupCumDay1 = calculateCDF('SC4', day1AllInds);
   357→      } else {
   358→        groupCumDay1 = day1AllInds['SC5'] ?? 0;
   359→      }
   360→
   361→      const newTrend: DayPoint[] = g.trend.map((dp, i) => {
   362→        if (i === 0) {
   363→          return { ...dp, individualProb: groupCumDay1, cumulativeProb: groupCumDay1 };
   364→        }
   365→        return dp;
   366→      });
   367→      return {
   368→        ...g,
   369→        trend: newTrend,
   370→        peakProbability: Math.max(groupCumDay1, g.peakProbability),
   371→      };
   372→    });
   373→
   374→    return { ...raw, scenarios: correctedScenarios, groups: correctedGroups };
   375→  }, [props.probabilityTrend, scenarios]);
   376→
   377→  // ── Meaningful node display values (Persian-formatted) ──────────
   378→  const nodeDisplayValues = useMemo(() => {
   379→    const display: Record<string, string> = {};
   380→    if (!decisionGraph) return display;
   381→
   382→    // Sub-branch parent mapping
   383→    const subParents: Record<string, string> = {
   384→      N_T_BULL: 'N_TREND', N_T_BEAR: 'N_TREND', N_T_FLAT: 'N_TREND',
   385→      N_B_UP: 'N_BREAK', N_B_DOWN: 'N_BREAK', N_B_NONE: 'N_BREAK',
   386→      N_R_BULL: 'N_REVERSAL', N_R_BEAR: 'N_REVERSAL', N_R_NONE: 'N_REVERSAL',
   387→    };
   388→
   389→    // Find edge probability from parent to child
   390→    const getEdgeProb = (from: string, to: string): number => {
   391→      for (let i = 0; i < edges.length; i++) {
   392→        if (edges[i].from === from && edges[i].to === to) {
   393→          return edgeProbabilities[i] ?? 0;
   394→        }
   395→      }
   396→      return 0;
   397→    };
   398→
   399→    // ROOT: no numeric value needed
   400→    display['ROOT'] = '';
   401→
   402→    // Main branches: absolute probability from root
   403→    display['N_TREND'] = toPersianDigits((branchProbs.trend * 100).toFixed(0)) + '٪';
   404→    display['N_BREAK'] = toPersianDigits((branchProbs.breakout * 100).toFixed(0)) + '٪';
   405→    display['N_REVERSAL'] = toPersianDigits((branchProbs.reversal * 100).toFixed(0)) + '٪';
   406→
   407→    // Sub-branches: conditional probability within parent (meaningful context)
   408→    for (const [sub, parent] of Object.entries(subParents)) {
   409→      const condProb = getEdgeProb(parent, sub);
   410→      display[sub] = toPersianDigits((condProb * 100).toFixed(0)) + '٪';
   411→    }
   412→
   413→    // Event/assessment nodes: no value (they represent conditions, not probability outcomes)
   414→    for (const node of nodes) {
   415→      if (node.type === 'event') {
   416→        display[node.id] = '';
   417→      }
   418→    }
   419→
   420→    // Terminal nodes (SC1-SC9): handled by the big badge, no extra text
   421→    for (const key of SCENARIO_KEYS) {
   422→      display[key] = '';
   423→    }
   424→
   425→    return display;
   426→  }, [decisionGraph, edges, edgeProbabilities, branchProbs, nodes]);
   427→
   428→  // ═══ Design dimensions ═══
   429→  const DESIGN_W = 1500;
   430→  const DESIGN_H = 1200;
   431→  const DISPLAY_W = 1200;
   432→  const DISPLAY_H = 1200;
   433→  const scaleX = DISPLAY_W / DESIGN_W;
   434→  const scaleY = DISPLAY_H / DESIGN_H;
   435→
   436→  // ── Loading state ────────────────────────────────────────────────
   437→  if (!decisionGraph) {
   438→    return <VdssGraphSkeleton />;
   439→  }
   440→
   441→  return (
   442→    <div className="space-y-3" dir="rtl">
   443→      {/* ═══ HEADER ═══ */}
   444→      <div className="flex items-center justify-between gap-4 px-5 py-4 rounded-2xl"
   445→        style={{
   446→          border: '1px solid rgba(58,213,219,.22)',
   447→          background: 'linear-gradient(105deg, rgba(14,35,53,.94), rgba(8,22,35,.77))',
   448→          boxShadow: D.shadow,
   449→        }}>
   450→        <div className="flex items-center gap-3.5">
   451→          <div style={{
   452→            width: 48, height: 48, display: 'grid', placeItems: 'center',
   453→            border: '1px solid rgba(255,177,27,.7)', borderRadius: 14,
   454→            color: D.gold, fontSize: 27,
   455→            boxShadow: 'inset 0 0 22px rgba(255,177,27,.12), 0 0 22px rgba(255,177,27,.08)',
   456→          }}>◈</div>
   457→          <div>
   458→            <h2 style={{ fontSize: 20, fontWeight: 700, margin: '0 0 6px', color: D.text }}>گراف تصمیم {symbolName}</h2>
   459→            <p style={{ fontSize: 12, color: D.muted, margin: 0 }}>مدل ۳‌شاخه‌ای | پیروی از روند، شکست، بازگشت | ۲۷ مسیر به ۹ سناریو</p>
   460→          </div>
   461→        </div>
   462→        <div className="text-left text-xs leading-loose pr-4"
   463→          style={{ color: '#ffffff', borderRight: '1px solid rgba(58,213,219,.24)' }}>
   464→          نقطهٔ مرجع: <b style={{ color: D.cyan }}>{toFa(currentPrice)}</b><br />
   465→          افق برآورد: ۱۰ تا ۲۵ جلسه معاملاتی
   466→        </div>
   467→      </div>
   468→
   469→      {/* ═══ Metric Cards ═══ */}
   470→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">

   802→
   803→export function generateDecisionGraphNarrative(p: {

   849→  const totalProb = ranked.reduce((s, r) => s + r.prob, 0);
   850→  lines.push(`
   851→مجموع: ${toPersianDigits(String(totalProb))}٪ — ${totalProb === 100 ? 'صحیح' : 'خطا در مجموع'}`);
   852→  lines.push('');
   853→
   854→  // Interpretation of top scenario
   855→  const topSc = ranked[0];
   856→  if (topSc) {
   857→    const group = SCENARIO_META[topSc.key]?.group;
   858→    const groupLabel = group === 'bullish' ? 'گاوی' : group === 'bearish' ? 'خرسی' : 'خنثی';
   859→    lines.push(`تفسیر: با احتمال ${toPersianDigits(String(topSc.prob))}٪، سناریو ${topSc.key} (${topSc.label}) بیشترین احتمال را دارد که در گروه ${groupLabel} قرار می‌گیرد. این بدان معناست که موتور تحلیلی انتظار حاصل ${topSc.label} را دارد.`);
   860→  }
   861→  lines.push('');
   862→
   863→  // ── 3. Cumulative probabilities ──
   864→  const bearishProb = ['SC1', 'SC2', 'SC3', 'SC4'].reduce((s, k) => s + (scenarios[k as keyof typeof scenarios]?.probability ?? 0), 0);
   865→  const neutralProb = scenarios.SC5?.probability ?? 0;
   866→  const bullishProb = ['SC6', 'SC7', 'SC8', 'SC9'].reduce((s, k) => s + (scenarios[k as keyof typeof scenarios]?.probability ?? 0), 0);
   867→
   868→  lines.push('احتمالات تجمعی (Cumulative Probabilities)');
   869→  lines.push('');
   870→  lines.push('احتمال تجمعی (CDF) در این مدل به این معنی است که از شدیدترین سناریو تا سناریوی مورد نظر را جمع می‌کند. به عنوان مثال، CDF سناریو SC6 (صعودی خفیف) مجموع احتمالات SC9+SC8+SC7+SC6 را شامل می‌شود که نشان‌دهنده احتمال حداقل یک بار صعود خفیف یا قوی‌تر است.');
   871→  lines.push('');
   872→  lines.push(`• گروه گاوی (SC6+SC7+SC8+SC9): ${toPersianDigits(String(bullishProb))}٪ (${toPersianDigits(String(Math.round(bullishProb)))} درصد)`);
   873→  lines.push(`• گروه خنثی (SC5): ${toPersianDigits(String(neutralProb))}٪ (${toPersianDigits(String(Math.round(neutralProb)))} درصد)`);
   874→  lines.push(`• گروه خرسی (SC1+SC2+SC3+SC4): ${toPersianDigits(String(bearishProb))}٪ (${toPersianDigits(String(Math.round(bearishProb)))} درصد)`);
   875→  lines.push('');
   876→
   877→  // Interpretation
   878→  if (bullishProb > bearishProb && bullishProb > neutralProb) {
   879→    lines.push(`تفسیر: با احتمال تجمعی ${toPersianDigits(String(Math.round(bullishProb)))} درصد برای سناریوهای گاوی، چشم‌انداز بازار به سمت صعود است. بازار با احتمال ${toPersianDigits(String(Math.round(bullishProb)))} درصد حداقل یک بار صعود خفیف یا قوی‌تر را تجربه خواهد کرد.`);
   880→  } else if (bearishProb > bullishProb && bearishProb > neutralProb) {
   881→    lines.push(`تفسیر: با احتمال تجمعی ${toPersianDigits(String(Math.round(bearishProb)))} درصد برای سناریوهای خرسی، چشم‌انداز بازار به سمت نزول است. بازار با احتمال ${toPersianDigits(String(Math.round(bearishProb)))} درصد حداقل یک بار نزول خفیف یا قوی‌تر را تجربه خواهد کرد.`);
   882→  } else {
   883→    lines.push(`تفسیر: بازار در وضعیت تعادل قرار دارد. احتمال گروه گاوی ${toPersianDigits(String(Math.round(bullishProb)))} درصد و گروه خرسی ${toPersianDigits(String(Math.round(bearishProb)))} درصد است که فاصله کمی بین آنها نشان‌دهنده عدم تعیین بازار است.`);
   884→  }
   885→  lines.push('');
   886→
   887→  // ── 4. Probability trends ──
   888→  if (probabilityTrend && probabilityTrend.scenarios && probabilityTrend.scenarios.length === 9 && probabilityTrend.scenarios[0]?.trend?.length > 1) {
   889→    lines.push('روند احتمالات (30 روزه)');
   890→    lines.push('');
   891→    lines.push('بررسی روند ۳۰ روزه احتمالات نشان می‌دهد که کدام سناریوها در حال تقویت و کدام در حال تضعیف هستند:');
   892→    lines.push('');
   893→
   894→    const rising: string[] = [];
   895→    const falling: string[] = [];
   896→    const stable: string[] = [];
   897→    const volatile: string[] = [];
   898→
   899→    for (const sc of probabilityTrend.scenarios) {
   900→      const meta = SCENARIO_META_LOCAL[sc.scenarioKey];
   901→      const entry = `${sc.scenarioKey} (${meta?.label ?? sc.label})`;
   902→      if (sc.trendDirection === 'rising') rising.push(entry);
   903→      else if (sc.trendDirection === 'falling') falling.push(entry);
   904→      else if (sc.trendDirection === 'stable') stable.push(entry);
   905→      else volatile.push(entry);
   906→    }
   907→
   908→    if (rising.length > 0) {
   909→      lines.push(`• سناریوهای صعودی (تقویت شده): ${rising.join('، ')}`);
   910→    }
   911→    if (falling.length > 0) {
   912→      lines.push(`• سناریوهای نزولی (تضعیف شده): ${falling.join('، ')}`);
   913→    }
   914→    if (stable.length > 0) {
   915→      lines.push(`• سناریوهای پایدار: ${stable.join('، ')}`);
   916→    }
   917→    if (volatile.length > 0) {
   918→      lines.push(`• سناریوهای ناپایدار: ${volatile.join('، ')}`);
   919→    }
   920→    lines.push('');
   921→
   922→    // Group trends
   923→    if (probabilityTrend.groups && probabilityTrend.groups.length === 3) {
   924→      lines.push('روند گروه‌ها:');
   925→      for (const g of probabilityTrend.groups) {
   926→        const dirLabel = g.trendDirection === 'rising' ? 'صعودی' : g.trendDirection === 'falling' ? 'نزولی' : g.trendDirection === 'volatile' ? 'ناپایدار' : 'پایدار';
   927→        const gLabel = g.group === 'bullish' ? 'گاوی' : g.group === 'bearish' ? 'خرسی' : 'خنثی';
   928→        lines.push(`• گروه ${gLabel}: روند ${dirLabel} | احتمال اختصاصی فعلی: ${faPct(g.trend[0]?.individualProb ?? 0)} درصد`);
   929→      }
   930→      lines.push('');
   931→    }
   932→  }
   933→
   934→  // ── 5. Branch contributions ──
   935→  lines.push('سهم استراتژی‌ها (Branch Contributions)');
   936→  lines.push('');
   937→  lines.push(`در این مدل، هر سناریو از سه مسیر مستقل غیرمتقاطع تغذیه می‌شود. سهم هر استراتژی نشان می‌دهد که چند درصد از احتمال نهایی سناریو از آن استراتژی تامین شده‌است.`);
   938→  lines.push('');
   939→  lines.push(`احتمال شاخه‌ها: پیرویی از روند ${faPct0(branchProbs.trend)}٪، شکست ${faPct0(branchProbs.breakout)}٪، بازگشت ${faPct0(branchProbs.reversal)}٪`);
   940→  lines.push('');
   941→
   942→  for (const key of SCENARIO_KEYS) {
   943→    const contrib = pathContributions[key] ?? { trend: 0, breakout: 0, reversal: 0 };
   944→    const meta = SCENARIO_META_LOCAL[key];
   945→    const s = scenarios[key as keyof typeof scenarios];
   946→    lines.push(`${key} (${meta?.label ?? ''}): پیرویی از روند ${faPctWhole(contrib.trend)}٪، شکست ${faPctWhole(contrib.breakout)}٪، بازگشت ${faPctWhole(contrib.reversal)}٪ — احتمال نهایی: ${toPersianDigits(String(s?.probability ?? 0))}٪`);
   947→  }
   948→  lines.push('');
   949→
   950→  return lines.join('\n');
   951→}

  1017→
  1018→function generateAdvancedDGAnalysis(p: {
  1019→  symbolName: string;
  1020→  currentPrice: number;
  1021→  currencyUnit: string;
  1022→  scenarios: VdssGraphProps['scenarios'];
  1023→  decisionGraph: GraphData | null;
  1024→  probabilityTrend: ProbabilityTrendResult | null;
  1025→  rsi: number;
  1026→  adx: number;
  1027→  atr: number;
  1028→  trendDirection: string;
  1029→  bullScore: number;
  1030→  resistances: number[];
  1031→  supports: number[];
  1032→  instrumentType?: string;
  1033→  instrumentCategory?: string;
  1034→  priceDecimals?: number;
  1035→}): string {
  1036→  const { symbolName, currentPrice, currencyUnit, scenarios, decisionGraph, probabilityTrend,
  1037→    rsi, adx, atr, trendDirection, bullScore, resistances, supports, instrumentCategory, priceDecimals } = p;
  1038→
  1039→  const unit = currencyUnit || 'ریال';
  1040→  const fa = (n: number) => {
  1041→    if (priceDecimals !== undefined && priceDecimals > 0) {
  1042→      return `\u200E${n.toLocaleString('fa-IR', { maximumFractionDigits: priceDecimals, minimumFractionDigits: 0 })}\u200E`;
  1043→    }
  1044→    return toPersianDigits(Math.round(n).toLocaleString('en-US'));
  1045→  };
  1046→  // pctWhole: values already 0–100 (scenario probabilities, pathContributions)
  1047→  const pctWhole = (n: number) => `${toPersianDigits(String(Math.round(n)))} درصد`;
  1048→  // pctFrac: values 0–1 (branchProbabilities) → multiply by 100
  1049→  const pctFrac = (n: number) => `${toPersianDigits(String(Math.round(n * 100)))} درصد`;
  1050→
  1051→  // ── Compute derived data ──
  1052→  const ranked = SCENARIO_KEYS.map(k => ({
  1053→    key: k,
  1054→    prob: scenarios[k as keyof typeof scenarios]?.probability ?? 0,
  1055→    label: SCENARIO_META_LOCAL[k].label,
  1056→    group: SCENARIO_META[k]?.group ?? 'neutral',
  1057→    targetMin: scenarios[k as keyof typeof scenarios]?.targetMin ?? 0,
  1058→    targetMax: scenarios[k as keyof typeof scenarios]?.targetMax ?? 0,
  1059→  })).sort((a, b) => b.prob - a.prob);
  1060→
  1061→  const bearProb = ranked.filter(r => r.group === 'bearish').reduce((s, r) => s + r.prob, 0);
  1062→  const bullProb = ranked.filter(r => r.group === 'bullish').reduce((s, r) => s + r.prob, 0);
  1063→  const neutProb = ranked.filter(r => r.group === 'neutral').reduce((s, r) => s + r.prob, 0);
  1064→
  1065→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.333, breakout: 0.333, reversal: 0.334 };
  1066→  const pathContribs = decisionGraph?.pathContributions ?? {} as Record<string, { trend: number; breakout: number; reversal: number }>;
  1067→
  1068→  const topSc = ranked[0];
  1069→  const secondSc = ranked[1];
  1070→  const topGroup = topSc?.group ?? 'neutral';
  1071→  const topGroupLabel = topGroup === 'bullish' ? 'گاوی' : topGroup === 'bearish' ? 'خرسی' : 'خنثی';
  1072→
  1073→  const R1 = resistances[0] ?? 0;
  1074→  const S1 = supports[0] ?? 0;
  1075→  const rrRatio = (R1 > 0 && S1 > 0 && currentPrice > S1)
  1076→    ? ((R1 - currentPrice) / (currentPrice - S1)).toFixed(1) : null;
  1077→
  1078→  const rsiZone = rsi > 70 ? 'اشباع خرید' : rsi > 60 ? 'نزدیک اشباع خرید' : rsi < 30 ? 'اشباع فروش' : rsi < 40 ? 'نزدیک اشباع فروش' : 'منطقه نرمال';
  1079→  const adxStrength = adx > 40 ? 'روند قدرتمند' : adx > 25 ? 'روند متوسط' : 'روند ضعیف یا رنج';
  1080→  const atrPct = currentPrice > 0 ? (atr / currentPrice * 100).toFixed(1) : '0';
  1081→  const volLevel = parseFloat(atrPct) > 3 ? 'نوسان بالا' : parseFloat(atrPct) > 1.5 ? 'نوسان متوسط' : 'نوسان کم';
  1082→
  1083→  // Branch dominance
  1084→  const branches = [
  1085→    { key: 'trend', label: 'پیروی از روند', prob: branchProbs.trend, color: 'emerald' as const },
  1086→    { key: 'breakout', label: 'شکست', prob: branchProbs.breakout, color: 'amber' as const },
  1087→    { key: 'reversal', label: 'بازگشت', prob: branchProbs.reversal, color: 'purple' as const },
  1088→  ].sort((a, b) => b.prob - a.prob);
  1089→  const dominantBranch = branches[0];
  1090→  const weakBranch = branches[2];
  1091→
  1092→  // Probability trend analysis
  1093→  const risingScs: string[] = [];
  1094→  const fallingScs: string[] = [];
  1095→  const stableScs: string[] = [];
  1096→  if (probabilityTrend?.scenarios?.length === 9) {
  1097→    for (const sc of probabilityTrend.scenarios) {
  1098→      const meta = SCENARIO_META_LOCAL[sc.scenarioKey];
  1099→      const entry = `${meta?.label ?? sc.label}`;
  1100→      if (sc.trendDirection === 'rising') risingScs.push(entry);
  1101→      else if (sc.trendDirection === 'falling') fallingScs.push(entry);
  1102→      else stableScs.push(entry);
  1103→    }
  1104→  }
  1105→
  1106→  // Dominant branch contribution to top scenarios
  1107→  const topScContrib = pathContribs[topSc?.key ?? 'SC5'] ?? { trend: 0, breakout: 0, reversal: 0 };
  1108→  const topScDominantStrategy = Object.entries(topScContrib).sort(([,a],[,b]) => (b as number) - (a as number))[0];
  1109→  const topScDominantLabel = topScDominantStrategy?.[0] === 'trend' ? 'پیروی از روند'
  1110→    : topScDominantStrategy?.[0] === 'breakout' ? 'شکست' : 'بازگشت';
  1111→
  1112→  // ── Build analysis paragraphs ──
  1113→  const lines: string[] = [];
  1114→
  1115→  // ── عنوان اصلی ──
  1116→  lines.push(`## تحلیل پیشرفته گراف تصمیم ${symbolName}`);
  1117→  lines.push('');
  1118→
  1119→  // ── بخش ۱: ساختار کلی گراف ──
  1120→  lines.push('### ساختار کلی گراف');
  1121→  lines.push('');
  1122→  lines.push(`گراف تصمیم ${symbolName} در نقطه مرجع {color:green}${fa(currentPrice)} ${unit}{/color}، یک مدل تحلیلی سه‌لایه با {color:blue}${toPersianDigits('27')} مسیر مستقل{/color} است که از ترکیب ${toPersianDigits('3')} استراتژی اصلی شامل پیروی از روند، شکست و بازگشت تشکیل شده‌است.`);
  1123→  lines.push('');
  1124→
  1125→  if (topSc) {
  1126→    const topColor = topGroup === 'bullish' ? 'green' : topGroup === 'bearish' ? 'red' : 'amber';
  1127→    lines.push(`سناریوی غالب با احتمال {color:${topColor}}${pctWhole(topSc.prob)}{/color}، **${topSc.label}** است که در گروه {color:${topColor}}${topGroupLabel}{/color} قرار می‌گیرد.`);
  1128→  }
  1129→
  1130→  if (secondSc) {
  1131→    lines.push(`سناریوی دوم با ${pctWhole(secondSc.prob)} احتمال، ${secondSc.label} می‌باشد و فاصله احتمالی بین دو سناریوی اول و دوم برابر ${toPersianDigits(String(Math.round(topSc.prob - secondSc.prob)))} درصد است${(topSc.prob - secondSc.prob) > 10 ? ' که نشان‌دهنده تمرکز بالای احتمالات است.' : '.'}`);
  1132→  }
  1133→
  1134→  lines.push(`مجموع احتمال گروه گاوی {color:green}${pctWhole(bullProb)}{/color}، گروه خنثی ${pctWhole(neutProb)} و گروه خرسی {color:red}${pctWhole(bearProb)}{/color} محاسبه شده‌است.`);
  1135→
  1136→  if (bullProb > bearProb + 0.2) {
  1137→    lines.push(`اختلاف قابل‌توجه ${toPersianDigits(String(Math.round(bullProb - bearProb)))} درصدی بین گروه گاوی و خرسی، چشم‌انداز صعودی بازار را تایید می‌کند.`);
  1138→  } else if (bearProb > bullProb + 0.2) {
  1139→    lines.push(`اختلاف ${toPersianDigits(String(Math.round(bearProb - bullProb)))} درصدی به نفع گروه خرسی، هشدار جدی برای معاملات خرید صادر می‌کند.`);
  1140→  } else {
  1141→    lines.push(`فاصله کم بین گروه گاوی و خرسی نشان‌دهنده **عدم قطعیت بازار** و لزوم مدیریت ریسک دقیق است.`);
  1142→  }
  1143→
  1144→  lines.push('');
  1145→
  1146→  // ── بخش ۲: تحلیل استراتژی‌ها ──
  1147→  lines.push('### تحلیل استراتژی‌ها');
  1148→  lines.push('');
  1149→  lines.push(`در ساختار درختی گراف تصمیم، استراتژی **${dominantBranch.label}** با سهم {color:${dominantBranch.color}}${pctFrac(dominantBranch.prob)}{/color} بیشترین تاثیر را در تعیین سناریوهای نهایی دارد.`);
  1150→
  1151→  if (weakBranch) {
  1152→    lines.push(`در مقابل، استراتژی ${weakBranch.label} تنها ${pctFrac(weakBranch.prob)} سهم دارد که نشان می‌دهد بازار کمتر انتظار ${weakBranch.key === 'reversal' ? 'واگرایی و بازگشت' : weakBranch.key === 'breakout' ? 'شکست سطوح' : 'ادامه روند'} را دارد.`);
  1153→  }

  1278→/** Parse {color:...}text{/color}, **bold**, ##/### headings from analysis text */
  1279→function renderDGAIText(text: string) {
  1280→  const lines = text.split('\n');
  1281→  const colorMap: Record<string, string> = {
  1282→    red: '#ef4444', green: '#22c55e', amber: '#f59e0b', blue: '#3b82f6',
  1283→    orange: '#f97316', purple: '#a855f7', emerald: '#10b981',
  1284→  };
  1285→  const processInline = (raw: string) => {
  1286→    let html = raw;
  1287→    html = html.replace(/\{color:([a-z]+)\}([^\{]*?)\{\/color\}/g, (_m: string, color: string, inner: string) => {
  1288→      return '<span style="color:' + (colorMap[color] || color) + ';font-weight:700">' + inner + '</span>';
  1289→    });
  1290→    html = html.replace(/\*\*([^*]+?)\*\*/g, '<strong>$1</strong>');
  1291→    return html;
  1292→  };
  1293→
  1294→  return lines.map((line, li) => {
  1295→    const trimmed = line.trim();
  1296→    if (!trimmed) return <div key={li} style={{ height: 6 }} />;
  1297→
  1298→    // ## Main heading
  1299→    if (trimmed.startsWith('## ')) {
  1300→      const html = processInline(trimmed.slice(3));
  1301→      return <h2 key={li} style={{ fontSize: 16, fontWeight: 800, color: D.text, marginTop: 4, marginBottom: 8, paddingBottom: 6, borderBottom: `1px solid ${D.line}` }} dangerouslySetInnerHTML={{ __html: html }} />;
  1302→    }
  1303→    // ### Section heading
  1304→    if (trimmed.startsWith('### ')) {
  1305→      const html = processInline(trimmed.slice(4));
  1306→      return <h3 key={li} style={{ fontSize: 14, fontWeight: 700, color: D.text, marginTop: 14, marginBottom: 6 }} dangerouslySetInnerHTML={{ __html: html }} />;
  1307→    }
  1308→    // **Bold line heading** (fallback for old-style headings)
  1309→    const isHeading = /^\*{0,2}[^*]{3,40}:$/.test(trimmed);
  1310→    if (isHeading) {
  1311→      const html = processInline(trimmed);
  1312→      return <h3 key={li} style={{ fontSize: 14, fontWeight: 700, color: D.text, marginTop: 12, marginBottom: 4 }} dangerouslySetInnerHTML={{ __html: html }} />;
  1313→    }
  1314→
  1315→    const html = processInline(trimmed);
  1316→    return <p key={li} style={{ margin: '4px 0', lineHeight: 2.2, fontSize: 13, color: D.text }} dangerouslySetInnerHTML={{ __html: html }} />;
  1317→  });
  1318→}
