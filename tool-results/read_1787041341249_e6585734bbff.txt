   102→   102→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   103→   103→function sma(closes: number[], period: number): number {
   104→   104→  if (closes.length < period) return 0;
   105→   105→  const slice = closes.slice(-period);
   106→   106→  return slice.reduce((a, b) => a + b, 0) / period;
   107→   107→}

   109→   109→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   110→   110→function emaCalc(closes: number[], period: number): number {
   111→   111→  if (closes.length < period) return 0;
   112→   112→  const k = 2 / (period + 1);
   113→   113→  let emaVal = sma(closes.slice(0, period), period);
   114→   114→  for (let i = period; i < closes.length; i++) {
   115→   115→    emaVal = closes[i] * k + emaVal * (1 - k);
   116→   116→  }
   117→   117→  return emaVal;
   118→   118→}

   120→   120→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   121→   121→function calcRSI(closes: number[], period: number = 14): number {
   122→   122→  if (closes.length < period + 1) return 50;
   123→   123→  let avgGain = 0;
   124→   124→  let avgLoss = 0;
   125→   125→  for (let i = 1; i <= period; i++) {
   126→   126→    const diff = closes[i] - closes[i - 1];
   127→   127→    if (diff > 0) avgGain += diff;
   128→   128→    else avgLoss += Math.abs(diff);
   129→   129→  }
   130→   130→  avgGain /= period;
   131→   131→  avgLoss /= period;
   132→   132→  for (let i = period + 1; i < closes.length; i++) {
   133→   133→    const diff = closes[i] - closes[i - 1];
   134→   134→    const gain = diff > 0 ? diff : 0;
   135→   135→    const loss = diff < 0 ? Math.abs(diff) : 0;
   136→   136→    avgGain = (avgGain * (period - 1) + gain) / period;
   137→   137→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   138→   138→  }
   139→   139→  if (avgLoss === 0) return 100;
   140→   140→  const rs = avgGain / avgLoss;
   141→   141→  return 100 - 100 / (1 + rs);
   142→   142→}

   144→   144→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   145→   145→function calcMFI(data: OHLCV[], period: number = 14): number {
   146→   146→  if (data.length < period + 1) return 50;
   147→   147→  let posFlow = 0;
   148→   148→  let negFlow = 0;
   149→   149→  for (let i = data.length - period; i < data.length; i++) {
   150→   150→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   151→   151→    const mf = tp * data[i].volume;
   152→   152→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   153→   153→    if (tp > prevTp) posFlow += mf;
   154→   154→    else negFlow += mf;
   155→   155→  }
   156→   156→  if (negFlow === 0) return 100;
   157→   157→  const mfr = posFlow / negFlow;
   158→   158→  return 100 - 100 / (1 + mfr);
   159→   159→}

   162→   162→function calcCCI(data: OHLCV[], period: number = 20): number {
   163→   163→  if (data.length < period) return 0;
   164→   164→  const tps: number[] = [];
   165→   165→  for (let i = data.length - period; i < data.length; i++) {
   166→   166→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   167→   167→  }
   168→   168→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   169→   169→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   170→   170→  if (meanDev === 0) return 0;
   171→   171→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   172→   172→}

   175→   175→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   176→   176→  if (data.length < kPeriod) return { k: 50, d: 50 };
   177→   177→  const rawKs: number[] = [];
   178→   178→  for (let i = kPeriod - 1; i < data.length; i++) {
   179→   179→    let lowest = Infinity;
   180→   180→    let highest = -Infinity;
   181→   181→    for (let j = i - kPeriod + 1; j <= i; j++) {
   182→   182→      if (data[j].low < lowest) lowest = data[j].low;
   183→   183→      if (data[j].high > highest) highest = data[j].high;
   184→   184→    }
   185→   185→    const range = highest - lowest;
   186→   186→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   187→   187→  }
   188→   188→  const smoothedK: number[] = [];
   189→   189→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   190→   190→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   191→   191→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   192→   192→  }
   193→   193→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   194→   194→  const dSlice = smoothedK.slice(-smoothD);
   195→   195→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   196→   196→  return { k, d };
   197→   197→}

   200→   200→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   201→   201→  if (data.length < period) return -50;
   202→   202→  const slice = data.slice(-period);
   203→   203→  let highest = -Infinity;
   204→   204→  let lowest = Infinity;
   205→   205→  for (const c of slice) {
   206→   206→    if (c.high > highest) highest = c.high;
   207→   207→    if (c.low < lowest) lowest = c.low;
   208→   208→  }
   209→   209→  const range = highest - lowest;
   210→   210→  if (range === 0) return -50;
   211→   211→  return ((highest - data[data.length - 1].close) / range) * -100;
   212→   212→}

   214→   214→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   215→   215→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   216→   216→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   217→   217→  const emaFast = emaCalc(closes, fast);
   218→   218→  const emaSlow = emaCalc(closes, slow);
   219→   219→  const macdLine = emaFast - emaSlow;
   220→   220→  // Calculate signal as EMA of MACD values
   221→   221→  const macdValues: number[] = [];
   222→   222→  const kFast = 2 / (fast + 1);
   223→   223→  const kSlow = 2 / (slow + 1);
   224→   224→  let eFast = sma(closes.slice(0, fast), fast);
   225→   225→  let eSlow = sma(closes.slice(0, slow), slow);
   226→   226→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   227→   227→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   228→   228→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   229→   229→    macdValues.push(eFast - eSlow);
   230→   230→  }
   231→   231→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   232→   232→  const kSig = 2 / (sig + 1);
   233→   233→  let signalVal = sma(macdValues.slice(0, sig), sig);
   234→   234→  for (let i = sig; i < macdValues.length; i++) {
   235→   235→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   236→   236→  }
   237→   237→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   238→   238→}

   240→   240→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   241→   241→function calcATR(data: OHLCV[], period: number = 14): number {
   242→   242→  if (data.length < 2) return 0;
   243→   243→  const trs: number[] = [];
   244→   244→  for (let i = 1; i < data.length; i++) {
   245→   245→    const tr = Math.max(
   246→   246→      data[i].high - data[i].low,
   247→   247→      Math.abs(data[i].high - data[i - 1].close),
   248→   248→      Math.abs(data[i].low - data[i - 1].close)
   249→   249→    );
   250→   250→    trs.push(tr);
   251→   251→  }
   252→   252→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   253→   253→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   254→   254→  for (let i = period; i < trs.length; i++) {
   255→   255→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   256→   256→  }
   257→   257→  return atrVal;
   258→   258→}

   260→   260→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   261→   261→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   262→   262→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   263→   263→  const plusDMs: number[] = [];
   264→   264→  const minusDMs: number[] = [];
   265→   265→  const trs: number[] = [];
   266→   266→  for (let i = 1; i < data.length; i++) {
   267→   267→    const upMove = data[i].high - data[i - 1].high;
   268→   268→    const downMove = data[i - 1].low - data[i].low;
   269→   269→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   270→   270→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   271→   271→    trs.push(Math.max(
   272→   272→      data[i].high - data[i].low,
   273→   273→      Math.abs(data[i].high - data[i - 1].close),
   274→   274→      Math.abs(data[i].low - data[i - 1].close)
   275→   275→    ));
   276→   276→  }
   277→   277→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   278→   278→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   279→   279→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   280→   280→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   281→   281→  const dxValues: number[] = [];
   282→   282→  for (let i = period; i < trs.length; i++) {
   283→   283→    smoothTR = smoothTR - smoothTR / period + trs[i];
   284→   284→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   285→   285→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   286→   286→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   287→   287→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   288→   288→    const diSum = diPlus + diMinus;
   289→   289→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   290→   290→    dxValues.push(dx);
   291→   291→  }
   292→   292→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   293→   293→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   294→   294→  for (let i = period; i < dxValues.length; i++) {
   295→   295→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   296→   296→  }
   297→   297→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   298→   298→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   299→   299→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   300→   300→}

   302→   302→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   303→   303→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   304→   304→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   305→   305→  let isLong = true;
   306→   306→  let af = step;
   307→   307→  let ep = data[0].high;
   308→   308→  let sar = data[0].low;
   309→   309→  for (let i = 1; i < data.length; i++) {
   310→   310→    const prevSar = sar;
   311→   311→    sar = prevSar + af * (ep - prevSar);
   312→   312→    // Ensure SAR is within bounds
   313→   313→    if (isLong) {
   314→   314→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   315→   315→      if (data[i].low < sar) {
   316→   316→        isLong = false;
   317→   317→        sar = ep;
   318→   318→        ep = data[i].low;
   319→   319→        af = step;
   320→   320→      } else {
   321→   321→        if (data[i].high > ep) {
   322→   322→          ep = data[i].high;
   323→   323→          af = Math.min(af + step, max);
   324→   324→        }
   325→   325→      }
   326→   326→    } else {
   327→   327→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   328→   328→      if (data[i].high > sar) {
   329→   329→        isLong = true;
   330→   330→        sar = ep;
   331→   331→        ep = data[i].high;
   332→   332→        af = step;
   333→   333→      } else {
   334→   334→        if (data[i].low < ep) {
   335→   335→          ep = data[i].low;
   336→   336→          af = Math.min(af + step, max);
   337→   337→        }
   338→   338→      }
   339→   339→    }
   340→   340→  }
   341→   341→  return sar;
   342→   342→}

   345→   345→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   346→   346→  if (closes.length < period) {
   347→   347→    const c = closes[closes.length - 1] ?? 0;
   348→   348→    return { upper: c, middle: c, lower: c };
   349→   349→  }
   350→   350→  const slice = closes.slice(-period);
   351→   351→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   352→   352→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   353→   353→  const sd = Math.sqrt(variance);
   354→   354→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   355→   355→}

   358→   358→function calcOBV(data: OHLCV[]): number {

   361→   361→    if (data[i].close > data[i - 1].close) obv += data[i].volume;

   363→   363→  }
   364→   364→  return obv;

   367→   367→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   368→   368→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   369→   369→  const n = values.length;
   370→   370→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   371→   371→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   372→   372→  for (let i = 0; i < n; i++) {
   373→   373→    sumX += i;
   374→   374→    sumY += values[i];
   375→   375→    sumXY += i * values[i];
   376→   376→    sumX2 += i * i;
   377→   377→    sumY2 += values[i] * values[i];
   378→   378→  }
   379→   379→  const denom = n * sumX2 - sumX * sumX;
   380→   380→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   381→   381→  const slope = (n * sumXY - sumX * sumY) / denom;
   382→   382→  const intercept = (sumY - slope * sumX) / n;
   383→   383→  const meanY = sumY / n;
   384→   384→  let ssRes = 0, ssTot = 0;
   385→   385→  for (let i = 0; i < n; i++) {
   386→   386→    const predicted = slope * i + intercept;
   387→   387→    ssRes += (values[i] - predicted) ** 2;
   388→   388→    ssTot += (values[i] - meanY) ** 2;
   389→   389→  }
   390→   390→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   391→   391→  return { slope, intercept, r2 };
   392→   392→}

   395→   395→function roundToNice(price: number): number {
   396→   396→  const abs = Math.abs(price);
   397→   397→  let step: number;
   398→   398→  if (abs >= 100000) step = 100;
   399→   399→  else if (abs >= 10000) step = 50;
   400→   400→  else if (abs >= 1000) step = 10;
   401→   401→  else if (abs >= 100) step = 5;
   402→   402→  else step = 1;
   403→   403→  return Math.round(price / step) * step;
   404→   404→}

   406→   406→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
   407→   407→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: number[]; lows: number[] } {
   408→   408→  const highs: number[] = [];
   409→   409→  const lows: number[] = [];
   410→   410→  for (let i = lookback; i < data.length - lookback; i++) {
   411→   411→    let isHigh = true;
   412→   412→    let isLow = true;
   413→   413→    for (let j = i - lookback; j <= i + lookback; j++) {
   414→   414→      if (j === i) continue;
   415→   415→      if (data[j].high >= data[i].high) isHigh = false;
   416→   416→      if (data[j].low <= data[i].low) isLow = false;
   417→   417→    }
   418→   418→    if (isHigh) highs.push(data[i].high);
   419→   419→    if (isLow) lows.push(data[i].low);
   420→   420→  }
   421→   421→  return { highs, lows };
   422→   422→}

   424→   424→// ─── Support / Resistance (6 each, adaptive gap, multi-factor strength) ───
   425→   425→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   426→   426→  resistances: number[];
   427→   427→  supports: number[];
   428→   428→  supportStrengths: LevelStrength[];
   429→   429→  resistanceStrengths: LevelStrength[];
   430→   430→  priceTargets: LevelStrength[];
   431→   431→} {
   432→   432→  const closes = data.map(d => d.close);
   433→   433→  const avgVolume = data.reduce((s, d) => s + d.volume, 0) / data.length;
   434→   434→
   435→   435→  // Tag each raw level with its source type for confluence scoring
   436→   436→  interface TaggedLevel { price: number; source: string };
   437→   437→  const allRaw: TaggedLevel[] = [];
   438→   438→
   439→   439→  // 1. Pivot Points (source: 'pivot')
   440→   440→  const prev = data[data.length - 2] ?? data[data.length - 1];
   441→   441→  if (prev) {
   442→   442→    const pp = (prev.high + prev.low + prev.close) / 3;
   443→   443→    const hl = prev.high - prev.low;
   444→   444→    const pivotR = [
   445→   445→      2 * pp - prev.low, pp + hl,
   446→   446→      prev.high + 2 * (pp - prev.low),
   447→   447→      prev.high + 2 * (pp - prev.low) + hl,
   448→   448→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   449→   449→    ];
   450→   450→    const pivotS = [
   451→   451→      2 * pp - prev.high, pp - hl,
   452→   452→      prev.low - 2 * (prev.high - pp),
   453→   453→      prev.low - 2 * (prev.high - pp) - hl,
   454→   454→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   455→   455→    ];
   456→   456→    for (const p of pivotR) allRaw.push({ price: p, source: 'pivot' });
   457→   457→    for (const p of pivotS) allRaw.push({ price: p, source: 'pivot' });
   458→   458→  }
   459→   459→
   460→   460→  // 2. Swing levels (source: 'swing')
   461→   461→  for (const lb of [3, 5, 7, 10]) {
   462→   462→    const swings = findSwingLevels(data, lb);
   463→   463→    for (const h of swings.highs) allRaw.push({ price: h, source: 'swing' });
   464→   464→    for (const l of swings.lows) allRaw.push({ price: l, source: 'swing' });
   465→   465→  }
   466→   466→
   467→   467→  // 3. Moving Average levels (source: 'ma')
   468→   468→  for (const period of [5, 10, 21, 50, 100, 200]) {
   469→   469→    const v = sma(closes, period);
   470→   470→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01)
   471→   471→      allRaw.push({ price: v, source: 'ma' });
   472→   472→  }
   473→   473→  const ema12 = emaCalc(closes, 12);
   474→   474→  const ema26 = emaCalc(closes, 26);
   475→   475→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01)
   476→   476→    allRaw.push({ price: ema12, source: 'ma' });
   477→   477→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01)
   478→   478→    allRaw.push({ price: ema26, source: 'ma' });
   479→   479→
   480→   480→  // 4. Bollinger Band levels (source: 'bb')
   481→   481→  const bbLevels = calcBollingerBands(closes);
   482→   482→  if (bbLevels.upper > 0) allRaw.push({ price: bbLevels.upper, source: 'bb' });
   483→   483→  if (bbLevels.lower > 0) allRaw.push({ price: bbLevels.lower, source: 'bb' });
   484→   484→
   485→   485→  // 5. Round number / psychological levels (source: 'round')
   486→   486→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
   487→   487→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
   488→   488→    const p = Math.round(currentPrice * mult / magnitude) * magnitude;
   489→   489→    allRaw.push({ price: p, source: 'round' });
   490→   490→  }
   491→   491→
   492→   492→  // 6. Recent high/low zones (source: 'hilo')
   493→   493→  for (const w of [5, 22, 66]) {
   494→   494→    const slice = data.slice(-Math.min(w, data.length));
   495→   495→    if (slice.length > 0) {
   496→   496→      allRaw.push({ price: Math.max(...slice.map(d => d.high)), source: 'hilo' });
   497→   497→      allRaw.push({ price: Math.min(...slice.map(d => d.low)), source: 'hilo' });
   498→   498→    }
   499→   499→  }
   500→   500→
   501→   501→  // 7. Round all and filter invalid
   502→   502→  const rounded: TaggedLevel[] = allRaw
   503→   503→    .map(l => ({ price: roundToNice(l.price), source: l.source }))
   504→   504→    .filter(l => l.price > 0);
   505→   505→
   506→   506→  // 8. Cluster nearby levels (within 0.5%) — merge sources, keep most common price
   507→   507→  interface Cluster {
   508→   508→    price: number;
   509→   509→    sources: Set<string>;
   510→   510→    touchCount: number;
   511→   511→    totalVolumeAtTouches: number;
   512→   512→    mostRecentTouchBar: number;
   513→   513→  }
   514→   514→  const clusters: Cluster[] = [];
   515→   515→  for (const lvl of rounded) {
   516→   516→    let merged = false;
   517→   517→    for (const c of clusters) {
   518→   518→      if (Math.abs(c.price - lvl.price) / Math.max(c.price, 1) < 0.005) {
   519→   519→        c.sources.add(lvl.source);
   520→   520→        merged = true;
   521→   521→        break;
   522→   522→      }
   523→   523→    }
   524→   524→    if (!merged) {
   525→   525→      clusters.push({ price: lvl.price, sources: new Set([lvl.source]), touchCount: 0, totalVolumeAtTouches: 0, mostRecentTouchBar: -1 });
   526→   526→    }
   527→   527→  }
   528→   528→
   529→   529→  // 9. Count actual price TOUCHES (rejections) and volume at each cluster level
   530→   530→  // A real touch = bar wick reached the level AND bar body closed away from it
   531→   531→  const touchThreshold = currentPrice * 0.004; // 0.4% tight proximity
   532→   532→  for (let i = 0; i < data.length; i++) {
   533→   533→    const bar = data[i];
   534→   534→    for (const c of clusters) {
   535→   535→      const isResistance = c.price > currentPrice;
   536→   536→      if (isResistance) {
   537→   537→        // Resistance: bar high reached near the level AND closed below it (rejection)
   538→   538→        if (bar.high >= c.price - touchThreshold && bar.close < c.price) {
   539→   539→          c.touchCount++;
   540→   540→          c.totalVolumeAtTouches += bar.volume;
   541→   541→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   542→   542→        }
   543→   543→      } else {
   544→   544→        // Support: bar low reached near the level AND closed above it (rejection)
   545→   545→        if (bar.low <= c.price + touchThreshold && bar.close > c.price) {
   546→   546→          c.touchCount++;
   547→   547→          c.totalVolumeAtTouches += bar.volume;
   548→   548→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   549→   549→        }
   550→   550→      }
   551→   551→    }
   552→   552→  }
   553→   553→
   554→   554→  // 10. Calculate multi-factor strength (0-10)
   555→   555→  const levelStrengths: LevelStrength[] = clusters.map(c => {
   556→   556→    // Factor 1: Source Confluence (0-2.5 pts)
   557→   557→    const sourceScore = Math.min(2.5, c.sources.size * 0.5);
   558→   558→    const hasSwingAndPivot = c.sources.has('swing') && c.sources.has('pivot') ? 0.5 : 0;
   559→   559→    const maConfluence = c.sources.has('ma') ? 0.5 : 0;
   560→   560→    const confluenceScore = Math.min(2.5, sourceScore + hasSwingAndPivot + maConfluence);
   561→   561→
   562→   562→    // Factor 2: Touch Count — CAPPED by recency (0-2 pts)
   563→   563→    // Only count touches in the most recent 50% of data to avoid ancient history inflating
   564→   564→    const recentBarStart = Math.floor(data.length * 0.5);
   565→   565→    const recentTouches = c.mostRecentTouchBar >= recentBarStart ? c.touchCount : Math.max(0, c.touchCount - 1);
   566→   566→    let touchScore: number;
   567→   567→    if (recentTouches === 0) touchScore = 0;
   568→   568→    else if (recentTouches === 1) touchScore = 0.4;
   569→   569→    else if (recentTouches <= 2) touchScore = 0.8;
   570→   570→    else if (recentTouches <= 4) touchScore = 1.2;
   571→   571→    else if (recentTouches <= 7) touchScore = 1.6;
   572→   572→    else touchScore = 2.0;
   573→   573→
   574→   574→    // Factor 3: Volume Confirmation (0-1.5 pts)
   575→   575→    let volumeScore = 0;
   576→   576→    if (c.touchCount > 0) {
   577→   577→      const avgVolAtTouch = c.totalVolumeAtTouches / c.touchCount;
   578→   578→      const volRatio = avgVolume > 0 ? avgVolAtTouch / avgVolume : 1;
   579→   579→      if (volRatio > 2.0) volumeScore = 1.5;
   580→   580→      else if (volRatio > 1.5) volumeScore = 1.2;
   581→   581→      else if (volRatio > 1.0) volumeScore = 0.8;
   582→   582→      else if (volRatio > 0.7) volumeScore = 0.4;
   583→   583→      else volumeScore = 0.2;
   584→   584→    }
   585→   585→
   586→   586→    // Factor 4: Freshness / Recency (0-1.5 pts)
   587→   587→    let freshnessScore = 0;
   588→   588→    if (c.touchCount > 0 && data.length > 0) {
   589→   589→      const barsAgo = data.length - 1 - c.mostRecentTouchBar;
   590→   590→      if (barsAgo <= 3) freshnessScore = 1.5;
   591→   591→      else if (barsAgo <= 10) freshnessScore = 1.2;
   592→   592→      else if (barsAgo <= 25) freshnessScore = 0.8;
   593→   593→      else if (barsAgo <= 50) freshnessScore = 0.4;
   594→   594→      else freshnessScore = 0.1;
   595→   595→    }
   596→   596→
   597→   597→    // Factor 5: Proximity to current price (0-1 pt)
   598→   598→    const distRatio = Math.abs(c.price - currentPrice) / currentPrice;
   599→   599→    let proximityScore = 0;
   600→   600→    if (distRatio < 0.015) proximityScore = 1.0;
   601→   601→    else if (distRatio < 0.03) proximityScore = 0.7;
   602→   602→    else if (distRatio < 0.06) proximityScore = 0.4;
   603→   603→    else if (distRatio < 0.10) proximityScore = 0.2;
   604→   604→
   605→   605→    const total = confluenceScore + touchScore + volumeScore + freshnessScore + proximityScore;
   606→   606→    const strength = Math.min(10, Math.max(1, Math.round(total)));
   607→   607→
   608→   608→    return { price: c.price, strength, isTarget: false };
   609→   609→  });
   610→   610→
   611→   611→  // 11. Separate above/below current price
   612→   612→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   613→   613→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   614→   614→
   615→   615→  // 12. Adaptive gap enforcement — keep highest strength in cluster
   616→   616→  function enforceGap(levels: LevelStrength[], targetCount: number): LevelStrength[] {
   617→   617→    const gapSteps = [0.05, 0.04, 0.03, 0.025, 0.02, 0.015, 0.01, 0.008, 0.005];
   618→   618→    for (const gapPct of gapSteps) {
   619→   619→      if (levels.length === 0) return [];
   620→   620→      const result: LevelStrength[] = [levels[0]];
   621→   621→      for (let i = 1; i < levels.length; i++) {
   622→   622→        if (result.length >= targetCount) break;
   623→   623→        const prevPrice = result[result.length - 1].price;
   624→   624→        const currPrice = levels[i].price;
   625→   625→        const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   626→   626→        if (gap >= gapPct) {
   627→   627→          result.push(levels[i]);
   628→   628→        } else {
   629→   629→          // Merge: keep stronger level's price, add 0.5 for confluence
   630→   630→          const keep = levels[i].strength >= result[result.length - 1].strength ? levels[i] : result[result.length - 1];
   631→   631→          const mergedStrength = Math.min(10, Math.max(keep.strength, result[result.length - 1].strength) + 0.5);
   632→   632→          result[result.length - 1] = { ...keep, strength: Math.round(mergedStrength) };
   633→   633→        }
   634→   634→      }
   635→   635→      if (result.length >= targetCount) return result;
   636→   636→    }
   637→   637→    return levels.slice(0, targetCount);
   638→   638→  }
   639→   639→
   640→   640→  const finalResistances = enforceGap(above, 6);
   641→   641→  const finalSupports = enforceGap(below, 6);
   642→   642→
   643→   643→  // 13. Determine price targets: top 2 by strength, or any above 7
   644→   644→  const allLevels = [...finalResistances, ...finalSupports];
   645→   645→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   646→   646→  let targets: LevelStrength[];
   647→   647→  if (sortedByStrength.some(l => l.strength >= 7)) {
   648→   648→    targets = sortedByStrength.filter(l => l.strength >= 7).slice(0, 3);
   649→   649→  } else {
   650→   650→    targets = sortedByStrength.slice(0, 2);
   651→   651→  }
   652→   652→
   653→   653→  // Mark targets
   654→   654→  targets = targets.map(t => ({ ...t, isTarget: true }));
   655→   655→
   656→   656→  // Also mark in final arrays
   657→   657→  finalResistances.forEach(r => {
   658→   658→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   659→   659→  });
   660→   660→  finalSupports.forEach(s => {
   661→   661→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   662→   662→  });
   663→   663→
   664→   664→  return {
   665→   665→    resistances: finalResistances.map(l => l.price),
   666→   666→    supports: finalSupports.map(l => l.price),
   667→   667→    supportStrengths: finalSupports,
   668→   668→    resistanceStrengths: finalResistances,
   669→   669→    priceTargets: targets,
   670→   670→  };
   671→   671→}

   674→   674→function calcTrend(closes: number[], period: number): TrendResult {
   675→   675→  const slice = closes.slice(-Math.min(period, closes.length));
   676→   676→  const { slope, r2 } = linearRegression(slice);
   677→   677→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   678→   678→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   679→   679→  const priceThreshold = avgPrice * 0.001; // 0.1%
   680→   680→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   681→   681→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   682→   682→}
