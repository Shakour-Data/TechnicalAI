    28→  ma21: number;
    29→  ma100: number;

    32→  cci: number;
    33→  adx: number;
    34→  stochK: number;
    35→  stochD: number;
    36→  macdLine: number;
    37→  macdSignal: number;
    38→  macdHist: number;
    39→  diPlus: number;
    40→  diMinus: number;
    41→  sar: number;

    43→  obv: number;
    44→  hasVolume: boolean;
    45→  bollingerUpper: number;

    67→
    68→const toFa = (n: number) => Math.round(n).toLocaleString('fa-IR');

    74→  R1: { label: 'تداوم صعود هیجانی', type: 'up', border: '#59e39b', badgeBg: 'rgba(89,227,155,0.2)', badgeColor: '#59e39b' },
    75→  R2: { label: 'پولبک سالم', type: 'pullback', border: '#ffb25f', badgeBg: 'rgba(255,178,95,0.2)', badgeColor: '#ffb25f' },
    76→  R3: { label: 'اصلاح کنترل‌شده', type: 'down', border: '#ff758a', badgeBg: 'rgba(255,117,138,0.2)', badgeColor: '#ff758a' },
    77→  R4: { label: 'اصلاح عمیق', type: 'down', border: '#ffb11b', badgeBg: 'rgba(255,177,27,0.2)', badgeColor: '#ffb11b' },
    78→  R5: { label: 'تضعیف ساختار', type: 'down', border: '#ef4d62', badgeBg: 'rgba(239,77,98,0.2)', badgeColor: '#ef4d62' },
    79→};
    80→
    81→const STRATEGY_MAP: Record<string, { text: string; tagCls: string }> = {
    82→  R1: { text: 'صعودی قوی — احتمال بالای عبور از مقاومت‌ها', tagCls: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30' },
    83→  R2: { text: 'صعود تدریجی — ورود در اصلاح توصیه می‌شود', tagCls: 'bg-cyan-500/15 text-cyan-400 border-cyan-500/30' },
    84→  R3: { text: 'بازار رنج — منتظر خروج از محدوده بمانید', tagCls: 'bg-amber-500/15 text-amber-400 border-amber-500/30' },
    85→  R4: { text: 'اصلاحی — احتیاط و کاهش حجم معاملات', tagCls: 'bg-orange-500/15 text-orange-400 border-orange-500/30' },
    86→  R5: { text: 'نزولی قوی — خروج فوری توصیه می‌شود', tagCls: 'bg-red-500/15 text-red-400 border-red-500/30' },
    87→};

    89→// ═══════════════════════════════════════════════════════════════════════════════
    90→// Component
    91→// ═══════════════════════════════════════════════════════════════════════════════
    92→
    93→export default function VdesAnalysis(props: VdesAnalysisProps) {
    94→  const {
    95→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
    96→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
    97→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
    98→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
    99→    supportStrengths, resistanceStrengths, priceTargets,
   100→  } = props;
   101→
   102→  const containerRef = useRef<HTMLDivElement>(null);
   103→  const textAnalysisRef = useRef<HTMLDivElement>(null);
   104→  const [exporting, setExporting] = useState(false);
   105→
   106→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   107→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   108→  const S1_level = supports[0] ?? currentPrice * 0.95;
   109→  const S2_level = supports[1] ?? currentPrice * 0.90;
   110→  const S3_level = supports[2] ?? currentPrice * 0.85;
   111→
   112→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   113→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   114→
   115→  // ── Dominant scenario ──────────────────────────────────────────
   116→  let highestKey = 'R3';
   117→  let highestProb = 0;
   118→  for (const key of SCENARIO_KEYS) {
   119→    if (scenarios[key].probability > highestProb) {
   120→      highestProb = scenarios[key].probability;
   121→      highestKey = key;
   122→    }
   123→  }
   124→  const strategy = STRATEGY_MAP[highestKey];
   125→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   126→
   127→  // ── RSI signal ─────────────────────────────────────────────────
   128→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   129→  const rsiColor = rsi > 70 ? 'text-red-400' : rsi > 60 ? 'text-amber-400' : rsi > 40 ? 'text-gray-300' : rsi > 30 ? 'text-amber-400' : 'text-emerald-400';
   130→
   131→  // ── Stochastic signal ──────────────────────────────────────────
   132→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   133→
   134→  // ── MACD signal ────────────────────────────────────────────────
   135→  const macdSignalText = macdHist > 0 ? 'مومنتوم مثبت (هیستوگرام بالای صفر)' : 'مومنتوم منفی (هیستوگرام زیر صفر)';
   136→  const macdBullish = macdLine > macdSignal;
   137→
   138→  // ── Trend text ─────────────────────────────────────────────────
   139→  const trendText = trendDirection === 'up'
   140→    ? `صعودی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${toFaD(trendR2 * 100)}٪`
   141→    : trendDirection === 'down'
   142→    ? `نزولی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${toFaD(trendR2 * 100)}٪`
   143→    : 'خنثی و بدون جهت مشخص';
   144→
   145→  // ── Bollinger Band position ────────────────────────────────
   146→  const bbRange = bollingerUpper - bollingerLower;
   147→  const bbPos = bbRange > 0 ? toPersianDigits(((currentPrice - bollingerLower) / bbRange * 100).toFixed(0)) : '۵۰';
   148→  const bbSignal = currentPrice > bollingerUpper ? 'بالای باند بالایی (اشباع خرید)'
   149→    : currentPrice < bollingerLower ? 'زیر باند پایینی (اشباع فروش)'
   150→    : `داخل باندها (${bbPos}٪ از بازه)`;
   151→
   152→  // ── TradingView scenario colors ────────────────────────────────
   153→  const chartCandles = useMemo(() => {
   154→    if (candles.length <= 2000) return candles;
   155→    return candles.slice(-2000);
   156→  }, [candles]);
   157→
   158→  const tvScenarios = useMemo(() => {
   159→    const colors: Record<string, string> = { R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62' };
   160→    return Object.fromEntries(
   161→      SCENARIO_KEYS.map(k => [k, { targetMin: scenarios[k].targetMin, targetMax: scenarios[k].targetMax, name: scenarios[k].name, probability: scenarios[k].probability, color: colors[k] }])
   162→    );
   163→  }, [scenarios]);
   164→
   165→  // ── Strategy recommendation text ────────────────────────────────
   166→  const mfiWarning = hasVolume ? ` (مقدار ${toFa(mfi)})` : '';
   167→  const strategyText = rsi > 70 || (hasVolume && mfi > 80)
   168→    ? `با توجه به هشدار اشباع خرید (قدرت نسبی: ${toFa(rsi)}${mfiWarning}) و فاصله قیمت تا مقاومت ${toFa(R1_level)}، استراتژی محتاطانه، انتظار برای اصلاح قیمت و ورود در محدوده حمایت ${toFa(S1_level)} تا ${toFa(S2_level)} ریال می‌باشد.`
   169→    : rsi < 30 || (hasVolume && mfi < 20)
   170→    ? `با توجه به اشباع فروش (قدرت نسبی: ${toFa(rsi)}${mfiWarning}) و نزدیکی به حمایت ${toFa(S1_level)}، فرصت خرید در محدوده فعلی با حد ضرر ${toFa(S2_level)} ریال قابل بررسی است. هدف اولیه ${toFa(R1_level)} و هدف ثانویه ${toFa(R2_level)} ریال.`
   171→    : `با توجه به وضعیت خنثی شاخص‌ها (قدرت نسبی: ${toFa(rsi)}، قدرت روند: ${adx > 25 ? 'قوی' : 'ضعیف'})، انتظار برای خروج قیمت از محدوده ${toFa(S1_level)} تا ${toFa(R1_level)} ریال و سپس تصمیم‌گیری توصیه می‌شود.`;
   172→
   173→  // ═══════════════════════════════════════════════════════════════════
   174→  // Export handlers
   175→  // ═══════════════════════════════════════════════════════════════════
   176→
   177→  // ── Export PDF (full page) ─────────────────────────────────────
   178→  const handleExportPDF = useCallback(async () => {
   179→    const el = containerRef.current;
   180→    if (!el) return;
   181→    setExporting(true);
   182→    try {
   183→      const { toPng } = await import('html-to-image');
   184→      const { jsPDF } = await import('jspdf');
   185→      const dataUrl = await toPng(el, { backgroundColor: '#060a13', pixelRatio: 2, quality: 0.95 });
   186→      const img = new Image();
   187→      img.src = dataUrl;
   188→      await new Promise((resolve) => { img.onload = resolve; });
   189→      const pdfWidth = 595.28;
   190→      const pdfHeight = (img.height * pdfWidth) / img.width;
   191→      const pdf = new jsPDF({ orientation: pdfHeight > pdfWidth ? 'portrait' : 'landscape', unit: 'pt', format: [pdfWidth, Math.min(pdfHeight, 841.89)] });
   192→      pdf.addImage(dataUrl, 'PNG', 0, 0, pdfWidth, pdfHeight);
   193→      pdf.save(`${symbolName}-تحلیل-تکنیکال.pdf`);
   194→    } catch (err) { console.error('PDF export failed:', err); }
   195→    finally { setExporting(false); }
   196→  }, [symbolName]);
   197→
   198→  // ── Export HTML (full page) ────────────────────────────────────
   199→  const handleExportHTML = useCallback(() => {
   200→    const el = containerRef.current;
   201→    if (!el) return;
   202→    setExporting(true);
   203→    try {
   204→      const now = new Date();
   205→      const dateStr = toShamsi(now, 'long');
   206→      const html = `<!DOCTYPE html>
   207→<html lang="fa" dir="rtl">
   208→<head>
   209→<meta charset="UTF-8">
   210→<title>تحلیل تکنیکال ${symbolName}</title>
   211→<style>
   212→  body { font-family: Tahoma, Arial, sans-serif; background: #060a13; color: #e5e7eb; direction: rtl; padding: 24px; line-height: 1.85; }
   213→  h1 { color: #f8e365; font-size: 20px; }
   214→  .badge { display: inline-block; padding: 4px 14px; border-radius: 20px; border: 1px solid rgba(255,255,255,0.05); background: rgba(255,255,255,0.04); font-size: 12px; color: #9ca3af; margin-left: 8px; }
   215→  .badge b { color: #e5e7eb; }
   216→  .section { background: rgba(14,22,40,0.7); border: 1px solid rgba(255,255,255,0.06); border-radius: 14px; padding: 20px; margin: 16px 0; }
   217→  .footer { text-align: center; font-size: 11px; color: #4b5563; margin-top: 16px; padding-top: 12px; border-top: 1px solid rgba(255,255,255,0.05); }
   218→</style>
   219→</head>
   220→<body>
   221→<h1>📈 تحلیل تکنیکال ${symbolName}</h1>
   222→<p style="margin-top:12px">
   223→  <span class="badge">📍 قیمت مرجع: <b>${toFa(currentPrice)} ریال</b></span>
   224→  <span class="badge">🎯 هدف: <b>${toFa(targetMin)} — ${toFa(targetMax)} ریال</b></span>
   225→  <span class="badge">📊 روند: <b>${trendText}</b></span>
   226→  <span class="badge">RSI: <b>${toFa(rsi)} (${rsiSignal})</b></span>
   227→</p>
   228→<div class="section">${el.innerHTML}</div>
   229→<div class="footer">تاریخ تهیه: ${dateStr} — صرفاً جنبه تحلیلی دارد و توصیه سرمایه‌گذاری نیست.</div>
   230→</body>
   231→</html>`;
   232→      const blob = new Blob([html], { type: 'text/html;charset=utf-8' });
   233→      const url = URL.createObjectURL(blob);
   234→      const a = document.createElement('a'); a.href = url; a.download = `${symbolName}-تحلیل-تکنیکال.html`; a.click();
   235→      URL.revokeObjectURL(url);
   236→    } catch (err) { console.error('HTML export failed:', err); }
   237→    finally { setExporting(false); }
   238→  }, [symbolName, currentPrice, targetMin, targetMax, trendText, rsi, rsiSignal]);
   239→
   240→  // ── Export Plain Text ──────────────────────────────────────────
   241→  const handleExportText = useCallback(() => {
   242→    const el = textAnalysisRef.current;
   243→    if (!el) return;
   244→    setExporting(true);
   245→    try {
   246→      const now = new Date();
   247→      const dateStr = toShamsi(now, 'long');
   248→      // Extract text content, clean up whitespace
   249→      const rawText = el.innerText || el.textContent || '';
   250→      const lines = rawText.split('\n').map(l => l.trim()).filter(Boolean);
   251→      const text = `تحلیل تکنیکال ${symbolName}
   252→${'═'.repeat(50)}
   253→تاریخ تهیه: ${dateStr}
   254→
   255→${lines.join('\n')}
   256→
   257→${'─'.repeat(50)}
   258→صرفاً جنبه تحلیلی دارد و توصیه سرمایه‌گذاری نیست.`;
   259→      const blob = new Blob([`\uFEFF${text}`], { type: 'text/plain;charset=utf-8' });
   260→      const url = URL.createObjectURL(blob);
   261→      const a = document.createElement('a'); a.href = url; a.download = `${symbolName}-تحلیل-متنی.txt`; a.click();
   262→      URL.revokeObjectURL(url);
   263→    } catch (err) { console.error('Text export failed:', err); }
   264→    finally { setExporting(false); }
   265→  }, [symbolName]);
   266→
   267→  return (
   268→    <div ref={containerRef} className="space-y-5" dir="rtl">
   269→      {/* ═══ HEADER ═══ */}
   270→      <div className="rounded-[32px] px-6 py-5 border border-amber-500/20"
   271→        style={{ background: 'rgba(18,28,46,0.7)', backdropFilter: 'blur(14px)', boxShadow: '0 25px 50px -12px rgba(0,0,0,0.6)' }}>
   272→        <div className="flex items-center justify-between flex-wrap gap-3">
   273→          <h1 className="text-xl font-bold" style={{ color: '#f8e365' }}>
   274→            📈 تحلیل تکنیکال {symbolName}
   275→          </h1>
   276→          {/* ── Export Buttons ── */}
   277→          <div className="flex items-center gap-2">
   278→            <ExportBtn icon={<Type className="w-3.5 h-3.5" />} label="خروجی متنی" onClick={handleExportText} disabled={exporting} loading={exporting} />
   279→            <ExportBtn icon={<FileText className="w-3.5 h-3.5" />} label="خروجی HTML" onClick={handleExportHTML} disabled={exporting} loading={exporting} />
   280→            <ExportBtn icon={<Download className="w-3.5 h-3.5" />} label="خروجی PDF" onClick={handleExportPDF} disabled={exporting} loading={exporting} />
   281→          </div>
   282→        </div>
   283→        <div className="flex flex-wrap gap-3 mt-3">
   284→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   285→            📍 قیمت مرجع: <b className="text-gray-200">{toFa(currentPrice)} ریال</b>
   286→          </span>
   287→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   288→            🎯 هدف کوتاه‌مدت: <b className="text-gray-200">{toFa(targetMin)} — {toFa(targetMax)} ریال</b>
   289→          </span>
   290→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   291→            📊 روند: <b className="text-gray-200">{trendText}</b>
   292→          </span>
   293→          <span className={`px-3.5 py-1 rounded-full border text-xs font-medium ${
   294→            rsi > 70 ? 'bg-red-500/15 text-red-400 border-red-500/30'
   295→            : rsi < 30 ? 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30'
   296→            : 'bg-amber-500/15 text-amber-400 border-amber-500/30'
   297→          }`}>
   298→            قدرت نسبی: {toFa(rsi)} ({rsiSignal})
   299→          </span>
   300→        </div>
   301→      </div>
   302→
   303→      {/* ═══ TRADINGVIEW CHART ═══ */}
   304→      <div className="rounded-[28px] p-4 border border-white/6"
   305→        style={{ background: 'rgba(14,22,40,0.7)', backdropFilter: 'blur(8px)', boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)' }}>
   306→        <h2 className="text-sm font-semibold text-gray-200 flex items-center gap-3 mb-3">
   307→          <span style={{ color: '#f8e365' }}>📊</span>
   308→          نمودار قیمتی سهم (روزانه) — به همراه MA21، MA100، سطوح حمایت/مقاومت و اهداف قیمتی
   309→        </h2>
   310→        <TradingViewChart
   311→          symbolName={symbolName}
   312→          candles={chartCandles}
   313→          supports={supports}
   314→          resistances={resistances}
   315→          supportStrengths={supportStrengths}
   316→          resistanceStrengths={resistanceStrengths}
   317→          ma21={ma21}
   318→          ma100={ma100}
   319→          scenarios={tvScenarios}
   320→        />
   321→      </div>
   322→
   323→      {/* ═══ SUPPORT / RESISTANCE LEVELS BOXES ═══ */}
   324→      <div className="rounded-[28px] p-5 border border-white/6"
   325→        style={{ background: 'rgba(14,22,40,0.7)', backdropFilter: 'blur(8px)', boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)' }}>
   326→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   327→          {/* ── Resistance Levels ── */}
   328→          <div>
   329→            <div className="flex items-center gap-2 mb-3">
   330→              <ShieldAlert className="w-4 h-4 text-red-400" />
   331→              <h3 className="text-sm font-bold text-red-400">سطوح مقاومت</h3>
   332→            </div>
   333→            <div className="space-y-2">
   334→              {resistanceStrengths.filter(r => r.price > 0).map((r, i) => (
   335→                <LevelBox key={`r-${i}`} index={i} label={`R${i + 1}`} price={r.price} strength={r.strength} isTarget={r.isTarget} type="resistance" />
   336→              ))}
   337→            </div>
   338→          </div>
   339→
   340→          {/* ── Support Levels ── */}
   341→          <div>
   342→            <div className="flex items-center gap-2 mb-3">
   343→              <Shield className="w-4 h-4 text-emerald-400" />
   344→              <h3 className="text-sm font-bold text-emerald-400">سطوح حمایت</h3>
   345→            </div>
   346→            <div className="space-y-2">
   347→              {supportStrengths.filter(s => s.price > 0).map((s, i) => (
   348→                <LevelBox key={`s-${i}`} index={i} label={`S${i + 1}`} price={s.price} strength={s.strength} isTarget={s.isTarget} type="support" />
   349→              ))}
   350→            </div>
   351→          </div>
   352→        </div>

   570→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   571→function LevelBox({ index, label, price, strength, isTarget, type }: {
   572→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target';
   573→}) {
   574→  const isSupport = type === 'support';
   575→  const isTargetType = type === 'target';
   576→
   577→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   578→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   579→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   580→
   581→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   582→  const strengthLabel = strength >= 8 ? 'بسیار قوی' : strength >= 6 ? 'قوی' : strength >= 4 ? 'متوسط' : strength >= 2 ? 'ضعیف' : 'بسیار ضعیف';
   583→  const strengthColor = strength >= 8 ? '#34c98b' : strength >= 6 ? '#3ad5db' : strength >= 4 ? '#ffb11b' : strength >= 2 ? '#ff7b32' : '#ef4d62';
   584→
   585→  // Distance from current price (if available from context)
   586→  return (
   587→    <div
   588→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   589→      style={{
   590→        background: bgAccent,
   591→        border: `1px solid ${borderAccent}`,
   592→        borderRight: `3px solid ${accentColor}`,
   593→      }}
   594→    >
   595→      {/* Label + Price */}
   596→      <div className="flex-1 min-w-0">
   597→        <div className="flex items-center gap-2">
   598→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   599→          {isTarget && <span className="text-[9px]">🎯</span>}
   600→        </div>
   601→        <span className="text-sm font-black text-gray-100 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   602→        <span className="text-[10px] text-gray-500 mr-1">ریال</span>
   603→      </div>
   604→
   605→      {/* Strength bar */}
   606→      <div className="flex flex-col items-end gap-1 min-w-[80px]">
   607→        <div className="flex items-center gap-1.5">
   608→          <span className="text-[9px] font-medium" style={{ color: strengthColor }}>{strengthLabel}</span>
   609→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(strength)}</span>
   610→          <span className="text-[9px] text-gray-500">/۱۰</span>
   611→        </div>
   612→        <div className="w-full h-1.5 rounded-full bg-white/5 overflow-hidden">
   613→          <div
   614→            className="h-full rounded-full transition-all duration-500"
   615→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   616→          />
   617→        </div>
   618→      </div>
   619→    </div>
   620→  );
   621→}
