    38→// ─── 429 Retry ──────────────────────────────────────────────────────
    39→async function withRetry<T>(fn: () => Promise<T>, label: string = 'V9'): Promise<T> {
    40→  const delays = [20000, 40000, 80000, 160000];
    41→  for (let attempt = 0; attempt <= delays.length; attempt++) {
    42→    try {
    43→      return await fn();
    44→    } catch (err: any) {
    45→      const status = err?.status || err?.statusCode;
    46→      if (status === 429 && attempt < delays.length) {
    47→        const jitter = Math.random() * 500;
    48→        const wait = delays[attempt] + jitter;
    49→        console.warn(`[VDES v9 429] ${label} Retry ${attempt + 1}/${delays.length} after ${Math.round(wait / 1000)}s`);
    50→        await new Promise(r => setTimeout(r, wait));
    51→        continue;
    52→      }
    53→      throw err;
    54→    }
    55→  }
    56→  throw new Error('unreachable');
    57→}

   133→    diPlus, diMinus, sar, atr, obv, hasVolume,
   134→    bollingerUpper, bollingerMiddle, bollingerLower,
   135→    trendDirection, trendAngle, trendR2, overallSignal,
   136→    scenarios, resistances, supports,
   137→    resistanceStrengths, supportStrengths,
   138→    isIndex, isTgju, currencyUnit,
   139→  } = body;
   140→
   141→  const unit = currencyUnit || 'ریال';
   142→  const label = isIndex ? `\u0634\u0627\u062E\u0635 ${symbolName}` : isTgju ? symbolName : `\u0633\u0647\u0645 ${symbolName}`;
   143→  const trendLabel = trendDirection === 'up' ? 'صعودی' : trendDirection === 'down' ? 'نزولی' : 'خنثی';
   144→  const r2Pct = (trendR2 * 100).toFixed(1);
   145→  const adxStrength = adx > 40 ? 'بسیار قوی' : adx > 25 ? 'قوی' : adx > 15 ? 'متوسط' : 'ضعیف';
   146→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   147→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   148→
   149→  const bbRange = bollingerUpper - bollingerLower;
   150→  const bbPos = bbRange > 0 ? Math.round((currentPrice - bollingerLower) / bbRange * 100) : 50;
   151→  const bbSignal = currentPrice > bollingerUpper
   152→    ? 'بالای باند بالایی'
   153→    : currentPrice < bollingerLower
   154→    ? 'زیر باند پایینی'
   155→    : `داخل باندها (${bbPos}٪)`;
   156→
   157→  const R1 = resistances[0] ?? Math.round(currentPrice * 1.05);
   158→  const R2 = resistances[1] ?? Math.round(currentPrice * 1.10);
   159→  const R3 = resistances[2] ?? Math.round(currentPrice * 1.15);
   160→  const S1 = supports[0] ?? Math.round(currentPrice * 0.95);
   161→  const S2 = supports[1] ?? Math.round(currentPrice * 0.90);
   162→  const S3 = supports[2] ?? Math.round(currentPrice * 0.85);
   163→
   164→  const r1Grade = resistanceStrengths?.[0]?.grade || (resistanceStrengths?.[0]?.strength ? srGrade(resistanceStrengths[0].strength) : 'نامشخص');
   165→  const s1Grade = supportStrengths?.[0]?.grade || (supportStrengths?.[0]?.strength ? srGrade(supportStrengths[0].strength) : 'نامشخص');

   167→  const s1Confirm = supportStrengths?.[0]?.overlapCount ? `(${supportStrengths[0].overlapCount} روش تأیید)` : '';
   168→
   169→  const overallLabel = overallSignal === 'bullish' ? 'صعودی' : overallSignal === 'bearish' ? 'نزولی' : 'خنثی';
   170→  const posVsMa21 = currentPrice > ma21 ? 'بالاتر' : 'پایین‌تر';
   171→  const posVsMa100 = currentPrice > ma100 ? 'بالاتر' : 'پایین‌تر';
   172→
   173→  const obvDesc = hasVolume
   174→    ? (obv > 0
   175→      ? `مثبت (OBV بالاست و نشان‌دهنده ورود پول هوشمند است)`

   182→    ? 'صعودی اما هیستوگرام در حال تضعیف'
   183→    : 'نزولی';
   184→
   185→  const cciDesc = cci > 100 ? 'بالاتر از +۱۰۰ (اشباع خرید)' : cci < -100 ? 'پایین‌تر از -۱۰۰ (اشباع فروش)' : 'در محدوده نرمال';

   224→    : '- الگوی هارمونیک قابل اعتمادی شناسایی نشد.'}
   225→${v9Extra.aiPatterns.elliottWave

   232→**داده‌های پایه (واقعی و غیرقابل تغییر):**
   233→- نام ابزار: **${label}**
   234→- قیمت مرجع: **${fmtGrouped(currentPrice)} ${unit}**

   364→// ─── V9 Parallel Data Gathering ─────────────────────────────────────
   365→async function gatherV9Data(body: VdesRequest): Promise<V9ExtraData> {
   366→  const result: V9ExtraData = {
   367→    candlePatterns: { patterns: [], dominantDirection: 'neutral', dominantScore: 0, summary: '' },
   368→    mlPrediction: null,
   369→    mlTrainingStatus: 'untrained',
   370→    bayesianSummary: '',
   371→    aiPatterns: null,
   372→    mlServiceAvailable: false,
   373→  };
   374→
   375→  // 1. Candlestick pattern detection (always available, local computation)
   376→  if (body.ohlcvHistory && body.ohlcvHistory.length >= 3) {
   377→    result.candlePatterns = scanCandlestickPatterns(body.ohlcvHistory as any, 20);
   378→  }
   379→
   380→  // 2. Bayesian weighting (always available, local computation)
   381→  result.bayesianSummary = generateBayesianSummary(body.symbolName);
   382→
   383→  // 3. ML prediction & AI patterns (requires ML service)
   384→  try {
   385→    const health = await checkMLHealth();
   386→    result.mlServiceAvailable = health.status === 'ok';
   387→  } catch {
   388→    result.mlServiceAvailable = false;
   389→  }
   390→
   391→  if (result.mlServiceAvailable && body.ohlcvHistory && body.ohlcvHistory.length >= 120) {
   392→    const symbol = body.symbolName;
   393→
   394→    // 3a. ML Prediction (with timeout fallback)
   395→    try {
   396→      const mlOHLCV = convertToMLOHLCV(body.ohlcvHistory);
   397→      const predResp = await Promise.race([
   398→        predict30Sessions(symbol, mlOHLCV, 30),
   399→        new Promise<import('@/lib/ml-predictor').MLPredictionResponse>((_, reject) =>
   400→          setTimeout(() => reject(new Error('ML prediction timeout')), 15000)
   401→        ),
   402→      ]);
   403→
   404→      if (predResp.status === 'ok' && predResp.prediction) {
   405→        result.mlPrediction = predResp.prediction;
   406→        result.mlTrainingStatus = 'trained';
   407→      } else {
   408→        // No model trained yet — try to train in background
   409→        console.log(`[V9 ML] No model for ${symbol}, attempting background training...`);
   410→        trainModelInBackground(symbol, convertToMLOHLCV(body.ohlcvHistory));
   411→      }
   412→    } catch (err) {
   413→      console.warn(`[V9 ML] Prediction failed for ${symbol}:`, err);
   414→    }
   415→
   416→    // 3b. AI Harmonic/Elliott patterns (via ZAI LLM, with timeout)
   417→    if (body.ohlcvHistory.length >= 50) {
   418→      try {
   419→        result.aiPatterns = await detectAIPatterns(body);
   420→      } catch (err) {
   421→        console.warn(`[V9 AI] Pattern detection failed:`, err);
   422→      }
   423→    }
   424→  } else if (body.ohlcvHistory && body.ohlcvHistory.length >= 60 && body.ohlcvHistory.length < 120) {
   425→    result.bayesianSummary += ' (برای فعال‌سازی پیش‌بینی ML به حداقل ۱۲۰ کندل داده نیاز است.)';
   426→  }
   427→
   428→  return result;
   429→}

   431→/** Background training (fire-and-forget) */
   432→function trainModelInBackground(symbol: string, ohlcv: import('@/lib/ml-predictor').OHLCVRow[]) {
   433→  // Fire in background, don't block the response
   434→  trainModel(symbol, ohlcv).then(resp => {
   435→    if (resp.status === 'ok') {
   436→      console.log(`[V9 ML] Background training complete for ${symbol}: ${resp.models_trained?.join(', ')}`);
   437→    } else {
   438→      console.warn(`[V9 ML] Background training failed for ${symbol}: ${resp.message}`);
   439→    }
   440→  }).catch(err => {
   441→    console.warn(`[V9 ML] Background training error for ${symbol}:`, err);
   442→  });
   443→}

   445→/** Detect AI patterns (harmonic + Elliott) via ZAI LLM */
   446→async function detectAIPatterns(
   447→  body: VdesRequest
   448→): Promise<import('@/lib/candlestick-patterns').AIPatternResult | null> {
   449→  if (!body.ohlcvHistory || body.ohlcvHistory.length < 50) return null;
   450→
   451→  const prompt = buildAIPatternPrompt({
   452→    ohlcv: body.ohlcvHistory as any,
   453→    currentPrice: body.currentPrice,
   454→    symbolName: body.symbolName,
   455→  });
   456→
   457→  try {
   458→    const zai = await getZAI();
   459→    const completion = await withRetry(() =>
   460→      zai.chat.completions.create({
   461→        messages: [
   462→          {
   463→            role: 'assistant',
   464→            content: 'شما یک تحلیلگر الگوهای هارمونیک و موج الیوت هستید. فقط JSON خالص پاسخ دهید. هیچ متن اضافی ننویسید.',
   465→          },
   466→          { role: 'user', content: prompt },
   467→        ],
   468→        thinking: { type: 'disabled' },
   469→      }),
   470→      'AI-Patterns'
   471→    );
   472→
   473→    const text = completion.choices[0]?.message?.content;
   474→    if (!text) return null;
   475→
   476→    // Extract JSON from response
   477→    const jsonMatch = text.match(/\{[\s\S]*\}/);
   478→    if (!jsonMatch) return null;
   479→
   480→    const parsed = JSON.parse(jsonMatch[0]);
   481→    return parsed as import('@/lib/candlestick-patterns').AIPatternResult;
   482→  } catch (err) {
   483→    console.warn('[V9 AI-Patterns] Parse/LLM error:', err);
   484→    return null;
   485→  }
   486→}

   488→// ─── POST Handler ─────────────────────────────────────────────────────
   489→export async function POST(req: NextRequest) {
   490→  const t0 = Date.now();
   491→
   492→  try {
   493→    const body: VdesRequest = await req.json();
   494→
   495→    // Validate required fields
   496→    if (!body.symbolName || body.currentPrice == null) {
   497→      return NextResponse.json({ error: 'symbolName و currentPrice الزامی است.' }, { status: 400 });
   498→    }
   499→    if (!body.scenarios || !Array.isArray(body.resistances) || !Array.isArray(body.supports)) {
   500→      return NextResponse.json({ error: 'scenarios، resistances و supports الزامی است.' }, { status: 400 });
   501→    }
   502→
   503→    // ── Step 1: Narrative selection (v8 engine, reused) ──
   504→    const narrativeInput = buildNarrativeInput(body);
   505→    const combo = selectNarrativeCombination(narrativeInput);
   506→
   507→    console.log(`[VDES v9] ${body.symbolName} | School: ${combo.school.nameEn} (${combo.schoolScore.toFixed(1)}) | Style: ${combo.style.nameEn} (${combo.styleScore.toFixed(1)}) | Tone: ${combo.tone.nameEn} (${combo.toneScore.toFixed(1)})`);
   508→
   509→    // ── Step 2: Gather V9 data (parallel: candlestick, ML, Bayesian, AI patterns) ──
   510→    const v9Extra = await gatherV9Data(body);
   511→
   512→    console.log(`[VDES v9] ${body.symbolName} | Candlestick: ${v9Extra.candlePatterns.patterns.length} patterns | ML: ${v9Extra.mlTrainingStatus} | AI-Patterns: ${v9Extra.aiPatterns ? 'detected' : 'skipped'} | (${Math.round(Date.now() - t0)}ms gathered)`);
   513→
   514→    // ── Step 3: Build prompt ──
   515→    const prompt = buildV9Prompt(body, combo, v9Extra);
   516→
   517→    // ── Step 4: Check cache ──
   518→    const key = cacheKey(body);
   519→    const cached = cache.get(key);
   520→    if (cached && cached.v === ANALYSIS_VERSION && (Date.now() - cached.ts) < CACHE_TTL_MS) {
   521→      console.log(`[VDES v9] Cache hit for ${body.symbolName} (${Math.round(Date.now() - t0)}ms)`);
   522→      return NextResponse.json({
   523→        analysis: cached.text,
   524→        _debug: {
   525→          ...cached.debug,
   526→          v9: {
   527→            candlePatterns: v9Extra.candlePatterns,
   528→            mlTrainingStatus: v9Extra.mlTrainingStatus,
   529→            mlServiceAvailable: v9Extra.mlServiceAvailable,
   530→            mlPrediction: v9Extra.mlPrediction ? {
   531→              overall_direction: v9Extra.mlPrediction.overall_direction,
   532→              overall_confidence: v9Extra.mlPrediction.overall_confidence,
   533→              target_price_min: v9Extra.mlPrediction.target_price_min,
   534→              target_price_max: v9Extra.mlPrediction.target_price_max,
   535→              risk_level: v9Extra.mlPrediction.risk_level,
   536→            } : null,
   537→            aiPatterns: v9Extra.aiPatterns,
   538→            bayesianSummary: v9Extra.bayesianSummary,
   539→          },
   540→          cached: true,
   541→          latencyMs: Date.now() - t0,
   542→        },
   543→      });
   544→    }
   545→
   546→    // ── Step 5: Call ZAI LLM ──
   547→    const zai = await getZAI();
   548→    const completion = await withRetry(() =>
   549→      zai.chat.completions.create({
   550→        messages: [
   551→          { role: 'assistant', content: SYSTEM_PROMPT },
   552→          { role: 'user', content: prompt },
   553→        ],
   554→        thinking: { type: 'disabled' },
   555→      })
   556→    );
   557→
   558→    const analysis = completion.choices[0]?.message?.content;
   559→
   560→    if (!analysis || analysis.trim().length === 0) {
   561→      return NextResponse.json({ error: 'مدل پاسخی تولید نکرد.' }, { status: 500 });
   562→    }
   563→
   564→    // ── Step 6: Cache & respond ──
   565→    const debugData = {
   566→      school: combo.school.name,
   567→      style: combo.style.name,
   568→      tone: combo.tone.name,
   569→      schoolScore: combo.schoolScore,
   570→      styleScore: combo.styleScore,
   571→      toneScore: combo.toneScore,
   572→      v9: {
   573→        candlePatterns: v9Extra.candlePatterns,
   574→        mlTrainingStatus: v9Extra.mlTrainingStatus,
   575→        mlServiceAvailable: v9Extra.mlServiceAvailable,
   576→        mlPrediction: v9Extra.mlPrediction ? {
   577→          overall_direction: v9Extra.mlPrediction.overall_direction,
   578→          overall_confidence: v9Extra.mlPrediction.overall_confidence,
   579→          target_price_min: v9Extra.mlPrediction.target_price_min,
   580→          target_price_max: v9Extra.mlPrediction.target_price_max,
   581→          risk_level: v9Extra.mlPrediction.risk_level,
   582→        } : null,
   583→        aiPatterns: v9Extra.aiPatterns,
   584→        bayesianSummary: v9Extra.bayesianSummary,
   585→      },
   586→    };
   587→
   588→    cache.set(key, { v: ANALYSIS_VERSION, text: analysis, ts: Date.now(), debug: debugData });
   589→
   590→    // Evict stale entries
   591→    for (const [k, v] of cache) {
   592→      if ((Date.now() - v.ts) > CACHE_TTL_MS) cache.delete(k);
   593→    }
   594→
   595→    console.log(`[VDES v9] Generated for ${body.symbolName} (${analysis.length} chars, ${Math.round(Date.now() - t0)}ms)`);
   596→
   597→    return NextResponse.json({
   598→      analysis,
   599→      _debug: debugData,
   600→    });
   601→  } catch (err) {
   602→    console.error('[VDES v9 Error]:', err);
   603→    return NextResponse.json(
   604→      { error: 'خطا در تولید تحلیل هوشمند v9. لطفاً دوباره تلاش کنید.' },
   605→      { status: 500 },
   606→    );
   607→  }
   608→}
