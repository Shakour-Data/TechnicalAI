    27→    27→// --- Global rate limiter (serial queue) ---
    28→    28→let lastAITime = 0;
    29→    29→const AI_MIN_INTERVAL = 20_000;
    30→    30→let cooldownUntil = 0;
    31→    31→let activeCall = false;
    32→    32→const pendingQueue: Array<{
    33→    33→  resolve: (text: string) => void;

    38→    38→function processAIQueue() {
    39→    39→  if (activeCall || pendingQueue.length === 0) return;
    40→    40→  const now = Date.now();
    41→    41→
    42→    42→  if (cooldownUntil > now) {
    43→    43→    setTimeout(processAIQueue, Math.min(cooldownUntil - now, 5000));
    44→    44→    return;
    45→    45→  }
    46→    46→
    47→    47→  const elapsed = now - lastAITime;
    48→    48→  if (elapsed < AI_MIN_INTERVAL) {
    49→    49→    setTimeout(processAIQueue, AI_MIN_INTERVAL - elapsed);
    50→    50→    return;
    51→    51→  }
    52→    52→
    53→    53→  const item = pendingQueue.shift()!;
    54→    54→  activeCall = true;
    55→    55→
    56→    56→  callZaiSDK(item.messages)
    57→    57→    .then(text => {
    58→    58→      lastAITime = Date.now();
    59→    59→      cooldownUntil = 0;
    60→    60→      item.resolve(text);
    61→    61→    })
    62→    62→    .catch(err => {
    63→    63→      const msg = err.message || '';
    64→    64→      if (msg.includes('429')) {
    65→    65→        const cd = Math.max(cooldownUntil - Date.now(), 0) + 90_000;
    66→    66→        cooldownUntil = Date.now() + Math.min(cd, 300_000);
    67→    67→        console.warn(`[AI v5] 429, cooldown ${Math.min(cd, 300_000) / 1000}s`);
    68→    68→      }
    69→    69→      item.reject(err);
    70→    70→    })
    71→    71→    .finally(() => {
    72→    72→      activeCall = false;
    73→    73→      setTimeout(processAIQueue, 2000);
    74→    74→    });
    75→    75→}

    77→    77→function queueAI(messages: { role: string; content: string }[], timeoutMs = 90_000): Promise<string> {
    78→    78→  return new Promise((resolve, reject) => {
    79→    79→    const timer = setTimeout(() => {
    80→    80→      const idx = pendingQueue.findIndex(q => q.messages === messages);
    81→    81→      if (idx >= 0) pendingQueue.splice(idx, 1);
    82→    82→      reject(new Error('AI request timeout'));
    83→    83→    }, timeoutMs);
    84→    84→
    85→    85→    pendingQueue.push({
    86→    86→      messages,
    87→    87→      resolve: (text) => { clearTimeout(timer); resolve(text); },
    88→    88→      reject: (err) => { clearTimeout(timer); reject(err); },
    89→    89→    });
    90→    90→
    91→    91→    processAIQueue();
    92→    92→  });
    93→    93→}

    96→    96→function sleep(ms: number) { return new Promise(r => setTimeout(r, ms)); }
    97→    97→function jitter(base: number): number { return base + Math.random() * base * 0.5; }

    99→    99→async function callZaiSDK(messages: { role: string; content: string }[], maxRetries = 4): Promise<string> {
   100→   100→  for (let attempt = 1; attempt <= maxRetries; attempt++) {
   101→   101→    try {
   102→   102→      const zai = await getZai();
   103→   103→      const completion = await zai.chat.completions.create({
   104→   104→        messages,
   105→   105→        thinking: { type: 'disabled' },
   106→   106→      });
   107→   107→      const raw = completion.choices[0]?.message?.content;
   108→   108→      if (!raw || raw.trim().length === 0) {
   109→   109→        throw new Error('Empty AI response');
   110→   110→      }
   111→   111→      const text = raw.trim();
   112→   112→      if (text.length < 10) {
   113→   113→        throw new Error('AI response too short: ' + text.slice(0, 100));
   114→   114→      }
   115→   115→      return text;
   116→   116→    } catch (err: unknown) {
   117→   117→      const msg = err instanceof Error ? err.message : String(err);
   118→   118→      if (msg.includes('429') && attempt < maxRetries) {
   119→   119→        const waitMs = Math.round(jitter(20_000 * Math.pow(2, attempt - 1))); // 20s, 40s, 80s, 160s + jitter
   120→   120→        console.warn(`[AI v5] 429 retry ${attempt}/${maxRetries}, waiting ${Math.round(waitMs / 1000)}s...`);
   121→   121→        await sleep(waitMs);
   122→   122→        continue;
   123→   123→      }
   124→   124→      throw err;
   125→   125→    }
   126→   126→  }
   127→   127→  throw new Error('Max retries exceeded');
   128→   128→}

   193→   193→  const dominantDirection: MSLContext['dominantDirection'] =
   194→   194→    trendDir === 'up' ? 'bullish' : trendDir === 'down' ? 'bearish' : 'neutral';
   195→   195→

   218→   218→      adx,
   219→   219→      cci: (body.cci as number) || 0,
   220→   220→      mfi: (body.mfi as number) || 0,
   221→   221→      atr,
   222→   222→      stochK: (body.stochK as number) || 50,
   223→   223→      stochD: (body.stochD as number) || 50,
   224→   224→      diPlus: (body.diPlus as number) || 0,
   225→   225→      diMinus: (body.diMinus as number) || 0,
   226→   226→      sar: (body.sar as number) || 0,
   227→   227→      bbWidth,
   228→   228→    },
