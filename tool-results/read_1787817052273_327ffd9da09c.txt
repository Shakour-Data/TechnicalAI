   257→  // Store last fetch params for auto-refresh
   258→  const lastFetchRef = useRef<{ symbol: string; category?: string; insCode?: string; tgjuKey?: string; finpySector?: string; finpyIndex?: string; webId?: number; yahooSymbol?: string } | null>(null);
   259→  const refreshTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   260→  const fetchControllerRef = useRef<AbortController | null>(null);
   261→
   262→  // Chart container ref for image export
   263→  const chartWrapperRef = useRef<HTMLDivElement>(null);
   264→
   265→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number, yahooSymbol?: string) => {
   266→    // Abort any previous in-flight fetch
   267→    if (fetchControllerRef.current) {
   268→      fetchControllerRef.current.abort();
   269→    }
   270→    const controller = new AbortController();
   271→    fetchControllerRef.current = controller;
   272→
   273→    // Yahoo Finance instrument: fetch historical data via Yahoo Finance API
   274→    // This handles ALL Yahoo instruments (stocks, indices, energy, metals, forex, crypto, ETFs, etc.)
   275→    if (yahooSymbol) {
   276→      setLoading(true);
   277→      setLoadingMessage('در حال دریافت داده‌های تاریخی از یاهو فایننس ...');
   278→      setError(null);
   279→      setData(null);
   280→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId, yahooSymbol };
   281→      try {
   282→        const res = await fetch('/api/yahoo-analysis?symbol=' + encodeURIComponent(yahooSymbol), { signal: controller.signal });
   283→        if (controller.signal.aborted) return;
   284→        if (!res.ok) {
   285→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   286→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   287→          throw new Error(errMsg);
   288→        }
   289→        const json = await res.json();
   290→        if (controller.signal.aborted) return;
   291→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   292→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   293→        }
   294→        setData({ ...json, isYahoo: true });
   295→        setActivePanel('visual');
   296→      } catch (err) {
   297→        if (controller.signal.aborted) return;
   298→        setError(String(err));
   299→      } finally {
   300→        if (!controller.signal.aborted) {
   301→          setLoading(false);
   302→          setLoadingMessage(null);
   303→        }
   304→      }
   305→      return;
   306→    }
   307→
   308→    // TGJU instrument: fetch historical data via tgju.org chart API
   309→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   310→      setLoading(true);
   311→      setLoadingMessage('در حال دریافت داده‌های تاریخی ...');
   312→      setError(null);
   313→      setData(null);
   314→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId, yahooSymbol };
   315→      try {
   316→        const res = await fetch('/api/tgju-analysis?key=' + encodeURIComponent(tgjuKey), { signal: controller.signal });
