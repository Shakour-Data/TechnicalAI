    30→    30→function renderAIText(text: string): string {
    31→    31→  // Normalize zero-width and look-alike characters that may interfere with regex
    32→    32→  const normalized = text.replace(/[\u200c\u200d\u200b\ufeff]/g, '');
    33→    33→
    34→    34→  // Color syntax: {color:hex}text{/color} or {color:named}text{/color} → <span>
    35→    35→  const colorMap: Record<string, string> = {
    36→    36→    'red': '#dc2626', 'red-600': '#dc2626', 'red-700': '#b91c1c',
    37→    37→    'green': '#16a34a', 'emerald': '#059669', 'emerald-600': '#059669', 'emerald-700': '#047857',
    38→    38→    'amber': '#d97706', 'amber-600': '#d97706', 'amber-700': '#b45309', 'amber-800': '#92400e',
    39→    39→    'blue': '#2563eb', 'blue-600': '#2563eb', 'blue-700': '#1d4ed8',
    40→    40→    'orange': '#ea580c',
    41→    41→    'purple': '#9333ea', 'purple-600': '#9333ea', 'purple-700': '#7e22ce',
    42→    42→  };
    43→    43→
    44→    44→  const withColors = normalized.replace(/\{color:([^}]+)\}([\s\S]*?)\{\/color\}/g, (_match, colorName: string, inner: string) => {
    45→    45→    const hex = colorMap[colorName.trim()] || (colorName.startsWith('#') ? colorName : null);
    46→    46→    if (!hex) return inner; // Unknown color: just return text without color
    47→    47→    return `<span style="color:${hex}">${inner}</span>`;
    48→    48→  });
    49→    49→
    50→    50→  return withColors
    51→    51→    .split('\n\n')
    52→    52→    .map(p => {
    53→    53→      // Convert **bold** to <strong>
    54→    54→      const html = p.replace(/\*\*(.+?)\*\*/g, '<strong class="font-bold text-gray-900">$1</strong>');
    55→    55→      // Convert newlines within paragraph to <br>
    56→    56→      return `<p class="mb-4">${html.replace(/\n/g, '<br>')}</p>`;
    57→    57→    })
    58→    58→    .join('');
    59→    59→}
