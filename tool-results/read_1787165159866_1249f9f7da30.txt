   131→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   132→function sma(closes: number[], period: number): number {
   133→  if (closes.length < period) return 0;
   134→  const slice = closes.slice(-period);
   135→  return slice.reduce((a, b) => a + b, 0) / period;
   136→}

   138→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   139→function emaCalc(closes: number[], period: number): number {
   140→  if (closes.length < period) return 0;
   141→  const k = 2 / (period + 1);
   142→  let emaVal = sma(closes.slice(0, period), period);
   143→  for (let i = period; i < closes.length; i++) {
   144→    emaVal = closes[i] * k + emaVal * (1 - k);
   145→  }
   146→  return emaVal;
   147→}

   149→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   150→function calcRSI(closes: number[], period: number = 14): number {
   151→  if (closes.length < period + 1) return 50;
   152→  let avgGain = 0;
   153→  let avgLoss = 0;
   154→  for (let i = 1; i <= period; i++) {
   155→    const diff = closes[i] - closes[i - 1];
   156→    if (diff > 0) avgGain += diff;
   157→    else avgLoss += Math.abs(diff);
   158→  }
   159→  avgGain /= period;
   160→  avgLoss /= period;
   161→  for (let i = period + 1; i < closes.length; i++) {
   162→    const diff = closes[i] - closes[i - 1];
   163→    const gain = diff > 0 ? diff : 0;
   164→    const loss = diff < 0 ? Math.abs(diff) : 0;
   165→    avgGain = (avgGain * (period - 1) + gain) / period;
   166→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   167→  }
   168→  if (avgLoss === 0) return 100;
   169→  const rs = avgGain / avgLoss;
   170→  return 100 - 100 / (1 + rs);
   171→}

   173→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   174→function calcMFI(data: OHLCV[], period: number = 14): number {
   175→  if (data.length < period + 1) return 50;
   176→  let posFlow = 0;
   177→  let negFlow = 0;
   178→  for (let i = data.length - period; i < data.length; i++) {
   179→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   180→    const mf = tp * data[i].volume;
   181→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   182→    if (tp > prevTp) posFlow += mf;
   183→    else negFlow += mf;
   184→  }
   185→  if (negFlow === 0) return 100;
   186→  const mfr = posFlow / negFlow;
   187→  return 100 - 100 / (1 + mfr);
   188→}

   190→// ─── Helper: CCI(20) ──────────────────────────────────────────────────────────
   191→function calcCCI(data: OHLCV[], period: number = 20): number {
   192→  if (data.length < period) return 0;
   193→  const tps: number[] = [];
   194→  for (let i = data.length - period; i < data.length; i++) {
   195→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   196→  }
   197→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   198→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   199→  if (meanDev === 0) return 0;
   200→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   201→}

   203→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   204→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   205→  if (data.length < kPeriod) return { k: 50, d: 50 };
   206→  const rawKs: number[] = [];
   207→  for (let i = kPeriod - 1; i < data.length; i++) {
   208→    let lowest = Infinity;
   209→    let highest = -Infinity;
   210→    for (let j = i - kPeriod + 1; j <= i; j++) {
   211→      if (data[j].low < lowest) lowest = data[j].low;
   212→      if (data[j].high > highest) highest = data[j].high;
   213→    }
   214→    const range = highest - lowest;
   215→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   216→  }
   217→  const smoothedK: number[] = [];
   218→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   219→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   220→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   221→  }
   222→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   223→  const dSlice = smoothedK.slice(-smoothD);
   224→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   225→  return { k, d };
   226→}

   228→// ─── Helper: Williams %R(14) ──────────────────────────────────────────────────
   229→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   230→  if (data.length < period) return -50;
   231→  const slice = data.slice(-period);
   232→  let highest = -Infinity;
   233→  let lowest = Infinity;
   234→  for (const c of slice) {
   235→    if (c.high > highest) highest = c.high;
   236→    if (c.low < lowest) lowest = c.low;
   237→  }
   238→  const range = highest - lowest;
   239→  if (range === 0) return -50;
   240→  return ((highest - data[data.length - 1].close) / range) * -100;
   241→}

   243→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   244→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   245→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   246→  const emaFast = emaCalc(closes, fast);
   247→  const emaSlow = emaCalc(closes, slow);
   248→  const macdLine = emaFast - emaSlow;
   249→  // Calculate signal as EMA of MACD values
   250→  const macdValues: number[] = [];
   251→  const kFast = 2 / (fast + 1);
   252→  const kSlow = 2 / (slow + 1);
   253→  let eFast = sma(closes.slice(0, fast), fast);
   254→  let eSlow = sma(closes.slice(0, slow), slow);
   255→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   256→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   257→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   258→    macdValues.push(eFast - eSlow);
   259→  }
   260→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   261→  const kSig = 2 / (sig + 1);
   262→  let signalVal = sma(macdValues.slice(0, sig), sig);
   263→  for (let i = sig; i < macdValues.length; i++) {
   264→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   265→  }
   266→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   267→}

   269→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   270→function calcATR(data: OHLCV[], period: number = 14): number {
   271→  if (data.length < 2) return 0;
   272→  const trs: number[] = [];
   273→  for (let i = 1; i < data.length; i++) {
   274→    const tr = Math.max(
   275→      data[i].high - data[i].low,
   276→      Math.abs(data[i].high - data[i - 1].close),
   277→      Math.abs(data[i].low - data[i - 1].close)
   278→    );
   279→    trs.push(tr);
   280→  }
   281→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   282→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   283→  for (let i = period; i < trs.length; i++) {
   284→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   285→  }
   286→  return atrVal;
   287→}

   289→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   290→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   291→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   292→  const plusDMs: number[] = [];
   293→  const minusDMs: number[] = [];
   294→  const trs: number[] = [];
   295→  for (let i = 1; i < data.length; i++) {
   296→    const upMove = data[i].high - data[i - 1].high;
   297→    const downMove = data[i - 1].low - data[i].low;
   298→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   299→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   300→    trs.push(Math.max(
   301→      data[i].high - data[i].low,
   302→      Math.abs(data[i].high - data[i - 1].close),
   303→      Math.abs(data[i].low - data[i - 1].close)
   304→    ));
   305→  }
   306→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   307→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   308→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   309→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   310→  const dxValues: number[] = [];
   311→  for (let i = period; i < trs.length; i++) {
   312→    smoothTR = smoothTR - smoothTR / period + trs[i];
   313→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   314→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   315→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   316→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   317→    const diSum = diPlus + diMinus;
   318→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   319→    dxValues.push(dx);
   320→  }
   321→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   322→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   323→  for (let i = period; i < dxValues.length; i++) {
   324→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   325→  }
   326→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   327→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   328→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   329→}

   331→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   332→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   333→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   334→  let isLong = true;
   335→  let af = step;
   336→  let ep = data[0].high;
   337→  let sar = data[0].low;
   338→  for (let i = 1; i < data.length; i++) {
   339→    const prevSar = sar;
   340→    sar = prevSar + af * (ep - prevSar);
   341→    // Ensure SAR is within bounds
   342→    if (isLong) {
   343→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   344→      if (data[i].low < sar) {
   345→        isLong = false;
   346→        sar = ep;
   347→        ep = data[i].low;
   348→        af = step;
   349→      } else {
   350→        if (data[i].high > ep) {
   351→          ep = data[i].high;
   352→          af = Math.min(af + step, max);
   353→        }
   354→      }
   355→    } else {
   356→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   357→      if (data[i].high > sar) {
   358→        isLong = true;
   359→        sar = ep;
   360→        ep = data[i].high;
   361→        af = step;
   362→      } else {
   363→        if (data[i].low < ep) {
   364→          ep = data[i].low;
   365→          af = Math.min(af + step, max);
   366→        }
   367→      }
   368→    }
   369→  }
   370→  return sar;
   371→}

   373→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   374→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   375→  if (closes.length < period) {
   376→    const c = closes[closes.length - 1] ?? 0;
   377→    return { upper: c, middle: c, lower: c };
   378→  }
   379→  const slice = closes.slice(-period);
   380→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   381→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   382→  const sd = Math.sqrt(variance);
   383→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   384→}

   386→// ─── Helper: OBV ──────────────────────────────────────────────────────────────
   387→function calcOBV(data: OHLCV[]): number {
   388→  let obv = 0;
   389→  for (let i = 1; i < data.length; i++) {
   390→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   391→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   392→  }
   393→  return obv;
   394→}

   396→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   397→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   398→  const n = values.length;
   399→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   400→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   401→  for (let i = 0; i < n; i++) {
   402→    sumX += i;
   403→    sumY += values[i];
   404→    sumXY += i * values[i];
   405→    sumX2 += i * i;
   406→    sumY2 += values[i] * values[i];
   407→  }
   408→  const denom = n * sumX2 - sumX * sumX;
   409→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   410→  const slope = (n * sumXY - sumX * sumY) / denom;
   411→  const intercept = (sumY - slope * sumX) / n;
   412→  const meanY = sumY / n;
   413→  let ssRes = 0, ssTot = 0;
   414→  for (let i = 0; i < n; i++) {
   415→    const predicted = slope * i + intercept;
   416→    ssRes += (values[i] - predicted) ** 2;
   417→    ssTot += (values[i] - meanY) ** 2;
   418→  }
   419→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   420→  return { slope, intercept, r2 };
   421→}

   423→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   424→function roundToNice(price: number): number {
   425→  const abs = Math.abs(price);
   426→  let step: number;
   427→  if (abs >= 100000) step = 100;
   428→  else if (abs >= 10000) step = 50;
   429→  else if (abs >= 1000) step = 10;
   430→  else if (abs >= 100) step = 5;
   431→  else step = 1;
   432→  return Math.round(price / step) * step;
   433→}

   476→// ─── Trend Analysis ───────────────────────────────────────────────────────────
   477→function calcTrend(closes: number[], period: number): TrendResult {
   478→  const slice = closes.slice(-Math.min(period, closes.length));
   479→  const { slope, r2 } = linearRegression(slice);
   480→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   481→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   482→  const priceThreshold = avgPrice * 0.001; // 0.1%
   483→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   484→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   485→}

   487→// ═══════════════════════════════════════════════════════════════════════════════
   488→// MAIN ANALYSIS FUNCTION
   489→// ═══════════════════════════════════════════════════════════════════════════════
   490→export async function analyze(data: OHLCV[], symbol?: string): Promise<TAResult> {
   491→  if (!data || data.length < 2) {
   492→    const empty = () => 0;
   493→    const emptyScenario = (): ScenarioResult => ({ name: '', nameEn: '', probability: 20, targetMin: 0, targetMax: 0, description: '' });
   494→    return {
   495→      sma: {}, ema: {}, rsi: 50, mfi: 50, cci: 0, stochK: 50, stochD: 50,
   496→      williamsR: -50, macd: { line: 0, signal: 0, histogram: 0 },
   497→      adx: 0, diPlus: 0, diMinus: 0, sar: 0, atr: 0,
   498→      bollingerBands: { upper: 0, middle: 0, lower: 0 }, obv: 0, hasVolume: false,
   499→      resistances: [], supports: [],
   500→      supportStrengths: [], resistanceStrengths: [], priceTargets: [],
   501→      srMlWeights: null, upwardTargets: [], downwardTargets: [],
   502→      trend: {
   503→        short: { direction: 'flat', slope: 0, angle: 0, r2: 0 },
   504→        medium: { direction: 'flat', slope: 0, angle: 0, r2: 0 },
   505→        long: { direction: 'flat', slope: 0, angle: 0, r2: 0 },
   506→      },
   507→      scenarios: { R1: emptyScenario(), R2: emptyScenario(), R3: emptyScenario(), R4: emptyScenario(), R5: emptyScenario() },
   508→      bullScore: 0.5, bearScore: 0.5, overallSignal: 'neutral',
   509→      edgeWeights: { up: 0.5, down: 0.5, pullback: 0.25, risk: 0.12 },
   510→      ml: null,
   511→    };
   512→  }
   513→
   514→  const closes = data.map(d => d.close);
   515→  const price = closes[closes.length - 1];
   516→
   517→  // ── Volume detection ───────────────────────────────────────────────────
   518→  const hasVolume = data.some(d => d.volume > 0);
   519→
   520→  // ── Moving Averages ──────────────────────────────────────────────────────
   521→  const smaPeriods = [5, 10, 21, 50, 100, 200];
   522→  const smaResult: Record<string, number> = {};
   523→  for (const p of smaPeriods) smaResult[`sma${p}`] = sma(closes, p);
   524→
   525→  const emaResult: Record<string, number> = {
   526→    ema12: emaCalc(closes, 12),
   527→    ema26: emaCalc(closes, 26),
   528→  };
   529→
   530→  // ── Oscillators ──────────────────────────────────────────────────────────
   531→  const rsi = calcRSI(closes);
   532→  const mfi = hasVolume ? calcMFI(data) : 50; // neutral when no volume
   533→  const cci = calcCCI(data);
   534→  const stoch = calcStochastic(data);
   535→  const williamsR = calcWilliamsR(data);
   536→  const macd = calcMACD(closes);
   537→
   538→  // ── Trend ────────────────────────────────────────────────────────────────
   539→  const adxResult = calcADX(data);
   540→  const sar = calcSAR(data);
   541→
   542→  // ── Volatility ───────────────────────────────────────────────────────────
   543→  const atr = calcATR(data);
   544→  const bb = calcBollingerBands(closes);
   545→
   546→  // ── Volume (only when data has volume) ─────────────────────────────────
   547→  const obv = hasVolume ? calcOBV(data) : 0;
   548→
   549→  // ── Support / Resistance ────────────────────────────────────────────────
   550→  const { resistances, supports, supportStrengths, resistanceStrengths, priceTargets, srMlWeights, upwardTargets, downwardTargets } = calcSupportResistance(data, price);
   551→
   552→  // ── Trend Lines ─────────────────────────────────────────────────────────
   553→  const trend = {
   554→    short: calcTrend(closes, 21),
   555→    medium: calcTrend(closes, 50),
   556→    long: calcTrend(closes, Math.min(100, closes.length)),
   557→  };
   558→
   559→  // ── S/R Level References ──────────────────────────────────────────────
   560→  const R1_level = resistances[0] ?? price * 1.05;
   561→  const R2_level = resistances[1] ?? price * 1.10;
   562→  const R3_level = resistances[2] ?? price * 1.15;
   563→  const R4_level = resistances[3] ?? price * 1.20;
   564→  const R5_level = resistances[4] ?? price * 1.30;
   565→  const R6_level = resistances[5] ?? price * 1.40;
   566→  const S1_level = supports[0] ?? price * 0.95;
   567→  const S2_level = supports[1] ?? price * 0.90;
   568→  const S3_level = supports[2] ?? price * 0.85;
   569→  const S4_level = supports[3] ?? price * 0.80;
   570→  const S5_level = supports[4] ?? price * 0.75;
   571→  const S6_level = supports[5] ?? price * 0.70;
   572→
   573→  // ═══════════════════════════════════════════════════════════════════════════
   574→  // ENHANCED PROBABILITY ENGINE V2 — 7 Layers (Logistic Regression, No Fixed Weights)
   575→  // L0: Indicator trends | L1: Raw scores | L2: Momentum-corrected + binary cross signals
   576→  // L3: Logistic Regression bull consensus | L4: Adaptive scenario distribution
   577→  // L5: ML-driven edge weights | L6: ML prediction overlay
   578→  // ═══════════════════════════════════════════════════════════════════════════
   579→
   580→  const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));
   581→
   582→  // ── LAYER 0: Indicator Trends (current vs 5 bars ago) ──────────────────
   583→  const TB = 5;
   584→  const pastData = data.length > TB + 15 ? data.slice(0, -TB) : data;
   585→  const pastCloses = pastData.map(d => d.close);
   586→  const rsiPast = calcRSI(pastCloses);
   587→  const mfiPast = hasVolume ? calcMFI(pastData) : 50;
   588→  const cciPast = calcCCI(pastData);
   589→  const macdPast = calcMACD(pastCloses);
   590→  const stochPast = calcStochastic(pastData);
   591→
   592→  const IT = {
   593→    rsi: clamp((rsi - rsiPast) / 100, -1, 1),
   594→    mfi: clamp((mfi - mfiPast) / 100, -1, 1),
   595→    cci: clamp((cci - cciPast) / 300, -1, 1),
   596→    macd: clamp((macd.histogram - macdPast.histogram) / (Math.abs(macd.histogram) + Math.abs(macdPast.histogram) + 1) * 3, -1, 1),
   597→    stoch: clamp((stoch.k - stochPast.k) / 50, -1, 1),
   598→  };
   599→
   600→  // ── LAYER 1: Raw Indicator Scores (s_*) ──────────────────────────────────
   601→  const s_rsi = rsi > 80 ? 0.95 : rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   602→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   603→  const s_mfi = mfi > 80 ? 0.95 : mfi > 70 ? 0.85 : mfi > 60 ? 0.7 : mfi > 50 ? 0.55
   604→    : mfi > 40 ? 0.45 : mfi > 30 ? 0.3 : mfi > 20 ? 0.15 : 0.05;
   605→  const s_cci = cci > 200 ? 0.9 : cci > 100 ? 0.75 : cci > 0 ? 0.6
   606→    : cci > -100 ? 0.4 : cci > -200 ? 0.25 : 0.1;
   607→  const s_adx = adxResult.diPlus > adxResult.diMinus
   608→    ? clamp(0.5 + (adxResult.diPlus - adxResult.diMinus) / 100, 0.5, 1)
   609→    : clamp(0.5 - (adxResult.diMinus - adxResult.diPlus) / 100, 0, 0.5);
   610→  const s_macd = macd.histogram > 0 && macd.line > macd.signal ? 0.9
   611→    : macd.histogram > 0 ? 0.7 : macd.line > macd.signal ? 0.55
   612→    : macd.histogram < 0 && macd.line < macd.signal ? 0.1 : 0.3;
   613→  const s_stoch = clamp(
   614→    (stoch.k > 80 ? 0.9 : stoch.k > 70 ? 0.8 : stoch.k > 50 ? 0.6
   615→      : stoch.k > 30 ? 0.4 : stoch.k > 20 ? 0.2 : 0.1)
   616→    + (stoch.k > stoch.d ? 0.1 : -0.1), 0, 1);
   617→  const bbRange = bb.upper - bb.lower;
   618→  const s_bb = bbRange === 0 ? 0.5 : clamp((price - bb.lower) / bbRange, 0, 1);
   619→  const ma21 = smaResult.sma21 ?? 0;
   620→  const s_ma21 = clamp(price > ma21 ? 0.7 + ((price - ma21) / ma21) * 0.5 : 0.3 + ((price - ma21) / ma21) * 0.5, 0, 1);
   621→  const ma100v = smaResult.sma100 ?? 0;
   622→  const s_ma100 = clamp(price > ma100v ? 0.65 + ((price - ma100v) / ma100v) * 0.3 : 0.35 + ((price - ma100v) / ma100v) * 0.3, 0, 1);
   623→  const e12v = emaResult.ema12 ?? 0;
   624→  const e26v = emaResult.ema26 ?? 0;
   625→  const emaGap = e12v > 0 && e26v > 0 ? (e12v - e26v) / e26v : 0;
   626→  const s_ema = clamp(0.5 + emaGap * 5, 0, 1);
   627→  const s_atr = clamp(1 - (price > 0 ? atr / price : 0) * 10, 0.2, 1);
   628→  const tStr = trend.medium.r2 * Math.abs(trend.medium.angle) / 45;
   629→  const s_trend = trend.medium.direction === 'up' ? clamp(0.5 + tStr / 2, 0.5, 1)
   630→    : trend.medium.direction === 'down' ? clamp(0.5 - tStr / 2, 0, 0.5) : 0.5;
   631→  const dR1 = R1_level > 0 ? Math.abs(price - R1_level) / R1_level : 1;
   632→  const dS1 = S1_level > 0 ? Math.abs(price - S1_level) / S1_level : 1;
   633→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
   634→
   635→  // ── LAYER 2: Momentum Correction + Binary Cross Signals ─────────────────
   636→  const f_rsi = clamp(s_rsi + IT.rsi * 0.08, 0, 1);
   637→  const f_mfi = clamp(s_mfi + IT.mfi * 0.08, 0, 1);
   638→  const f_cci = clamp(s_cci + IT.cci * 0.06, 0, 1);
   639→  const f_macd = clamp(s_macd + IT.macd * 0.08, 0, 1);
   640→  const f_stoch = clamp(s_stoch + IT.stoch * 0.06, 0, 1);
   641→
   642→  // Binary cross signals per spec: 0.9 for bullish cross, 0.1 otherwise
   643→  const f_stochCross = stochPast.k < stochPast.d && stoch.k > stoch.d ? 0.9 : 0.1;
   644→  const f_macdCross = macdPast.line < macdPast.signal && macd.line > macd.signal ? 0.9 : 0.1;
   645→
   646→  // Divergence detection (binary: 0.85 if detected, 0.15 otherwise)
   647→  let f_div = 0.15;
   648→  if (closes.length > 40) {
   649→    const p20 = closes[closes.length - 21];
   650→    const rsi20 = calcRSI(closes.slice(0, -20));
   651→    const pChange = price - p20;
   652→    const rsiChange = rsi - rsi20;
   653→    if (pChange !== 0 && rsiChange !== 0) {
   654→      const isDivergence = (pChange > 0 && rsiChange < 0) || (pChange < 0 && rsiChange > 0);
   655→      f_div = isDivergence ? 0.85 : 0.15;
   656→    }
   657→  }
   658→
   659→  // ── LAYER 3: Logistic Regression ML Consensus (NO FIXED WEIGHTS) ─────────
   660→  // Build 16-feature vector for current candle
   661→  const currentFeatures = [
   662→    f_rsi, f_mfi, f_cci, s_adx,
   663→    f_macd, f_stoch, s_bb, s_ma21, s_ma100, s_ema,
   664→    s_atr, s_trend, s_sr,
   665→    f_stochCross, f_macdCross, f_div,
   666→  ];
   667→
   668→  let mlResult: {
   669→    weightsUsed: boolean;
   670→    indicatorWeights: Record<string, number>;
   671→    accuracy: number;
   672→    isTrained: boolean;
   673→    sampleCount: number;
   674→    predictionProb: number | null;
   675→    predictedReturn5d: number;
   676→    predictionConfidence: number;
   677→    mlScore: number;
   678→    adaptiveParams: { momentumFactor: number; volatilityFactor: number; trendFactor: number };
   679→    scenarioFactors: { momentum: number; volatility: number; trend: number };
   680→  } | null = null;
   681→
   682→  // Train/Load Adaptive Model
   683→  let adaptiveResult: import('./ml-engine').AdaptiveModelResult | null = null;
   684→  if (symbol && data.length >= 70) {
   685→    try {
   686→      const { trainAdaptiveModel, getCachedAdaptiveModel, calculateBullConsensus, VDSS_FEATURE_NAMES, predictPrices, evaluatePredictedIndicators } = await import('./ml-engine');
   687→
   688→      // Try cache first, then train
   689→      adaptiveResult = getCachedAdaptiveModel(symbol);
   690→      if (!adaptiveResult) {
   691→        adaptiveResult = trainAdaptiveModel(data, symbol);
   692→      }
   693→
   694→      if (adaptiveResult && adaptiveResult.isTrained) {
   695→        // Build indicator weights map for display
   696→        const indicatorWeights: Record<string, number> = {};
   697→        for (let i = 0; i < VDSS_FEATURE_NAMES.length; i++) {
   698→          indicatorWeights[VDSS_FEATURE_NAMES[i]] = adaptiveResult.weights[i];
   699→        }
   700→
   701→        mlResult = {
   702→          weightsUsed: true,
   703→          indicatorWeights,
   704→          accuracy: adaptiveResult.recentAccuracy,
   705→          isTrained: true,
   706→          sampleCount: adaptiveResult.sampleCount,
   707→          predictionProb: adaptiveResult.predictionProb,
   708→          predictedReturn5d: 0,
   709→          predictionConfidence: 0,
   710→          mlScore: adaptiveResult.predictionProb ?? 0.5,
   711→          adaptiveParams: adaptiveResult.adaptiveParams,
   712→          scenarioFactors: {
   713→            momentum: adaptiveResult.adaptiveParams.momentumFactor,
   714→            volatility: adaptiveResult.adaptiveParams.volatilityFactor,
   715→            trend: adaptiveResult.adaptiveParams.trendFactor,
   716→          },
   717→        };
   718→      }
   719→
   720→      // ML Price Prediction (L6 overlay)
   721→      if (data.length >= 40) {
   722→        const prediction = predictPrices(data);
   723→        if (prediction) {
   724→          const { mlScore } = evaluatePredictedIndicators(data, prediction);
   725→          if (mlResult) {
   726→            mlResult.predictedReturn5d = prediction.predictedReturn5d;
   727→            mlResult.predictionConfidence = prediction.confidence;
   728→            mlResult.mlScore = mlScore;
   729→          } else {
   730→            mlResult = {
   731→              weightsUsed: false,
   732→              indicatorWeights: {},
   733→              accuracy: 0.5,
   734→              isTrained: false,
   735→              sampleCount: 0,
   736→              predictionProb: null,
   737→              predictedReturn5d: prediction.predictedReturn5d,
   738→              predictionConfidence: prediction.confidence,
   739→              mlScore,
   740→              adaptiveParams: { momentumFactor: 0.7, volatilityFactor: 0.5, trendFactor: 0.6 },
   741→              scenarioFactors: { momentum: 0.7, volatility: 0.5, trend: 0.6 },
   742→            };
   743→          }
   744→        }
   745→      }
   746→    } catch {
   747→      // ML engine failed — use fallback
   748→    }
   749→  }
   750→
   751→  // Calculate bullConsensus using ML weights or fallback defaults
   752→  const { calculateBullConsensus: calcBC } = await import('./ml-engine');
   753→  const { bullConsensus } = calcBC(currentFeatures, adaptiveResult, hasVolume);
   754→
   755→  const bullScore = bullConsensus;
   756→  const bearScore = 1 - bullConsensus;
   757→
   758→  // ── LAYER 4: 5-Scenario Probability Distribution (Adaptive) ──────────────
   759→  const { calculateScenarioProbabilities: calcSP } = await import('./ml-engine');
   760→  const scenarioResult = calcSP(
   761→    bullConsensus, price,
   762→    R1_level, S1_level, ma100v,
   763→    rsi, mfi, stoch.k,
   764→    hasVolume,
   765→    adaptiveResult,
   766→  );
   767→  const { pR1, pR2, pR3, pR4, pR5 } = scenarioResult;
   768→
   769→  // Update mlResult with scenario factors
   770→  if (mlResult) {
   771→    mlResult.scenarioFactors = scenarioResult.factors;
   772→  }
   773→
   774→  // ── LAYER 5: Edge Weights for VDss Graph (ML-driven) ────────────────────
   775→  const { calculateEdgeWeights: calcEW } = await import('./ml-engine');
   776→  const edgeWeights = calcEW(bullConsensus, adxResult.adx, adaptiveResult);
   777→
   778→  // ── Overall Signal ─────────────────────────────────────────────────────
   779→  const overallSignal: 'bullish' | 'bearish' | 'neutral' =
   780→    bullScore > 0.58 ? 'bullish' : bullScore < 0.42 ? 'bearish' : 'neutral';
   781→
   782→  // ── Scenario Descriptions (context-aware: names adapt to trend) ──────
   783→  const fmt = (n: number) => Math.round(n).toLocaleString('fa-IR');
   784→
   785→  // Dynamic scenario names based on market direction
   786→  const scenarioNames: Record<string, { name: string; nameEn: string }> = overallSignal === 'bearish'
   787→    ? {
   788→        R1: { name: 'بازگشت صعودی', nameEn: 'Strong Bullish Reversal' },
   789→        R2: { name: 'پولبک صعودی', nameEn: 'Bullish Pullback' },
   790→        R3: { name: 'نوسان در محدوده', nameEn: 'Range-bound' },
   791→        R4: { name: 'نزول متوسط', nameEn: 'Moderate Decline' },
   792→        R5: { name: 'نزول هیجانی', nameEn: 'Strong Bearish' },
   793→      }
   794→    : overallSignal === 'bullish'
   795→      ? {
   796→          R1: { name: 'صعود هیجانی', nameEn: 'Strong Bullish' },
   797→          R2: { name: 'صعود تدریجی', nameEn: 'Gradual Uptrend' },
   798→          R3: { name: 'نوسان در محدوده', nameEn: 'Range-bound' },
   799→          R4: { name: 'اصلاح متوسط', nameEn: 'Moderate Correction' },
   800→          R5: { name: 'اصلاح عمیق', nameEn: 'Deep Correction' },
   801→        }
   802→      : {
   803→          R1: { name: 'صعود هیجانی', nameEn: 'Strong Bullish' },
   804→          R2: { name: 'صعود تدریجی', nameEn: 'Gradual Uptrend' },
   805→          R3: { name: 'نوسان در محدوده', nameEn: 'Range-bound' },
   806→          R4: { name: 'اصلاح متوسط', nameEn: 'Moderate Correction' },
   807→          R5: { name: 'افت شدید', nameEn: 'Sharp Decline' },
   808→        };
   809→
   810→  const scenarios = {
   811→    R1: {
   812→      ...scenarioNames.R1,
   813→      probability: pR1,
   814→      targetMin: R2_level,
   815→      targetMax: R5_level,
   816→      description: `عبور از مقاومت‌های ${fmt(R1_level)} و ${fmt(R2_level)} با شتاب بالا. هدف: ${fmt(R2_level)} تا ${fmt(R5_level)}.`,
   817→    },
   818→    R2: {
   819→      ...scenarioNames.R2,
   820→      probability: pR2,
   821→      targetMin: R1_level,
   822→      targetMax: R3_level,
   823→      description: `حرکت صعودی آرام با پولبک‌های موقت. هدف: ${fmt(R1_level)} تا ${fmt(R3_level)}.`,
   824→    },
   825→    R3: {
   826→      ...scenarioNames.R3,
   827→      probability: pR3,
   828→      targetMin: S1_level,
   829→      targetMax: R1_level,
   830→      description: `نوسان بین حمایت ${fmt(S1_level)} و مقاومت ${fmt(R1_level)}.`,
   831→    },
   832→    R4: {
   833→      ...scenarioNames.R4,
   834→      probability: pR4,
   835→      targetMin: S3_level,
   836→      targetMax: S1_level,
   837→      description: `اصلاح تا محدوده حمایت ${fmt(S3_level)} تا ${fmt(S1_level)}.`,
   838→    },
   839→    R5: {
   840→      ...scenarioNames.R5,
   841→      probability: pR5,
   842→      targetMin: S5_level,
   843→      targetMax: S3_level,
   844→      description: `شکست ساختار و افت تا حمایت‌های ${fmt(S5_level)} تا ${fmt(S3_level)}.`,
   845→    },
   846→  };
   847→
   848→  return {
   849→    sma: smaResult,
   850→    ema: emaResult,
   851→    rsi,
   852→    mfi,
   853→    cci,
   854→    stochK: stoch.k,
   855→    stochD: stoch.d,
   856→    williamsR,
   857→    macd,
   858→    adx: adxResult.adx,
   859→    diPlus: adxResult.diPlus,
   860→    diMinus: adxResult.diMinus,
   861→    sar,
   862→    atr,
   863→    bollingerBands: bb,
   864→    obv,
   865→    hasVolume,
   866→    resistances: resistances.length ? resistances : [R1_level, R2_level, R3_level, R4_level, R5_level, R6_level],
   867→    supports: supports.length ? supports : [S1_level, S2_level, S3_level, S4_level, S5_level, S6_level],
   868→    supportStrengths,
   869→    resistanceStrengths,
   870→    priceTargets,
   871→    srMlWeights,
   872→    upwardTargets,
   873→    downwardTargets,
   874→    trend,
   875→    scenarios,
   876→    bullScore,
   877→    bearScore,
   878→    overallSignal,
   879→    edgeWeights,
   880→    ml: mlResult,
   881→  };
   882→}
