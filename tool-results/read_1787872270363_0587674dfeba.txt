   221→// Helper: resolve asset type
   222→// ═══════════════════════════════════════════════════════════════

   224→function resolveAsset(raw: string): AssetType {
   225→  const v = raw.toLowerCase();
   226→  if (v.includes('crypto') || v.includes('btc') || v.includes('eth') || v.includes('usdt')) return 'cryptocurrency';
   227→  if (v.includes('forex') || v.includes('eur') || v.includes('usd') || v.includes('jpy')) return 'forex';
   228→  if (v.includes('gold') || v.includes('طلا') || v.includes('xau') || v.includes('oil') || v.includes('نفت')) return 'commodities';
   229→  return 'stocks';
   230→}
   231→
   232→function resolveTimeframe(raw: string): TimeframeType {
   233→  const v = raw.toLowerCase();
   234→  if (v.includes('h') || v.includes('hour') || v.includes('4h') || v.includes('1h')) return 'hourly';
   235→  if (v.includes('w') || v.includes('week')) return 'weekly';
   236→  if (v.includes('m') || v.includes('month')) return 'monthly';
   237→  return 'daily';
   238→}
   239→
   240→function resolveRegime(raw: string): RegimeType {
   241→  if (raw.includes('Strong Bull') || raw.includes('strong_bull')) return 'Strong Bull';
   242→  if (raw.includes('Weak Bull') || raw.includes('weak_bull')) return 'Weak Bull';
   243→  if (raw.includes('Strong Bear') || raw.includes('strong_bear')) return 'Strong Bear';
   244→  if (raw.includes('Weak Bear') || raw.includes('weak_bear')) return 'Weak Bear';
   245→  return 'Range';
   246→}

   248→function toPersianNum(n: number): string {
   249→  if (!isFinite(n) || isNaN(n)) return '\u06f0';
   250→  const fixed = Math.round(n * 100) / 100;
   251→  return String(fixed).replace(/\d/g, d => String.fromCharCode('\u0660'.charCodeAt(0) + parseInt(d)));
   252→}

   255→// School Selection
   256→// ═══════════════════════════════════════════════════════════════
   257→
   258→function selectSchools(ctx: MSLV4Context) {
   259→  const asset = resolveAsset(ctx.asset);
   260→  const tf = resolveTimeframe(ctx.timeframe);
   261→  const regime = resolveRegime(ctx.regime);
   262→  const am = ASSET_MULT[asset] ?? ASSET_MULT.stocks;
   263→  const tm = TF_MULT[tf] ?? TF_MULT.daily;
   264→  const rm = REGIME_MULT[regime] ?? REGIME_MULT.Range;
   265→
   266→  const pt = ctx.probTrend;
   267→
   268→  const rawWeights: Record<SchoolId, number> = {} as Record<SchoolId, number>;
   269→  for (const s of SCHOOLS) {
   270→    let w = s.base_weight * am[s.id] * tm[s.id] * rm[s.id];
   271→    // Adaptive: zero out volume school if no volume data
   272→    if (s.id === 'volume' && ctx.hasVolume === false) w *= 0.2;
   273→
   274→    // Probability trend adjustments (cumulative weighted 2x more than individual)
   275→    if (pt) {
   276→      // Oscillator school: if cumulative bearish trend rising → divergence/reversal signals
   277→      if (s.id === 'oscillator' && pt.bearGroupDir === 'rising' && pt.bullGroupCum && pt.bullGroupCum > 0.3) {
   278→        w *= 1.25; // Bearish momentum building, oscillators useful for reversal timing
   279→      }
   280→      // Classical school: strong directional cumulative trend → support/resistance focus
   281→      if (s.id === 'classical' && (pt.bullGroupDir === 'rising' || pt.bearGroupDir === 'rising')) {
   282→        w *= 1.15; // Clear trend direction, classical S/R analysis
   283→      }
   284→      // Elliott school: volatile or changing cumulative trends → wave analysis
   285→      if (s.id === 'elliott' && (pt.bullGroupDir === 'volatile' || pt.bearGroupDir === 'volatile')) {
   286→        w *= 1.2; // Volatile probability shifts suggest wave structure changes
   287→      }
   288→      // Harmonic school: trend reversal in cumulative probabilities
   289→      if (s.id === 'harmonic' && (
   290→        (pt.bullGroupDir === 'falling' && pt.bullGroupCum && pt.bullGroupCum > 0.4) ||
   291→        (pt.bearGroupDir === 'falling' && pt.bearGroupCum && pt.bearGroupCum > 0.4)
   292→      )) {
   293→        w *= 1.2; // Declining dominant group suggests reversal zone
   294→      }
   295→      // Hybrid school: neutral dominant and stable trends
   296→      if (s.id === 'hybrid' && pt.neutralGroupDir === 'stable' && pt.neutralGroupCum && pt.neutralGroupCum > 0.2) {
   297→        w *= 1.15; // Range-bound market, hybrid approach best
   298→      }
   299→      // Volume school: strong directional 7-day change
   300→      if (s.id === 'volume' && ctx.hasVolume !== false) {
   301→        const maxChange = Math.max(Math.abs(pt.bullGroupChange7d ?? 0), Math.abs(pt.bearGroupChange7d ?? 0));
   302→        if (maxChange > 0.15) w *= 1.15; // Significant probability shift, volume confirms
   303→      }
   304→    }
   305→
   306→    rawWeights[s.id] = w;
   307→  }
   308→
   309→  const total = Object.values(rawWeights).reduce((a, b) => a + b, 0);
   310→  const normWeights: Record<string, number> = {};
   311→  for (const id of SCHOOL_IDS) normWeights[id] = rawWeights[id] / total;
   312→
   313→  const sorted = SCHOOL_IDS.map(id => ({ id, w: normWeights[id] })).sort((a, b) => b.w - a.w);
   314→  const primary = sorted[0];
   315→  const secondary = sorted[1];
   316→
   317→  let status: string;
   318→  if (primary.id === 'hybrid' && primary.w < 0.35) {
   319→    status = 'hybrid';
   320→  } else if (primary.w > 0.45) {
   321→    status = 'confirmed';
   322→  } else if (primary.w > 0.35) {
   323→    status = 'tentative';
   324→  } else {
   325→    status = 'hybrid';
   326→  }
   327→
   328→  const primaryDef = SCHOOLS.find(s => s.id === primary.id)!;
   329→  const secondaryDef = SCHOOLS.find(s => s.id === secondary.id)!;
   330→
   331→  const reasoning = buildSchoolReasoning(primaryDef, normWeights, ctx, asset, tf, regime);
   332→
   333→  return {
   334→    primary: {
   335→      id: primary.id, name: primaryDef.name, nameEn: primaryDef.nameEn,
   336→      weight: Math.round(primary.w * 100) / 100,
   337→      confidence: ctx.confidence > 0.75 ? 'high' : ctx.confidence > 0.5 ? 'medium' : 'low',
   338→      status, reasoning,
   339→    },
   340→    secondary: {
   341→      id: secondary.id, name: secondaryDef.name,
   342→      weight: Math.round(secondary.w * 100) / 100,
   343→      reasoning: `مکتب ${secondaryDef.name} با وزن ${toPersianNum(secondary.w)} به عنوان مکتب مکمل انتخاب شد`,
   344→    },
   345→    all_schools: Object.fromEntries(SCHOOL_IDS.map(id => [id, Math.round(normWeights[id] * 100) / 100])),
   346→  };
   347→}

   349→function buildSchoolReasoning(s: SchoolDef, w: Record<string, number>, ctx: MSLV4Context, asset: string, tf: string, regime: string): string[] {
   350→  const r: string[] = [];
   351→  const pc = ctx.pattern_counts?.classical ?? 0;
   352→  const phc = ctx.pattern_counts?.harmonic ?? 0;
   353→  const ec = ctx.pattern_counts?.elliott ?? 0;
   354→  if (s.id === 'classical' && pc >= 2) r.push(`تشخیص ${toPersianNum(pc)} الگوی کلاسیک`);
   355→  if (s.id === 'oscillator' && ctx.divergence_present) r.push('واگرایی تشخیص داده شده');
   356→  if (s.id === 'oscillator' && ctx.rsi !== undefined && (ctx.rsi > 70 || ctx.rsi < 30)) r.push('RSI در منطقه اشباع');
   357→  if (s.id === 'harmonic' && phc >= 2) r.push(`تشخیص ${toPersianNum(phc)} الگوی هارمونیک`);
   358→  if (s.id === 'elliott' && ec >= 3) r.push(`تشخیص ${toPersianNum(ec)} موج الیوت`);
   359→  if (s.id === 'volume' && ctx.hasVolume) r.push('داده‌های حجم موجود و تأییدکننده');
   360→  r.push(`رژیم بازار ${regime} تطابق دارد`);
   361→  r.push(`نوع دارایی: ${asset}، تایم‌فریم: ${tf}`);
   362→  return r;
   363→}

   369→function selectStyles(primarySchoolId: SchoolId, ctx: MSLV4Context) {
   370→  const scores: Record<StyleId, number> = {} as Record<StyleId, number>;
   371→  const audience = ctx.audience_level ?? 'pro';
   372→  const pt = ctx.probTrend;
   373→
   374→  for (const style of STYLES) {
   375→    const compatibility = SS_MATRIX[primarySchoolId]?.[style.id] ?? 0.5;
   376→    const audienceMatch = getAudienceMatch(style.id, audience);
   377→    const marketMatch = getMarketContextMatch(style.id, ctx);
   378→    const schoolMatch = SS_MATRIX[primarySchoolId]?.[style.id] ?? 0.5;
   379→
   380→    // Probability trend influence on style (cumulative weighted 2x)
   381→    let trendStyleBonus = 0;
   382→    if (pt) {
   383→      const cumDir = pt.bullGroupDir === 'rising' ? 1 : pt.bearGroupDir === 'rising' ? -1 : 0;
   384→      const indivDir = pt.dominantIndivDir === 'rising' ? 1 : pt.dominantIndivDir === 'falling' ? -1 : 0;
   385→      const trendSignal = cumDir * 0.67 + indivDir * 0.33; // cumulative 2x weight
   386→
   387→      if (style.id === 'forecasting' && Math.abs(trendSignal) > 0.5) trendStyleBonus = 0.1;
   388→      if (style.id === 'trading' && Math.abs(trendSignal) > 0.3) trendStyleBonus = 0.08;
   389→      if (style.id === 'analytical' && Math.abs(trendSignal) < 0.3) trendStyleBonus = 0.08;
   390→      if (style.id === 'executive' && (pt.bullGroupDir === 'stable' || pt.bearGroupDir === 'stable')) trendStyleBonus = 0.05;
   391→    }
   392→
   393→    scores[style.id] = compatibility * 0.35 + audienceMatch * 0.25 + marketMatch * 0.2 + schoolMatch * 0.1 + trendStyleBonus;
   394→  }
   395→
   396→  const sorted = (Object.entries(scores) as [StyleId, number][]).sort((a, b) => b[1] - a[1]);
   397→  const [pId, pScore] = sorted[0];
   398→  const [sId, sScore] = sorted[1];
   399→  const pDef = STYLES.find(s => s.id === pId)!;
   400→  const sDef = STYLES.find(s => s.id === sId)!;
   401→
   402→  return {
   403→    primary: {
   404→      id: pId, name: pDef.name, nameEn: pDef.nameEn,
   405→      score: Math.round(pScore * 100) / 100,
   406→      detail_level: pDef.detailLevel, length: pDef.length, focus: pDef.focus,
   407→      reasoning: buildStyleReasoning(pDef, primarySchoolId, ctx),
   408→    },
   409→    secondary: { id: sId, name: sDef.name, score: Math.round(sScore * 100) / 100 },
   410→  };
   411→}

   413→function getAudienceMatch(styleId: StyleId, audience: string): number {
   414→  const map: Record<string, Record<StyleId, number>> = {
   415→    pro:    { executive: 0.9, analytical: 1.0, forecasting: 0.8, trading: 0.9, educational: 0.3 },
   416→    mid:    { executive: 0.7, analytical: 0.8, forecasting: 0.7, trading: 0.8, educational: 0.6 },
   417→    begin:  { executive: 0.4, analytical: 0.5, forecasting: 0.4, trading: 0.5, educational: 1.0 },
   418→  };
   419→  const level = audience.includes('begin') || audience.includes('مبتدی') ? 'begin'
   420→    : audience.includes('mid') || audience.includes('متوسط') ? 'mid' : 'pro';
   421→  return map[level]?.[styleId] ?? 0.6;
   422→}
   423→
   424→function getMarketContextMatch(styleId: StyleId, ctx: MSLV4Context): number {
   425→  let score = 0.5;
   426→  const isBull = ctx.regime.includes('Bull');
   427→  const isBear = ctx.regime.includes('Bear');
   428→  if (styleId === 'executive' && ctx.confidence > 0.7) score = 0.9;
   429→  if (styleId === 'trading' && (isBull || isBear) && ctx.trend_strength > 0.5) score = 0.9;
   430→  if (styleId === 'forecasting' && ctx.confidence > 0.5) score = 0.8;
   431→  if (styleId === 'analytical' && ctx.confidence > 0.5) score = 0.85;
   432→  if (styleId === 'educational' && ctx.confidence < 0.5) score = 0.85;
   433→  return score;
   434→}

   436→function buildStyleReasoning(s: StyleDef, schoolId: SchoolId, ctx: MSLV4Context): string[] {
   437→  const r: string[] = [];
   438→  const schoolName = SCHOOLS.find(x => x.id === schoolId)?.name ?? '';
   439→  r.push(`مکتب ${schoolName} نیاز به تحلیل ${s.focus === 'decision' ? 'تصمیم‌محور' : s.focus === 'analysis' ? 'دقیق' : s.focus === 'forecast' ? 'پیش‌بینی' : s.focus === 'action' ? 'عملیاتی' : 'آموزشی'} دارد`);
   440→  if (ctx.confidence > 0.7) r.push(`اطمینان کلی ${toPersianNum(ctx.confidence)} (بالا)`);
   441→  else if (ctx.confidence > 0.5) r.push(`اطمینان کلی ${toPersianNum(ctx.confidence)} (متوسط)`);

   449→function selectTones(schoolId: SchoolId, styleId: StyleId, ctx: MSLV4Context) {
   450→  const scores: Record<ToneId, number> = {} as Record<ToneId, number>;
   451→
   452→  for (const tone of TONES) {
   453→    const condMatch = checkToneConditions(tone.id, ctx);
   454→    const schoolMatch = ST_MATRIX[schoolId]?.[tone.id] ?? 0.5;
   455→    const styleMatch = STYLE_TONE_MATRIX[styleId]?.[tone.id] ?? 0.5;
   456→    const intensity = 0.5 + ctx.confidence * 0.3 + ctx.volatility * 0.2;
   457→    scores[tone.id] = condMatch * 0.5 + schoolMatch * 0.2 + styleMatch * 0.2 + Math.min(intensity, 1) * 0.1;
   458→  }
   459→
   460→  const sorted = (Object.entries(scores) as [ToneId, number][]).sort((a, b) => b[1] - a[1]);
   461→  const [pId, pScore] = sorted[0];
   462→  const [sId, sScore] = sorted[1];
   463→  const pDef = TONES.find(t => t.id === pId)!;
   464→  const sDef = TONES.find(t => t.id === sId)!;
   465→  const intensity = calcIntensity(pId, ctx);
   466→
   467→  return {
   468→    primary: {
   469→      id: pId, name: pDef.name, nameEn: pDef.nameEn,
   470→      score: Math.round(pScore * 100) / 100,
   471→      intensity, keywords: pDef.keywords.slice(0, 3),
   472→      reasoning: buildToneReasoning(pDef, ctx),
   473→    },
   474→    secondary: { id: sId, name: sDef.name, score: Math.round(sScore * 100) / 100 },
   475→  };
   476→}
   477→
   478→function checkToneConditions(toneId: ToneId, ctx: MSLV4Context): number {
   479→  const isBull = ctx.regime.includes('Bull');
   480→  const isBear = ctx.regime.includes('Bear');
   481→  switch (toneId) {
   482→    case 'conservative': return ctx.confidence < 0.6 ? 0.9 : ctx.volatility > 0.03 ? 0.7 : 0.4;
   483→    case 'aggressive': return ctx.confidence > 0.8 && isBull ? 0.9 : ctx.confidence > 0.7 ? 0.6 : 0.3;
   484→    case 'balanced': return ctx.confidence >= 0.6 && ctx.confidence <= 0.8 ? 0.9 : 0.5;
   485→    case 'warning': return ctx.divergence_present ? 0.9 : isBear && ctx.volatility > 0.03 ? 0.8 : 0.3;
   486→    case 'optimistic': return isBull && ctx.confidence > 0.7 ? 0.9 : isBull ? 0.5 : 0.2;
   487→    case 'realistic': return ctx.confidence >= 0.4 && ctx.confidence <= 0.7 ? 0.9 : 0.5;
   488→    default: return 0.5;
   489→  }
   490→}
   491→
   492→function calcIntensity(toneId: ToneId, ctx: MSLV4Context): string {
   493→  const base = ctx.confidence + ctx.volatility;
   494→  if (toneId === 'warning') return base > 0.9 ? 'high' : base > 0.6 ? 'medium' : 'low';
   495→  if (toneId === 'aggressive' || toneId === 'optimistic') return base > 0.85 ? 'high' : base > 0.55 ? 'medium' : 'low';
   496→  if (toneId === 'conservative') return ctx.volatility > 0.03 ? 'high' : ctx.volatility > 0.015 ? 'medium' : 'low';
   497→  return base > 0.8 ? 'medium-high' : base > 0.5 ? 'medium' : 'low';
   498→}

   502→  r.push(`اطمینان کلی = ${toPersianNum(ctx.confidence)}`);

   505→  if (ctx.divergence_present) r.push('واگرایی تشخیص داده شده');
   506→  return r;

   537→    analysis_style: styles,
   538→    analysis_tone: tones,
