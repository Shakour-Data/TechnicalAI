   699→function scoreSchool(s: SchoolDef, inp: NarrativeInput): number {
   700→  let s_ = 0;
   701→  const { trendDirection, rsi, adx, bbPosition, diPlus, diMinus } = inp;
   702→  switch (s.id) {
   703→    case 'trend': s_ = adx > 25 ? 8 : adx > 15 ? 5 : 2; break;
   704→    case 'classical': s_ = (inp.nearResistance || inp.nearSupport) ? 9 : 5; break;
   705→    case 'candlestick': s_ = (inp.nearResistance || inp.nearSupport) ? 8 : 4; break;
   706→    case 'oscillator': s_ = (rsi > 70 || rsi < 30) ? 9 : Math.abs(rsi - 50) > 15 ? 6 : 3; break;
   707→    case 'harmonic': s_ = (inp.nearResistance || inp.nearSupport) ? 7 : 3; break;
   708→    case 'elliott': s_ = inp.trendR2 > 0.7 ? 9 : inp.trendR2 > 0.4 ? 5 : 2; break;
   709→    case 'volume': s_ = inp.hasVolume ? 8 : 2; break;
   710→    case 'sr': s_ = (inp.nearResistance || inp.nearSupport) ? 9 : 4; break;
   711→    case 'volatility': s_ = inp.atr / inp.price > 0.03 ? 8 : inp.atr / inp.price > 0.01 ? 5 : 2; break;
   712→    case 'hybrid': s_ = 6; break;
   713→    default: s_ = 4;
   714→  }
   715→  if (s.id === 'trend' && trendDirection === 'up') s_ += 1;
   716→  if (s.id === 'trend' && diPlus > diMinus) s_ += 1;
   717→  return s_;
   718→}

   720→function scoreStyle(st: StyleDef, inp: NarrativeInput): number {
   721→  let s_ = 4;
   722→  const { trendDirection, adx, rsi, dominantScenarioKey } = inp;
   723→  switch (st.id) {
   724→    case 'executive': s_ = adx > 25 ? 7 : 4; break;

   726→    case 'trading': s_ = (dominantScenarioKey === 'R1' || dominantScenarioKey === 'R2') ? 9 : 4; break;
   727→    case 'forecast': s_ = 6; break;
   728→    case 'educational': s_ = 5; break;
   729→  }
   730→  return s_;
   731→}

   733→function scoreTone(t: ToneDef, inp: NarrativeInput): number {
   734→  let s_ = 3;
   735→  const { trendDirection, adx, rsi, dominantScenarioKey, macdHist } = inp;
   736→  switch (t.id) {
   737→    case 'conservative': s_ = 5; if (rsi > 65 || rsi < 35) s_ += 3; break;
   738→    case 'aggressive': s_ = 3; if (adx > 30 && (rsi > 60 || rsi < 40)) s_ += 4; break;
   739→    case 'balanced': s_ = 6; break;
   740→    case 'alert': s_ = 4; if (rsi > 75 || rsi < 25) s_ += 3; break;
   741→    case 'optimistic': s_ = 3; if (trendDirection === 'up' && macdHist > 0) s_ += 4; break;
   742→    case 'realistic': s_ = 6; break;
   743→  }
   744→  if (dominantScenarioKey === 'R1' || dominantScenarioKey === 'R2') s_ += 2;
   745→  return s_ + 1;
   746→}

   748→interface Scored<T> { item: T; score: number; }
   749→function scoreAll<T>(items: T[], inp: NarrativeInput, fn: (item: T, inp: NarrativeInput) => number): Scored<T>[] {
   750→  return items.map(item => ({ item, score: fn(item, inp) })).sort((a, b) => b.score - a.score);
   751→}

   772→export function buildNarrativeInput(body: {
   773→  currentPrice: number; trendDirection: string; rsi: number; adx: number;
   774→  stochK: number; stochD: number; macdLine: number; macdSignal: number; macdHist: number;
   775→  diPlus: number; diMinus: number; obv: number; hasVolume: boolean;
   776→  cci: number; mfi: number; atr: number; bollingerUpper: number; bollingerMiddle: number; bollingerLower: number;
   777→  trendAngle: number; trendR2: number; overallSignal: string;
   778→  scenarios: Record<string, { probability: number }>;
   779→  resistances: number[]; supports: number[]; ma21: number; ma100: number;
   780→}): NarrativeInput {
   781→  const bbRange = body.bollingerUpper - body.bollingerLower;
   782→  const bbPosition = bbRange > 0 ? Math.max(0, Math.min(100, ((body.currentPrice - body.bollingerLower) / bbRange) * 100)) : 50;
   783→  const R1 = body.resistances[0] || body.currentPrice * 1.05;
   784→  const S1 = body.supports[0] || body.currentPrice * 0.95;
   785→  let dominantKey = 'R3'; let dominantProb = 0;
   786→  const allScenarioKeys = ['R1', 'R2', 'R3', 'R4', 'R5', 'R6', 'R7', 'R8', 'R9'] as const;
   787→  for (const k of allScenarioKeys) {
   788→    const p = body.scenarios[k]?.probability ?? 0;
   789→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
   790→  }
   791→  return {
   792→    price: body.currentPrice, trendDirection: body.trendDirection, rsi: body.rsi, adx: body.adx,
   793→    stochK: body.stochK, stochD: body.stochD, macdLine: body.macdLine, macdSignal: body.macdSignal,
   794→    macdHist: body.macdHist, diPlus: body.diPlus, diMinus: body.diMinus, obv: body.obv,
   795→    hasVolume: body.hasVolume, cci: body.cci, mfi: body.mfi, atr: body.atr,
   796→    bbPosition, bbUpper: body.bollingerUpper, bbLower: body.bollingerLower,
   797→    dominantScenarioKey: dominantKey,
   798→    scenarioProbabilities: Object.fromEntries(allScenarioKeys.map(k => [k, body.scenarios[k]?.probability ?? 0])),
   799→    nearResistance: R1 > 0 ? Math.abs(body.currentPrice - R1) / R1 < 0.03 : false,
   800→    nearSupport: S1 > 0 ? Math.abs(body.currentPrice - S1) / S1 < 0.03 : false,
   801→    priceVsMa21: body.currentPrice > body.ma21 ? 'above' : 'below',
   802→    priceVsMa100: body.currentPrice > body.ma100 ? 'above' : 'below',
   803→    trendR2: body.trendR2,
   804→  };
   805→}

   811→/** Simple deterministic hash for date+instrument rotation */
   812→function simpleHash(str: string): number {
   813→  let hash = 0;
   814→  for (let i = 0; i < str.length; i++) {
   815→    const ch = str.charCodeAt(i);
   816→    hash = ((hash << 5) - hash) + ch;
   817→    hash |= 0;
   818→  }
   819→  return Math.abs(hash);
   820→}

   822→/**
   823→ * Selects a V9 persona, template format, and variation seed
   824→ * based on deterministic hash of date + instrument + school.
   825→ * This ensures: different persona per day, different per instrument,
   826→ * and school affinity matching.
   827→ */
   828→export function selectV9Persona(
   829→  instrument: string,
   830→  date: string,
   831→  schoolId: string
   832→): { persona: V9Persona; template: V9TemplateFormat; variationSeed: number } {
   833→  const hashInput = `${date}:${instrument}:v9:2024`;
   834→  const dayHash = simpleHash(hashInput);
   835→
   836→  // Step 1: Find personas matching school affinity
   837→  const matching = PERSONAS.filter(p => p.schoolAffinity === schoolId);
   838→  // Step 2: If no match, use all personas
   839→  const pool = matching.length > 0 ? matching : PERSONAS;
   840→  // Step 3: Select persona by hash
   841→  const personaIndex = dayHash % pool.length;
   842→  const persona = pool[personaIndex];
   843→
   844→  // Step 4: Select template — vary by hash % 7
   845→  let templateIndex: number;
   846→  const templateMod = dayHash % 7;
   847→  if (templateMod < 2) templateIndex = 1; // technical (most common)
   848→  else if (templateMod < 3) templateIndex = 0; // executive
   849→  else if (templateMod < 4) templateIndex = 3; // forecast
   850→  else if (templateMod < 5) templateIndex = 4; // educational
   851→  else templateIndex = 2; // trading
   852→  // Occasional variation: hash % 5 == 0 → use a different template
   853→  if (dayHash % 5 === 0) {
   854→    templateIndex = dayHash % TEMPLATE_FORMATS.length;
   855→  }
   856→  const template = TEMPLATE_FORMATS[templateIndex];
   857→
   858→  return {
   859→    persona,
   860→    template,
   861→    variationSeed: dayHash,
   862→  };
   863→}

   869→/**
   870→ * Selects N phrases from a section of the phrase bank, rotated by seed.
   871→ * Returns a comma-separated string of selected phrases.
   872→ */
   873→function selectPhrases(section: string, subsection: string, count: number, seed: number): string {
   874→  const sectionData = PHRASE_BANK[section];
   875→  if (!sectionData) return '';
   876→  const phrases = sectionData[subsection] || sectionData[Object.keys(sectionData)[0]] || [];
   877→  if (phrases.length === 0) return '';
   878→  const startIdx = seed % phrases.length;
   879→  const selected: string[] = [];
   880→  for (let i = 0; i < Math.min(count, phrases.length); i++) {
   881→    selected.push(phrases[(startIdx + i) % phrases.length]);
   882→  }
   883→  return selected.join('\n- ');
   884→}
