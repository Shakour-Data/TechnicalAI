    95→        const res = await fetch(`/api/tgju-analysis?key=${encodeURIComponent(params.tgjuKey)}`);
    96→        if (res.ok) {
    97→          const json = await res.json();
    98→          if (json.candles && json.candles.length > 0 && json.ta) {
    99→            setData({ ...json, isTgju: true, category: params.category });
   100→          }
   101→        }
   102→        return;
   103→      }
   104→      // TSE Index (finpy-tse)
   105→      if (params.category === 'tse_index' && params.insCode) {
   106→        const res = await fetch(`/api/index-analysis?webId=${encodeURIComponent(params.insCode)}`);
   107→        if (res.ok) {
   108→          const json = await res.json();
   109→          if (json.candles && json.candles.length > 0 && json.ta) {
   110→            setData({ ...json, isIndex: true, category: params.category });
   111→          }
   112→        }
   113→        return;
   114→      }
   115→
   116→      // Regular TSE
   117→      const res = await fetch(`/api/analysis?symbol=${encodeURIComponent(params.symbol)}`);
   118→      if (res.ok) {
   119→        const json = await res.json();
   120→        setData(json);
   121→      }
   122→    } finally {
   123→      refreshingRef.current = false;
   124→      setRefreshing(false);
   125→    }
   126→  }, []);

   132→    return () => clearInterval(id);
   133→  }, [doRefresh]);

   135→  // ── Refresh on tab focus ──

   139→    return () => window.removeEventListener('focus', onFocus);
   140→  }, [doRefresh]);

   160→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');

   163→        setActivePanel('visual'); setCurrentPage('analysis');

   165→        setError(String(err));

   167→        setLoading(false);
   168→        setLoadingMessage(null);
   169→      }
   170→      return;

   182→          const err = await res.json();
   183→          throw new Error(err.error || 'خطا در دریافت داده‌های شاخص');

   190→        setActivePanel('visual'); setCurrentPage('analysis');

   192→        setError(String(err));

   194→        setLoading(false);
   195→        setLoadingMessage(null);

   212→      setActivePanel('visual'); setCurrentPage('analysis');

   214→      setError(String(err));

   216→      setLoading(false);

   273→                    {data.info.change >= 0 ? '▲' : '▼'} {toFa(Math.abs(data.info.change))}%

   276→              </div>
   277→              {isTgjuData && (
   278→                <span className="px-2 py-1 rounded-lg bg-teal-50 border border-teal-200 text-teal-700 text-[10px] font-bold flex items-center gap-1">
   279→                  <Coins className="w-3 h-3" /> TGJU

   284→                  <BarChart3 className="w-3 h-3" /> شاخص (finpy-tse)

   286→              )}
   287→              <div className={`px-3 py-1.5 rounded-lg border ${signalBg}`}>
   288→                <div className={`flex items-center gap-1.5 text-xs font-bold ${signalColor}`}>

   301→            </div>

   303→        </div>
   304→      </header>
   305→
   306→      {/* ── MAIN CONTENT ──────────────────────────────────────── */}
   307→      <main className="flex-1 max-w-[1600px] w-full mx-auto px-4 py-4">
   308→        {/* ── Help Page ── */}

   310→          setCurrentPage('analysis');
   311→          setTimeout(() => {
   312→            const searchInput = document.querySelector('input[placeholder*="جستجو"]') as HTMLInputElement;
   313→            if (searchInput) searchInput.focus();
   314→          }, 100);
   315→        }} />}
   316→
   317→        {/* ── Documentation Page ── */}
   318→        {currentPage === 'docs' && <DocsPage />}
   319→
   320→        {/* ── Index Manager Page ── */}
   321→        {currentPage === 'indices' && <IndexManager onBack={() => setCurrentPage('home')} />}
   322→
   323→        {/* ── Home Page ── */}
   324→        {currentPage === 'home' && !data && !loading && !error && (
   325→          <Homepage onSelectCategory={(cat) => {
   326→            if (cat === 'indices') {
   327→              setCurrentPage('indices');
   328→              return;
   329→            }
   330→            const searchInput = document.querySelector('input[placeholder*="جستجو"]') as HTMLInputElement;
   331→            if (searchInput) {
   332→              searchInput.focus();
   333→            }
   334→          }} />

   336→
   337→        {/* ── Analysis View ── */}

   347→            <Skeleton className="h-[500px] w-full bg-gray-200 rounded-xl" />
   348→            <div className="grid grid-cols-3 gap-4">
