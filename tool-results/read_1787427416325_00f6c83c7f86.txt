   230→function parsePersianNum(str: string | null | undefined): number {
   231→  if (!str) return 0;
   232→  return Number(str.replace(/[,۰-۹]/g, (c) => {
   233→    const persianDigits = '۰۱۲۳۴۵۶۷۸۹';
   234→    const idx = persianDigits.indexOf(c);
   235→    return idx >= 0 ? String(idx) : c === ',' ? '' : c;
   236→  }));
   237→}

   272→    if (result.size === keys.length) return result;
   273→  }
   274→
   275→  const result = new Map<string, StaticPriceInfo>();
   276→
   277→  // Skip page_reader batch to preserve rate limit for TSE index data

   326→/* ─── Fetch live instruments (currency + gold from tgju-api) + static list ── */
   327→
   328→export async function fetchTgjuInstruments(): Promise<TgjuInstrument[]> {
   329→  const now = Date.now();
   330→  if (liveInstrumentsCache.data && now - liveInstrumentsCache.ts < INSTRUMENTS_TTL) {
   331→    return liveInstrumentsCache.data;
   332→  }
   333→
   334→  const controller = new AbortController();
   335→  const timeout = setTimeout(() => controller.abort(), 10000);
   336→
   337→  try {
   338→    const [currencyRes, goldRes] = await Promise.all([
   339→      fetch(`${TGJU_API_BASE}/api/price/currency`, { signal: controller.signal }),
   340→      fetch(`${TGJU_API_BASE}/api/price/gold`, { signal: controller.signal }),
   341→    ]);
   342→
   343→    clearTimeout(timeout);
   344→
   345→    const instruments: TgjuInstrument[] = [];
   346→    const currencyGoldKeys: string[] = []; // keys to fetch chart data for accurate change
   347→
   348→    if (currencyRes.ok) {
   349→      const currencies: TgjuPriceItem[] = await currencyRes.json();
   350→      for (const c of currencies) {
   351→        const price = parsePersianNum(c.price);
   352→        const low = parsePersianNum(c.low_price);
   353→        const high = parsePersianNum(c.high_price);
   354→        instruments.push({
   355→          title: c.title,
   356→          key: c.key,
   357→          price,
   358→          highPrice: high,
   359→          lowPrice: low,
   360→          change: 0, // Will be updated from chart data
   361→          changePercent: 0, // Will be updated from chart data
   362→          category: 'currency',
   363→        });
   364→        currencyGoldKeys.push(c.key);
   365→      }
   366→    }
   367→
   368→    if (goldRes.ok) {
   369→      const goldCategories: TgjuGoldCategory[] = await goldRes.json();
   370→      for (const cat of goldCategories) {
   371→        for (const item of cat.prices) {
   372→          const price = parsePersianNum(item.price);
   373→          const low = parsePersianNum(item.low_price);
   374→          const high = parsePersianNum(item.high_price);
   375→
   376→          let category: TgjuCategoryType = 'gold';
   377→          if (item.key.startsWith('ime_fund_')) category = 'gold_etf';
   378→          else if (item.key.startsWith('silver_')) category = 'silver';
   379→
   380→          instruments.push({
   381→            title: item.title,
   382→            key: item.key,
   383→            price,
   384→            highPrice: high,
   385→            lowPrice: low,
   386→            change: 0, // Will be updated from chart data
   387→            changePercent: 0, // Will be updated from chart data
   388→            category,
   389→            groupTitle: cat.title,
   390→          });
   391→          currencyGoldKeys.push(item.key);
   392→        }
   393→      }
   394→    }
   395→
   396→    // Add static instruments (prices will be populated by background batch fetch)
   397→    for (const s of STATIC_INSTRUMENTS) {
   398→      const cached = staticPriceCache.data.get(s.key);
   399→      instruments.push({
   400→        title: s.title,
   401→        key: s.key,
   402→        price: cached?.close ?? 0,
   403→        highPrice: cached?.high ?? 0,
   404→        lowPrice: cached?.low ?? 0,
   405→        change: cached ? (cached.close - cached.prevClose) : 0,
   406→        changePercent: cached && cached.prevClose > 0 ? ((cached.close - cached.prevClose) / cached.prevClose) * 100 : 0,
   407→        category: s.category,
   408→        groupTitle: s.groupTitle,
   409→      });
   410→    }
   411→
   412→    liveInstrumentsCache.data = instruments;
   413→    liveInstrumentsCache.ts = now;
   414→
   415→    // ── Background: fetch chart data for ALL TGJU keys ──
   416→    // This gives us accurate prevClose for currency/gold (instead of using intraday low)
   417→    // and populates prices for static instruments
   418→    const allKeys = [
   419→      ...currencyGoldKeys,
   420→      ...STATIC_INSTRUMENTS.map(s => s.key),
   421→    ];
   422→    batchFetchChartPrices(allKeys).then((priceMap) => {
   423→      if (priceMap.size > 0 && liveInstrumentsCache.data) {
   424→        for (const inst of liveInstrumentsCache.data) {
   425→          const p = priceMap.get(inst.key);
   426→          if (!p) continue;
   427→
   428→          if (inst.category === 'currency' || inst.category === 'gold' ||
   429→              inst.category === 'silver' || inst.category === 'gold_etf') {
   430→            // Currency/gold: keep live price, but use chart prevClose for accurate change
   431→            if (p.prevClose > 0) {
   432→              inst.change = inst.price - p.prevClose;
   433→              inst.changePercent = (inst.change / p.prevClose) * 100;
   434→            }
   435→          } else if (inst.price === 0) {
   436→            // Static instruments: update all fields from chart data
   437→            inst.price = p.close;
   438→            inst.highPrice = p.high;
   439→            inst.lowPrice = p.low;
   440→            inst.change = p.close - p.prevClose;
   441→            inst.changePercent = p.prevClose > 0 ? ((p.close - p.prevClose) / p.prevClose) * 100 : 0;
   442→          }
   443→        }
   444→      }
   445→    }).catch(() => {});
   446→
   447→    return instruments;
   448→  } catch {
   449→    clearTimeout(timeout);
   450→    return liveInstrumentsCache.data || [];
   451→  }
   452→}

   454→/* ─── Fetch historical OHLC data via z-ai page_reader ── */
   455→
   456→export async function fetchTgjuHistory(tgjuKey: string): Promise<TgjuOHLC[]> {
   457→  const cached = historyCache.get(tgjuKey);
   458→  const now = Date.now();
   459→  if (cached && now - cached.ts < HISTORY_TTL) {
   460→    return cached.data;
   461→  }
   462→
   463→  // Historical data requires page_reader (rate-limited) — on-demand fetch per user click
   464→  try {
   465→    const { rateLimitedPageReader } = await import('@/lib/zai-shared');
   466→
   467→    const url = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=5000`;
   468→
   469→    const html = await rateLimitedPageReader(url, 30_000);
   470→    const jsonStr = html.replace(/<[^>]+>/g, '').trim();
   471→
   472→    if (!jsonStr.startsWith('{')) {
   473→      return cached?.data || [];
   474→    }
   475→
   476→    const data = JSON.parse(jsonStr);
   477→    const rows: string[][] = data.data;
   478→
   479→    if (!rows || rows.length === 0) {
   480→      return cached?.data || [];
   481→    }
   482→
   483→    // Parse: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   484→    const candles: TgjuOHLC[] = rows
   485→      .map((row) => ({
   486→        date: (row[6] || '').replace(/\//g, '-'),
   487→        open: parsePersianNum(row[0]),
   488→        high: parsePersianNum(row[2]),
   489→        low: parsePersianNum(row[1]),
   490→        close: parsePersianNum(row[3]),
   491→      }))
   492→      .filter((c) => c.open > 0 && c.high > 0 && c.low > 0 && c.close > 0)
   493→      .reverse(); // newest-last for charting
   494→
   495→    historyCache.set(tgjuKey, { data: candles, ts: now });
   496→    return candles;
   497→  } catch (err) {
   498→    console.error(`[TGJU] Failed to fetch history for ${tgjuKey}:`, err);
   499→    return cached?.data || [];
   500→  }
   501→}
