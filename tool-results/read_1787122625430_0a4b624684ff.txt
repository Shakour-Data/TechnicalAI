   117→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   118→function sma(closes: number[], period: number): number {
   119→  if (closes.length < period) return 0;
   120→  const slice = closes.slice(-period);
   121→  return slice.reduce((a, b) => a + b, 0) / period;
   122→}

   124→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   125→function emaCalc(closes: number[], period: number): number {
   126→  if (closes.length < period) return 0;
   127→  const k = 2 / (period + 1);
   128→  let emaVal = sma(closes.slice(0, period), period);
   129→  for (let i = period; i < closes.length; i++) {
   130→    emaVal = closes[i] * k + emaVal * (1 - k);
   131→  }
   132→  return emaVal;
   133→}

   135→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   136→function calcRSI(closes: number[], period: number = 14): number {
   137→  if (closes.length < period + 1) return 50;
   138→  let avgGain = 0;
   139→  let avgLoss = 0;
   140→  for (let i = 1; i <= period; i++) {
   141→    const diff = closes[i] - closes[i - 1];
   142→    if (diff > 0) avgGain += diff;
   143→    else avgLoss += Math.abs(diff);
   144→  }
   145→  avgGain /= period;
   146→  avgLoss /= period;
   147→  for (let i = period + 1; i < closes.length; i++) {
   148→    const diff = closes[i] - closes[i - 1];
   149→    const gain = diff > 0 ? diff : 0;
   150→    const loss = diff < 0 ? Math.abs(diff) : 0;
   151→    avgGain = (avgGain * (period - 1) + gain) / period;
   152→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   153→  }
   154→  if (avgLoss === 0) return 100;
   155→  const rs = avgGain / avgLoss;
   156→  return 100 - 100 / (1 + rs);
   157→}

   159→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   160→function calcMFI(data: OHLCV[], period: number = 14): number {
   161→  if (data.length < period + 1) return 50;
   162→  let posFlow = 0;
   163→  let negFlow = 0;
   164→  for (let i = data.length - period; i < data.length; i++) {
   165→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   166→    const mf = tp * data[i].volume;
   167→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   168→    if (tp > prevTp) posFlow += mf;
   169→    else negFlow += mf;
   170→  }
   171→  if (negFlow === 0) return 100;
   172→  const mfr = posFlow / negFlow;
   173→  return 100 - 100 / (1 + mfr);
   174→}

   176→// ─── Helper: CCI(20) ──────────────────────────────────────────────────────────
   177→function calcCCI(data: OHLCV[], period: number = 20): number {
   178→  if (data.length < period) return 0;
   179→  const tps: number[] = [];
   180→  for (let i = data.length - period; i < data.length; i++) {
   181→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   182→  }
   183→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   184→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   185→  if (meanDev === 0) return 0;
   186→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   187→}

   189→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   190→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   191→  if (data.length < kPeriod) return { k: 50, d: 50 };
   192→  const rawKs: number[] = [];
   193→  for (let i = kPeriod - 1; i < data.length; i++) {
   194→    let lowest = Infinity;
   195→    let highest = -Infinity;
   196→    for (let j = i - kPeriod + 1; j <= i; j++) {
   197→      if (data[j].low < lowest) lowest = data[j].low;
   198→      if (data[j].high > highest) highest = data[j].high;
   199→    }
   200→    const range = highest - lowest;
   201→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   202→  }
   203→  const smoothedK: number[] = [];
   204→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   205→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   206→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   207→  }
   208→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   209→  const dSlice = smoothedK.slice(-smoothD);
   210→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   211→  return { k, d };
   212→}

   214→// ─── Helper: Williams %R(14) ──────────────────────────────────────────────────
   215→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   216→  if (data.length < period) return -50;
   217→  const slice = data.slice(-period);
   218→  let highest = -Infinity;
   219→  let lowest = Infinity;
   220→  for (const c of slice) {
   221→    if (c.high > highest) highest = c.high;
   222→    if (c.low < lowest) lowest = c.low;
   223→  }
   224→  const range = highest - lowest;
   225→  if (range === 0) return -50;
   226→  return ((highest - data[data.length - 1].close) / range) * -100;
   227→}

   229→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   230→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   231→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   232→  const emaFast = emaCalc(closes, fast);
   233→  const emaSlow = emaCalc(closes, slow);
   234→  const macdLine = emaFast - emaSlow;
   235→  // Calculate signal as EMA of MACD values
   236→  const macdValues: number[] = [];
   237→  const kFast = 2 / (fast + 1);
   238→  const kSlow = 2 / (slow + 1);
   239→  let eFast = sma(closes.slice(0, fast), fast);
   240→  let eSlow = sma(closes.slice(0, slow), slow);
   241→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   242→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   243→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   244→    macdValues.push(eFast - eSlow);
   245→  }
   246→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   247→  const kSig = 2 / (sig + 1);
   248→  let signalVal = sma(macdValues.slice(0, sig), sig);
   249→  for (let i = sig; i < macdValues.length; i++) {
   250→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   251→  }
   252→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   253→}

   255→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   256→function calcATR(data: OHLCV[], period: number = 14): number {
   257→  if (data.length < 2) return 0;
   258→  const trs: number[] = [];
   259→  for (let i = 1; i < data.length; i++) {
   260→    const tr = Math.max(
   261→      data[i].high - data[i].low,
   262→      Math.abs(data[i].high - data[i - 1].close),
   263→      Math.abs(data[i].low - data[i - 1].close)
   264→    );
   265→    trs.push(tr);
   266→  }
   267→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   268→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   269→  for (let i = period; i < trs.length; i++) {
   270→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   271→  }
   272→  return atrVal;
   273→}

   275→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   276→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   277→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   278→  const plusDMs: number[] = [];
   279→  const minusDMs: number[] = [];
   280→  const trs: number[] = [];
   281→  for (let i = 1; i < data.length; i++) {
   282→    const upMove = data[i].high - data[i - 1].high;
   283→    const downMove = data[i - 1].low - data[i].low;
   284→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   285→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   286→    trs.push(Math.max(
   287→      data[i].high - data[i].low,
   288→      Math.abs(data[i].high - data[i - 1].close),
   289→      Math.abs(data[i].low - data[i - 1].close)
   290→    ));
   291→  }
   292→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   293→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   294→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   295→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   296→  const dxValues: number[] = [];
   297→  for (let i = period; i < trs.length; i++) {
   298→    smoothTR = smoothTR - smoothTR / period + trs[i];
   299→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   300→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   301→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   302→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   303→    const diSum = diPlus + diMinus;
   304→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   305→    dxValues.push(dx);
   306→  }
   307→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   308→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   309→  for (let i = period; i < dxValues.length; i++) {
   310→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   311→  }
   312→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   313→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   314→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   315→}

   317→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   318→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   319→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   320→  let isLong = true;
   321→  let af = step;
   322→  let ep = data[0].high;
   323→  let sar = data[0].low;
   324→  for (let i = 1; i < data.length; i++) {
   325→    const prevSar = sar;
   326→    sar = prevSar + af * (ep - prevSar);
   327→    // Ensure SAR is within bounds
   328→    if (isLong) {
   329→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   330→      if (data[i].low < sar) {
   331→        isLong = false;
   332→        sar = ep;
   333→        ep = data[i].low;
   334→        af = step;
   335→      } else {
   336→        if (data[i].high > ep) {
   337→          ep = data[i].high;
   338→          af = Math.min(af + step, max);
   339→        }
   340→      }
   341→    } else {
   342→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   343→      if (data[i].high > sar) {
   344→        isLong = true;
   345→        sar = ep;
   346→        ep = data[i].high;
   347→        af = step;
   348→      } else {
   349→        if (data[i].low < ep) {
   350→          ep = data[i].low;
   351→          af = Math.min(af + step, max);
   352→        }
   353→      }
   354→    }
   355→  }
   356→  return sar;
   357→}

   359→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   360→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   361→  if (closes.length < period) {
   362→    const c = closes[closes.length - 1] ?? 0;
   363→    return { upper: c, middle: c, lower: c };
   364→  }
   365→  const slice = closes.slice(-period);
   366→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   367→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   368→  const sd = Math.sqrt(variance);
   369→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   370→}

   373→function calcOBV(data: OHLCV[]): number {

   382→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   383→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   384→  const n = values.length;
   385→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   386→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   387→  for (let i = 0; i < n; i++) {
   388→    sumX += i;
   389→    sumY += values[i];
   390→    sumXY += i * values[i];
   391→    sumX2 += i * i;
   392→    sumY2 += values[i] * values[i];
   393→  }
   394→  const denom = n * sumX2 - sumX * sumX;
   395→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   396→  const slope = (n * sumXY - sumX * sumY) / denom;
   397→  const intercept = (sumY - slope * sumX) / n;
   398→  const meanY = sumY / n;
   399→  let ssRes = 0, ssTot = 0;
   400→  for (let i = 0; i < n; i++) {
   401→    const predicted = slope * i + intercept;
   402→    ssRes += (values[i] - predicted) ** 2;
   403→    ssTot += (values[i] - meanY) ** 2;
   404→  }
   405→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   406→  return { slope, intercept, r2 };
   407→}

   409→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   410→function roundToNice(price: number): number {
   411→  const abs = Math.abs(price);
   412→  let step: number;
   413→  if (abs >= 100000) step = 100;
   414→  else if (abs >= 10000) step = 50;
   415→  else if (abs >= 1000) step = 10;
   416→  else if (abs >= 100) step = 5;
   417→  else step = 1;
   418→  return Math.round(price / step) * step;
   419→}

   421→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
   422→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: number[]; lows: number[] } {
   423→  const highs: number[] = [];
   424→  const lows: number[] = [];
   425→  for (let i = lookback; i < data.length - lookback; i++) {
   426→    let isHigh = true;
   427→    let isLow = true;
   428→    for (let j = i - lookback; j <= i + lookback; j++) {
   429→      if (j === i) continue;
   430→      if (data[j].high >= data[i].high) isHigh = false;
   431→      if (data[j].low <= data[i].low) isLow = false;
   432→    }
   433→    if (isHigh) highs.push(data[i].high);
   434→    if (isLow) lows.push(data[i].low);
   435→  }
   436→  return { highs, lows };
   437→}

   439→// ─── Support / Resistance (6 each, adaptive gap, multi-source, 10% window strength) ─
   440→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   441→  resistances: number[];
   442→  supports: number[];
   443→  supportStrengths: LevelStrength[];
   444→  resistanceStrengths: LevelStrength[];
   445→  priceTargets: LevelStrength[];
   446→} {
   447→  const allRawLevels: number[] = [];
   448→  const closes = data.map(d => d.close);
   449→
   450→  // 1. Pivot Points from previous day (5R + 5S)
   451→  const prev = data[data.length - 2] ?? data[data.length - 1];
   452→  if (prev) {
   453→    const pp = (prev.high + prev.low + prev.close) / 3;
   454→    const hl = prev.high - prev.low;
   455→    const pivotR = [
   456→      2 * pp - prev.low,
   457→      pp + hl,
   458→      prev.high + 2 * (pp - prev.low),
   459→      prev.high + 2 * (pp - prev.low) + hl,
   460→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   461→    ];
   462→    const pivotS = [
   463→      2 * pp - prev.high,
   464→      pp - hl,
   465→      prev.low - 2 * (prev.high - pp),
   466→      prev.low - 2 * (prev.high - pp) - hl,
   467→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   468→    ];
   469→    allRawLevels.push(...pivotR, ...pivotS);
   470→  }
   471→
   472→  // 2. Swing levels (multiple lookback periods for better coverage)
   473→  for (const lb of [3, 5, 7, 10]) {
   474→    const swings = findSwingLevels(data, lb);
   475→    allRawLevels.push(...swings.highs, ...swings.lows);
   476→  }
   477→
   478→  // 3. Moving Average levels (potential dynamic S/R)
   479→  for (const period of [5, 10, 21, 50, 100, 200]) {
   480→    const v = sma(closes, period);
   481→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01) allRawLevels.push(v);
   482→  }
   483→  const ema12 = emaCalc(closes, 12);
   484→  const ema26 = emaCalc(closes, 26);
   485→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01) allRawLevels.push(ema12);
   486→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01) allRawLevels.push(ema26);
   487→
   488→  // 4. Bollinger Band levels
   489→  const bbLevels = calcBollingerBands(closes);
   490→  if (bbLevels.upper > 0) allRawLevels.push(bbLevels.upper);
   491→  if (bbLevels.lower > 0) allRawLevels.push(bbLevels.lower);
   492→
   493→  // 5. Round number / psychological levels
   494→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
   495→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
   496→    allRawLevels.push(Math.round(currentPrice * mult / magnitude) * magnitude);
   497→  }
   498→
   499→  // 6. Recent high/low zones (5-day, 22-day, 66-day)
   500→  for (const window of [5, 22, 66]) {
   501→    const slice = data.slice(-Math.min(window, data.length));
   502→    if (slice.length > 0) {
   503→      allRawLevels.push(Math.max(...slice.map(d => d.high)));
   504→      allRawLevels.push(Math.min(...slice.map(d => d.low)));
   505→    }
   506→  }
   507→
   508→  // 7. Round all and filter invalid
   509→  const rounded = allRawLevels.map(roundToNice).filter(l => l > 0);
   510→
   511→  // 8. Remove duplicates (within 0.5%)
   512→  const unique: number[] = [];
   513→  for (const level of rounded) {
   514→    const isDup = unique.some(u => Math.abs(u - level) / Math.max(u, 1) < 0.005);
   515→    if (!isDup) unique.push(level);
   516→  }
   517→
   518→  // 9. Calculate strength for each unique level — LOGICAL MULTI-FACTOR FORMULA
   519→  //
   520→  // Formula: strength = min(10, max(1, round(
   521→  //   touchFactor × convergenceFactor × recencyFactor × proximityFactor
   522→  // )))
   523→  //
   524→  // touchFactor: How many times price touched (bounced off) this level.
   525→  //   Count candles where high/low is within 0.5% of the level.
   526→  //   More touches = stronger level. Scale: 1 + touches * 0.3 (max 3.0)
   527→  //
   528→  // convergenceFactor: How many different sources identified this same level.
   529→  //   Count of raw levels within 1% of this level. More convergence = stronger.
   530→  //   Scale: 0.5 + convergences * 0.15 (max 2.5)
   531→  //
   532→  // recencyFactor: Recent touches are more relevant than old ones.
   533→  //   Weighted average of touch recency. Scale: 0.5 to 1.5
   534→  //
   535→  // proximityFactor: Distance from current price affects relevance.
   536→  //   Levels too far away are less relevant. Scale: 0.5 to 1.2
   537→  //
   538→  // volumeTouchFactor (when has volume): Average volume at touch points
   539→  //   relative to average volume. Higher volume touches = stronger.
   540→  //   Scale: 0.7 to 1.3
   541→
   542→  const hasVol = data.some(d => d.volume > 0);
   543→  const avgVolume = hasVol ? data.slice(-50).reduce((s, d) => s + d.volume, 0) / Math.min(50, data.length) : 0;
   544→
   545→  const levelStrengths: LevelStrength[] = unique.map(level => {
   546→    // ── touchFactor: count price touches within 0.5% ──
   547→    const touchThreshold = level * 0.005;
   548→    let touchCount = 0;
   549→    let touchRecencySum = 0; // weighted by recency
   550→    let touchVolumeSum = 0;
   551→
   552→    for (let i = 0; i < data.length; i++) {
   553→      const d = data[i];
   554→      const nearHigh = Math.abs(d.high - level) <= touchThreshold;
   555→      const nearLow = Math.abs(d.low - level) <= touchThreshold;
   556→      if (nearHigh || nearLow) {
   557→        touchCount++;
   558→        // Recency: more recent touches get higher weight (0 to 1, linear)
   559→        const recencyWeight = (i + 1) / data.length;
   560→        touchRecencySum += recencyWeight;
   561→        // Volume at touch
   562→        if (hasVol && d.volume > 0) {
   563→          touchVolumeSum += d.volume;
   564→        }
   565→      }
   566→    }
   567→
   568→    const touchFactor = Math.min(3.0, 1.0 + touchCount * 0.3);
   569→
   570→    // ── convergenceFactor: how many raw levels cluster near this one ──
   571→    const convergenceCount = unique.filter(other =>
   572→      other !== level && Math.abs(other - level) / Math.max(level, 1) < 0.01
   573→    ).length;
   574→    const convergenceFactor = Math.min(2.5, 0.5 + convergenceCount * 0.15);
   575→
   576→    // ── recencyFactor: weighted average of how recent the touches are ──
   577→    const recencyFactor = touchCount > 0
   578→      ? Math.min(1.5, 0.5 + (touchRecencySum / touchCount) * 1.0)
   579→      : 0.7;
   580→
   581→    // ── proximityFactor: closer to current price = more relevant ──
   582→    const distFromPrice = Math.abs(level - currentPrice) / currentPrice;
   583→    const proximityFactor = Math.min(1.2, Math.max(0.5, 1.2 - distFromPrice * 3));
   584→
   585→    // ── volumeTouchFactor: volume at touches vs average ──
   586→    let volumeTouchFactor = 1.0;
   587→    if (hasVol && touchCount > 0 && avgVolume > 0) {
   588→      const avgTouchVolume = touchVolumeSum / touchCount;
   589→      const volRatio = avgTouchVolume / avgVolume;
   590→      volumeTouchFactor = Math.min(1.3, Math.max(0.7, 0.7 + volRatio * 0.3));
   591→    }
   592→
   593→    // ── Final strength calculation ──
   594→    const rawStrength = touchFactor * convergenceFactor * recencyFactor * proximityFactor * volumeTouchFactor;
   595→    const strength = Math.min(10, Math.max(1, Math.round(rawStrength)));
   596→
   597→    return { price: level, strength, isTarget: false };
   598→  });
   599→
   600→  // 10. Separate above/below current price
   601→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   602→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   603→
   604→  // 11. Enforce 2%-5% gap, always return exactly targetCount levels
   605→  //    Step 1: Merge levels closer than 2%
   606→  //    Step 2: Walk from nearest level, keep only chain within 2-5% gap
   607→  //    Step 3: If fewer than targetCount, extend chain by computing ~3% steps outward
   608→  //    Step 4: If more than targetCount, trim by keeping highest strength
   609→  function enforceGap(
   610→    levels: LevelStrength[],
   611→    targetCount: number,
   612→    _allUnique: LevelStrength[],
   613→  ): LevelStrength[] {
   614→    if (levels.length === 0) return [];
   615→
   616→    const MIN_GAP = 0.02;
   617→    const MAX_GAP = 0.05;
   618→    const EXTEND_GAP = 0.035; // 3.5% — middle of 2-5% range
   619→    const step = currentPrice >= 100000 ? 100 : currentPrice >= 10000 ? 50 : currentPrice >= 1000 ? 10 : currentPrice >= 100 ? 5 : 1;
   620→    const isDescending = levels.length > 0 && levels[0].price > (levels[levels.length - 1]?.price ?? 0);
   621→
   622→    // Step 1: Merge levels closer than MIN_GAP
   623→    const merged: LevelStrength[] = [levels[0]];
   624→    for (let i = 1; i < levels.length; i++) {
   625→      const prevPrice = merged[merged.length - 1].price;
   626→      const currPrice = levels[i].price;
   627→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   628→      if (gap < MIN_GAP) {
   629→        const mergedStrength = Math.min(10, Math.max(merged[merged.length - 1].strength, levels[i].strength) + 1);
   630→        merged[merged.length - 1] = {
   631→          ...(levels[i].strength > merged[merged.length - 1].strength ? levels[i] : merged[merged.length - 1]),
   632→          strength: mergedStrength,
   633→        };
   634→      } else {
   635→        merged.push(levels[i]);
   636→      }
   637→    }
   638→
   639→    // Step 2: Walk from first level (nearest to price), keep only if gap is <= MAX_GAP
   640→    const chain: LevelStrength[] = [merged[0]];
   641→    for (let i = 1; i < merged.length; i++) {
   642→      const prevPrice = chain[chain.length - 1].price;
   643→      const currPrice = merged[i].price;
   644→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   645→      if (gap <= MAX_GAP) {
   646→        chain.push(merged[i]);
   647→      }
   648→    }
   649→
   650→    // Step 3: If fewer than targetCount, extend the chain outward at ~3.5% intervals
   651→    while (chain.length < targetCount) {
   652→      const lastPrice = chain[chain.length - 1].price;
   653→      const nextRaw = isDescending
   654→        ? lastPrice * (1 - EXTEND_GAP)  // supports go down
   655→        : lastPrice * (1 + EXTEND_GAP); // resistances go up
   656→      const nextPrice = Math.round(nextRaw / step) * step;
   657→      if (nextPrice <= 0) break;
   658→      if (nextPrice === lastPrice) break; // rounding made no change (very low price)
   659→      chain.push({ price: nextPrice, strength: 2, isTarget: false });
   660→    }
   661→
   662→    // Step 4: If more than targetCount, trim by keeping highest strength
   663→    if (chain.length > targetCount) {
   664→      chain.sort((a, b) => b.strength - a.strength);
   665→      const kept = chain.slice(0, targetCount);
   666→      kept.sort((a, b) => isDescending ? b.price - a.price : a.price - b.price);
   667→      return kept;
   668→    }
   669→
   670→    return chain;
   671→  }
   672→
   673→  const finalResistances = enforceGap(above, 6, levelStrengths);
   674→  const finalSupports = enforceGap(below, 6, levelStrengths);
   675→
   676→  // 12. Determine price targets: strength > 7 is a target
   677→  const allLevels = [...finalResistances, ...finalSupports];
   678→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   679→  let targets: LevelStrength[];
   680→  if (sortedByStrength.some(l => l.strength > 7)) {
   681→    targets = sortedByStrength.filter(l => l.strength > 7);
   682→  } else {
   683→    targets = sortedByStrength.slice(0, 2);
   684→  }
   685→
   686→  // Mark targets
   687→  targets = targets.map(t => ({ ...t, isTarget: true }));
   688→
   689→  // Also mark in final arrays
   690→  finalResistances.forEach(r => {
   691→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   692→  });
   693→  finalSupports.forEach(s => {
   694→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   695→  });
   696→
   697→  return {
   698→    resistances: finalResistances.map(l => l.price),
   699→    supports: finalSupports.map(l => l.price),
   700→    supportStrengths: finalSupports,
   701→    resistanceStrengths: finalResistances,
   702→    priceTargets: targets,
   703→  };
   704→}

   706→// ─── Trend Analysis ───────────────────────────────────────────────────────────
   707→function calcTrend(closes: number[], period: number): TrendResult {
   708→  const slice = closes.slice(-Math.min(period, closes.length));
   709→  const { slope, r2 } = linearRegression(slice);
   710→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   711→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   712→  const priceThreshold = avgPrice * 0.001; // 0.1%
   713→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   714→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   715→}

   877→  // ── LAYER 2: Trend-Adjusted Scores ────────────────────────────────────
   878→  const f_rsi = clamp(s_rsi + IT.rsi * 0.08, 0, 1);

   881→  const f_macd = clamp(s_macd + IT.macd * 0.08, 0, 1);
   882→  const f_stoch = clamp(s_stoch + IT.stoch * 0.06, 0, 1);
   883→  const f_bb = clamp(s_bb + IT.bb * 0.04, 0, 1);

   893→  let divergence = 0;

   906→  // ── LAYER 3: ML-Optimized Weighted Bull Consensus ─────────────────────

   949→        }
   950→
   951→        // Add cross-signal weights (not in ML training)
   952→        mlWeightMap.stochCross = 6;
   953→        mlWeightMap.macdCross = 6;
   954→        mlWeightMap.div = 8;
   955→
   956→        // Don't override MFI weight if no volume
   957→        if (!hasVolume) delete mlWeightMap.mfi;
   958→
   959→        activeWeights = mlWeightMap;
   960→        mlResult = {
   961→          weightsUsed: true,
   962→          indicatorWeights: mlWeights.normalizedWeights,
   963→          mlR2: mlWeights.r2Score,
   964→          predictionUsed: false,
   965→          predictedReturn5d: 0,
   966→          predictionConfidence: 0,
   967→          mlScore: 0.5,
   968→        };
   969→      }
   970→
   971→      // ML Price Prediction (L6)
   972→      if (data.length >= 40) {

   978→            mlResult.predictedReturn5d = prediction.predictedReturn5d;
   979→            mlResult.predictionConfidence = prediction.confidence;
   980→            mlResult.mlScore = mlScore;

   983→              weightsUsed: false,
   984→              indicatorWeights: {},
   985→              mlR2: 0,
   986→              predictionUsed: true,
   987→              predictedReturn5d: prediction.predictedReturn5d,
   988→              predictionConfidence: prediction.confidence,
   989→              mlScore,
   990→            };
   991→          }
   992→        }
   993→      }

   996→    }
   997→  }

  1023→  // Blend ML prediction score with bull consensus (20% weight for ML)

  1030→  const bearScore = 1 - bullConsensus;
  1031→
  1032→  // ── LAYER 4: 5-Scenario Probability Distribution ──────────────────────

  1036→  const oversoldBounce = (rsi < 30 || (hasVolume && mfi < 20) || stoch.k < 20) ? 1 : 0;
  1037→

  1041→  const raw_R4 = (1 - bullConsensus) * distS1 * 0.7 + overboughtRisk * 0.5;
  1042→  const raw_R5 = (1 - bullConsensus) ** 2 * 0.5 + (price < ma100v ? 0.2 : 0);

  1046→  const pR1 = rawSum === 0 ? 20 : Math.round(raw_R1 / rawSum * 100);
  1047→  const pR2 = rawSum === 0 ? 20 : Math.round(raw_R2 / rawSum * 100);
  1048→  const pR3 = rawSum === 0 ? 20 : Math.round(raw_R3 / rawSum * 100);
  1049→  const pR4 = rawSum === 0 ? 20 : Math.round(raw_R4 / rawSum * 100);
  1050→  const pR5 = 100 - (pR1 + pR2 + pR3 + pR4); // remainder ensures sum=100
  1051→
