   171→def fetch_via_proxy(web_id: str, timeout: int = 180) -> str:
   172→    """
   173→    Fetch raw JSON from cdn.tsetmc.com via the Next.js proxy.
   174→    The proxy uses z-ai SDK page_reader (the only way to reach Iran-only CDN).
   175→    Retries on transient errors (500, 502, 504).
   176→    """
   177→    url = f'{NEXTJS_PROXY}?webId={web_id}'
   178→    max_retries = 3
   179→    last_error = ''
   180→    
   181→    for attempt in range(1, max_retries + 1):
   182→        try:
   183→            print(f'[PROXY] Attempt {attempt}/{max_retries} for webId={web_id}...')
   184→            resp = req_lib.get(url, timeout=timeout)
   185→            if resp.status_code != 200:
   186→                body = resp.text[:200]
   187→                if resp.status_code in (500, 502, 504) and attempt < max_retries:
   188→                    print(f'[PROXY] Status {resp.status_code}, retrying in 10s...')
   189→                    last_error = f'Proxy returned status {resp.status_code}: {body}'
   190→                    time.sleep(10)
   191→                    continue
   192→                raise Exception(f'Proxy returned status {resp.status_code}: {body}')
   193→            data = resp.json()
   194→            if data.get('error'):
   195→                if attempt < max_retries:
   196→                    print(f'[PROXY] Error from proxy, retrying in 10s: {data["error"][:100]}')
   197→                    last_error = f"Proxy error: {data['error']}"
   198→                    time.sleep(10)
   199→                    continue
   200→                raise Exception(f"Proxy error: {data['error']}")
   201→            raw = data.get('raw', '')
   202→            if not raw or len(raw) < 10:
   203→                if attempt < max_retries:
   204→                    print(f'[PROXY] Empty raw data (len={len(raw)}), retrying in 10s...')
   205→                    last_error = f'Empty raw data (len={len(raw)})'
   206→                    time.sleep(10)
   207→                    continue
   208→                raise Exception(f'Empty raw data from proxy (len={len(raw)})')
   209→            return raw
   210→        except req_lib.exceptions.Timeout:
   211→            if attempt < max_retries:
   212→                print(f'[PROXY] Timeout, retrying in 10s...')
   213→                last_error = 'Request timeout'
   214→                time.sleep(10)
   215→                continue
   216→            raise Exception(f'Proxy timeout after {timeout}s')
   217→        except req_lib.exceptions.ConnectionError as e:
   218→            if attempt < max_retries:
   219→                print(f'[PROXY] Connection error, retrying in 10s: {e}')
   220→                last_error = str(e)
   221→                time.sleep(10)
   222→                continue
   223→            raise Exception(f'Proxy connection error: {e}')
   224→    
   225→    raise Exception(f'Failed after {max_retries} retries. Last error: {last_error}')

   231→def pad2(n: int) -> str:
   232→    return f'{n:02d}'
   233→
   234→def parse_b2_json(json_str: str):
   235→    """Parse B2 JSON with truncation repair (same logic as finpy-tse/TypeScript)."""
   236→    try:
   237→        parsed = json.loads(json_str)
   238→        return parsed.get('indexB2', [])
   239→    except json.JSONDecodeError:
   240→        # JSON is truncated — repair it
   241→        repaired = json_str.strip()
   242→        last_brace = repaired.rfind('}')
   243→        if last_brace > 0:
   244→            repaired = repaired[:last_brace + 1]
   245→            repaired = repaired.rstrip(',').rstrip() + ']}'
   246→            try:
   247→                parsed = json.loads(repaired)
   248→                return parsed.get('indexB2', [])
   249→            except json.JSONDecodeError:
   250→                pass
   251→        # Try removing last entry
   252→        last_complete = repaired.rfind('},{')
   253→        if last_complete > 0:
   254→            repaired = repaired[:last_complete + 1] + ']}'
   255→            try:
   256→                parsed = json.loads(repaired)
   257→                return parsed.get('indexB2', [])
   258→            except json.JSONDecodeError:
   259→                pass
   260→        raise Exception('Cannot repair truncated JSON')

   262→def process_entries(entries, sector_name='خودرو'):
   263→    """
   264→    Process B2 entries into candle data using finpy-tse methodology.
   265→    Returns list of dicts: {date, jDate, open, high, low, close, volume}
   266→    """
   267→    candles = []
   268→    for i, e in enumerate(entries):
   269→        d_even = str(e.get('dEven', ''))
   270→        if len(d_even) != 8:
   271→            continue
   272→
   273→        g_year = int(d_even[:4])
   274→        g_month = int(d_even[4:6])
   275→        g_day = int(d_even[6:8])
   276→
   277→        try:
   278→            g_date = datetime(g_year, g_month, g_day)
   279→        except ValueError:
   280→            continue
   281→
   282→        # Convert to Jalali using jdatetime (same as finpy-tse)
   283→        try:
   284→            j_date = jdatetime.date.fromgregorian(date=g_date.date())
   285→            j_date_str = f'{pad2(j_date.year)}-{pad2(j_date.month)}-{pad2(j_date.day)}'
   286→        except:
   287→            continue
   288→
   289→        close_val = float(e.get('xNivInuClMresIbs', 0)) or 0
   290→        base_val = float(e.get('xNivInuPbMresIbs', 0)) or 0
   291→        high_val = float(e.get('xNivInuPhMresIbs', 0)) or 0
   292→
   293→        if close_val <= 0:
   294→            continue
   295→
   296→        prev_close = float(entries[i - 1].get('xNivInuClMresIbs', 0)) if i > 0 else close_val
   297→        open_val = close_val if i == 0 else prev_close
   298→        candle_high = max(high_val, close_val, open_val)
   299→        candle_low = min(base_val if base_val > 0 else close_val, close_val, open_val)
   300→
   301→        candles.append({
   302→            'date': f'{pad2(g_year)}-{pad2(g_month)}-{pad2(g_day)}',
   303→            'jDate': j_date_str,
   304→            'open': round(open_val, 2),
   305→            'high': round(candle_high, 2),
   306→            'low': round(candle_low, 2),
   307→            'close': round(close_val, 2),
   308→            'volume': 0,
   309→        })
   310→
   311→    return candles

   313→def fetch_and_store_index(idx_def: dict) -> dict:
   314→    """
   315→    Fetch a single index and store in SQLite.
   316→    Returns {code, name, count, status, error?}
   317→    """
   318→    code = idx_def['code']
   319→    web_id = idx_def['webId']
   320→    name = idx_def['name']
   321→    sector_name = name.replace('شاخص گروه ', '') if idx_def['category'] == 'sector' else name
   322→
   323→    conn = get_db()
   324→    try:
   325→        # Update status to fetching
   326→        now = prisma_now()
   327→        conn.execute('''
   328→            UPDATE IndexDef SET status = 'fetching', updatedAt = ?
   329→            WHERE code = ?
   330→        ''', (now, code,))
   331→        conn.commit()
   332→
   333→        print(f'[FETCH] {code} ({name}) — fetching via proxy...')
   334→
   335→        # Fetch raw data via z-ai SDK proxy
   336→        raw_json = fetch_via_proxy(web_id)
   337→        entries = parse_b2_json(raw_json)
   338→        print(f'[FETCH] {code} — parsed {len(entries)} raw entries')
   339→
   340→        if not entries:
   341→            conn.execute('''
   342→                UPDATE IndexDef SET status = 'error', errorMsg = 'No entries parsed from CDN',
   343→                updatedAt = ? WHERE code = ?
   344→            ''', (now, code,))
   345→            conn.commit()
   346→            return {'code': code, 'name': name, 'count': 0, 'status': 'error', 'error': 'No entries parsed'}
   347→
   348→        # Process entries (finpy-tse methodology)
   349→        candles = process_entries(entries, sector_name)
   350→        print(f'[FETCH] {code} — processed {len(candles)} candles')
   351→
   352→        if not candles:
   353→            conn.execute('''
   354→                UPDATE IndexDef SET status = 'error', errorMsg = 'No valid candles after processing',
   355→                updatedAt = ? WHERE code = ?
   356→            ''', (now, code,))
   357→            conn.commit()
   358→            return {'code': code, 'name': name, 'count': 0, 'status': 'error', 'error': 'No valid candles'}
   359→
   360→        # Get IndexDef ID
   361→        row = conn.execute('SELECT id FROM IndexDef WHERE code = ?', (code,)).fetchone()
   362→        if not row:
   363→            return {'code': code, 'name': name, 'count': 0, 'status': 'error', 'error': 'IndexDef not found in DB'}
   364→        index_def_id = row['id']
   365→
   366→        # Delete old candles and insert new ones (upsert)
   367→        conn.execute('DELETE FROM IndexHistory WHERE indexDefId = ?', (index_def_id,))
   368→
   369→        for c in candles:
   370→            conn.execute('''
   371→                INSERT INTO IndexHistory (id, indexDefId, date, jDate, open, high, low, close, volume)
   372→                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
   373→            ''', (f'{code}_{c["date"]}', index_def_id, c['date'], c['jDate'],
   374→                  c['open'], c['high'], c['low'], c['close'], c['volume']))
   375→
   376→        # Update IndexDef status
   377→        now = prisma_now()
   378→        conn.execute('''
   379→            UPDATE IndexDef SET status = 'done', candleCount = ?, fetchedAt = ?,
   380→            errorMsg = NULL, updatedAt = ? WHERE code = ?
   381→        ''', (len(candles), now, now, code))
   382→        conn.commit()
   383→
   384→        print(f'[FETCH] {code} — stored {len(candles)} candles ✓')
   385→        return {'code': code, 'name': name, 'count': len(candles), 'status': 'done'}
   386→
   387→    except Exception as ex:
   388→        err_msg = str(ex)[:500]
   389→        print(f'[FETCH] {code} — ERROR: {err_msg}')
   390→        try:
   391→            now = prisma_now()
   392→            conn.execute('''
   393→                UPDATE IndexDef SET status = 'error', errorMsg = ?, updatedAt = ?
   394→                WHERE code = ?
   395→            ''', (err_msg, now, code))
   396→            conn.commit()
   397→        except:
   398→            pass
   399→        return {'code': code, 'name': name, 'count': 0, 'status': 'error', 'error': err_msg}
   400→    finally:
   401→        conn.close()

   403→# ═══════════════════════════════════════════════════════════════
   404→# Bulk fetch (background thread)
   405→# ═══════════════════════════════════════════════════════════════
   406→
   407→class BulkFetcher:
   408→    def __init__(self):
   409→        self.is_running = False
   410→        self.progress = []  # list of result dicts
   411→        self.current_index = 0
   412→        self.total = len(ALL_INDICES)
   413→        self.start_time = None
   414→        self.thread = None
   415→        self.error = None
   416→
   417→    def start(self, category='all'):
   418→        if self.is_running:
   419→            return {'error': 'Bulk fetch already in progress'}
   420→
   421→        indices = ALL_INDICES
   422→        if category == 'main':
   423→            indices = MAIN_INDICES
   424→        elif category == 'sector':
   425→            indices = SECTOR_INDICES
   426→
   427→        self.total = len(indices)
   428→        self.current_index = 0
   429→        self.progress = []
   430→        self.is_running = True
   431→        self.start_time = time.time()
   432→        self.error = None
   433→
   434→        def run():
   435→            try:
   436→                for i, idx in enumerate(indices):
   437→                    if not self.is_running:
   438→                        self.progress.append({'code': idx['code'], 'status': 'cancelled'})
   439→                        break
   440→                    self.current_index = i + 1
   441→                    result = fetch_and_store_index(idx)
   442→                    self.progress.append(result)
   443→                    # Delay between fetches to avoid overwhelming z-ai SDK
   444→                    if i < len(indices) - 1:
   445→                        time.sleep(2)
   446→            except Exception as ex:
   447→                self.error = str(ex)
   448→                print(f'[BULK] Fatal error: {ex}')
   449→            finally:
   450→                self.is_running = False
   451→                print(f'[BULK] Finished. {len(self.progress)} indices processed.')
   452→
   453→        self.thread = threading.Thread(target=run, daemon=True)
   454→        self.thread.start()
   455→        return {'status': 'started', 'total': self.total}
   456→
   457→    def stop(self):
   458→        self.is_running = False
   459→        return {'status': 'stopping'}
   460→
   461→    def get_status(self):
   462→        elapsed = time.time() - self.start_time if self.start_time else 0
   463→        done_count = len(self.progress)
   464→        error_count = sum(1 for p in self.progress if p.get('status') == 'error')
   465→        success_count = sum(1 for p in self.progress if p.get('status') == 'done')
   466→        return {
   467→            'is_running': self.is_running,
   468→            'current': self.current_index,
   469→            'total': self.total,
   470→            'done': done_count,
   471→            'success': success_count,
   472→            'errors': error_count,
   473→            'elapsed_seconds': round(elapsed, 1),
   474→            'error': self.error,
   475→            'progress': self.progress[-10:],  # Last 10 results
   476→        }
   477→
   478→bulk_fetcher = BulkFetcher()

   480→# ═══════════════════════════════════════════════════════════════
   481→# HTTP Server
   482→# ═══════════════════════════════════════════════════════════════
   483→
   484→def json_response(data, status=200):
   485→    body = json.dumps(data, ensure_ascii=False).encode('utf-8')
   486→    return (status, {
   487→        'Content-Type': 'application/json; charset=utf-8',
   488→        'Access-Control-Allow-Origin': '*',
   489→        'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
   490→        'Access-Control-Allow-Headers': 'Content-Type',
   491→    }, body)
   492→
   493→class Handler(BaseHTTPRequestHandler):
   494→    def log_message(self, format, *args):
   495→        pass  # Suppress default logging
   496→
   497→    def do_OPTIONS(self):
   498→        self.send_response(204)
   499→        self.send_header('Access-Control-Allow-Origin', '*')
   500→        self.send_header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
   501→        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
   502→        self.end_headers()
   503→
   504→    def do_GET(self):
   505→        parsed = urlparse(self.path)
   506→        path = parsed.path
   507→        params = parse_qs(parsed.query)
   508→
   509→        try:
   510→            # Health check
   511→            if path == '/api/health':
   512→                return self._send(*json_response({
   513→                    'status': 'ok',
   514→                    'service': 'finpy-index-fetcher',
   515→                    'version': '2.0.0',
   516→                    'source': 'finpy-tse + z-ai SDK proxy',
   517→                    'total_indices': len(ALL_INDICES),
   518→                }))
   519→
   520→            # List all index definitions and their fetch status
   521→            if path == '/api/indices':
   522→                conn = get_db()
   523→                try:
   524→                    rows = conn.execute('''
   525→                        SELECT code, name, webId, category, status, candleCount, fetchedAt, errorMsg
   526→                        FROM IndexDef ORDER BY category, code
   527→                    ''').fetchall()
   528→                    indices = [dict(r) for r in rows]
   529→                finally:
   530→                    conn.close()
   531→                return self._send(*json_response({'indices': indices}))
   532→
   533→            # Get candles for a specific index (from SQLite)
   534→            if path == '/api/index-data':
   535→                code = params.get('code', [''])[0]
   536→                if not code:
   537→                    return self._send(*json_response({'error': 'code is required'}, 400))
   538→
   539→                conn = get_db()
   540→                try:
   541→                    # Get IndexDef
   542→                    def_row = conn.execute('SELECT id, code, name, webId, category FROM IndexDef WHERE code = ?', (code,)).fetchone()
   543→                    if not def_row:
   544→                        return self._send(*json_response({'error': f'Index {code} not found'}, 404))
   545→
   546→                    index_def = dict(def_row)
   547→                    candles = conn.execute('''
   548→                        SELECT date, jDate as j_date, open, high, low, close, volume
   549→                        FROM IndexHistory WHERE indexDefId = ? ORDER BY date
   550→                    ''', (index_def['id'],)).fetchall()
   551→
   552→                    return self._send(*json_response({
   553→                        'index': index_def,
   554→                        'candles': [dict(c) for c in candles],
   555→                        'count': len(candles),
   556→                    }))
   557→                finally:
   558→                    conn.close()
   559→
   560→            # Bulk fetch status
   561→            if path == '/api/bulk-status':
   562→                return self._send(*json_response(bulk_fetcher.get_status()))
   563→
   564→            return self._send(*json_response({'error': 'Not found'}, 404))
   565→
   566→        except Exception as ex:
   567→            traceback.print_exc()
   568→            return self._send(*json_response({'error': str(ex)}, 500))
   569→
   570→    def do_POST(self):
   571→        parsed = urlparse(self.path)
   572→        path = parsed.path
   573→        params = parse_qs(parsed.query)
   574→
   575→        try:
   576→            # Start bulk fetch
   577→            if path == '/api/bulk-fetch':
   578→                category = params.get('category', ['all'])[0]
   579→                result = bulk_fetcher.start(category)
   580→                return self._send(*json_response(result))
   581→
   582→            # Stop bulk fetch
   583→            if path == '/api/bulk-stop':
   584→                result = bulk_fetcher.stop()
   585→                return self._send(*json_response(result))
   586→
   587→            # Fetch single index
   588→            if path == '/api/fetch-one':
   589→                code = params.get('code', [''])[0]
   590→                if not code:
   591→                    return self._send(*json_response({'error': 'code is required'}, 400))
   592→
   593→                idx_def = next((i for i in ALL_INDICES if i['code'] == code), None)
   594→                if not idx_def:
   595→                    return self._send(*json_response({'error': f'Unknown code: {code}'}, 404))
   596→
   597→                result = fetch_and_store_index(idx_def)
   598→                return self._send(*json_response(result))
   599→
   600→            return self._send(*json_response({'error': 'Not found'}, 404))
   601→
   602→        except Exception as ex:
   603→            traceback.print_exc()
   604→            return self._send(*json_response({'error': str(ex)}, 500))
   605→
   606→    def _send(self, status, headers, body):
   607→        self.send_response(status)
   608→        for key, val in headers.items():
   609→            self.send_header(key, val)
   610→        self.end_headers()
   611→        self.wfile.write(body)
