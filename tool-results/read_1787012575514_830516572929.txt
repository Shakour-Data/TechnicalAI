   118→const TYPE_FILTERS = [
   119→  { key: 'all', label: 'همه مسیرها' },
   120→  { key: 'up', label: 'صعودی' },
   121→  { key: 'pullback', label: 'پولبک و بازگشت' },

   129→// Component
   130→// ═══════════════════════════════════════════════════════════════════════════════
   131→
   132→export default function VdssGraph(props: VdssGraphProps) {
   133→  const { symbolName, currentPrice, resistances, supports, ma100, scenarios, rsi, adx, bullScore, edgeWeights, currencyUnit } = props;
   134→  const unit = currencyUnit || 'ریال';
   135→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   136→  const S1_level = supports[0] ?? currentPrice * 0.95;
   137→
   138→  const [activeFilter, setActiveFilter] = useState('all');
   139→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   140→  const graphRef = useRef<HTMLDivElement>(null);
   141→  const svgRef = useRef<SVGSVGElement>(null);
   142→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   143→  const shellRef = useRef<HTMLDivElement>(null);
   144→
   145→  // ── Edge probabilities (using TA engine's bullScore) ──────────────
   146→  const edgeProbs = useMemo(() =>
   147→    calcEdgeProbabilities(edgeWeights),
   148→  [edgeWeights]
   149→  );
   150→
   151→  // ── All paths with raw probabilities ─────────────────────────────
   152→  const allPathsRaw = useMemo(() => findAllPaths(edgeProbs), [edgeProbs]);
   153→
   154→  // ── Calibrate path probabilities to match TA engine scenario probabilities ──
   155→  const allPaths = useMemo(() => {
   156→    // Compute raw sum per target
   157→    const rawSums: Record<string, number> = {};
   158→    for (const key of SCENARIO_KEYS) rawSums[key] = 0;
   159→    for (const p of allPathsRaw) rawSums[p.target] += p.prob;
   160→
   161→    // Scale each path so the sum per target matches scenario probability
   162→    return allPathsRaw.map(p => {
   163→      const rawSum = rawSums[p.target] || 0.0001;
   164→      const targetProb = scenarios[p.target as keyof typeof scenarios].probability / 100;
   165→      const scale = targetProb / rawSum;
   166→      return { ...p, prob: p.prob * scale };
   167→    });
   168→  }, [allPathsRaw, scenarios]);
   169→
   170→  // ── Filtered paths ────────────────────────────────────────────────
   171→  const filteredPaths = useMemo(() => {
   172→    if (activeFilter === 'all') return allPaths;
   173→    // Check if filter is a scenario key
   174→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   175→      return allPaths.filter(p => p.target === activeFilter);
   176→    }
   177→    // Type filter: paths must use only edges of this type
   178→    return allPaths.filter(p =>
   179→      p.edges.every(ei => EDGES[ei][3] === activeFilter)
   180→    );
   181→  }, [allPaths, activeFilter]);
   182→
   183→  // ── Visible edges/nodes based on filter ────────────────────────────
   184→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   185→    if (activeFilter === 'all') {
   186→      const allEdgeIdx = EDGES.map((_, i) => i);
   187→      const allNodes = new Set(Object.keys(NODE_POSITIONS));
   188→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   189→    }
   190→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   191→      // Show all edges/nodes on paths to this scenario
   192→      const edgeSet = new Set<number>();
   193→      const nodeSet = new Set<string>();
   194→      for (const p of filteredPaths) {
   195→        p.edges.forEach(ei => edgeSet.add(ei));
   196→        p.nodes.forEach(n => nodeSet.add(n));
   197→      }
   198→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   199→    }
   200→    // Type filter
   201→    const edgeSet = new Set<number>();
   202→    const nodeSet = new Set<string>();
   203→    EDGES.forEach((e, i) => {
   204→      if (e[3] === activeFilter) {
   205→        edgeSet.add(i);
   206→        nodeSet.add(e[0]);
   207→        nodeSet.add(e[1]);
   208→      }
   209→    });
   210→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   211→  }, [activeFilter, filteredPaths]);
   212→
   213→  // ── Node values ──────────────────────────────────────────────────
   214→  const nodeValues: Record<string, string> = {
   215→    A: toFa(currentPrice),
   216→    B: toFa(R1_level),
   217→    D: toFa(resistances[1] ?? currentPrice * 1.10),
   218→    E: `${toFa(Math.round(R1_level + (resistances[1] ?? R1_level * 1.05 - R1_level) * 0.5))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 0.8))}`,
   219→    F: `${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 1.2))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 2.0))}`,
   220→    G: toFa(S1_level),
   221→    H: toFa(supports[1] ?? currentPrice * 0.90),
   222→    I: toFa(supports[2] ?? currentPrice * 0.85),
   223→    J: toFa(supports[3] ?? currentPrice * 0.80),
   224→    K: toFa(ma100),
   225→    L: props.trendDirection === 'up' ? 'صعودی' : props.trendDirection === 'down' ? 'نزولی' : 'خنثی',
   226→  };
   227→  for (const key of SCENARIO_KEYS) {
   228→    const s = scenarios[key];
   229→    nodeValues[key] = `${toFa(s.targetMin)} — ${toFa(s.targetMax)} ${unit}`;
   230→  }
   231→
   232→  // ── Draw SVG edges ───────────────────────────────────────────────
   233→  const drawEdges = useCallback(() => {
   234→    const svg = svgRef.current;
   235→    const graph = graphRef.current;
   236→    if (!svg || !graph) return;
   237→
   238→    let markersSvg = '';
   239→    for (const [k, c] of Object.entries(EDGE_COLORS)) {
   240→      markersSvg += `<marker id="arrow-${k}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   241→    }
   242→
   243→    let pathsSvg = '';
   244→    const graphRect = graph.getBoundingClientRect();
   245→
   246→    for (let ei = 0; ei < EDGES.length; ei++) {
   247→      const [fromId, toId, label, type] = EDGES[ei];
   248→      const isVisible = visibleEdgeIndices.includes(ei);
   249→      const fromEl = nodeRefs.current[fromId];
   250→      const toEl = nodeRefs.current[toId];
   251→      if (!fromEl || !toEl) continue;
   252→
   253→      const fRect = fromEl.getBoundingClientRect();
   254→      const tRect = toEl.getBoundingClientRect();
   255→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   256→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   257→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   258→      const by = tRect.top - graphRect.top + tRect.height / 2;
   259→      const dx = bx - ax;
   260→      const dy = by - ay;
   261→      const dist = Math.sqrt(dx * dx + dy * dy);
   262→      if (dist === 0) continue;
   263→
   264→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   265→      const mx = (ax + bx) / 2;
   266→      const my = (ay + by) / 2;
   267→      const nx = -dy / dist;
   268→      const ny = dx / dist;
   269→      const cx = mx + nx * bend;
   270→      const cy = my + ny * bend;
   271→
   272→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   273→      const prob = edgeProbs[String(ei)] ?? 0;
   274→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   275→
   276→      pathsSvg += `<path d="${d}" stroke="${EDGE_COLORS[type]}" stroke-width="${isVisible ? 2 : 1}" opacity="${isVisible ? 0.72 : 0.06}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   277→
   278→      // Edge probability label (only for visible edges)
   279→      if (isVisible) {
   280→        pathsSvg += `<text x="${cx}" y="${cy - 6}" fill="#cde4ef" font-size="9" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3.5" stroke-linejoin="round" opacity="0.85" data-type="${type}" class="edge-label">${label}</text>`;
   281→        pathsSvg += `<text x="${cx}" y="${cy + 8}" fill="${EDGE_COLORS[type]}" font-size="10" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   282→      }
   283→    }
   284→
   285→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   286→  }, [visibleEdgeIndices, edgeProbs]);
   287→
   288→  useEffect(() => { drawEdges(); }, [drawEdges]);
   289→  useEffect(() => {
   290→    const el = shellRef.current;
   291→    if (!el) return;
   292→    const observer = new ResizeObserver(() => drawEdges());
   293→    observer.observe(el);
   294→    return () => observer.disconnect();
   295→  }, [drawEdges]);
   296→
   297→  // ── Aggregated path probabilities per target (calibrated = matches scenario prob) ──
   298→  const pathProbsByTarget = useMemo(() => {
   299→    const result: Record<string, number> = {};
   300→    for (const key of SCENARIO_KEYS) result[key] = 0;
   301→    for (const p of filteredPaths) {
   302→      result[p.target] = (result[p.target] ?? 0) + p.prob;
   303→    }
   304→    // After calibration, these should match scenario probabilities
   305→    return result;
   306→  }, [filteredPaths]);
   307→
   308→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   309→
   310→  // ── Detail panel content ─────────────────────────────────────────
   311→  const detailContent = useMemo(() => {
   312→    if (!selectedNode) return null;
   313→    const def = NODE_DEFS[selectedNode];
   314→    if (!def) return null;
   315→    const relations = EDGES.filter(e => e[0] === selectedNode || e[1] === selectedNode);
   316→    const inputs = relations.filter(e => e[1] === selectedNode);
   317→    const outputs = relations.filter(e => e[0] === selectedNode);
   318→
   319→    return (
   320→      <div className="space-y-3">
   321→        <h3 className="text-sm font-bold text-gray-100">{def.title}</h3>
   322→        <p className="text-xs text-gray-300"><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   323→        <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border border-white/10 bg-white/5 text-gray-300">{def.type}</span>
   324→        <p className="text-xs text-gray-400 leading-relaxed">{def.desc}</p>
   325→        {inputs.length > 0 && (
   326→          <div>
   327→            <p className="text-xs font-medium text-gray-300 mb-1">مسیرهای ورودی ({inputs.length}):</p>
   328→            <ul className="space-y-1">
   329→              {inputs.map((e, i) => {
   330→                const ei = EDGES.indexOf(e);
   331→                const ep = edgeProbs[String(ei)] ?? 0;
   332→                return (
   333→                  <li key={i} className="text-[11px] text-gray-500 leading-relaxed border-t border-dashed border-white/10 pt-1.5">
   334→                    <b className="text-gray-400">{e[0]} ← {e[1]}</b>
   335→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${EDGE_COLORS[e[3]]}20`, color: EDGE_COLORS[e[3]] }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   336→                    <br />{e[2]}
   337→                  </li>
   338→                );
   339→              })}
   340→            </ul>
   341→          </div>
   342→        )}
   343→        {outputs.length > 0 && (
   344→          <div>
   345→            <p className="text-xs font-medium text-gray-300 mb-1">مسیرهای خروجی ({outputs.length}):</p>
   346→            <ul className="space-y-1">
   347→              {outputs.map((e, i) => {
   348→                const ei = EDGES.indexOf(e);
   349→                const ep = edgeProbs[String(ei)] ?? 0;
   350→                return (
   351→                  <li key={i} className="text-[11px] text-gray-500 leading-relaxed border-t border-dashed border-white/10 pt-1.5">
   352→                    <b className="text-gray-400">{e[0]} → {e[1]}</b>
   353→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${EDGE_COLORS[e[3]]}20`, color: EDGE_COLORS[e[3]] }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   354→                    <br />{e[2]}
   355→                  </li>
   356→                );
   357→              })}
   358→            </ul>
   359→          </div>
   360→        )}
   361→      </div>
   362→    );
   363→  }, [selectedNode, nodeValues, edgeProbs]);
   364→
   365→  // ── All filter buttons ────────────────────────────────────────────
   366→  const allFilters = [
   367→    ...TYPE_FILTERS,
   368→    { key: 'sep1', label: '│', isSep: true as const },
   369→    ...SCENARIO_KEYS.map(k => ({ key: k, label: `${k} | ${SCENARIO_META[k].label}`, scenarioKey: k })),
   370→  ];
   371→
   372→  return (
   373→    <div className="space-y-3" dir="rtl">
   374→      {/* ═══ HEADER ═══ */}
   375→      <div className="flex items-center justify-between gap-4 px-4 py-3 rounded-2xl border border-cyan-500/20"
   376→        style={{ background: 'linear-gradient(105deg, rgba(14,35,53,.94), rgba(8,22,35,.77))', boxShadow: '0 18px 55px rgba(0,0,0,.35)' }}>
   377→        <div className="flex items-center gap-3">
   378→          <div className="w-10 h-10 rounded-xl border border-amber-500/60 flex items-center justify-center text-amber-400 text-xl font-bold"
   379→            style={{ boxShadow: 'inset 0 0 22px rgba(255,177,27,.12), 0 0 22px rgba(255,177,27,.08)' }}>◈</div>
   380→          <div>
   381→            <h2 className="text-base font-bold text-gray-100">گراف تصمیم {symbolName}</h2>
