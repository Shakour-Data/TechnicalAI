   123→export default function VdesAnalysis(props: VdesAnalysisProps) {
   124→  const {
   125→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
   126→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,

   131→    edgeWeights,
   132→  } = props;
   133→
   134→  const unit = currencyUnit || 'ریال';
   135→
   136→  // ── Instrument Context (for intelligent naming) ─────────────────
   137→  const instrumentLabel = useMemo(() => {
   138→    if (isIndex) return { name: `شاخص ${symbolName}`, type: 'index' as const, verb: 'قرار دارد' };
   139→    if (isTgju) return { name: symbolName, type: 'tgju' as const, verb: 'قرار دارد' };
   140→    return { name: `سهم ${symbolName}`, type: 'stock' as const, verb: 'قرار دارد' };
   141→  }, [symbolName, isIndex, isTgju]);
   142→
   143→  const IL = instrumentLabel;
   144→
   145→  const containerRef = useRef<HTMLDivElement>(null);
   146→  const textAnalysisRef = useRef<HTMLDivElement>(null);
   147→  const chartApiRef = useRef<IChartApi | null>(null);
   148→  const [exporting, setExporting] = useState(false);
   149→  const [exportMenuOpen, setExportMenuOpen] = useState(false);
   150→  const exportMenuRef = useRef<HTMLDivElement>(null);
   151→
   152→  // ── AI Analysis state ───────────────────────────────────────────
   153→  const [aiAnalysis, setAiAnalysis] = useState<string | null>(null);
   154→  const [aiLoading, setAiLoading] = useState(false);
   155→  const [aiError, setAiError] = useState<string | null>(null);
   156→  const aiFetchIdRef = useRef(0);
   157→  // Track the current AbortController + its timeout for proper cleanup
   158→  const abortControllerRef = useRef<{ controller: AbortController; timeoutId: ReturnType<typeof setTimeout> } | null>(null);
   159→
   160→  // Cleanup any in-flight fetch (abort + clear timeout)
   161→  const cancelInFlightFetch = useCallback(() => {
   162→    if (abortControllerRef.current) {
   163→      clearTimeout(abortControllerRef.current.timeoutId);
   164→      try {
   165→        abortControllerRef.current.controller.abort('cancelled');
   166→      } catch { /* ignore */ }
   167→      abortControllerRef.current = null;
   168→    }
   169→  }, []);
   170→
   171→  // Cleanup on unmount
   172→  useEffect(() => {
   173→    return () => cancelInFlightFetch();
   174→  }, [cancelInFlightFetch]);
   175→
   176→  // ── V11 Debug data ──────────────────────────────────────────────
   177→  const [v11Debug, setV11Debug] = useState<any>(null);
   178→
   179→  // ── V11 Scenarios (built from TA data + S/R levels) ───────────────
   180→  const v11Result = useMemo(() => {
   181→    return buildV11Scenarios({
   182→      currentPrice,
   183→      atr,
   184→      overallSignal: overallSignal as 'bullish' | 'bearish' | 'neutral',
   185→      v9Scenarios: scenarios,
   186→      supports,
   187→      resistances,
   188→      bollingerUpper,
   189→      bollingerLower,
   190→      bollingerMiddle,
   191→      ma21,
   192→      ma100,
   193→    });
   194→  }, [currentPrice, atr, overallSignal, scenarios, supports, resistances,
   195→      bollingerUpper, bollingerLower, bollingerMiddle, ma21, ma100]);
   196→
   197→  // ── Fetch AI analysis (robust with logging + auto-retry) ─────────────
   198→  const fetchAiAnalysis = useCallback(async (isRetry = false) => {
   199→    // Cancel any previous in-flight fetch first
   200→    cancelInFlightFetch();
   201→
   202→    const fetchId = ++aiFetchIdRef.current;
   203→    setAiLoading(true);
   204→    setAiError(null);
   205→    try {
   206→      const apiUrl = '/api/v11-analysis';
   207→      const activeScenarios = v11Result.scenarios;
   208→      const requestBody: any = {
   209→          symbolName, currentPrice, ma21, ma100,
   210→          rsi, mfi, cci, adx, stochK, stochD,
   211→          macdLine, macdSignal, macdHist,
   212→          diPlus, diMinus, sar, atr, obv, hasVolume,
   213→          bollingerUpper, bollingerMiddle, bollingerLower,
   214→          trendDirection, trendAngle, trendR2, overallSignal,
   215→          scenarios: activeScenarios, resistances, supports,
   216→          resistanceStrengths, supportStrengths,
   217→          isIndex, isTgju, currencyUnit,
   218→          v11DominantKey: v11Result.dominantKey,
   219→          v11HighestCumulativeKey: v11Result.highestCumulativeKey,
   220→        };
   221→      // Include OHLCV history for ML training/prediction
   222→      // Cap at last 500 candles to keep request payload small (ML needs ~120, patterns ~50)
   223→      if (candles.length > 0) {
   224→        const recentCandles = candles.length > 500 ? candles.slice(-500) : candles;
   225→        requestBody.ohlcvHistory = recentCandles.map(c => ({
   226→          date: c.date, open: c.open, high: c.high,
   227→          low: c.low, close: c.close, volume: c.volume,
   228→        }));
   229→      }
   230→
   231→      // AbortController for 120s client-side timeout
   232→      const controller = new AbortController();
   233→      const timeoutId = setTimeout(() => {
   234→        controller.abort('Client timeout after 120s');
   235→      }, 120000);
   236→      // Store for cleanup on re-fetch / unmount
   237→      abortControllerRef.current = { controller, timeoutId };
   238→
   239→      console.log(`[VdesAnalysis v11] Fetching ${apiUrl} for ${symbolName} (retry=${isRetry})`);
   240→      const res = await fetch(apiUrl, {
   241→        method: 'POST',
   242→        headers: { 'Content-Type': 'application/json' },
   243→        body: JSON.stringify(requestBody),
   244→        signal: controller.signal,
   245→      });
   246→      // Fetch completed — clear timeout and ref
   247→      clearTimeout(timeoutId);
   248→      abortControllerRef.current = null;
   249→
   250→      if (!res.ok) {
   251→        let errorMsg = `خطای سرور (${res.status})`;
   252→        try {
   253→          const errData = await res.json();
   254→          errorMsg = errData.error || errorMsg;
   255→        } catch {
   256→          // Response body is not JSON (e.g. 502 from Caddy)
   257→          if (res.status === 502) errorMsg = 'سرور در حال بارگذاری مجدد است. لطفاً چند ثانیه صبر کنید و دوباره تلاش کنید.';
   258→        }
   259→        console.error(`[VdesAnalysis v11] API error ${res.status}:`, errorMsg);
   260→        throw new Error(errorMsg);
   261→      }
   262→      const data = await res.json();
   263→      if (fetchId === aiFetchIdRef.current) {
   264→        if (data.analysis && data.analysis.trim().length > 0) {
   265→          setAiAnalysis(data.analysis);
   266→          console.log(`[VdesAnalysis v11] Got analysis for ${symbolName}: ${data.analysis.length} chars`);
   267→        } else {
   268→          console.error(`[VdesAnalysis v11] Empty analysis for ${symbolName}`, data);
   269→          setAiError('مدل پاسخ خالی تولید کرد. لطفاً دوباره تلاش کنید.');
   270→        }
   271→        if (data._debug?.v11) setV11Debug({ ...data._debug.v11, v11Persona: data._debug.v11Persona, v11Template: data._debug.v11Template });
   272→        else setV11Debug(null);
   273→      }
   274→    } catch (err) {
   275→      // Clear timeout and ref on error too
   276→      clearTimeout(abortControllerRef.current?.timeoutId);
   277→      abortControllerRef.current = null;
   278→
   279→      // Determine if this is an abort
   280→      const errMsg = err instanceof Error ? err.message : String(err);
   281→      const isAbort = err instanceof Error && err.name === 'AbortError';
   282→      // Browser prefixes the reason: "The operation was aborted. cancelled"
   283→      // so we use .includes() to catch intentional cancellations regardless of browser prefix
   284→      const isIntentionalCancel = isAbort && errMsg.includes('cancelled');
   285→
   286→      // Silently ignore intentional cancellations (superseded by new fetch or unmount)
   287→      if (isIntentionalCancel) {
   288→        return;
   289→      }
   290→
   291→      console.error(`[VdesAnalysis v11] Error for ${symbolName}:`, err);
   292→      if (fetchId === aiFetchIdRef.current) {
   293→        if (isAbort) {
   294→          setAiError('زمان پاسخگویی سرور به پایان رسید. لطفاً دوباره تلاش کنید.');
   295→        } else {
   296→          setAiError(errMsg || 'خطای ناشناخته');
   297→        }
   298→        // Auto-retry once on transient errors (503/429/timeout/abort) after 5s
   299→        if (!isRetry) {
   300→          const isTransient = errMsg.includes('503') || errMsg.includes('429') || errMsg.includes('پایان رسید') || errMsg.includes('بارگذاری مجدد') || errMsg.includes('timeout') || isAbort;
   301→          if (isTransient) {
   302→            console.log(`[VdesAnalysis v11] Auto-retry in 5s for ${symbolName}`);
   303→            setTimeout(() => {
   304→              if (fetchId === aiFetchIdRef.current) fetchAiAnalysis(true).catch(() => {});
   305→            }, 5000);
   306→          }
   307→        }
   308→      }
   309→    } finally {
   310→      if (fetchId === aiFetchIdRef.current) {
   311→        setAiLoading(false);
   312→      }
   313→    }
   314→  }, [symbolName, currentPrice, ma21, ma100, rsi, mfi, cci, adx, stochK, stochD,
   315→    macdLine, macdSignal, macdHist, diPlus, diMinus, sar, atr, obv, hasVolume,
   316→    bollingerUpper, bollingerMiddle, bollingerLower, trendDirection, trendAngle, trendR2,
   317→    overallSignal, scenarios, resistances, supports, resistanceStrengths, supportStrengths,
   318→    isIndex, isTgju, currencyUnit, v11Result, cancelInFlightFetch]);
   319→
   320→  // ── Auto-fetch AI analysis only on symbol change ──
   321→  // NOT on every data refresh (auto-refresh every 15min would re-trigger
   322→  // a 10-25s LLM call, making the text disappear each time)
   323→  const fetchTriggerKey = symbolName;
   324→  const lastAutoFetchKeyRef = useRef('');
   325→  // Keep a ref to fetchAiAnalysis so the effect always calls the latest version
   326→  const fetchAiAnalysisRef = useRef(fetchAiAnalysis);
   327→  fetchAiAnalysisRef.current = fetchAiAnalysis;
   328→  useEffect(() => {
   329→    if (fetchTriggerKey === lastAutoFetchKeyRef.current) return;
   330→    if (!symbolName || !currentPrice || currentPrice <= 0) {
   331→      console.log('[VdesAnalysis] Skipping fetch: no symbol or invalid price', { symbolName, currentPrice });
   332→      return;
   333→    }
   334→    lastAutoFetchKeyRef.current = fetchTriggerKey;
   335→    // Reset old text when switching symbols/versions
   336→    setAiAnalysis(null);
   337→    setAiError(null);
   338→    // Small delay to ensure all useMemo values (v11Result etc.) are computed
   339→    const timer = setTimeout(() => {
   340→      fetchAiAnalysisRef.current().catch(() => {});
   341→    }, 150);
   342→    return () => {
   343→      clearTimeout(timer);
   344→      // Also cancel any in-flight fetch when this effect re-runs or unmounts
   345→      cancelInFlightFetch();
   346→    };
   347→  }, [fetchTriggerKey, symbolName, currentPrice, cancelInFlightFetch]);
   348→
   349→  // ── Close export menu on outside click ───────────────────────
   350→  useEffect(() => {
   351→    if (!exportMenuOpen) return;
   352→    const handler = (e: MouseEvent) => {
   353→      if (exportMenuRef.current && !exportMenuRef.current.contains(e.target as Node)) {
   354→        setExportMenuOpen(false);
   355→      }
   356→    };
   357→    document.addEventListener('mousedown', handler);
   358→    return () => document.removeEventListener('mousedown', handler);
   359→  }, [exportMenuOpen]);
   360→
   361→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   362→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   363→  const S1_level = supports[0] ?? currentPrice * 0.95;
   364→  const S2_level = supports[1] ?? currentPrice * 0.90;
   365→  const S3_level = supports[2] ?? currentPrice * 0.85;
   366→
   367→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   368→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   369→
   370→  // ── Dominant scenario (V9) ──────────────────────────────────────
   371→  let highestKey = 'R3';
   372→  let highestProb = 0;
   373→  for (const key of SCENARIO_KEYS) {
   374→    if (scenarios[key].probability > highestProb) {
   375→      highestProb = scenarios[key].probability;
   376→      highestKey = key;
   377→    }
   378→  }
   379→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   380→
   381→  // ── V11 dominant + highest cumulative + total ────────────
   382→  const v11DominantKey = v11Result?.dominantKey || 'S5';
   383→  const v11HighestCumKey = v11Result?.highestCumulativeKey || 'S5';
   384→  const v11TotalProb = v11Result
   385→    ? V11_KEYS.reduce((sum, k) => sum + (v11Result.scenarios[k]?.probability || 0), 0)
   386→    : 0;
   387→
   388→  // ── Sorted scenarios by target price (ascending) ─────────────
   389→  const sortedScenarios = useMemo(() => {
   390→    return [...SCENARIO_KEYS]
   391→      .map(k => ({ key: k, ...scenarios[k] }))
   392→      .sort((a, b) => a.targetMin - b.targetMin);
   393→  }, [scenarios]);
   394→
   395→  // ── Dynamic scenario labels (context-aware based on overallSignal) ─
   396→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal), [overallSignal]);
   397→  const strategy = dynamicMeta[highestKey];
   398→
   399→  // ── Text variety: hash-based selector for varied language ──────────
   400→  const varietySeed = useMemo(() => {
   401→    const raw = `${rsi}${macdHist}${cci}${adx}${stochK}${trendAngle}${Math.round(currentPrice)}`;
   402→    let hash = 0;
   403→    for (let i = 0; i < raw.length; i++) hash = ((hash << 5) - hash + raw.charCodeAt(i)) | 0;
   404→    return Math.abs(hash);
   405→  }, [rsi, macdHist, cci, adx, stochK, trendAngle, currentPrice]);
   406→
   407→  // ── RSI signal ─────────────────────────────────────────────────
   408→  const rsiZone = rsi > 70 ? 'overbought' : rsi > 60 ? 'weak_ob' : rsi > 40 ? 'neutral' : rsi > 30 ? 'weak_os' : 'oversold';
   409→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   410→  const rsiColor = rsi > 70 ? 'text-red-700' : rsi > 60 ? 'text-amber-800' : rsi > 40 ? 'text-gray-600' : rsi > 30 ? 'text-amber-800' : 'text-emerald-700';
   411→
   412→  // ── Stochastic signal ──────────────────────────────────────────
   413→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   414→
   415→  // ── Trend text ─────────────────────────────────────────────────
   416→  const isUp = trendDirection === 'up';
   417→  const isDown = trendDirection === 'down';
   418→  const trendLabel = isUp ? 'صعودی' : isDown ? 'نزولی' : 'خنثی';
   419→  const trendColorCls = isUp ? 'text-emerald-700' : isDown ? 'text-red-700' : 'text-amber-800';
   420→  const adxStrength = adx > 40 ? 'بسیار قوی' : adx > 25 ? 'قوی' : adx > 15 ? 'متوسط' : 'ضعیف';
   421→  const adxColor = adx > 40 ? 'text-emerald-700' : adx > 25 ? 'text-amber-800' : 'text-gray-500';
   422→  const r2Pct = (trendR2 * 100).toFixed(1);
   423→

   929→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   930→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   931→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   932→  grade?: string; overlapCount?: number; touchCount?: number;
   933→}) {
   934→  const isSupport = type === 'support';
   935→  const isTargetType = type === 'target';
   936→
   937→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   938→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   939→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   940→
   941→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   942→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   943→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   944→
   945→  return (
   946→    <div
   947→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   948→      style={{
   949→        background: bgAccent,
   950→        border: `1px solid ${borderAccent}`,
   951→        borderRight: `3px solid ${accentColor}`,
   952→      }}
   953→    >
   954→      {/* Label + Price */}
   955→      <div className="flex-1 min-w-0">
   956→        <div className="flex items-center gap-2">
   957→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   958→          {isTarget && <span className="text-[9px]">🎯</span>}
   959→          {overlapCount !== undefined && overlapCount > 0 && (
   960→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   961→              {toPersianDigits(overlapCount)} هم‌پوشانی
   962→            </span>
   963→          )}
   964→        </div>
   965→        <div className="flex items-baseline gap-1">
   966→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   967→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   968→        </div>
   969→      </div>
   970→
   971→      {/* Strength + Grade */}
   972→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   973→        <div className="flex items-center gap-1.5">
   974→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   975→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   976→          <span className="text-[9px] text-gray-500">/۱۰</span>
   977→        </div>
   978→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   979→          <div
   980→            className="h-full rounded-full transition-all duration-500"
   981→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   982→          />
   983→        </div>
   984→        {touchCount !== undefined && touchCount > 0 && (
   985→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   986→        )}
   987→      </div>
   988→    </div>
   989→  );
   990→}
