   136→   136→
   137→   137→export default function VdssGraph(props: VdssGraphProps) {
   138→   138→  const { colors: C } = useTheme();
   139→   139→  const {
   140→   140→    symbolName,
   141→   141→    currentPrice,
   142→   142→    scenarios,
   143→   143→    decisionGraph,
   144→   144→  } = props;
   145→   145→
   146→   146→  const [activeFilter, setActiveFilter] = useState('all');
   147→   147→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   148→   148→  const graphRef = useRef<HTMLDivElement>(null);
   149→   149→  const svgRef = useRef<SVGSVGElement>(null);
   150→   150→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   151→   151→  const shellRef = useRef<HTMLDivElement>(null);
   152→   152→
   153→   153→  // ── Derived data from decisionGraph ──────────────────────────────
   154→   154→  const edges = decisionGraph?.edges ?? [];
   155→   155→  const nodes = decisionGraph?.nodes ?? [];
   156→   156→  const nodePositions = decisionGraph?.nodePositions ?? {};
   157→   157→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   158→   158→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   159→   159→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   160→   160→  const pathContributions = decisionGraph?.pathContributions ?? {};
   161→   161→  const nodeValues = decisionGraph?.nodeValues ?? {};
   162→   162→
   163→   163→  // Build node lookup map
   164→   164→  const nodeMap = useMemo(() => {
   165→   165→    const map: Record<string, (typeof nodes)[number]> = {};
   166→   166→    for (const n of nodes) map[n.id] = n;
   167→   167→    return map;
   168→   168→  }, [nodes]);
   169→   169→
   170→   170→  // ── Visible edges/nodes based on filter ──────────────────────────
   171→   171→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   172→   172→    if (activeFilter === 'all') {
   173→   173→      const allEdgeIdx = edges.map((_, i) => i);
   174→   174→      const allNodes = new Set(nodes.map(n => n.id));
   175→   175→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   176→   176→    }
   177→   177→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   178→   178→      // Show edges going TO this scenario node, plus their source nodes
   179→   179→      const edgeSet = new Set<number>();
   180→   180→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   181→   181→      edges.forEach((e, i) => {
   182→   182→        if (e.to === activeFilter) {
   183→   183→          edgeSet.add(i);
   184→   184→          nodeSet.add(e.from);
   185→   185→        }
   186→   186→      });
   187→   187→      // Also show intermediate nodes connected to those sources
   188→   188→      edges.forEach((e, i) => {
   189→   189→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   190→   190→          edgeSet.add(i);
   191→   191→          nodeSet.add(e.from);
   192→   192→        }
   193→   193→      });
   194→   194→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   195→   195→    }
   196→   196→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   197→   197→      const branchType = `branch-${activeFilter}` as const;
   198→   198→      const edgeSet = new Set<number>();
   199→   199→      const nodeSet = new Set<string>(['ROOT']);
   200→   200→      edges.forEach((e, i) => {
   201→   201→        if (e.type === branchType || (e.type !== `branch-trend` && e.type !== `branch-breakout` && e.type !== `branch-reversal` && nodeSet.has(e.from))) {
   202→   202→          // Include branch-type edges
   203→   203→          if (e.type === branchType) {
   204→   204→            edgeSet.add(i);
   205→   205→            nodeSet.add(e.from);
   206→   206→            nodeSet.add(e.to);
   207→   207→          }
   208→   208→        }
   209→   209→      });
   210→   210→      // Now add all edges whose source is already in nodeSet
   211→   211→      const expanded = new Set<number>();
   212→   212→      const expandedNodes = new Set<string>(nodeSet);
   213→   213→      edges.forEach((e, i) => {
   214→   214→        if (expandedNodes.has(e.from)) {
   215→   215→          expanded.add(i);
   216→   216→          expandedNodes.add(e.to);
   217→   217→        }
   218→   218→      });
   219→   219→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   220→   220→    }
   221→   221→    // Type filter (up/pullback/down/risk)
   222→   222→    const edgeSet = new Set<number>();
   223→   223→    const nodeSet = new Set<string>(['ROOT']);
   224→   224→    edges.forEach((e, i) => {
   225→   225→      if (e.type === activeFilter) {
   226→   226→        edgeSet.add(i);
   227→   227→        nodeSet.add(e.from);
   228→   228→        nodeSet.add(e.to);
   229→   229→      }
   230→   230→    });
   231→   231→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   232→   232→  }, [activeFilter, edges, nodes]);
   233→   233→
   234→   234→  // ── Draw SVG edges ───────────────────────────────────────────────
   235→   235→  const drawEdges = useCallback(() => {
   236→   236→    const svg = svgRef.current;
   237→   237→    const graph = graphRef.current;
   238→   238→    if (!svg || !graph || edges.length === 0) return;
   239→   239→
   240→   240→    let markersSvg = '';
   241→   241→    const seenColors = new Set<string>();
   242→   242→    for (const e of edges) {
   243→   243→      const c = EDGE_COLORS[e.type];
   244→   244→      if (c && !seenColors.has(e.type)) {
   245→   245→        seenColors.add(e.type);
   246→   246→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   247→   247→      }
   248→   248→    }
   249→   249→
   250→   250→    let pathsSvg = '';
   251→   251→    const graphRect = graph.getBoundingClientRect();
   252→   252→
   253→   253→    for (let ei = 0; ei < edges.length; ei++) {
   254→   254→      const { from: fromId, to: toId, label, type } = edges[ei];
   255→   255→      const isVisible = visibleEdgeIndices.includes(ei);
   256→   256→      const fromEl = nodeRefs.current[fromId];
   257→   257→      const toEl = nodeRefs.current[toId];
   258→   258→      if (!fromEl || !toEl) continue;
   259→   259→
   260→   260→      const fRect = fromEl.getBoundingClientRect();
   261→   261→      const tRect = toEl.getBoundingClientRect();
   262→   262→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   263→   263→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   264→   264→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   265→   265→      const by = tRect.top - graphRect.top + tRect.height / 2;
   266→   266→      const dx = bx - ax;
   267→   267→      const dy = by - ay;
   268→   268→      const dist = Math.sqrt(dx * dx + dy * dy);
   269→   269→
   270→   270→      if (dist === 0) continue;
   271→   271→
   272→   272→      // Curved edges with bend proportional to distance
   273→   273→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   274→   274→      const mx = (ax + bx) / 2;
   275→   275→      const my = (ay + by) / 2;
   276→   276→      const nx = -dy / dist;
   277→   277→      const ny = dx / dist;
   278→   278→      const cx = mx + nx * bend;
   279→   279→      const cy = my + ny * bend;
   280→   280→
   281→   281→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   282→   282→      const prob = edgeProbabilities[ei] ?? 0;
   283→   283→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   284→   284→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   285→   285→
   286→   286→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.72 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   287→   287→
   288→   288→      // Edge labels — skip deterministic root→branch and branch→event edges (index < 6)
   289→   289→      if (isVisible && ei >= 6) {
   290→   290→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#cde4ef" font-size="10" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="4" stroke-linejoin="round" opacity="0.9" data-type="${type}" class="edge-label">${label}</text>`;
   291→   291→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3" stroke-linejoin="round" opacity="0.85" class="edge-prob">${probLabel}</text>`;
   292→   292→      }
   293→   293→    }
   294→   294→
   295→   295→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   296→   296→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   297→   297→
   298→   298→  useEffect(() => { drawEdges(); }, [drawEdges]);
   299→   299→
   300→   300→  useEffect(() => {
   301→   301→    const el = shellRef.current;
   302→   302→    if (!el) return;
   303→   303→    const observer = new ResizeObserver(() => drawEdges());
   304→   304→    observer.observe(el);
   305→   305→    return () => observer.disconnect();
   306→   306→  }, [drawEdges]);
   307→   307→
   308→   308→
   309→   309→  // ── Detail panel content ─────────────────────────────────────────
   310→   310→  const detailContent = useMemo(() => {
   311→   311→    if (!selectedNode) return null;
   312→   312→    const node = nodeMap[selectedNode];
   313→   313→    if (!node) return null;
   314→   314→
   315→   315→    // For terminal scenario nodes, show branch breakdown
   316→   316→    if (node.isTerminal && SCENARIO_KEYS.includes(selectedNode as typeof SCENARIO_KEYS[number])) {
   317→   317→      const contrib = pathContributions[selectedNode] ?? { trend: 0, breakout: 0, reversal: 0 };
   318→   318→      const meta = SCENARIO_META[selectedNode];
   319→   319→      const s = scenarios[selectedNode as keyof typeof scenarios];
   320→   320→      return (
   321→   321→        <div style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}>
   322→   322→          <h3 style={{ fontSize: '14px', fontWeight: 700, margin: 0, color: meta.color }}>{node.title}</h3>
   323→   323→          <p style={{ fontSize: '12px', color: '#c4d8e3', lineHeight: 1.95, margin: 0 }}>{node.desc}</p>
   324→   324→          <span style={{
   325→   325→            display: 'inline-block', padding: '4px 8px', margin: '3px 2px', borderRadius: 8,
   326→   326→            color: '#d7eaf1', fontSize: '10px',
   327→   327→            background: 'rgba(255,255,255,.06)', border: '1px solid rgba(255,255,255,.09)',
   328→   328→          }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   329→   329→          <div style={{ fontSize: '12px', color: '#c4d8e3' }} dir="ltr">{nodeValues[selectedNode] ?? '--'}</div>
   330→   330→
   331→   331→          <div style={{ marginTop: 12, paddingTop: 12, borderTop: `1px solid ${D.line}` }}>
   332→   332→            <p style={{ fontSize: '12px', fontWeight: 700, marginBottom: 8, color: '#d8eff9' }}>سهم هر استراتژی:</p>
   333→   333→            {Object.entries(BRANCH_META).map(([bKey, bMeta]) => {
   334→   334→              const val = contrib[bKey as 'trend' | 'breakout' | 'reversal'];
   335→   335→              const pct = (val * 100).toFixed(1);
   336→   336→              return (
   337→   337→                <div key={bKey} style={{
   338→   338→                  display: 'flex', alignItems: 'center', justifyContent: 'space-between',
   339→   339→                  padding: '9px 0', borderBottom: `1px dashed rgba(175,210,225,.15)`,
   340→   340→                }}>
   341→   341→                  <div style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
   342→   342→                    <span style={{ display: 'inline-block', width: 8, height: 8, borderRadius: '50%', background: bMeta.color }} />
   343→   343→                    <span style={{ fontSize: '11px', color: '#d8eff9' }}>{bMeta.label}</span>
   344→   344→                  </div>
   345→   345→                  <span style={{ fontSize: '11px', fontWeight: 700, color: bMeta.color }}>{toPersianDigits(pct)}٪</span>
   346→   346→                </div>
   347→   347→              );
   348→   348→            })}
   349→   349→            <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '9px 0' }}>
   350→   350→              <span style={{ fontSize: '11px', fontWeight: 700, color: '#d8eff9' }}>مجموع</span>
   351→   351→              <span style={{ fontSize: '11px', fontWeight: 900, color: meta.color }}>{toFa(s?.probability ?? 0)}٪</span>
   352→   352→            </div>
   353→   353→          </div>
   354→   354→        </div>
   355→   355→      );
   356→   356→    }
   357→   357→
   358→   358→    // For non-terminal nodes, show incoming/outgoing edges
   359→   359→    const inputs = edges.filter(e => e.to === selectedNode);
   360→   360→    const outputs = edges.filter(e => e.from === selectedNode);
   361→   361→
   362→   362→    return (
   363→   363→      <div style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}>
   364→   364→        <h3 style={{ fontSize: '14px', fontWeight: 700, margin: 0, color: '#d8eff9' }}>{node.title}</h3>
   365→   365→        <p style={{ fontSize: '12px', color: '#c4d8e3', lineHeight: 1.95, margin: 0 }}><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   366→   366→        <span style={{
   367→   367→          display: 'inline-block', padding: '4px 8px', margin: '3px 2px', borderRadius: 8,
   368→   368→          color: '#d7eaf1', fontSize: '10px',
   369→   369→          background: 'rgba(255,255,255,.06)', border: '1px solid rgba(255,255,255,.09)',
   370→   370→        }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   371→   371→        <p style={{ fontSize: '12px', color: '#c4d8e3', lineHeight: 1.95, margin: 0 }}>{node.desc}</p>
   372→   372→        {inputs.length > 0 && (
   373→   373→          <div>
   374→   374→            <p style={{ fontSize: '12px', fontWeight: 600, marginBottom: 4, color: '#d8eff9' }}>مسیرهای ورودی ({toFa(inputs.length)}):</p>
   375→   375→            <ul style={{ margin: '12px 0 0', padding: 0, listStyle: 'none' }}>
