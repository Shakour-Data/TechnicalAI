   767→   767→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   768→   768→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   769→   769→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   770→   770→  grade?: string; overlapCount?: number; touchCount?: number;
   771→   771→}) {
   772→   772→  const isSupport = type === 'support';
   773→   773→  const isTargetType = type === 'target';
   774→   774→
   775→   775→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   776→   776→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   777→   777→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   778→   778→
   779→   779→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   780→   780→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   781→   781→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   782→   782→
   783→   783→  return (
   784→   784→    <div
   785→   785→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   786→   786→      style={{
   787→   787→        background: bgAccent,
   788→   788→        border: `1px solid ${borderAccent}`,
   789→   789→        borderRight: `3px solid ${accentColor}`,
   790→   790→      }}
   791→   791→    >
   792→   792→      {/* Label + Price */}
   793→   793→      <div className="flex-1 min-w-0">
   794→   794→        <div className="flex items-center gap-2">
   795→   795→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   796→   796→          {isTarget && <span className="text-[9px]">🎯</span>}
   797→   797→          {overlapCount !== undefined && overlapCount > 0 && (
   798→   798→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   799→   799→              {toPersianDigits(overlapCount)} هم‌پوشانی
   800→   800→            </span>
   801→   801→          )}
   802→   802→        </div>
   803→   803→        <div className="flex items-baseline gap-1">
   804→   804→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   805→   805→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   806→   806→        </div>
   807→   807→      </div>
   808→   808→
   809→   809→      {/* Strength + Grade */}
   810→   810→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   811→   811→        <div className="flex items-center gap-1.5">
   812→   812→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   813→   813→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   814→   814→          <span className="text-[9px] text-gray-500">/۱۰</span>
   815→   815→        </div>
   816→   816→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   817→   817→          <div
   818→   818→            className="h-full rounded-full transition-all duration-500"
   819→   819→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   820→   820→          />
   821→   821→        </div>
   822→   822→        {touchCount !== undefined && touchCount > 0 && (
   823→   823→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   824→   824→        )}
   825→   825→      </div>
   826→   826→    </div>
   827→   827→  );
