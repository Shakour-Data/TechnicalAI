   810→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   811→function LevelBox({ index, label, price, strength, isTarget, type, unit }: {
   812→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   813→}) {
   814→  const isSupport = type === 'support';
   815→  const isTargetType = type === 'target';
   816→
   817→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   818→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   819→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   820→
   821→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   822→  const strengthLabel = strength >= 8 ? 'بسیار قوی' : strength >= 6 ? 'قوی' : strength >= 4 ? 'متوسط' : strength >= 2 ? 'ضعیف' : 'بسیار ضعیف';
   823→  const strengthColor = strength >= 8 ? '#34c98b' : strength >= 6 ? '#3ad5db' : strength >= 4 ? '#ffb11b' : strength >= 2 ? '#ff7b32' : '#ef4d62';
   824→
   825→  // Distance from current price (if available from context)
   826→  return (
   827→    <div
   828→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   829→      style={{
   830→        background: bgAccent,
   831→        border: `1px solid ${borderAccent}`,
   832→        borderRight: `3px solid ${accentColor}`,
   833→      }}
   834→    >
   835→      {/* Label + Price */}
   836→      <div className="flex-1 min-w-0">
   837→        <div className="flex items-center gap-2">
   838→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   839→          {isTarget && <span className="text-[9px]">🎯</span>}
   840→        </div>
   841→        <span className="text-sm font-black text-gray-100 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   842→        <span className="text-[10px] text-gray-500 mr-1">{unit || 'ریال'}</span>
   843→      </div>
   844→
   845→      {/* Strength bar */}
   846→      <div className="flex flex-col items-end gap-1 min-w-[80px]">
   847→        <div className="flex items-center gap-1.5">
   848→          <span className="text-[9px] font-medium" style={{ color: strengthColor }}>{strengthLabel}</span>
   849→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(strength)}</span>
   850→          <span className="text-[9px] text-gray-500">/۱۰</span>
   851→        </div>
   852→        <div className="w-full h-1.5 rounded-full bg-white/5 overflow-hidden">
   853→          <div
   854→            className="h-full rounded-full transition-all duration-500"
   855→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   856→          />
   857→        </div>
   858→      </div>
   859→    </div>
   860→  );
   861→}
