   327→  if (id === 'sr') {
   328→    if (adx < 20) s += 5;
   329→    if (nearResistance) s += 4;
   330→    if (nearSupport) s += 4;
   331→  }
   332→
   333→  if (id === 'harmonic') {
   334→    if ((bbPosition > 80 || bbPosition < 20) && (nearResistance || nearSupport)) s += 5;
   335→    if (input.priceVsMa21 !== input.priceVsMa100) s += 3;
   336→    if (nearResistance || nearSupport) s += 2;
   337→  }
   338→
   339→  if (id === 'elliott') {
   340→    if (adx > 30 && trendR2 > 0.8) s += 5;
   341→    if (trendDirection === 'up' && bbPosition > 75) s += 3;
   342→    if (trendDirection === 'down' && bbPosition < 25) s += 3;
   343→  }
   344→
   345→  if (id === 'psychology') {
   346→    if (adx < 15 || (rsi > 70 && trendDirection === 'down') || (rsi < 30 && trendDirection === 'up')) s += 5;
   347→    if (hasVolume && obv !== 0 && Math.sign(obv) !== (trendDirection === 'up' ? 1 : -1)) s += 4;
   348→    if (rsi > 75 || rsi < 25) s += 2;
   349→  }
   350→
   351→  // Baseline so no school gets zero
   352→  return s + 1;
   353→}

   355→// ─── Style Scoring (per-item: checks style.id) ─────────────────
   356→function scoreStyle(item: StyleDef, input: NarrativeInput): number {
   357→  let s = 0;
   358→  const id = item.id;
   359→  const { adx, rsi, stochK, macdHist, obv, hasVolume, cci, bbPosition, trendDirection, nearResistance, nearSupport } = input;
   360→
   361→  if (id === 'conservative') {
   362→    if (adx < 20) s += 6;
   363→    if (rsi > 40 && rsi < 60) s += 3;
   364→    if (bbPosition > 30 && bbPosition < 70) s += 2;
   365→  }
   366→
   367→  if (id === 'scalper') {
   368→    if (adx > 25 && trendDirection === 'up' && macdHist > 0) s += 6;
   369→    if (stochK > 50 && stochK < 80) s += 2;
   370→    if (bbPosition > 50) s += 2;
   371→  }
   372→
   373→  if (id === 'trendFollower') {
   374→    if (adx > 25) s += 5;
   375→    if (trendDirection === 'up' && macdHist > 0) s += 3;
   376→    if (trendDirection === 'down' && macdHist < 0) s += 3;
   377→  }
   378→
   379→  if (id === 'bearish') {
   380→    if ((rsi > 70 || stochK > 80) && nearResistance) s += 7;
   381→    if (macdHist < 0 && rsi > 60) s += 4;
   382→    if (cci > 100) s += 2;
   383→  }
   384→
   385→  if (id === 'narrative') {
   386→    if (adx < 20) s += 5;
   387→    if ((macdHist > 0 && rsi < 50) || (macdHist < 0 && rsi > 50)) s += 3;
   388→  }
   389→
   390→  if (id === 'decision') {
   391→    if (nearResistance || nearSupport) s += 5;
   392→    if (adx > 25) s += 3;
   393→  }
   394→
   395→  if (id === 'volumeAnalyst') {
   396→    if (hasVolume && obv !== 0) s += 5;
   397→    if (hasVolume && Math.sign(obv) !== (trendDirection === 'up' ? 1 : -1)) s += 3;
   398→  }
   399→
   400→  if (id === 'patternAnalyst') {
   401→    if (nearResistance || nearSupport) s += 5;
   402→    if (bbPosition > 85 || bbPosition < 15) s += 3;
   403→  }
   404→
   405→  if (id === 'psychological') {
   406→    if (rsi > 75 || rsi < 25) s += 5;
   407→    if (stochK > 85 || stochK < 15) s += 3;
   408→    if (hasVolume && obv !== 0 && Math.sign(obv) !== (trendDirection === 'up' ? 1 : -1)) s += 2;
   409→  }
   410→
   411→  if (id === 'volatilityAnalyst') {
   412→    if (bbPosition > 20 && bbPosition < 80 && adx < 20) s += 6;
   413→    if (bbPosition > 90 || bbPosition < 10) s += 4;
   414→  }
   415→
   416→  return s + 1;
   417→}

   419→// ─── Tone Scoring (per-item: checks tone.id) ───────────────────
   420→function scoreTone(item: ToneDef, input: NarrativeInput): number {
   421→  let s = 0;
   422→  const id = item.id;
   423→  const { adx, rsi, stochK, macdHist, obv, hasVolume, bbPosition, trendDirection, nearResistance, nearSupport, dominantScenarioKey, trendR2 } = input;
   424→
   425→  if (id === 'formal') {
   426→    if (adx > 25) s += 3;
   427→    if (trendR2 > 0.7) s += 2;
   428→  }
   429→
   430→  if (id === 'fast') {
   431→    if (adx > 30 && macdHist > 0 && trendDirection === 'up') s += 5;
   432→    if (stochK > 60) s += 2;
   433→  }
   434→
   435→  if (id === 'philosophical') {
   436→    if (adx > 15 && adx < 35) s += 4;
   437→    if (rsi > 40 && rsi < 60) s += 2;
   438→  }
   439→
   440→  if (id === 'warning') {
   441→    if ((rsi > 70 || stochK > 80) && nearResistance) s += 7;
   442→    if (macdHist > 0 && rsi < 50) s += 3;
   443→    if ((rsi < 30 || stochK < 20) && nearSupport && trendDirection === 'down') s += 4;
   444→  }
   445→
   446→  if (id === 'storytelling') {
   447→    if (adx < 25) s += 4;
   448→    if (bbPosition > 30 && bbPosition < 70) s += 2;
   449→  }
   450→
   451→  if (id === 'stepByStep') {
   452→    if (nearResistance || nearSupport) s += 4;
   453→    if (adx > 25) s += 2;
   454→  }
   455→
   456→  if (id === 'skeptical') {
   457→    if ((macdHist > 0 && rsi < 50) || (macdHist < 0 && rsi > 50)) s += 6;
   458→    if (rsi > 60 && trendDirection === 'down') s += 3;
   459→    if (rsi < 40 && trendDirection === 'up') s += 3;
   460→  }
   461→
   462→  if (id === 'optimistic') {
   463→    if (trendDirection === 'up' && adx > 25 && rsi > 50 && rsi < 70) s += 6;
   464→    if (macdHist > 0 && obv > 0) s += 2;
   465→  }
   466→
   467→  if (id === 'simple') {
   468→    if (adx > 20 && adx < 40 && rsi > 40 && rsi < 60) s += 4;
   469→    if (!nearResistance && !nearSupport) s += 2;
   470→  }
   471→
   472→  if (id === 'quantitative') {
   473→    if (hasVolume && adx > 25) s += 3;
   474→    if (rsi > 70 || rsi < 30) s += 2;
   475→  }
   476→
   477→  if (id === 'mysterious') {
   478→    if (adx < 20) s += 4;
   479→    if ((macdHist > 0 && rsi < 50) || (macdHist < 0 && rsi > 50)) s += 3;
   480→    if (dominantScenarioKey === 'SC3') s += 2;
   481→  }
   482→
   483→  if (id === 'educational') {
   484→    if (nearResistance || nearSupport) s += 3;
   485→    if (rsi > 65 || rsi < 35) s += 2;
   486→  }
   487→
   488→  if (id === 'comparative') {
   489→    const probs = Object.values(input.scenarioProbabilities);
   490→    const maxProb = Math.max(...probs);
   491→    if (maxProb < 35) s += 5;
   492→    if (probs.filter(p => p > 20).length >= 3) s += 3;
   493→  }
   494→
   495→  if (id === 'balanced') {
   496→    if (adx < 20) s += 5;
   497→    if (rsi > 40 && rsi < 60) s += 3;
   498→  }
   499→
   500→  if (id === 'conclusive') {
   501→    if (adx > 25 && (rsi > 60 || rsi < 40)) s += 4;
   502→    if (dominantScenarioKey === 'SC1' || dominantScenarioKey === 'SC2') s += 2;
   503→  }
   504→
   505→  return s + 1;
   506→}

   508→// ═══════════════════════════════════════════════════════════════════════════════
   509→// Main Selector: Returns exactly ONE combination per request
   510→// ═══════════════════════════════════════════════════════════════════════════════
   511→
   512→export function selectNarrativeCombination(input: NarrativeInput): NarrativeCombination {
   513→  const schoolResults = scoreAll(SCHOOLS, input, scoreSchool);
   514→  const styleResults = scoreAll(STYLES, input, scoreStyle);
   515→  const toneResults = scoreAll(TONES, input, scoreTone);
   516→
   517→  return {
   518→    school: schoolResults[0].item,
   519→    style: styleResults[0].item,
   520→    tone: toneResults[0].item,
   521→    schoolScore: schoolResults[0].score,
   522→    styleScore: styleResults[0].score,
   523→    toneScore: toneResults[0].score,
   524→  };
   525→}

   527→// Helper: build NarrativeInput from VdesRequest data
   528→export function buildNarrativeInput(body: {
   529→  currentPrice: number;
   530→  trendDirection: string;
   531→  rsi: number;
   532→  adx: number;
   533→  stochK: number;
   534→  stochD: number;
   535→  macdLine: number;
   536→  macdSignal: number;
   537→  macdHist: number;
   538→  diPlus: number;
   539→  diMinus: number;
   540→  obv: number;
   541→  hasVolume: boolean;
   542→  cci: number;
   543→  mfi: number;
   544→  atr: number;
   545→  bollingerUpper: number;
   546→  bollingerMiddle: number;
   547→  bollingerLower: number;
   548→  trendAngle: number;
   549→  trendR2: number;
   550→  overallSignal: string;
   551→  scenarios: Record<string, { probability: number }>;
   552→  resistances: number[];
   553→  supports: number[];
   554→  ma21: number;
   555→  ma100: number;
   556→}): NarrativeInput {
   557→  const bbRange = body.bollingerUpper - body.bollingerLower;
   558→  const bbPosition = bbRange > 0
   559→    ? Math.max(0, Math.min(100, ((body.currentPrice - body.bollingerLower) / bbRange) * 100))
   560→    : 50;
   561→
   562→  const R1 = body.resistances[0] || body.currentPrice * 1.05;
   563→  const S1 = body.supports[0] || body.currentPrice * 0.95;
   564→
   565→  // Find dominant scenario
   566→  let dominantKey = 'SC3';
   567→  let dominantProb = 0;
   568→  for (const k of ['SC1', 'SC2', 'SC3', 'SC4', 'SC5'] as const) {
   569→    const p = body.scenarios[k]?.probability ?? 0;
   570→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
   571→  }
   572→
   573→  return {
   574→    price: body.currentPrice,
   575→    trendDirection: body.trendDirection,
   576→    rsi: body.rsi,
   577→    adx: body.adx,
   578→    stochK: body.stochK,
   579→    stochD: body.stochD,
   580→    macdLine: body.macdLine,
   581→    macdSignal: body.macdSignal,
   582→    macdHist: body.macdHist,
   583→    diPlus: body.diPlus,
   584→    diMinus: body.diMinus,
   585→    obv: body.obv,
   586→    hasVolume: body.hasVolume,
   587→    cci: body.cci,
   588→    mfi: body.mfi,
   589→    atr: body.atr,
   590→    bbPosition,
   591→    bbUpper: body.bollingerUpper,
   592→    bbLower: body.bollingerLower,
   593→    dominantScenarioKey: dominantKey,
   594→    scenarioProbabilities: {
   595→      SC1: body.scenarios.SC1?.probability ?? 0,
   596→      SC2: body.scenarios.SC2?.probability ?? 0,
   597→      SC3: body.scenarios.SC3?.probability ?? 0,
   598→      SC4: body.scenarios.SC4?.probability ?? 0,
   599→      SC5: body.scenarios.SC5?.probability ?? 0,
   600→    },
   601→    nearResistance: R1 > 0 ? Math.abs(body.currentPrice - R1) / R1 < 0.03 : false,
   602→    nearSupport: S1 > 0 ? Math.abs(body.currentPrice - S1) / S1 < 0.03 : false,
   603→    priceVsMa21: body.currentPrice > body.ma21 ? 'above' : 'below',
   604→    priceVsMa100: body.currentPrice > body.ma100 ? 'above' : 'below',
   605→    trendR2: body.trendR2,
   606→  };
   607→}
