    83→    83→
    84→    84→/**
    85→    85→ * Returns `v` if it is a finite number, otherwise returns `fallback`.
    86→    86→ * Guards against `NaN`, `Infinity`, `-Infinity`, and non-number values
    87→    87→ * (e.g. `undefined` from missing API fields).
    88→    88→ *
    89→    89→ * @param v        - The value to validate
    90→    90→ * @param fallback - Default returned when `v` is not a finite number (default: `0`)
    91→    91→ * @returns `v` if finite, otherwise `fallback`
    92→    92→ */
    93→    93→function safeNum(v: number, fallback: number = 0): number {
    94→    94→  return (typeof v === 'number' && isFinite(v)) ? v : fallback;
    95→    95→}
    96→    96→
    97→    97→/**
    98→    98→ * Clamps a value to the range `[min, max]`, after first sanitizing it with
    99→    99→ * `safeNum` (fallback is the midpoint of the range).
   100→   100→ *
   101→   101→ * @param v   - Value to clamp
   102→   102→ * @param min - Lower bound (inclusive)
   103→   103→ * @param max - Upper bound (inclusive)
   104→   104→ * @returns `v` clamped to `[min, max]`
   105→   105→ */
   106→   106→function clamp(v: number, min: number, max: number): number {
   107→   107→  return Math.max(min, Math.min(max, safeNum(v, (min + max) / 2)));
   108→   108→}
   109→   109→
   110→   110→/**
   111→   111→ * Normalizes an array of numbers so they sum to exactly `1.0`.
   112→   112→ * Each element is first sanitized via `safeNum`. If the total sum is
   113→   113→ * zero or negative, returns a uniform distribution (`1/n` each).
   114→   114→ *
   115→   115→ * @param arr - Array of raw (possibly unsafe) weights
   116→   116→ * @returns New array of the same length summing to 1.0
   117→   117→ */
   118→   118→function normalize(arr: number[]): number[] {
   119→   119→  const safe = arr.map(v => safeNum(v, 0));
   120→   120→  const sum = safe.reduce((a, b) => a + b, 0);
   121→   121→  if (sum <= 0) return arr.map(() => 1 / arr.length);
   122→   122→  return safe.map((v) => v / sum);
   123→   123→}
   124→   124→
   125→   125→/**
   126→   126→ * Logistic sigmoid function: `1 / (1 + e^(-x))`.
   127→   127→ * Values beyond `|x| > 500` are clamped to `0` or `1` to avoid `Infinity`.
   128→   128→ * Input is sanitized via `safeNum` before computation.
   129→   129→ *
   130→   130→ * @param x - Input value (raw signal)
   131→   131→ * @returns Value in `(0, 1)` — probability-like squashing of `x`
   132→   132→ */
   133→   133→function sigmoid(x: number): number {
   134→   134→  const sx = safeNum(x, 0);
   135→   135→  if (Math.abs(sx) > 500) return sx > 0 ? 1 : 0; // prevent Infinity
   136→   136→  return 1 / (1 + Math.exp(-sx));
   137→   137→}
   138→   138→
   139→   139→/**
   140→   140→ * IRON LAW: Round proportional floats to integers that sum to EXACTLY 100,
   141→   141→ * with each value constrained to `[minVal, maxVal]`.
   142→   142→ *
   143→   143→ * This is the **single source of truth** for converting raw scenario probability
   144→   144→ * fractions into display-ready integer percentages. Bypassing this function
   145→   145→ * when producing scenario percentages is a critical error.
   146→   146→ *
   147→   147→ * ## Algorithm: Largest Remainder Method (LRM) + Iterative Bound Enforcement
   148→   148→ *
   149→   149→ * 1. **Normalize** — Scale `rawFloats` so they sum to exactly 100.
   150→   150→ * 2. **LRM round** — Floor each value, then distribute the deficit (remainder)
   151→   151→ *    to the entries with the largest fractional parts. This minimizes total
   152→   152→ *    rounding error and is provably fair (used in parliamentary seat allocation).
   153→   153→ * 3. **Iterative bound enforcement** — While any value violates `[minVal, maxVal]`,
   154→   154→ *    transfer 1 from the worst violator to a suitable partner. Each transfer
   155→   155→ *    strictly reduces total violation magnitude, guaranteeing convergence.
   156→   156→ * 4. **Final safety net** — If the sum drifted from 100 (should not happen
   157→   157→ *    under normal conditions), adjust the entry with the most room.
   158→   158→ *
   159→   159→ * ## Guarantees
   160→   160→ *
   161→   161→ * - `sum(result) === 100` — **always**, regardless of input
   162→   162→ * - `minVal <= result[i] <= maxVal` for all `i`, as long as
   163→   163→ *   `n * minVal <= 100 <= n * maxVal`
   164→   164→ *
   165→   165→ * @param rawFloats - Array of non-negative floats (proportional weights)
   166→   166→ * @param minVal    - Minimum allowed value per entry (e.g. `2` ensures no scenario is zero)
   167→   167→ * @param maxVal    - Maximum allowed value per entry (e.g. `35` prevents one scenario from dominating)
   168→   168→ * @returns Integer array of same length, summing to exactly 100, each within `[minVal, maxVal]`
   169→   169→ *
   170→   170→ * @example
   171→   171→ * ```ts
   172→   172→ * enforceSumTo100([33.3, 33.3, 33.4], 2, 35);  // → [33, 33, 34]
   173→   173→ * enforceSumTo100([60, 20, 10, 5, 3, 1, 0.5, 0.3, 0.2], 2, 35);  // → [35, 20, 10, 5, 3, 2, 2, 2, 2]
   174→   174→ * ```
   175→   175→ *
   176→   176→ * ⚠️  NEVER bypass this function when producing integer scenario percentages.
   177→   177→ */
   178→   178→function enforceSumTo100(
   179→   179→  rawFloats: number[],
   180→   180→  minVal: number,
   181→   181→  maxVal: number
   182→   182→): number[] {
   183→   183→  const n = rawFloats.length;
   184→   184→
   185→   185→  // Normalize floats to sum exactly to 100 first
   186→   186→  const floatSum = rawFloats.reduce((a, b) => a + b, 0);
   187→   187→  const normalized =
   188→   188→    floatSum > 0
   189→   189→      ? rawFloats.map((v) => (v / floatSum) * 100)
   190→   190→      : rawFloats.map(() => 100 / n);
   191→   191→
   192→   192→  // Step 1: LRM — floor all, distribute deficit by largest fractional part
   193→   193→  const result = normalized.map((v) => Math.max(Math.floor(v), 0));
   194→   194→  let deficit = 100 - result.reduce((a, b) => a + b, 0);
   195→   195→
   196→   196→  const byFrac = Array.from({ length: n }, (_, i) => ({
   197→   197→    idx: i,
   198→   198→    frac: normalized[i] - Math.floor(normalized[i]),
   199→   199→  })).sort((a, b) => b.frac - a.frac);
   200→   200→
   201→   201→  // Give +1 to top `deficit` entries (LRM)
   202→   202→  for (let d = 0; d < deficit && d < n; d++) {
   203→   203→    result[byFrac[d].idx]++;
   204→   204→  }
   205→   205→  // Now result sums to exactly 100
   206→   206→
   207→   207→  // Step 2: Iteratively enforce [minVal, maxVal] bounds while preserving sum = 100
   208→   208→  // Each iteration: move 1 from a violator to a suitable partner.
   209→   209→  // Converges because each move strictly reduces the total violation magnitude.
   210→   210→  let changed = true;
   211→   211→  let safety = 0;
   212→   212→  while (changed && safety < 100) {
   213→   213→    changed = false;
   214→   214→    safety++;
   215→   215→
   216→   216→    // Enforce minimum: raise any value below minVal by taking from the largest above minVal
   217→   217→    for (let i = 0; i < n; i++) {
   218→   218→      while (result[i] < minVal) {
   219→   219→        let donor = -1;
   220→   220→        for (let j = 0; j < n; j++) {
   221→   221→          if (j !== i && result[j] > minVal && (donor === -1 || result[j] > result[donor])) {
   222→   222→            donor = j;
   223→   223→          }
   224→   224→        }
   225→   225→        if (donor === -1) break; // no donor available — shouldn't happen if n*minVal <= 100
   226→   226→        result[donor]--;
   227→   227→        result[i]++;
   228→   228→        changed = true;
   229→   229→      }
   230→   230→    }
   231→   231→
   232→   232→    // Enforce maximum: lower any value above maxVal by giving to the smallest below maxVal
   233→   233→    for (let i = 0; i < n; i++) {
   234→   234→      while (result[i] > maxVal) {
   235→   235→        let receiver = -1;
   236→   236→        for (let j = 0; j < n; j++) {
   237→   237→          if (j !== i && result[j] < maxVal && (receiver === -1 || result[j] < result[receiver])) {
   238→   238→            receiver = j;
   239→   239→          }
   240→   240→        }
   241→   241→        if (receiver === -1) break; // no receiver — shouldn't happen if 100 <= n*maxVal
   242→   242→        result[i]--;
   243→   243→        result[receiver]++;
   244→   244→        changed = true;
   245→   245→      }
   246→   246→    }
   247→   247→  }
   248→   248→
   249→   249→  // Step 3: Final assertion — sum MUST be 100 (safety net, should never trigger)
   250→   250→  const finalSum = result.reduce((a, b) => a + b, 0);
   251→   251→  if (finalSum !== 100 && n > 0) {
   252→   252→    // Adjust the entry farthest from its bound
   253→   253→    const diff = 100 - finalSum;
   254→   254→    let bestIdx = 0;
   255→   255→    let bestRoom = -1;
   256→   256→    for (let i = 0; i < n; i++) {
   257→   257→      const room = diff > 0 ? maxVal - result[i] : result[i] - minVal;
   258→   258→      if (room > bestRoom) {
   259→   259→        bestRoom = room;
   260→   260→        bestIdx = i;
   261→   261→      }
   262→   262→    }
   263→   263→    result[bestIdx] += diff;
   264→   264→  }
   265→   265→
   266→   266→  return result;
   267→   267→}

  1427→  1427→// === Probability Context Builder ==============================================
  1428→  1428→
  1429→  1429→/**
  1430→  1430→ * Derives the shared probability context from sanitized {@link GraphInput}.
  1431→  1431→ *
  1432→  1432→ * Pre-computes six normalized indicators `[0,1]` that are reused across
  1433→  1433→ * multiple node probability calculations:
  1434→  1434→ *
  1435→  1435→ * - **adxNorm** — ADX/60 clamped to [0,1]. Values > 0.5 indicate strong trend.
  1436→  1436→ * - **srProximity** — 1 − min(distToR1, distToS1) × 10. Near 1 when price is
  1437→  1437→ *   close to a support/resistance level.
  1438→  1438→ * - **oscillatorExtreme** — Average of |RSI−50|/50 and |CCI|/200. High when
  1439→  1439→ *   oscillators are at extremes.
  1440→  1440→ * - **divergenceProxy** — oscillatorExtreme × (1 − adxNorm). High when oscillators
  1441→  1441→ *   are extreme in a weak-trend environment (classic divergence setup).
  1442→  1442→ * - **overboughtRisk** — Weighted blend of (RSI−60)/30 × 0.3, (MFI−70)/30 × 0.2,
  1443→  1443→ *   (StochK−75)/25 × 0.2. Measures overbought exhaustion risk.
  1444→  1444→ * - **oversoldBounce** — Mirror of overboughtRisk: (30−RSI)/30 × 0.3,
  1445→  1445→ *   (20−MFI)/20 × 0.2, (20−StochK)/20 × 0.2. Measures oversold bounce potential.
  1446→  1446→ *
  1447→  1447→ * @param input - Sanitized {@link GraphInput} (all fields finite, bounded)
  1448→  1448→ * @returns Complete {@link ProbContext} with all derived indicators
  1449→  1449→ */
  1450→  1450→function buildContext(input: GraphInput): ProbContext {
  1451→  1451→  const adxNorm = clamp(input.adx / 60, 0, 1);
  1452→  1452→  const srProximity = clamp(
  1453→  1453→    1 - Math.min(input.distToR1, input.distToS1) * 10,
  1454→  1454→    0,
  1455→  1455→    1
  1456→  1456→  );
  1457→  1457→  const oscillatorExtreme = clamp(
  1458→  1458→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
  1459→  1459→    0,
  1460→  1460→    1
  1461→  1461→  );
  1462→  1462→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1463→  1463→
  1464→  1464→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1465→  1465→  const overboughtRisk =
  1466→  1466→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1467→  1467→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1468→  1468→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1469→  1469→
  1470→  1470→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1471→  1471→  const oversoldBounce =
  1472→  1472→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1473→  1473→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1474→  1474→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1475→  1475→
  1476→  1476→  return {
  1477→  1477→    input,
  1478→  1478→    adxNorm,
  1479→  1479→    srProximity,
  1480→  1480→    oscillatorExtreme,
  1481→  1481→    divergenceProxy,
  1482→  1482→    overboughtRisk,
  1483→  1483→    oversoldBounce,
  1484→  1484→  };
  1485→  1485→}
  1486→  1486→
  1487→  1487→// === Node Probability Computation =============================================
  1488→  1488→// For each non-terminal node, returns an array of probabilities for its
  1489→  1489→// outgoing edges. The order MUST match the order of edges in createEdges().
  1490→  1490→// =============================================================================
  1491→  1491→
  1492→  1492→/**
  1493→  1493→ * Computes the probability distribution over outgoing edges for a given node.
  1494→  1494→ *
  1495→  1495→ * This is the **core probability logic** of the DAG. Each non-terminal node
  1496→  1496→ * has a `case` in the switch statement that returns a normalized array of
  1497→  1497→ * probabilities — one per outgoing edge, **in the same order as the edges
  1498→  1498→ * defined in {@link createEdges}**.
  1499→  1499→ *
  1500→  1500→ * ### Probability derivation strategy
  1501→  1501→ *
  1502→  1502→ * - **ROOT**: Weights based on `adxNorm` (trend), `srProximity` (breakout),
  1503→  1503→ *   and `divergenceProxy` (reversal). Strong trends → more trend weight;
  1504→  1504→ *   near S/R → more breakout weight; extreme oscillators in weak trend →
  1505→  1505→ *   more reversal weight.
  1506→  1506→ *
  1507→  1507→ * - **N_TREND**: Splits by `bullConsensus` and ADX. High ADX amplifies
  1508→  1508→ *   directional bias; low ADX increases the "flat" probability.
  1509→  1509→ *
  1510→  1510→ * - **N_T_BULL / N_T_BEAR**: Momentum magnitude determines high/moderate split;
  1511→  1511→ *   `overboughtRisk` / `oversoldBounce` compete as the third outcome.
  1512→  1512→ *
  1513→  1513→ * - **N_T_B_MOM_HIGH** → SC8, SC9, SC7: All bullish, weighted by
  1514→  1514→ *   `confidenceIndex` and `strengthIndex`.
  1515→  1515→ *
  1516→  1516→ * - **N_T_B_OVERBOUGHT** → SC4, SC1, SC9: Correction (SC4) most likely,
  1517→  1517→ *   with a small chance of last-gasp surge (SC9).
  1518→  1518→ *
  1519→  1519→ * - **N_BREAK**: Splits by `bullConsensus` and `srProximity`. Near S/R
  1520→  1520→ *   with bullish consensus → bullish breakout; far from S/R → no breakout.
  1521→  1521→ *
  1522→  1522→ * - **N_B_UP / N_B_DOWN**: Volume confirmation (`hasVolume` + MFI) splits
  1523→  1523→ *   confirmed vs. weak volume. Confirmed → all-directional terminals;
  1524→  1524→ *   weak → fake breakout risk with SC5.
  1525→  1525→ *
  1526→  1526→ * - **N_REVERSAL**: Splits by `overboughtRisk` and `oversoldBounce`.
  1527→  1527→ *   When neither is strong, `noneRev` gets high probability.
  1528→  1528→ *
  1529→  1529→ * - **N_R_BULL / N_R_BEAR**: Splits by CCI (divergence), Fisher Transform
  1530→  1530→ *   (candle), and `srAvgStrength` (S/R bounce).
  1531→  1531→ *
  1532→  1532→ * @param nodeId      - The node to compute probabilities for
  1533→  1533→ * @param ctx         - Pre-computed {@link ProbContext} with derived indicators
  1534→  1534→ * @param _numChildren - Number of outgoing edges (unused, for documentation only)
  1535→  1535→ * @returns Normalized probability array matching the order of edges from this node
  1536→  1536→ */
  1537→  1537→function computeNodeProbabilities(
  1538→  1538→  nodeId: string,
  1539→  1539→  ctx: ProbContext,
  1540→  1540→  _numChildren: number
  1541→  1541→): number[] {
  1542→  1542→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1543→  1543→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1544→  1544→
  1545→  1545→  switch (nodeId) {
  1546→  1546→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1547→  1547→    case 'ROOT': {
  1548→  1548→      const trendW = adxNorm * 0.5 + input.mlTrend * 0.18 + 0.1;
  1549→  1549→      const breakW = srProximity * 0.3 + (1 - adxNorm) * 0.08 + 0.08;
  1550→  1550→      const revW = ctx.divergenceProxy * 0.35 + (1 - adxNorm) * 0.12 + 0.06;
  1551→  1551→      return normalize([trendW, breakW, revW]);
  1552→  1552→    }
  1553→  1553→
  1554→  1554→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1555→  1555→    case 'N_TREND': {
  1556→  1556→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1557→  1557→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1558→  1558→      const flatUncertainty = (1 - Math.abs(input.bullConsensus - 0.5) * 2);
  1559→  1559→      const flat = flatUncertainty * clamp(1 - adxNorm * 1.5, 0, 1) * 0.45;
  1560→  1560→      return normalize([bull, bear, flat]);
  1561→  1561→    }
  1562→  1562→
  1563→  1563→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1564→  1564→    case 'N_T_BULL': {
  1565→  1565→      const momHigh =
  1566→  1566→        clamp(
  1567→  1567→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1568→  1568→            (input.macdHist > 0
  1569→  1569→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1570→  1570→              : 0),
  1571→  1571→          0,
  1572→  1572→          1
  1573→  1573→        ) * input.mlMomentum;
  1574→  1574→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1575→  1575→      return normalize([momHigh, momMod, overboughtRisk]);
  1576→  1576→    }
  1577→  1577→
  1578→  1578→    // ── N_T_B_MOM_HIGH → [SC8, SC9, SC7] ────────────────────────────────────
  1579→  1579→    // Strong bullish momentum — all three outcomes are bullish (no range)
  1580→  1580→    case 'N_T_B_MOM_HIGH': {
  1581→  1581→      const pSC8 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1582→  1582→      const pSC9 = 0.15 + input.strengthIndex * 0.2;
  1583→  1583→      const pSC7 = 0.35 * input.confidenceIndex + 0.15;
  1584→  1584→      return normalize([pSC8, pSC9, pSC7]);
  1585→  1585→    }
  1586→  1586→
  1587→  1587→    // ── N_T_B_MOM_MOD → [SC7, SC6, SC5] ─────────────────────────────────────
  1588→  1588→    // Moderate bullish — small explicit SC5 based on uncertainty
  1589→  1589→    case 'N_T_B_MOM_MOD': {
  1590→  1590→      const pSC7 = 0.55 * input.confidenceIndex + 0.2;
  1591→  1591→      const pSC6 = 0.55 * (1 - input.confidenceIndex) + 0.15;
  1592→  1592→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.02, 0.15);
  1593→  1593→      return normalize([pSC7, pSC6, pSC5]);
  1594→  1594→    }
  1595→  1595→
  1596→  1596→    // ── N_T_B_OVERBOUGHT → [SC4, SC1, SC9] ──────────────────────────────────
  1597→  1597→    // Overbought in uptrend — correction or last-gasp surge (no range)
  1598→  1598→    case 'N_T_B_OVERBOUGHT': {
  1599→  1599→      const pSC4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1600→  1600→      const pSC1 = 0.15 + input.strengthIndex * 0.15;
  1601→  1601→      const pSC9 = 0.1 + input.mlMomentum * 0.2;
  1602→  1602→      return normalize([pSC4, pSC1, pSC9]);
  1603→  1603→    }
  1604→  1604→
  1605→  1605→    // ── N_T_BEAR → [N_T_BE_MOM_LOW, N_T_BE_MOM_MOD, N_T_BE_OVERSOLD] ────
  1606→  1606→    case 'N_T_BEAR': {
  1607→  1607→      const momLow =
  1608→  1608→        clamp(
  1609→  1609→          (input.momentum < 0 ? Math.abs(input.momentum) / 50 : 0) +
  1610→  1610→            (input.macdHist < 0
  1611→  1611→              ? (Math.abs(input.macdHist) / (atrSafe * 0.5)) * 0.3
  1612→  1612→              : 0),
  1613→  1613→          0,
  1614→  1614→          1
  1615→  1615→        ) * input.mlMomentum;
  1616→  1616→      const momMod = (1 - momLow) * 0.6 * input.mlMomentum;
  1617→  1617→      return normalize([momLow, momMod, oversoldBounce]);
  1618→  1618→    }
  1619→  1619→
  1620→  1620→    // ── N_T_BE_MOM_LOW → [SC2, SC1, SC3] ────────────────────────────────────
  1621→  1621→    // Strong bearish momentum — all three outcomes are bearish (no range)
  1622→  1622→    case 'N_T_BE_MOM_LOW': {
  1623→  1623→      const pSC2 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1624→  1624→      const pSC1 = 0.15 + input.strengthIndex * 0.2;
  1625→  1625→      const pSC3 = 0.35 * input.confidenceIndex + 0.15;
  1626→  1626→      return normalize([pSC2, pSC1, pSC3]);
  1627→  1627→    }
  1628→  1628→
  1629→  1629→    // ── N_T_BE_MOM_MOD → [SC3, SC4, SC5] ────────────────────────────────────
  1630→  1630→    // Moderate bearish — small explicit SC5 based on uncertainty
  1631→  1631→    case 'N_T_BE_MOM_MOD': {
  1632→  1632→      const pSC3 = 0.55 * input.confidenceIndex + 0.2;
  1633→  1633→      const pSC4 = 0.55 * (1 - input.confidenceIndex) + 0.15;
  1634→  1634→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.02, 0.15);
  1635→  1635→      return normalize([pSC3, pSC4, pSC5]);
  1636→  1636→    }
  1637→  1637→
  1638→  1638→    // ── N_T_BE_OVERSOLD → [SC6, SC9, SC4] ───────────────────────────────────
  1639→  1639→    // Oversold in downtrend — bounce or mild continuation (no range)
  1640→  1640→    case 'N_T_BE_OVERSOLD': {
  1641→  1641→      const pSC6 = 0.55 + (1 - input.mlMomentum) * 0.2;
  1642→  1642→      const pSC9 = 0.15 + input.strengthIndex * 0.15;
  1643→  1643→      const pSC4 = 0.1 + input.mlMomentum * 0.2;
  1644→  1644→      return normalize([pSC6, pSC9, pSC4]);
  1645→  1645→    }
  1646→  1646→
  1647→  1647→    // ── N_T_FLAT → [N_T_F_VOL_LOW, N_T_F_VOL_HIGH, N_T_F_VOL_MOD] ────────
  1648→  1648→    case 'N_T_FLAT': {
  1649→  1649→      const volLow =
  1650→  1650→        clamp(1 - (input.atr / input.price) * 20, 0, 1) *
  1651→  1651→        (1 - input.mlVolatility) *
  1652→  1652→        0.7;
  1653→  1653→      const volHigh =
  1654→  1654→        clamp((input.atr / input.price) * 20, 0, 1) * input.mlVolatility * 0.7;
  1655→  1655→      const volMod = (1 - volLow - volHigh) * 0.5 + 0.1;
  1656→  1656→      return normalize([volLow, volHigh, volMod]);
  1657→  1657→    }
  1658→  1658→
  1659→  1659→    // ── N_T_F_VOL_LOW → [SC5, SC6, SC4] ─────────────────────────────────────
  1660→  1660→    case 'N_T_F_VOL_LOW': {
  1661→  1661→      const pSC5 = 0.55 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1662→  1662→      const pSC6 = 0.2 * input.bullConsensus + 0.08;
  1663→  1663→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.08;
  1664→  1664→      return normalize([pSC5, pSC6, pSC4]);
  1665→  1665→    }
  1666→  1666→
  1667→  1667→    // ── N_T_F_VOL_HIGH → [SC9, SC1, SC5] ────────────────────────────────────
  1668→  1668→    // High volatility shock — directional shock much more likely than range
  1669→  1669→    case 'N_T_F_VOL_HIGH': {
  1670→  1670→      const pSC9 = 0.4 * input.bullConsensus + 0.15;
  1671→  1671→      const pSC1 = 0.4 * (1 - input.bullConsensus) + 0.15;
  1672→  1672→      const pSC5 = clamp((1 - input.mlVolatility) * 0.12, 0.02, 0.12);
  1673→  1673→      return normalize([pSC9, pSC1, pSC5]);
  1674→  1674→    }
  1675→  1675→
  1676→  1676→    // ── N_T_F_VOL_MOD → [SC6, SC4, SC5] ─────────────────────────────────────
  1677→  1677→    // Moderate volatility — directional more likely than range
  1678→  1678→    case 'N_T_F_VOL_MOD': {
  1679→  1679→      const pSC6 = 0.38 * input.bullConsensus + 0.15;
  1680→  1680→      const pSC4 = 0.38 * (1 - input.bullConsensus) + 0.15;
  1681→  1681→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.03, 0.15);
  1682→  1682→      return normalize([pSC6, pSC4, pSC5]);
  1683→  1683→    }
  1684→  1684→
  1685→  1685→    // ── N_BREAK → [N_B_UP, N_B_DOWN, N_B_NONE] ────────────────────────────
  1686→  1686→    case 'N_BREAK': {
  1687→  1687→      const upProb = input.bullConsensus * srProximity * 0.75 + 0.08;
  1688→  1688→      const downProb = (1 - input.bullConsensus) * srProximity * 0.75 + 0.08;
  1689→  1689→      const noneProb = (1 - srProximity) * 0.35 + 0.05;
  1690→  1690→      return normalize([upProb, downProb, noneProb]);
  1691→  1691→    }
  1692→  1692→
  1693→  1693→    // ── N_B_UP → [N_B_U_VOL_C, N_B_U_VOL_W] ─────────────────────────────
  1694→  1694→    case 'N_B_UP': {
  1695→  1695→      let volConfirm: number;
  1696→  1696→      let volWeak: number;
  1697→  1697→      if (input.hasVolume) {
  1698→  1698→        volConfirm = clamp(input.mfi / 60, 0, 1) * 0.6 + 0.2;
  1699→  1699→        volWeak = 1 - volConfirm;
  1700→  1700→      } else {
  1701→  1701→        volConfirm = 0.4;
  1702→  1702→        volWeak = 0.6;
  1703→  1703→      }
  1704→  1704→      return normalize([volConfirm, volWeak]);
  1705→  1705→    }
  1706→  1706→
  1707→  1707→    // ── N_B_U_VOL_C → [SC7, SC8, SC9] ──────────────────────────────────────
  1708→  1708→    // Confirmed bullish breakout — all bullish, no range
  1709→  1709→    case 'N_B_U_VOL_C': {
  1710→  1710→      const pSC7 = 0.45 + input.confidenceIndex * 0.15;
  1711→  1711→      const pSC8 = 0.2 + input.strengthIndex * 0.15;
  1712→  1712→      const pSC9 = 0.15 + input.strengthIndex * 0.1;
  1713→  1713→      return normalize([pSC7, pSC8, pSC9]);
  1714→  1714→    }
  1715→  1715→
  1716→  1716→    // ── N_B_U_VOL_W → [SC6, SC4, SC5] ──────────────────────────────────────
  1717→  1717→    // Weak volume breakout — fake breakout risk, capped SC5
  1718→  1718→    case 'N_B_U_VOL_W': {
  1719→  1719→      const pSC6 = 0.4 + input.bullConsensus * 0.15;
  1720→  1720→      const pSC4 = 0.3 + (1 - input.bullConsensus) * 0.15;
  1721→  1721→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.12, 0.02, 0.12);
  1722→  1722→      return normalize([pSC6, pSC4, pSC5]);
  1723→  1723→    }
  1724→  1724→
  1725→  1725→    // ── N_B_DOWN → [N_B_D_VOL_C, N_B_D_VOL_W] ────────────────────────────
  1726→  1726→    case 'N_B_DOWN': {
  1727→  1727→      let volConfirm: number;
  1728→  1728→      if (input.hasVolume) {
  1729→  1729→        volConfirm = clamp((100 - input.mfi) / 60, 0, 1) * 0.6 + 0.2;
  1730→  1730→      } else {
  1731→  1731→        volConfirm = 0.4;
  1732→  1732→      }
  1733→  1733→      const volWeak = 1 - volConfirm;
  1734→  1734→      return normalize([volConfirm, volWeak]);
  1735→  1735→    }
  1736→  1736→
  1737→  1737→    // ── N_B_D_VOL_C → [SC3, SC2, SC1] ──────────────────────────────────────
  1738→  1738→    // Confirmed bearish breakout — all bearish, no range
  1739→  1739→    case 'N_B_D_VOL_C': {
  1740→  1740→      const pSC3 = 0.45 + input.confidenceIndex * 0.15;
  1741→  1741→      const pSC2 = 0.2 + input.strengthIndex * 0.15;
  1742→  1742→      const pSC1 = 0.15 + input.strengthIndex * 0.1;
  1743→  1743→      return normalize([pSC3, pSC2, pSC1]);
  1744→  1744→    }
  1745→  1745→
  1746→  1746→    // ── N_B_D_VOL_W → [SC4, SC6, SC5] ──────────────────────────────────────
  1747→  1747→    // Weak volume bearish breakout — fake breakout risk, capped SC5
  1748→  1748→    case 'N_B_D_VOL_W': {
  1749→  1749→      const pSC4 = 0.4 + (1 - input.bullConsensus) * 0.15;
  1750→  1750→      const pSC6 = 0.3 + input.bullConsensus * 0.15;
  1751→  1751→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.12, 0.02, 0.12);
  1752→  1752→      return normalize([pSC4, pSC6, pSC5]);
  1753→  1753→    }
  1754→  1754→
  1755→  1755→    // ── N_B_NONE → [SC4, SC5, SC6] ──────────────────────────────────────────
  1756→  1756→    case 'N_B_NONE': {
  1757→  1757→      const pSC5 = 0.45 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1758→  1758→      const pSC6 = 0.2 * input.bullConsensus + 0.08;
  1759→  1759→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.08;
  1760→  1760→      return normalize([pSC4, pSC5, pSC6]);
  1761→  1761→    }
  1762→  1762→
  1763→  1763→    // ── N_REVERSAL → [N_R_BULL, N_R_BEAR, N_R_NONE] ──────────────────────
  1764→  1764→    case 'N_REVERSAL': {
  1765→  1765→      const bullRev = overboughtRisk;
  1766→  1766→      const bearRev = oversoldBounce;
  1767→  1767→      const maxRevSignal = Math.max(overboughtRisk, oversoldBounce);
  1768→  1768→      const noneRev = 1 - maxRevSignal * 0.85;
  1769→  1769→      return normalize([bullRev, bearRev, noneRev]);
  1770→  1770→    }
  1771→  1771→
  1772→  1772→    // ── N_R_BULL → [N_R_B_DIV, N_R_B_CANDLE, N_R_B_SR] ──────────────────
  1773→  1773→    case 'N_R_BULL': {
  1774→  1774→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1775→  1775→      const candleProb =
  1776→  1776→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1777→  1777→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1778→  1778→      return normalize([divProb, candleProb, srProb]);
  1779→  1779→    }
  1780→  1780→
  1781→  1781→    // ── N_R_B_DIV → [SC6, SC8, SC5] ────────────────────────────────────────
  1782→  1782→    // Bullish divergence — explicit small SC5
  1783→  1783→    case 'N_R_B_DIV': {
  1784→  1784→      const pSC6 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1785→  1785→      const pSC8 = 0.2 + input.mlMomentum * 0.15;
  1786→  1786→      const pSC5 = clamp((1 - adxNorm) * 0.1, 0.02, 0.1);
  1787→  1787→      return normalize([pSC6, pSC8, pSC5]);
  1788→  1788→    }
  1789→  1789→
  1790→  1790→    // ── N_R_B_CANDLE → [SC7, SC6, SC5] ─────────────────────────────────────
  1791→  1791→    // Bullish candle pattern — explicit small SC5
  1792→  1792→    case 'N_R_B_CANDLE': {
  1793→  1793→      const pSC7 = 0.5 * input.confidenceIndex + 0.2;
  1794→  1794→      const pSC6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1795→  1795→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.1, 0.02, 0.1);
  1796→  1796→      return normalize([pSC7, pSC6, pSC5]);
  1797→  1797→    }
  1798→  1798→
  1799→  1799→    // ── N_R_B_SR → [SC6, SC5, SC4] ─────────────────────────────────────────
  1800→  1800→    // Support bounce — explicit small SC5
  1801→  1801→    // Edge order: SC6, SC5, SC4 — probabilities computed then reordered
  1802→  1802→    case 'N_R_B_SR': {
  1803→  1803→      const pSC6 = 0.5 + input.srAvgStrength * 0.2;
  1804→  1804→      const pSC4 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1805→  1805→      const pSC5 = clamp((1 - input.srAvgStrength) * 0.1, 0.02, 0.1);
  1806→  1806→      // Return in edge order: [SC6, SC5, SC4]
  1807→  1807→      return normalize([pSC6, pSC5, pSC4]);
  1808→  1808→    }
  1809→  1809→
  1810→  1810→    // ── N_R_BEAR → [N_R_BE_DIV, N_R_BE_CANDLE, N_R_BE_SR] ───────────────
  1811→  1811→    // Mirror of N_R_BULL
  1812→  1812→    case 'N_R_BEAR': {
  1813→  1813→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1814→  1814→      const candleProb =
  1815→  1815→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1816→  1816→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1817→  1817→      return normalize([divProb, candleProb, srProb]);
  1818→  1818→    }
  1819→  1819→
  1820→  1820→    // ── N_R_BE_DIV → [SC4, SC2, SC5] ───────────────────────────────────────
  1821→  1821→    // Bearish divergence — explicit small SC5
  1822→  1822→    case 'N_R_BE_DIV': {
  1823→  1823→      const pSC4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1824→  1824→      const pSC2 = 0.2 + input.mlMomentum * 0.15;
  1825→  1825→      const pSC5 = clamp((1 - adxNorm) * 0.1, 0.02, 0.1);
  1826→  1826→      return normalize([pSC4, pSC2, pSC5]);
  1827→  1827→    }
  1828→  1828→
  1829→  1829→    // ── N_R_BE_CANDLE → [SC3, SC4, SC5] ────────────────────────────────────
  1830→  1830→    // Bearish candle pattern — explicit small SC5
  1831→  1831→    case 'N_R_BE_CANDLE': {
  1832→  1832→      const pSC3 = 0.5 * input.confidenceIndex + 0.2;
  1833→  1833→      const pSC4 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1834→  1834→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.1, 0.02, 0.1);
  1835→  1835→      return normalize([pSC3, pSC4, pSC5]);
  1836→  1836→    }
  1837→  1837→
  1838→  1838→    // ── N_R_BE_SR → [SC4, SC5, SC6] ─────────────────────────────────────────
  1839→  1839→    // Resistance rejection — explicit small SC5
  1840→  1840→    // Edge order: SC4, SC5, SC6 — probabilities computed then reordered
  1841→  1841→    case 'N_R_BE_SR': {
  1842→  1842→      const pSC4 = 0.5 + input.srAvgStrength * 0.2;
  1843→  1843→      const pSC6 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1844→  1844→      const pSC5 = clamp((1 - input.srAvgStrength) * 0.1, 0.02, 0.1);
  1845→  1845→      // Return in edge order: [SC4, SC5, SC6]
  1846→  1846→      return normalize([pSC4, pSC5, pSC6]);
  1847→  1847→    }
  1848→  1848→
  1849→  1849→    // ── N_R_NONE → [SC4, SC5, SC6] ──────────────────────────────────────────
  1850→  1850→    case 'N_R_NONE': {
  1851→  1851→      const pSC5 = 0.4 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1852→  1852→      const pSC6 = 0.2 * input.bullConsensus + 0.1;
  1853→  1853→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.1;
  1854→  1854→      return normalize([pSC4, pSC5, pSC6]);
  1855→  1855→    }
  1856→  1856→
  1857→  1857→    // ── Terminal nodes (SC1–SC9) have no outgoing edges ─────────────────────
  1858→  1858→    default:
  1859→  1859→      return [];
  1860→  1860→  }
  1861→  1861→}
  1862→  1862→
  1863→  1863→// === Graph Traversal ==========================================================
  1864→  1864→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1865→  1865→// =============================================================================
  1866→  1866→
  1867→  1867→/**
  1868→  1868→ * Recursively traverses the DAG from a given node, accumulating path
  1869→  1869→ * probabilities at terminal nodes (SC1–SC9).
  1870→  1870→ *
  1871→  1871→ * ### Algorithm
  1872→  1872→ *
  1873→  1873→ * 1. Add `pathProb` to the node's accumulator (`nodeAccum`).
  1874→  1874→ * 2. Compute outgoing edge probabilities via {@link computeNodeProbabilities}.
  1875→  1875→ * 3. For each child edge:
  1876→  1876→ *    a. Store the edge probability in `edgeProbabilities`.
  1877→  1877→ *    b. Determine the branch context (trend/breakout/reversal) if at ROOT.
  1878→  1878→ *    c. If the child is a terminal (SC*), accumulate `childPathProb` into
  1879→  1879→ *       `scenarios` and `contributions` (per-branch breakdown).
  1880→  1880→ *    d. Recurse into the child node.
  1881→  1881→ *
  1882→  1882→ * Because the graph is a DAG (no cycles), recursion terminates at the
  1883→  1883→ * terminal layer. Multiple paths can reach the same terminal, so
  1884→  1884→ * probabilities are **summed** across all contributing paths.
  1885→  1885→ *
  1886→  1886→ * @param nodeId           - Current node being visited
  1887→  1887→ * @param pathProb         - Probability of reaching this node from ROOT (product of all edge probs)
  1888→  1888→ * @param currentBranch    - Which strategy branch we're under (null at ROOT)
  1889→  1889→ * @param ctx              - Pre-computed probability context
  1890→  1890→ * @param adj              - Adjacency map: nodeId → array of edge indices
  1891→  1891→ * @param edges            - All edges in the graph
  1892→  1892→ * @param edgeProbabilities - Mutable: stores computed probability per edge index
  1893→  1893→ * @param nodeAccum        - Mutable: accumulates total path probability at each node
  1894→  1894→ * @param scenarios        - Mutable: accumulates total probability at each SC terminal
  1895→  1895→ * @param contributions    - Mutable: per-branch probability contribution to each SC terminal
  1896→  1896→ */
  1897→  1897→function traverseGraph(
  1898→  1898→  nodeId: string,
  1899→  1899→  pathProb: number,
  1900→  1900→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1901→  1901→  ctx: ProbContext,
  1902→  1902→  adj: Map<string, number[]>,
  1903→  1903→  edges: GraphEdge[],
  1904→  1904→  edgeProbabilities: Record<number, number>,
  1905→  1905→  nodeAccum: Record<string, number>,
  1906→  1906→  scenarios: Record<string, number>,
  1907→  1907→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1908→  1908→): void {
  1909→  1909→  // Accumulate path probability at this node
  1910→  1910→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1911→  1911→
  1912→  1912→  const outEdgeIndices = adj.get(nodeId);
  1913→  1913→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1914→  1914→
  1915→  1915→  // Compute outgoing edge probabilities for this node
  1916→  1916→  const probs = computeNodeProbabilities(
  1917→  1917→    nodeId,
  1918→  1918→    ctx,
  1919→  1919→    outEdgeIndices.length
  1920→  1920→  );
  1921→  1921→
  1922→  1922→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1923→  1923→    const edgeIdx = outEdgeIndices[i];
  1924→  1924→    const edge = edges[edgeIdx];
  1925→  1925→    const prob = probs[i] || 0;
  1926→  1926→
  1927→  1927→    // Store edge probability
  1928→  1928→    edgeProbabilities[edgeIdx] = prob;
  1929→  1929→
  1930→  1930→    // Determine branch for child nodes
  1931→  1931→    let childBranch = currentBranch;
  1932→  1932→    if (nodeId === 'ROOT') {
  1933→  1933→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1934→  1934→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
