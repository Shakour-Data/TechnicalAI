    17→    17→function sma(closes: number[], period: number): number {
    18→    18→  if (closes.length < period) return 0;
    19→    19→  const slice = closes.slice(-period);
    20→    20→  return slice.reduce((a, b) => a + b, 0) / period;
    21→    21→}
    22→    22→
    23→    23→function emaCalc(closes: number[], period: number): number {
    24→    24→  if (closes.length < period) return 0;
    25→    25→  const k = 2 / (period + 1);
    26→    26→  let v = sma(closes.slice(0, period), period);
    27→    27→  for (let i = period; i < closes.length; i++) {
    28→    28→    v = closes[i] * k + v * (1 - k);
    29→    29→  }
    30→    30→  return v;
    31→    31→}

    33→    33→function calcRSI(closes: number[], period = 14): number {
    34→    34→  if (closes.length < period + 1) return 50;
    35→    35→  let avgGain = 0, avgLoss = 0;
    36→    36→  for (let i = 1; i <= period; i++) {
    37→    37→    const diff = closes[i] - closes[i - 1];
    38→    38→    if (diff > 0) avgGain += diff; else avgLoss += Math.abs(diff);
    39→    39→  }
    40→    40→  avgGain /= period; avgLoss /= period;
    41→    41→  for (let i = period + 1; i < closes.length; i++) {
    42→    42→    const diff = closes[i] - closes[i - 1];
    43→    43→    avgGain = (avgGain * (period - 1) + (diff > 0 ? diff : 0)) / period;
    44→    44→    avgLoss = (avgLoss * (period - 1) + (diff < 0 ? Math.abs(diff) : 0)) / period;
    45→    45→  }
    46→    46→  if (avgLoss === 0) return 100;
    47→    47→  return 100 - 100 / (1 + avgGain / avgLoss);
    48→    48→}

    50→    50→function calcCCI(data: OHLCV[], period = 20): number {
    51→    51→  if (data.length < period) return 0;
    52→    52→  const tps: number[] = [];
    53→    53→  for (let i = data.length - period; i < data.length; i++) {
    54→    54→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
    55→    55→  }
    56→    56→  const mean = tps.reduce((a, b) => a + b, 0) / tps.length;
    57→    57→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - mean), 0) / tps.length;
    58→    58→  return meanDev === 0 ? 0 : (tps[tps.length - 1] - mean) / (0.015 * meanDev);
    59→    59→}

    61→    61→function calcMFI(data: OHLCV[], period = 14): number {
    62→    62→  if (data.length < period + 1) return 50;
    63→    63→  let posFlow = 0, negFlow = 0;
    64→    64→  for (let i = data.length - period; i < data.length; i++) {
    65→    65→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
    66→    66→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
    67→    67→    const mf = tp * data[i].volume;
    68→    68→    if (tp > prevTp) posFlow += mf; else negFlow += mf;
    69→    69→  }
    70→    70→  return negFlow === 0 ? 100 : 100 - 100 / (1 + posFlow / negFlow);
    71→    71→}

    73→    73→function calcStochastic(data: OHLCV[], kPeriod = 14, smoothK = 3, smoothD = 3): { k: number; d: number } {
    74→    74→  const rawKs: number[] = [];
    75→    75→  for (let i = kPeriod - 1; i < data.length; i++) {
    76→    76→    let lowest = Infinity, highest = -Infinity;
    77→    77→    for (let j = i - kPeriod + 1; j <= i; j++) {
    78→    78→      if (data[j].low < lowest) lowest = data[j].low;
    79→    79→      if (data[j].high > highest) highest = data[j].high;
    80→    80→    }
    81→    81→    const range = highest - lowest;
    82→    82→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
    83→    83→  }
    84→    84→  const smoothedK: number[] = [];
    85→    85→  for (let i = smoothK - 1; i < rawKs.length; i++) {
    86→    86→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
    87→    87→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
    88→    88→  }
    89→    89→  const k = smoothedK[smoothedK.length - 1] ?? 50;
    90→    90→  const dSlice = smoothedK.slice(-smoothD);
    91→    91→  const d = dSlice.length > 0 ? dSlice.reduce((a, b) => a + b, 0) / dSlice.length : 50;
    92→    92→  return { k, d };
    93→    93→}

    95→    95→function calcMACD(closes: number[], fast = 12, slow = 26, sig = 9): { line: number; signal: number; histogram: number } {
    96→    96→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
    97→    97→  const kFast = 2 / (fast + 1), kSlow = 2 / (slow + 1), kSig = 2 / (sig + 1);
    98→    98→  let eFast = sma(closes.slice(0, fast), fast);
    99→    99→  let eSlow = sma(closes.slice(0, slow), slow);
   100→   100→  const macdVals: number[] = [];
   101→   101→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   102→   102→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   103→   103→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   104→   104→    macdVals.push(eFast - eSlow);
   105→   105→  }
   106→   106→  if (macdVals.length < sig) return { line: 0, signal: 0, histogram: 0 };
   107→   107→  let s = sma(macdVals.slice(0, sig), sig);
   108→   108→  for (let i = sig; i < macdVals.length; i++) {
   109→   109→    s = macdVals[i] * kSig + s * (1 - kSig);
   110→   110→  }
   111→   111→  const line = macdVals[macdVals.length - 1];
   112→   112→  return { line, signal: s, histogram: line - s };
   113→   113→}

   115→   115→function calcBollingerBands(closes: number[], period = 20, mult = 2): { upper: number; middle: number; lower: number } {
   116→   116→  if (closes.length < period) return { upper: 0, middle: 0, lower: 0 };
   117→   117→  const slice = closes.slice(-period);
   118→   118→  const mid = slice.reduce((a, b) => a + b, 0) / period;
   119→   119→  const variance = slice.reduce((a, b) => a + (b - mid) ** 2, 0) / period;
   120→   120→  const sd = Math.sqrt(variance);
   121→   121→  return { upper: mid + mult * sd, middle: mid, lower: mid - mult * sd };
   122→   122→}

   124→   124→function calcADX(data: OHLCV[], period = 14): { adx: number; diPlus: number; diMinus: number } {
   125→   125→  if (data.length < period * 2) return { adx: 0, diPlus: 0, diMinus: 0 };
   126→   126→  let sumPDM = 0, sumMDM = 0, sumTR = 0;
   127→   127→  for (let i = data.length - period; i < data.length; i++) {
   128→   128→    const upMove = data[i].high - data[i - 1].high;
   129→   129→    const downMove = data[i - 1].low - data[i].low;
   130→   130→    sumPDM += upMove > downMove && upMove > 0 ? upMove : 0;
   131→   131→    sumMDM += downMove > upMove && downMove > 0 ? downMove : 0;
   132→   132→    sumTR += Math.max(
   133→   133→      data[i].high - data[i].low,
   134→   134→      Math.abs(data[i].high - data[i - 1].close),
   135→   135→      Math.abs(data[i].low - data[i - 1].close)
   136→   136→    );
   137→   137→  }
   138→   138→  const diPlus = sumTR > 0 ? (sumPDM / sumTR) * 100 : 0;
   139→   139→  const diMinus = sumTR > 0 ? (sumMDM / sumTR) * 100 : 0;
   140→   140→  const dx = (diPlus + diMinus) > 0 ? Math.abs(diPlus - diMinus) / (diPlus + diMinus) * 100 : 0;
   141→   141→  return { adx: dx, diPlus, diMinus };
   142→   142→}

   144→   144→function calcATR(data: OHLCV[], period = 14): number {
   145→   145→  if (data.length < period + 1) return 0;
   146→   146→  let sum = 0;
   147→   147→  for (let i = data.length - period; i < data.length; i++) {
   148→   148→    sum += Math.max(
   149→   149→      data[i].high - data[i].low,
   150→   150→      Math.abs(data[i].high - data[i - 1].close),
   151→   151→      Math.abs(data[i].low - data[i - 1].close)
   152→   152→    );
   153→   153→  }
   154→   154→  return sum / period;
   155→   155→}

   163→   163→// ─── 16-Feature Extraction for VDss ─────────────────────────────────────────

   165→   165→const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));
   166→   166→
   167→   167→/**
   168→   168→ * Extract the 16 VDSS features at a given point in the data.
   169→   169→ * Feature order must match VDSS_FEATURE_NAMES exactly.
   170→   170→ *
   171→   171→ * @param data - Full OHLCV history (chronological)
   172→   172→ * @param endIdx - Index of the candle to compute features for
   173→   173→ * @param hasVolume - Whether data has valid volume
   174→   174→ * @returns Array of 16 feature values in [0,1] range
   175→   175→ */
   176→   176→export function extractVDSSFeatures(data: OHLCV[], endIdx: number, hasVolume: boolean): number[] {
   177→   177→  const slice = data.slice(0, endIdx + 1);
   178→   178→  const closes = slice.map(d => d.close);
   179→   179→  const price = closes[closes.length - 1];
   180→   180→  if (price <= 0) return new Array(16).fill(0.5);
   181→   181→
   182→   182→  // ── Compute indicators ──
   183→   183→  const rsi = calcRSI(closes);
   184→   184→  const mfi = hasVolume ? calcMFI(slice) : 50;
   185→   185→  const cci = calcCCI(slice);
   186→   186→  const adxResult = calcADX(slice);
   187→   187→  const macd = calcMACD(closes);
   188→   188→  const stoch = calcStochastic(slice);
   189→   189→  const bb = calcBollingerBands(closes);
   190→   190→  const atr = calcATR(slice);
   191→   191→  const ma21 = sma(closes, 21);
   192→   192→  const ma100 = sma(closes, 100);
   193→   193→  const ema12 = emaCalc(closes, 12);
   194→   194→  const ema26 = emaCalc(closes, 26);
   195→   195→
   196→   196→  // ── LAYER 1: Raw scores (s_*) ──
   197→   197→  const s_rsi = rsi > 80 ? 0.95 : rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   198→   198→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   199→   199→  const s_mfi = mfi > 80 ? 0.95 : mfi > 70 ? 0.85 : mfi > 60 ? 0.7 : mfi > 50 ? 0.55
   200→   200→    : mfi > 40 ? 0.45 : mfi > 30 ? 0.3 : mfi > 20 ? 0.15 : 0.05;
   201→   201→  const s_cci = cci > 200 ? 0.9 : cci > 100 ? 0.75 : cci > 0 ? 0.6
   202→   202→    : cci > -100 ? 0.4 : cci > -200 ? 0.25 : 0.1;
   203→   203→  const s_adx = adxResult.diPlus > adxResult.diMinus
   204→   204→    ? clamp(0.5 + (adxResult.diPlus - adxResult.diMinus) / 100, 0.5, 1)
   205→   205→    : clamp(0.5 - (adxResult.diMinus - adxResult.diPlus) / 100, 0, 0.5);
   206→   206→  const s_macd = macd.histogram > 0 && macd.line > macd.signal ? 0.9
   207→   207→    : macd.histogram > 0 ? 0.7 : macd.line > macd.signal ? 0.55
   208→   208→    : macd.histogram < 0 && macd.line < macd.signal ? 0.1 : 0.3;
   209→   209→  const s_stoch = clamp(
   210→   210→    (stoch.k > 80 ? 0.9 : stoch.k > 70 ? 0.8 : stoch.k > 50 ? 0.6
   211→   211→      : stoch.k > 30 ? 0.4 : stoch.k > 20 ? 0.2 : 0.1)
   212→   212→    + (stoch.k > stoch.d ? 0.1 : -0.1), 0, 1);
   213→   213→  const bbRange = bb.upper - bb.lower;
   214→   214→  const s_bb = bbRange === 0 ? 0.5 : clamp((price - bb.lower) / bbRange, 0, 1);
   215→   215→  const s_ma21 = clamp(price > ma21 ? 0.7 + ((price - ma21) / ma21) * 0.5 : 0.3 + ((price - ma21) / ma21) * 0.5, 0, 1);
   216→   216→  const s_ma100 = clamp(price > ma100 ? 0.65 + ((price - ma100) / ma100) * 0.3 : 0.35 + ((price - ma100) / ma100) * 0.3, 0, 1);
   217→   217→  const emaGap = ema12 > 0 && ema26 > 0 ? (ema12 - ema26) / ema26 : 0;
   218→   218→  const s_ema = clamp(0.5 + emaGap * 5, 0, 1);
   219→   219→  const s_atr = clamp(1 - atr / price * 10, 0.2, 1);
   220→   220→
   221→   221→  // Trend (R² + angle)
   222→   222→  const trendPeriod = Math.min(21, closes.length);
   223→   223→  const trendSlice = closes.slice(-trendPeriod);
   224→   224→  const n = trendSlice.length;
   225→   225→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   226→   226→  for (let i = 0; i < n; i++) { sumX += i; sumY += trendSlice[i]; sumXY += i * trendSlice[i]; sumX2 += i * i; sumY2 += trendSlice[i] ** 2; }
   227→   227→  const denom = n * sumX2 - sumX * sumX;
   228→   228→  let r2 = 0, angle = 0;
   229→   229→  if (denom !== 0) {
   230→   230→    const slope = (n * sumXY - sumX * sumY) / denom;
   231→   231→    const meanY = sumY / n;
   232→   232→    let ssRes = 0, ssTot = 0;
   233→   233→    for (let i = 0; i < n; i++) {
   234→   234→      ssRes += (trendSlice[i] - (slope * i + meanY - slope * (n - 1) / 2)) ** 2;
   235→   235→      ssTot += (trendSlice[i] - meanY) ** 2;
   236→   236→    }
   237→   237→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
   238→   238→    angle = meanY > 0 ? Math.atan(slope / meanY) * (180 / Math.PI) : 0;
   239→   239→  }
   240→   240→  const tStr = r2 * Math.abs(angle) / 45;
   241→   241→  const direction = denom !== 0 ? (n * sumXY - sumX * sumY) / denom : 0;
   242→   242→  const s_trend = direction > 0 ? clamp(0.5 + tStr / 2, 0.5, 1)
   243→   243→    : direction < 0 ? clamp(0.5 - tStr / 2, 0, 0.5) : 0.5;
   244→   244→
   245→   245→  // Simplified S/R: distance to nearest local high/low
   246→   246→  let s_sr = 0.5;
   247→   247→  if (slice.length > 20) {
   248→   248→    const lookback = slice.slice(-30);
   249→   249→    let nearestHigh = Infinity, nearestLow = 0;
   250→   250→    for (const d of lookback) {
   251→   251→      if (d.high > price && d.high < nearestHigh) nearestHigh = d.high;
   252→   252→      if (d.low < price && d.low > nearestLow) nearestLow = d.low;
   253→   253→    }
   254→   254→    const dR = nearestHigh < Infinity ? Math.abs(price - nearestHigh) / nearestHigh : 1;
   255→   255→    const dS = nearestLow > 0 ? Math.abs(price - nearestLow) / nearestLow : 1;
   256→   256→    s_sr = clamp(0.5 - dR * 1.5 + dS * 1.0, 0, 1);
   257→   257→  }
   258→   258→
   259→   259→  // ── LAYER 2: Momentum correction ──
   260→   260→  const TB = 5;
   261→   261→  let f_rsi = s_rsi, f_mfi = s_mfi, f_cci = s_cci, f_macd = s_macd, f_stoch = s_stoch;
   262→   262→  if (slice.length > TB + 15) {
   263→   263→    const pastSlice = slice.slice(0, -TB);
   264→   264→    const pastCloses = pastSlice.map(d => d.close);
   265→   265→    const rsiPast = calcRSI(pastCloses);
   266→   266→    const mfiPast = hasVolume ? calcMFI(pastSlice) : 50;
   267→   267→    const cciPast = calcCCI(pastSlice);
   268→   268→    const macdPast = calcMACD(pastCloses);
   269→   269→    const stochPast = calcStochastic(pastSlice);
   270→   270→
   271→   271→    f_rsi = clamp(s_rsi + (rsi - rsiPast) / 100 * 0.08, 0, 1);
   272→   272→    f_mfi = clamp(s_mfi + (mfi - mfiPast) / 100 * 0.08, 0, 1);
   273→   273→    f_cci = clamp(s_cci + (cci - cciPast) / 300 * 0.06, 0, 1);
   274→   274→    f_macd = clamp(s_macd + (macd.histogram - macdPast.histogram) / (Math.abs(macd.histogram) + Math.abs(macdPast.histogram) + 1) * 3 * 0.08, 0, 1);
   275→   275→    f_stoch = clamp(s_stoch + (stoch.k - stochPast.k) / 50 * 0.06, 0, 1);
   276→   276→  }
   277→   277→
   278→   278→  // ── LAYER 2: Cross signals (binary per spec) ──
   279→   279→  let f_stochCross = 0.1;
   280→   280→  let f_macdCross = 0.1;
   281→   281→  let f_div = 0.15;
   282→   282→
   283→   283→  if (slice.length > 20) {
   284→   284→    const prevSlice = slice.slice(0, -1);
   285→   285→    const prevStoch = calcStochastic(prevSlice);
   286→   286→    const prevMacd = calcMACD(prevSlice.map(d => d.close));
   287→   287→
   288→   288→    // Stochastic bullish cross: K_prev < D_prev AND K_curr > D_curr
   289→   289→    f_stochCross = (prevStoch.k < prevStoch.d && stoch.k > stoch.d) ? 0.9 : 0.1;
   290→   290→
   291→   291→    // MACD bullish cross: line_prev < signal_prev AND line_curr > signal_curr
   292→   292→    f_macdCross = (prevMacd.line < prevMacd.signal && macd.line > macd.signal) ? 0.9 : 0.1;
   293→   293→
   294→   294→    // Divergence detection
   295→   295→    if (closes.length > 40) {
   296→   296→      const p20 = closes[closes.length - 21];
   297→   297→      const rsi20 = calcRSI(closes.slice(0, -20));
   298→   298→      const pChange = price - p20;
   299→   299→      const rsiChange = rsi - rsi20;
   300→   300→      if (pChange !== 0 && rsiChange !== 0) {
   301→   301→        f_div = (pChange > 0 && rsiChange < 0) ? 0.85 : (pChange < 0 && rsiChange > 0) ? 0.85 : 0.15;
   302→   302→      }
   303→   303→    }
   304→   304→  }
   305→   305→
   306→   306→  // ── Return 16 features in exact order ──
   307→   307→  return [
   308→   308→    f_rsi, f_mfi, f_cci, s_adx,
   309→   309→    f_macd, f_stoch, s_bb, s_ma21, s_ma100, s_ema,
   310→   310→    s_atr, s_trend, s_sr,
   311→   311→    f_stochCross, f_macdCross, f_div,
   312→   312→  ];
   313→   313→}

   335→   335→// ─── Main Training Function ─────────────────────────────────────────────────
   336→   336→
   337→   337→/**
   338→   338→ * Train the Adaptive Weight Model for a given symbol.
   339→   339→ * Extracts 16 VDSS features for each historical candle, creates binary labels,
   340→   340→ * trains Logistic Regression with TimeSeriesSplit validation.
   341→   341→ *
   342→   342→ * @param data - Full OHLCV history (chronological, oldest first)
   343→   343→ * @param symbol - Symbol identifier for caching
   344→   344→ * @param minSamples - Minimum samples needed (default 70)
   345→   345→ * @returns AdaptiveModelResult or null
   346→   346→ */
   347→   347→export function trainAdaptiveModel(data: OHLCV[], symbol: string, minSamples = 70): AdaptiveModelResult | null {
   348→   348→  const FORWARD_DAYS = 5;
   349→   349→  const START_IDX = 40; // Need enough history for all indicators to stabilize
   350→   350→
   351→   351→  if (data.length < START_IDX + FORWARD_DAYS + minSamples) return null;
   352→   352→
   353→   353→  const hasVolume = data.some(d => d.volume > 0);
   354→   354→
   355→   355→  const X: number[][] = [];
   356→   356→  const y: number[] = [];
   357→   357→
   358→   358→  for (let i = START_IDX; i < data.length - FORWARD_DAYS; i++) {
   359→   359→    const features = extractVDSSFeatures(data, i, hasVolume);
   360→   360→    const currentPrice = data[i].close;
   361→   361→    const futurePrice = data[i + FORWARD_DAYS].close;
   362→   362→
   363→   363→    if (currentPrice <= 0) continue;
   364→   364→    if (features.some(f => !isFinite(f))) continue;
   365→   365→
   366→   366→    // Label: 1 if 5-day forward return > 1%, else 0
   367→   367→    const forwardReturn = (futurePrice - currentPrice) / currentPrice;
   368→   368→    const label = forwardReturn > 0.01 ? 1 : 0;
   369→   369→
   370→   370→    X.push(features);
   371→   371→    y.push(label);
   372→   372→  }
   373→   373→
   374→   374→  if (X.length < minSamples) return null;
   375→   375→
   376→   376→  const model = new AdaptiveWeightModel(minSamples);
   377→   377→  const success = model.train(X, y);
   378→   378→
   379→   379→  if (!success) return null;
   380→   380→
   381→   381→  // Predict current features
   382→   382→  const currentFeatures = extractVDSSFeatures(data, data.length - 1, hasVolume);
   383→   383→  model.predictScore(currentFeatures);
   384→   384→
   385→   385→  const result = model.getResult();
   386→   386→  setCachedAdaptiveModel(symbol, result);
   387→   387→
   388→   388→  return result;
   389→   389→}

   391→   391→// ─── Calculate bullConsensus with ML Weights (Layer 3) ──────────────────────
   392→   392→
   393→   393→/**
   394→   394→ * Calculate bullConsensus using ML weights.
   395→   395→ * Falls back to default weights if ML model is not trained.
   396→   396→ *
   397→   397→ * @param features - 16 VDSS features for the current candle
   398→   398→ * @param mlResult - Adaptive model result (or null for fallback)
   399→   399→ * @param hasVolume - Whether MFI feature is valid
   400→   400→ */
   401→   401→export function calculateBullConsensus(
   402→   402→  features: number[],
   403→   403→  mlResult: AdaptiveModelResult | null,
   404→   404→  hasVolume: boolean,
   405→   405→): { bullConsensus: number; usedML: boolean } {
   406→   406→  if (mlResult && mlResult.isTrained && mlResult.weights.length === 16) {
   407→   407→    const weights = mlResult.weights;
   408→   408→    let bullConsensus = 0;
   409→   409→    let totalWeight = 0;
   410→   410→
   411→   411→    for (let j = 0; j < 16; j++) {
   412→   412→      // Skip MFI (index 1) if no volume
   413→   413→      if (j === 1 && !hasVolume) continue;
   414→   414→      bullConsensus += weights[j] * features[j];
   415→   415→      totalWeight += weights[j];
   416→   416→    }
   417→   417→
   418→   418→    if (totalWeight > 0) bullConsensus /= totalWeight;
   419→   419→
   420→   420→    // Blend with ML direct prediction (dynamic weight based on accuracy)
   421→   421→    if (mlResult.predictionProb !== null) {
   422→   422→      const mlWeight = Math.max(0, Math.min(mlResult.recentAccuracy * 0.30, 0.30));
   423→   423→      bullConsensus = bullConsensus * (1 - mlWeight) + mlResult.predictionProb * mlWeight;
   424→   424→    }
   425→   425→
   426→   426→    return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: true };
   427→   427→  }
   428→   428→
   429→   429→  // Fallback: default weights (per spec, only when ML is unavailable)
   430→   430→  const DEFAULT_WEIGHTS = [
   431→   431→    0.15, 0.12, 0.10, 0.08,  // rsi, mfi, cci, adx
   432→   432→    0.14, 0.10, 0.08,         // macd, stoch, bb
   433→   433→    0.10, 0.08, 0.10,         // ma21, ma100, ema
   434→   434→    0.05, 0.10, 0.05,         // atr, trend, sr
   435→   435→    0.06, 0.06, 0.08,         // stochCross, macdCross, div
   436→   436→  ];
   437→   437→
   438→   438→  let bullConsensus = 0;
   439→   439→  let totalWeight = 0;
   440→   440→  for (let j = 0; j < 16; j++) {
   441→   441→    if (j === 1 && !hasVolume) continue;
   442→   442→    bullConsensus += DEFAULT_WEIGHTS[j] * features[j];
   443→   443→    totalWeight += DEFAULT_WEIGHTS[j];
   444→   444→  }
   445→   445→  if (totalWeight > 0) bullConsensus /= totalWeight;
   446→   446→
   447→   447→  return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: false };
   448→   448→}

   450→   450→// ─── Scenario Probabilities with Adaptive Parameters (Layer 4) ─────────────
   451→   451→
   452→   452→/**
   453→   453→ * Calculate 5-scenario probabilities with ML-adaptive parameters.
   454→   454→ */
   455→   455→export function calculateScenarioProbabilities(
   456→   456→  bullConsensus: number,
   457→   457→  price: number,
   458→   458→  R1: number,
   459→   459→  S1: number,
   460→   460→  MA100: number,
   461→   461→  rsi: number,
   462→   462→  mfi: number,
   463→   463→  stochK: number,
   464→   464→  hasVolume: boolean,
   465→   465→  mlResult: AdaptiveModelResult | null,
   466→   466→): { pR1: number; pR2: number; pR3: number; pR4: number; pR5: number; factors: { momentum: number; volatility: number; trend: number } } {
   467→   467→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0.5;
   468→   468→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0.5;
   469→   469→  const overboughtRisk = (rsi > 70 || (hasVolume && mfi > 80) || stochK > 80) ? 1 : 0;
   470→   470→  const oversoldBounce = (rsi < 30 || (hasVolume && mfi < 20) || stochK < 20) ? 1 : 0;
   471→   471→
   472→   472→  // Adaptive parameters from ML coefficients
   473→   473→  const factors = mlResult?.adaptiveParams ?? { momentum: 0.7, volatility: 0.5, trend: 0.6 };
   474→   474→
   475→   475→  // Raw scenario calculations with adaptive parameters
   476→   476→  const raw_R1 = (
   477→   477→    bullConsensus * distR1 * (1 - overboughtRisk * factors.momentum * 0.7) +
   478→   478→    oversoldBounce * 0.3 * distR1 * factors.momentum
   479→   479→  );
   480→   480→
   481→   481→  const raw_R2 = (
   482→   482→    (1 - Math.abs(bullConsensus - 0.65)) * 0.8 +
   483→   483→    (bullConsensus > 0.4 ? 0.2 : 0)
   484→   484→  ) * factors.trend;
   485→   485→
   486→   486→  const raw_R3 = (1 - Math.abs(bullConsensus - 0.5)) * 0.6 * (1 + (1 - factors.volatility) * 0.2);
   487→   487→
   488→   488→  const raw_R4 = (
   489→   489→    (1 - bullConsensus) * distS1 * 0.7 +
   490→   490→    overboughtRisk * 0.5 * factors.momentum
   491→   491→  ) * (0.8 + 0.2 * (1 - factors.trend));
   492→   492→
   493→   493→  const raw_R5 = (
   494→   494→    (1 - bullConsensus) ** 2 * 0.5 +
   495→   495→    (price < MA100 ? 0.2 : 0)
   496→   496→  ) * (1 + (1 - factors.trend) * 0.3);
   497→   497→
   498→   498→  // Normalize to sum = 100
   499→   499→  const rawSum = raw_R1 + raw_R2 + raw_R3 + raw_R4 + raw_R5;
   500→   500→  if (rawSum <= 0) {
   501→   501→    return { pR1: 20, pR2: 20, pR3: 20, pR4: 20, pR5: 20, factors };
   502→   502→  }
   503→   503→
   504→   504→  const pR1 = Math.round(raw_R1 / rawSum * 100);
   505→   505→  const pR2 = Math.round(raw_R2 / rawSum * 100);
   506→   506→  const pR3 = Math.round(raw_R3 / rawSum * 100);
   507→   507→  const pR4 = Math.round(raw_R4 / rawSum * 100);
   508→   508→  const pR5 = 100 - (pR1 + pR2 + pR3 + pR4); // remainder ensures sum=100
   509→   509→
   510→   510→  return { pR1, pR2, pR3, pR4, pR5, factors };
   511→   511→}

   513→   513→// ─── Edge Weights with ML-Driven Parameters (Layer 5) ─────────────────────
   514→   514→
   515→   515→/**
   516→   516→ * Calculate graph edge weights with ML-driven adaptive parameters.
   517→   517→ */
   518→   518→export function calculateEdgeWeights(
   519→   519→  bullConsensus: number,
   520→   520→  adx: number,
   521→   521→  mlResult: AdaptiveModelResult | null,
   522→   522→): { up: number; down: number; pullback: number; risk: number } {
   523→   523→  // Default coefficients
   524→   524→  let trendCoef = 1 / 3;
   525→   525→  let momentumCoef = 1 / 3;
   526→   526→  let volatilityCoef = 1 / 3;
   527→   527→
   528→   528→  if (mlResult?.isTrained && mlResult.coefficients.length === 16) {
   529→   529→    const ec = mlResult.coefficients;
   530→   530→    const absTrend = Math.abs(ec[11] ?? 0);
   531→   531→    const absMomentum = Math.abs(ec[0] ?? 0);
   532→   532→    const absVolatility = Math.abs(ec[10] ?? 0);
   533→   533→    const total = absTrend + absMomentum + absVolatility;
   534→   534→    if (total > 0) {
   535→   535→      trendCoef = absTrend / total;
   536→   536→      momentumCoef = absMomentum / total;
   537→   537→      volatilityCoef = absVolatility / total;
   538→   538→    }
   539→   539→  }
   540→   540→
   541→   541→  // Dynamic base calculations
   542→   542→  const up_base = bullConsensus * 0.7 + 0.15;
   543→   543→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
   544→   544→  const pullback_base = 0.25;
   545→   545→  const risk_base = 0.12 * (1.2 - Math.min(adx / 100, 1));
   546→   546→
   547→   547→  // Apply ML-driven adjustments
   548→   548→  const up = clamp(up_base * (1 + (momentumCoef - 1 / 3) * 0.3), 0.05, 0.95);
   549→   549→  const down = clamp(down_base * (1 + (trendCoef - 1 / 3) * 0.3), 0.05, 0.95);
   550→   550→  const pullback = clamp(pullback_base * (1 + (volatilityCoef - 1 / 3) * 0.2), 0.05, 0.50);
   551→   551→  const risk = clamp(risk_base * (1 - (trendCoef - 1 / 3) * 0.5), 0.02, 0.30);
   552→   552→
   553→   553→  return { up, down, pullback, risk };
   554→   554→}

   620→   620→export interface PricePrediction {
   621→   621→  predictedPrices: number[];
   622→   622→  predictedReturn5d: number;
   623→   623→  confidence: number;
   624→   624→  trendDirection: 'up' | 'down' | 'flat';
   625→   625→}
   626→   626→
   627→   627→export function predictPrices(data: OHLCV[]): PricePrediction | null {
   628→   628→  if (data.length < 40) return null;
   629→   629→  const PREDICT_DAYS = 5;
   630→   630→  const closes = data.map(d => d.close);
   631→   631→  const price = closes[closes.length - 1];
   632→   632→  if (price <= 0) return null;
   633→   633→
   634→   634→  const LOOKBACK = 20;
   635→   635→  const returns: number[] = [];
   636→   636→  for (let i = closes.length - LOOKBACK; i < closes.length; i++) {
   637→   637→    returns.push(i > 0 && closes[i - 1] > 0 ? (closes[i] - closes[i - 1]) / closes[i - 1] : 0);
   638→   638→  }
   639→   639→
   640→   640→  const mom3 = returns.slice(-3).reduce((a, b) => a + b, 0) / 3;
   641→   641→  const mom10 = returns.slice(-10).reduce((a, b) => a + b, 0) / 10;
   642→   642→  const ma20 = sma(closes, 20);
   643→   643→  const deviation = price > 0 && ma20 > 0 ? (price - ma20) / ma20 : 0;
   644→   644→
   645→   645→  const trendSlice = closes.slice(-20);
   646→   646→  const tn = trendSlice.length;
   647→   647→  let sx = 0, sy = 0, sxy = 0, sx2 = 0;
   648→   648→  for (let i = 0; i < tn; i++) { sx += i; sy += trendSlice[i]; sxy += i * trendSlice[i]; sx2 += i * i; }
   649→   649→  const d = tn * sx2 - sx * sx;
   650→   650→  let slope = 0, r2 = 0;
   651→   651→  if (d !== 0) {
   652→   652→    slope = (tn * sxy - sx * sy) / d;
   653→   653→    const meanY = sy / tn;
   654→   654→    let ssRes = 0, ssTot = 0;
   655→   655→    for (let i = 0; i < tn; i++) { ssRes += (trendSlice[i] - (slope * i + meanY - slope * (tn - 1) / 2)) ** 2; ssTot += (trendSlice[i] - meanY) ** 2; }
   656→   656→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
   657→   657→  }
   658→   658→
   659→   659→  let atr = 0;
   660→   660→  for (let i = Math.max(1, data.length - 14); i < data.length; i++) {
   661→   661→    atr += Math.max(data[i].high - data[i].low, Math.abs(data[i].high - data[i - 1].close), Math.abs(data[i].low - data[i - 1].close));
   662→   662→  }
   663→   663→  atr /= Math.min(14, data.length - 1);
   664→   664→  const dailyVol = price > 0 ? atr / price : 0.02;
   665→   665→
   666→   666→  const trendDaily = price > 0 ? slope / price : 0;
   667→   667→  const meanRevDaily = -deviation * 0.05;
   668→   668→  const momentumDaily = mom3 * 0.4 + mom10 * 0.1;
   669→   669→  const trendW = r2 * 0.6, mrW = Math.min(Math.abs(deviation) * 5, 0.4), momW = 0.2;
   670→   670→  const predDaily = trendDaily * trendW + meanRevDaily * mrW + momentumDaily * momW;
   671→   671→
   672→   672→  const predictedPrices: number[] = [];
   673→   673→  let cp = price;
   674→   674→  for (let dd = 1; dd <= PREDICT_DAYS; dd++) {
   675→   675→    const decay = Math.exp(-0.1 * (dd - 1));
   676→   676→    cp = cp * (1 + predDaily * decay);
   677→   677→    predictedPrices.push(Math.round(cp));
   678→   678→  }
   679→   679→
   680→   680→  const predictedReturn5d = (predictedPrices[PREDICT_DAYS - 1] - price) / price;
   681→   681→  return {
   682→   682→    predictedPrices,
   683→   683→    predictedReturn5d,
   684→   684→    confidence: Math.min(0.9, Math.max(0.1, r2 * 0.7 + 0.2)),
   685→   685→    trendDirection: predictedReturn5d > 0.005 ? 'up' : predictedReturn5d < -0.005 ? 'down' : 'flat',
   686→   686→  };
   687→   687→}

   689→   689→export function evaluatePredictedIndicators(data: OHLCV[], prediction: PricePrediction): { mlScore: number; predictedBullConsensus: number } {
   690→   690→  const price = data[data.length - 1].close;
   691→   691→  if (price <= 0 || prediction.predictedPrices.length === 0) return { mlScore: 0.5, predictedBullConsensus: 0.5 };
   692→   692→  const avgP = prediction.predictedPrices.reduce((a, b) => a + b, 0) / prediction.predictedPrices.length;
   693→   693→  const closes = data.map(d => d.close);
   694→   694→  const ma21 = sma(closes, 21), ma100 = sma(closes, 100), rsi = calcRSI(closes);
   695→   695→  const rsiP = Math.max(0, Math.min(1, (rsi + (avgP > price ? 5 : -5)) / 100));
   696→   696→  const ma21S = ma21 > 0 ? (avgP > ma21 ? 0.7 : 0.3) : 0.5;
   697→   697→  const ma100S = ma100 > 0 ? (avgP > ma100 ? 0.65 : 0.35) : 0.5;
   698→   698→  const dirS = avgP > price ? 0.65 : 0.35;
   699→   699→  return {
   700→   700→    mlScore: Math.max(0, Math.min(1, 0.5 + prediction.predictedReturn5d * 10 * prediction.confidence)),
   701→   701→    predictedBullConsensus: Math.max(0, Math.min(1, rsiP * 0.3 + ma21S * 0.25 + ma100S * 0.2 + dirS * 0.25)),
   702→   702→  };
   703→   703→}
