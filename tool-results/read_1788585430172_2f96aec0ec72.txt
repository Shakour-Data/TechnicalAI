    42→function renderAIText(text: string): string {
    43→  // Remove CJK (Chinese/Japanese/Korean) characters — prohibited in output
    44→  const noCjk = text.replace(/[\u4e00-\u9fff\u3400-\u4dbf\u3040-\u309f\u30a0-\u30ff\uac00-\ud7af\uff00-\uffef]/g, '');
    45→
    46→  // Normalize zero-width BOM and look-alike characters that may interfere with regex
    47→  // but PRESERVE ZWNJ (\u200C) which is essential for Persian compound words
    48→  const normalized = noCjk.replace(/[\u200d\u200b\ufeff]/g, '');
    49→
    50→  // Color syntax: {color:hex}text{/color} or {color:named}text{/color} → <span>
    51→  const colorMap: Record<string, string> = {
    52→    'red': '#dc2626', 'red-600': '#dc2626', 'red-700': '#b91c1c',
    53→    'green': '#16a34a', 'emerald': '#059669', 'emerald-600': '#059669', 'emerald-700': '#047857',
    54→    'amber': '#d97706', 'amber-600': '#d97706', 'amber-700': '#b45309', 'amber-800': '#92400e',
    55→    'blue': '#2563eb', 'blue-600': '#2563eb', 'blue-700': '#1d4ed8',
    56→    'orange': '#ea580c',
    57→    'purple': '#9333ea', 'purple-600': '#9333ea', 'purple-700': '#7e22ce',
    58→  };
    59→
    60→  const processInline = (raw: string) => {
    61→    let html = raw;
    62→    html = html.replace(/\{color:([^}]+)\}([\s\S]*?)\{\/color\}/g, (_match, colorName: string, inner: string) => {
    63→      const hex = colorMap[colorName.trim()] || (colorName.startsWith('#') ? colorName : null);
    64→      if (!hex) return inner; // Unknown color: just return text without color
    65→      return `<span style="color:${hex};font-weight:700">${inner}</span>`;
    66→    });
    67→    html = html.replace(/\*\*(.+?)\*\*/g, '<strong class="font-bold">$1</strong>');
    68→    return html;
    69→  };
    70→
    71→  // Process line by line to detect headings
    72→  const lines = normalized.split('\n');
    73→  const htmlParts: string[] = [];
    74→  let currentParagraph: string[] = [];
    75→
    76→  const flushParagraph = () => {
    77→    if (currentParagraph.length > 0) {
    78→      const content = currentParagraph.join('<br>');
    79→      htmlParts.push(`<p class="mb-3" style="line-height:2.2">${content}</p>`);
    80→      currentParagraph = [];
    81→    }
    82→  };
    83→
    84→  for (const line of lines) {
    85→    const trimmed = line.trim();
    86→
    87→    // ## Main heading
    88→    if (trimmed.startsWith('## ')) {
    89→      flushParagraph();
    90→      htmlParts.push(`<h2 class="mb-2 pb-1" style="font-size:16px;font-weight:800;border-bottom:1px solid #e5e7eb">${processInline(trimmed.slice(3))}</h2>`);
    91→      continue;
    92→    }
    93→    // ### Section heading
    94→    if (trimmed.startsWith('### ')) {
    95→      flushParagraph();
    96→      htmlParts.push(`<h3 class="mt-3 mb-1" style="font-size:14px;font-weight:700">${processInline(trimmed.slice(4))}</h3>`);
    97→      continue;
    98→    }
    99→
   100→    // Empty line = paragraph break
   101→    if (!trimmed) {
   102→      flushParagraph();
   103→      continue;
   104→    }
   105→
   106→    // Regular line — accumulate into current paragraph
   107→    currentParagraph.push(processInline(trimmed));
   108→  }
   109→  flushParagraph();
   110→
   111→  return htmlParts.join('');
   112→}

   326→function generateAnalysisText(ctx: AnalysisContext) {
   327→  const {
   328→    symbolName, currentPrice, ma21, ma100, rsi, mfi, cci, adx,
   329→    stochK, stochD, macdLine, macdSignal, macdHist,
   330→    diPlus, diMinus, sar, atr, obv,
   331→    bollingerUpper, bollingerMiddle, bollingerLower,
   332→    trendDirection, trendAngle, trendR2, overallSignal,
   333→    highestKey, highestProb, scenarios, S1, R1, R2,
   334→    hasVolume, resistanceStrengths, supportStrengths,
   335→    v11Result, currencyUnit: ctxCurrencyUnit,
   336→    instrumentCategory, priceDecimals: ctxPriceDecimals,
   337→  } = ctx;
   338→  const unit = ctxCurrencyUnit || 'ریال';
   339→  // Dynamic price formatter using data-source-derived decimals
   340→  const toFa = (n: number) => formatPriceFa(n, ctxPriceDecimals ?? 0);
   341→  const terms = getInstrumentTerms(instrumentCategory);
   342→
   343→  const R1_info = resistanceStrengths[0];
   344→  const S1_info = supportStrengths[0];
   345→  const R1_grade = R1_info?.grade ? (GRADE_MAP[R1_info.grade]?.label ?? R1_info.grade) : '';
   346→  const S1_grade = S1_info?.grade ? (GRADE_MAP[S1_info.grade]?.label ?? S1_info.grade) : '';
   347→  const R1_methods = R1_info?.methods?.length ? ` با ${toPersianDigits(String(R1_info.methods.length))} روش تأیید شده` : '';
   348→  const S1_methods = S1_info?.methods?.length ? ` با ${toPersianDigits(String(S1_info.methods.length))} روش تأیید شده` : '';
   349→
   350→  const bullCum = v11Result.bullishCumulative;
   351→  const bearCum = v11Result.bearishCumulative;
   352→  const rangeCum = v11Result.neutralCumulative;
   353→
   354→  const dominant = SCENARIO_META[highestKey].label;
   355→
   356→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   357→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   358→  const macdBullish = macdLine > macdSignal;
   359→  const bbRange = bollingerUpper - bollingerLower;
   360→  const bbPos = toPersianDigits(bbRange > 0 ? ((currentPrice - bollingerLower) / bbRange * 100).toFixed(0) : '50');
   361→  const bbSignal = currentPrice > bollingerUpper ? 'بالای باند بالایی (اشباع خرید)'
   362→    : currentPrice < bollingerLower ? 'زیر باند پایینی (اشباع فروش)'
   363→    : `داخل باندها (${bbPos}٪ از بازه)`;
   364→  const diSignal = diPlus > diMinus
   365→    ? `DI+ (${toFa(diPlus)}) بالاتر از DI- (${toFa(diMinus)}) — فشار خرید غالب`
