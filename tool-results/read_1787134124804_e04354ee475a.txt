   772→   772→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   773→   773→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   774→   774→  grade?: string; overlapCount?: number; touchCount?: number;
   775→   775→}) {
   776→   776→  const isSupport = type === 'support';
   777→   777→  const isTargetType = type === 'target';
   778→   778→
   779→   779→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   780→   780→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';

   785→   785→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   786→   786→
   787→   787→  return (
   788→   788→    <div
   789→   789→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   790→   790→      style={{

   803→   803→              {toPersianDigits(overlapCount)} هم‌پوشانی
