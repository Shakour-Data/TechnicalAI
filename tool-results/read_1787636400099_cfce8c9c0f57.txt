  1718→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
  1719→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
  1720→  const slice = data.slice(0, barIdx + 1);
  1721→  const price = slice[slice.length - 1].close;
  1722→  let nearestR = price * 1.05;
  1723→  let nearestS = price * 0.95;
  1724→
  1725→  // Collect all swing highs (resistance) and swing lows (support)
  1726→  const resistances: number[] = [];
  1727→  const supports: number[] = [];
  1728→  for (const lb of [3, 5, 7]) {
  1729→    const swings = findSwingLevels(slice, lb);
  1730→    resistances.push(...swings.highs);
  1731→    supports.push(...swings.lows);
  1732→  }
  1733→
  1734→  // Find nearest resistance above price
  1735→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
  1736→  if (abovePrice.length > 0) nearestR = abovePrice[0];
  1737→
  1738→  // Find nearest support below price
  1739→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
  1740→  if (belowPrice.length > 0) nearestS = belowPrice[0];
  1741→
  1742→  return { R1: nearestR, S1: nearestS };
  1743→}

  1745→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
  1746→// Returns true if bullish divergence detected, false if bearish, null if none
  1747→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
  1748→  const lookback = 20;
  1749→  const start = Math.max(0, barIdx - lookback);
  1750→  const slice = data.slice(start, barIdx + 1);
  1751→  if (slice.length < 10) return null;
  1752→
  1753→  const mid = Math.floor(slice.length / 2);
  1754→  const firstHalf = slice.slice(0, mid);
  1755→  const secondHalf = slice.slice(mid);
  1756→
  1757→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
  1758→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
  1759→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
  1760→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
  1761→
  1762→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
  1763→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
  1764→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
  1765→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
  1766→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
  1767→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
  1768→
  1769→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
  1770→    return false; // bearish divergence
  1771→  }
  1772→
  1773→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
  1774→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
  1775→
  1776→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
  1777→    return true; // bullish divergence
  1778→  }
  1779→
  1780→  return null;
  1781→}

  1784→// LAYER 1 & 2 — Feature Computation at a Historical Bar
  1785→// ═══════════════════════════════════════════════════════════════════════════════

  1787→/**
  1788→ * Computes all 16 ML features at a specific historical bar.
  1789→ * Uses data[0..barIdx] (inclusive) for all indicator calculations.
  1790→ * Returns an object keyed by feature name matching FEATURES_ORDER.
  1791→ */
  1792→function computeFeaturesAtBar(data: OHLCV[], barIdx: number): Record<string, number> {
  1793→  // Need at least 30 bars for meaningful indicators
  1794→  if (barIdx < 30) {
  1795→    const empty: Record<string, number> = {};
  1796→    for (const key of FEATURES_ORDER) empty[key] = 0.5;
  1797→    return empty;
  1798→  }
  1799→
  1800→  const slice = data.slice(0, barIdx + 1);
  1801→  const closes = slice.map(d => d.close);
  1802→  const price = closes[closes.length - 1];
  1803→  const eps = 1e-10;
  1804→
  1805→  // ── Compute all base indicators ───────────────────────────────────────────
  1806→  const rsiVal = calcRSI(closes);
  1807→  const sliceHasVolume = slice.some(d => d.volume > 0);
  1808→  const mfiVal = sliceHasVolume ? calcMFI(slice) : 50; // neutral when no volume
  1809→  const cciVal = calcCCI(slice);
  1810→  const stoch = calcStochastic(slice);
  1811→  const macd = calcMACD(closes);
  1812→  const adxResult = calcADX(slice);
  1813→  const atrVal = calcATR(slice);
  1814→  const bb = calcBollingerBands(closes);
  1815→  const ma21 = sma(closes, 21);
  1816→  const ma100 = sma(closes, 100);
  1817→  const ema12 = emaCalc(closes, 12);
  1818→  const ema26 = emaCalc(closes, 26);
  1819→  const trendResult = calcTrend(closes, 21);
  1820→  const sr = simplifiedSR(data, barIdx);
  1821→
  1822→  // ── Layer 1: Raw Scores (s_*) ────────────────────────────────────────────
  1823→
  1824→  // s_rsi — RSI mapped to 0-1
  1825→  const s_rsi = rsiVal > 80 ? 0.95
  1826→    : rsiVal > 70 ? 0.85
  1827→    : rsiVal > 60 ? 0.70
  1828→    : rsiVal > 50 ? 0.55
  1829→    : rsiVal > 40 ? 0.45
  1830→    : rsiVal > 30 ? 0.30
  1831→    : rsiVal > 20 ? 0.15
  1832→    : 0.05;
  1833→
  1834→  // s_mfi — MFI mapped to 0-1 (same formula as RSI)
  1835→  const s_mfi = mfiVal > 80 ? 0.95
  1836→    : mfiVal > 70 ? 0.85
  1837→    : mfiVal > 60 ? 0.70
  1838→    : mfiVal > 50 ? 0.55
  1839→    : mfiVal > 40 ? 0.45
  1840→    : mfiVal > 30 ? 0.30
  1841→    : mfiVal > 20 ? 0.15
  1842→    : 0.05;
  1843→
  1844→  // s_cci — CCI mapped to 0-1
  1845→  const s_cci = cciVal > 200 ? 0.90
  1846→    : cciVal > 100 ? 0.75
  1847→    : cciVal > 0 ? 0.60
  1848→    : cciVal > -100 ? 0.40
  1849→    : cciVal > -200 ? 0.25
  1850→    : 0.10;
  1851→
  1852→  // s_adx — Directional bias
  1853→  const s_adx = (adxResult.diPlus - adxResult.diMinus + 100) / 200;
  1854→
  1855→  // s_macd — MACD histogram + signal combined
  1856→  const absMacdLine = Math.abs(macd.line) + 1;
  1857→  const s_macd = macd.line > macd.signal
  1858→    ? 0.5 + Math.min(0.5, macd.histogram / absMacdLine)
  1859→    : 0.5 - Math.min(0.5, Math.abs(macd.histogram) / absMacdLine);
  1860→
  1861→  // s_stoch — Stochastic %K mapped to 0-1
  1862→  const s_stoch = stoch.k / 100;
  1863→
  1864→  // s_bb — Bollinger Band position
  1865→  const bbRange = bb.upper - bb.lower;
  1866→  const s_bb = bbRange > eps
  1867→    ? clamp((price - bb.lower) / bbRange, 0, 1)
  1868→    : 0.5;
  1869→
  1870→  // s_ma21 — Price relative to 21-period SMA
  1871→  const s_ma21 = ma21 > eps
  1872→    ? clamp(0.5 + (price - ma21) / ma21, 0, 1)
  1873→    : 0.5;
  1874→
  1875→  // s_ma100 — Price relative to 100-period SMA
  1876→  const s_ma100 = ma100 > eps
  1877→    ? clamp(0.5 + (price - ma100) / ma100, 0, 1)
  1878→    : 0.5;
  1879→
  1880→  // s_ema — EMA12 vs EMA26 spread
  1881→  const s_ema = ema26 > eps
  1882→    ? clamp(0.5 + (ema12 - ema26) / ema26 * 5, 0, 1)
  1883→    : 0.5;
  1884→
  1885→  // s_atr — Inverse volatility (lower ATR = higher score)
  1886→  const s_atr = price > eps ? 1 - Math.min(atrVal / price * 10, 1) : 0.5;
  1887→
  1888→  // s_trend — R² + angle combination
  1889→  const s_trend = (trendResult.r2 * 0.7) +
  1890→    (Math.max(0, Math.min(1, (trendResult.angle + 45) / 90)) * 0.3);
  1891→
  1892→  // s_sr — Support/Resistance proximity
  1893→  const dR1 = (sr.R1 - price) / (price > eps ? price : 1);
  1894→  const dS1 = (price - sr.S1) / (price > eps ? price : 1);
  1895→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
  1896→
  1897→  // ── Layer 2: Corrections (f_*) ───────────────────────────────────────────
  1898→
  1899→  // f_rsi — RSI with momentum correction
  1900→  const rsiCurrent = rsiVal;
  1901→  const rsi5ago = barIdx >= 5
  1902→    ? calcRSI(closes.slice(0, barIdx - 5 + 1))
  1903→    : rsiCurrent;
  1904→  const f_rsi = clamp(s_rsi + (rsiCurrent - rsi5ago) / 100 * 0.08, 0, 1);
  1905→
  1906→  // f_mfi — MFI with momentum correction (skip when no volume)
  1907→  const mfiCurrent = mfiVal;
  1908→  const mfi5ago = (barIdx >= 5 && sliceHasVolume)
  1909→    ? calcMFI(data.slice(0, barIdx - 5 + 1))
  1910→    : mfiCurrent;
  1911→  const f_mfi = sliceHasVolume
  1912→    ? clamp(s_mfi + (mfiCurrent - mfi5ago) / 100 * 0.08, 0, 1)
  1913→    : 0.5; // neutral when no volume
  1914→
  1915→  // f_cci — CCI with momentum correction (scale: /200 instead of /100)
  1916→  const cciCurrent = cciVal;
  1917→  const cci5ago = barIdx >= 5
  1918→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
  1919→    : cciCurrent;
  1920→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
  1921→
  1922→  // f_macd — Same as s_macd (no additional correction)
  1923→  const f_macd = s_macd;
  1924→
  1925→  // f_stoch — Same as s_stoch
  1926→  const f_stoch = s_stoch;
  1927→
  1928→  // f_stochCross — Stochastic K/D crossover detection
  1929→  let f_stochCross = 0.1;
  1930→  if (barIdx >= 1) {
  1931→    const prevSlice = data.slice(0, barIdx);
  1932→    const prevStoch = calcStochastic(prevSlice);
  1933→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
  1934→      f_stochCross = 0.9; // K crossed above D (bullish)
  1935→    }
  1936→  }
  1937→
  1938→  // f_macdCross — MACD/Signal crossover detection
  1939→  let f_macdCross = 0.1;
  1940→  if (barIdx >= 1) {
  1941→    const prevCloses = closes.slice(0, -1);
  1942→    if (prevCloses.length >= 35) {
  1943→      const prevMacd = calcMACD(prevCloses);
  1944→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
  1945→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
  1946→      }
  1947→    }
  1948→  }
  1949→
  1950→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
  1951→  const divResult = detectDivergenceSimple(data, barIdx);
  1952→  const f_div = divResult === true ? 0.85 : 0.15;
  1953→
  1954→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
  1955→  return {
  1956→    rsi: f_rsi,
  1957→    mfi: f_mfi,
  1958→    cci: f_cci,
  1959→    adx: s_adx,
  1960→    macd: f_macd,
  1961→    stoch: f_stoch,
  1962→    bb: s_bb,
  1963→    ma21: s_ma21,
  1964→    ma100: s_ma100,
  1965→    ema: s_ema,
  1966→    atr: s_atr,
  1967→    trend: s_trend,
  1968→    sr: s_sr,
  1969→    stochCross: f_stochCross,
  1970→    macdCross: f_macdCross,
  1971→    div: f_div,
  1972→  };
  1973→}

  1976→// TRAINING DATA CREATION (Layer 0 — Data Preparation)

  1978→
  1979→/**
  1980→ * Creates supervised training data from historical OHLCV data.
  1981→ * For each bar, computes features and labels based on future price movement.
  1982→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
  1983→ * Samples every `sampleStep` bars to reduce computation.
  1984→ */
  1985→function createTrainingData(
  1986→  data: OHLCV[],
  1987→  horizon: number = 5,
  1988→  sampleStep: number = 5
  1989→): { X: number[][]; y: number[] } {
  1990→  const X: number[][] = [];
  1991→  const y: number[] = [];
  1992→
  1993→  // Start from horizon+30 to ensure enough data for indicators
  1994→  // End at data.length - horizon - 1 to have future bars for labeling
  1995→  // Sample every sampleStep bars for performance
  1996→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
  1997→    const features = computeFeaturesAtBar(data, i);
  1998→
  1999→    // Convert to array following FEATURES_ORDER
  2000→    const featureVec: number[] = [];
  2001→    let valid = true;
  2002→    for (const key of FEATURES_ORDER) {
  2003→      const val = features[key];
  2004→      if (val === undefined || val === null || Number.isNaN(val)) {
  2005→        valid = false;
  2006→        break;
  2007→      }
  2008→      featureVec.push(val);
  2009→    }
  2010→    if (!valid) continue;
  2011→
  2012→    // Label: 1 if future close > 1% above current close
  2013→    const currentClose = data[i].close;
  2014→    const futureClose = data[i + horizon].close;
  2015→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
  2016→
  2017→    X.push(featureVec);
  2018→    y.push(label);
  2019→  }
  2020→
  2021→  return { X, y };
  2022→}

  2024→// ═══════════════════════════════════════════════════════════════════════════════
  2025→// VDss GRAPH STRUCTURE (Layer 5 — State Machine)
  2026→// ═══════════════════════════════════════════════════════════════════════════════
  2027→
  2028→/**
  2029→ * VDss Graph: Each node represents a market state.
  2030→ * Edges represent transitions with types: 'up', 'down', 'pullback', 'risk', 'terminal'.
  2031→ * Terminal nodes map to scenario results (R1=strong bullish, R5=deep correction).
  2032→ */
  2033→const VDSS_GRAPH = {
  2034→  // outgoing edges from each node: [target, type]
  2035→  // v11: 9 scenario terminals — R1(most bullish) to R9(most bearish)
  2036→  outgoing: {
  2037→    A: [['B', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  2038→    B: [['C', 'up'], ['G', 'pullback'], ['A', 'risk'], ['R3', 'terminal']] as [string, string][],
  2039→    C: [['D', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  2040→    D: [['E', 'up'], ['B', 'pullback'], ['G', 'pullback'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  2041→    E: [['F', 'up'], ['D', 'pullback'], ['G', 'pullback'], ['H', 'down'], ['R2', 'terminal']] as [string, string][],
  2042→    F: [['R1', 'terminal'], ['E', 'pullback'], ['D', 'down'], ['G', 'risk']] as [string, string][],
  2043→    G: [['B', 'up'], ['A', 'pullback'], ['H', 'down'], ['L', 'risk'], ['R4', 'terminal']] as [string, string][],
  2044→    H: [['G', 'pullback'], ['I', 'down'], ['B', 'up'], ['L', 'risk'], ['R5', 'terminal']] as [string, string][],
  2045→    I: [['H', 'pullback'], ['J', 'down'], ['G', 'up'], ['L', 'risk'], ['R6', 'terminal']] as [string, string][],
  2046→    J: [['I', 'pullback'], ['K', 'down'], ['H', 'up'], ['L', 'risk'], ['R7', 'terminal']] as [string, string][],
  2047→    K: [['J', 'pullback'], ['I', 'up'], ['L', 'risk'], ['R8', 'terminal'], ['R9', 'terminal']] as [string, string][],
  2048→    L: [['C', 'up'], ['H', 'pullback'], ['J', 'down'], ['K', 'risk'], ['R9', 'terminal']] as [string, string][],
  2049→  },
  2050→  // which result scenario each terminal node maps to
  2051→  terminalMap: { R1: 'R1', R2: 'R2', R3: 'R3', R4: 'R4', R5: 'R5', R6: 'R6', R7: 'R7', R8: 'R8', R9: 'R9' } as Record<string, string>,
  2052→  startNode: 'A' as string,
  2053→};
  2054→
  2055→// ═══════════════════════════════════════════════════════════════════════════════
  2056→// LAYER 5 — Edge Weight Calculation
  2057→// ═══════════════════════════════════════════════════════════════════════════════
  2058→
  2059→/**
  2060→ * Calculates transition edge weights for the VDss graph.
  2061→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
  2062→ * to determine the relative probability of up/down/pullback/risk transitions.
  2063→ */
  2064→function calculateEdgeWeights(
  2065→  bullConsensus: number,
  2066→  adx: number,
  2067→  mlModel: AdaptiveWeightModel
  2068→): { up: number; down: number; pullback: number; risk: number } {
  2069→  // Extract ML coefficient insights if model is trained
  2070→  let trend_coef = 1 / 3;
  2071→  let momentum_coef = 1 / 3;
  2072→  let volatility_coef = 1 / 3;
  2073→
  2074→  if (mlModel.isTrained && mlModel.weights) {
  2075→    const coefs = mlModel.getCoefficients();
  2076→    // Map feature indices to coefficient categories
  2077→    // FEATURES_ORDER: ['rsi','mfi','cci','adx','macd','stoch','bb','ma21','ma100','ema','atr','trend','sr','stochCross','macdCross','div']
  2078→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER
  2079→    const rsiIdx = 0;     // 'rsi' in FEATURES_ORDER
  2080→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
  2081→
  2082→    const absTrend = Math.abs(coefs[trendIdx] ?? 0);
  2083→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
  2084→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
  2085→
  2086→    const coefSum = absTrend + absRsi + absAtr;
  2087→    if (coefSum > 1e-10) {
  2088→      trend_coef = absTrend / coefSum;
  2089→      momentum_coef = absRsi / coefSum;
  2090→      volatility_coef = absAtr / coefSum;
  2091→    }
  2092→  }
  2093→
  2094→  // Base weights driven by bull consensus
  2095→  const up_base = bullConsensus * 0.7 + 0.15;
  2096→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
  2097→  const pullback_base = 0.25;
  2098→  const risk_base = 0.12 * (1.2 - adx / 100);
  2099→
  2100→  // Apply ML coefficient adjustments
  2101→  const up = clamp(
  2102→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
  2103→    0.05, 0.95
  2104→  );
  2105→  const down = clamp(
  2106→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
  2107→    0.05, 0.95
  2108→  );
  2109→  const pullback = clamp(
  2110→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
  2111→    0.05, 0.50
  2112→  );
  2113→  const risk = clamp(
  2114→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
  2115→    0.02, 0.30
  2116→  );
  2117→
  2118→  return { up, down, pullback, risk };
  2119→}
  2120→
  2121→// ═══════════════════════════════════════════════════════════════════════════════
  2122→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)
  2123→// ═══════════════════════════════════════════════════════════════════════════════
  2124→
  2125→/**
  2126→ * Performs iterative DFS from the start node through the VDss graph,
  2127→ * computing raw path probabilities, then calibrating them against
  2128→ * Layer 4 scenario probabilities.
  2129→ *
  2130→ * Returns:
  2131→ * - calibratedPaths: Map<pathString, probability>
  2132→ * - calibrationFactors: per-scenario multiplier used
  2133→ * - scenarioSums: raw probability sums per scenario before calibration
  2134→ */
  2135→function calculatePathProbabilities(
  2136→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
  2137→  scenarioProbs: Record<string, number>
  2138→): {
  2139→  calibratedPaths: Map<string, number>;
  2140→  calibrationFactors: Record<string, number>;
  2141→  scenarioSums: Record<string, number>;
  2142→} {
  2143→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
  2144→  const MAX_PATHS = 200;
  2145→
  2146→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
  2147→  type StackEntry = [string, string, number];
  2148→
  2149→  // Collect all terminal paths with raw probabilities
  2150→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
  2151→
  2152→  // Iterative DFS
  2153→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
  2154→
  2155→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
  2156→    const [node, pathStr, pathProb] = stack.pop()!;
  2157→
  2158→    const edges = outgoing[node as keyof typeof outgoing];
  2159→    if (!edges) continue;
  2160→
  2161→    // Normalize outgoing edge weights to probabilities for this node
  2162→    // Map edge types to their weights
  2163→    let totalWeight = 0;
  2164→    const edgeWeightsList: number[] = [];
  2165→    for (const [, edgeType] of edges) {
  2166→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
  2167→      edgeWeightsList.push(w);
  2168→      totalWeight += w;
  2169→    }
  2170→
  2171→    if (totalWeight < 1e-10) continue;
  2172→
  2173→    // Expand edges
  2174→    for (let i = 0; i < edges.length; i++) {
  2175→      const [target, edgeType] = edges[i];
  2176→      const edgeProb = edgeWeightsList[i] / totalWeight;
  2177→      const newProb = pathProb * edgeProb;
  2178→      const newPath = pathStr + '→' + target;
  2179→
  2180→      if (edgeType === 'terminal') {
  2181→        // Reached a terminal node
  2182→        const scenario = terminalMap[target] ?? 'R3';
  2183→        rawPaths.push({ path: newPath, scenario, prob: newProb });
  2184→      } else {
  2185→        // Continue DFS
  2186→        stack.push([target, newPath, newProb]);
  2187→      }
  2188→    }
  2189→  }
  2190→

  2202→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
  2203→  }
  2204→
  2205→  // Apply calibration to each path
  2206→  const calibratedPaths = new Map<string, number>();
  2207→  for (const rp of rawPaths) {
  2208→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
  2209→    calibratedPaths.set(rp.path, rp.prob * factor);
  2210→  }
  2211→
  2212→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
  2213→}

  2228→  for (let i = 0; i < 4; i++) {
  2229→    const base = above[i] ?? (price + (i + 1) * 0.25 * atr);
  2230→    // Ensure base is above previous target and above price
  2231→    const safeBase = Math.max(base, prevMax + 1);
  2232→    let min = rnd(Math.max(safeBase - 0.25 * atr, prevMax + 1));
  2233→    let max = rnd(Math.min(safeBase + 0.5 * atr, min + atr));
  2234→    if (max <= min) max = rnd(min + 0.25 * atr);
  2235→    if (min <= prevMax) min = rnd(prevMax + 1);
  2236→    if (max <= min) max = rnd(min + 0.25 * atr);
  2237→    // Enforce max 1 ATR range
  2238→    if (max - min > atr) max = rnd(min + atr);
  2239→    results.push({min, max});
  2240→    prevMax = max;
  2241→  }
  2242→  return results;
  2243→}
