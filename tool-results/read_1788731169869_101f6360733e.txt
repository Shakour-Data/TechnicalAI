   125→   125→// ─── EMA (full array) ───────────────────────────────────────────────────────────
   126→   126→function emaArray(data: number[], period: number): number[] {
   127→   127→  const result: number[] = new Array(data.length).fill(0);
   128→   128→  if (data.length < period) return result;
   129→   129→  let sum = 0;
   130→   130→  for (let i = 0; i < period; i++) sum += data[i];
   131→   131→  result[period - 1] = sum / period;
   132→   132→  const k = 2 / (period + 1);
   133→   133→  for (let i = period; i < data.length; i++) {
   134→   134→    result[i] = data[i] * k + result[i - 1] * (1 - k);
   135→   135→  }
   136→   136→  return result;
   137→   137→}

   143→   143→// ─── SMA (full array) ──────────────────────────────────────────────────────────
   144→   144→function smaArray(closes: number[], period: number): number[] {
   145→   145→  const result: number[] = new Array(closes.length).fill(0);
   146→   146→  for (let i = period - 1; i < closes.length; i++) {
   147→   147→    let sum = 0;
   148→   148→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   149→   149→    result[i] = sum / period;
   150→   150→  }
   151→   151→  return result;
   152→   152→}

   154→   154→// ─── WMA — Weighted Moving Average ─────────────────────────────────────────────
   155→   155→function wmaArray(data: number[], period: number): number[] {
   156→   156→  const result: number[] = new Array(data.length).fill(0);
   157→   157→  if (data.length < period) return result;
   158→   158→  const denom = period * (period + 1) / 2;
   159→   159→  for (let i = period - 1; i < data.length; i++) {
   160→   160→    let sum = 0;
   161→   161→    for (let j = 0; j < period; j++) sum += data[i - j] * (period - j);
   162→   162→    result[i] = sum / denom;
   163→   163→  }
   164→   164→  return result;
   165→   165→}

   167→   167→// ─── HMA — Hull Moving Average ─────────────────────────────────────────────────
   168→   168→function hmaArray(closes: number[], period: number): number[] {
   169→   169→  const halfPeriod = Math.max(1, Math.floor(period / 2));
   170→   170→  const sqrtPeriod = Math.max(2, Math.floor(Math.sqrt(period)));
   171→   171→  const wmaHalf = wmaArray(closes, halfPeriod);
   172→   172→  const wmaFull = wmaArray(closes, period);
   173→   173→  const diff = new Array(closes.length).fill(0);
   174→   174→  for (let i = 0; i < closes.length; i++) diff[i] = 2 * wmaHalf[i] - wmaFull[i];
   175→   175→  return wmaArray(diff, sqrtPeriod);
   176→   176→}

   179→   179→function tmaArray(closes: number[], period: number): number[] {
   180→   180→  const halfPeriod = Math.max(2, Math.ceil(period / 2));
   181→   181→  const firstPass = smaArray(closes, period);
   182→   182→  return smaArray(firstPass, halfPeriod);
   183→   183→}

   185→   185→// ─── LMA — Linear Regression (Least Squares) Moving Average ────────────────────
   186→   186→function lmaArray(closes: number[], period: number): number[] {
   187→   187→  const result: number[] = new Array(closes.length).fill(0);
   188→   188→  if (closes.length < period) return result;
   189→   189→  const sumX = period * (period - 1) / 2;
   190→   190→  const sumX2 = (period - 1) * period * (2 * period - 1) / 6;
   191→   191→  const denom = period * sumX2 - sumX * sumX;
   192→   192→  if (denom === 0) return result;
   193→   193→  for (let i = period - 1; i < closes.length; i++) {
   194→   194→    let sumY = 0;
   195→   195→    let sumXY = 0;
   196→   196→    for (let j = 0; j < period; j++) {
   197→   197→      const idx = i - period + 1 + j;
   198→   198→      sumY += closes[idx];
   199→   199→      sumXY += j * closes[idx];
   200→   200→    }
   201→   201→    const b = (period * sumXY - sumX * sumY) / denom;
   202→   202→    const a = (sumY - b * sumX) / period;
   203→   203→    result[i] = a + b * (period - 1);
   204→   204→  }
   205→   205→  return result;
   206→   206→}

   208→   208→// ─── VWMA — Volume Weighted Moving Average ──────────────────────────────────────
   209→   209→function vwmaArray(closes: number[], volumes: number[], period: number): number[] {
   210→   210→  const result: number[] = new Array(closes.length).fill(0);
   211→   211→  if (closes.length < period) return result;
   212→   212→  for (let i = period - 1; i < closes.length; i++) {
   213→   213→    let pvSum = 0;
   214→   214→    let vSum = 0;
   215→   215→    for (let j = i - period + 1; j <= i; j++) {
   216→   216→      pvSum += closes[j] * volumes[j];
   217→   217→      vSum += volumes[j];
   218→   218→    }
   219→   219→    result[i] = vSum === 0 ? 0 : pvSum / vSum;
   220→   220→  }
   221→   221→  return result;
   222→   222→}

   311→   311→// ─── RSI (full array) ──────────────────────────────────────────────────────────
   312→   312→function rsiArray(closes: number[], period: number = 14): number[] {
   313→   313→  const result: number[] = new Array(closes.length).fill(0);
   314→   314→  if (closes.length < period + 1) return result;
   315→   315→
   316→   316→  let gainSum = 0,
   317→   317→    lossSum = 0;
   318→   318→  for (let i = 1; i <= period; i++) {
   319→   319→    const diff = closes[i] - closes[i - 1];
   320→   320→    if (diff > 0) gainSum += diff;
   321→   321→    else lossSum += Math.abs(diff);
   322→   322→  }
   323→   323→  let avgGain = gainSum / period;
   324→   324→  let avgLoss = lossSum / period;
   325→   325→  result[period] = avgLoss === 0 ? 100 : 100 - 100 / (1 + avgGain / avgLoss);
   326→   326→
   327→   327→  for (let i = period + 1; i < closes.length; i++) {
   328→   328→    const diff = closes[i] - closes[i - 1];
   329→   329→    const gain = diff > 0 ? diff : 0;
   330→   330→    const loss = diff < 0 ? Math.abs(diff) : 0;
   331→   331→    avgGain = (avgGain * (period - 1) + gain) / period;
   332→   332→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   333→   333→    result[i] = avgLoss === 0 ? 100 : 100 - 100 / (1 + avgGain / avgLoss);
   334→   334→  }
   335→   335→  return result;
   336→   336→}

   338→   338→// ─── Stochastic (full array) ───────────────────────────────────────────────────
   339→   339→function stochasticArray(
   340→   340→  highs: number[],
   341→   341→  lows: number[],
   342→   342→  closes: number[],
   343→   343→  kPeriod: number = 14,
   344→   344→  dPeriod: number = 3,
   345→   345→): { k: number[]; d: number[] } {
   346→   346→  const kArr: number[] = new Array(closes.length).fill(0);
   347→   347→  const dArr: number[] = new Array(closes.length).fill(0);
   348→   348→
   349→   349→  for (let i = kPeriod - 1; i < closes.length; i++) {
   350→   350→    let highest = -Infinity,
   351→   351→      lowest = Infinity;
   352→   352→    for (let j = i - kPeriod + 1; j <= i; j++) {
   353→   353→      if (highs[j] > highest) highest = highs[j];
   354→   354→      if (lows[j] < lowest) lowest = lows[j];
   355→   355→    }
   356→   356→    kArr[i] = highest === lowest ? 50 : ((closes[i] - lowest) / (highest - lowest)) * 100;
   357→   357→  }
   358→   358→
   359→   359→  for (let i = kPeriod - 1 + dPeriod - 1; i < closes.length; i++) {
   360→   360→    let sum = 0;
   361→   361→    for (let j = i - dPeriod + 1; j <= i; j++) sum += kArr[j];
   362→   362→    dArr[i] = sum / dPeriod;
   363→   363→  }
   364→   364→  return { k: kArr, d: dArr };
   365→   365→}

   368→   368→function stochRsiArray(
   369→   369→  closes: number[],
   370→   370→  rsiPeriod: number = 14,
   371→   371→  stochPeriod: number = 14,
   372→   372→  smoothPeriod: number = 3,
   373→   373→): { k: number[]; d: number[] } {
   374→   374→  const rsiValues = rsiArray(closes, rsiPeriod);
   375→   375→  const kArr: number[] = new Array(closes.length).fill(0);
   376→   376→  const dArr: number[] = new Array(closes.length).fill(0);
   377→   377→  const startIdx = rsiPeriod - 1 + stochPeriod - 1;
   378→   378→
   379→   379→  for (let i = startIdx; i < closes.length; i++) {
   380→   380→    let minRsi = Infinity,
   381→   381→      maxRsi = -Infinity;
   382→   382→    for (let j = i - stochPeriod + 1; j <= i; j++) {
   383→   383→      if (rsiValues[j] < minRsi) minRsi = rsiValues[j];
   384→   384→      if (rsiValues[j] > maxRsi) maxRsi = rsiValues[j];
   385→   385→    }
   386→   386→    kArr[i] = maxRsi === minRsi ? 50 : ((rsiValues[i] - minRsi) / (maxRsi - minRsi)) * 100;
   387→   387→  }
   388→   388→
   389→   389→  for (let i = startIdx + smoothPeriod - 1; i < closes.length; i++) {
   390→   390→    let sum = 0;
   391→   391→    for (let j = i - smoothPeriod + 1; j <= i; j++) sum += kArr[j];
   392→   392→    dArr[i] = sum / smoothPeriod;
   393→   393→  }
   394→   394→  return { k: kArr, d: dArr };
   395→   395→}

   426→   426→// ─── ROC — Rate of Change ──────────────────────────────────────────────────────
   427→   427→function rocArray(closes: number[], period: number = 12): number[] {
   428→   428→  const result: number[] = new Array(closes.length).fill(0);
   429→   429→  for (let i = period; i < closes.length; i++) {
   430→   430→    result[i] = closes[i - period] === 0 ? 0 : ((closes[i] - closes[i - period]) / closes[i - period]) * 100;
   431→   431→  }
   432→   432→  return result;
   433→   433→}

   436→   436→function trixArray(closes: number[], period: number = 9): number[] {
   437→   437→  const result: number[] = new Array(closes.length).fill(0);
   438→   438→  const startIdx = 3 * (period - 1) + 1;
   439→   439→  if (closes.length < startIdx) return result;
   440→   440→
   441→   441→  const ema1 = emaArray(closes, period);
   442→   442→  const ema2 = emaArray(ema1, period);
   443→   443→  const ema3 = emaArray(ema2, period);
   444→   444→
   445→   445→  for (let i = startIdx; i < closes.length; i++) {
   446→   446→    if (ema3[i - 1] === 0) {
   447→   447→      result[i] = 0;
   448→   448→    } else {
   449→   449→      result[i] = ((ema3[i] - ema3[i - 1]) / Math.abs(ema3[i - 1])) * 100;
   450→   450→    }
   451→   451→  }
   452→   452→  return result;
   453→   453→}

   455→   455→// ─── CCI (full array) ──────────────────────────────────────────────────────────
   456→   456→function cciArray(
   457→   457→  highs: number[],
   458→   458→  lows: number[],
   459→   459→  closes: number[],
   460→   460→  period: number = 20,
   461→   461→): number[] {
   462→   462→  const len = closes.length;
   463→   463→  const result: number[] = new Array(len).fill(0);
   464→   464→
   465→   465→  for (let i = period - 1; i < len; i++) {
   466→   466→    let tpSum = 0;
   467→   467→    for (let j = i - period + 1; j <= i; j++) {
   468→   468→      tpSum += (highs[j] + lows[j] + closes[j]) / 3;
   469→   469→    }
   470→   470→    const tpMean = tpSum / period;
   471→   471→    let md = 0;
   472→   472→    for (let j = i - period + 1; j <= i; j++) {
   473→   473→      md += Math.abs((highs[j] + lows[j] + closes[j]) / 3 - tpMean);
   474→   474→    }
   475→   475→    md /= period;
   476→   476→    result[i] = md === 0 ? 0 : ((highs[i] + lows[i] + closes[i]) / 3 - tpMean) / (0.015 * md);
   477→   477→  }
   478→   478→  return result;
   479→   479→}

   482→   482→function mfiArray(
   483→   483→  highs: number[],
   484→   484→  lows: number[],
   485→   485→  closes: number[],
   486→   486→  volumes: number[],
   487→   487→  period: number = 14,
   488→   488→): number[] {
   489→   489→  const len = closes.length;
   490→   490→  const result: number[] = new Array(len).fill(0);
   491→   491→
   492→   492→  for (let i = period; i < len; i++) {
   493→   493→    let posFlow = 0,
   494→   494→      negFlow = 0;
   495→   495→    for (let j = i - period + 1; j <= i; j++) {
   496→   496→      const tp = (highs[j] + lows[j] + closes[j]) / 3;
   497→   497→      const prevTp = (highs[j - 1] + lows[j - 1] + closes[j - 1]) / 3;
   498→   498→      const mf = tp * volumes[j];
   499→   499→      if (tp > prevTp) posFlow += mf;
   500→   500→      else negFlow += mf;
   501→   501→    }
   502→   502→    result[i] = negFlow === 0 ? 100 : 100 - 100 / (1 + posFlow / negFlow);
   503→   503→  }
   504→   504→  return result;

   535→   535→// ─── ADX + DMI (full array) ────────────────────────────────────────────────────
   536→   536→function adxArray(
   537→   537→  highs: number[],
   538→   538→  lows: number[],
   539→   539→  closes: number[],
   540→   540→  period: number = 14,
   541→   541→): { adx: number[]; dmiPlus: number[]; dmiMinus: number[] } {
   542→   542→  const len = closes.length;
   543→   543→  const adxResult: number[] = new Array(len).fill(0);
   544→   544→  const dmiPlusResult: number[] = new Array(len).fill(0);
   545→   545→  const dmiMinusResult: number[] = new Array(len).fill(0);
   546→   546→  if (len < period * 2 + 1) return { adx: adxResult, dmiPlus: dmiPlusResult, dmiMinus: dmiMinusResult };
   547→   547→
   548→   548→  const tr: number[] = new Array(len).fill(0);
   549→   549→  const plusDM: number[] = new Array(len).fill(0);
   550→   550→  const minusDM: number[] = new Array(len).fill(0);
   551→   551→
   552→   552→  for (let i = 1; i < len; i++) {
   553→   553→    tr[i] = Math.max(highs[i] - lows[i], Math.abs(highs[i] - closes[i - 1]), Math.abs(lows[i] - closes[i - 1]));
   554→   554→    const upMove = highs[i] - highs[i - 1];
   555→   555→    const downMove = lows[i - 1] - lows[i];
   556→   556→    plusDM[i] = upMove > downMove && upMove > 0 ? upMove : 0;
   557→   557→    minusDM[i] = downMove > upMove && downMove > 0 ? downMove : 0;
   558→   558→  }
   559→   559→
   560→   560→  const smooth = (arr: number[], start: number) => {
   561→   561→    let sum = 0;
   562→   562→    for (let i = start; i < start + period; i++) sum += arr[i];
   563→   563→    const res: number[] = new Array(len).fill(0);
   564→   564→    res[start + period - 1] = sum;
   565→   565→    for (let i = start + period; i < len; i++) {
   566→   566→      res[i] = res[i - 1] - res[i - 1] / period + arr[i];
   567→   567→    }
   568→   568→    return res;
   569→   569→  };
   570→   570→
   571→   571→  const atrSmooth = smooth(tr, 1);
   572→   572→  const sPlusDM = smooth(plusDM, 1);
   573→   573→  const sMinusDM = smooth(minusDM, 1);
   574→   574→
   575→   575→  // Compute DI+ and DI-
   576→   576→  for (let i = 1; i < len; i++) {
   577→   577→    const atrVal = atrSmooth[i];
   578→   578→    dmiPlusResult[i] = atrVal === 0 ? 0 : (sPlusDM[i] / atrVal) * 100;
   579→   579→    dmiMinusResult[i] = atrVal === 0 ? 0 : (sMinusDM[i] / atrVal) * 100;
   580→   580→  }
   581→   581→
   582→   582→  const dx: number[] = new Array(len).fill(0);
   583→   583→  for (let i = 1; i < len; i++) {
   584→   584→    const sum = sPlusDM[i] + sMinusDM[i];
   585→   585→    dx[i] = sum === 0 ? 0 : (Math.abs(sPlusDM[i] - sMinusDM[i]) / sum) * 100;
   586→   586→  }
   587→   587→
   588→   588→  // ADX = smoothed DX
   589→   589→  let adxSum = 0;
   590→   590→  const startIdx = period * 2;
   591→   591→  for (let i = period + 1; i <= startIdx; i++) adxSum += dx[i];
   592→   592→  adxResult[startIdx] = adxSum / period;
   593→   593→  for (let i = startIdx + 1; i < len; i++) {
   594→   594→    adxResult[i] = (adxResult[i - 1] * (period - 1) + dx[i]) / period;
   595→   595→  }
   596→   596→
   597→   597→  return { adx: adxResult, dmiPlus: dmiPlusResult, dmiMinus: dmiMinusResult };
   598→   598→}

   600→   600→// ─── Parabolic SAR (full array) ────────────────────────────────────────────────
   601→   601→function sarArray(highs: number[], lows: number[], closes: number[]): number[] {
   602→   602→  const len = closes.length;
   603→   603→  const result: number[] = new Array(len).fill(0);
   604→   604→  if (len < 2) return result;
   605→   605→
   606→   606→  let isLong = closes[1] > closes[0];
   607→   607→  let af = 0.02;
   608→   608→  let ep = isLong ? highs[1] : lows[1];
   609→   609→  result[0] = lows[0];
   610→   610→  result[1] = isLong ? lows[0] : highs[0];
   611→   611→
   612→   612→  for (let i = 2; i < len; i++) {
   613→   613→    result[i] = result[i - 1] + af * (ep - result[i - 1]);
   614→   614→
   615→   615→    if (isLong) {
   616→   616→      if (lows[i] < result[i]) {
   617→   617→        isLong = false;
   618→   618→        result[i] = ep;
   619→   619→        ep = lows[i];
   620→   620→        af = 0.02;
   621→   621→      } else {
   622→   622→        if (highs[i] > ep) {
   623→   623→          ep = highs[i];
   624→   624→          af = Math.min(af + 0.02, 0.2);
   625→   625→        }
   626→   626→      }
   627→   627→    } else {
   628→   628→      if (highs[i] > result[i]) {
   629→   629→        isLong = true;
   630→   630→        result[i] = ep;
   631→   631→        ep = highs[i];
   632→   632→        af = 0.02;
   633→   633→      } else {
   634→   634→        if (lows[i] < ep) {
   635→   635→          ep = lows[i];
   636→   636→          af = Math.min(af + 0.02, 0.2);
   637→   637→        }
   638→   638→      }
   639→   639→    }
   640→   640→  }
   641→   641→  return result;
   642→   642→}

   656→   656→// ─── Fisher Transform ──────────────────────────────────────────────────────────
   657→   657→function fisherTransformArray(highs: number[], lows: number[], period: number = 9): number[] {
   658→   658→  const result: number[] = new Array(highs.length).fill(0);
   659→   659→  if (highs.length < period) return result;
   660→   660→
   661→   661→  for (let i = period - 1; i < highs.length; i++) {
   662→   662→    let maxH = -Infinity,
   663→   663→      minL = Infinity;
   664→   664→    for (let j = i - period + 1; j <= i; j++) {
   665→   665→      if (highs[j] > maxH) maxH = highs[j];
   666→   666→      if (lows[j] < minL) minL = lows[j];
   667→   667→    }
   668→   668→    const range = maxH - minL;
   669→   669→    if (range === 0) {
   670→   670→      result[i] = 0;
   671→   671→      continue;
   672→   672→    }
   673→   673→    const raw = ((highs[i] + lows[i]) / 2 - minL) / range;
   674→   674→    const x = Math.max(-0.999, Math.min(0.999, 2 * raw - 1));
   675→   675→    result[i] = 0.5 * Math.log((1 + x) / (1 - x));
   676→   676→  }
   677→   677→  return result;
   678→   678→}

   685→   685→function atrArray(highs: number[], lows: number[], closes: number[], period: number = 14): number[] {
   686→   686→  const len = closes.length;
   687→   687→  const tr: number[] = new Array(len).fill(0);
   688→   688→  const result: number[] = new Array(len).fill(0);
   689→   689→
   690→   690→  for (let i = 1; i < len; i++) {
   691→   691→    tr[i] = Math.max(
   692→   692→      highs[i] - lows[i],
   693→   693→      Math.abs(highs[i] - closes[i - 1]),
   694→   694→      Math.abs(lows[i] - closes[i - 1]),
   695→   695→    );
   696→   696→  }
   697→   697→
   698→   698→  if (len < period + 1) return result;
   699→   699→
   700→   700→  let sum = 0;
   701→   701→  for (let i = 1; i <= period; i++) sum += tr[i];
   702→   702→  result[period] = sum / period;
   703→   703→  for (let i = period + 1; i < len; i++) {
   704→   704→    result[i] = (result[i - 1] * (period - 1) + tr[i]) / period;
   705→   705→  }
   706→   706→  return result;
   707→   707→}

   711→   711→  closes: number[],
   712→   712→  period: number = 20,
   713→   713→  mult: number = 2,
   714→   714→): { upper: number[]; middle: number[]; lower: number[]; width: number[] } {
   715→   715→  const middle = smaArray(closes, period);
   716→   716→  const upper: number[] = new Array(closes.length).fill(0);
   717→   717→  const lower: number[] = new Array(closes.length).fill(0);
   718→   718→  const width: number[] = new Array(closes.length).fill(0);
   719→   719→
   720→   720→  for (let i = period - 1; i < closes.length; i++) {
   721→   721→    let sumSq = 0;
   722→   722→    for (let j = i - period + 1; j <= i; j++) {
   723→   723→      sumSq += (closes[j] - middle[i]) ** 2;
   724→   724→    }
   725→   725→    const stdDev = Math.sqrt(sumSq / period);
   726→   726→    upper[i] = middle[i] + mult * stdDev;
   727→   727→    lower[i] = middle[i] - mult * stdDev;
   728→   728→    width[i] = middle[i] === 0 ? 0 : ((upper[i] - lower[i]) / middle[i]) * 100;
   729→   729→  }
   730→   730→  return { upper, middle, lower, width };
   731→   731→}

   772→   772→function histVolatilityArray(closes: number[], period: number = 20): number[] {
   773→   773→  const result: number[] = new Array(closes.length).fill(0);
   774→   774→  if (closes.length < period + 1) return result;
   775→   775→
   776→   776→  const logReturns: number[] = new Array(closes.length).fill(0);
   777→   777→  for (let i = 1; i < closes.length; i++) {
   778→   778→    logReturns[i] = closes[i - 1] === 0 ? 0 : Math.log(closes[i] / closes[i - 1]);
   779→   779→  }
   780→   780→
   781→   781→  for (let i = period; i < closes.length; i++) {
   782→   782→    let sum = 0;

   784→   784→    const mean = sum / period;
   785→   785→    let sumSq = 0;
   786→   786→    for (let j = i - period + 1; j <= i; j++) sumSq += (logReturns[j] - mean) ** 2;
   787→   787→    result[i] = Math.sqrt(sumSq / period) * Math.sqrt(252);
   788→   788→  }
   789→   789→  return result;
   790→   790→}

   836→   836→  const smaLong = smaArray(volumes, longP);
   837→   837→  const result: number[] = new Array(volumes.length).fill(0);
   838→   838→  for (let i = 0; i < volumes.length; i++) {
   839→   839→    result[i] = smaLong[i] === 0 ? 0 : ((smaShort[i] - smaLong[i]) / smaLong[i]) * 100;
   840→   840→  }
   841→   841→  return result;
   842→   842→}

   844→   844→// ─── Chaikin A/D Line ──────────────────────────────────────────────────────────
   845→   845→function chaikinADArray(highs: number[], lows: number[], closes: number[], volumes: number[]): number[] {
   846→   846→  const result: number[] = new Array(closes.length).fill(0);
   847→   847→  for (let i = 0; i < closes.length; i++) {
   848→   848→    const range = highs[i] - lows[i];
   849→   849→    const clv = range === 0 ? 0 : ((closes[i] - lows[i]) - (highs[i] - closes[i])) / range;
   850→   850→    result[i] = (i === 0 ? 0 : result[i - 1]) + clv * volumes[i];
   851→   851→  }
   852→   852→  return result;
   853→   853→}

   855→   855→// ─── Force Index (EMA-smoothed) ────────────────────────────────────────────────
   856→   856→function forceIndexArray(closes: number[], volumes: number[], period: number = 13): number[] {
   857→   857→  const len = closes.length;
   858→   858→  const result: number[] = new Array(len).fill(0);
   859→   859→  if (len < 2) return result;
   860→   860→  const raw = new Array(len).fill(0);
   861→   861→  for (let i = 1; i < len; i++) raw[i] = (closes[i] - closes[i - 1]) * volumes[i];
   862→   862→  const smoothed = emaArray(raw, period);
   863→   863→  return smoothed;
   864→   864→}

   866→   866→// ─── VAP — Volume at Price (simplified: rolling VWAP using typical price) ──────
   867→   867→function vapArray(highs: number[], lows: number[], closes: number[], volumes: number[], period: number = 20): number[] {
   868→   868→  const result: number[] = new Array(closes.length).fill(0);
   869→   869→  if (closes.length < period) return result;
   870→   870→  for (let i = period - 1; i < closes.length; i++) {
   871→   871→    let pvSum = 0;
   872→   872→    let vSum = 0;
   873→   873→    for (let j = i - period + 1; j <= i; j++) {
   874→   874→      const tp = (highs[j] + lows[j] + closes[j]) / 3;
   875→   875→      pvSum += tp * volumes[j];
   876→   876→      vSum += volumes[j];
   877→   877→    }
   878→   878→    result[i] = vSum === 0 ? 0 : pvSum / vSum;
   879→   879→  }
   880→   880→  return result;
   881→   881→}

   883→   883→// ─── VWAP — Volume Weighted Average Price (cumulative) ─────────────────────────
   884→   884→function vwapArray(highs: number[], lows: number[], closes: number[], volumes: number[]): number[] {
   885→   885→  const result: number[] = new Array(closes.length).fill(0);
   886→   886→  if (closes.length === 0) return result;
   887→   887→  let cumPV = 0;
   888→   888→  let cumV = 0;
   889→   889→  for (let i = 0; i < closes.length; i++) {
   890→   890→    const tp = (highs[i] + lows[i] + closes[i]) / 3;
   891→   891→    cumPV += tp * volumes[i];
   892→   892→    cumV += volumes[i];
   893→   893→    result[i] = cumV === 0 ? 0 : cumPV / cumV;
   894→   894→  }
   895→   895→  return result;
   896→   896→}
