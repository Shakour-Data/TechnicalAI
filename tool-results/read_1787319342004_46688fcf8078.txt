   133→   133→
   134→   134→export default function VdssGraph(props: VdssGraphProps) {
   135→   135→  const { symbolName, currentPrice, resistances, supports, ma100, scenarios, rsi, adx, bullScore, edgeWeights, currencyUnit, overallSignal } = props;
   136→   136→
   137→   137→  // ── Dynamic scenario labels (context-aware) ──────────────────
   138→   138→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal || 'neutral'), [overallSignal]);
   139→   139→
   140→   140→  // ── Dynamic NODE_DEFS: R1-R5 titles/descriptions adapt to signal ──
   141→   141→  const nodeDefs = useMemo(() => ({
   142→   142→    ...NODE_DEFS,
   143→   143→    R1: { ...NODE_DEFS.R1, title: dynamicMeta.R1.nodeTitle, desc: dynamicMeta.R1.nodeDesc },
   144→   144→    R2: { ...NODE_DEFS.R2, title: dynamicMeta.R2.nodeTitle, desc: dynamicMeta.R2.nodeDesc },
   145→   145→    R3: { ...NODE_DEFS.R3, title: dynamicMeta.R3.nodeTitle, desc: dynamicMeta.R3.nodeDesc },
   146→   146→    R4: { ...NODE_DEFS.R4, title: dynamicMeta.R4.nodeTitle, desc: dynamicMeta.R4.nodeDesc },
   147→   147→    R5: { ...NODE_DEFS.R5, title: dynamicMeta.R5.nodeTitle, desc: dynamicMeta.R5.nodeDesc },
   148→   148→  }), [dynamicMeta]);
   149→   149→  const unit = currencyUnit || 'ریال';
   150→   150→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   151→   151→  const S1_level = supports[0] ?? currentPrice * 0.95;
   152→   152→
   153→   153→  const [activeFilter, setActiveFilter] = useState('all');
   154→   154→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   155→   155→  const [selectedPathIdx, setSelectedPathIdx] = useState<number | null>(null);
   156→   156→  const graphRef = useRef<HTMLDivElement>(null);
   157→   157→  const svgRef = useRef<SVGSVGElement>(null);
   158→   158→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   159→   159→  const shellRef = useRef<HTMLDivElement>(null);
   160→   160→
   161→   161→  // ── Edge probabilities (using TA engine's bullScore) ──────────────
   162→   162→  const edgeProbs = useMemo(() =>
   163→   163→    edgeWeights ? calcEdgeProbabilities(edgeWeights) : {},
   164→   164→  [edgeWeights]
   165→   165→  );
   166→   166→
   167→   167→  // ── All paths with raw probabilities ─────────────────────────────
   168→   168→  const allPathsRaw = useMemo(() => {
   169→   169→    if (Object.keys(edgeProbs).length === 0) return [];
   170→   170→    return findAllPaths(edgeProbs);
   171→   171→  }, [edgeProbs]);
   172→   172→
   173→   173→  // ── Calibrate path probabilities to match TA engine scenario probabilities ──
   174→   174→  const allPaths = useMemo(() => {
   175→   175→    // Compute raw sum per target
   176→   176→    const rawSums: Record<string, number> = {};
   177→   177→    for (const key of SCENARIO_KEYS) rawSums[key] = 0;
   178→   178→    for (const p of allPathsRaw) rawSums[p.target] += p.prob;
   179→   179→
   180→   180→    // Scale each path so the sum per target matches scenario probability
   181→   181→    return allPathsRaw.map(p => {
   182→   182→      const rawSum = rawSums[p.target] || 0.0001;
   183→   183→      const targetProb = scenarios[p.target as keyof typeof scenarios].probability / 100;
   184→   184→      const scale = targetProb / rawSum;
   185→   185→      return { ...p, prob: p.prob * scale };
   186→   186→    });
   187→   187→  }, [allPathsRaw, scenarios]);
   188→   188→
   189→   189→  // ── Filtered paths (sorted by probability descending) ─────────────────
   190→   190→  const filteredPaths = useMemo(() => {
   191→   191→    if (activeFilter === 'all') return [...allPaths].sort((a, b) => b.prob - a.prob);
   192→   192→    // Check if filter is a scenario key
   193→   193→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   194→   194→      return allPaths.filter(p => p.target === activeFilter).sort((a, b) => b.prob - a.prob);
   195→   195→    }
   196→   196→    // Type filter: paths must use only edges of this type
   197→   197→    return allPaths.filter(p =>
   198→   198→      p.edges.every(ei => EDGES[ei][3] === activeFilter)
   199→   199→    ).sort((a, b) => b.prob - a.prob);
   200→   200→  }, [allPaths, activeFilter]);
   201→   201→
   202→   202→  // ── Visible edges/nodes based on filter (or selected path) ──────────────
   203→   203→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   204→   204→    // If a specific path is selected, show only that path's nodes/edges
   205→   205→    if (selectedPathIdx !== null && filteredPaths[selectedPathIdx]) {
   206→   206→      const p = filteredPaths[selectedPathIdx];
   207→   207→      return {
   208→   208→        visibleEdgeIndices: p.edges,
   209→   209→        visibleNodes: new Set([...p.nodes, 'A']),
   210→   210→      };
   211→   211→    }
   212→   212→    if (activeFilter === 'all') {
   213→   213→      const allEdgeIdx = EDGES.map((_, i) => i);
   214→   214→      const allNodes = new Set(Object.keys(NODE_POSITIONS));
   215→   215→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   216→   216→    }
   217→   217→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   218→   218→      // Show all edges/nodes on paths to this scenario
   219→   219→      const edgeSet = new Set<number>();
   220→   220→      const nodeSet = new Set<string>();
   221→   221→      for (const p of filteredPaths) {
   222→   222→        p.edges.forEach(ei => edgeSet.add(ei));
   223→   223→        p.nodes.forEach(n => nodeSet.add(n));
   224→   224→      }
   225→   225→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   226→   226→    }
   227→   227→    // Type filter
   228→   228→    const edgeSet = new Set<number>();
   229→   229→    const nodeSet = new Set<string>();
   230→   230→    EDGES.forEach((e, i) => {
   231→   231→      if (e[3] === activeFilter) {
   232→   232→        edgeSet.add(i);
   233→   233→        nodeSet.add(e[0]);
   234→   234→        nodeSet.add(e[1]);
   235→   235→      }
   236→   236→    });
   237→   237→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   238→   238→  }, [activeFilter, filteredPaths, selectedPathIdx]);
   239→   239→
   240→   240→  // ── Node values ──────────────────────────────────────────────────
   241→   241→  const nodeValues: Record<string, string> = {
   242→   242→    A: toFa(currentPrice),
   243→   243→    B: toFa(R1_level),
   244→   244→    D: toFa(resistances[1] ?? currentPrice * 1.10),
   245→   245→    E: `${toFa(Math.round(R1_level + (resistances[1] ?? R1_level * 1.05 - R1_level) * 0.5))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 0.8))}`,
   246→   246→    F: `${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 1.2))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 2.0))}`,
   247→   247→    G: toFa(S1_level),
   248→   248→    H: toFa(supports[1] ?? currentPrice * 0.90),
   249→   249→    I: toFa(supports[2] ?? currentPrice * 0.85),
   250→   250→    J: toFa(supports[3] ?? currentPrice * 0.80),
   251→   251→    K: toFa(ma100),
   252→   252→    L: props.trendDirection === 'up' ? 'صعودی' : props.trendDirection === 'down' ? 'نزولی' : 'خنثی',
   253→   253→  };
   254→   254→  for (const key of SCENARIO_KEYS) {
   255→   255→    const s = scenarios[key];
   256→   256→    nodeValues[key] = `${toFa(s.targetMin)} — ${toFa(s.targetMax)} ${unit}`;
   257→   257→  }
   258→   258→
   259→   259→  // ── Draw SVG edges ───────────────────────────────────────────────
   260→   260→  const drawEdges = useCallback(() => {
   261→   261→    const svg = svgRef.current;
   262→   262→    const graph = graphRef.current;
   263→   263→    if (!svg || !graph) return;
   264→   264→
   265→   265→    let markersSvg = '';
   266→   266→    for (const [k, c] of Object.entries(EDGE_COLORS)) {
   267→   267→      markersSvg += `<marker id="arrow-${k}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   268→   268→    }
   269→   269→
   270→   270→    let pathsSvg = '';
   271→   271→    const graphRect = graph.getBoundingClientRect();
   272→   272→
   273→   273→    for (let ei = 0; ei < EDGES.length; ei++) {
   274→   274→      const [fromId, toId, label, type] = EDGES[ei];
   275→   275→      const isVisible = visibleEdgeIndices.includes(ei);
   276→   276→      const fromEl = nodeRefs.current[fromId];
   277→   277→      const toEl = nodeRefs.current[toId];
   278→   278→      if (!fromEl || !toEl) continue;
   279→   279→
   280→   280→      const fRect = fromEl.getBoundingClientRect();
   281→   281→      const tRect = toEl.getBoundingClientRect();
   282→   282→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   283→   283→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   284→   284→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   285→   285→      const by = tRect.top - graphRect.top + tRect.height / 2;
   286→   286→      const dx = bx - ax;
   287→   287→      const dy = by - ay;
   288→   288→      const dist = Math.sqrt(dx * dx + dy * dy);
   289→   289→      if (dist === 0) continue;
   290→   290→
   291→   291→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   292→   292→      const mx = (ax + bx) / 2;
   293→   293→      const my = (ay + by) / 2;
   294→   294→      const nx = -dy / dist;
   295→   295→      const ny = dx / dist;
   296→   296→      const cx = mx + nx * bend;
   297→   297→      const cy = my + ny * bend;
   298→   298→
   299→   299→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   300→   300→      const prob = edgeProbs[String(ei)] ?? 0;
   301→   301→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   302→   302→
   303→   303→      pathsSvg += `<path d="${d}" stroke="${EDGE_COLORS[type]}" stroke-width="${isVisible ? 2 : 1}" opacity="${isVisible ? 0.72 : 0.06}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   304→   304→
   305→   305→      // Edge probability label (only for visible edges)
   306→   306→      if (isVisible) {
   307→   307→        pathsSvg += `<text x="${cx}" y="${cy - 6}" fill="#374151" font-size="9" text-anchor="middle" paint-order="stroke" stroke="#f3f4f6" stroke-width="3.5" stroke-linejoin="round" opacity="0.85" data-type="${type}" class="edge-label">${label}</text>`;
   308→   308→        pathsSvg += `<text x="${cx}" y="${cy + 8}" fill="${EDGE_COLORS[type]}" font-size="10" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#f3f4f6" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   309→   309→      }
   310→   310→    }
   311→   311→
   312→   312→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   313→   313→  }, [visibleEdgeIndices, edgeProbs]);
   314→   314→
   315→   315→  useEffect(() => { drawEdges(); }, [drawEdges]);
   316→   316→  useEffect(() => {
   317→   317→    const el = shellRef.current;
   318→   318→    if (!el) return;
   319→   319→    const observer = new ResizeObserver(() => drawEdges());
   320→   320→    observer.observe(el);
   321→   321→    return () => observer.disconnect();
   322→   322→  }, [drawEdges]);
   323→   323→
   324→   324→  // ── Aggregated path probabilities per target (calibrated = matches scenario prob) ──
   325→   325→  const pathProbsByTarget = useMemo(() => {
   326→   326→    const result: Record<string, number> = {};
   327→   327→    for (const key of SCENARIO_KEYS) {
   328→   328→      result[key] = 0;
   329→   329→      for (const p of filteredPaths) {
   330→   330→        if (p.target === key) result[key] = (result[key] ?? 0) + p.prob;
   331→   331→      }
