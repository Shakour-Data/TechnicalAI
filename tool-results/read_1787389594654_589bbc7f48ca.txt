   142→  } = props;
   143→
   144→  const unit = currencyUnit || 'ریال';
   145→
   146→  // ── Instrument Context (for intelligent naming) ─────────────────

   149→    if (isTgju) return { name: symbolName, type: 'tgju' as const, verb: 'قرار دارد' };

   151→  }, [symbolName, isIndex, isTgju]);
   152→
   153→  const IL = instrumentLabel;
   154→
   155→  const containerRef = useRef<HTMLDivElement>(null);
   156→  const textAnalysisRef = useRef<HTMLDivElement>(null);

   158→
   159→  // ── AI Analysis state ───────────────────────────────────────────
   160→  const [aiAnalysis, setAiAnalysis] = useState<string | null>(null);
   161→  const [aiLoading, setAiLoading] = useState(false);
   162→  const [aiError, setAiError] = useState<string | null>(null);

   165→  // ── Analysis version selector (V7 locked / V8 / V9 locked / V10 active) ─────────
   166→  const [analysisVersion, setAnalysisVersion] = useState<7 | 8 | 9 | 10>(10);
   167→  // ── V10 Debug data ──────────────────────────────────────────────
   168→  const [v10Debug, setV10Debug] = useState<any>(null);

   170→  // ── V10 Scenarios (built from TA data + S/R levels) ───────────────

   175→      atr,
   176→      overallSignal: overallSignal as 'bullish' | 'bearish' | 'neutral',

   193→    setAiError(null);
   194→    try {

   197→      const activeScenarios = analysisVersion === 10 && v10Result
   198→        ? v10Result.scenarios

   201→          symbolName, currentPrice, ma21, ma100,
   202→          rsi, mfi, cci, adx, stochK, stochD,
   203→          macdLine, macdSignal, macdHist,
   204→          diPlus, diMinus, sar, atr, obv, hasVolume,
   205→          bollingerUpper, bollingerMiddle, bollingerLower,
   206→          trendDirection, trendAngle, trendR2, overallSignal,

   209→          isIndex, isTgju, currencyUnit,
   210→        };

   212→      if (analysisVersion === 10 && v10Result) {
   213→        requestBody.v10DominantKey = v10Result.dominantKey;
   214→        requestBody.v10HighestCumulativeKey = v10Result.highestCumulativeKey;
   215→      }

   217→      if ((analysisVersion === 9 || analysisVersion === 10) && candles.length > 0) {

   219→          date: c.date, open: c.open, high: c.high,
   220→          low: c.low, close: c.close, volume: c.volume,
   221→        }));
   222→      }

   225→      const controller = new AbortController();
   226→      const timeoutId = setTimeout(() => controller.abort(), 90000);

   228→      const res = await fetch(apiUrl, {
   229→        method: 'POST',
   230→        headers: { 'Content-Type': 'application/json' },
   231→        body: JSON.stringify(requestBody),
   232→        signal: controller.signal,
   233→      });
   234→      clearTimeout(timeoutId);
   235→
   236→      if (!res.ok) {
   237→        let errorMsg = `خطای سرور (${res.status})`;

   239→          const errData = await res.json();
   240→          errorMsg = errData.error || errorMsg;

   243→          if (res.status === 502) errorMsg = 'سرور در حال بارگذاری مجدد است. لطفاً چند ثانیه صبر کنید و دوباره تلاش کنید.';
   244→        }
   245→        throw new Error(errorMsg);
   246→      }
   247→      const data = await res.json();
   248→      if (fetchId === aiFetchIdRef.current) {
   249→        setAiAnalysis(data.analysis);
   250→        if (data._debug?.v10) setV10Debug({ ...data._debug.v10, v10Persona: data._debug.v10Persona, v10Template: data._debug.v10Template });
   251→        else setV10Debug(null);
   252→      }
   253→    } catch (err) {
   254→      if (fetchId === aiFetchIdRef.current) {
   255→        if (err instanceof Error && err.name === 'AbortError') {
   256→          setAiError('زمان پاسخگویی به پایان رسید. لطفاً دوباره تلاش کنید.');

   259→        }
   260→      }
   261→    } finally {
   262→      if (fetchId === aiFetchIdRef.current) {
   263→        setAiLoading(false);
   264→      }
   265→    }

   274→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   275→  const R2_level = resistances[1] ?? currentPrice * 1.10;

   277→  const S2_level = supports[1] ?? currentPrice * 0.90;
   278→  const S3_level = supports[2] ?? currentPrice * 0.85;
   279→
   280→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   281→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   282→

   286→  for (const key of SCENARIO_KEYS) {
   287→    if (scenarios[key].probability > highestProb) {
   288→      highestProb = scenarios[key].probability;
   289→      highestKey = key;
   290→    }
   291→  }
   292→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   293→
   294→  // ── V10 dominant + highest cumulative + total ────────────
   295→  const v10DominantKey = v10Result?.dominantKey || 'S5';
   296→  const v10HighestCumKey = v10Result?.highestCumulativeKey || 'S5';
   297→  const v10TotalProb = v10Result
   298→    ? V10_KEYS.reduce((sum, k) => sum + (v10Result.scenarios[k]?.probability || 0), 0)

   300→

   302→  const sortedScenarios = useMemo(() => {

   304→      .map(k => ({ key: k, ...scenarios[k] }))

   325→  // ── Stochastic signal ──────────────────────────────────────────
   326→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';

   328→  // ── Trend text ─────────────────────────────────────────────────
   329→  const isUp = trendDirection === 'up';
   330→  const isDown = trendDirection === 'down';
   331→  const trendLabel = isUp ? 'صعودی' : isDown ? 'نزولی' : 'خنثی';

   337→  // ── Bollinger position ─────────────────────────────────────────

   340→  const bbAbove = currentPrice > bollingerUpper;
   341→  const bbBelow = currentPrice < bollingerLower;
   342→  const bbSignal = bbAbove ? 'بالای باند بالایی' : bbBelow ? 'زیر باند پایینی' : `داخل باندها (${bbPos}٪)`;

   344→  // ── Distance to MA ─────────────────────────────────────────────
   345→  const distMa21 = ((currentPrice - ma21) / ma21 * 100).toFixed(1);
   346→  const distMa100 = ((currentPrice - ma100) / ma100 * 100).toFixed(1);

   349→  const chartCandles = useMemo(() => {

   351→    return candles.slice(-2000);
   352→  }, [candles]);

   354→  const tvScenarios = useMemo(() => {

   356→      R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62',

   931→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   932→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   933→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   934→  grade?: string; overlapCount?: number; touchCount?: number;
   935→}) {
   936→  const isSupport = type === 'support';
   937→  const isTargetType = type === 'target';
   938→
   939→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   940→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   941→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   942→
   943→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   944→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   945→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   946→
   947→  return (
   948→    <div
   949→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   950→      style={{
   951→        background: bgAccent,
   952→        border: `1px solid ${borderAccent}`,
   953→        borderRight: `3px solid ${accentColor}`,
   954→      }}
   955→    >
   956→      {/* Label + Price */}
   957→      <div className="flex-1 min-w-0">
   958→        <div className="flex items-center gap-2">
   959→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   960→          {isTarget && <span className="text-[9px]">🎯</span>}
   961→          {overlapCount !== undefined && overlapCount > 0 && (
   962→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   963→              {toPersianDigits(overlapCount)} هم‌پوشانی
   964→            </span>
   965→          )}
   966→        </div>
   967→        <div className="flex items-baseline gap-1">
   968→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   969→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   970→        </div>
   971→      </div>
   972→
   973→      {/* Strength + Grade */}
   974→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   975→        <div className="flex items-center gap-1.5">
   976→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   977→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   978→          <span className="text-[9px] text-gray-500">/۱۰</span>
   979→        </div>
   980→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   981→          <div
   982→            className="h-full rounded-full transition-all duration-500"
   983→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   984→          />
   985→        </div>
   986→        {touchCount !== undefined && touchCount > 0 && (
   987→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   988→        )}
   989→      </div>
   990→    </div>
   991→  );
   992→}
