   458→   458→/* ─── In-memory cache ────────────────────────────────── */
   459→   459→
   460→   460→let quotesCache: YahooQuote[] | null = null;
   461→   461→let quotesCacheTime = 0;
   462→   462→const CACHE_TTL = 120_000; // 2 minutes
   463→   463→
   464→   464→/* ─── Fetch live quotes ──────────────────────────────── */
   465→   465→
   466→   466→export async function fetchYahooQuotes(): Promise<YahooQuote[]> {
   467→   467→  const now = Date.now();
   468→   468→  if (quotesCache && now - quotesCacheTime < CACHE_TTL) {
   469→   469→    return quotesCache;
   470→   470→  }
   471→   471→
   472→   472→  const symbols = ALL_YAHOO_INSTRUMENTS.map((s) => s.symbol);
   473→   473→  const quotes: YahooQuote[] = [];
   474→   474→
   475→   475→  // Fetch in batches of 20 (Yahoo API limit)
   476→   476→  const BATCH_SIZE = 20;
   477→   477→  for (let i = 0; i < symbols.length; i += BATCH_SIZE) {
   478→   478→    const batch = symbols.slice(i, i + BATCH_SIZE);
   479→   479→    try {
   480→   480→      const results = await yahooFinance.quote(batch, {
   481→   481→        fields: [
   482→   482→          'symbol', 'regularMarketPrice', 'regularMarketChange', 'regularMarketChangePercent',
   483→   483→          'regularMarketPreviousClose', 'regularMarketOpen', 'regularMarketDayHigh',
   484→   484→          'regularMarketDayLow', 'regularMarketVolume', 'marketCap', 'currency',
   485→   485→          'shortName', 'longName',
   486→   486→        ],
   487→   487→      });
   488→   488→
   489→   489→      for (const r of results) {
   490→   490→        const def = ALL_YAHOO_INSTRUMENTS.find((s) => s.symbol === r.symbol);
   491→   491→        if (!def) continue;
   492→   492→        const unit = r.currency || 'USD';
   493→   493→        quotes.push({
   494→   494→          symbol: r.symbol,
   495→   495→          name: def.name,
   496→   496→          nameEn: def.nameEn,
   497→   497→          country: def.country,
   498→   498→          countryEn: def.countryEn,
   499→   499→          exchange: def.exchange,
   500→   500→          groupTitle: def.groupTitle,
   501→   501→          category: def.category,
   502→   502→          unit,
   503→   503→          price: r.regularMarketPrice ?? 0,
   504→   504→          change: r.regularMarketChange ?? 0,
   505→   505→          changePercent: r.regularMarketChangePercent ?? 0,
   506→   506→          previousClose: r.regularMarketPreviousClose ?? 0,
   507→   507→          open: r.regularMarketOpen ?? 0,
   508→   508→          high: r.regularMarketDayHigh ?? 0,
   509→   509→          low: r.regularMarketDayLow ?? 0,
   510→   510→          volume: r.regularMarketVolume ?? 0,
   511→   511→          marketCap: r.marketCap ?? 0,
   512→   512→          currency: r.currency || 'USD',
   513→   513→        });
   514→   514→      }
   515→   515→    } catch (err) {
   516→   516→      const errMsg = err instanceof Error ? err.message : String(err);
   517→   517→      if (errMsg.includes('FailedYahooValidationError') || errMsg.includes('validation')) {
   518→   518→        // Batch failed due to schema validation — retry individually to salvage valid symbols
   519→   519→        console.warn(`[Yahoo Finance] Batch ${Math.floor(i / BATCH_SIZE) + 1} validation error, retrying individually...`);
   520→   520→        for (const symbol of batch) {
   521→   521→          try {
   522→   522→            const singleResults = await yahooFinance.quote([symbol], {
   523→   523→              fields: [
   524→   524→                'symbol', 'regularMarketPrice', 'regularMarketChange', 'regularMarketChangePercent',
   525→   525→                'regularMarketPreviousClose', 'regularMarketOpen', 'regularMarketDayHigh',
   526→   526→                'regularMarketDayLow', 'regularMarketVolume', 'marketCap', 'currency',
   527→   527→              ],
   528→   528→            });
   529→   529→            for (const r of singleResults) {
   530→   530→              const def = ALL_YAHOO_INSTRUMENTS.find((s) => s.symbol === r.symbol);
   531→   531→              if (!def) continue;
   532→   532→              const unit = r.currency || 'USD';
   533→   533→              quotes.push({
   534→   534→                symbol: r.symbol, name: def.name, nameEn: def.nameEn,
   535→   535→                country: def.country, countryEn: def.countryEn,
   536→   536→                exchange: def.exchange, groupTitle: def.groupTitle, category: def.category,
   537→   537→                unit, price: r.regularMarketPrice ?? 0, change: r.regularMarketChange ?? 0,
   538→   538→                changePercent: r.regularMarketChangePercent ?? 0, previousClose: r.regularMarketPreviousClose ?? 0,
   539→   539→                open: r.regularMarketOpen ?? 0, high: r.regularMarketDayHigh ?? 0,
   540→   540→                low: r.regularMarketDayLow ?? 0, volume: r.regularMarketVolume ?? 0,
   541→   541→                marketCap: r.marketCap ?? 0, currency: r.currency || 'USD',
   542→   542→              });
   543→   543→            }
   544→   544→          } catch {
   545→   545→            // Skip individual symbol errors
   546→   546→          }
   547→   547→        }
   548→   548→      } else {
   549→   549→        console.error(`[Yahoo Finance] Error fetching batch ${Math.floor(i / BATCH_SIZE) + 1}:`, err);
   550→   550→      }
   551→   551→    }
   552→   552→  }
   553→   553→
   554→   554→  quotesCache = quotes;
   555→   555→  quotesCacheTime = now;
   556→   556→  return quotes;
   557→   557→}

   559→   559→/* ─── Fetch historical data ──────────────────────────── */
   560→   560→
   561→   561→export async function fetchYahooHistory(symbol: string, days: number = 365): Promise<YahooOHLC[]> {
   562→   562→  const end = new Date();
   563→   563→  const start = new Date();
   564→   564→  start.setDate(end.getDate() - days);
   565→   565→
   566→   566→  const result = await yahooFinance.chart(symbol, {
   567→   567→    period1: start,
   568→   568→    period2: end,
   569→   569→    interval: '1d',
   570→   570→  });
   571→   571→
   572→   572→  if (!result.quotes || result.quotes.length === 0) {
   573→   573→    return [];
   574→   574→  }
   575→   575→
   576→   576→  return result.quotes
   577→   577→    .filter((q) => q.close != null && q.open != null)
   578→   578→    .map((q) => ({
   579→   579→      date: q.date?.toISOString().split('T')[0] || '',
   580→   580→      open: q.open!,
   581→   581→      high: q.high || q.close!,
   582→   582→      low: q.low || q.close!,
   583→   583→      close: q.close!,
   584→   584→      volume: q.volume || 0,
   585→   585→    }));
   586→   586→}
