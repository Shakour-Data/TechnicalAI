   248→   248→    // TGJU instrument: fetch historical data via tgju.org chart API
   249→   249→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   250→   250→      setLoading(true);

   269→   269→        setError(String(err));

   271→   271→        setLoading(false);
   272→   272→        setLoadingMessage(null);
   273→   273→      }
   274→   274→      return;
   275→   275→    }
   276→   276→
   277→   277→    // Index: fetch historical data via TSETMC CDN (webId), then finpy-tse (finpyIndex/finpySector)
   278→   278→    if (category === 'index') {
   279→   279→      setLoading(true);
   280→   280→      setLoadingMessage('در حال دریافت داده‌های شاخص از TSETMC ... (حدود ۳۰ ثانیه برای بار اول)');
