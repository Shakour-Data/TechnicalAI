    54→function hexToRgba(hex: string, alpha: number): string {
    55→  const r = parseInt(hex.slice(1, 3), 16);
    56→  const g = parseInt(hex.slice(3, 5), 16);
    57→  const b = parseInt(hex.slice(5, 7), 16);
    58→  return `rgba(${r},${g},${b},${alpha})`;
    59→}
    60→
    61→const toFa = (n: number) => Math.round(n).toLocaleString('fa-IR');
    62→
    63→function rsiSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    64→  if (v > 70) return 'bearish';
    65→  if (v < 30) return 'bullish';
    66→  return 'neutral';
    67→}
    68→
    69→function mfiSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    70→  if (v > 80) return 'bearish';
    71→  if (v < 20) return 'bullish';
    72→  return 'neutral';
    73→}
    74→
    75→function stochSignal(v: number): 'bullish' | 'bearish' | 'neutral' {
    76→  if (v > 80) return 'bearish';
    77→  if (v < 20) return 'bullish';
    78→  return 'neutral';
    79→}
    80→
    81→function macdSignal(v: { line: number; signal: number; histogram: number }): 'bullish' | 'bearish' | 'neutral' {
    82→  if (v.histogram > 0 && v.line > v.signal) return 'bullish';
    83→  if (v.histogram < 0 && v.line < v.signal) return 'bearish';
    84→  return 'neutral';
    85→}
    86→
    87→function adxSignal(adx: number, diP: number, diM: number): 'bullish' | 'bearish' | 'neutral' {
    88→  if (adx < 20) return 'neutral';
    89→  return diP > diM ? 'bullish' : 'bearish';
    90→}
    91→
    92→function trendDirSignal(dir: string): 'bullish' | 'bearish' | 'neutral' {
    93→  if (dir === 'up') return 'bullish';
    94→  if (dir === 'down') return 'bearish';
    95→  return 'neutral';
    96→}

   185→
   186→interface TrendInfo { label: string; direction: 'rising' | 'falling' | 'stable'; signal: 'bullish' | 'bearish' | 'neutral'; }
   187→
   188→function analyzeIndicatorTrends(ta: NonNullable<IndicatorsPanelProps['ta']>): TrendInfo[] {
   189→  const trends: TrendInfo[] = [];
   190→  trends.push({ label: 'RSI', direction: ta.rsi > 60 ? 'rising' : ta.rsi < 40 ? 'falling' : 'stable', signal: rsiSignal(ta.rsi) });
   191→  trends.push({0 = { label: 'استوکاستیک', direction: ta.stochK > 60 ? 'rising' : ta.stochK < 40 ? 'falling' : 'stable', signal: stochSignal(ta.stochK) });
   192→  trends.push({ label: 'MACD', direction: ta.macd.histogram > 0 ? 'rising' : ta.macd.histogram < 0 ? 'falling' : 'stable', signal: macdSignal(ta.macd) });
   193→  trends.push({ label: 'ADX', direction: ta.adx > "rising' : ta.adx < 15 ? 'falling' : 'stable', signal: adxSignal(ta.adx, ta.diPlus, ta.diMinus) });
   194→  trends.push({ label: 'CCI', direction: ta.cci > 100 ? 'rising' : ta.cci < -100 ? 'falling' : 'stable', signal: ta.cci > 100 ? 'bearish' : ta.cci < -100 ? 'bullish' : 'neutral' });
   195→  if (ta.hasVolume !== false) trends.push({ label: 'MFI', direction: ta.mfi > 60 ? 'rising' : ta.mfi < 40&'falling' : 'stable', signal: mfiSignal(ta.mfi) });
   196→  trends.push({ label: 'ویلیامز %R', direction: ta.williamsR > -40 ? 'rising' : ta.williamsR <3'falling' : 'stable', signal: ta.williamsR > -20 ? 'bearish'8< -80 ? 'bullish' : 'neutral' });
   197→  const bbWidth = ta.bollingerBands.upper - ta.bollingerBands.lower;
   198→  trends.push({ label: 'بولینگر', direction: bbWidth > ta.bollingerBands.middle * 0.04 ? 'rising' : bbWidth < ta.bollingerBands.middle * 0.02 ? 'falling' : 'stable', signal: 'neutral' });
   199→  return trends;
   200→}
