    26→    26→// --- Global rate limiter (serial queue) ---
    27→    27→let lastAITime = 0;
    28→    28→const AI_MIN_INTERVAL = 15_000;
    29→    29→let cooldownUntil = 0;
    30→    30→let activeCall = false;
    31→    31→const pendingQueue: Array<{
    32→    32→  resolve: (text: string) => void;

    37→    37→function processAIQueue() {
    38→    38→  if (activeCall || pendingQueue.length === 0) return;
    39→    39→  const now = Date.now();
    40→    40→
    41→    41→  if (cooldownUntil > now) {
    42→    42→    setTimeout(processAIQueue, Math.min(cooldownUntil - now, 5000));
    43→    43→    return;
    44→    44→  }
    45→    45→
    46→    46→  const elapsed = now - lastAITime;
    47→    47→  if (elapsed < AI_MIN_INTERVAL) {
    48→    48→    setTimeout(processAIQueue, AI_MIN_INTERVAL - elapsed);
    49→    49→    return;
    50→    50→  }
    51→    51→
    52→    52→  const item = pendingQueue.shift()!;
    53→    53→  activeCall = true;
    54→    54→
    55→    55→  callZaiSDK(item.messages)
    56→    56→    .then(text => {
    57→    57→      lastAITime = Date.now();
    58→    58→      cooldownUntil = 0;
    59→    59→      item.resolve(text);
    60→    60→    })
    61→    61→    .catch(err => {
    62→    62→      const msg = err.message || '';
    63→    63→      if (msg.includes('429')) {
    64→    64→        cooldownUntil = Date.now() + 60_000;
    65→    65→        console.warn('[AI] 429, cooldown 60s');
    66→    66→      }
    67→    67→      item.reject(err);
    68→    68→    })
    69→    69→    .finally(() => {
    70→    70→      activeCall = false;
    71→    71→      setTimeout(processAIQueue, 2000);
    72→    72→    });
    73→    73→}

    75→    75→function queueAI(messages: { role: string; content: string }[], timeoutMs = 90_000): Promise<string> {
    76→    76→  return new Promise((resolve, reject) => {
    77→    77→    const timer = setTimeout(() => {
    78→    78→      const idx = pendingQueue.findIndex(q => q.messages === messages);
    79→    79→      if (idx >= 0) pendingQueue.splice(idx, 1);
    80→    80→      reject(new Error('AI request timeout'));
    81→    81→    }, timeoutMs);
    82→    82→
    83→    83→    pendingQueue.push({
    84→    84→      messages,
    85→    85→      resolve: (text) => { clearTimeout(timer); resolve(text); },
    86→    86→      reject: (err) => { clearTimeout(timer); reject(err); },
    87→    87→    });
    88→    88→
    89→    89→    processAIQueue();
    90→    90→  });
    91→    91→}

    94→    94→function sleep(ms: number) { return new Promise(r => setTimeout(r, ms)); }
    95→    95→function jitter(base: number): number { return base + Math.random() * base * 0.3; }
    96→    96→
    97→    97→async function callZaiSDK(messages: { role: string; content: string }[], maxRetries = 3): Promise<string> {
    98→    98→  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    99→    99→    try {
   100→   100→      const zai = await getZai();
   101→   101→      const completion = await zai.chat.completions.create({
   102→   102→        messages,
   103→   103→        thinking: { type: 'disabled' },
   104→   104→      });
   105→   105→      const content = completion.choices[0]?.message?.content;
   106→   106→      if (!content || content.trim().length === 0) {
   107→   107→        throw new Error('Empty AI response');
   108→   108→      }
   109→   109→      // Strip emoji prefix lines
   110→   110→      const lines = content.split('\n').filter(l => !l.startsWith('\ud83d\ude80'));
   111→   111→      const text = lines.join('\n').trim();
   112→   112→      if (text.length < 10) {
   113→   113→        throw new Error('AI response too short: ' + text.slice(0, 100));
   114→   114→      }
   115→   115→      return text;
   116→   116→    } catch (err: unknown) {
   117→   117→      const msg = err instanceof Error ? err.message : String(err);
   118→   118→      if (msg.includes('429') && attempt < maxRetries) {
   119→   119→        const waitMs = Math.round(jitter(15_000 * Math.pow(2, attempt - 1))); // 15s, 30s, 60s + jitter
   120→   120→        console.warn(`[AI v5.1] 429 retry ${attempt}/${maxRetries}, waiting ${waitMs}ms...`);
   121→   121→        await sleep(waitMs);
   122→   122→        continue;
   123→   123→      }
   124→   124→      throw err;
   125→   125→    }
   126→   126→  }
   127→   127→  throw new Error('Max retries exceeded');
   128→   128→}

   137→   137→  if (strength >= 8.5) return '\u0628\u0633\u06cc\u0627\u0631 \u0642\u0648\u06cc';

   140→   140→  if (strength >= 3) return '\u0636\u0639\u06cc\u0641';
