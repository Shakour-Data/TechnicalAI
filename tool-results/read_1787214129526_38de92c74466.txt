    42→    42→function srGrade(strength: number): string {
    43→    43→  if (strength >= 8.5) return '\u0628\u0633\u06cc\u0627\u0631 \u0642\u0648\u06cc';
    44→    44→  if (strength >= 7) return '\u0642\u0648\u06cc';
    45→    45→  if (strength >= 5) return '\u0645\u062a\u0648\u0633\u0637';
    46→    46→  if (strength >= 3) return '\u0636\u0639\u06cc\u0641';

    51→    51→function buildMLInput(body: Record<string, unknown>): MLSelectorInput {
    52→    52→  const price = (body.currentPrice as number) || 0;
    53→    53→  const trendDir = (body.trendDirection as string) || 'range';
    54→    54→  const adx = (body.adx as number) || 0;
    55→    55→  const rsi = (body.rsi as number) || 50;
    56→    56→  const stochK = (body.stochK as number) || 50;
    57→    57→  const macdHist = (body.macdHist as number) || 0;
    58→    58→  const obv = (body.obv as number) || 0;
    59→    59→  const bollingerUpper = (body.bollingerUpper as number) || 0;
    60→    60→  const bollingerLower = (body.bollingerLower as number) || 0;
    61→    61→  const resistance = (body.resistanceStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
    62→    62→  const support = (body.supportStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
    63→    63→  const atr = (body.atr as number) || 0;
    64→    64→
    65→    65→  const bbRange = bollingerUpper - bollingerLower;
    66→    66→  const bbPosition = bbRange > 0 ? Math.min(100, Math.max(0, Math.round((price - bollingerLower) / bbRange * 100))) : 50;
    67→    67→
    68→    68→  const scenarios = body.scenarios as Record<string, { probability: number }> | undefined;
    69→    69→  let dominantKey = 'R3';
    70→    70→  let dominantProb = 0;
    71→    71→  for (const k of ['R1', 'R2', 'R3', 'R4', 'R5'] as const) {
    72→    72→    const p = scenarios?.[k]?.probability ?? 0;
    73→    73→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
    74→    74→  }
    75→    75→
    76→    76→  return {
    77→    77→    price,
    78→    78→    trend: trendDir === 'up' ? 'up' : trendDir === 'down' ? 'down' : 'range',
    79→    79→    adx,
    80→    80→    diPlus: (body.diPlus as number) || 0,
    81→    81→    diMinus: (body.diMinus as number) || 0,
    82→    82→    rsi,
    83→    83→    stochK,
    84→    84→    macdHist,
    85→    85→    obv,
    86→    86→    bbPosition,
    87→    87→    resistance,
    88→    88→    support,
    89→    89→    atr,
    90→    90→    scenarioDominant: dominantKey,
    91→    91→    hasVolume: (body.hasVolume as boolean) ?? false,
    92→    92→  };
    93→    93→}

   325→   325→function sleep(ms: number) { return new Promise(r => setTimeout(r, ms)); }
   326→   326→
   327→   327→function jitter(base: number): number {
   328→   328→  return base + Math.random() * base * 0.3;
   329→   329→}
   330→   330→
   331→   331→async function aiCompletionWithRetry(
   332→   332→  zai: Awaited<ReturnType<typeof getZai>>,
   333→   333→  messages: { role: string; content: string }[],
   334→   334→  maxRetries = 3,
   335→   335→): Promise<string> {
   336→   336→  for (let attempt = 1; attempt <= maxRetries; attempt++) {
   337→   337→    try {
   338→   338→      const completion = await zai.chat.completions.create({
   339→   339→        messages,
   340→   340→        thinking: { type: 'disabled' },
   341→   341→      });
   342→   342→      const content = completion.choices[0]?.message?.content;
   343→   343→      if (!content || content.trim().length === 0) {
   344→   344→        throw new Error('Empty AI response');
   345→   345→      }
   346→   346→      return content;
   347→   347→    } catch (err: unknown) {
   348→   348→      const msg = err instanceof Error ? err.message : String(err);
   349→   349→      if (msg.includes('429') && attempt < maxRetries) {
   350→   350→        const waitMs = Math.round(jitter(15000 * Math.pow(2, attempt - 1))); // 15s, 30s, 60s + jitter
   351→   351→        console.warn(`[AI Analysis v5] 429 retry ${attempt}/${maxRetries}, waiting ${waitMs}ms...`);
   352→   352→        await sleep(waitMs);
   353→   353→        continue;
   354→   354→      }
   355→   355→      throw err;
   356→   356→    }
   357→   357→  }
   358→   358→  throw new Error('Max retries exceeded');
   359→   359→}
