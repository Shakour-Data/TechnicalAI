     1→/**
     2→ * @module decision-graph
     3→ *
     4→ * Decision Graph Probability Engine — Technical Analysis System
     5→ * =============================================================================
     6→ *
     7→ * This module is the **primary probability engine** for the technical analysis system.
     8→ * All 9 scenario probabilities (SC1–SC9) are derived by aggregating path
     9→ * probabilities through a **directed acyclic graph (DAG)**.
    10→ *
    11→ * ## DAG Structure Overview
    12→ *
    13→ * The graph contains **34 nodes** across 5 layers and **55+ edges**:
    14→ *
    15→ * | Layer | Description                          | Nodes                         |
    16→ * |-------|--------------------------------------|-------------------------------|
    17→ * | 0     | Root decision point                  | `ROOT`                        |
    18→ * | 1     | Three main strategy branches         | `N_TREND`, `N_BREAK`, `N_REVERSAL` |
    19→ * | 2     | Directional sub-branches (9 nodes)  | Bull/Bear/Flat per branch     |
    20→ * | 3     | Assessment/event nodes (19 nodes)   | Momentum, volume, divergence… |
    21→ * | 4     | Terminal scenario nodes (9 nodes)    | `SC1`–`SC9`                   |
    22→ *
    23→ * ## Three Strategy Branches
    24→ *
    25→ * 1. **Trend Following** (`branch: 'trend'`) — Driven by ADX trend strength,
    26→ *    bull consensus, and momentum indicators. Sub-branches: Bullish, Bearish, Flat.
    27→ *
    28→ * 2. **Breakout** (`branch: 'breakout'`) — Driven by support/resistance proximity,
    29→ *    volume confirmation, and price distance to S/R levels. Sub-branches:
    30→ *    Bullish Breakout, Bearish Breakout, No Breakout.
    31→ *
    32→ * 3. **Reversal** (`branch: 'reversal'`) — Driven by oscillator extremes,
    33→ *    divergence proxies, and overbought/oversold conditions. Sub-branches:
    34→ *    Bullish Reversal, Bearish Reversal, No Reversal.
    35→ *
    36→ * ## Scenario Terminal Nodes (SC1–SC9)
    37→ *
    38→ * Ordered from most bearish to most bullish:
    39→ * - **SC1** — Bearish Shock (sudden severe drop)
    40→ * - **SC2** — Accelerating Bearish (strong downtrend momentum)
    41→ * - **SC3** — Strong Bearish (sustained downtrend)
    42→ * - **SC4** — Weak Bearish (mild downtrend, possible reversal)
    43→ * - **SC5** — Range-bound (no clear direction)
    44→ * - **SC6** — Weak Bullish (mild uptrend, correction risk)
    45→ * - **SC7** — Strong Bullish (sustained uptrend)
    46→ * - **SC8** — Accelerating Bullish (strong uptrend momentum)
    47→ * - **SC9** — Bullish Shock (sudden severe rally)
    48→ *
    49→ * ## The IRON LAW: `enforceSumTo100()`
    50→ *
    51→ * All 9 scenario probabilities are converted to **integers that sum to EXACTLY 100**
    52→ * using the **Largest Remainder Method (LRM)** with iterative bound enforcement.
    53→ * This guarantees `sum(SC1..SC9) === 100` and `2 ≤ SCi ≤ 35` for all i.
    54→ * **Never bypass this function** when producing integer scenario percentages.

    71→ *

    82→// === Helper Functions =========================================================
    83→
    84→/**
    85→ * Returns `v` if it is a finite number, otherwise returns `fallback`.
    86→ * Guards against `NaN`, `Infinity`, `-Infinity`, and non-number values
    87→ * (e.g. `undefined` from missing API fields).
    88→ *
    89→ * @param v        - The value to validate
    90→ * @param fallback - Default returned when `v` is not a finite number (default: `0`)
    91→ * @returns `v` if finite, otherwise `fallback`
    92→ */
    93→function safeNum(v: number, fallback: number = 0): number {
    94→  return (typeof v === 'number' && isFinite(v)) ? v : fallback;
    95→}
    96→
    97→/**
    98→ * Clamps a value to the range `[min, max]`, after first sanitizing it with
    99→ * `safeNum` (fallback is the midpoint of the range).
   100→ *
   101→ * @param v   - Value to clamp
   102→ * @param min - Lower bound (inclusive)
   103→ * @param max - Upper bound (inclusive)
   104→ * @returns `v` clamped to `[min, max]`
   105→ */
   106→function clamp(v: number, min: number, max: number): number {
   107→  return Math.max(min, Math.min(max, safeNum(v, (min + max) / 2)));
   108→}
   109→
   110→/**
   111→ * Normalizes an array of numbers so they sum to exactly `1.0`.
   112→ * Each element is first sanitized via `safeNum`. If the total sum is
   113→ * zero or negative, returns a uniform distribution (`1/n` each).
   114→ *
   115→ * @param arr - Array of raw (possibly unsafe) weights
   116→ * @returns New array of the same length summing to 1.0
   117→ */
   118→function normalize(arr: number[]): number[] {
   119→  const safe = arr.map(v => safeNum(v, 0));
   120→  const sum = safe.reduce((a, b) => a + b, 0);
   121→  if (sum <= 0) return arr.map(() => 1 / arr.length);
   122→  return safe.map((v) => v / sum);
   123→}
   124→
   125→/**
   126→ * Logistic sigmoid function: `1 / (1 + e^(-x))`.
   127→ * Values beyond `|x| > 500` are clamped to `0` or `1` to avoid `Infinity`.
   128→ * Input is sanitized via `safeNum` before computation.
   129→ *
   130→ * @param x - Input value (raw signal)
   131→ * @returns Value in `(0, 1)` — probability-like squashing of `x`
   132→ */
   133→function sigmoid(x: number): number {
   134→  const sx = safeNum(x, 0);
   135→  if (Math.abs(sx) > 500) return sx > 0 ? 1 : 0; // prevent Infinity
   136→  return 1 / (1 + Math.exp(-sx));
   137→}

   139→/**
   140→ * IRON LAW: Round proportional floats to integers that sum to EXACTLY 100,
   141→ * with each value constrained to `[minVal, maxVal]`.
   142→ *
   143→ * This is the **single source of truth** for converting raw scenario probability
   144→ * fractions into display-ready integer percentages. Bypassing this function
   145→ * when producing scenario percentages is a critical error.
   146→ *
   147→ * ## Algorithm: Largest Remainder Method (LRM) + Iterative Bound Enforcement
   148→ *
   149→ * 1. **Normalize** — Scale `rawFloats` so they sum to exactly 100.
   150→ * 2. **LRM round** — Floor each value, then distribute the deficit (remainder)
   151→ *    to the entries with the largest fractional parts. This minimizes total
   152→ *    rounding error and is provably fair (used in parliamentary seat allocation).
   153→ * 3. **Iterative bound enforcement** — While any value violates `[minVal, maxVal]`,
   154→ *    transfer 1 from the worst violator to a suitable partner. Each transfer
   155→ *    strictly reduces total violation magnitude, guaranteeing convergence.
   156→ * 4. **Final safety net** — If the sum drifted from 100 (should not happen
   157→ *    under normal conditions), adjust the entry with the most room.
   158→ *
   159→ * ## Guarantees
   160→ *
   161→ * - `sum(result) === 100` — **always**, regardless of input
   162→ * - `minVal <= result[i] <= maxVal` for all `i`, as long as
   163→ *   `n * minVal <= 100 <= n * maxVal`
   164→ *
   165→ * @param rawFloats - Array of non-negative floats (proportional weights)
   166→ * @param minVal    - Minimum allowed value per entry (e.g. `2` ensures no scenario is zero)
   167→ * @param maxVal    - Maximum allowed value per entry (e.g. `35` prevents one scenario from dominating)
   168→ * @returns Integer array of same length, summing to exactly 100, each within `[minVal, maxVal]`
   169→ *
   170→ * @example
   171→ * ```ts
   172→ * enforceSumTo100([33.3, 33.3, 33.4], 2, 35);  // → [33, 33, 34]
   173→ * enforceSumTo100([60, 20, 10, 5, 3, 1, 0.5, 0.3, 0.2], 2, 35);  // → [35, 20, 10, 5, 3, 2, 2, 2, 2]
   174→ * ```
   175→ *
   176→ * ⚠️  NEVER bypass this function when producing integer scenario percentages.
   177→ */
   178→function enforceSumTo100(
   179→  rawFloats: number[],
   180→  minVal: number,
   181→  maxVal: number
   182→): number[] {
   183→  const n = rawFloats.length;
   184→
   185→  // Normalize floats to sum exactly to 100 first
   186→  const floatSum = rawFloats.reduce((a, b) => a + b, 0);
   187→  const normalized =
   188→    floatSum > 0
   189→      ? rawFloats.map((v) => (v / floatSum) * 100)
   190→      : rawFloats.map(() => 100 / n);
   191→
   192→  // Step 1: LRM — floor all, distribute deficit by largest fractional part
   193→  const result = normalized.map((v) => Math.max(Math.floor(v), 0));
   194→  let deficit = 100 - result.reduce((a, b) => a + b, 0);
   195→
   196→  const byFrac = Array.from({ length: n }, (_, i) => ({
   197→    idx: i,
   198→    frac: normalized[i] - Math.floor(normalized[i]),
   199→  })).sort((a, b) => b.frac - a.frac);
   200→
   201→  // Give +1 to top `deficit` entries (LRM)
   202→  for (let d = 0; d < deficit && d < n; d++) {
   203→    result[byFrac[d].idx]++;
   204→  }
   205→  // Now result sums to exactly 100
   206→
   207→  // Step 2: Iteratively enforce [minVal, maxVal] bounds while preserving sum = 100
   208→  // Each iteration: move 1 from a violator to a suitable partner.
   209→  // Converges because each move strictly reduces the total violation magnitude.
   210→  let changed = true;
   211→  let safety = 0;
   212→  while (changed && safety < 100) {
   213→    changed = false;
   214→    safety++;
   215→
   216→    // Enforce minimum: raise any value below minVal by taking from the largest above minVal
   217→    for (let i = 0; i < n; i++) {
   218→      while (result[i] < minVal) {
   219→        let donor = -1;
   220→        for (let j = 0; j < n; j++) {
   221→          if (j !== i && result[j] > minVal && (donor === -1 || result[j] > result[donor])) {
   222→            donor = j;
   223→          }
   224→        }
   225→        if (donor === -1) break; // no donor available — shouldn't happen if n*minVal <= 100
   226→        result[donor]--;
   227→        result[i]++;
   228→        changed = true;
   229→      }
   230→    }
   231→
   232→    // Enforce maximum: lower any value above maxVal by giving to the smallest below maxVal
   233→    for (let i = 0; i < n; i++) {
   234→      while (result[i] > maxVal) {
   235→        let receiver = -1;
   236→        for (let j = 0; j < n; j++) {
   237→          if (j !== i && result[j] < maxVal && (receiver === -1 || result[j] < result[receiver])) {
   238→            receiver = j;
   239→          }
   240→        }
   241→        if (receiver === -1) break; // no receiver — shouldn't happen if 100 <= n*maxVal
   242→        result[i]--;
   243→        result[receiver]++;
   244→        changed = true;
   245→      }
   246→    }
   247→  }
   248→
   249→  // Step 3: Final assertion — sum MUST be 100 (safety net, should never trigger)
   250→  const finalSum = result.reduce((a, b) => a + b, 0);
   251→  if (finalSum !== 100 && n > 0) {
   252→    // Adjust the entry farthest from its bound
   253→    const diff = 100 - finalSum;
   254→    let bestIdx = 0;
   255→    let bestRoom = -1;
   256→    for (let i = 0; i < n; i++) {
   257→      const room = diff > 0 ? maxVal - result[i] : result[i] - minVal;
   258→      if (room > bestRoom) {
   259→        bestRoom = room;
   260→        bestIdx = i;
   261→      }
   262→    }
   263→    result[bestIdx] += diff;
   264→  }
   265→
   266→  return result;
   267→}

  1427→// === Probability Context Builder ==============================================
  1428→
  1429→/**
  1430→ * Derives the shared probability context from sanitized {@link GraphInput}.
  1431→ *
  1432→ * Pre-computes six normalized indicators `[0,1]` that are reused across
  1433→ * multiple node probability calculations:
  1434→ *
  1435→ * - **adxNorm** — ADX/60 clamped to [0,1]. Values > 0.5 indicate strong trend.
  1436→ * - **srProximity** — 1 − min(distToR1, distToS1) × 10. Near 1 when price is
  1437→ *   close to a support/resistance level.
  1438→ * - **oscillatorExtreme** — Average of |RSI−50|/50 and |CCI|/200. High when
  1439→ *   oscillators are at extremes.
  1440→ * - **divergenceProxy** — oscillatorExtreme × (1 − adxNorm). High when oscillators
  1441→ *   are extreme in a weak-trend environment (classic divergence setup).
  1442→ * - **overboughtRisk** — Weighted blend of (RSI−60)/30 × 0.3, (MFI−70)/30 × 0.2,
  1443→ *   (StochK−75)/25 × 0.2. Measures overbought exhaustion risk.
  1444→ * - **oversoldBounce** — Mirror of overboughtRisk: (30−RSI)/30 × 0.3,
  1445→ *   (20−MFI)/20 × 0.2, (20−StochK)/20 × 0.2. Measures oversold bounce potential.
  1446→ *
  1447→ * @param input - Sanitized {@link GraphInput} (all fields finite, bounded)
  1448→ * @returns Complete {@link ProbContext} with all derived indicators
  1449→ */
  1450→function buildContext(input: GraphInput): ProbContext {
  1451→  const adxNorm = clamp(input.adx / 60, 0, 1);
  1452→  const srProximity = clamp(
  1453→    1 - Math.min(input.distToR1, input.distToS1) * 10,
  1454→    0,
  1455→    1
  1456→  );
  1457→  const oscillatorExtreme = clamp(
  1458→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
  1459→    0,
  1460→    1
  1461→  );
  1462→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1463→
  1464→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1465→  const overboughtRisk =
  1466→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1467→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1468→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1469→
  1470→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1471→  const oversoldBounce =
  1472→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1473→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1474→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1475→
  1476→  return {
  1477→    input,
  1478→    adxNorm,
  1479→    srProximity,
  1480→    oscillatorExtreme,
  1481→    divergenceProxy,
  1482→    overboughtRisk,
  1483→    oversoldBounce,
  1484→  };
  1485→}
  1486→
  1487→// === Node Probability Computation =============================================
  1488→// For each non-terminal node, returns an array of probabilities for its
  1489→// outgoing edges. The order MUST match the order of edges in createEdges().
  1490→// =============================================================================
  1491→
  1492→/**
  1493→ * Computes the probability distribution over outgoing edges for a given node.
  1494→ *
  1495→ * This is the **core probability logic** of the DAG. Each non-terminal node
  1496→ * has a `case` in the switch statement that returns a normalized array of
  1497→ * probabilities — one per outgoing edge, **in the same order as the edges
  1498→ * defined in {@link createEdges}**.
  1499→ *
  1500→ * ### Probability derivation strategy
  1501→ *
  1502→ * - **ROOT**: Weights based on `adxNorm` (trend), `srProximity` (breakout),
  1503→ *   and `divergenceProxy` (reversal). Strong trends → more trend weight;
  1504→ *   near S/R → more breakout weight; extreme oscillators in weak trend →
  1505→ *   more reversal weight.
  1506→ *
  1507→ * - **N_TREND**: Splits by `bullConsensus` and ADX. High ADX amplifies
  1508→ *   directional bias; low ADX increases the "flat" probability.
  1509→ *
  1510→ * - **N_T_BULL / N_T_BEAR**: Momentum magnitude determines high/moderate split;
  1511→ *   `overboughtRisk` / `oversoldBounce` compete as the third outcome.
  1512→ *
  1513→ * - **N_T_B_MOM_HIGH** → SC8, SC9, SC7: All bullish, weighted by
  1514→ *   `confidenceIndex` and `strengthIndex`.
  1515→ *
  1516→ * - **N_T_B_OVERBOUGHT** → SC4, SC1, SC9: Correction (SC4) most likely,
  1517→ *   with a small chance of last-gasp surge (SC9).
  1518→ *
  1519→ * - **N_BREAK**: Splits by `bullConsensus` and `srProximity`. Near S/R
  1520→ *   with bullish consensus → bullish breakout; far from S/R → no breakout.
  1521→ *
  1522→ * - **N_B_UP / N_B_DOWN**: Volume confirmation (`hasVolume` + MFI) splits
  1523→ *   confirmed vs. weak volume. Confirmed → all-directional terminals;
  1524→ *   weak → fake breakout risk with SC5.
  1525→ *
  1526→ * - **N_REVERSAL**: Splits by `overboughtRisk` and `oversoldBounce`.
  1527→ *   When neither is strong, `noneRev` gets high probability.
  1528→ *
  1529→ * - **N_R_BULL / N_R_BEAR**: Splits by CCI (divergence), Fisher Transform
  1530→ *   (candle), and `srAvgStrength` (S/R bounce).
  1531→ *
  1532→ * @param nodeId      - The node to compute probabilities for
  1533→ * @param ctx         - Pre-computed {@link ProbContext} with derived indicators
  1534→ * @param _numChildren - Number of outgoing edges (unused, for documentation only)
  1535→ * @returns Normalized probability array matching the order of edges from this node
  1536→ */
  1537→function computeNodeProbabilities(
  1538→  nodeId: string,
  1539→  ctx: ProbContext,
  1540→  _numChildren: number
  1541→): number[] {
  1542→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1543→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1544→
  1545→  switch (nodeId) {
  1546→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1547→    case 'ROOT': {
  1548→      const trendW = adxNorm * 0.5 + input.mlTrend * 0.18 + 0.1;
  1549→      const breakW = srProximity * 0.3 + (1 - adxNorm) * 0.08 + 0.08;
  1550→      const revW = ctx.divergenceProxy * 0.35 + (1 - adxNorm) * 0.12 + 0.06;
  1551→      return normalize([trendW, breakW, revW]);
  1552→    }
  1553→
  1554→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1555→    case 'N_TREND': {
  1556→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1557→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1558→      const flatUncertainty = (1 - Math.abs(input.bullConsensus - 0.5) * 2);
  1559→      const flat = flatUncertainty * clamp(1 - adxNorm * 1.5, 0, 1) * 0.45;
  1560→      return normalize([bull, bear, flat]);
  1561→    }
  1562→
  1563→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1564→    case 'N_T_BULL': {
  1565→      const momHigh =
  1566→        clamp(
  1567→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1568→            (input.macdHist > 0
  1569→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1570→              : 0),
  1571→          0,
  1572→          1
  1573→        ) * input.mlMomentum;
  1574→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1575→      return normalize([momHigh, momMod, overboughtRisk]);
  1576→    }
  1577→
  1578→    // ── N_T_B_MOM_HIGH → [SC8, SC9, SC7] ────────────────────────────────────
  1579→    // Strong bullish momentum — all three outcomes are bullish (no range)
  1580→    case 'N_T_B_MOM_HIGH': {
  1581→      const pSC8 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1582→      const pSC9 = 0.15 + input.strengthIndex * 0.2;
  1583→      const pSC7 = 0.35 * input.confidenceIndex + 0.15;
  1584→      return normalize([pSC8, pSC9, pSC7]);
  1585→    }
  1586→
  1587→    // ── N_T_B_MOM_MOD → [SC7, SC6, SC5] ─────────────────────────────────────
  1588→    // Moderate bullish — small explicit SC5 based on uncertainty
  1589→    case 'N_T_B_MOM_MOD': {
  1590→      const pSC7 = 0.55 * input.confidenceIndex + 0.2;
  1591→      const pSC6 = 0.55 * (1 - input.confidenceIndex) + 0.15;
  1592→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.02, 0.15);
  1593→      return normalize([pSC7, pSC6, pSC5]);
  1594→    }
  1595→
  1596→    // ── N_T_B_OVERBOUGHT → [SC4, SC1, SC9] ──────────────────────────────────
  1597→    // Overbought in uptrend — correction or last-gasp surge (no range)
  1598→    case 'N_T_B_OVERBOUGHT': {
  1599→      const pSC4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1600→      const pSC1 = 0.15 + input.strengthIndex * 0.15;
  1601→      const pSC9 = 0.1 + input.mlMomentum * 0.2;
  1602→      return normalize([pSC4, pSC1, pSC9]);
  1603→    }
  1604→
  1605→    // ── N_T_BEAR → [N_T_BE_MOM_LOW, N_T_BE_MOM_MOD, N_T_BE_OVERSOLD] ────
  1606→    case 'N_T_BEAR': {
  1607→      const momLow =
  1608→        clamp(
  1609→          (input.momentum < 0 ? Math.abs(input.momentum) / 50 : 0) +
  1610→            (input.macdHist < 0
  1611→              ? (Math.abs(input.macdHist) / (atrSafe * 0.5)) * 0.3
  1612→              : 0),
  1613→          0,
  1614→          1
  1615→        ) * input.mlMomentum;
  1616→      const momMod = (1 - momLow) * 0.6 * input.mlMomentum;
  1617→      return normalize([momLow, momMod, oversoldBounce]);
  1618→    }
  1619→
  1620→    // ── N_T_BE_MOM_LOW → [SC2, SC1, SC3] ────────────────────────────────────
  1621→    // Strong bearish momentum — all three outcomes are bearish (no range)
  1622→    case 'N_T_BE_MOM_LOW': {
  1623→      const pSC2 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1624→      const pSC1 = 0.15 + input.strengthIndex * 0.2;
  1625→      const pSC3 = 0.35 * input.confidenceIndex + 0.15;
  1626→      return normalize([pSC2, pSC1, pSC3]);
  1627→    }
  1628→
  1629→    // ── N_T_BE_MOM_MOD → [SC3, SC4, SC5] ────────────────────────────────────
  1630→    // Moderate bearish — small explicit SC5 based on uncertainty
  1631→    case 'N_T_BE_MOM_MOD': {
  1632→      const pSC3 = 0.55 * input.confidenceIndex + 0.2;
  1633→      const pSC4 = 0.55 * (1 - input.confidenceIndex) + 0.15;
  1634→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.02, 0.15);
  1635→      return normalize([pSC3, pSC4, pSC5]);
  1636→    }
  1637→
  1638→    // ── N_T_BE_OVERSOLD → [SC6, SC9, SC4] ───────────────────────────────────
  1639→    // Oversold in downtrend — bounce or mild continuation (no range)
  1640→    case 'N_T_BE_OVERSOLD': {
  1641→      const pSC6 = 0.55 + (1 - input.mlMomentum) * 0.2;
  1642→      const pSC9 = 0.15 + input.strengthIndex * 0.15;
  1643→      const pSC4 = 0.1 + input.mlMomentum * 0.2;
  1644→      return normalize([pSC6, pSC9, pSC4]);
  1645→    }
  1646→
  1647→    // ── N_T_FLAT → [N_T_F_VOL_LOW, N_T_F_VOL_HIGH, N_T_F_VOL_MOD] ────────
  1648→    case 'N_T_FLAT': {
  1649→      const volLow =
  1650→        clamp(1 - (input.atr / input.price) * 20, 0, 1) *
  1651→        (1 - input.mlVolatility) *
  1652→        0.7;
  1653→      const volHigh =
  1654→        clamp((input.atr / input.price) * 20, 0, 1) * input.mlVolatility * 0.7;
  1655→      const volMod = (1 - volLow - volHigh) * 0.5 + 0.1;
  1656→      return normalize([volLow, volHigh, volMod]);
  1657→    }
  1658→
  1659→    // ── N_T_F_VOL_LOW → [SC5, SC6, SC4] ─────────────────────────────────────
  1660→    case 'N_T_F_VOL_LOW': {
  1661→      const pSC5 = 0.55 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1662→      const pSC6 = 0.2 * input.bullConsensus + 0.08;
  1663→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.08;
  1664→      return normalize([pSC5, pSC6, pSC4]);
  1665→    }
  1666→
  1667→    // ── N_T_F_VOL_HIGH → [SC9, SC1, SC5] ────────────────────────────────────
  1668→    // High volatility shock — directional shock much more likely than range
  1669→    case 'N_T_F_VOL_HIGH': {
  1670→      const pSC9 = 0.4 * input.bullConsensus + 0.15;
  1671→      const pSC1 = 0.4 * (1 - input.bullConsensus) + 0.15;
  1672→      const pSC5 = clamp((1 - input.mlVolatility) * 0.12, 0.02, 0.12);
  1673→      return normalize([pSC9, pSC1, pSC5]);
  1674→    }
  1675→
  1676→    // ── N_T_F_VOL_MOD → [SC6, SC4, SC5] ─────────────────────────────────────
  1677→    // Moderate volatility — directional more likely than range
  1678→    case 'N_T_F_VOL_MOD': {
  1679→      const pSC6 = 0.38 * input.bullConsensus + 0.15;
  1680→      const pSC4 = 0.38 * (1 - input.bullConsensus) + 0.15;
  1681→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.03, 0.15);
  1682→      return normalize([pSC6, pSC4, pSC5]);
  1683→    }
  1684→
  1685→    // ── N_BREAK → [N_B_UP, N_B_DOWN, N_B_NONE] ────────────────────────────
  1686→    case 'N_BREAK': {
  1687→      const upProb = input.bullConsensus * srProximity * 0.75 + 0.08;
  1688→      const downProb = (1 - input.bullConsensus) * srProximity * 0.75 + 0.08;
  1689→      const noneProb = (1 - srProximity) * 0.35 + 0.05;
  1690→      return normalize([upProb, downProb, noneProb]);
  1691→    }
  1692→
  1693→    // ── N_B_UP → [N_B_U_VOL_C, N_B_U_VOL_W] ─────────────────────────────
  1694→    case 'N_B_UP': {
  1695→      let volConfirm: number;
  1696→      let volWeak: number;
  1697→      if (input.hasVolume) {
  1698→        volConfirm = clamp(input.mfi / 60, 0, 1) * 0.6 + 0.2;
  1699→        volWeak = 1 - volConfirm;
  1700→      } else {
  1701→        volConfirm = 0.4;
  1702→        volWeak = 0.6;
  1703→      }
  1704→      return normalize([volConfirm, volWeak]);
  1705→    }
  1706→
  1707→    // ── N_B_U_VOL_C → [SC7, SC8, SC9] ──────────────────────────────────────
  1708→    // Confirmed bullish breakout — all bullish, no range
  1709→    case 'N_B_U_VOL_C': {
  1710→      const pSC7 = 0.45 + input.confidenceIndex * 0.15;
  1711→      const pSC8 = 0.2 + input.strengthIndex * 0.15;
  1712→      const pSC9 = 0.15 + input.strengthIndex * 0.1;
  1713→      return normalize([pSC7, pSC8, pSC9]);
  1714→    }
  1715→
  1716→    // ── N_B_U_VOL_W → [SC6, SC4, SC5] ──────────────────────────────────────
  1717→    // Weak volume breakout — fake breakout risk, capped SC5
  1718→    case 'N_B_U_VOL_W': {
  1719→      const pSC6 = 0.4 + input.bullConsensus * 0.15;
  1720→      const pSC4 = 0.3 + (1 - input.bullConsensus) * 0.15;
  1721→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.12, 0.02, 0.12);
  1722→      return normalize([pSC6, pSC4, pSC5]);
  1723→    }
  1724→
  1725→    // ── N_B_DOWN → [N_B_D_VOL_C, N_B_D_VOL_W] ────────────────────────────
  1726→    case 'N_B_DOWN': {
  1727→      let volConfirm: number;
  1728→      if (input.hasVolume) {
  1729→        volConfirm = clamp((100 - input.mfi) / 60, 0, 1) * 0.6 + 0.2;
  1730→      } else {
  1731→        volConfirm = 0.4;
  1732→      }
  1733→      const volWeak = 1 - volConfirm;
  1734→      return normalize([volConfirm, volWeak]);
  1735→    }
  1736→
  1737→    // ── N_B_D_VOL_C → [SC3, SC2, SC1] ──────────────────────────────────────
  1738→    // Confirmed bearish breakout — all bearish, no range
  1739→    case 'N_B_D_VOL_C': {
  1740→      const pSC3 = 0.45 + input.confidenceIndex * 0.15;
  1741→      const pSC2 = 0.2 + input.strengthIndex * 0.15;
  1742→      const pSC1 = 0.15 + input.strengthIndex * 0.1;
  1743→      return normalize([pSC3, pSC2, pSC1]);
  1744→    }
  1745→
  1746→    // ── N_B_D_VOL_W → [SC4, SC6, SC5] ──────────────────────────────────────
  1747→    // Weak volume bearish breakout — fake breakout risk, capped SC5
  1748→    case 'N_B_D_VOL_W': {
  1749→      const pSC4 = 0.4 + (1 - input.bullConsensus) * 0.15;
  1750→      const pSC6 = 0.3 + input.bullConsensus * 0.15;
  1751→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.12, 0.02, 0.12);
  1752→      return normalize([pSC4, pSC6, pSC5]);
  1753→    }
  1754→
  1755→    // ── N_B_NONE → [SC4, SC5, SC6] ──────────────────────────────────────────
  1756→    case 'N_B_NONE': {
  1757→      const pSC5 = 0.45 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1758→      const pSC6 = 0.2 * input.bullConsensus + 0.08;
  1759→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.08;
  1760→      return normalize([pSC4, pSC5, pSC6]);
  1761→    }
  1762→
  1763→    // ── N_REVERSAL → [N_R_BULL, N_R_BEAR, N_R_NONE] ──────────────────────
  1764→    case 'N_REVERSAL': {
  1765→      const bullRev = overboughtRisk;
  1766→      const bearRev = oversoldBounce;
  1767→      const maxRevSignal = Math.max(overboughtRisk, oversoldBounce);
  1768→      const noneRev = 1 - maxRevSignal * 0.85;
  1769→      return normalize([bullRev, bearRev, noneRev]);
  1770→    }
  1771→
  1772→    // ── N_R_BULL → [N_R_B_DIV, N_R_B_CANDLE, N_R_B_SR] ──────────────────
  1773→    case 'N_R_BULL': {
  1774→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1775→      const candleProb =
  1776→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1777→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1778→      return normalize([divProb, candleProb, srProb]);
  1779→    }
  1780→
  1781→    // ── N_R_B_DIV → [SC6, SC8, SC5] ────────────────────────────────────────
  1782→    // Bullish divergence — explicit small SC5
  1783→    case 'N_R_B_DIV': {
  1784→      const pSC6 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1785→      const pSC8 = 0.2 + input.mlMomentum * 0.15;
  1786→      const pSC5 = clamp((1 - adxNorm) * 0.1, 0.02, 0.1);
  1787→      return normalize([pSC6, pSC8, pSC5]);
  1788→    }
  1789→
  1790→    // ── N_R_B_CANDLE → [SC7, SC6, SC5] ─────────────────────────────────────
  1791→    // Bullish candle pattern — explicit small SC5
  1792→    case 'N_R_B_CANDLE': {
  1793→      const pSC7 = 0.5 * input.confidenceIndex + 0.2;
  1794→      const pSC6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1795→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.1, 0.02, 0.1);
  1796→      return normalize([pSC7, pSC6, pSC5]);
  1797→    }
  1798→
  1799→    // ── N_R_B_SR → [SC6, SC5, SC4] ─────────────────────────────────────────
  1800→    // Support bounce — explicit small SC5
  1801→    // Edge order: SC6, SC5, SC4 — probabilities computed then reordered
  1802→    case 'N_R_B_SR': {
  1803→      const pSC6 = 0.5 + input.srAvgStrength * 0.2;
  1804→      const pSC4 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1805→      const pSC5 = clamp((1 - input.srAvgStrength) * 0.1, 0.02, 0.1);
  1806→      // Return in edge order: [SC6, SC5, SC4]
  1807→      return normalize([pSC6, pSC5, pSC4]);
  1808→    }
  1809→
  1810→    // ── N_R_BEAR → [N_R_BE_DIV, N_R_BE_CANDLE, N_R_BE_SR] ───────────────
  1811→    // Mirror of N_R_BULL
  1812→    case 'N_R_BEAR': {
  1813→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1814→      const candleProb =
  1815→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1816→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1817→      return normalize([divProb, candleProb, srProb]);
  1818→    }
  1819→
  1820→    // ── N_R_BE_DIV → [SC4, SC2, SC5] ───────────────────────────────────────
  1821→    // Bearish divergence — explicit small SC5
  1822→    case 'N_R_BE_DIV': {
  1823→      const pSC4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1824→      const pSC2 = 0.2 + input.mlMomentum * 0.15;
  1825→      const pSC5 = clamp((1 - adxNorm) * 0.1, 0.02, 0.1);
  1826→      return normalize([pSC4, pSC2, pSC5]);
  1827→    }
  1828→
  1829→    // ── N_R_BE_CANDLE → [SC3, SC4, SC5] ────────────────────────────────────
  1830→    // Bearish candle pattern — explicit small SC5
  1831→    case 'N_R_BE_CANDLE': {
  1832→      const pSC3 = 0.5 * input.confidenceIndex + 0.2;
  1833→      const pSC4 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1834→      const pSC5 = clamp((1 - input.confidenceIndex) * 0.1, 0.02, 0.1);
  1835→      return normalize([pSC3, pSC4, pSC5]);
  1836→    }
  1837→
  1838→    // ── N_R_BE_SR → [SC4, SC5, SC6] ─────────────────────────────────────────
  1839→    // Resistance rejection — explicit small SC5
  1840→    // Edge order: SC4, SC5, SC6 — probabilities computed then reordered
  1841→    case 'N_R_BE_SR': {
  1842→      const pSC4 = 0.5 + input.srAvgStrength * 0.2;
  1843→      const pSC6 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1844→      const pSC5 = clamp((1 - input.srAvgStrength) * 0.1, 0.02, 0.1);
  1845→      // Return in edge order: [SC4, SC5, SC6]
  1846→      return normalize([pSC4, pSC5, pSC6]);
  1847→    }
  1848→
  1849→    // ── N_R_NONE → [SC4, SC5, SC6] ──────────────────────────────────────────
  1850→    case 'N_R_NONE': {
  1851→      const pSC5 = 0.4 + (1 - Math.abs(input.bullConsensus - 0.5) * 2) * 0.15;
  1852→      const pSC6 = 0.2 * input.bullConsensus + 0.1;
  1853→      const pSC4 = 0.2 * (1 - input.bullConsensus) + 0.1;
  1854→      return normalize([pSC4, pSC5, pSC6]);
  1855→    }
  1856→
  1857→    // ── Terminal nodes (SC1–SC9) have no outgoing edges ─────────────────────
  1858→    default:
  1859→      return [];
  1860→  }
  1861→}
  1862→
  1863→// === Graph Traversal ==========================================================
  1864→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1865→// =============================================================================
  1866→
  1867→/**
  1868→ * Recursively traverses the DAG from a given node, accumulating path
  1869→ * probabilities at terminal nodes (SC1–SC9).
  1870→ *
  1871→ * ### Algorithm
  1872→ *
  1873→ * 1. Add `pathProb` to the node's accumulator (`nodeAccum`).
  1874→ * 2. Compute outgoing edge probabilities via {@link computeNodeProbabilities}.
  1875→ * 3. For each child edge:
  1876→ *    a. Store the edge probability in `edgeProbabilities`.
  1877→ *    b. Determine the branch context (trend/breakout/reversal) if at ROOT.
  1878→ *    c. If the child is a terminal (SC*), accumulate `childPathProb` into
  1879→ *       `scenarios` and `contributions` (per-branch breakdown).
  1880→ *    d. Recurse into the child node.
  1881→ *
  1882→ * Because the graph is a DAG (no cycles), recursion terminates at the
  1883→ * terminal layer. Multiple paths can reach the same terminal, so
  1884→ * probabilities are **summed** across all contributing paths.
  1885→ *
  1886→ * @param nodeId           - Current node being visited
  1887→ * @param pathProb         - Probability of reaching this node from ROOT (product of all edge probs)
  1888→ * @param currentBranch    - Which strategy branch we're under (null at ROOT)
  1889→ * @param ctx              - Pre-computed probability context
  1890→ * @param adj              - Adjacency map: nodeId → array of edge indices
  1891→ * @param edges            - All edges in the graph
  1892→ * @param edgeProbabilities - Mutable: stores computed probability per edge index
  1893→ * @param nodeAccum        - Mutable: accumulates total path probability at each node
  1894→ * @param scenarios        - Mutable: accumulates total probability at each SC terminal
  1895→ * @param contributions    - Mutable: per-branch probability contribution to each SC terminal
  1896→ */
  1897→function traverseGraph(
  1898→  nodeId: string,
  1899→  pathProb: number,
  1900→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1901→  ctx: ProbContext,
  1902→  adj: Map<string, number[]>,
  1903→  edges: GraphEdge[],
  1904→  edgeProbabilities: Record<number, number>,
  1905→  nodeAccum: Record<string, number>,
  1906→  scenarios: Record<string, number>,
  1907→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1908→): void {
  1909→  // Accumulate path probability at this node
  1910→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1911→
  1912→  const outEdgeIndices = adj.get(nodeId);
  1913→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1914→
  1915→  // Compute outgoing edge probabilities for this node
  1916→  const probs = computeNodeProbabilities(
  1917→    nodeId,
  1918→    ctx,
  1919→    outEdgeIndices.length
  1920→  );
  1921→
  1922→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1923→    const edgeIdx = outEdgeIndices[i];
  1924→    const edge = edges[edgeIdx];
  1925→    const prob = probs[i] || 0;
  1926→
  1927→    // Store edge probability
  1928→    edgeProbabilities[edgeIdx] = prob;
  1929→
  1930→    // Determine branch for child nodes
  1931→    let childBranch = currentBranch;
  1932→    if (nodeId === 'ROOT') {
  1933→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1934→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1935→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1936→    }
  1937→
  1938→    const childPathProb = pathProb * prob;
  1939→
  1940→    // Check if target is a terminal node (SC1–SC9)
  1941→    const isTerminal =
  1942→      edge.to.length === 3 &&
  1943→      edge.to[0] === 'S' &&
  1944→      edge.to[1] === 'C' &&
  1945→      edge.to[2] >= '1' &&
  1946→      edge.to[2] <= '9';
  1947→
  1948→    if (isTerminal) {
  1949→      // Accumulate scenario probability
  1950→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1951→      // Accumulate branch contribution
  1952→      if (childBranch) {
  1953→        contributions[edge.to][childBranch] += childPathProb;
  1954→      }
  1955→    }
  1956→
  1957→    // Recurse into child
  1958→    traverseGraph(
  1959→      edge.to,
  1960→      childPathProb,
  1961→      childBranch,
  1962→      ctx,
  1963→      adj,
  1964→      edges,
  1965→      edgeProbabilities,
  1966→      nodeAccum,
  1967→      scenarios,
  1968→      contributions
  1969→    );
  1970→  }
  1971→}
