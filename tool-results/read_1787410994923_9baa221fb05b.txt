   138→export default function VdesAnalysis(props: VdesAnalysisProps) {
   139→  const {
   140→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
   141→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
   142→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
   143→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
   144→    supportStrengths, resistanceStrengths, priceTargets,
   145→    isIndex, isTgju, currencyUnit,
   146→    edgeWeights,
   147→  } = props;
   148→
   149→  const unit = currencyUnit || 'ریال';
   150→
   151→  // ── Instrument Context (for intelligent naming) ─────────────────
   152→  const instrumentLabel = useMemo(() => {
   153→    if (isIndex) return { name: `شاخص ${symbolName}`, type: 'index' as const, verb: 'قرار دارد' };
   154→    if (isTgju) return { name: symbolName, type: 'tgju' as const, verb: 'قرار دارد' };
   155→    return { name: `سهم ${symbolName}`, type: 'stock' as const, verb: 'قرار دارد' };
   156→  }, [symbolName, isIndex, isTgju]);
   157→
   158→  const IL = instrumentLabel;
   159→
   160→  const containerRef = useRef<HTMLDivElement>(null);
   161→  const textAnalysisRef = useRef<HTMLDivElement>(null);
   162→  const chartApiRef = useRef<IChartApi | null>(null);
   163→  const [exporting, setExporting] = useState(false);
   164→  const [exportMenuOpen, setExportMenuOpen] = useState(false);
   165→  const exportMenuRef = useRef<HTMLDivElement>(null);
   166→
   167→  // ── AI Analysis state ───────────────────────────────────────────
   168→  const [aiAnalysis, setAiAnalysis] = useState<string | null>(null);
   169→  const [aiLoading, setAiLoading] = useState(false);
   170→  const [aiError, setAiError] = useState<string | null>(null);
   171→  const aiFetchIdRef = useRef(0);
   172→
   173→  // ── Analysis version selector (V7 locked / V8 / V9 locked / V10 locked / V11 active) ─────────
   174→  const [analysisVersion, setAnalysisVersion] = useState<7 | 8 | 9 | 10 | 11>(11);
   175→  // ── V11 Debug data ──────────────────────────────────────────────
   176→  const [v11Debug, setV11Debug] = useState<any>(null);
   177→
   178→  // ── V11 Scenarios (built from TA data + S/R levels) ───────────────
   179→  const v11Result = useMemo(() => {
   180→    if (analysisVersion !== 11) return null;
   181→    return buildV11Scenarios({
   182→      currentPrice,
   183→      atr,
   184→      overallSignal: overallSignal as 'bullish' | 'bearish' | 'neutral',
   185→      v9Scenarios: scenarios,
   186→      supports,
   187→      resistances,
   188→      bollingerUpper,
   189→      bollingerLower,
   190→      bollingerMiddle,
   191→      ma21,
   192→      ma100,
   193→    });
   194→  }, [analysisVersion, currentPrice, atr, overallSignal, scenarios, supports, resistances,
   195→      bollingerUpper, bollingerLower, bollingerMiddle, ma21, ma100]);
   196→
   197→  // ── V10 Scenarios (LOCKED — backward compat) ───────────────
   198→  const v10Result = useMemo(() => {
   199→    if (analysisVersion !== 10) return null;
   200→    return buildV10Locked({
   201→      currentPrice,
   202→      atr,
   203→      overallSignal: overallSignal as 'bullish' | 'bearish' | 'neutral',
   204→      v9Scenarios: scenarios,
   205→      supports,
   206→      resistances,
   207→      bollingerUpper,
   208→      bollingerLower,
   209→      bollingerMiddle,
   210→      ma21,
   211→      ma100,
   212→    });
   213→  }, [analysisVersion, currentPrice, atr, overallSignal, scenarios, supports, resistances,
   214→      bollingerUpper, bollingerLower, bollingerMiddle, ma21, ma100]);
   215→
   216→  // ── Fetch AI analysis ───────────────────────────────────────────
   217→  const fetchAiAnalysis = useCallback(async () => {
   218→    const fetchId = ++aiFetchIdRef.current;
   219→    setAiLoading(true);
   220→    setAiError(null);
   221→    try {
   222→      const apiUrl = analysisVersion === 11 ? '/api/v11-analysis' : analysisVersion === 10 ? '/api/v10-analysis' : analysisVersion === 9 ? '/api/v9-analysis' : analysisVersion === 8 ? '/api/v8-analysis' : '/api/vdes-analysis';
   223→      // V10/V11: use S-key scenarios, otherwise use V9 R-key scenarios
   224→      const sKeyResult = analysisVersion === 11 ? v11Result : analysisVersion === 10 ? v10Result : null;
   225→      const activeScenarios = sKeyResult ? sKeyResult.scenarios : scenarios;
   226→      const requestBody: any = {
   227→          symbolName, currentPrice, ma21, ma100,
   228→          rsi, mfi, cci, adx, stochK, stochD,
   229→          macdLine, macdSignal, macdHist,
   230→          diPlus, diMinus, sar, atr, obv, hasVolume,
   231→          bollingerUpper, bollingerMiddle, bollingerLower,
   232→          trendDirection, trendAngle, trendR2, overallSignal,
   233→          scenarios: activeScenarios, resistances, supports,
   234→          resistanceStrengths, supportStrengths,
   235→          isIndex, isTgju, currencyUnit,
   236→        };
   237→      // V10/V11: pass dominant + highest cumulative keys for narrative consistency
   238→      if (sKeyResult) {
   239→        if (analysisVersion === 11) {
   240→          requestBody.v11DominantKey = sKeyResult.dominantKey;
   241→          requestBody.v11HighestCumulativeKey = sKeyResult.highestCumulativeKey;
   242→        } else if (analysisVersion === 10) {
   243→          requestBody.v10DominantKey = sKeyResult.dominantKey;
   244→          requestBody.v10HighestCumulativeKey = sKeyResult.highestCumulativeKey;
   245→        }
   246→      }
   247→      // V9/V10/V11: include OHLCV history for ML training/prediction
   248→      if ((analysisVersion === 9 || analysisVersion === 10 || analysisVersion === 11) && candles.length > 0) {
   249→        requestBody.ohlcvHistory = candles.map(c => ({
   250→          date: c.date, open: c.open, high: c.high,
   251→          low: c.low, close: c.close, volume: c.volume,
   252→        }));
   253→      }
   254→
   255→      // AbortController for 90s client-side timeout
   256→      const controller = new AbortController();
   257→      const timeoutId = setTimeout(() => controller.abort(), 90000);
   258→
   259→      const res = await fetch(apiUrl, {
   260→        method: 'POST',
   261→        headers: { 'Content-Type': 'application/json' },
   262→        body: JSON.stringify(requestBody),
   263→        signal: controller.signal,
   264→      });
   265→      clearTimeout(timeoutId);
   266→
   267→      if (!res.ok) {
   268→        let errorMsg = `خطای سرور (${res.status})`;
   269→        try {
   270→          const errData = await res.json();
   271→          errorMsg = errData.error || errorMsg;
   272→        } catch {
   273→          // Response body is not JSON (e.g. 502 from Caddy)
   274→          if (res.status === 502) errorMsg = 'سرور در حال بارگذاری مجدد است. لطفاً چند ثانیه صبر کنید و دوباره تلاش کنید.';
   275→        }
   276→        throw new Error(errorMsg);
   277→      }
   278→      const data = await res.json();
   279→      if (fetchId === aiFetchIdRef.current) {
   280→        setAiAnalysis(data.analysis);
   281→        if (data._debug?.v11) setV11Debug({ ...data._debug.v11, v11Persona: data._debug.v11Persona, v11Template: data._debug.v11Template });
   282→        else if (data._debug?.v10) setV11Debug({ ...data._debug.v10, v11Persona: data._debug.v10Persona, v10Template: data._debug.v10Template });
   283→        else setV11Debug(null);
   284→      }
   285→    } catch (err) {
   286→      if (fetchId === aiFetchIdRef.current) {
   287→        if (err instanceof Error && err.name === 'AbortError') {
   288→          setAiError('زمان پاسخگویی به پایان رسید. لطفاً دوباره تلاش کنید.');
   289→        } else {
   290→          setAiError(err instanceof Error ? err.message : 'خطای ناشناخته');
   291→        }
   292→      }
   293→    } finally {
   294→      if (fetchId === aiFetchIdRef.current) {
   295→        setAiLoading(false);
   296→      }
   297→    }
   298→  }, [symbolName, currentPrice, ma21, ma100, rsi, mfi, cci, adx, stochK, stochD,
   299→    macdLine, macdSignal, macdHist, diPlus, diMinus, sar, atr, obv, hasVolume,
   300→    bollingerUpper, bollingerMiddle, bollingerLower, trendDirection, trendAngle, trendR2,
   301→    overallSignal, scenarios, resistances, supports, resistanceStrengths, supportStrengths,
   302→    isIndex, isTgju, currencyUnit, analysisVersion, v11Result, v10Result]);
   303→
   304→  // ── Auto-fetch AI analysis only on symbol or version change ──
   305→  // NOT on every data refresh (auto-refresh every 60s would re-trigger
   306→  // a 10-25s LLM call, making the text disappear each minute)
   307→  const fetchTriggerKey = `${symbolName}:${analysisVersion}`;
   308→  const lastAutoFetchKeyRef = useRef('');
   309→  useEffect(() => {
   310→    if (fetchTriggerKey === lastAutoFetchKeyRef.current) return;
   311→    lastAutoFetchKeyRef.current = fetchTriggerKey;
   312→    // Reset old text when switching symbols/versions
   313→    setAiAnalysis(null);
   314→    setAiError(null);
   315→    fetchAiAnalysis().catch(() => {});
   316→  }, [fetchTriggerKey]);
   317→
   318→  // ── Close export menu on outside click ───────────────────────
   319→  useEffect(() => {
   320→    if (!exportMenuOpen) return;
   321→    const handler = (e: MouseEvent) => {
   322→      if (exportMenuRef.current && !exportMenuRef.current.contains(e.target as Node)) {
   323→        setExportMenuOpen(false);
   324→      }
   325→    };
   326→    document.addEventListener('mousedown', handler);
   327→    return () => document.removeEventListener('mousedown', handler);
   328→  }, [exportMenuOpen]);
   329→
   330→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   331→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   332→  const S1_level = supports[0] ?? currentPrice * 0.95;
   333→  const S2_level = supports[1] ?? currentPrice * 0.90;
   334→  const S3_level = supports[2] ?? currentPrice * 0.85;
   335→
   336→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   337→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   338→
   339→  // ── Dominant scenario (V9) ──────────────────────────────────────
   340→  let highestKey = 'R3';
   341→  let highestProb = 0;
   342→  for (const key of SCENARIO_KEYS) {
   343→    if (scenarios[key].probability > highestProb) {
   344→      highestProb = scenarios[key].probability;
   345→      highestKey = key;
   346→    }
   347→  }
   348→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   349→
   350→  // ── V10/V11 dominant + highest cumulative + total ────────────
   351→  const sKeyResult = analysisVersion === 11 ? v11Result : analysisVersion === 10 ? v10Result : null;
   352→  const v11DominantKey = sKeyResult?.dominantKey || 'S5';
   353→  const v11HighestCumKey = sKeyResult?.highestCumulativeKey || 'S5';
   354→  const v11TotalProb = sKeyResult
   355→    ? V11_KEYS.reduce((sum, k) => sum + (sKeyResult.scenarios[k]?.probability || 0), 0)
   356→    : 0;
   357→
   358→  // ── Sorted scenarios by target price (ascending) ─────────────
   359→  const sortedScenarios = useMemo(() => {
   360→    return [...SCENARIO_KEYS]
   361→      .map(k => ({ key: k, ...scenarios[k] }))
   362→      .sort((a, b) => a.targetMin - b.targetMin);
   363→  }, [scenarios]);
   364→
   365→  // ── Dynamic scenario labels (context-aware based on overallSignal) ─
   366→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal), [overallSignal]);
   367→  const strategy = dynamicMeta[highestKey];
   368→
   369→  // ── Text variety: hash-based selector for varied language ──────────
   370→  const varietySeed = useMemo(() => {
   371→    const raw = `${rsi}${macdHist}${cci}${adx}${stochK}${trendAngle}${Math.round(currentPrice)}`;
   372→    let hash = 0;
   373→    for (let i = 0; i < raw.length; i++) hash = ((hash << 5) - hash + raw.charCodeAt(i)) | 0;
   374→    return Math.abs(hash);
   375→  }, [rsi, macdHist, cci, adx, stochK, trendAngle, currentPrice]);
   376→
   377→  // ── RSI signal ─────────────────────────────────────────────────
   378→  const rsiZone = rsi > 70 ? 'overbought' : rsi > 60 ? 'weak_ob' : rsi > 40 ? 'neutral' : rsi > 30 ? 'weak_os' : 'oversold';
   379→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   380→  const rsiColor = rsi > 70 ? 'text-red-700' : rsi > 60 ? 'text-amber-800' : rsi > 40 ? 'text-gray-600' : rsi > 30 ? 'text-amber-800' : 'text-emerald-700';
   381→
   382→  // ── Stochastic signal ──────────────────────────────────────────
   383→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   384→
   385→  // ── Trend text ─────────────────────────────────────────────────
   386→  const isUp = trendDirection === 'up';
   387→  const isDown = trendDirection === 'down';
   388→  const trendLabel = isUp ? 'صعودی' : isDown ? 'نزولی' : 'خنثی';
   389→  const trendColorCls = isUp ? 'text-emerald-700' : isDown ? 'text-red-700' : 'text-amber-800';
   390→  const adxStrength = adx > 40 ? 'بسیار قوی' : adx > 25 ? 'قوی' : adx > 15 ? 'متوسط' : 'ضعیف';
   391→  const adxColor = adx > 40 ? 'text-emerald-700' : adx > 25 ? 'text-amber-800' : 'text-gray-500';
   392→  const r2Pct = (trendR2 * 100).toFixed(1);
   393→
   394→  // ── Bollinger position ─────────────────────────────────────────
   395→  const bbRange = bollingerUpper - bollingerLower;
   396→  const bbPos = bbRange > 0 ? toPersianDigits(((currentPrice - bollingerLower) / bbRange * 100).toFixed(0)) : '۵۰';
   397→  const bbAbove = currentPrice > bollingerUpper;
   398→  const bbBelow = currentPrice < bollingerLower;
   399→  const bbSignal = bbAbove ? 'بالای باند بالایی' : bbBelow ? 'زیر باند پایینی' : `داخل باندها (${bbPos}٪)`;

   958→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   959→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   960→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   961→  grade?: string; overlapCount?: number; touchCount?: number;
   962→}) {
   963→  const isSupport = type === 'support';
   964→  const isTargetType = type === 'target';
   965→
   966→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   967→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   968→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   969→
   970→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   971→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   972→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   973→
   974→  return (
   975→    <div
   976→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   977→      style={{
   978→        background: bgAccent,
   979→        border: `1px solid ${borderAccent}`,
   980→        borderRight: `3px solid ${accentColor}`,
   981→      }}
   982→    >
   983→      {/* Label + Price */}
   984→      <div className="flex-1 min-w-0">
   985→        <div className="flex items-center gap-2">
   986→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   987→          {isTarget && <span className="text-[9px]">🎯</span>}
   988→          {overlapCount !== undefined && overlapCount > 0 && (
   989→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   990→              {toPersianDigits(overlapCount)} هم‌پوشانی
   991→            </span>
   992→          )}
   993→        </div>
   994→        <div className="flex items-baseline gap-1">
   995→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   996→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   997→        </div>
   998→      </div>
   999→
  1000→      {/* Strength + Grade */}
  1001→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
  1002→        <div className="flex items-center gap-1.5">
  1003→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
  1004→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
  1005→          <span className="text-[9px] text-gray-500">/۱۰</span>
  1006→        </div>
  1007→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
  1008→          <div
  1009→            className="h-full rounded-full transition-all duration-500"
  1010→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
  1011→          />
  1012→        </div>
  1013→        {touchCount !== undefined && touchCount > 0 && (
  1014→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
  1015→        )}
  1016→      </div>
  1017→    </div>
  1018→  );
  1019→}
