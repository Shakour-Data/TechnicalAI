   147→
   148→def adx(prices, period=14):
   149→    """Average Directional Index (simplified)."""
   150→    if len(prices) < 2 * period + 1:
   151→        return 25.0  # neutral
   152→    # Compute +DM, -DM, TR
   153→    plus_dm_list = []
   154→    minus_dm_list = []
   155→    tr_list = []
   156→    for i in range(1, len(prices)):
   157→        high = max(prices[i], prices[i - 1])  # proxy when no OHLC
   158→        low = min(prices[i], prices[i - 1])
   159→        prev_high = max(prices[i - 1], prices[max(0, i - 2)]) if i >= 2 else prices[i - 1]
   160→        prev_low = min(prices[i - 1], prices[max(0, i - 2)]) if i >= 2 else prices[i - 1]
   161→        up_move = high - prev_high
   162→        down_move = prev_low - low
   163→        plus_dm = max(up_move, 0) if up_move > down_move and up_move > 0 else 0
   164→        minus_dm = max(down_move, 0) if down_move > up_move and down_move > 0 else 0
   165→        tr = max(high - low, abs(high - prices[i - 1]), abs(low - prices[i - 1]))
   166→        plus_dm_list.append(plus_dm)
   167→        minus_dm_list.append(minus_dm)
   168→        tr_list.append(tr)
   169→    if len(tr_list) < period:
   170→        return 25.0
   171→    # Smooth with Wilder's method
   172→    atr = _mean(tr_list[-period:])
   173→ plus_smooth = _mean(plus_dm_list[-period:])
   174→    minus_smooth = _mean(minus_dm_list[-period:])
   175→    if atr == 0:
   176→        return 25.0
   177→    plus_di = 100 * plus_smooth / atr
   178→    minus_di = 100 * minus_smooth / atr
   179→    di_sum = plus_di + minus_di
   180→    if di_sum == 0:
   181→        return 0.0
   182→    dx = 100 * abs(plus_di - minus_di) / di_sum
   183→    return dx  # simplified single-period ADX
   184→
   185→
   186→def macd(prices, fast=12, slow=26, signal=9):
   187→    """MACD histogram value."""
   188→    if len(prices) < slow + signal:
   189→        return 0.0
   190→    ema_fast = ema(prices, fast)
   191→    ema_slow = ema(prices, slow)
   192→    if ema_fast is None or ema_slow is None:
   193→        return 0.0
   194→    macd_line = ema_fast - ema_slow
   195→    # Approximate signal line
   196→    diffs = []
   197→    k = 2.0 / (slow + 1)
   198→    ef = _mean(prices[:fast])
   199→    es = _mean(prices[:slow])
   200→    for v in prices[slow:]:
   201→        ef = v * (2 / (fast + 1)) + ef * (1 - 2 / (fast + 1))
   202→        es = v * k + es * (1 - k)
   203→        diffs.append(ef - es)
   204→    if len(diffs) < signal:
   205→        return macd_line
   206→    sig = ema(diffs, signal)
   207→    if sig is None:
   208→        return macd_line
   209→    return macd_line - sig

   212→def rolling_std(values, period):
   213→    """Rolling standard deviation over last `period` values."""
   214→    if len(values) < period:
   215→        return 0.0
   216→    window = values[-period:]
   217→    return _stdev(window)

   220→def rolling_max(values, period):
   221→    if len(values) < period:
   222→        return max(values) if values else 0
   223→    return max(values[-period:])

   226→def rolling_min(values, period):
   227→    if len(values) < period:
   228→        return min(values) if values else 0
   229→    return min(values[-period:])

   237→def compute_returns(prices):
   238→    """Simple returns."""
   239→    return [(prices[i] - prices[i - 1]) / prices[i - 1] if prices[i - 1] != 0 else 0
   240→            for i in range(1, len(prices))]
   241→
   242→
   243→def compute_log_returns(prices):
   244→    """Log returns."""
   245→    return [math.log(prices[i] / prices[i - 1]) if prices[i - 1] > 0 and prices[i] > 0 else 0
   246→            for i in range(1, len(prices))]

   248→
   249→def build_features(prices, volume=None):
   250→    """Build feature vector from price series.
   251→    Returns list of feature values.
   252→    """
   253→    if len(prices) < 5:
   254→        return []
   255→
   256→    feats = []
   257→    n = len(prices)
   258→
   259→    # --- Lag features (normalized) ---
   260→    for lag in [1, 2, 3, 5, 10]:
   261→        if n > lag and prices[-lag - 1] != 0:
   262→            feats.append(prices[-1] / prices[-lag - 1] - 1)
   263→        else:
   264→            feats.append(0.0)
   265→
   266→    # --- Returns ---
   267→    rets = compute_returns(prices)
   268→    if rets:
   269→        feats.append(rets[-1])  # 1-day return
   270→    else:
   271→        feats.append(0.0)
   272→    if len(rets) >= 5:
   273→        feats.append(_mean(rets[-5:]))  # 5-day mean return
   274→    else:
   275→        feats.append(_mean(rets) if rets else 0.0)
   276→
   277→    # --- Volatility ---
   278→    if len(rets) >= 5:
   279→        feats.append(_stdev(rets[-5:]))  # 5-day vol
   280→    else:
   281→        feats.append(0.0)
   282→    if len(rets) >= 10:
   283→        feats.append(_stdev(rets[-10:]))  # 10-day vol
   284→    else:
   285→        feats.append(_stdev(rets) if rets else 0.0)
   286→
   287→    # --- MA ratios ---
   288→    for period in [5, 10, 20]:
   289→        m = sma(prices, period)
   290→        if m and m != 0:
   291→            feats.append(prices[-1] / m - 1)
   292→        else:
   293→            feats.append(0.0)
   294→
   295→    # --- EMA ratios ---
   296→    for period in [12, 26]:
   297→        e = ema(prices, period)
   298→        if e and e != 0:
   299→            feats.append(prices[-1] / e - 1)
   300→        else:
   301→            feats.append(0.0)
   302→
   303→    # --- RSI ---
   304→    feats.append(rsi(prices) / 100.0)
   305→
   306→    # --- ADX ---
   307→    feats.append(adx(prices) / 100.0)
   308→
   309→    # --- MACD histogram (normalized) ---
   310→    mh = macd(prices)
   311→    if prices[-1] != 0:
   312→        feats.append(mh / prices[-1])
   313→    else:
   314→        feats.append(0.0)
   315→
   316→    # --- Bollinger position ---
   317→    for period in [20]:
   318→        if len(prices) >= period:
   319→            m = sma(prices, period)
   320→            s = rolling_std(prices, period)
   321→            if s and s != 0:
   322→                feats.append((prices[-1] - m) / (2 * s))
   323→            else:
   324→                feats.append(0.0)
   325→        else:
   326→            feats.append(0.0)
   327→
   328→    # --- Volume features ---
   329→    if volume and len(volume) >= 5:
   330→        vol_rets = [(volume[i] - volume[i - 1]) / volume[i - 1]
   331→                     if volume[i - 1] != 0 else 0
   332→                     for i in range(1, len(volume))]
   333→        feats.append(_mean(vol_rets[-5:]) if len(vol_rets) >= 5 else _mean(vol_rets) if vol_rets else 0.0)
   334→        feats.append(_stdev(vol_rets[-5:]) if len(vol_rets) >= 5 else 0.0)
   335→        # Price-volume correlation proxy
   336→        if len(rets) >= 5 and len(vol_rets) >= 5:
   337→            r = rets[-5:]
   338→            v = vol_rets[-5:]
   339→            rm = _mean(r)
   340→            vm = _mean(v)
   341→            num = sum((ri - rm) * (vi - vm) for ri, vi in zip(r, v))
   342→            den = _stdev(r) * _stdev(v) * 5
   343→            feats.append(num / den if den != 0 else 0.0)
   344→        else:
   345→            feats.append(0.0)
   346→    else:
   347→        feats.extend([0.0, 0.0, 0.0])
   348→
   349→    # --- Momentum ---
   350→    if n >= 5:
   351→        feats.append((prices[-1] / prices[-5] - 1))
   352→    else:
   353→        feats.append(0.0)
   354→    if n >= 10:
   355→        feats.append((prices[-1] / prices[-10] - 1))
   356→    else:
   357→        feats.append(0.0)
   358→
   359→    return feats
   360→

   362→def build_target(prices, step=1):
   363→    """Target: return at step ahead."""
   364→    if len(prices) <= step:
   365→        return 0.0
   366→    return (prices[-step - 1] / prices[-1] - 1) if prices[-1] != 0 else 0.0

   369→def build_dataset(prices, volume=None, max_lag=30):
   370→    """Build X, y dataset from rolling windows."""
   371→    X = []
   372→    y = []
   373→    min_len = 30  # minimum data needed
   374→    if len(prices) < min_len:
   375→        return X, y
   376→    for i in range(min_len, len(prices) - 1):
   377→        window_prices = prices[:i]
   378→        window_vol = volume[:i] if volume else None
   379→        feats = build_features(window_prices, window_vol)
   380→        if not feats:
   381→            continue
   382→        X.append(feats)
   383→        y.append(prices[i] / prices[i - 1] - 1 if prices[i - 1] != 0 else 0.0)
   384→    return X, y

   388→# Model 1: Ridge Regression (from scratch)
   389→# ─────────────────────────────────────────────────────────────────────────────

   391→
   392→class RidgeRegression:
   393→    """Ridge regression: w = (X^T X + λI)^{-1} X^T y"""
   394→
   395→    def __init__(self, alpha=1.0):
   396→        self.alpha = alpha
   397→        self.weights = None
   398→        self.bias = 0.0
   399→        self.fitted = False
   400→
   401→    def fit(self, X, y):
   402→        if len(X) < 3 or len(X[0]) < 1:
   403→            self.fitted = False
   404→            return self
   405→        n = len(X)
   406→        p = len(X[0])
   407→        # Center y
   408→        y_mean = _mean(y)
   409→        y_c = [yi - y_mean for yi in y]
   410→        # Center X
   411→        x_means = [_mean(X[j][i] for j in range(n)) for i in range(p)]
   412→        X_c = [[X[j][i] - x_means[i] for i in range(p)] for j in range(n)]
   413→        # X^T X
   414→        XT = _transpose(X_c)
   415→        XTX = _mat_mul(XT, X_c)
   416→        # Regularize
   417→        reg = _identity(p)
   418→        XTX_reg = [[XTX[i][j] + (self.alpha if i == j else 0) for j in range(p)] for i in range(p)]
   419→        XTX_inv = _invert_matrix(XTX_reg)
   420→        # X^T y
   421→        XTy = [sum(XT[i][j] * y_c[j] for j in range(n)) for i in range(p)]
   422→        # w = XTX_inv @ XTy
   423→        self.weights = [sum(XTX_inv[i][j] * XTy[j] for j in range(p)) for i in range(p)]
   424→        self.bias = y_mean - sum(self.weights[i] * x_means[i] for i in range(p))
   425→        self.fitted = True
   426→        return self
   427→
   428→    def predict(self, X):
   429→        if not self.fitted or not self.weights:
   430→            return [0.0] * len(X)
   431→        return [sum(self.weights[j] * row[j] for j in range(len(row))) + self.bias for row in X]
   432→
   433→    def predict_single(self, x):
   434→        if not self.fitted or not self.weights:
   435→            return 0.0
   436→        return sum(self.weights[j] * x[j] for j in range(len(x))) + self.bias
   437→

   439→# ─────────────────────────────────────────────────────────────────────────────
   440→# Model 2: Weighted Moving Average Ensemble
   441→# ─────────────────────────────────────────────────────────────────────────────
   442→
   443→
   444→def wma_predict(prices, steps=5):
   445→    """Predict using weighted moving average of recent returns."""
   446→    if len(prices) < 3:
   447→        return [0.0] * steps
   448→
   449→    rets = compute_returns(prices)
   450→    if not rets:
   451→        return [0.0] * steps
   452→
   453→    # Use exponential weights for recent returns
   454→    n_rets = min(len(rets), 20)
   455→    recent_rets = rets[-n_rets:]
   456→    weights = [math.exp(-0.1 * (n_rets - 1 - i)) for i in range(n_rets)]
   457→    w_sum = sum(weights)
   458→    weights = [w / w_sum for w in weights]
   459→    expected_return = sum(w * r for w, r in zip(weights, recent_rets))
   460→
   461→    # Volatility for decay
   462→    if len(recent_rets) >= 5:
   463→        vol = _stdev(recent_rets[-5:])
   464→    else:
   465→        vol = _stdev(recent_rets)
   466→
   467→    predictions = []
   468→    last_price = prices[-1]
   469→    for step in range(steps):
   470→        # Decay the expected return for longer horizons
   471→        decay = math.exp(-0.05 * step)
   472→        ret = expected_return * decay
   473→        last_price = last_price * (1 + ret)
   474→        predictions.append(last_price)
   475→
   476→    return predictions
   477→

   480→# Model 3: Mean-Reversion Model
   481→# ─────────────────────────────────────────────────────────────────────────────
   482→
   483→
   484→def mean_reversion_predict(prices, steps=5):
   485→    """Predict based on mean-reversion tendency."""
   486→    if len(prices) < 10:
   487→        return [0.0] * steps
   488→
   489→    # Compute multiple lookback means
   490→    lookbacks = [10, 20, 50]
   491→    means = []
   492→    for lb in lookbacks:
   493→        if len(prices) >= lb:
   494→            means.append(sma(prices, lb))
   495→        else:
   496→            means.append(sma(prices, len(prices)))
   497→
   498→    # Weight longer lookbacks more (they're more stable)
   499→    weights = [0.2, 0.3, 0.5][:len(means)]
   500→    w_sum = sum(weights)
   501→    weights = [w / w_sum for w in weights]
   502→    target_mean = sum(w * m for w, m in zip(weights, means) if m is not None)
   503→
   504→    if target_mean is None or target_mean == 0:
   505→        return [prices[-1]] * steps
   506→
   507→    # Speed of reversion: how far from mean / recent volatility
   508→    deviation = (prices[-1] - target_mean) / target_mean
   509→    rets = compute_returns(prices)
   510→    vol = _stdev(rets[-10:]) if len(rets) >= 10 else (_stdev(rets) if rets else 0.01)
   511→    if vol == 0:
   512→        vol = 0.01
   513→
   514→    # Half-life estimation (simplified Ornstein-Uhlenbeck)
   515→    reversion_speed = 0.1  # base reversion per step
   516→    reversion_strength = -reversion_speed * deviation
   517→
   518→    predictions = []
   519→    last_price = prices[-1]
   520→    for step in range(steps):
   521→        # Reversion decays as we approach mean
   522→        current_dev = (last_price - target_mean) / target_mean
   523→        adj_return = -reversion_speed * current_dev * 0.5
   524→        last_price = last_price * (1 + adj_return)
   525→        predictions.append(last_price)
   526→
   527→    return predictions
   528→

   531→# Ensemble Prediction
   532→# ─────────────────────────────────────────────────────────────────────────────

   534→
   535→def ensemble_predict(prices, volume=None, steps=5):
   536→    """Combine three models with adaptive weights."""
   537→    if len(prices) < 10:
   538→        # Not enough data — return simple extrapolation
   539→        last = prices[-1]
   540→        ret = (prices[-1] / prices[-2] - 1) if len(prices) >= 2 and prices[-2] != 0 else 0
   541→        return {
   542→            "predictions": [last * (1 + ret * (i + 1) * 0.3) for i in range(steps)],
   543→            "confidence": 0.1,
   544→            "conf_intervals": [[0, 0]] * steps,
   545→            "predicted_return_5d": ret * 1.5,
   546→        }
   547→
   548→    steps = max(1, min(steps, 30))
   549→
   550→    # --- Build dataset & fit Ridge ---
   551→    X, y = build_dataset(prices, volume)
   552→    ridge_preds = None
   553→    ridge_score = 0.0
   554→    residuals = []
   555→
   556→    if len(X) >= 10:
   557→        # Use last 30% for validation
   558→        split = int(len(X) * 0.7)
   559→        X_train, X_val = X[:split], X[split:]
   560→        y_train, y_val = y[:split], y[split:]
   561→
   562→        model = RidgeRegression(alpha=0.5)
   563→        model.fit(X_train, y_train)
   564→
   565→        if model.fitted and X_val:
   566→            val_preds = model.predict(X_val)
   567→            # R² score
   568→            ym = _mean(y_val)
   569→            ss_res = sum((yp - yt) ** 2 for yp, yt in zip(val_preds, y_val))
   570→            ss_tot = sum((yt - ym) ** 2 for yt in y_val)
   571→            ridge_score = 1 - ss_res / ss_tot if ss_tot > 0 else 0.0
   572→            ridge_score = max(0.0, min(1.0, ridge_score))
   573→            residuals = [yp - yt for yp, yt in zip(val_preds, y_val)]
   574→
   575→        # Multi-step prediction
   576→        ridge_preds = []
   577→        last_price = prices[-1]
   578→        for step in range(steps):
   579→            feats = build_features(prices, volume)
   580→            if feats and model.fitted:
   581→                pred_return = model.predict_single(feats)
   582→                pred_return = max(-0.05, min(0.05, pred_return))  # clamp
   583→            else:
   584→                rets = compute_returns(prices)
   585→                pred_return = _mean(rets[-5:]) if len(rets) >= 5 else (_mean(rets) if rets else 0)
   586→
   587→            last_price = last_price * (1 + pred_return)
   588→            ridge_preds.append(last_price)
   589→            # Append prediction to prices for next step's features
   590→            prices = prices + [last_price]
   591→            if volume:
   592→                avg_vol = _mean(volume[-5:]) if len(volume) >= 5 else _mean(volume)
   593→                volume = volume + [avg_vol]
   594→
   595→        prices = prices[:-(steps)]  # restore original
   596→        if volume:
   597→            volume = volume[:-(steps)]
   598→    else:
   599→        ridge_preds = [prices[-1]] * steps
   600→        ridge_score = 0.0
   601→
   602→    # --- WMA predictions ---
   603→    wma_preds = wma_predict(prices, steps)
   604→
   605→    # --- Mean-reversion predictions ---
   606→    mr_preds = mean_reversion_predict(prices, steps)
   607→
   608→    # --- Compute model weights based on recent performance ---
   609→    # Validate WMA and MR on recent data
   610→    wma_score = 0.0
   611→    mr_score = 0.0
   612→
   613→    if len(prices) >= 15:
   614→        recent = prices[-15:]
   615→        actual_next = prices[-5:] if len(prices) >= 20 else prices[-(len(prices) - 15):]
   616→        wma_test = wma_predict(prices[-20:-5] if len(prices) >= 20 else prices[:-5], len(actual_next))
   617→        mr_test = mean_reversion_predict(prices[-20:-5] if len(prices) >= 20 else prices[:-5], len(actual_next))
   618→
   619→        for pred_list, score_ref in [(wma_test, 'wma'), (mr_test, 'mr')]:
   620→            if len(pred_list) == len(actual_next) and actual_next:
   621→                ss_res = sum((p - a) ** 2 for p, a in zip(pred_list, actual_next))
   622→                ss_tot = sum((a - _mean(actual_next)) ** 2 for a in actual_next)
   623→                sc = 1 - ss_res / ss_tot if ss_tot > 0 else 0.0
   624→                sc = max(0.0, min(1.0, sc))
   625→                if score_ref == 'wma':
   626→                    wma_score = sc
   627→                else:
   628→                    mr_score = sc
   629→
   630→    # Adaptive ensemble weights
   631→    scores = [ridge_score + 0.05, wma_score + 0.05, mr_score + 0.05]  # floor at 0.05
   632→    total_score = sum(scores)
   633→    w = [s / total_score for s in scores]
   634→
   635→    # If ridge has very little data, favor simpler models
   636→    if len(X) < 10:
   637→        w = [0.1, 0.45, 0.45]
   638→
   639→    # --- Combine predictions ---
   640→    final_preds = []
   641→    for i in range(steps):
   642→        p = w[0] * ridge_preds[i] + w[1] * wma_preds[i] + w[2] * mr_preds[i]
   643→        final_preds.append(p)
   644→
   645→    # --- Confidence & intervals ---
   646→    if residuals:
   647→        resid_std = _stdev(residuals)
   648→    else:
   649→        rets = compute_returns(prices)
   650→        resid_std = _stdev(rets[-10:]) if len(rets) >= 10 else (_stdev(rets) if rets else 0.01)
   651→
   652→    confidence = max(0.1, min(0.95, ridge_score * 0.6 + wma_score * 0.2 + mr_score * 0.2 + 0.1))
   653→
   654→    conf_intervals = []
   655→    for i, p in enumerate(final_preds):
   656→        # Widen intervals for longer horizons
   657→        width = resid_std * p * (1 + 0.15 * i) * 1.96
   658→        conf_intervals.append([p - width, p + width])
   659→
   660→    # 5-day predicted return
   661→    if len(final_preds) >= 5:
   662→        predicted_return_5d = final_preds[4] / final_preds[0] - 1 if final_preds[0] != 0 else 0
   663→    elif final_preds:
   664→        predicted_return_5d = final_preds[-1] / prices[-1] - 1 if prices[-1] != 0 else 0
   665→    else:
   666→        predicted_return_5d = 0.0
   667→
   668→    return {
   669→        "predictions": [round(p, 6) for p in final_preds],
   670→        "confidence": round(confidence, 4),
   671→        "conf_intervals": [[round(lo, 6), round(hi, 6)] for lo, hi in conf_intervals],
   672→        "predicted_return_5d": round(predicted_return_5d, 6),
   673→    }
   674→

   677→# Endpoint 2: Bayesian Weights
   678→# ─────────────────────────────────────────────────────────────────────────────
   679→
   680→
   681→def compute_bayesian_weights(features, regime="sideways", has_volume=True):
   682→    """
   683→    Compute Bayesian weights for technical indicators.
   684→    W_i = P(Success|Tool_i) * Prior_i / sum(P(Success|Tool_j) * Prior_j)
   685→    """
   686→    # Default priors
   687→    priors = {
   688→        "trend": 0.28,
   689→        "oscillator": 0.25,
   690→        "volume": 0.15,
   691→        "volatility": 0.10,
   692→        "leading": 0.12,
   693→        "patterns": 0.10,
   694→    }
   695→
   696→    # Redistribute volume prior if no volume data
   697→    if not has_volume:
   698→        vol_prior = priors["volume"]
   699→        del priors["volume"]
   700→        total_other = sum(priors.values())
   701→        for k in priors:
   702→            priors[k] += vol_prior * (priors[k] / total_other)
   703→
   704→    # Likelihood P(Success|Tool_i) based on regime and feature values
   705→    likelihoods = {}
   706→
   707→    # Trend indicators: more useful in trending markets
   708→    trend_indicators = ["sma_20", "ema_12", "ema_26", "macd_signal", "adx"]
   709→    oscillator_indicators = ["rsi", "stoch_k", "stoch_d", "williams_r", "cci"]
   710→    volume_indicators = ["obv", "vwap", "vol_sma", "vol_ratio"]
   711→    volatility_indicators = ["atr", "bb_width", "bb_position", "keltner_width"]
   712→    leading_indicators = ["ichimoku", "pivot", "fibonacci", "donchian"]
   713→    pattern_indicators = ["candle_patterns", "harmonic", "chart_patterns", "support_resistance"]
   714→
   715→    # Count available features per category
   716→    def count_matching(keys):
   717→        return sum(1 for k in keys if k in features and features[k] is not None)
   718→
   719→    trend_count = count_matching(trend_indicators)
   720→    osc_count = count_matching(oscillator_indicators)
   721→    vol_count = count_matching(volume_indicators) if has_volume else 0
   722→    var_count = count_matching(volatility_indicators)
   723→    lead_count = count_matching(leading_indicators)
   724→    pat_count = count_matching(pattern_indicators)
   725→
   726→    # Adjust likelihoods based on regime
   727→    regime_boost = {
   728→        "strong_bull": {"trend": 1.3, "oscillator": 0.8, "volatility": 0.7, "leading": 1.1, "patterns": 0.9},
   729→        "weak_bull": {"trend": 1.1, "oscillator": 1.0, "volatility": 0.9, "leading": 1.0, "patterns": 1.0},
   730→        "sideways": {"trend": 0.7, "oscillator": 1.2, "volatility": 1.2, "leading": 0.9, "patterns": 1.1},
   731→        "weak_bear": {"trend": 1.0, "oscillator": 1.1, "volatility": 1.0, "leading": 1.0, "patterns": 1.0},
   732→        "strong_bear": {"trend": 1.3, "oscillator": 0.8, "volatility": 0.7, "leading": 1.1, "patterns": 0.9},
   733→    }
   734→
   735→    boost = regime_boost.get(regime, regime_boost["sideways"])
   736→
   737→    # Feature availability boost (more features = higher confidence)
   738→    def feature_boost(count):
   739→        if count == 0:
   740→            return 0.3
   741→        elif count == 1:
   742→            return 0.7
   743→        elif count == 2:
   744→            return 0.9
   745→        else:
   746→            return 1.0
   747→
   748→    # Base likelihood for each category
   749→    base_likelihood = {
   750→        "trend": 0.6,
   751→        "oscillator": 0.6,
   752→        "volatility": 0.55,
   753→        "leading": 0.5,
   754→        "patterns": 0.5,
   755→    }
   756→
   757→    if has_volume:
   758→        base_likelihood["volume"] = 0.55
   759→
   760→    # Compute final likelihoods
   761→    for cat in base_likelihood:
   762→        count = {
   763→            "trend": trend_count,
   764→            "oscillator": osc_count,
   765→            "volume": vol_count,
   766→            "volatility": var_count,
   767→            "leading": lead_count,
   768→            "patterns": pat_count,
   769→        }.get(cat, 0)
   770→
   771→        fb = feature_boost(count)
   772→        rb = boost.get(cat, 1.0)
   773→        likelihoods[cat] = base_likelihood[cat] * fb * rb
   774→
   775→    # Extra: boost based on extreme feature values
   776→    # e.g., extreme RSI means oscillators are more informative
   777→    rsi_val = features.get("rsi")
   778→    if rsi_val is not None:
   779→        if rsi_val > 70 or rsi_val < 30:
   780→            likelihoods["oscillator"] *= 1.2
   781→
   782→    macd_val = features.get("macd_hist")
   783→    if macd_val is not None and abs(macd_val) > 0:
   784→        likelihoods["trend"] *= 1.1
   785→
   786→    adx_val = features.get("adx")
   787→    if adx_val is not None and adx_val > 25:
   788→        likelihoods["trend"] *= 1.15
   789→
   790→    # Compute posterior weights: W_i = P(Success|i) * Prior_i / Z
   791→    posterior = {}
   792→    Z = 0.0
   793→    for cat in priors:
   794→        posterior[cat] = likelihoods.get(cat, 0.5) * priors[cat]
   795→        Z += posterior[cat]
   796→
   797→    if Z > 0:
   798→        for cat in posterior:
   799→            posterior[cat] /= Z
   800→
   801→    # Build per-indicator weights
   802→    weights = {}
   803→    all_indicators = {
   804→        "trend": trend_indicators,
   805→        "oscillator": oscillator_indicators,
   806→        "volume": volume_indicators,
   807→        "volatility": volatility_indicators,
   808→        "leading": leading_indicators,
   809→        "patterns": pattern_indicators,
   810→    }
   811→
   812→    for cat, indicators in all_indicators.items():
   813→        cat_weight = posterior.get(cat, 0.0)
   814→        present = [ind for ind in indicators if ind in features and features[ind] is not None]
   815→        if present:
   816→            per_ind = cat_weight / len(present)
   817→            for ind in present:
   818→                weights[ind] = round(per_ind, 4)
   819→        elif cat in priors:
   820→            # No features in this category — distribute to others later
   821→            pass
   822→
   823→    # Normalize so sum = 1
   824→    total = sum(weights.values())
   825→    if total > 0:
   826→        weights = {k: round(v / total, 4) for k, v in weights.items()}
   827→    else:
   828→        # Fallback: equal weights
   829→        n_keys = len(features)
   830→        if n_keys > 0:
   831→            w = 1.0 / n_keys
   832→            weights = {k: round(w, 4) for k in features if features[k] is not None}
   833→
   834→    return {
   835→        "weights": weights,
   836→        "prior_trend": round(priors.get("trend", 0), 4),
   837→        "prior_oscillator": round(priors.get("oscillator", 0), 4),
   838→        "prior_volume": round(priors.get("volume", 0), 4) if has_volume else 0,
   839→        "prior_volatility": round(priors.get("volatility", 0), 4),
   840→        "prior_leading": round(priors.get("leading", 0), 4),
   841→        "prior_patterns": round(priors.get("patterns", 0), 4),
   842→    }
   843→
   844→
   845→# ─────────────────────────────────────────────────────────────────────────────
   846→# Endpoint 3: Regime Detection
   847→# ─────────────────────────────────────────────────────────────────────────────
   848→
   849→
   850→def detect_regime(prices):
   851→    """Detect market regime from price series.
   852→    Returns one of: strong_bull, weak_bull, sideways, weak_bear, strong_bear
   853→    """
   854→    if len(prices) < 10:
   855→        return {"regime": "sideways", "probabilities": {
   856→            "strong_bull": 0.1, "weak_bull": 0.2, "sideways": 0.4,
   857→            "weak_bear": 0.2, "strong_bear": 0.1,
   858→        }}
   859→
   860→    # --- ADX-based trend strength ---
   861→    adx_val = adx(prices)
   862→    trend_strength = adx_val / 100.0  # 0-1
   863→
   864→    # --- Returns-based direction ---
   865→    rets = compute_returns(prices)
   866→    if not rets:
   867→        return {"regime": "sideways", "probabilities": {
   868→            "strong_bull": 0.1, "weak_bull": 0.2, "sideways": 0.4,
   869→            "weak_bear": 0.2, "strong_bear": 0.1,
   870→        }}
   871→
   872→    # Multi-period returns
   873→    ret_5 = sum(rets[-5:]) if len(rets) >= 5 else sum(rets)
   874→    ret_10 = sum(rets[-10:]) if len(rets) >= 10 else sum(rets)
   875→    ret_20 = sum(rets[-20:]) if len(rets) >= 20 else ret_10
   876→
   877→    # Average positive vs negative returns
   878→    pos_rets = [r for r in rets if r > 0]
   879→    neg_rets = [r for r in rets if r < 0]
   880→    pos_ratio = len(pos_rets) / len(rets) if rets else 0.5
   881→
   882→    # Direction score: positive = bullish, negative = bearish
   883→    direction_score = 0.0
   884→    # Recent 5-day return (most important)
   885→ direction_score += ret_5 * 3.0
   886→    # 10-day return
   887→    direction_score += ret_10 * 1.5
   888→    # 20-day return
   889→    direction_score += ret_20 * 1.0
   890→    # Positive ratio
   891→    direction_score += (pos_ratio - 0.5) * 0.3
   892→    # RSI contribution
   893→    rsi_val = rsi(prices)
   894→    direction_score += (rsi_val - 50) / 100.0 * 0.5
   895→
   896→    # --- Volatility-based ---
   897→    if len(rets) >= 10:
   898→        vol = _stdev(rets[-10:])
   899→    else:
   900→        vol = _stdev(rets)
   901→
   902→    # High volatility regime scoring
   903→    vol_score = min(1.0, vol * 10)  # normalize
   904→
   905→    # --- Classify into 5 regimes ---
   906→    # Thresholds
   907→    strong_trend_threshold = 0.4  # ADX equivalent
   908→    moderate_trend_threshold = 0.2
   909→
   910→    # Regime probabilities (softmax-like)
   911→    regimes = ["strong_bull", "weak_bull", "sideways", "weak_bear", "strong_bear"]
   912→    scores = {}
   913→
   914→    # Strong bull: high trend + positive direction + high vol
   915→    scores["strong_bull"] = (
   916→        max(0, trend_strength - moderate_trend_threshold) * 2.0 +
   917→        max(0, direction_score) * 1.5 +
   918→        vol_score * 0.3
   919→    )
   920→
   921→    # Weak bull: moderate trend + positive direction + low vol
   922→    scores["weak_bull"] = (
   923→        max(0, trend_strength - moderate_trend_threshold * 0.5) * 1.0 +
   924→        max(0, direction_score) * 1.0 +
   925→        (1 - vol_score) * 0.3
   926→    )
   927→
   928→    # Sideways: low trend + low direction
   929→    scores["sideways"] = (
   930→        (1 - trend_strength) * 1.5 +
   931→        (1 - min(1, abs(direction_score) * 3)) * 1.0 +
   932→        (1 - vol_score) * 0.2
   933→    )
   934→
   935→    # Weak bear: moderate trend + negative direction + low vol
   936→    scores["weak_bear"] = (
   937→        max(0, trend_strength - moderate_trend_threshold * 0.5) * 1.0 +
   938→        max(0, -direction_score) * 1.0 +
   939→        (1 - vol_score) * 0.3
   940→    )
   941→
   942→    # Strong bear: high trend + negative direction + high vol
   943→    scores["strong_bear"] = (
   944→        max(0, trend_strength - moderate_trend_threshold) * 2.0 +
   945→        max(0, -direction_score) * 1.5 +
   946→        vol_score * 0.3
   947→    )
   948→
   949→    # Softmax to get probabilities
   950→    max_score = max(scores.values())
   951→    exp_scores = {r: math.exp(s - max_score) for r, s in scores.items()}
   952→    total = sum(exp_scores.values())
   953→    probs = {r: round(exp_scores[r] / total, 4) for r in regimes}
   954→
   955→    # Determine dominant regime
   956→    best_regime = max(regimes, key=lambda r: probs[r])
   957→
   958→    return {
   959→        "regime": best_regime,
   960→        "probabilities": probs,
   961→    }
   962→

   964→# ─────────────────────────────────────────────────────────────────────────────
   965→# Endpoint 4: Divergence Detection
   966→# ─────────────────────────────────────────────────────────────────────────────
   967→
   968→
   969→def detect_divergence(prices, indicator_values, indicator_name="rsi"):
   970→    """Detect bullish/bearish divergence between price and an indicator."""
   971→    min_len = 10
   972→    if len(prices) < min_len or len(indicator_values) < min_len:
   973→        return {"type": "none", "strength": 0.0, "start_idx": -1}
   974→
   975→    # Align lengths
   976→    n = min(len(prices), len(indicator_values))
   977→    p = prices[-n:]
   978→    ind = indicator_values[-n:]
   979→
   980→    # Find recent swing highs and lows in price
   981→    lookback = min(n, 30)
   982→    window_p = p[-lookback:]
   983→    window_ind = ind[-lookback:]
   984→
   985→    # Find local maxima and minima (at least 3 bars apart)
   986→    def find_extrema(values, min_distance=3):
   987→        highs = []
   988→        lows = []
   989→        for i in range(1, len(values) - 1):
   990→            is_high = True
   991→            is_low = True
   992→            for j in range(max(0, i - min_distance), min(len(values), i + min_distance + 1))
   993→                if values[j] > values[i]:
   994→                    is_high = False
   995→                if values[j] < values[i]:
   996→                    is_low = False
   997→            if is_high and (not highs or i - highs[-1][0] >= min_distance):
   998→                highs.append((i, values[i]))
   999→            if is_low and (not lows or i - lows[-1][0] >= min_distance):
  1000→                lows.append((i, values[i]))
  1001→        return highs, lows
  1002→
  1003→    p_highs, p_lows = find_extrema(window_p)
  1004→    i_highs, i_lows = find_extrema(window_ind)
  1005→
  1006→    # --- Check bearish divergence (price higher high, indicator lower high) ---
  1007→    bearish_strength = 0.0
  1008→    bearish_start = -1
  1009→
  1010→    if len(p_highs) >= 2 and len(i_highs) >= 2:
  1011→        # Match the last two highs
  1012→        ph_last = p_highs[-1]
  1013→        ph_prev = p_highs[-2]
  1014→        ih_last = i_highs[-1]
  1015→        ih_prev = i_highs[-2]
  1016→
  1017→        # Price makes higher high, indicator makes lower high
  1018→        if ph_last[1] > ph_prev[1] and ih_last[1] < ih_prev[1]:
  1019→            price_diff = (ph_last[1] - ph_prev[1]) / ph_prev[1] if ph_prev[1] != 0 else 0
  1020→            ind_diff = (ih_prev[1] - ih_last[1]) / ih_prev[1] if ih_prev[1] != 0 else 0
  1021→            bearish_strength = min(1.0, (price_diff + ind_diff) * 10)
  1022→            bearish_start = ph_prev[0]
  1023→
  1024→    # --- Check bullish divergence (price lower low, indicator higher low) ---
  1025→    bullish_strength = 0.0
  1026→    bullish_start = -1
  1027→
  1028→    if len(p_lows) >= 2 and len(i_lows) >= 2:
  1029→        pl_last = p_lows[-1]
  1030→        pl_prev = p_lows[-2]
  1031→        il_last = i_lows[-1]
  1032→        il_prev = i_lows[-2]
  1033→
  1034→        # Price makes lower low, indicator makes higher low
  1035→        if pl_last[1] < pl_prev[1] and il_last[1] > il_prev[1]:
  1036→            price_diff = (pl_prev[1] - pl_last[1]) / pl_prev[1] if pl_prev[1] != 0 else 0
  1037→            ind_diff = (il_last[1] - il_prev[1]) / il_prev[1] if il_prev[1] != 0 else 0
  1038→            bullish_strength = min(1.0, (price_diff + ind_diff) * 10)
  1039→            bullish_start = pl_prev[0]
  1040→
  1041→    # Also check for simpler slope-based divergence
  1042→    if bearish_strength == 0 and bullish_strength == 0 and len(window_p) >= 10:
  1043→        # Compare slopes of price and indicator over the window
  1044→        p_slope = (window_p[-1] - window_p[-10]) / window_p[-10] if window_p[-10] != 0 else 0
  1045→        i_slope = (window_ind[-1] - window_ind[-10]) / abs(window_ind[-10]) if window_ind[-10] != 0 else 0
  1046→
  1047→        # Normalize indicator slope
  1048→        ind_range = max(window_ind) - min(window_ind)
  1049→        if ind_range > 0:
  1050→            i_slope_norm = (window_ind[-1] - window_ind[-10]) / ind_range
  1051→        else:
  1052→            i_slope_norm = 0
  1053→
  1054→        p_range = max(window_p) - min(window_p)
  1055→        if p_range > 0:
  1056→            p_slope_norm = (window_p[-1] - window_p[-10]) / p_range
  1057→        else:
  1058→            p_slope_norm = 0
  1059→
  1060→        # Divergence: opposite slopes
  1061→        if p_slope_norm > 0.1 and i_slope_norm < -0.1:
  1062→            bearish_strength = min(1.0, abs(p_slope_norm) + abs(i_slope_norm))
  1063→            bearish_start = len(window_p) - 10
  1064→        elif p_slope_norm < -0.1 and i_slope_norm > 0.1:
  1065→            bullish_strength = min(1.0, abs(p_slope_norm) + abs(i_slope_norm))
  1066→            bullish_start = len(window_p) - 10
  1067→
  1068→    # Determine result
  1069→    min_strength_threshold = 0.15
  1070→    if bullish_strength > bearish_strength and bullish_strength >= min_strength_threshold:
  1071→        return {
  1072→            "type": "bullish",
  1073→            "strength": round(bullish_strength, 4),
  1074→            "start_idx": bullish_start,
  1075→        }
  1076→    elif bearish_strength > bullish_strength and bearish_strength >= min_strength_threshold:
  1077→        return {
  1078→            "type": "bearish",
  1079→            "strength": round(bearish_strength, 4),
  1080→            "start_idx": bearish_start,
  1081→        }
  1082→    else:
  1083→        return {
  1084→            "type": "none",
  1085→            "strength": 0.0,
  1086→            "start_idx": -1,
  1087→        }
  1088→

  1162→    def _handle_predict(self, body):
  1163→        prices = body.get("prices")
  1164→        volume = body.get("volume")
  1165→        steps = body.get("steps", 5)
  1166→
  1167→        if not isinstance(prices, list) or len(prices) < 2:
  1168→            self._send_error("'prices' must be an array with at least 2 numbers")
  1169→            return
  1170→        if not all(isinstance(p, (int, float)) for p in prices):
  1171→            self._send_error("All price values must be numbers")
  1172→            return
  1173→        if any(p <= 0 for p in prices):
  1174→            self._send_error("All prices must be positive")
  1175→            return

  1177→            if not isinstance(volume, list) or len(volume) < 2:
  1178→                self._send_error("'volume' must be an array with at least 2 numbers or null")
