    32→// --- Global rate limiter (serial queue) ---
    33→let lastAITime = 0;
    34→const AI_MIN_INTERVAL = 20_000;
    35→let cooldownUntil = 0;
    36→let activeCall = false;
    37→const pendingQueue: Array<{
    38→  resolve: (text: string) => void;
    39→  reject: (err: Error) => void;
    40→  messages: { role: string; content: string }[];
    41→}> = [];
    42→
    43→function processAIQueue() {
    44→  if (activeCall || pendingQueue.length === 0) return;
    45→  const now = Date.now();
    46→
    47→  if (cooldownUntil > now) {
    48→    setTimeout(processAIQueue, Math.min(cooldownUntil - now, 5000));
    49→    return;
    50→  }
    51→
    52→  const elapsed = now - lastAITime;
    53→  if (elapsed < AI_MIN_INTERVAL) {
    54→    setTimeout(processAIQueue, AI_MIN_INTERVAL - elapsed);
    55→    return;
    56→  }
    57→
    58→  const item = pendingQueue.shift()!;
    59→  activeCall = true;
    60→
    61→  callZaiSDK(item.messages)
    62→    .then(text => {
    63→      lastAITime = Date.now();
    64→      cooldownUntil = 0;
    65→      item.resolve(text);
    66→    })
    67→    .catch(err => {
    68→      const msg = err.message || '';
    69→      if (msg.includes('429')) {
    70→        const cd = Math.max(cooldownUntil - Date.now(), 0) + 120_000;
    71→        cooldownUntil = Date.now() + Math.min(cd, 300_000);
    72→        console.warn(`[AI v5] 429, cooldown ${Math.min(cd, 300_000) / 1000}s`);
    73→      }
    74→      item.reject(err);
    75→    })
    76→    .finally(() => {
    77→      activeCall = false;
    78→      setTimeout(processAIQueue, 2000);
    79→    });
    80→}

    82→function queueAI(messages: { role: string; content: string }[], timeoutMs = 45_000): Promise<string> {
    83→  return new Promise((resolve, reject) => {
    84→    const timer = setTimeout(() => {
    85→      const idx = pendingQueue.findIndex(q => q.messages === messages);
    86→      if (idx >= 0) pendingQueue.splice(idx, 1);
    87→      reject(new Error('AI request timeout'));
    88→    }, timeoutMs);
    89→
    90→    pendingQueue.push({
    91→      messages,
    92→      resolve: (text) => { clearTimeout(timer); resolve(text); },
    93→      reject: (err) => { clearTimeout(timer); reject(err); },
    94→    });
    95→
    96→    processAIQueue();
    97→  });
    98→}

   100→// --- Z.ai SDK call with retry ---
   101→function sleep(ms: number) { return new Promise(r => setTimeout(r, ms)); }
   102→function jitter(base: number): number { return base + Math.random() * base * 0.5; }

   104→async function callZaiSDK(messages: { role: string; content: string }[], maxRetries = 1): Promise<string> {
   105→  for (let attempt = 1; attempt <= maxRetries; attempt++) {
   106→    try {
   107→      let zai;
   108→      try {
   109→        zai = await getZai();
   110→      } catch (initErr: unknown) {
   111→        const initMsg = initErr instanceof Error ? initErr.message : String(initErr);
   112→        console.error(`[AI v5] SDK init failed (attempt ${attempt}):`, initMsg);
   113→        if (attempt < maxRetries) {
   114→          await sleep(3000);
   115→          continue;
   116→        }
   117→        throw new Error('خطا در اتصال به سرور هوش مصنوعی. لطفاً صفحه را رفرش کنید.');
   118→      }
   119→
   120→      const completion = await Promise.race([
   121→        zai.chat.completions.create({
   122→          messages,
   123→          thinking: { type: 'disabled' },
   124→        }),
   125→        new Promise<never>((_, reject) => setTimeout(() => reject(new Error('AI SDK timeout')), 30_000)),
   126→      ]);
   127→      const raw = completion.choices[0]?.message?.content;
   128→      if (!raw || raw.trim().length === 0) {
   129→        throw new Error('Empty AI response');
   130→      }
   131→      const text = raw.trim();
   132→      if (text.length < 10) {
   133→        throw new Error('AI response too short: ' + text.slice(0, 100));
   134→      }
   135→      return text;
   136→    } catch (err: unknown) {
   137→      const msg = err instanceof Error ? err.message : String(err);
   138→      console.error(`[AI v5] attempt ${attempt}/${maxRetries} error:`, msg);
   139→      if (msg.includes('429') && attempt < maxRetries) {
   140→        console.warn(`[AI v5] 429 retry ${attempt}/${maxRetries}, waiting 10s...`);
   141→        await sleep(10_000);
   142→        continue;
   143→      }
   144→      throw err;
   145→    }
   146→  }
   147→  throw new Error('Max retries exceeded');
   148→}

   151→function toPersianNum(n: number): string {
   152→  if (!isFinite(n) || isNaN(n)) return '\u06f0';
   153→  return Math.round(n).toLocaleString('fa-IR');

   156→function srGrade(strength: number): string {
   157→  if (strength >= 8.5) return '\u0628\u0633\u06cc\u0627\u0631 \u0642\u0648\u06cc';
   158→  if (strength >= 7) return '\u0642\u0648\u06cc';
   159→  if (strength >= 5) return '\u0645\u062a\u0648\u0633\u0637';
   160→  if (strength >= 3) return '\u0636\u0639\u06cc\u0641';
   161→  return '\u0628\u0633\u06cc\u0627\u0631 \u0636\u0639\u06cc\u0641';
   162→}

   165→function buildMLInput(body: Record<string, unknown>): MLSelectorInput {
   166→  const price = (body.currentPrice as number) || 0;
   167→  const trendDir = (body.trendDirection as string) || 'range';
   168→  const adx = (body.adx as number) || 0;
   169→  const rsi = (body.rsi as number) || 50;
   170→  const stochK = (body.stochK as number) || 50;
   171→  const macdHist = (body.macdHist as number) || 0;
   172→  const obv = (body.obv as number) || 0;
   173→  const bollingerUpper = (body.bollingerUpper as number) || 0;
   174→  const bollingerLower = (body.bollingerLower as number) || 0;
   175→  const resistance = (body.resistanceStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
   176→  const support = (body.supportStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
   177→  const atr = (body.atr as number) || 0;
   178→  const bbRange = bollingerUpper - bollingerLower;
   179→  const bbPosition = bbRange > 0 ? Math.min(100, Math.max(0, Math.round((price - bollingerLower) / bbRange * 100))) : 50;
   180→  const scenarios = body.scenarios as Record<string, { probability: number }> | undefined;
   181→  let dominantKey = 'R3';
   182→  let dominantProb = 0;
   183→  for (const k of ['R1', 'R2', 'R3', 'R4', 'R5'] as const) {
   184→    const p = scenarios?.[k]?.probability ?? 0;
   185→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
   186→  }
   187→  return { price, trend: trendDir === 'up' ? 'up' : trendDir === 'down' ? 'down' : 'range', adx, diPlus: (body.diPlus as number) || 0, diMinus: (body.diMinus as number) || 0, rsi, stochK, macdHist, obv, bbPosition, resistance, support, atr, scenarioDominant: dominantKey, hasVolume: (body.hasVolume as boolean) ?? false };
   188→}

   190→// --- Build MSL v4 Context ---
   191→function buildMSLV4Context(body: Record<string, unknown>): MSLV4Context {
   192→  const price = (body.currentPrice as number) || 0;
   193→  const atr = (body.atr as number) || 0;
   194→  const atrRatio = price > 0 ? atr / price : 0;
   195→
   196→  // Dominant scenario from v11 or legacy scenarios
   197→  const v11Probs = body.v11Probabilities as {
   198→    scenarios: Array<{ key: string; name: string; rawProbability: number }>;
   199→  } | undefined;
   200→  let dominantScenario = '';
   201→  if (v11Probs?.scenarios) {
   202→    const sorted = [...v11Probs.scenarios].sort((a, b) => b.rawProbability - a.rawProbability);
   203→    dominantScenario = sorted[0]?.name ?? '';
   204→  } else {
   205→    const scenarios = body.scenarios as Record<string, { name?: string; probability: number }> | undefined;
   206→    let maxProb = 0;
   207→    for (const k of ['R1', 'R2', 'R3', 'R4', 'R5', 'R6', 'R7', 'R8', 'R9']) {
   208→      const p = scenarios?.[k]?.probability ?? 0;
   209→      if (p > maxProb) { maxProb = p; dominantScenario = scenarios?.[k]?.name ?? k; }
   210→    }
   211→  }
   212→
   213→  const trendDir = (body.trendDirection as string) || 'range';
   214→  const isBull = trendDir === 'up';
   215→  const isBear = trendDir === 'down';
   216→  const adx = (body.adx as number) || 0;
   217→  const trendStrength = adx > 40 ? 0.9 : adx > 25 ? 0.7 : adx > 15 ? 0.4 : 0.15;
   218→
   219→  let regime: MSLV4Context['regime'] = 'Range';
   220→  if (isBull && trendStrength > 0.6) regime = 'Strong Bull';
   221→  else if (isBull) regime = 'Weak Bull';
   222→  else if (isBear && trendStrength > 0.6) regime = 'Strong Bear';
   223→  else if (isBear) regime = 'Weak Bear';
   224→
   225→  const instrumentType = (body.instrumentType as string) ||
   226→    (body.symbolName as string) || 'stock';
   227→
   228→  const ta = body.technicalAnalysis as {
   229→    classicPatternCount?: number;
   230→    harmonicPatternCount?: number;
   231→    elliottWaveCount?: number;
   232→    divergenceCount?: number;
   233→  } | undefined;
   234→
   235→  const overallConfidence = (body.overallConfidence as number) ??
   236→    (body.confidence as number) ?? 0.6;
   237→
   238→  return {
   239→    asset: instrumentType,
   240→    timeframe: (body.timeframe as string) || 'daily',
   241→    regime,
   242→    confidence: Math.max(0, Math.min(1, overallConfidence)),
   243→    volatility: atrRatio,
   244→    trend_strength: trendStrength,
   245→    dominantScenario,
   246→    pattern_counts: {
   247→      classical: ta?.classicPatternCount ?? 0,
   248→      harmonic: ta?.harmonicPatternCount ?? 0,
   249→      elliott: ta?.elliottWaveCount ?? 0,
   250→    },
   251→    divergence_present: (ta?.divergenceCount ?? 0) > 0,
   252→    hasVolume: (body.hasVolume as boolean) ?? false,
   253→    rsi: (body.rsi as number) || 50,
   254→    adx,
   255→    audience_level: (body.audienceLevel as string) || 'pro',
   256→  };
   257→}
