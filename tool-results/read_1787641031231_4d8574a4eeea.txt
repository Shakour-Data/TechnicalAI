   108→  { key: 'up', label: 'صعودی' },
   109→  { key: 'pullback', label: 'رنج و خنثی' },

   115→// Component
   116→// ═══════════════════════════════════════════════════════════════════════════════
   117→
   118→export default function VdssGraph(props: VdssGraphProps) {
   119→  const {
   120→    symbolName,
   121→    currentPrice,
   122→    scenarios,
   123→    decisionGraph,
   124→  } = props;
   125→
   126→  const [activeFilter, setActiveFilter] = useState('all');
   127→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   128→  const graphRef = useRef<HTMLDivElement>(null);
   129→  const svgRef = useRef<SVGSVGElement>(null);
   130→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   131→  const shellRef = useRef<HTMLDivElement>(null);
   132→
   133→  // ── Derived data from decisionGraph ──────────────────────────────
   134→  const edges = decisionGraph?.edges ?? [];
   135→  const nodes = decisionGraph?.nodes ?? [];
   136→  const nodePositions = decisionGraph?.nodePositions ?? {};
   137→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   138→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   139→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   140→  const pathContributions = decisionGraph?.pathContributions ?? {};
   141→  const nodeValues = decisionGraph?.nodeValues ?? {};
   142→
   143→  // Build node lookup map
   144→  const nodeMap = useMemo(() => {
   145→    const map: Record<string, (typeof nodes)[number]> = {};
   146→    for (const n of nodes) map[n.id] = n;
   147→    return map;
   148→  }, [nodes]);
   149→
   150→  // ── Visible edges/nodes based on filter ──────────────────────────
   151→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   152→    if (activeFilter === 'all') {
   153→      const allEdgeIdx = edges.map((_, i) => i);
   154→      const allNodes = new Set(nodes.map(n => n.id));
   155→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   156→    }
   157→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   158→      // Show edges going TO this scenario node, plus their source nodes
   159→      const edgeSet = new Set<number>();
   160→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   161→      edges.forEach((e, i) => {
   162→        if (e.to === activeFilter) {
   163→          edgeSet.add(i);
   164→          nodeSet.add(e.from);
   165→        }
   166→      });
   167→      // Also show intermediate nodes connected to those sources
   168→      edges.forEach((e, i) => {
   169→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   170→          edgeSet.add(i);
   171→          nodeSet.add(e.from);
   172→        }
   173→      });
   174→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   175→    }
   176→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   177→      const branchType = `branch-${activeFilter}` as const;
   178→      const edgeSet = new Set<number>();
   179→      const nodeSet = new Set<string>(['ROOT']);
   180→      edges.forEach((e, i) => {
   181→        if (e.type === branchType || (e.type !== `branch-trend` && e.type !== `branch-breakout` && e.type !== `branch-reversal` && nodeSet.has(e.from))) {
   182→          // Include branch-type edges
   183→          if (e.type === branchType) {
   184→            edgeSet.add(i);
   185→            nodeSet.add(e.from);
   186→            nodeSet.add(e.to);
   187→          }
   188→        }
   189→      });
   190→      // Now add all edges whose source is already in nodeSet
   191→      const expanded = new Set<number>();
   192→      const expandedNodes = new Set<string>(nodeSet);
   193→      edges.forEach((e, i) => {
   194→        if (expandedNodes.has(e.from)) {
   195→          expanded.add(i);
   196→          expandedNodes.add(e.to);
   197→        }
   198→      });
   199→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   200→    }
   201→    // Type filter (up/pullback/down/risk)
   202→    const edgeSet = new Set<number>();
   203→    const nodeSet = new Set<string>(['ROOT']);
   204→    edges.forEach((e, i) => {
   205→      if (e.type === activeFilter) {
   206→        edgeSet.add(i);
   207→        nodeSet.add(e.from);
   208→        nodeSet.add(e.to);
   209→      }
   210→    });
   211→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   212→  }, [activeFilter, edges, nodes]);
   213→
   214→  // ── Draw SVG edges ───────────────────────────────────────────────
   215→  const drawEdges = useCallback(() => {
   216→    const svg = svgRef.current;
   217→    const graph = graphRef.current;
   218→    if (!svg || !graph || edges.length === 0) return;
   219→
   220→    let markersSvg = '';
   221→    const seenColors = new Set<string>();
   222→    for (const e of edges) {
   223→      const c = EDGE_COLORS[e.type];
   224→      if (c && !seenColors.has(e.type)) {
   225→        seenColors.add(e.type);
   226→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="8" markerHeight="8" refX="7" refY="4" orient="auto"><path d="M0,0 L8,4 L0,8 z" fill="${c}"/></marker>`;
   227→      }
   228→    }
   229→
   230→    let pathsSvg = '';
   231→    const graphRect = graph.getBoundingClientRect();
   232→
   233→    for (let ei = 0; ei < edges.length; ei++) {
   234→      const { from: fromId, to: toId, label, type } = edges[ei];
   235→      const isVisible = visibleEdgeIndices.includes(ei);
   236→      const fromEl = nodeRefs.current[fromId];
   237→      const toEl = nodeRefs.current[toId];
   238→      if (!fromEl || !toEl) continue;
   239→
   240→      const fRect = fromEl.getBoundingClientRect();
   241→      const tRect = toEl.getBoundingClientRect();
   242→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   243→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   244→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   245→      const by = tRect.top - graphRect.top + tRect.height / 2;
   246→      const dx = bx - ax;
   247→      const dy = by - ay;
   248→      const dist = Math.sqrt(dx * dx + dy * dy);
   249→      if (dist === 0) continue;
   250→
   251→      // Curved edges with bend proportional to distance
   252→      const bend = Math.min(40, Math.max(12, dist * 0.08));
   253→      const mx = (ax + bx) / 2;
   254→      const my = (ay + by) / 2;
   255→      const nx = -dy / dist;
   256→      const ny = dx / dist;
   257→      const cx = mx + nx * bend;
   258→      const cy = my + ny * bend;
   259→
   260→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   261→      const prob = edgeProbabilities[ei] ?? 0;
   262→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   263→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   264→
   265→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.7 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   266→
   267→      // Edge labels — skip deterministic root→branch and branch→event edges (index < 6)
   268→      if (isVisible && ei >= 6) {
   269→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#374151" font-size="8" text-anchor="middle" paint-order="stroke" stroke="#ffffff" stroke-width="3" stroke-linejoin="round" opacity="0.8" data-type="${type}" class="edge-label">${label}</text>`;
   270→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#ffffff" stroke-width="2.5" stroke-linejoin="round" opacity="0.85" class="edge-prob">${probLabel}</text>`;
   271→      }
   272→    }
   273→
   274→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   275→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   276→
   277→  useEffect(() => { drawEdges(); }, [drawEdges]);
   278→
   279→  useEffect(() => {
   280→    const el = shellRef.current;
   281→    if (!el) return;
   282→    const observer = new ResizeObserver(() => drawEdges());
   283→    observer.observe(el);
   284→    return () => observer.disconnect();
   285→  }, [drawEdges]);
   286→
   287→
   288→  // ── Detail panel content ─────────────────────────────────────────
   289→  const detailContent = useMemo(() => {
   290→    if (!selectedNode) return null;
   291→    const node = nodeMap[selectedNode];
   292→    if (!node) return null;
   293→
   294→    // For terminal scenario nodes, show branch breakdown
   295→    if (node.isTerminal && SCENARIO_KEYS.includes(selectedNode as typeof SCENARIO_KEYS[number])) {
   296→      const contrib = pathContributions[selectedNode] ?? { trend: 0, breakout: 0, reversal: 0 };
   297→      const meta = SCENARIO_META[selectedNode];
   298→      const s = scenarios[selectedNode as keyof typeof scenarios];
   299→      return (
   300→        <div className="space-y-3">
   301→          <h3 className="text-sm font-bold" style={{ color: meta.color }}>{node.title}</h3>
   302→          <p className="text-xs text-[#374151]">{node.desc}</p>
   303→          <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border border-[#e5e7eb] bg-[#f3f4f6] text-[#374151]">{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   304→          <div className="text-xs text-[#6b7280]" dir="ltr">{nodeValues[selectedNode] ?? '--'}</div>
   305→
   306→          <div className="mt-3 pt-3 border-t border-[#e5e7eb]">
   307→            <p className="text-xs font-bold text-[#374151] mb-2">سهم هر استراتژی:</p>
   308→            {Object.entries(BRANCH_META).map(([bKey, bMeta]) => {
   309→              const val = contrib[bKey as 'trend' | 'breakout' | 'reversal'];
   310→              const pct = (val * 100).toFixed(1);
   311→              return (
   312→                <div key={bKey} className="flex items-center justify-between py-1.5 border-b border-dashed border-[#e5e7eb]">
   313→                  <div className="flex items-center gap-1.5">
   314→                    <span className="inline-block w-2 h-2 rounded-full" style={{ background: bMeta.color }} />
   315→                    <span className="text-[11px] text-[#374151]">{bMeta.label}</span>
   316→                  </div>
   317→                  <span className="text-[11px] font-bold" style={{ color: bMeta.color }}>{toPersianDigits(pct)}٪</span>
   318→                </div>
   319→              );
   320→            })}
   321→            <div className="flex items-center justify-between py-1.5">
   322→              <span className="text-[11px] font-bold text-[#111827]">مجموع</span>
   323→              <span className="text-[11px] font-black" style={{ color: meta.color }}>{toFa(s?.probability ?? 0)}٪</span>
   324→            </div>
   325→          </div>
   326→        </div>
   327→      );
   328→    }
   329→
   330→    // For non-terminal nodes, show incoming/outgoing edges
   331→    const inputs = edges.filter(e => e.to === selectedNode);
   332→    const outputs = edges.filter(e => e.from === selectedNode);
   333→
   334→    return (
   335→      <div className="space-y-3">
   336→        <h3 className="text-sm font-bold text-[#111827]">{node.title}</h3>
   337→        <p className="text-xs text-[#374151]"><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   338→        <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border border-[#e5e7eb] bg-[#f3f4f6] text-[#374151]">{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   339→        <p className="text-xs text-[#6b7280] leading-relaxed">{node.desc}</p>
   340→        {inputs.length > 0 && (
   341→          <div>
   342→            <p className="text-xs font-medium text-[#374151] mb-1">مسیرهای ورودی ({toFa(inputs.length)}):</p>
   343→            <ul className="space-y-1">
   344→              {inputs.map((e, i) => {
   345→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   346→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   347→                return (
   348→                  <li key={i} className="text-[11px] text-[#6b7280] leading-relaxed border-t border-dashed border-[#e5e7eb] pt-1.5">
   349→                    <b className="text-[#374151]">{SCENARIO_DISPLAY[e.from] || e.from} ← {SCENARIO_DISPLAY[e.to] || e.to}</b>
   350→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${edgeColor}20`, color: edgeColor }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   351→                    <br />{e.label}
   352→                  </li>
   353→                );
   354→              })}
   355→            </ul>
   356→          </div>
   357→        )}
   358→        {outputs.length > 0 && (
   359→          <div>
   360→            <p className="text-xs font-medium text-[#374151] mb-1">مسیرهای خروجی ({toFa(outputs.length)}):</p>
   361→            <ul className="space-y-1">
   362→              {outputs.map((e, i) => {
   363→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   364→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   365→                return (
   366→                  <li key={i} className="text-[11px] text-[#6b7280] leading-relaxed border-t border-dashed border-[#e5e7eb] pt-1.5">
   367→                    <b className="text-[#374151]">{SCENARIO_DISPLAY[e.from] || e.from} → {SCENARIO_DISPLAY[e.to] || e.to}</b>
   368→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${edgeColor}20`, color: edgeColor }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   369→                    <br />{e.label}
   370→                  </li>
   371→                );
   372→              })}
   373→            </ul>
   374→          </div>
   375→        )}
   376→      </div>
   377→    );
   378→  }, [selectedNode, nodeMap, nodeValues, edges, edgeProbabilities, pathContributions, scenarios]);
   379→
   380→  // ── All filter buttons ────────────────────────────────────────────
   381→  const allFilters = [
   382→    ...TYPE_FILTERS,
   383→    { key: 'sep1', label: '│', isSep: true as const },
   384→    ...Object.entries(BRANCH_META).map(([k, v]) => ({ key: k, label: v.label, branchKey: k, branchColor: v.color })),
   385→    { key: 'sep2', label: '│', isSep: true as const },
   386→    ...SCENARIO_KEYS.map(k => ({ key: k, label: SCENARIO_META[k].label, scenarioKey: k })),
   387→  ];
   388→
   389→  // ═══ Design dimensions ═══
   390→  const DESIGN_W = 1500;
   391→  const DESIGN_H = 820;
   392→  const DISPLAY_W = 1200;
   393→  const DISPLAY_H = DESIGN_H;
   394→  const scaleX = DISPLAY_W / DESIGN_W;
   395→  const scaleY = DISPLAY_H / DESIGN_H;
   396→
   397→  // ── Loading state ────────────────────────────────────────────────
   398→  if (!decisionGraph) {
   399→    return <VdssGraphSkeleton />;
   400→  }
   401→
   402→  return (
   403→    <div className="space-y-3" dir="rtl">
   404→      {/* ═══ HEADER ═══ */}
   405→      <div className="flex items-center justify-between gap-4 px-4 py-3 rounded-2xl border border-[#e5e7eb]"
   406→        style={{ background: 'linear-gradient(105deg, #ffffff, #f3f4f6)', boxShadow: '0 4px 16px rgba(0,0,0,.06)' }}>
   407→        <div className="flex items-center gap-3">
   408→          <div className="w-10 h-10 rounded-xl border border-amber-200 flex items-center justify-center text-amber-800 text-xl font-bold"
   409→            style={{ boxShadow: 'inset 0 0 22px rgba(146,64,14,.06), 0 0 22px rgba(146,64,14,.04)' }}>◈</div>
   410→          <div>
   411→            <h2 className="text-base font-bold text-[#111827]">گراف تصمیم {symbolName}</h2>
   412→            <p className="text-[11px] text-[#6b7280]">مدل ۳‌شاخه‌ای | پیروی از روند، شکست، بازگشت | ۲۷ مسیر به ۹ سناریو</p>
   413→          </div>
   414→        </div>
   415→        <div className="text-left text-xs text-[#6b7280] leading-relaxed pr-4 border-r border-[#e5e7eb]">
   416→          نقطه مرجع: <b className="text-cyan-700">{toFa(currentPrice)}</b><br />
   417→          افق برآورد: ۱۰ تا ۲۵ جلسه معاملاتی
   418→        </div>
   419→      </div>
   420→
   421→      {/* ═══ Metric Cards ═══ */}

   812→                <TableHead className="text-right text-[#374151] font-bold px-2 py-2 min-w-[120px]">سناریو</TableHead>
   813→                {TREND_DAYS.map(day => (
   814→                  <TableHead key={day} colSpan={2} className="text-center text-[#374151] font-bold px-1.5 py-2 border-r border-[#e5e7eb]">
   815→                    روز {toPersianDigits(day.toString())}
   816→                  </TableHead>
   817→                ))}
   818→                <TableHead className="text-center text-[#374151] font-bold px-2 py-2 border-r border-[#e5e7eb]">اوج</TableHead>
   819→              </TableRow>
   820→              <TableRow className="border-b border-[#e5e7eb] bg-[#f9fafb]">
   821→                <TableHead className="px-2 py-1" />
   822→                {TREND_DAYS.map(day => (
   823→                  <React.Fragment key={day}>
   824→                    <TableHead className="text-center text-[10px] text-[#6b7280] font-medium px-1 py-1">اختصاصی</TableHead>
   825→                    <TableHead className="text-center text-[10px] text-[#6b7280] font-medium px-1 py-1">تجمعی</TableHead>
   826→                  </React.Fragment>
   827→                ))}
   828→                <TableHead className="text-center text-[10px] text-[#6b7280] font-medium px-2 py-1">روز</TableHead>
   829→              </TableRow>
   830→            </TableHeader>
   831→            <TableBody>
   832→              {/* Scenario rows */}
   833→              {['R1','R2','R3','R4','R5','R6','R7','R8','R9'].map(key => {
   834→                const sc = scenarioMap[key];
   835→                if (!sc) return null;
   836→                const gc = TREND_GROUP_COLOR[sc.group];
   837→                return (
   838→                  <TableRow key={key} className="border-b border-[#e5e7eb]/60 hover:bg-[#f9fafb]">
   839→                    <TableCell className="px-2 py-1.5 font-bold" style={{ color: gc.text, minWidth: 120 }}>
   840→                      <span className="inline-block w-2 h-2 rounded-full ml-1.5" style={{ background: gc.text }} />
   841→                      {sc.label}
   842→                    </TableCell>
   843→                    {TREND_DAYS.map(day => {
   844→                      const dp = getDay(sc.trend, day);
   845→                      const isPeak = sc.peakDay === day;
   846→                      return (
   847→                        <React.Fragment key={day}>
   848→                          <TableCell
   849→                            className={`text-center px-1 py-1 tabular-nums ${isPeak ? 'font-black' : 'font-medium'}`}
   850→                            style={isPeak ? { background: gc.badge, color: gc.text, borderRadius: 4 } : { color: '#374151' }}
   851→                          >
   852→                            {dp ? fmtPct(dp.individualProb) : '—'}
   853→                          </TableCell>
   854→                          <TableCell
   855→                            className={`text-center px-1 py-1 tabular-nums ${isPeak ? 'font-black' : 'font-medium'}`}
   856→                            style={isPeak ? { background: gc.badge, color: gc.text, borderRadius: 4 } : { color: '#6b7280' }}
   857→                          >
   858→                            {dp ? fmtPct(dp.cumulativeProb) : '—'}
   859→                          </TableCell>
   860→                        </React.Fragment>
   861→                      );
   862→                    })}
   863→                    <TableCell className="text-center px-2 py-1 font-bold tabular-nums" style={{ color: gc.text }}>
   864→                      {toPersianDigits(sc.peakDay.toString())}
   865→                    </TableCell>
   866→                  </TableRow>
   867→                );
   868→              })}
   869→
   870→              {/* Group trend rows */}
   871→              <TableRow className="border-t-2 border-[#e5e7eb] bg-[#f3f4f6]">
   872→                <TableCell className="px-2 py-2 font-black text-[#111827]" colSpan={1}>

   874→                </TableCell>
   875→                {TREND_DAYS.map(day => {
   876→                  const bearG = groupMap['bearish'];
   877→                  const neutG = groupMap['neutral'];
   878→                  const bullG = groupMap['bullish'];
   879→                  const bCum = getDay(bearG?.trend ?? [], day)?.cumulativeProb ?? 0;
   880→                  const nCum = getDay(neutG?.trend ?? [], day)?.cumulativeProb ?? 0;
   881→                  const buCum = getDay(bullG?.trend ?? [], day)?.cumulativeProb ?? 0;
   882→                  const total = bCum + nCum + buCum;
   883→                  const bullPct = total > 0 ? buCum / total : 0;
   884→                  const barColor = bullPct > 0.55 ? '#16a34a' : bullPct < 0.45 ? '#dc2626' : '#6b7280';
   885→                  return (
   886→                    <TableCell key={day} colSpan={2} className="text-center px-1 py-2">
   887→                      <div className="flex items-center justify-center gap-1">
   888→                        <div className="w-12 h-1.5 rounded-full bg-[#e5e7eb] overflow-hidden">
   889→                          <div className="h-full rounded-full" style={{ width: `${Math.min(100, bullPct * 100)}%`, background: barColor }} />
   890→                        </div>
   891→                        <span className="text-[10px] font-bold tabular-nums" style={{ color: barColor }}>
   892→                          {fmtPct(bullPct)}
   893→                        </span>
   894→                      </div>
   895→                    </TableCell>
   896→                  );
   897→                })}
   898→                <TableCell className="px-2 py-2" />
   899→              </TableRow>
   900→              {/* Group detail rows */}
   901→              {(['bearish', 'neutral', 'bullish'] as const).map(g => {
   902→                const gData = groupMap[g];
   903→                if (!gData) return null;
   904→                const gc = TREND_GROUP_COLOR[g];
   905→                const lastDay = getDay(gData.trend, 30);
   906→                return (
   907→                  <TableRow key={g} className="border-b border-[#e5e7eb]/40">
   908→                    <TableCell className="px-2 py-1.5 font-bold" style={{ color: gc.text }}>
   909→                      <span className="inline-block w-1.5 h-1.5 rounded-full ml-1" style={{ background: gc.text }} />
   910→                      {TREND_GROUP_LABEL[g]}
   911→                    </TableCell>
   912→                    {TREND_DAYS.map(day => {
   913→                      const dp = getDay(gData.trend, day);
   914→                      return (
   915→                        <TableCell key={day} colSpan={2} className="text-center px-1 py-1 tabular-nums font-medium" style={{ color: gc.text }}>
   916→                          {dp ? fmtPct(dp.cumulativeProb) : '—'}
   917→                        </TableCell>
   918→                      );
   919→                    })}
   920→                    <TableCell className="text-center px-2 py-1 font-bold tabular-nums text-[10px]" style={{ color: gc.text }}>
   921→                      {lastDay ? fmtPct(lastDay.cumulativeProb) : '—'}
   922→                    </TableCell>
   923→                  </TableRow>
   924→                );
   925→              })}
   926→            </TableBody>
   927→          </Table>
   928→
   929→          <div className="mt-3 px-4 py-2.5 rounded-lg border-r-3 border-emerald-700/60 bg-emerald-50 text-[11px] text-[#374151] leading-relaxed">
   930→            <b>توضیح:</b> احتمال اختصاصی = احتمال وقوع سناریو در آن روز خاص. احتمال تجمعی = مجموع تجمعی از روز ۱ تا آن روز.
   931→            سلول‌های برجسته نشان‌دهنده روز اوج احتمال هر سناریو هستند. نوار روند گروهی نسبت تجمعی گاوی به کل را نشان می‌دهد.
   932→          </div>
   933→        </div>
   934→      </CollapsibleContent>
   935→    </Collapsible>
   936→  );
   937→}
