   282→   282→function scoreSchool(input: NarrativeInput): ScoreFn {
   283→   283→  return () => {

   286→   286→

   296→   296→
   297→   297→    // Candlestick: near key levels (reversal zone)
   298→   298→    if (nearResistance && stochK > 75) s += 4;

   488→   488→export function buildNarrativeInput(body: {
   489→   489→  currentPrice: number;
   490→   490→  trendDirection: string;
   491→   491→  rsi: number;
   492→   492→  adx: number;
   493→   493→  stochK: number;
   494→   494→  stochD: number;
   495→   495→  macdLine: number;
   496→   496→  macdSignal: number;
   497→   497→  macdHist: number;
   498→   498→  diPlus: number;
   499→   499→  diMinus: number;
   500→   500→  obv: number;
   501→   501→  hasVolume: boolean;
   502→   502→  cci: number;
   503→   503→  mfi: number;
   504→   504→  atr: number;
   505→   505→  bollingerUpper: number;
   506→   506→  bollingerMiddle: number;
   507→   507→  bollingerLower: number;
   508→   508→  trendAngle: number;
   509→   509→  trendR2: number;
   510→   510→  overallSignal: string;
   511→   511→  scenarios: Record<string, { probability: number }>;
   512→   512→  resistances: number[];
   513→   513→  supports: number[];
   514→   514→  ma21: number;
   515→   515→  ma100: number;
   516→   516→}): NarrativeInput {
   517→   517→  const bbRange = body.bollingerUpper - body.bollingerLower;
   518→   518→  const bbPosition = bbRange > 0
   519→   519→    ? Math.max(0, Math.min(100, ((body.currentPrice - body.bollingerLower) / bbRange) * 100))
   520→   520→    : 50;
   521→   521→
   522→   522→  const R1 = body.resistances[0] || body.currentPrice * 1.05;
   523→   523→  const S1 = body.supports[0] || body.currentPrice * 0.95;
   524→   524→
   525→   525→  // Find dominant scenario
   526→   526→  let dominantKey = 'R3';
   527→   527→  let dominantProb = 0;
   528→   528→  for (const k of ['R1', 'R2', 'R3', 'R4', 'R5'] as const) {
   529→   529→    const p = body.scenarios[k]?.probability ?? 0;
   530→   530→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
   531→   531→  }
   532→   532→
   533→   533→  return {
   534→   534→    price: body.currentPrice,
   535→   535→    trendDirection: body.trendDirection,
   536→   536→    rsi: body.rsi,
   537→   537→    adx: body.adx,
   538→   538→    stochK: body.stochK,
   539→   539→    stochD: body.stochD,
   540→   540→    macdLine: body.macdLine,
   541→   541→    macdSignal: body.macdSignal,
   542→   542→    macdHist: body.macdHist,
   543→   543→    diPlus: body.diPlus,
   544→   544→    diMinus: body.diMinus,
   545→   545→    obv: body.obv,
   546→   546→    hasVolume: body.hasVolume,
   547→   547→    cci: body.cci,
   548→   548→    mfi: body.mfi,
   549→   549→    atr: body.atr,
   550→   550→    bbPosition,
   551→   551→    bbUpper: body.bollingerUpper,
   552→   552→    bbLower: body.bollingerLower,
   553→   553→    dominantScenarioKey: dominantKey,
   554→   554→    scenarioProbabilities: {
   555→   555→      R1: body.scenarios.R1?.probability ?? 0,
   556→   556→      R2: body.scenarios.R2?.probability ?? 0,
   557→   557→      R3: body.scenarios.R3?.probability ?? 0,
   558→   558→      R4: body.scenarios.R4?.probability ?? 0,
   559→   559→      R5: body.scenarios.R5?.probability ?? 0,
   560→   560→    },
   561→   561→    nearResistance: R1 > 0 ? Math.abs(body.currentPrice - R1) / R1 < 0.03 : false,
   562→   562→    nearSupport: S1 > 0 ? Math.abs(body.currentPrice - S1) / S1 < 0.03 : false,
   563→   563→    priceVsMa21: body.currentPrice > body.ma21 ? 'above' : 'below',
   564→   564→    priceVsMa100: body.currentPrice > body.ma100 ? 'above' : 'below',
   565→   565→    trendR2: body.trendR2,
   566→   566→  };
   567→   567→}
