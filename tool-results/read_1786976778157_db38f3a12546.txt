   143→/* --─ Helpers -------------------------------------------- */
   144→
   145→function formatNum(num: number): string {
   146→  if (num == null) return '—';
   147→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 0 }).format(num);
   148→}
   149→
   150→function formatIdx(num: number): string {
   151→  if (num == null) return '—';
   152→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 2 }).format(num);
   153→}
   154→

   160→    </svg>
   161→  );
   162→}

   164→/* --─ Component ------------------------------------------ */
   165→
   166→export default function SymbolSearch({
   167→  onSelect,
   168→  placeholder = 'جستجوی ابزار مالی ...',
   169→  className,
   170→}: SymbolSearchProps) {
   171→  const [query, setQuery] = React.useState('');
   172→  const [open, setOpen] = React.useState(false);
   173→  const [results, setResults] = React.useState<InstrumentItem[]>([]);
   174→  const [loading, setLoading] = React.useState(false);
   175→  const [activeIndex, setActiveIndex] = React.useState(-1);
   176→  const [totalMatched, setTotalMatched] = React.useState(0);
   177→  const [activeCategory, setActiveCategory] = React.useState<CategoryKey>('all');
   178→  const [activeIndustry, setActiveIndustry] = React.useState<string | null>(null);
   179→  const [showIndustryPicker, setShowIndustryPicker] = React.useState(false);
   180→  const [indexSubFilter, setIndexSubFilter] = React.useState<'all' | 'main' | 'sector'>('all');
   181→
   182→  const cacheRef = React.useRef<InstrumentsData | null>(null);
   183→  const tgjuCacheRef = React.useRef<TgjuData | null>(null);
   184→  const fetchRef = React.useRef<Promise<InstrumentsData> | null>(null);
   185→  const tgjuFetchRef = React.useRef<Promise<TgjuData | null> | null>(null);
   186→  const inputRef = React.useRef<HTMLInputElement>(null);
   187→  const dropdownRef = React.useRef<HTMLDivElement>(null);
   188→  const itemRefs = React.useRef<(HTMLDivElement | null)[]>([]);
   189→  const industryScrollRef = React.useRef<HTMLDivElement>(null);
   190→
   191→  /* -- Data fetching -- */
   192→  const fetchData = React.useCallback(async (): Promise<InstrumentsData> => {
   193→    if (cacheRef.current) return cacheRef.current;
   194→    if (fetchRef.current) return fetchRef.current;
   195→    const p = async () => {
   196→      try {
   197→        const r = await fetch('/api/instruments');
   198→        if (!r.ok) throw new Error(`${r.status}`);
   199→        const d: InstrumentsData = await r.json();
   200→        cacheRef.current = d;
   201→        return d;
   202→      } finally {
   203→        fetchRef.current = null;
   204→      }
   205→    };
   206→    fetchRef.current = p();
   207→    return fetchRef.current;
   208→  }, []);
   209→
   210→  const fetchTgjuData = React.useCallback(async (): Promise<TgjuData | null> => {
   211→    if (tgjuCacheRef.current) return tgjuCacheRef.current;
   212→    if (tgjuFetchRef.current) return tgjuFetchRef.current;
   213→    const p = async () => {
   214→      try {
   215→        const r = await fetch('/api/tgju-instruments');
   216→        if (!r.ok) return null;
   217→        const d: TgjuData = await r.json();
   218→        tgjuCacheRef.current = d;
   219→        return d;
   220→      } catch {
   221→        return null;
   222→      } finally {
   223→        tgjuFetchRef.current = null;
   224→      }
   225→    };
   226→    tgjuFetchRef.current = p();
   227→    return p();
   228→  }, []);
   229→
   230→  /* -- Check if category is TGJU-based -- */
   231→  const isTgjuCategory = (cat: CategoryKey) => TGJU_CATEGORIES.has(cat);
   232→
   233→  /* -- Get all items for current category -- */
   234→  const getItemsForCategory = React.useCallback(async (cat: CategoryKey): Promise<InstrumentItem[]> => {
   235→    if (isTgjuCategory(cat)) {
   236→      const tgju = tgjuCacheRef.current || await fetchTgjuData();
   237→      if (!tgju) return [];
   238→      switch (cat) {
   239→        case 'currency': return tgju.currencies;
   240→        case 'gold': return [...tgju.gold, ...tgju.silver];
   241→        case 'gold_etf': return tgju.goldEtfs;
   242→        case 'silver': return tgju.silver;
   243→        case 'crypto': return tgju.crypto;
   244→        case 'world_index': return tgju.worldIndices;
   245→        case 'foreign_stock': return tgju.foreignStocks;
   246→        case 'forex': return tgju.forex;
   247→        case 'energy': return tgju.energy;
   248→        case 'metal': return tgju.metals;
   249→        case 'commodity': return tgju.commodities;
   250→        default: return [];
   251→      }
   252→    }
   253→
   254→    // TSE categories
   255→    const data = cacheRef.current || await fetchData();
   256→    switch (cat) {
   257→      case 'indices': {
   258→        let idxItems = data.indices;
   259→        if (indexSubFilter === 'main') {
   260→          idxItems = idxItems.filter((i) => i.cs === 'شاخص اصلی');
   261→        } else if (indexSubFilter === 'sector') {
   262→          idxItems = idxItems.filter((i) => i.cs === 'صنعت');
   263→        }
   264→        return idxItems;
   265→      }
   266→      case 'stocks':
   267→        return activeIndustry
   268→          ? data.stocks.filter((s) => s.cs === activeIndustry)
   269→          : data.stocks;
   270→      case 'etf': return data.etfs;
   271→      case 'bond': return [...data.bonds, ...data.mortgage];
   272→      case 'derivative': return [...data.futures, ...data.salaf];
   273→      case 'all':
   274→      default: {
   275→        const tgju = tgjuCacheRef.current;
   276→        return [
   277→          ...data.indices,
   278→          ...data.stocks,
   279→          ...data.etfs,
   280→          ...(tgju?.currencies || []),
   281→          ...(tgju?.forex || []),
   282→          ...(tgju?.crypto || []),
   283→          ...(tgju?.gold || []),
   284→          ...(tgju?.silver || []),
   285→          ...(tgju?.goldEtfs || []),
   286→          ...(tgju?.worldIndices || []),
   287→          ...(tgju?.foreignStocks || []),
   288→          ...(tgju?.energy || []),
   289→          ...(tgju?.metals || []),
   290→          ...(tgju?.commodities || []),
   291→          ...data.bonds,
   292→          ...data.futures,
   293→          ...data.salaf,
   294→          ...data.mortgage,
   295→        ];
   296→      }
   297→    }
   298→  }, [activeIndustry, indexSubFilter, fetchData, fetchTgjuData]);
   299→
   300→  /* -- Popular items -- */
   301→  const getPopular = React.useCallback(async () => {
   302→    const data = cacheRef.current || await fetchData();
   303→    const tgju = tgjuCacheRef.current;
   304→    const all = [
   305→      ...(data.indices.map((i) => ({ ...i, tval: i.index || 0 }))),
   306→      ...data.stocks,
   307→      ...data.etfs,
   308→      ...(tgju?.currencies || []),
   309→      ...(tgju?.crypto?.slice(0, 5) || []),
   310→      ...(tgju?.foreignStocks?.slice(0, 3) || []),
   311→      ...(tgju?.gold?.slice(0, 3) || []),
   312→    ];
   313→    const sorted = [...all].sort((a, b) => (b.tval || 0) - (a.tval || 0)).slice(0, 15);
   314→    setResults(sorted);
   315→    setTotalMatched(all.length);
   316→  }, [fetchData]);
   317→
   318→  /* -- Filter by search query -- */
   319→  const doFilter = React.useCallback((items: InstrumentItem[], q: string) => {
   320→    const n = q.trim().toLowerCase();
   321→    const f = items.filter(
   322→      (s) =>
   323→        s.l18?.toLowerCase().includes(n) ||
   324→        s.l30?.toLowerCase().includes(n) ||
   325→        s.cs?.toLowerCase().includes(n),
   326→    );
   327→    setTotalMatched(f.length);
   328→    setResults(f.slice(0, MAX_RESULTS));
   329→    setActiveIndex(-1);
   330→  }, []);
   331→
   332→  /* -- Show dropdown with data -- */
   333→  const showDropdown = React.useCallback(async (q: string) => {
   334→    setOpen(true);
   335→    await Promise.all([fetchData(), fetchTgjuData()]);
   336→    const items = await getItemsForCategory(activeCategory);
   337→    if (q.trim().length === 0) {
   338→      if (activeCategory === 'all') {
   339→        await getPopular();
   340→      } else {
   341→        setResults(items.slice(0, MAX_RESULTS));
   342→        setTotalMatched(items.length);
   343→      }
   344→    } else {
   345→      doFilter(items, q);
   346→    }
   347→    setLoading(false);
   348→  }, [fetchData, fetchTgjuData, getItemsForCategory, activeCategory, getPopular, doFilter]);
   349→
   350→  /* -- Category change -- */
   351→  const handleCategoryChange = React.useCallback(async (cat: CategoryKey) => {
   352→    setActiveCategory(cat);
   353→    if (cat !== 'stocks') setActiveIndustry(null);
   354→    if (cat !== 'indices') setIndexSubFilter('all');
   355→    setShowIndustryPicker(false);
   356→
   357→    const items = await getItemsForCategory(cat);
   358→    if (query.trim().length > 0) {
   359→      doFilter(items, query);
   360→    } else if (cat === 'all') {
   361→      await getPopular();
   362→    } else {
   363→      setResults(items.slice(0, MAX_RESULTS));
   364→      setTotalMatched(items.length);
   365→    }
   366→    setActiveIndex(-1);
   367→  }, [getItemsForCategory, query, doFilter, getPopular]);
   368→
   369→  /* -- Industry change -- */
   370→  const handleIndustryChange = React.useCallback((industry: string | null) => {
   371→    setActiveIndustry(industry);
   372→    setShowIndustryPicker(false);
   373→    if (cacheRef.current) {
   374→      const items = industry
   375→        ? cacheRef.current.stocks.filter((s) => s.cs === industry)
   376→        : cacheRef.current.stocks;
   377→      if (query.trim().length > 0) {
   378→        doFilter(items, query);
   379→      } else {
   380→        setResults(items.slice(0, MAX_RESULTS));
   381→        setTotalMatched(items.length);
   382→      }
   383→      setActiveIndex(-1);
   384→    }
   385→  }, [query, doFilter]);
   386→
   387→  const handleChange = React.useCallback((value: string) => {
   388→    setQuery(value);
   389→    getItemsForCategory(activeCategory).then((items) => {
   390→      if (value.trim().length === 0) {
   391→        if (activeCategory === 'all') getPopular();
   392→        else {
   393→          setResults(items.slice(0, MAX_RESULTS));
   394→          setTotalMatched(items.length);
   395→        }
   396→      } else {
   397→        doFilter(items, value);
   398→      }
   399→    });
   400→  }, [activeCategory, getItemsForCategory, getPopular, doFilter]);
   401→
   402→  const handleFocus = React.useCallback(() => {
   403→    setLoading(true);
   404→    showDropdown(query);
   405→  }, [query, showDropdown]);
   406→
   407→  const selectSymbol = React.useCallback((s: InstrumentItem) => {
   408→    setQuery(s.l18);
   409→    setOpen(false);
   410→    onSelect?.(s.l18, s.category, s.insCode, s.tgjuKey);
   411→    inputRef.current?.blur();
   412→  }, [onSelect]);
   413→
   414→  const handleKeyDown = React.useCallback((e: React.KeyboardEvent) => {
   415→    if (!open || results.length === 0) {
   416→      if (e.key === 'ArrowDown' || e.key === 'Enter') { e.preventDefault(); handleFocus(); }
   417→      return;
   418→    }
   419→    if (e.key === 'ArrowDown') { e.preventDefault(); setActiveIndex((p) => p < results.length - 1 ? p + 1 : 0); }
   420→    else if (e.key === 'ArrowUp') { e.preventDefault(); setActiveIndex((p) => p > 0 ? p - 1 : results.length - 1); }
   421→    else if (e.key === 'Enter') { e.preventDefault(); if (activeIndex >= 0) selectSymbol(results[activeIndex]); }
   422→    else if (e.key === 'Escape') { e.preventDefault(); setOpen(false); inputRef.current?.blur(); }
   423→  }, [open, results, activeIndex, selectSymbol, handleFocus]);
   424→
   425→  React.useEffect(() => {
   426→    if (activeIndex >= 0) itemRefs.current[activeIndex]?.scrollIntoView({ block: 'nearest' });
   427→  }, [activeIndex]);
   428→
   429→  React.useEffect(() => {
   430→    if (!open) return;
   431→    const handler = (e: MouseEvent) => {
   432→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node) &&
   433→          inputRef.current && !inputRef.current.contains(e.target as Node)) {
   434→        setOpen(false);
   435→        setShowIndustryPicker(false);
   436→      }
   437→    };
   438→    document.addEventListener('mousedown', handler);
   439→    return () => document.removeEventListener('mousedown', handler);
   440→  }, [open]);
   441→
   442→  const clear = () => { setQuery(''); setActiveIndustry(null); setActiveCategory('all'); setResults([]); inputRef.current?.focus(); };
   443→  const isPopular = query.trim().length === 0 && activeCategory === 'all';
   444→  const isSearch = query.trim().length > 0;
   445→
   446→  /* -- Render instrument price -- */
   447→  const renderPrice = (item: InstrumentItem) => {
   448→    const isTgju = isTgjuCategory(item.category as CategoryKey);
   449→    if (item.category === 'index' || item.category === 'tse_index') {
   450→      return (
   451→        <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   452→          <span className='text-xs font-semibold text-gray-200'>{formatIdx(item.pl)}</span>
   453→          <span className={cn(
   454→            'text-[11px] font-bold px-1.5 py-0.5 rounded',
   455→            item.pcp > 0 ? 'bg-emerald-500/15 text-emerald-400' :
   456→            item.pcp < 0 ? 'bg-red-500/15 text-red-400' : 'text-gray-500',
   457→          )}>
   458→            {item.pcp > 0 ? '+' : ''}{item.pcp?.toFixed(2)}%
   459→          </span>
   460→        </div>
   461→      );
   462→    }
   463→    // For static TGJU items with no live price, show dash
   464→    if (isTgju && item.pl === 0) {
   465→      return (
   466→        <div className='flex shrink-0 flex-col items-end gap-0.5'>
   467→          <span className='text-xs text-gray-500'>—</span>
   468→        </div>
   469→      );
   470→    }
   471→    const isUp = item.pcp > 0;
   472→    const isDown = item.pcp < 0;
   473→    return (
   474→      <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   475→        <span className='text-xs font-semibold text-gray-200'>{formatNum(item.pl)}</span>
   476→        <span className={cn(
   477→          'text-[11px] font-bold px-1.5 py-0.5 rounded',
   478→          isUp ? 'bg-emerald-500/15 text-emerald-400' :
   479→          isDown ? 'bg-red-500/15 text-red-400' : 'text-gray-500',
   480→        )}>
   481→          {isUp ? '+' : ''}{item.pcp?.toFixed(2)}%
   482→        </span>
   483→      </div>
   484→    );
   485→  };
   486→
   487→  const industries = cacheRef.current?.industries || [];
   488→
   489→  return (
   490→    <div dir='rtl' className={cn('relative', className)}>
   491→
   492→      <div className='relative'>
   493→        <Search className='absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-500 pointer-events-none' />
   494→        <Input
   495→          ref={inputRef}
   496→          type='text'
   497→          value={query}
   498→          onChange={(e) => handleChange(e.target.value)}
   499→          onFocus={handleFocus}
   500→          onKeyDown={handleKeyDown}
   501→          placeholder={placeholder}
   502→          suppressHydrationWarning
   503→          className={cn(
   504→            'h-10 w-full rounded-lg border border-gray-700/80 bg-gray-900/90',
   505→            'text-sm text-gray-100 placeholder:text-gray-500',
