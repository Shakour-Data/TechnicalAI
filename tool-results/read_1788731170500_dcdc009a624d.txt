    44→    44→function findSwingPoints(data: OHLCV[], order: number = 3): SwingPoint[] {
    45→    45→  const swings: SwingPoint[] = [];
    46→    46→  const len = data.length;
    47→    47→  for (let i = order; i < len - order; i++) {
    48→    48→    let isHigh = true;
    49→    49→    let isLow = true;
    50→    50→    for (let j = 1; j <= order; j++) {
    51→    51→      if (data[i].high <= data[i - j].high || data[i].high <= data[i + j].high) isHigh = false;
    52→    52→      if (data[i].low >= data[i - j].low || data[i].low >= data[i + j].low) isLow = false;
    53→    53→    }
    54→    54→    if (isHigh) swings.push({ index: i, price: data[i].high, type: 'high' });
    55→    55→    if (isLow) swings.push({ index: i, price: data[i].low, type: 'low' });
    56→    56→  }
    57→    57→  return swings;
    58→    58→}

   104→   104→function ema(values: number[], period: number): number {
   105→   105→  if (values.length < period) return 0;
   106→   106→  const k = 2 / (period + 1);
   107→   107→  let result = values.slice(0, period).reduce((a, b) => a + b, 0) / period;
   108→   108→  for (let i = period; i < values.length; i++) {
   109→   109→    result = values[i] * k + result * (1 - k);
   110→   110→  }
   111→   111→  return result;
   112→   112→}

   129→   129→function detectHeadAndShoulders(data: OHLCV[]): PatternResult[] {
   130→   130→  const results: PatternResult[] = [];
   131→   131→  if (data.length < 30) return results;
   132→   132→  const swings = findSwingPoints(data, 3);
   133→   133→  const highs = swings.filter(s => s.type === 'high');
   134→   134→  const lows = swings.filter(s => s.type === 'low');
   135→   135→
   136→   136→  // Bearish Head & Shoulders: left shoulder < head > right shoulder, neckline support
   137→   137→  for (let i = 2; i < highs.length; i++) {
   138→   138→    const ls = highs[i - 2];
   139→   139→    const hd = highs[i - 1];
   140→   140→    const rs = highs[i];
   141→   141→    if (hd.price > ls.price && hd.price > rs.price) {
   142→   142→      const sim = 1 - Math.abs(ls.price - rs.price) / hd.price;
   143→   143→      if (sim > 0.85) {
   144→   144→        const neckline = (ls.price + rs.price) / 2;
   145→   145→        const last = data[data.length - 1];
   146→   146→        const dist = (last.close - neckline) / neckline;
   147→   147→        const status = dist > 0.01 ? 'forming' : dist < -0.005 ? 'completed' : 'forming';
   148→   148→        results.push({
   149→   149→          name: 'سر و شانه', nameEn: 'Head & Shoulders', category: 'classic',
   150→   150→          direction: 'bearish', strength: sim * 0.8 + (status === 'completed' ? 0.2 : 0),
   151→   151→          status, priceLevel: neckline,
   152→   152→          description: `الگوی سر و شانه نزولی با خط گردن ${neckline.toFixed(0)}`
   153→   153→        });
   154→   154→      }
   155→   155→    }
   156→   156→  }
   157→   157→
   158→   158→  // Bullish Inverse Head & Shoulders
   159→   159→  for (let i = 2; i < lows.length; i++) {
   160→   160→    const ls = lows[i - 2];
   161→   161→    const hd = lows[i - 1];
   162→   162→    const rs = lows[i];
   163→   163→    if (hd.price < ls.price && hd.price < rs.price) {
   164→   164→      const sim = 1 - Math.abs(ls.price - rs.price) / hd.price;
   165→   165→      if (sim > 0.85) {
   166→   166→        const neckline = (ls.price + rs.price) / 2;
   167→   167→        const last = data[data.length - 1];
   168→   168→        const dist = (last.close - neckline) / neckline;
   169→   169→        const status = dist < -0.01 ? 'forming' : dist > 0.005 ? 'completed' : 'forming';
   170→   170→        results.push({
   171→   171→          name: 'سر و شانه معکوس', nameEn: 'Inverse Head & Shoulders', category: 'classic',
   172→   172→          direction: 'bullish', strength: sim * 0.8 + (status === 'completed' ? 0.2 : 0),
   173→   173→          status, priceLevel: neckline,
   174→   174→          description: `الگوی سر و شانه معکوس صعودی با خط گردن ${neckline.toFixed(0)}`
   175→   175→        });
   176→   176→      }
   177→   177→    }
   178→   178→  }
   179→   179→
   180→   180→  return results;
   181→   181→}

   339→   339→function detectSymmetricTriangle(data: OHLCV[]): PatternResult {
   340→   340→  if (data.length < 20) return null!;
   341→   341→  const swings = findSwingPoints(data, 3);
   342→   342→  const highs = swings.filter(s => s.type === 'high');
   343→   343→  const lows = swings.filter(s => s.type === 'low');
   344→   344→  if (highs.length < 3 || lows.length < 3) return null!;
   345→   345→
   346→   346→  const rh = highs.slice(-3);
   347→   347→  const rl = lows.slice(-3);
   348→   348→  const hConverging = rh[rh.length - 1].price < rh[0].price;
   349→   349→  const lConverging = rl[rl.length - 1].price > rl[0].price;
   350→   350→  const hSlope = (rh[rh.length - 1].price - rh[0].price) / (rh[rh.length - 1].index - rh[0].index);
   351→   351→  const lSlope = (rl[rl.length - 1].price - rl[0].price) / (rl[rl.length - 1].index - rl[0].index);
   352→   352→  const slopeSim = 1 - Math.abs(Math.abs(hSlope) - Math.abs(lSlope)) / (Math.abs(hSlope) + Math.abs(lSlope) + 0.001);
   353→   353→
   354→   354→  if (!hConverging || !lConverging || slopeSim < 0.5) return null!;
   355→   355→
   356→   356→  const apex = (rh[rh.length - 1].price + rl[rl.length - 1].price) / 2;
   357→   357→  const last = data[data.length - 1];
   358→   358→  const dir = last.close > apex ? 'bullish' : 'bearish';
   359→   359→
   360→   360→  return {
   361→   361→    name: 'مثلث متقارن', nameEn: 'Symmetric Triangle', category: 'classic',
   362→   362→    direction: dir as 'bullish' | 'bearish',
   363→   363→    strength: slopeSim * 0.7 + 0.3,
   364→   364→    status: 'forming',
   365→   365→    priceLevel: apex,
   366→   366→    description: `الگوی مثلث متقارن با رأس تقریبی ${apex.toFixed(0)}`
   367→   367→  };
   368→   368→}

   427→   427→function detectBullFlag(data: OHLCV[]): PatternResult {

   429→   429→  const poleLen = Math.min(10, Math.floor(data.length * 0.3));
   430→   430→  const flagLen = Math.min(10, Math.floor(data.length * 0.2));
   431→   431→  const flagStart = data.length - flagLen;
   432→   432→  const poleEnd = flagStart;
   433→   433→  const poleStart = Math.max(0, poleEnd - poleLen);
   434→   434→
   435→   435→  if (poleStart >= poleEnd) return null!;
   436→   436→  const poleRise = data[poleEnd - 1].close - data[poleStart].close;
   437→   437→  if (poleRise <= 0) return null!;
   438→   438→
   439→   439→  const flagHighs = data.slice(flagStart).map(d => d.high);
   440→   440→  const flagLows = data.slice(flagStart).map(d => d.low);
   441→   441→  const flagTrend = sma(flagHighs, flagLen) - sma(flagLows, flagLen);
   442→   442→  const flagSlope = (flagHighs[flagHighs.length - 1] - flagHighs[0]) / (flagLen || 1);
   443→   443→
   444→   444→  if (flagSlope > 0 || flagTrend > poleRise * 0.3) return null!;
   445→   445→
   446→   446→  const last = data[data.length - 1];
   447→   447→  return {
   448→   448→    name: 'پرچم صعودی', nameEn: 'Bull Flag', category: 'classic',
   449→   449→    direction: 'bullish', strength: 0.5 + Math.min(poleRise / (last.close || 1), 0.3) + (flagSlope < 0 ? 0.2 : 0),
   450→   450→    status: 'forming',
   451→   451→    priceLevel: Math.max(...flagHighs),

   522→   522→function detectCupAndHandle(data: OHLCV[]): PatternResult {
   523→   523→  if (data.length < 40) return null!;
   524→   524→  const swings = findSwingPoints(data, 3);
   525→   525→  const lows = swings.filter(s => s.type === 'low');
   526→   526→  if (lows.length < 3) return null!;
   527→   527→
   528→   528→  // Look for U-shape: high, low, high with a small dip (handle)
   529→   529→  const recentLows = lows.slice(-5);
   530→   530→  if (recentLows.length < 3) return null!;
   531→   531→
   532→   532→  // Cup: first and last swing lows should be near same level, middle ones lower
   533→   533→  const cupL = recentLows.slice(0, Math.ceil(recentLows.length * 0.7));
   534→   534→  const cupBottom = Math.min(...cupL.map(l => l.price));
   535→   535→  const cupRim = avg([recentLows[0].price, cupL[cupL.length - 1].price]);
   536→   536→  const depth = (cupRim - cupBottom) / cupRim;
   537→   537→  if (depth < 0.05 || depth > 0.5) return null!;
   538→   538→
   539→   539→  const handleLows = recentLows.slice(Math.ceil(recentLows.length * 0.7));
   540→   540→  if (handleLows.length === 0) return null!;
   541→   541→  const handleDepth = (cupRim - avg(handleLows.map(l => l.price))) / cupRim;
   542→   542→  if (handleDepth > 0.3 * depth) return null!;
   543→   543→
   544→   544→  const last = data[data.length - 1];
   545→   545→  const broken = last.close > cupRim;
   546→   546→  return {
   547→   547→    name: 'فنجان و دسته', nameEn: 'Cup & Handle', category: 'classic',
   548→   548→    direction: 'bullish', strength: 0.5 + (broken ? 0.3 : 0) + Math.min(depth, 0.2),
   549→   549→    status: broken ? 'completed' : 'forming',
   550→   550→    priceLevel: cupRim,
   551→   551→    description: `الگوی فنجان و دسته صعودی با سطح مقاومت ${cupRim.toFixed(0)}`
   552→   552→  };
   553→   553→}

   555→   555→function detectRectangle(data: OHLCV[]): PatternResult {
   556→   556→  if (data.length < 20) return null!;
   557→   557→  const swings = findSwingPoints(data, 3);
   558→   558→  const highs = swings.filter(s => s.type === 'high');

   560→   560→  if (highs.length < 2 || lows.length < 2) return null!;
   561→   561→
   562→   562→  const rh = highs.slice(-3);
   563→   563→  const rl = lows.slice(-3);
   564→   564→  const hAvg = avg(rh.map(h => h.price));
   565→   565→  const lAvg = avg(rl.map(l => l.price));
   566→   566→  const hVar = rh.reduce((s, h) => s + Math.abs(h.price - hAvg), 0) / hAvg;
   567→   567→  const lVar = rl.reduce((s, l) => s + Math.abs(l.price - lAvg), 0) / lAvg;
   568→   568→
   569→   569→  if (hVar > 0.03 || lVar > 0.03) return null!;
   570→   570→  if ((hAvg - lAvg) / lAvg < 0.02) return null!;
   571→   571→
   572→   572→  const last = data[data.length - 1];
   573→   573→  const mid = (hAvg + lAvg) / 2;
   574→   574→  const dir = last.close > mid ? 'bullish' : 'bearish';
   575→   575→
   576→   576→  return {
   577→   577→    name: 'مستطیل / کانال', nameEn: 'Rectangle / Channel', category: 'classic',
   578→   578→    direction: dir as 'bullish' | 'bearish',
   579→   579→    strength: (1 - hVar * 20) * (1 - lVar * 20),
   580→   580→    status: 'forming',
   581→   581→    priceLevel: mid,
   582→   582→    description: `الگوی مستطیل با محدوده ${lAvg.toFixed(0)} تا ${hAvg.toFixed(0)}`
   583→   583→  };

   738→   738→function detectHarmonicFromPoints(
   739→   739→  X: number, A: number, B: number, C: number, D: number,
   740→   740→  spec: HarmonicSpec,
   741→   741→  direction: 'bullish' | 'bearish'
   742→   742→): PatternResult | null {
   743→   743→  const xLeg = Math.abs(A - X);
   744→   744→  const aLeg = Math.abs(B - A);
   745→   745→  const bLeg = Math.abs(C - B);
   746→   746→  const cLeg = Math.abs(D - C);
   747→   747→
   748→   748→  if (xLeg === 0 || aLeg === 0 || bLeg === 0) return null;
   749→   749→
   750→   750→  const abR = aLeg / xLeg;
   751→   751→  const bcR = bLeg / aLeg;

  1576→  1576→function detectImpulseWave(data: OHLCV[]): PatternResult | null {
  1577→  1577→  if (data.length < 30) return null;
  1578→  1578→  const swings = findSwingPoints(data, 3);
  1579→  1579→  if (swings.length < 7) return null;
  1580→  1580→
  1581→  1581→  // Look for 5-wave structure in recent swings
  1582→  1582→  const recent = swings.slice(-7);
  1583→  1583→  // Impulse: wave 1 > wave 2 < wave 3 > wave 4 < wave 5
  1584→  1584→  // with wave 3 not shortest, wave 2 doesn't retrace past wave 1 start
  1585→  1585→  const prices = recent.map(s => s.price);
  1586→  1586→
  1587→  1587→  // Try to find 5-wave pattern: 1(up) 2(down) 3(up) 4(down) 5(up)
  1588→  1588→  for (let i = 0; i <= prices.length - 5; i++) {
  1589→  1589→    const w = prices.slice(i, i + 5);
  1590→  1590→    const isUp = (a: number, b: number) => b > a;
  1591→  1591→    const isDown = (a: number, b: number) => b < a;
  1592→  1592→
  1593→  1593→    if (isUp(w[0], w[1]) && isDown(w[1], w[2]) && isUp(w[2], w[3]) && isDown(w[3], w[4])) {
  1594→  1594→      // Wave 3 should be the longest (or at least not shortest) impulse
  1595→  1595→      const impulse1 = Math.abs(w[1] - w[0]);
  1596→  1596→      const impulse3 = Math.abs(w[3] - w[2]);
  1597→  1597→      const impulse5 = Math.abs(w[4] - w[3]);
  1598→  1598→      if (impulse3 >= impulse1 && impulse3 >= impulse5 * 0.8) {
  1599→  1599→        // Wave 2 should not retrace beyond wave 1 start
  1600→  1600→        const retrace2 = Math.abs(w[2] - w[1]) / impulse1;
  1601→  1601→        if (retrace2 < 1 && retrace2 > 0.2) {
  1602→  1602→          return {
  1603→  1603→            name: 'موج تکانه‌ای', nameEn: 'Impulse Wave', category: 'elliott',
  1604→  1604→            direction: 'bullish', strength: 0.7,
  1605→  1605→            status: 'forming',
  1606→  1606→            priceLevel: w[3],
  1607→  1607→            description: 'الگوی موج تکانه‌ای ۵ موجی الیوت — روند صعودی'
  1608→  1608→          };
  1609→  1609→        }
  1610→  1610→      }
  1611→  1611→    }
  1612→  1612→  }
  1613→  1613→  return null;
  1614→  1614→}

  1868→  1868→    p.nameEn.includes('Doji') || p.nameEn.includes('Star') ||

  1883→  1883→  const elliottScore = elliott.length > 0

  1888→  1888→  const hybridScore = allPatterns.length > 0

  1892→  1892→  return {
