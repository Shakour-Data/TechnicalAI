   135→   135→export default function VdssGraph(props: VdssGraphProps) {
   136→   136→  const { colors: C } = useTheme();
   137→   137→  const {
   138→   138→    symbolName,
   139→   139→    currentPrice,
   140→   140→    scenarios,
   141→   141→    decisionGraph,
   142→   142→  } = props;
   143→   143→
   144→   144→  const [activeFilter, setActiveFilter] = useState('all');
   145→   145→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   146→   146→  const graphRef = useRef<HTMLDivElement>(null);
   147→   147→  const svgRef = useRef<SVGSVGElement>(null);
   148→   148→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   149→   149→  const shellRef = useRef<HTMLDivElement>(null);
   150→   150→
   151→   151→  // ── Derived data from decisionGraph ──────────────────────────────
   152→   152→  const edges = decisionGraph?.edges ?? [];
   153→   153→  const nodes = decisionGraph?.nodes ?? [];
   154→   154→  const nodePositions = decisionGraph?.nodePositions ?? {};
   155→   155→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   156→   156→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   157→   157→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   158→   158→  const pathContributions = decisionGraph?.pathContributions ?? {};
   159→   159→
   160→   160→  // Build node lookup map
   161→   161→  const nodeMap = useMemo(() => {
   162→   162→    const map: Record<string, (typeof nodes)[number]> = {};
   163→   163→    for (const n of nodes) map[n.id] = n;
   164→   164→    return map;
   165→   165→  }, [nodes]);
   166→   166→
   167→   167→  // ── Visible edges/nodes based on filter ──────────────────────────
   168→   168→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   169→   169→    if (activeFilter === 'all') {
   170→   170→      const allEdgeIdx = edges.map((_, i) => i);
   171→   171→      const allNodes = new Set(nodes.map(n => n.id));
   172→   172→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   173→   173→    }
   174→   174→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   175→   175→      const edgeSet = new Set<number>();
   176→   176→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   177→   177→      edges.forEach((e, i) => {
   178→   178→        if (e.to === activeFilter) {
   179→   179→          edgeSet.add(i);
   180→   180→          nodeSet.add(e.from);
   181→   181→        }
   182→   182→      });
   183→   183→      edges.forEach((e, i) => {
   184→   184→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   185→   185→          edgeSet.add(i);
   186→   186→          nodeSet.add(e.from);
   187→   187→        }
   188→   188→      });
   189→   189→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   190→   190→    }
   191→   191→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   192→   192→      const branchType = `branch-${activeFilter}` as const;
   193→   193→      const edgeSet = new Set<number>();
   194→   194→      const nodeSet = new Set<string>(['ROOT']);
   195→   195→      edges.forEach((e, i) => {
   196→   196→        if (e.type === branchType) {
   197→   197→          edgeSet.add(i);
   198→   198→          nodeSet.add(e.from);
   199→   199→          nodeSet.add(e.to);
   200→   200→        }
   201→   201→      });
   202→   202→      const expanded = new Set<number>();
   203→   203→      const expandedNodes = new Set<string>(nodeSet);
   204→   204→      edges.forEach((e, i) => {
   205→   205→        if (expandedNodes.has(e.from)) {
   206→   206→          expanded.add(i);
   207→   207→          expandedNodes.add(e.to);
   208→   208→        }
   209→   209→      });
   210→   210→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   211→   211→    }
   212→   212→    // Type filter (up/pullback/down/risk)
   213→   213→    const edgeSet = new Set<number>();
   214→   214→    const nodeSet = new Set<string>(['ROOT']);
   215→   215→    edges.forEach((e, i) => {
   216→   216→      if (e.type === activeFilter) {
   217→   217→        edgeSet.add(i);
   218→   218→        nodeSet.add(e.from);
   219→   219→        nodeSet.add(e.to);
   220→   220→      }
   221→   221→    });
   222→   222→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   223→   223→  }, [activeFilter, edges, nodes]);
   224→   224→
   225→   225→  // ── Draw SVG edges ───────────────────────────────────────────────
   226→   226→  const drawEdges = useCallback(() => {
   227→   227→    const svg = svgRef.current;
   228→   228→    const graph = graphRef.current;
   229→   229→    if (!svg || !graph || edges.length === 0) return;
   230→   230→
   231→   231→    let markersSvg = '';
   232→   232→    const seenColors = new Set<string>();
   233→   233→    for (const e of edges) {
   234→   234→      const c = EDGE_COLORS[e.type];
   235→   235→      if (c && !seenColors.has(e.type)) {
   236→   236→        seenColors.add(e.type);
   237→   237→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   238→   238→      }
   239→   239→    }
   240→   240→
   241→   241→    let pathsSvg = '';
   242→   242→    const graphRect = graph.getBoundingClientRect();
   243→   243→
   244→   244→    for (let ei = 0; ei < edges.length; ei++) {
   245→   245→      const { from: fromId, to: toId, label, type } = edges[ei];
   246→   246→      const isVisible = visibleEdgeIndices.includes(ei);
   247→   247→      const fromEl = nodeRefs.current[fromId];
   248→   248→      const toEl = nodeRefs.current[toId];
   249→   249→      if (!fromEl || !toEl) continue;
   250→   250→
   251→   251→      const fRect = fromEl.getBoundingClientRect();
   252→   252→      const tRect = toEl.getBoundingClientRect();
   253→   253→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   254→   254→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   255→   255→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   256→   256→      const by = tRect.top - graphRect.top + tRect.height / 2;
   257→   257→      const dx = bx - ax;
   258→   258→      const dy = by - ay;
   259→   259→      const dist = Math.sqrt(dx * dx + dy * dy);
   260→   260→
   261→   261→      if (dist === 0) continue;
   262→   262→
   263→   263→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   264→   264→      const mx = (ax + bx) / 2;
   265→   265→      const my = (ay + by) / 2;
   266→   266→      const nx = -dy / dist;
   267→   267→      const ny = dx / dist;
   268→   268→      const cx = mx + nx * bend;
   269→   269→      const cy = my + ny * bend;
   270→   270→
   271→   271→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   272→   272→      const prob = edgeProbabilities[ei] ?? 0;
   273→   273→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   274→   274→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   275→   275→
   276→   276→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.72 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   277→   277→
   278→   278→      if (isVisible && ei >= 6) {
   279→   279→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#ffffff" font-size="10" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="4" stroke-linejoin="round" opacity="0.95" data-type="${type}" class="edge-label">${label}</text>`;
   280→   280→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   281→   281→      }
   282→   282→    }
   283→   283→
   284→   284→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   285→   285→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   286→   286→
   287→   287→  useEffect(() => { drawEdges(); }, [drawEdges]);
   288→   288→
   289→   289→  useEffect(() => {
   290→   290→    const el = shellRef.current;
   291→   291→    if (!el) return;
   292→   292→    let observer: ResizeObserver | null = null;
   293→   293→    try {
   294→   294→      observer = new ResizeObserver(() => { try { drawEdges(); } catch {} });
   295→   295→      observer.observe(el);
   296→   296→    } catch {}
   297→   297→    return () => { try { observer?.disconnect(); } catch {} };
   298→   298→  }, [drawEdges]);
   299→   299→
   300→   300→
   301→   301→  // ── All filter buttons ────────────────────────────────────────────
   302→   302→  const allFilters = [
   303→   303→    ...TYPE_FILTERS,
   304→   304→    { key: 'sep1', label: '│', isSep: true as const },
   305→   305→    ...Object.entries(BRANCH_META).map(([k, v]) => ({ key: k, label: v.label, branchKey: k, branchColor: v.color })),
   306→   306→    { key: 'sep2', label: '│', isSep: true as const },
   307→   307→    ...SCENARIO_KEYS.map(k => ({ key: k, label: SCENARIO_META_LOCAL[k].label, scenarioKey: k })),
   308→   308→  ];
   309→   309→
   310→   310→  // ── Corrected probabilityTrend: day 1 overridden with final scenario probabilities ──
   311→   311→  const correctedTrend = useMemo(() => {
   312→   312→    const raw = props.probabilityTrend;
   313→   313→    if (!raw || !raw.scenarios || raw.scenarios.length === 0) return raw;
   314→   314→
   315→   315→    // Build final individual probabilities (matching boxes) for day 1
   316→   316→    const day1Individuals: Record<string, number> = {};
   317→   317→    for (const key of SCENARIO_KEYS) {
   318→   318→      const s = scenarios[key as keyof typeof scenarios];
   319→   319→      day1Individuals[key] = (s?.probability ?? 0) / 100;
   320→   320→    }
   321→   321→
   322→   322→    // Corrected scenarios array
   323→   323→    const correctedScenarios: ScenarioTrend[] = raw.scenarios.map(sc => {
   324→   324→      const newInd = day1Individuals[sc.scenarioKey] ?? sc.trend[0]?.individualProb ?? 0;
   325→   325→      const newCum = calculateCDF(sc.scenarioKey, day1Individuals);
   326→   326→      const newTrend: DayPoint[] = sc.trend.map((dp, i) => {
   327→   327→        if (i === 0) {
   328→   328→          return { ...dp, individualProb: newInd, cumulativeProb: newCum };
   329→   329→        }
   330→   330→        return dp;
   331→   331→      });
   332→   332→      return {
   333→   333→        ...sc,
   334→   334→        currentProbability: newInd,
   335→   335→        trend: newTrend,
   336→   336→        peakProbability: Math.max(newInd, sc.peakProbability),
   337→   337→      };
   338→   338→    });
   339→   339→
   340→   340→    // Corrected groups
   341→   341→    const correctedGroups = raw.groups.map(g => {
   342→   342→      // Group cumulative for day 1
   343→   343→      const day1AllInds: Record<string, number> = {};
   344→   344→      for (const sc of raw.scenarios) {
   345→   345→        day1AllInds[sc.scenarioKey] = day1Individuals[sc.scenarioKey] ?? sc.trend[0]?.individualProb ?? 0;
   346→   346→      }
   347→   347→      let groupCumDay1 = 0;
   348→   348→      if (g.group === 'bullish') {
   349→   349→        groupCumDay1 = calculateCDF('SC6', day1AllInds);
   350→   350→      } else if (g.group === 'bearish') {
   351→   351→        groupCumDay1 = calculateCDF('SC4', day1AllInds);
   352→   352→      } else {
   353→   353→        groupCumDay1 = day1AllInds['SC5'] ?? 0;
   354→   354→      }
   355→   355→
   356→   356→      const newTrend: DayPoint[] = g.trend.map((dp, i) => {
   357→   357→        if (i === 0) {
   358→   358→          return { ...dp, individualProb: groupCumDay1, cumulativeProb: groupCumDay1 };
   359→   359→        }
   360→   360→        return dp;
   361→   361→      });
   362→   362→      return {
   363→   363→        ...g,
   364→   364→        trend: newTrend,
   365→   365→        peakProbability: Math.max(groupCumDay1, g.peakProbability),
   366→   366→      };
   367→   367→    });
   368→   368→
   369→   369→    return { ...raw, scenarios: correctedScenarios, groups: correctedGroups };
   370→   370→  }, [props.probabilityTrend, scenarios]);
   371→   371→
   372→   372→  // ── Meaningful node display values (Persian-formatted) ──────────
   373→   373→  const nodeDisplayValues = useMemo(() => {
   374→   374→    const display: Record<string, string> = {};
   375→   375→    if (!decisionGraph) return display;
   376→   376→
   377→   377→    // Sub-branch parent mapping
   378→   378→    const subParents: Record<string, string> = {
   379→   379→      N_T_BULL: 'N_TREND', N_T_BEAR: 'N_TREND', N_T_FLAT: 'N_TREND',
   380→   380→      N_B_UP: 'N_BREAK', N_B_DOWN: 'N_BREAK', N_B_NONE: 'N_BREAK',
   381→   381→      N_R_BULL: 'N_REVERSAL', N_R_BEAR: 'N_REVERSAL', N_R_NONE: 'N_REVERSAL',
   382→   382→    };
   383→   383→
   384→   384→    // Find edge probability from parent to child
   385→   385→    const getEdgeProb = (from: string, to: string): number => {
   386→   386→      for (let i = 0; i < edges.length; i++) {
   387→   387→        if (edges[i].from === from && edges[i].to === to) {
   388→   388→          return edgeProbabilities[i] ?? 0;
   389→   389→        }
   390→   390→      }
   391→   391→      return 0;
   392→   392→    };
   393→   393→
   394→   394→    // ROOT: no numeric value needed
   395→   395→    display['ROOT'] = '';
   396→   396→
   397→   397→    // Main branches: absolute probability from root
   398→   398→    display['N_TREND'] = toPersianDigits((branchProbs.trend * 100).toFixed(0)) + '٪';
   399→   399→    display['N_BREAK'] = toPersianDigits((branchProbs.breakout * 100).toFixed(0)) + '٪';
   400→   400→    display['N_REVERSAL'] = toPersianDigits((branchProbs.reversal * 100).toFixed(0)) + '٪';
   401→   401→
   402→   402→    // Sub-branches: conditional probability within parent (meaningful context)
   403→   403→    for (const [sub, parent] of Object.entries(subParents)) {
   404→   404→      const condProb = getEdgeProb(parent, sub);
   405→   405→      display[sub] = toPersianDigits((condProb * 100).toFixed(0)) + '٪';
   406→   406→    }
   407→   407→

   870→   870→function MultiSelectTrendChart({ data, selectedKeys, onToggleKey, mode, title }: {
   871→   871→  data: ProbabilityTrendResult;
   872→   872→  selectedKeys: Set<string>;
   873→   873→  onToggleKey: (k: string) => void;
   874→   874→  mode: 'individual' | 'cumulative';
   875→   875→  title: string;
   876→   876→}) {
   877→   877→  const svgRef = useRef<SVGSVGElement>(null);
   878→   878→  const tooltipRef = useRef<HTMLDivElement>(null);
   879→   879→  const [hoveredDay, setHoveredDay] = useState<number | null>(null);
   880→   880→
   881→   881→  const scenarios = data.scenarios.filter(s => selectedKeys.has(s.scenarioKey));
   882→   882→  const maxDays = data.scenarios[0]?.trend?.length ?? 0;
   883→   883→
   884→   884→  // Compute Y range — dynamic based on selected scenarios
   885→   885→  let yMin = 0;
   886→   886→  let yMax = 0;
   887→   887→  for (const sc of scenarios) {
   888→   888→    for (const d of sc.trend) {
   889→   889→      const v = mode === 'individual' ? d.individualProb : d.cumulativeProb;
   890→   890→      if (v > yMax) yMax = v;
   891→   891→    }
   892→   892→  }
   893→   893→  // For cumulative mode, ensure minimum range
   894→   894→  if (yMax <= 0) yMax = 0.1;
   895→   895→

   901→   901→
   902→   902→  // Find a nice step: aim for ~5 grid lines
   903→   903→  const niceSteps = [0.01, 0.02, 0.025, 0.05, 0.1, 0.15, 0.2, 0.25, 0.5, 1.0];
   904→   904→  const rawStep = (paddedMax - paddedMin) / 5;
   905→   905→  let step = niceSteps[0];
   906→   906→  for (const ns of niceSteps) {
   907→   907→    if (ns >= rawStep) { step = ns; break; }
   908→   908→  }
   909→   909→  // Round yMin down and yMax up to step boundaries
   910→   910→  const gridYMin = Math.floor(paddedMin / step) * step;

   925→   925→
   926→   926→  // Dynamic grid lines based on computed range

   931→   931→    const pctVal = v * 100;

   933→   933→    gridEls.push(

   941→   941→    gridEls.push(

   943→   943→        {toPersianDigits(String(d))}
   944→   944→      </text>,
   945→   945→    );
   946→   946→  }

   948→   948→  // Vertical hover line

   951→   951→  ) : null;

   963→   963→        const dp = sc.trend.find(d => d.day === hoveredDay);
   964→   964→        if (!dp) return null;
   965→   965→        const val = mode === 'individual' ? dp.individualProb : dp.cumulativeProb;
   966→   966→        const color = SCENARIO_LINE_COLORS[sc.scenarioKey] || '#9ca3af';
   967→   967→        return (
   968→   968→          <div key={sc.scenarioKey} className="flex items-center justify-between gap-3">

   972→   972→            </span>

   975→   975→        );
   976→   976→      })}
   977→   977→    </div>
   978→   978→  ) : null;
   979→   979→
   980→   980→  return (
   981→   981→    <div className="rounded-2xl border border-[#e5e7eb] bg-[#ffffff] overflow-hidden"
   982→   982→      style={{ boxShadow: '0 2px 12px rgba(0,0,0,.05)' }}>

   987→   987→          {data.scenarios.map(sc => {
   988→   988→            const color = SCENARIO_LINE_COLORS[sc.scenarioKey] || '#9ca3af';
   989→   989→            const isActive = selectedKeys.has(sc.scenarioKey);
   990→   990→            const trendIcon = sc.trendDirection === 'rising' ? '↑' : sc.trendDirection === 'falling' ? '↓' : '→';
   991→   991→            const curVal = sc.trend[0] ? (mode === 'individual' ? sc.trend[0].individualProb : sc.trend[0].cumulativeProb) : 0;
   992→   992→            return (
