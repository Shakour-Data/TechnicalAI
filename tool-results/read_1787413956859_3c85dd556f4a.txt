   148→   148→
   149→   149→  const unit = currencyUnit || 'ریال';

   159→   159→
   160→   160→  const containerRef = useRef<HTMLDivElement>(null);

   168→   168→  const [aiAnalysis, setAiAnalysis] = useState<string | null>(null);

   197→   197→  // ── V10 Scenarios (LOCKED — backward compat) ───────────────
   198→   198→  const v10Result = useMemo(() => {
   199→   199→    if (analysisVersion !== 10) return null;
   200→   200→    return buildV10Locked({
   201→   201→      currentPrice,
   202→   202→      atr,
   203→   203→      overallSignal: overallSignal as 'bullish' | 'bearish' | 'neutral',
   204→   204→      v9Scenarios: scenarios,
   205→   205→      supports,
   206→   206→      resistances,
   207→   207→      bollingerUpper,
   208→   208→      bollingerLower,
   209→   209→      bollingerMiddle,
   210→   210→      ma21,
   211→   211→      ma100,
   212→   212→    });
   213→   213→  }, [analysisVersion, currentPrice, atr, overallSignal, scenarios, supports, resistances,
   214→   214→      bollingerUpper, bollingerLower, bollingerMiddle, ma21, ma100]);
   215→   215→
   216→   216→  // ── Fetch AI analysis (robust with logging + auto-retry) ─────────────
   217→   217→  const fetchAiAnalysis = useCallback(async (isRetry = false) => {
   218→   218→    const fetchId = ++aiFetchIdRef.current;
   219→   219→    setAiLoading(true);
   220→   220→    setAiError(null);
   221→   221→    try {
   222→   222→      const apiUrl = analysisVersion === 11 ? '/api/v11-analysis' : analysisVersion === 10 ? '/api/v10-analysis' : analysisVersion === 9 ? '/api/v9-analysis' : analysisVersion === 8 ? '/api/v8-analysis' : '/api/vdes-analysis';
   223→   223→      // V10/V11: use S-key scenarios, otherwise use V9 R-key scenarios
   224→   224→      const sKeyResult = analysisVersion === 11 ? v11Result : analysisVersion === 10 ? v10Result : null;
   225→   225→      const activeScenarios = sKeyResult ? sKeyResult.scenarios : scenarios;
   226→   226→      const requestBody: any = {
   227→   227→          symbolName, currentPrice, ma21, ma100,
   228→   228→          rsi, mfi, cci, adx, stochK, stochD,
   229→   229→          macdLine, macdSignal, macdHist,
   230→   230→          diPlus, diMinus, sar, atr, obv, hasVolume,
   231→   231→          bollingerUpper, bollingerMiddle, bollingerLower,
   232→   232→          trendDirection, trendAngle, trendR2, overallSignal,
   233→   233→          scenarios: activeScenarios, resistances, supports,
   234→   234→          resistanceStrengths, supportStrengths,
   235→   235→          isIndex, isTgju, currencyUnit,
   236→   236→        };
   237→   237→      // V10/V11: pass dominant + highest cumulative keys for narrative consistency
   238→   238→      if (sKeyResult) {
   239→   239→        if (analysisVersion === 11) {
   240→   240→          requestBody.v11DominantKey = sKeyResult.dominantKey;
   241→   241→          requestBody.v11HighestCumulativeKey = sKeyResult.highestCumulativeKey;
   242→   242→        } else if (analysisVersion === 10) {
   243→   243→          requestBody.v10DominantKey = sKeyResult.dominantKey;
   244→   244→          requestBody.v10HighestCumulativeKey = sKeyResult.highestCumulativeKey;
   245→   245→        }
   246→   246→      }
   247→   247→      // V9/V10/V11: include OHLCV history for ML training/prediction
   248→   248→      if ((analysisVersion === 9 || analysisVersion === 10 || analysisVersion === 11) && candles.length > 0) {
   249→   249→        requestBody.ohlcvHistory = candles.map(c => ({
   250→   250→          date: c.date, open: c.open, high: c.high,
   251→   251→          low: c.low, close: c.close, volume: c.volume,
   252→   252→        }));
   253→   253→      }
   254→   254→
   255→   255→      // AbortController for 90s client-side timeout
   256→   256→      const controller = new AbortController();
   257→   257→      const timeoutId = setTimeout(() => controller.abort(), 90000);
   258→   258→
   259→   259→      console.log(`[VdesAnalysis v${analysisVersion}] Fetching ${apiUrl} for ${symbolName} (retry=${isRetry})`);
   260→   260→      const res = await fetch(apiUrl, {
   261→   261→        method: 'POST',
   262→   262→        headers: { 'Content-Type': 'application/json' },
   263→   263→        body: JSON.stringify(requestBody),
   264→   264→        signal: controller.signal,
   265→   265→      });
   266→   266→      clearTimeout(timeoutId);
   267→   267→
   268→   268→      if (!res.ok) {
   269→   269→        let errorMsg = `خطای سرور (${res.status})`;
   270→   270→        try {
   271→   271→          const errData = await res.json();
   272→   272→          errorMsg = errData.error || errorMsg;
   273→   273→        } catch {
   274→   274→          // Response body is not JSON (e.g. 502 from Caddy)
   275→   275→          if (res.status === 502) errorMsg = 'سرور در حال بارگذاری مجدد است. لطفاً چند ثانیه صبر کنید و دوباره تلاش کنید.';
   276→   276→        }
   277→   277→        console.error(`[VdesAnalysis v${analysisVersion}] API error ${res.status}:`, errorMsg);
   278→   278→        throw new Error(errorMsg);
   279→   279→      }
   280→   280→      const data = await res.json();
   281→   281→      if (fetchId === aiFetchIdRef.current) {
   282→   282→        if (data.analysis && data.analysis.trim().length > 0) {
   283→   283→          setAiAnalysis(data.analysis);
   284→   284→          console.log(`[VdesAnalysis v${analysisVersion}] Got analysis for ${symbolName}: ${data.analysis.length} chars`);
   285→   285→        } else {
   286→   286→          console.error(`[VdesAnalysis v${analysisVersion}] Empty analysis for ${symbolName}`, data);
   287→   287→          setAiError('مدل پاسخ خالی تولید کرد. لطفاً دوباره تلاش کنید.');
   288→   288→        }
   289→   289→        if (data._debug?.v11) setV11Debug({ ...data._debug.v11, v11Persona: data._debug.v11Persona, v11Template: data._debug.v11Template });
   290→   290→        else if (data._debug?.v10) setV11Debug({ ...data._debug.v10, v11Persona: data._debug.v10Persona, v10Template: data._debug.v10Template });
   291→   291→        else setV11Debug(null);
   292→   292→      }
   293→   293→    } catch (err) {
   294→   294→      console.error(`[VdesAnalysis v${analysisVersion}] Error for ${symbolName}:`, err);
   295→   295→      if (fetchId === aiFetchIdRef.current) {
   296→   296→        if (err instanceof Error && err.name === 'AbortError') {
   297→   297→          setAiError('زمان پاسخگویی به پایان رسید. لطفاً دوباره تلاش کنید.');
   298→   298→        } else {
   299→   299→          setAiError(err instanceof Error ? err.message : 'خطای ناشناخته');
   300→   300→        }
   301→   301→        // Auto-retry once on transient errors (503/429/timeout) after 5s
   302→   302→        if (!isRetry) {
   303→   303→          const errMsg = err instanceof Error ? err.message : '';
   304→   304→          const isTransient = errMsg.includes('503') || errMsg.includes('429') || errMsg.includes('پایان رسید') || errMsg.includes('بارگذاری مجدد');
   305→   305→          if (isTransient) {
   306→   306→            console.log(`[VdesAnalysis] Auto-retry in 5s for ${symbolName}`);
   307→   307→            setTimeout(() => {
   308→   308→              if (fetchId === aiFetchIdRef.current) fetchAiAnalysis(true).catch(() => {});
   309→   309→            }, 5000);
   310→   310→          }
   311→   311→        }
   312→   312→      }
   313→   313→    } finally {
   314→   314→      if (fetchId === aiFetchIdRef.current) {
   315→   315→        setAiLoading(false);
   316→   316→      }
   317→   317→    }
   318→   318→  }, [symbolName, currentPrice, ma21, ma100, rsi, mfi, cci, adx, stochK, stochD,
   319→   319→    macdLine, macdSignal, macdHist, diPlus, diMinus, sar, atr, obv, hasVolume,
   320→   320→    bollingerUpper, bollingerMiddle, bollingerLower, trendDirection, trendAngle, trendR2,
   321→   321→    overallSignal, scenarios, resistances, supports, resistanceStrengths, supportStrengths,
   322→   322→    isIndex, isTgju, currencyUnit, analysisVersion, v11Result, v10Result]);
   323→   323→
   324→   324→  // ── Auto-fetch AI analysis only on symbol or version change ──
   325→   325→  // NOT on every data refresh (auto-refresh every 15min would re-trigger
   326→   326→  // a 10-25s LLM call, making the text disappear each time)
   327→   327→  const fetchTriggerKey = `${symbolName}:${analysisVersion}`;
   328→   328→  const lastAutoFetchKeyRef = useRef('');
   329→   329→  // Keep a ref to fetchAiAnalysis so the effect always calls the latest version
   330→   330→  const fetchAiAnalysisRef = useRef(fetchAiAnalysis);
   331→   331→  fetchAiAnalysisRef.current = fetchAiAnalysis;
   332→   332→  useEffect(() => {
   333→   333→    if (fetchTriggerKey === lastAutoFetchKeyRef.current) return;
   334→   334→    if (!symbolName || !currentPrice || currentPrice <= 0) {
   335→   335→      console.log('[VdesAnalysis] Skipping fetch: no symbol or invalid price', { symbolName, currentPrice });
   336→   336→      return;
   337→   337→    }
   338→   338→    lastAutoFetchKeyRef.current = fetchTriggerKey;
   339→   339→    // Reset old text when switching symbols/versions
   340→   340→    setAiAnalysis(null);
   341→   341→    setAiError(null);
   342→   342→    // Small delay to ensure all useMemo values (v11Result etc.) are computed
   343→   343→    const timer = setTimeout(() => {
   344→   344→      fetchAiAnalysisRef.current().catch((e) => {
   345→   345→        console.error('[VdesAnalysis] Auto-fetch failed:', e);
   346→   346→      });
   347→   347→    }, 150);
   348→   348→    return () => clearTimeout(timer);
   349→   349→  }, [fetchTriggerKey, symbolName, currentPrice]);
   350→   350→
   351→   351→  // ── Close export menu on outside click ───────────────────────
   352→   352→  useEffect(() => {
   353→   353→    if (!exportMenuOpen) return;
   354→   354→    const handler = (e: MouseEvent) => {
   355→   355→      if (exportMenuRef.current && !exportMenuRef.current.contains(e.target as Node)) {
   356→   356→        setExportMenuOpen(false);
   357→   357→      }

  1014→  1014→  grade?: string; overlapCount?: number; touchCount?: number;
