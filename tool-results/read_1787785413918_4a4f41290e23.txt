   237→/* --─ Recent Searches ------------------------------------ */
   238→
   239→const RECENT_KEY = 'fin-search-recent';
   240→const MAX_RECENT = 8;
   241→
   242→function getRecentSearches(): string[] {
   243→  if (typeof window === 'undefined') return [];
   244→  try { const raw = localStorage.getItem(RECENT_KEY); return raw ? JSON.parse(raw) : []; }
   245→  catch { return []; }
   246→}
   247→
   248→function addRecentSearch(name: string) {
   249→  if (typeof window === 'undefined' || !name) return;
   250→  try {
   251→    let list = getRecentSearches();
   252→    list = list.filter((r) => r !== name);
   253→    list.unshift(name);
   254→    if (list.length > MAX_RECENT) list = list.slice(0, MAX_RECENT);
   255→    localStorage.setItem(RECENT_KEY, JSON.stringify(list));
   256→  } catch { /* ignore */ }
   257→}
