   118→// Component
   119→// ═══════════════════════════════════════════════════════════════════════════════
   120→
   121→export default function VdssGraph(props: VdssGraphProps) {
   122→  const { symbolName, currentPrice, resistances, supports, ma100, scenarios, rsi, adx, bullScore, edgeWeights, currencyUnit, overallSignal } = props;
   123→
   124→  // ── Dynamic scenario labels (context-aware) ──────────────────
   125→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal || 'neutral'), [overallSignal]);
   126→
   127→  // ── Dynamic NODE_DEFS: R1-R5 titles/descriptions adapt to signal ──
   128→  const nodeDefs = useMemo(() => {
   129→    const base = { ...NODE_DEFS };
   130→    for (const key of SCENARIO_KEYS) {
   131→      if (base[key]) {
   132→        base[key] = { ...base[key], title: dynamicMeta[key].nodeTitle, desc: dynamicMeta[key].nodeDesc };
   133→      }
   134→    }
   135→    return base;
   136→  }, [dynamicMeta]);
   137→  const unit = currencyUnit || 'ریال';
   138→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   139→  const S1_level = supports[0] ?? currentPrice * 0.95;
   140→
   141→  const [activeFilter, setActiveFilter] = useState('all');
   142→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   143→  const [selectedPathIdx, setSelectedPathIdx] = useState<number | null>(null);
   144→  const graphRef = useRef<HTMLDivElement>(null);
   145→  const svgRef = useRef<SVGSVGElement>(null);
   146→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   147→  const shellRef = useRef<HTMLDivElement>(null);
   148→
   149→  // ── Edge probabilities (using TA engine's bullScore) ──────────────
   150→  const edgeProbs = useMemo(() =>
   151→    edgeWeights ? calcEdgeProbabilities(edgeWeights) : {},
   152→  [edgeWeights]
   153→  );
   154→
   155→  // ── All paths with raw probabilities ─────────────────────────────
   156→  const allPathsRaw = useMemo(() => {
   157→    if (Object.keys(edgeProbs).length === 0) return [];
   158→    return findAllPaths(edgeProbs);
   159→  }, [edgeProbs]);
   160→
   161→  // ── Calibrate path probabilities to match TA engine scenario probabilities ──
   162→  const allPaths = useMemo(() => {
   163→    // Compute raw sum per target
   164→    const rawSums: Record<string, number> = {};
   165→    for (const key of SCENARIO_KEYS) rawSums[key] = 0;
   166→    for (const p of allPathsRaw) rawSums[p.target] += p.prob;
   167→
   168→    // Scale each path so the sum per target matches scenario probability
   169→    return allPathsRaw.map(p => {
   170→      const rawSum = rawSums[p.target] || 0.0001;
   171→      const targetProb = scenarios[p.target as keyof typeof scenarios].probability / 100;
   172→      const scale = targetProb / rawSum;
   173→      return { ...p, prob: p.prob * scale };
   174→    });
   175→  }, [allPathsRaw, scenarios]);
   176→
   177→  // ── Filtered paths (sorted by probability descending) ─────────────────
   178→  const filteredPaths = useMemo(() => {
   179→    if (activeFilter === 'all') return [...allPaths].sort((a, b) => b.prob - a.prob);
   180→    // Check if filter is a scenario key
   181→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   182→      return allPaths.filter(p => p.target === activeFilter).sort((a, b) => b.prob - a.prob);
   183→    }
   184→    // Type filter: paths must use only edges of this type
   185→    return allPaths.filter(p =>
   186→      p.edges.every(ei => EDGES[ei][3] === activeFilter)
   187→    ).sort((a, b) => b.prob - a.prob);
   188→  }, [allPaths, activeFilter]);
   189→
   190→  // ── Visible edges/nodes based on filter (or selected path) ──────────────
   191→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   192→    // If a specific path is selected, show only that path's nodes/edges
   193→    if (selectedPathIdx !== null && filteredPaths[selectedPathIdx]) {
   194→      const p = filteredPaths[selectedPathIdx];
   195→      return {
   196→        visibleEdgeIndices: p.edges,
   197→        visibleNodes: new Set([...p.nodes, 'A']),
   198→      };
   199→    }
   200→    if (activeFilter === 'all') {
   201→      const allEdgeIdx = EDGES.map((_, i) => i);
   202→      const allNodes = new Set(Object.keys(NODE_POSITIONS));
   203→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   204→    }
   205→    if (SCENARIO_KEYS.includes(activeFilter as any)) {
   206→      // Show all edges/nodes on paths to this scenario
   207→      const edgeSet = new Set<number>();
   208→      const nodeSet = new Set<string>();
   209→      for (const p of filteredPaths) {
   210→        p.edges.forEach(ei => edgeSet.add(ei));
   211→        p.nodes.forEach(n => nodeSet.add(n));
   212→      }
   213→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   214→    }
   215→    // Type filter
   216→    const edgeSet = new Set<number>();
   217→    const nodeSet = new Set<string>();
   218→    EDGES.forEach((e, i) => {
   219→      if (e[3] === activeFilter) {
   220→        edgeSet.add(i);
   221→        nodeSet.add(e[0]);
   222→        nodeSet.add(e[1]);
   223→      }
   224→    });
   225→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   226→  }, [activeFilter, filteredPaths, selectedPathIdx]);
   227→
   228→  // ── Node values ──────────────────────────────────────────────────
   229→  const nodeValues: Record<string, string> = {
   230→    A: toFa(currentPrice),
   231→    B: toFa(R1_level),
   232→    D: toFa(resistances[1] ?? currentPrice * 1.10),
   233→    E: `${toFa(Math.round(R1_level + (resistances[1] ?? R1_level * 1.05 - R1_level) * 0.5))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 0.8))}`,
   234→    F: `${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 1.2))} — ${toFa(Math.round((resistances[1] ?? R1_level * 1.05) + ((resistances[1] ?? R1_level * 1.05) - R1_level) * 2.0))}`,
   235→    G: toFa(S1_level),
   236→    H: toFa(supports[1] ?? currentPrice * 0.90),
   237→    I: toFa(supports[2] ?? currentPrice * 0.85),
   238→    J: toFa(supports[3] ?? currentPrice * 0.80),
   239→    K: toFa(ma100),
   240→    L: props.trendDirection === 'up' ? 'صعودی' : props.trendDirection === 'down' ? 'نزولی' : 'خنثی',
   241→  };
   242→  for (const key of SCENARIO_KEYS) {
   243→    const s = scenarios[key];
   244→    nodeValues[key] = `${toFa(s.targetMin)} — ${toFa(s.targetMax)} ${unit}`;
   245→  }
   246→
   247→  // ── Draw SVG edges ───────────────────────────────────────────────
   248→  const drawEdges = useCallback(() => {
   249→    const svg = svgRef.current;
   250→    const graph = graphRef.current;
   251→    if (!svg || !graph) return;
   252→
   253→    let markersSvg = '';
   254→    for (const [k, c] of Object.entries(EDGE_COLORS)) {
   255→      markersSvg += `<marker id="arrow-${k}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   256→    }
   257→
   258→    let pathsSvg = '';
   259→    const graphRect = graph.getBoundingClientRect();
   260→
   261→    for (let ei = 0; ei < EDGES.length; ei++) {
   262→      const [fromId, toId, label, type] = EDGES[ei];
   263→      const isVisible = visibleEdgeIndices.includes(ei);
   264→      const fromEl = nodeRefs.current[fromId];
   265→      const toEl = nodeRefs.current[toId];
   266→      if (!fromEl || !toEl) continue;
   267→
   268→      const fRect = fromEl.getBoundingClientRect();
   269→      const tRect = toEl.getBoundingClientRect();
   270→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   271→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   272→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   273→      const by = tRect.top - graphRect.top + tRect.height / 2;
   274→      const dx = bx - ax;
   275→      const dy = by - ay;
   276→      const dist = Math.sqrt(dx * dx + dy * dy);
   277→      if (dist === 0) continue;
   278→
   279→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   280→      const mx = (ax + bx) / 2;
   281→      const my = (ay + by) / 2;
   282→      const nx = -dy / dist;
   283→      const ny = dx / dist;
   284→      const cx = mx + nx * bend;
   285→      const cy = my + ny * bend;
   286→
   287→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   288→      const prob = edgeProbs[String(ei)] ?? 0;
   289→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   290→
   291→      pathsSvg += `<path d="${d}" stroke="${EDGE_COLORS[type]}" stroke-width="${isVisible ? 2 : 1}" opacity="${isVisible ? 0.72 : 0.06}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   292→
   293→      // Edge probability label (only for visible edges)
   294→      if (isVisible) {
   295→        pathsSvg += `<text x="${cx}" y="${cy - 6}" fill="#374151" font-size="9" text-anchor="middle" paint-order="stroke" stroke="#f3f4f6" stroke-width="3.5" stroke-linejoin="round" opacity="0.85" data-type="${type}" class="edge-label">${label}</text>`;
   296→        pathsSvg += `<text x="${cx}" y="${cy + 8}" fill="${EDGE_COLORS[type]}" font-size="10" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#f3f4f6" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   297→      }
   298→    }
   299→
   300→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   301→  }, [visibleEdgeIndices, edgeProbs]);
   302→
   303→  useEffect(() => { drawEdges(); }, [drawEdges]);
   304→  useEffect(() => {
   305→    const el = shellRef.current;
   306→    if (!el) return;
   307→    const observer = new ResizeObserver(() => drawEdges());
   308→    observer.observe(el);
   309→    return () => observer.disconnect();
   310→  }, [drawEdges]);
   311→
   312→  // ── Aggregated path probabilities per target (calibrated = matches scenario prob) ──
   313→  const pathProbsByTarget = useMemo(() => {
   314→    const result: Record<string, number> = {};
   315→    for (const key of SCENARIO_KEYS) {
   316→      result[key] = 0;
   317→      for (const p of filteredPaths) {
   318→        if (p.target === key) result[key] = (result[key] ?? 0) + p.prob;
   319→      }
   320→      // Fallback: if path probability is 0 but scenario has probability, use scenario directly
   321→      if (result[key] === 0 && scenarios[key].probability > 0) {
   322→        result[key] = scenarios[key].probability / 100;
   323→      }
   324→    }
   325→    return result;
   326→  }, [filteredPaths, scenarios]);
   327→
   328→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   329→
   330→  // ── Detail panel content ─────────────────────────────────────────
   331→  const detailContent = useMemo(() => {
   332→    if (!selectedNode) return null;
   333→    const def = nodeDefs[selectedNode];
   334→    if (!def) return null;
   335→    const relations = EDGES.filter(e => e[0] === selectedNode || e[1] === selectedNode);
   336→    const inputs = relations.filter(e => e[1] === selectedNode);
   337→    const outputs = relations.filter(e => e[0] === selectedNode);
   338→
   339→    return (
   340→      <div className="space-y-3">
   341→        <h3 className="text-sm font-bold text-gray-900">{def.title}</h3>
   342→        <p className="text-xs text-gray-600"><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   343→        <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border border-gray-300 bg-gray-100 text-gray-600">{def.type}</span>
   344→        <p className="text-xs text-gray-500 leading-relaxed">{def.desc}</p>
   345→        {inputs.length > 0 && (
   346→          <div>
   347→            <p className="text-xs font-medium text-gray-600 mb-1">مسیرهای ورودی ({inputs.length}):</p>
   348→            <ul className="space-y-1">
   349→              {inputs.map((e, i) => {
   350→                const ei = EDGES.indexOf(e);
   351→                const ep = edgeProbs[String(ei)] ?? 0;
   352→                return (
   353→                  <li key={i} className="text-[11px] text-gray-500 leading-relaxed border-t border-dashed border-gray-300 pt-1.5">
   354→                    <b className="text-gray-500">{e[0]} ← {e[1]}</b>
   355→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${EDGE_COLORS[e[3]]}20`, color: EDGE_COLORS[e[3]] }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   356→                    <br />{e[2]}
   357→                  </li>
   358→                );
   359→              })}
   360→            </ul>
   361→          </div>
   362→        )}
   363→        {outputs.length > 0 && (
   364→          <div>
   365→            <p className="text-xs font-medium text-gray-600 mb-1">مسیرهای خروجی ({outputs.length}):</p>
   366→            <ul className="space-y-1">
   367→              {outputs.map((e, i) => {
   368→                const ei = EDGES.indexOf(e);
   369→                const ep = edgeProbs[String(ei)] ?? 0;
   370→                return (
   371→                  <li key={i} className="text-[11px] text-gray-500 leading-relaxed border-t border-dashed border-gray-300 pt-1.5">
   372→                    <b className="text-gray-500">{e[0]} → {e[1]}</b>
