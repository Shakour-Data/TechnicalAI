   223→   223→
   224→   224→function resolveAsset(raw: string): AssetType {
   225→   225→  const v = raw.toLowerCase();
   226→   226→  if (v.includes('crypto') || v.includes('btc') || v.includes('eth') || v.includes('usdt')) return 'cryptocurrency';
   227→   227→  if (v.includes('forex') || v.includes('eur') || v.includes('usd') || v.includes('jpy')) return 'forex';
   228→   228→  if (v.includes('gold') || v.includes('طلا') || v.includes('xau') || v.includes('oil') || v.includes('نفت')) return 'commodities';
   229→   229→  return 'stocks';
   230→   230→}
   231→   231→
   232→   232→function resolveTimeframe(raw: string): TimeframeType {
   233→   233→  const v = raw.toLowerCase();
   234→   234→  if (v.includes('h') || v.includes('hour') || v.includes('4h') || v.includes('1h')) return 'hourly';
   235→   235→  if (v.includes('w') || v.includes('week')) return 'weekly';
   236→   236→  if (v.includes('m') || v.includes('month')) return 'monthly';
   237→   237→  return 'daily';
   238→   238→}

   240→   240→function resolveRegime(raw: string): RegimeType {
   241→   241→  // Support new 5-state regime from regime-engine.ts
   242→   242→  if (raw === 'TRENDING_UP') return 'Strong Bull';
   243→   243→  if (raw === 'TRENDING_DOWN') return 'Strong Bear';
   244→   244→  if (raw === 'RANGING' || raw === 'VOLATILE') return 'Range';
   245→   245→  if (raw === 'BREAKOUT') return 'Strong Bull'; // breakout assumed bullish by default
   246→   246→  // Legacy support
   247→   247→  if (raw.includes('Strong Bull') || raw.includes('strong_bull')) return 'Strong Bull';
   248→   248→  if (raw.includes('Weak Bull') || raw.includes('weak_bull')) return 'Weak Bull';
   249→   249→  if (raw.includes('Strong Bear') || raw.includes('strong_bear')) return 'Strong Bear';
   250→   250→  if (raw.includes('Weak Bear') || raw.includes('weak_bear')) return 'Weak Bear';
   251→   251→  return 'Range';
   252→   252→}

   254→   254→function toPersianNum(n: number): string {
   255→   255→  if (!isFinite(n) || isNaN(n)) return '\u06f0';
   256→   256→  const fixed = Math.round(n * 100) / 100;
   257→   257→  return String(fixed).replace(/\d/g, d => String.fromCharCode('\u0660'.charCodeAt(0) + parseInt(d)));
   258→   258→}

   264→   264→function selectSchools(ctx: MSLV4Context) {
   265→   265→  const asset = resolveAsset(ctx.asset);
   266→   266→  const tf = resolveTimeframe(ctx.timeframe);
   267→   267→  const regime = resolveRegime(ctx.regime);
   268→   268→  const am = ASSET_MULT[asset] ?? ASSET_MULT.stocks;
   269→   269→  const tm = TF_MULT[tf] ?? TF_MULT.daily;
   270→   270→  const rm = REGIME_MULT[regime] ?? REGIME_MULT.Range;
   271→   271→
   272→   272→  const pt = ctx.probTrend;
   273→   273→
   274→   274→  const rawWeights: Record<SchoolId, number> = {} as Record<SchoolId, number>;
   275→   275→  for (const s of SCHOOLS) {
   276→   276→    let w = s.base_weight * am[s.id] * tm[s.id] * rm[s.id];
   277→   277→    // Adaptive: zero out volume school if no volume data
   278→   278→    if (s.id === 'volume' && ctx.hasVolume === false) w *= 0.2;
   279→   279→
   280→   280→    // Probability trend adjustments (cumulative weighted 2x more than individual)
   281→   281→    if (pt) {
   282→   282→      // Oscillator school: if cumulative bearish trend rising → divergence/reversal signals
   283→   283→      if (s.id === 'oscillator' && pt.bearGroupDir === 'rising' && pt.bullGroupCum && pt.bullGroupCum > 0.3) {
   284→   284→        w *= 1.25; // Bearish momentum building, oscillators useful for reversal timing
   285→   285→      }
   286→   286→      // Classical school: strong directional cumulative trend → support/resistance focus
   287→   287→      if (s.id === 'classical' && (pt.bullGroupDir === 'rising' || pt.bearGroupDir === 'rising')) {
   288→   288→        w *= 1.15; // Clear trend direction, classical S/R analysis
   289→   289→      }
   290→   290→      // Elliott school: volatile or changing cumulative trends → wave analysis
   291→   291→      if (s.id === 'elliott' && (pt.bullGroupDir === 'volatile' || pt.bearGroupDir === 'volatile')) {
   292→   292→        w *= 1.2; // Volatile probability shifts suggest wave structure changes
   293→   293→      }
   294→   294→      // Harmonic school: trend reversal in cumulative probabilities
   295→   295→      if (s.id === 'harmonic' && (
   296→   296→        (pt.bullGroupDir === 'falling' && pt.bullGroupCum && pt.bullGroupCum > 0.4) ||
   297→   297→        (pt.bearGroupDir === 'falling' && pt.bearGroupCum && pt.bearGroupCum > 0.4)
   298→   298→      )) {
   299→   299→        w *= 1.2; // Declining dominant group suggests reversal zone
   300→   300→      }
   301→   301→      // Hybrid school: neutral dominant and stable trends
   302→   302→      if (s.id === 'hybrid' && pt.neutralGroupDir === 'stable' && pt.neutralGroupCum && pt.neutralGroupCum > 0.2) {
   303→   303→        w *= 1.15; // Range-bound market, hybrid approach best
   304→   304→      }
   305→   305→      // Volume school: strong directional 7-day change
   306→   306→      if (s.id === 'volume' && ctx.hasVolume !== false) {
   307→   307→        const maxChange = Math.max(Math.abs(pt.bullGroupChange7d ?? 0), Math.abs(pt.bearGroupChange7d ?? 0));
   308→   308→        if (maxChange > 0.15) w *= 1.15; // Significant probability shift, volume confirms
   309→   309→      }
   310→   310→    }
   311→   311→
   312→   312→    rawWeights[s.id] = w;
   313→   313→  }
   314→   314→
   315→   315→  const total = Object.values(rawWeights).reduce((a, b) => a + b, 0);
   316→   316→  const normWeights: Record<string, number> = {};
   317→   317→  for (const id of SCHOOL_IDS) normWeights[id] = rawWeights[id] / total;
   318→   318→
   319→   319→  const sorted = SCHOOL_IDS.map(id => ({ id, w: normWeights[id] })).sort((a, b) => b.w - a.w);
   320→   320→  const primary = sorted[0];
   321→   321→  const secondary = sorted[1];
   322→   322→
   323→   323→  let status: string;
   324→   324→  if (primary.id === 'hybrid' && primary.w < 0.35) {
   325→   325→    status = 'hybrid';
   326→   326→  } else if (primary.w > 0.45) {
   327→   327→    status = 'confirmed';
   328→   328→  } else if (primary.w > 0.35) {
   329→   329→    status = 'tentative';
   330→   330→  } else {
   331→   331→    status = 'hybrid';
   332→   332→  }
   333→   333→
   334→   334→  const primaryDef = SCHOOLS.find(s => s.id === primary.id)!;
   335→   335→  const secondaryDef = SCHOOLS.find(s => s.id === secondary.id)!;
   336→   336→
   337→   337→  const reasoning = buildSchoolReasoning(primaryDef, normWeights, ctx, asset, tf, regime);
   338→   338→
   339→   339→  return {
   340→   340→    primary: {
   341→   341→      id: primary.id, name: primaryDef.name, nameEn: primaryDef.nameEn,
   342→   342→      weight: Math.round(primary.w * 100) / 100,
   343→   343→      confidence: ctx.confidence > 0.75 ? 'high' : ctx.confidence > 0.5 ? 'medium' : 'low',
   344→   344→      status, reasoning,
   345→   345→    },
   346→   346→    secondary: {
   347→   347→      id: secondary.id, name: secondaryDef.name,
   348→   348→      weight: Math.round(secondary.w * 100) / 100,
   349→   349→      reasoning: `مکتب ${secondaryDef.name} با وزن ${toPersianNum(secondary.w)} به عنوان مکتب مکمل انتخاب شد`,
   350→   350→    },
   351→   351→    all_schools: Object.fromEntries(SCHOOL_IDS.map(id => [id, Math.round(normWeights[id] * 100) / 100])),
   352→   352→  };
   353→   353→}

   355→   355→function buildSchoolReasoning(s: SchoolDef, w: Record<string, number>, ctx: MSLV4Context, asset: string, tf: string, regime: string): string[] {
   356→   356→  const r: string[] = [];
   357→   357→  const pc = ctx.pattern_counts?.classical ?? 0;
   358→   358→  const phc = ctx.pattern_counts?.harmonic ?? 0;
   359→   359→  const ec = ctx.pattern_counts?.elliott ?? 0;
   360→   360→  if (s.id === 'classical' && pc >= 2) r.push(`تشخیص ${toPersianNum(pc)} الگوی کلاسیک`);
   361→   361→  if (s.id === 'oscillator' && ctx.divergence_present) r.push('واگرایی تشخیص داده شده');
   362→   362→  if (s.id === 'oscillator' && ctx.rsi !== undefined && (ctx.rsi > 70 || ctx.rsi < 30)) r.push('RSI در منطقه اشباع');
   363→   363→  if (s.id === 'harmonic' && phc >= 2) r.push(`تشخیص ${toPersianNum(phc)} الگوی هارمونیک`);
   364→   364→  if (s.id === 'elliott' && ec >= 3) r.push(`تشخیص ${toPersianNum(ec)} موج الیوت`);
   365→   365→  if (s.id === 'volume' && ctx.hasVolume) r.push('داده‌های حجم موجود و تأییدکننده');
   366→   366→  r.push(`رژیم بازار ${regime} تطابق دارد`);
   367→   367→  r.push(`نوع دارایی: ${asset}، تایم‌فریم: ${tf}`);
   368→   368→  return r;
   369→   369→}

   375→   375→function selectStyles(primarySchoolId: SchoolId, ctx: MSLV4Context) {
   376→   376→  const scores: Record<StyleId, number> = {} as Record<StyleId, number>;
   377→   377→  const audience = ctx.audience_level ?? 'pro';
   378→   378→  const pt = ctx.probTrend;
   379→   379→
   380→   380→  for (const style of STYLES) {
   381→   381→    const compatibility = SS_MATRIX[primarySchoolId]?.[style.id] ?? 0.5;
   382→   382→    const audienceMatch = getAudienceMatch(style.id, audience);
   383→   383→    const marketMatch = getMarketContextMatch(style.id, ctx);
   384→   384→    const schoolMatch = SS_MATRIX[primarySchoolId]?.[style.id] ?? 0.5;
   385→   385→
   386→   386→    // Probability trend influence on style (cumulative weighted 2x)
   387→   387→    let trendStyleBonus = 0;
   388→   388→    if (pt) {
   389→   389→      const cumDir = pt.bullGroupDir === 'rising' ? 1 : pt.bearGroupDir === 'rising' ? -1 : 0;
   390→   390→      const indivDir = pt.dominantIndivDir === 'rising' ? 1 : pt.dominantIndivDir === 'falling' ? -1 : 0;
   391→   391→      const trendSignal = cumDir * 0.67 + indivDir * 0.33; // cumulative 2x weight
   392→   392→
   393→   393→      if (style.id === 'forecasting' && Math.abs(trendSignal) > 0.5) trendStyleBonus = 0.1;
   394→   394→      if (style.id === 'trading' && Math.abs(trendSignal) > 0.3) trendStyleBonus = 0.08;
   395→   395→      if (style.id === 'analytical' && Math.abs(trendSignal) < 0.3) trendStyleBonus = 0.08;
   396→   396→      if (style.id === 'executive' && (pt.bullGroupDir === 'stable' || pt.bearGroupDir === 'stable')) trendStyleBonus = 0.05;
   397→   397→    }
   398→   398→
   399→   399→    scores[style.id] = compatibility * 0.35 + audienceMatch * 0.25 + marketMatch * 0.2 + schoolMatch * 0.1 + trendStyleBonus;
   400→   400→  }
   401→   401→
   402→   402→  const sorted = (Object.entries(scores) as [StyleId, number][]).sort((a, b) => b[1] - a[1]);
   403→   403→  const [pId, pScore] = sorted[0];
   404→   404→  const [sId, sScore] = sorted[1];
   405→   405→  const pDef = STYLES.find(s => s.id === pId)!;
   406→   406→  const sDef = STYLES.find(s => s.id === sId)!;
   407→   407→
   408→   408→  return {
   409→   409→    primary: {
   410→   410→      id: pId, name: pDef.name, nameEn: pDef.nameEn,
   411→   411→      score: Math.round(pScore * 100) / 100,
   412→   412→      detail_level: pDef.detailLevel, length: pDef.length, focus: pDef.focus,
   413→   413→      reasoning: buildStyleReasoning(pDef, primarySchoolId, ctx),
   414→   414→    },
   415→   415→    secondary: { id: sId, name: sDef.name, score: Math.round(sScore * 100) / 100 },
   416→   416→  };
   417→   417→}

   426→   426→    : audience.includes('mid') || audience.includes('متوسط') ? 'mid' : 'pro';

   430→   430→function getMarketContextMatch(styleId: StyleId, ctx: MSLV4Context): number {

   436→   436→  if (styleId === 'forecasting' && ctx.confidence > 0.5) score = 0.8;
   437→   437→  if (styleId === 'analytical' && ctx.confidence > 0.5) score = 0.85;
   438→   438→  if (styleId === 'educational' && ctx.confidence < 0.5) score = 0.85;
   439→   439→  return score;

   455→   455→function selectTones(schoolId: SchoolId, styleId: StyleId, ctx: MSLV4Context) {
   456→   456→  const scores: Record<ToneId, number> = {} as Record<ToneId, number>;
   457→   457→  const pt = ctx.probTrend;
   458→   458→
   459→   459→  for (const tone of TONES) {
   460→   460→    const condMatch = checkToneConditions(tone.id, ctx);
   461→   461→    const schoolMatch = ST_MATRIX[schoolId]?.[tone.id] ?? 0.5;
   462→   462→    const styleMatch = STYLE_TONE_MATRIX[styleId]?.[tone.id] ?? 0.5;
   463→   463→    const intensity = 0.5 + ctx.confidence * 0.3 + ctx.volatility * 0.2;
   464→   464→
   465→   465→    // Probability trend influence on tone (cumulative weighted 2x)
   466→   466→    let trendToneBonus = 0;
   467→   467→    if (pt) {
   468→   468→      const cumDir = pt.bullGroupDir === 'rising' ? 1 : pt.bearGroupDir === 'rising' ? -1 : 0;
   469→   469→      const indivDir = pt.dominantIndivDir === 'rising' ? 1 : pt.dominantIndivDir === 'falling' ? -1 : 0;
   470→   470→      const trendSignal = cumDir * 0.67 + indivDir * 0.33;
   471→   471→
   472→   472→      // Cumulative trend overrides basic regime check
   473→   473→      if (tone.id === 'optimistic' && pt.bullGroupDir === 'rising' && (pt.bullGroupChange7d ?? 0) > 0.05) {
   474→   474→        trendToneBonus = 0.15;
   475→   475→      }
   476→   476→      if (tone.id === 'warning' && pt.bearGroupDir === 'rising' && (pt.bearGroupChange7d ?? 0) > 0.05) {
   477→   477→        trendToneBonus = 0.15;
   478→   478→      }
   479→   479→      if (tone.id === 'conservative' && pt.neutralGroupDir === 'stable' && (pt.neutralGroupCum ?? 0) > 0.15) {
   480→   480→        trendToneBonus = 0.12;
   481→   481→      }
   482→   482→      if (tone.id === 'aggressive' && Math.abs(trendSignal) > 0.67 && Math.abs(pt.bullGroupChange7d ?? 0) > 0.1) {
   483→   483→        trendToneBonus = 0.1;
   484→   484→      }
   485→   485→      if (tone.id === 'realistic' && (pt.bullGroupDir === 'volatile' || pt.bearGroupDir === 'volatile')) {
   486→   486→        trendToneBonus = 0.12;
   487→   487→      }
   488→   488→    }
   489→   489→
   490→   490→    scores[tone.id] = condMatch * 0.4 + schoolMatch * 0.15 + styleMatch * 0.15 + Math.min(intensity, 1) * 0.1 + trendToneBonus;
   491→   491→  }
   492→   492→
   493→   493→  const sorted = (Object.entries(scores) as [ToneId, number][]).sort((a, b) => b[1] - a[1]);
   494→   494→  const [pId, pScore] = sorted[0];
   495→   495→  const [sId, sScore] = sorted[1];
   496→   496→  const pDef = TONES.find(t => t.id === pId)!;
   497→   497→  const sDef = TONES.find(t => t.id === sId)!;
   498→   498→  const intensity = calcIntensity(pId, ctx);
   499→   499→
   500→   500→  return {
   501→   501→    primary: {
   502→   502→      id: pId, name: pDef.name, nameEn: pDef.nameEn,
   503→   503→      score: Math.round(pScore * 100) / 100,
   504→   504→      intensity, keywords: pDef.keywords.slice(0, 3),
   505→   505→      reasoning: buildToneReasoning(pDef, ctx),
   506→   506→    },
   507→   507→    secondary: { id: sId, name: sDef.name, score: Math.round(sScore * 100) / 100 },
   508→   508→  };
   509→   509→}

   511→   511→function checkToneConditions(toneId: ToneId, ctx: MSLV4Context): number {
   512→   512→  const isBull = ctx.regime.includes('Bull');
   513→   513→  const isBear = ctx.regime.includes('Bear');
   514→   514→  const pt = ctx.probTrend;
   515→   515→
   516→   516→  switch (toneId) {
   517→   517→    case 'conservative': {
   518→   518→      let s = ctx.confidence < 0.6 ? 0.9 : ctx.volatility > 0.03 ? 0.7 : 0.4;
   519→   519→      if (pt?.neutralGroupDir === 'stable' && (pt.neutralGroupCum ?? 0) > 0.3) s = Math.max(s, 0.85);
   520→   520→      if (pt?.neutralGroupDir === 'rising' && (pt.neutralGroupCum ?? 0) > 0.25) s = Math.max(s, 0.8);
   521→   521→      return s;
   522→   522→    }
   523→   523→    case 'aggressive': {
   524→   524→      let s = ctx.confidence > 0.8 && isBull ? 0.9 : ctx.confidence > 0.7 ? 0.6 : 0.3;
   525→   525→      if (pt?.bullGroupDir === 'rising' && (pt.bullGroupChange7d ?? 0) > 0.08) s = Math.max(s, 0.85);
   526→   526→      if (pt?.bearGroupDir === 'rising' && (pt.bearGroupChange7d ?? 0) > 0.08) s = Math.max(s, 0.8);
   527→   527→      return s;
   528→   528→    }
   529→   529→    case 'balanced': {
   530→   530→      let s = ctx.confidence >= 0.6 && ctx.confidence <= 0.8 ? 0.9 : 0.5;
   531→   531→      if (pt?.bullGroupDir === 'stable' && pt.bearGroupDir === 'stable') s = Math.max(s, 0.85);
   532→   532→      return s;
   533→   533→    }
   534→   534→    case 'warning': {
   535→   535→      let s = ctx.divergence_present ? 0.9 : isBear && ctx.volatility > 0.03 ? 0.8 : 0.3;
   536→   536→      if (pt?.bearGroupDir === 'rising' && (pt.bearGroupChange7d ?? 0) > 0.05) s = Math.max(s, 0.85);
   537→   537→      if (pt?.bullGroupDir === 'falling' && (pt.bullGroupChange7d ?? 0) < -0.05) s = Math.max(s, 0.8);
   538→   538→      return s;
   539→   539→    }
   540→   540→    case 'optimistic': {
   541→   541→      let s = isBull && ctx.confidence > 0.7 ? 0.9 : isBull ? 0.5 : 0.2;
   542→   542→      if (pt?.bullGroupDir === 'rising' && (pt.bullGroupChange7d ?? 0) > 0.05) s = Math.max(s, 0.85);
   543→   543→      if (pt?.bearGroupDir === 'falling' && (pt.bearGroupChange7d ?? 0) < -0.03) s = Math.max(s, 0.7);
   544→   544→      return s;
   545→   545→    }
   546→   546→    case 'realistic': {
   547→   547→      let s = ctx.confidence >= 0.4 && ctx.confidence <= 0.7 ? 0.9 : 0.5;
   548→   548→      if (pt?.bullGroupDir === 'volatile' || pt?.bearGroupDir === 'volatile') s = Math.max(s, 0.9);
   549→   549→      return s;
   550→   550→    }
   551→   551→    default: return 0.5;
   552→   552→  }
   553→   553→}
   554→   554→
   555→   555→function calcIntensity(toneId: ToneId, ctx: MSLV4Context): string {
   556→   556→  const base = ctx.confidence + ctx.volatility;
   557→   557→  if (toneId === 'warning') return base > 0.9 ? 'high' : base > 0.6 ? 'medium' : 'low';
   558→   558→  if (toneId === 'aggressive' || toneId === 'optimistic') return base > 0.85 ? 'high' : base > 0.55 ? 'medium' : 'low';
   559→   559→  if (toneId === 'conservative') return ctx.volatility > 0.03 ? 'high' : ctx.volatility > 0.015 ? 'medium' : 'low';
   560→   560→  return base > 0.8 ? 'medium-high' : base > 0.5 ? 'medium' : 'low';
   561→   561→}

   563→   563→function buildToneReasoning(t: ToneDef, ctx: MSLV4Context): string[] {
   564→   564→  const r: string[] = [];
   565→   565→  r.push(`اطمینان کلی = ${toPersianNum(ctx.confidence)}`);
   566→   566→  r.push(`رژیم بازار = ${ctx.regime}`);
   567→   567→  r.push(`نوسان = ${ctx.volatility > 0.03 ? 'بالا' : ctx.volatility > 0.01 ? 'متوسط' : 'پایین'} (ATR/Price = ${toPersianNum(ctx.volatility)})`);
   568→   568→  if (ctx.divergence_present) r.push('واگرایی تشخیص داده شده');
   569→   569→  return r;
   570→   570→}
