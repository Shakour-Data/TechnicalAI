    17→function sma(closes: number[], period: number): number {
    18→  if (closes.length < period) return 0;
    19→  const slice = closes.slice(-period);
    20→  return slice.reduce((a, b) => a + b, 0) / period;
    21→}
    22→
    23→function emaCalc(closes: number[], period: number): number {
    24→  if (closes.length < period) return 0;
    25→  const k = 2 / (period + 1);
    26→  let v = sma(closes.slice(0, period), period);
    27→  for (let i = period; i < closes.length; i++) {
    28→    v = closes[i] * k + v * (1 - k);
    29→  }
    30→  return v;
    31→}

    33→function calcRSI(closes: number[], period = 14): number {
    34→  if (closes.length < period + 1) return 50;
    35→  let avgGain = 0, avgLoss = 0;
    36→  for (let i = 1; i <= period; i++) {
    37→    const diff = closes[i] - closes[i - 1];
    38→    if (diff > 0) avgGain += diff; else avgLoss += Math.abs(diff);
    39→  }
    40→  avgGain /= period; avgLoss /= period;
    41→  for (let i = period + 1; i < closes.length; i++) {
    42→    const diff = closes[i] - closes[i - 1];
    43→    avgGain = (avgGain * (period - 1) + (diff > 0 ? diff : 0)) / period;
    44→    avgLoss = (avgLoss * (period - 1) + (diff < 0 ? Math.abs(diff) : 0)) / period;
    45→  }
    46→  if (avgLoss === 0) return 100;
    47→  return 100 - 100 / (1 + avgGain / avgLoss);
    48→}

    50→function calcCCI(data: OHLCV[], period = 20): number {
    51→  if (data.length < period) return 0;
    52→  const tps: number[] = [];
    53→  for (let i = data.length - period; i < data.length; i++) {
    54→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
    55→  }
    56→  const mean = tps.reduce((a, b) => a + b, 0) / tps.length;
    57→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - mean), 0) / tps.length;
    58→  return meanDev === 0 ? 0 : (tps[tps.length - 1] - mean) / (0.015 * meanDev);
    59→}
    60→
    61→function calcMFI(data: OHLCV[], period = 14): number {
    62→  if (data.length < period + 1) return 50;
    63→  let posFlow = 0, negFlow = 0;
    64→  for (let i = data.length - period; i < data.length; i++) {
    65→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
    66→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
    67→    const mf = tp * data[i].volume;
    68→    if (tp > prevTp) posFlow += mf; else negFlow += mf;
    69→  }
    70→  return negFlow === 0 ? 100 : 100 - 100 / (1 + posFlow / negFlow);
    71→}

    73→function calcStochastic(data: OHLCV[], kPeriod = 14, smoothK = 3, smoothD = 3): { k: number; d: number } {
    74→  const rawKs: number[] = [];
    75→  for (let i = kPeriod - 1; i < data.length; i++) {
    76→    let lowest = Infinity, highest = -Infinity;
    77→    for (let j = i - kPeriod + 1; j <= i; j++) {
    78→      if (data[j].low < lowest) lowest = data[j].low;
    79→      if (data[j].high > highest) highest = data[j].high;
    80→    }
    81→    const range = highest - lowest;
    82→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
    83→  }
    84→  const smoothedK: number[] = [];
    85→  for (let i = smoothK - 1; i < rawKs.length; i++) {
    86→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
    87→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
    88→  }
    89→  const k = smoothedK[smoothedK.length - 1] ?? 50;
    90→  const dSlice = smoothedK.slice(-smoothD);
    91→  const d = dSlice.length > 0 ? dSlice.reduce((a, b) => a + b, 0) / dSlice.length : 50;
    92→  return { k, d };
    93→}

    95→function calcMACD(closes: number[], fast = 12, slow = 26, sig = 9): { line: number; signal: number; histogram: number } {
    96→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
    97→  const kFast = 2 / (fast + 1), kSlow = 2 / (slow + 1), kSig = 2 / (sig + 1);
    98→  let eFast = sma(closes.slice(0, fast), fast);
    99→  let eSlow = sma(closes.slice(0, slow), slow);
   100→  const macdVals: number[] = [];
   101→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   102→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   103→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   104→    macdVals.push(eFast - eSlow);
   105→  }
   106→  if (macdVals.length < sig) return { line: 0, signal: 0, histogram: 0 };
   107→  let s = sma(macdVals.slice(0, sig), sig);
   108→  for (let i = sig; i < macdVals.length; i++) {
   109→    s = macdVals[i] * kSig + s * (1 - kSig);
   110→  }
   111→  const line = macdVals[macdVals.length - 1];
   112→  return { line, signal: s, histogram: line - s };
   113→}

   115→function calcBollingerBands(closes: number[], period = 20, mult = 2): { upper: number; middle: number; lower: number } {
   116→  if (closes.length < period) return { upper: 0, middle: 0, lower: 0 };
   117→  const slice = closes.slice(-period);
   118→  const mid = slice.reduce((a, b) => a + b, 0) / period;
   119→  const variance = slice.reduce((a, b) => a + (b - mid) ** 2, 0) / period;
   120→  const sd = Math.sqrt(variance);
   121→  return { upper: mid + mult * sd, middle: mid, lower: mid - mult * sd };
   122→}

   124→function calcADX(data: OHLCV[], period = 14): { adx: number; diPlus: number; diMinus: number } {
   125→  if (data.length < period * 2) return { adx: 0, diPlus: 0, diMinus: 0 };
   126→  let sumPDM = 0, sumMDM = 0, sumTR = 0;
   127→  for (let i = data.length - period; i < data.length; i++) {
   128→    const upMove = data[i].high - data[i - 1].high;
   129→    const downMove = data[i - 1].low - data[i].low;
   130→    sumPDM += upMove > downMove && upMove > 0 ? upMove : 0;
   131→    sumMDM += downMove > upMove && downMove > 0 ? downMove : 0;
   132→    sumTR += Math.max(
   133→      data[i].high - data[i].low,
   134→      Math.abs(data[i].high - data[i - 1].close),
   135→      Math.abs(data[i].low - data[i - 1].close)
   136→    );
   137→  }
   138→  const diPlus = sumTR > 0 ? (sumPDM / sumTR) * 100 : 0;
   139→  const diMinus = sumTR > 0 ? (sumMDM / sumTR) * 100 : 0;
   140→  const dx = (diPlus + diMinus) > 0 ? Math.abs(diPlus - diMinus) / (diPlus + diMinus) * 100 : 0;
   141→  return { adx: dx, diPlus, diMinus };
   142→}
   143→
   144→function calcATR(data: OHLCV[], period = 14): number {
   145→  if (data.length < period + 1) return 0;
   146→  let sum = 0;
   147→  for (let i = data.length - period; i < data.length; i++) {
   148→    sum += Math.max(
   149→      data[i].high - data[i].low,
   150→      Math.abs(data[i].high - data[i - 1].close),
   151→      Math.abs(data[i].low - data[i - 1].close)
   152→    );
   153→  }
   154→  return sum / period;
   155→}

   157→function calcSAR(data: OHLCV[]): number {
   158→  if (data.length < 5) return 0;
   159→  // Simplified Parabolic SAR
   160→  return data[data.length - 1].low; // placeholder
   161→}

   163→// ─── 16-Feature Extraction for VDss ─────────────────────────────────────────

   165→const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));
   166→
   167→/**
   168→ * Extract the 16 VDSS features at a given point in the data.
   169→ * Feature order must match VDSS_FEATURE_NAMES exactly.
   170→ *
   171→ * @param data - Full OHLCV history (chronological)
   172→ * @param endIdx - Index of the candle to compute features for
   173→ * @param hasVolume - Whether data has valid volume
   174→ * @returns Array of 16 feature values in [0,1] range
   175→ */
   176→export function extractVDSSFeatures(data: OHLCV[], endIdx: number, hasVolume: boolean): number[] {
   177→  const slice = data.slice(0, endIdx + 1);
   178→  const closes = slice.map(d => d.close);
   179→  const price = closes[closes.length - 1];
   180→  if (price <= 0) return new Array(16).fill(0.5);
   181→
   182→  // ── Compute indicators ──
   183→  const rsi = calcRSI(closes);
   184→  const mfi = hasVolume ? calcMFI(slice) : 50;
   185→  const cci = calcCCI(slice);
   186→  const adxResult = calcADX(slice);
   187→  const macd = calcMACD(closes);
   188→  const stoch = calcStochastic(slice);
   189→  const bb = calcBollingerBands(closes);
   190→  const atr = calcATR(slice);
   191→  const ma21 = sma(closes, 21);
   192→  const ma100 = sma(closes, 100);
   193→  const ema12 = emaCalc(closes, 12);
   194→  const ema26 = emaCalc(closes, 26);
   195→
   196→  // ── LAYER 1: Raw scores (s_*) ──
   197→  const s_rsi = rsi > 80 ? 0.95 : rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   198→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   199→  const s_mfi = mfi > 80 ? 0.95 : mfi > 70 ? 0.85 : mfi > 60 ? 0.7 : mfi > 50 ? 0.55
   200→    : mfi > 40 ? 0.45 : mfi > 30 ? 0.3 : mfi > 20 ? 0.15 : 0.05;
   201→  const s_cci = cci > 200 ? 0.9 : cci > 100 ? 0.75 : cci > 0 ? 0.6
   202→    : cci > -100 ? 0.4 : cci > -200 ? 0.25 : 0.1;
   203→  const s_adx = adxResult.diPlus > adxResult.diMinus
   204→    ? clamp(0.5 + (adxResult.diPlus - adxResult.diMinus) / 100, 0.5, 1)
   205→    : clamp(0.5 - (adxResult.diMinus - adxResult.diPlus) / 100, 0, 0.5);
   206→  const s_macd = macd.histogram > 0 && macd.line > macd.signal ? 0.9
   207→    : macd.histogram > 0 ? 0.7 : macd.line > macd.signal ? 0.55
   208→    : macd.histogram < 0 && macd.line < macd.signal ? 0.1 : 0.3;
   209→  const s_stoch = clamp(
   210→    (stoch.k > 80 ? 0.9 : stoch.k > 70 ? 0.8 : stoch.k > 50 ? 0.6
   211→      : stoch.k > 30 ? 0.4 : stoch.k > 20 ? 0.2 : 0.1)
   212→    + (stoch.k > stoch.d ? 0.1 : -0.1), 0, 1);
   213→  const bbRange = bb.upper - bb.lower;
   214→  const s_bb = bbRange === 0 ? 0.5 : clamp((price - bb.lower) / bbRange, 0, 1);
   215→  const s_ma21 = clamp(price > ma21 ? 0.7 + ((price - ma21) / ma21) * 0.5 : 0.3 + ((price - ma21) / ma21) * 0.5, 0, 1);
   216→  const s_ma100 = clamp(price > ma100 ? 0.65 + ((price - ma100) / ma100) * 0.3 : 0.35 + ((price - ma100) / ma100) * 0.3, 0, 1);
   217→  const emaGap = ema12 > 0 && ema26 > 0 ? (ema12 - ema26) / ema26 : 0;
   218→  const s_ema = clamp(0.5 + emaGap * 5, 0, 1);
   219→  const s_atr = clamp(1 - atr / price * 10, 0.2, 1);
   220→
   221→  // Trend (R² + angle)
   222→  const trendPeriod = Math.min(21, closes.length);
   223→  const trendSlice = closes.slice(-trendPeriod);
   224→  const n = trendSlice.length;
   225→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   226→  for (let i = 0; i < n; i++) { sumX += i; sumY += trendSlice[i]; sumXY += i * trendSlice[i]; sumX2 += i * i; sumY2 += trendSlice[i] ** 2; }
   227→  const denom = n * sumX2 - sumX * sumX;
   228→  let r2 = 0, angle = 0;
   229→  if (denom !== 0) {
   230→    const slope = (n * sumXY - sumX * sumY) / denom;
   231→    const meanY = sumY / n;
   232→    let ssRes = 0, ssTot = 0;
   233→    for (let i = 0; i < n; i++) {
   234→      ssRes += (trendSlice[i] - (slope * i + meanY - slope * (n - 1) / 2)) ** 2;
   235→      ssTot += (trendSlice[i] - meanY) ** 2;
   236→    }
   237→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
   238→    // Convert slope/meanY to percentage before atan (same fix as ta-engine.ts)
   239→    // so that 1% daily rise → 45°, not ~0.57°
   240→    angle = meanY > 0 ? Math.atan((slope / meanY) * 100) * (180 / Math.PI) : 0;
   241→  }
   242→  const tStr = r2 * Math.abs(angle) / 45;
   243→  const direction = denom !== 0 ? (n * sumXY - sumX * sumY) / denom : 0;
   244→  const s_trend = direction > 0 ? clamp(0.5 + tStr / 2, 0.5, 1)
   245→    : direction < 0 ? clamp(0.5 - tStr / 2, 0, 0.5) : 0.5;
   246→
   247→  // Simplified S/R: distance to nearest local high/low
   248→  let s_sr = 0.5;
   249→  if (slice.length > 20) {
   250→    const lookback = slice.slice(-30);
   251→    let nearestHigh = Infinity, nearestLow = 0;
   252→    for (const d of lookback) {
   253→      if (d.high > price && d.high < nearestHigh) nearestHigh = d.high;
   254→      if (d.low < price && d.low > nearestLow) nearestLow = d.low;
   255→    }
   256→    const dR = nearestHigh < Infinity ? Math.abs(price - nearestHigh) / nearestHigh : 1;
   257→    const dS = nearestLow > 0 ? Math.abs(price - nearestLow) / nearestLow : 1;
   258→    s_sr = clamp(0.5 - dR * 1.5 + dS * 1.0, 0, 1);
   259→  }
   260→
   261→  // ── LAYER 2: Momentum correction ──
   262→  const TB = 5;
   263→  let f_rsi = s_rsi, f_mfi = s_mfi, f_cci = s_cci, f_macd = s_macd, f_stoch = s_stoch;
   264→  if (slice.length > TB + 15) {
   265→    const pastSlice = slice.slice(0, -TB);
   266→    const pastCloses = pastSlice.map(d => d.close);
   267→    const rsiPast = calcRSI(pastCloses);
   268→    const mfiPast = hasVolume ? calcMFI(pastSlice) : 50;
   269→    const cciPast = calcCCI(pastSlice);
   270→    const macdPast = calcMACD(pastCloses);
   271→    const stochPast = calcStochastic(pastSlice);
   272→
   273→    f_rsi = clamp(s_rsi + (rsi - rsiPast) / 100 * 0.08, 0, 1);
   274→    f_mfi = clamp(s_mfi + (mfi - mfiPast) / 100 * 0.08, 0, 1);
   275→    f_cci = clamp(s_cci + (cci - cciPast) / 300 * 0.06, 0, 1);
   276→    f_macd = clamp(s_macd + (macd.histogram - macdPast.histogram) / (Math.abs(macd.histogram) + Math.abs(macdPast.histogram) + 1) * 3 * 0.08, 0, 1);
   277→    f_stoch = clamp(s_stoch + (stoch.k - stochPast.k) / 50 * 0.06, 0, 1);
   278→  }
   279→
   280→  // ── LAYER 2: Cross signals (binary per spec) ──
   281→  let f_stochCross = 0.1;
   282→  let f_macdCross = 0.1;
   283→  let f_div = 0.15;
   284→
   285→  if (slice.length > 20) {
   286→    const prevSlice = slice.slice(0, -1);
   287→    const prevStoch = calcStochastic(prevSlice);
   288→    const prevMacd = calcMACD(prevSlice.map(d => d.close));
   289→
   290→    // Stochastic bullish cross: K_prev < D_prev AND K_curr > D_curr
   291→    f_stochCross = (prevStoch.k < prevStoch.d && stoch.k > stoch.d) ? 0.9 : 0.1;
   292→
   293→    // MACD bullish cross: line_prev < signal_prev AND line_curr > signal_curr
   294→    f_macdCross = (prevMacd.line < prevMacd.signal && macd.line > macd.signal) ? 0.9 : 0.1;
   295→
   296→    // Divergence detection
   297→    if (closes.length > 40) {
   298→      const p20 = closes[closes.length - 21];
   299→      const rsi20 = calcRSI(closes.slice(0, -20));
   300→      const pChange = price - p20;
   301→      const rsiChange = rsi - rsi20;
   302→      if (pChange !== 0 && rsiChange !== 0) {
   303→        f_div = (pChange > 0 && rsiChange < 0) ? 0.85 : (pChange < 0 && rsiChange > 0) ? 0.85 : 0.15;
   304→      }
   305→    }
   306→  }
   307→
   308→  // ── Return 16 features in exact order ──
   309→  return [
   310→    f_rsi, f_mfi, f_cci, s_adx,
   311→    f_macd, f_stoch, s_bb, s_ma21, s_ma100, s_ema,
   312→    s_atr, s_trend, s_sr,
   313→    f_stochCross, f_macdCross, f_div,
   314→  ];
   315→}

   319→interface CachedModel {
   320→  result: AdaptiveModelResult;
   321→  time: number;
   322→}
   323→
   324→const modelCache = new Map<string, CachedModel>();
   325→const CACHE_TTL = 60 * 60 * 1000; // 1 hour
   326→
   327→export function getCachedAdaptiveModel(symbol: string): AdaptiveModelResult | null {
   328→  const cached = modelCache.get(symbol);
   329→  if (cached && Date.now() - cached.time < CACHE_TTL) return cached.result;
   330→  return null;
   331→}

   333→export function setCachedAdaptiveModel(symbol: string, result: AdaptiveModelResult): void {
   334→  modelCache.set(symbol, { result, time: Date.now() });
   335→}

   337→// ─── Main Training Function ─────────────────────────────────────────────────
   338→
   339→/**
   340→ * Train the Adaptive Weight Model for a given symbol.
   341→ * Extracts 16 VDSS features for each historical candle, creates binary labels,
   342→ * trains Logistic Regression with TimeSeriesSplit validation.
   343→ *
   344→ * @param data - Full OHLCV history (chronological, oldest first)
   345→ * @param symbol - Symbol identifier for caching
   346→ * @param minSamples - Minimum samples needed (default 70)
   347→ * @returns AdaptiveModelResult or null
   348→ */
   349→export function trainAdaptiveModel(data: OHLCV[], symbol: string, minSamples = 70): AdaptiveModelResult | null {
   350→  const FORWARD_DAYS = 5;
   351→  const START_IDX = 40; // Need enough history for all indicators to stabilize
   352→
   353→  if (data.length < START_IDX + FORWARD_DAYS + minSamples) return null;
   354→
   355→  const hasVolume = data.some(d => d.volume > 0);
   356→
   357→  const X: number[][] = [];
   358→  const y: number[] = [];
   359→
   360→  for (let i = START_IDX; i < data.length - FORWARD_DAYS; i++) {
   361→    const features = extractVDSSFeatures(data, i, hasVolume);
   362→    const currentPrice = data[i].close;
   363→    const futurePrice = data[i + FORWARD_DAYS].close;
   364→
   365→    if (currentPrice <= 0) continue;
   366→    if (features.some(f => !isFinite(f))) continue;
   367→
   368→    // Label: 1 if 5-day forward return > 1%, else 0
   369→    const forwardReturn = (futurePrice - currentPrice) / currentPrice;
   370→    const label = forwardReturn > 0.01 ? 1 : 0;
   371→
   372→    X.push(features);
   373→    y.push(label);
   374→  }
   375→
   376→  if (X.length < minSamples) return null;
   377→
   378→  const model = new AdaptiveWeightModel(minSamples);
   379→  const success = model.train(X, y);
   380→
   381→  if (!success) return null;
   382→
   383→  // Predict current features
   384→  const currentFeatures = extractVDSSFeatures(data, data.length - 1, hasVolume);
   385→  model.predictScore(currentFeatures);
   386→
   387→  const result = model.getResult();
   388→  setCachedAdaptiveModel(symbol, result);
   389→
   390→  return result;
   391→}

   393→// ─── Calculate bullConsensus with ML Weights (Layer 3) ──────────────────────
   394→
   395→/**
   396→ * Calculate bullConsensus using ML weights.
   397→ * Falls back to default weights if ML model is not trained.
   398→ *
   399→ * @param features - 16 VDSS features for the current candle
   400→ * @param mlResult - Adaptive model result (or null for fallback)
   401→ * @param hasVolume - Whether MFI feature is valid
   402→ */
   403→export function calculateBullConsensus(
   404→  features: number[],
   405→  mlResult: AdaptiveModelResult | null,
   406→  hasVolume: boolean,
   407→): { bullConsensus: number; usedML: boolean } {
   408→  if (mlResult && mlResult.isTrained && mlResult.weights.length === 16) {
   409→    const weights = mlResult.weights;
   410→    let bullConsensus = 0;
   411→    let totalWeight = 0;
   412→
   413→    for (let j = 0; j < 16; j++) {
   414→      // Skip MFI (index 1) if no volume
   415→      if (j === 1 && !hasVolume) continue;
   416→      bullConsensus += weights[j] * features[j];
   417→      totalWeight += weights[j];
   418→    }
   419→
   420→    if (totalWeight > 0) bullConsensus /= totalWeight;
   421→
   422→    // Blend with ML direct prediction (dynamic weight based on accuracy)
   423→    if (mlResult.predictionProb !== null) {
   424→      const mlWeight = Math.max(0, Math.min(mlResult.recentAccuracy * 0.30, 0.30));
   425→      bullConsensus = bullConsensus * (1 - mlWeight) + mlResult.predictionProb * mlWeight;
   426→    }
   427→
   428→    return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: true };
   429→  }
   430→
   431→  // Fallback: default weights (per spec, only when ML is unavailable)
   432→  const DEFAULT_WEIGHTS = [
   433→    0.15, 0.12, 0.10, 0.08,  // rsi, mfi, cci, adx
   434→    0.14, 0.10, 0.08,         // macd, stoch, bb
   435→    0.10, 0.08, 0.10,         // ma21, ma100, ema
   436→    0.05, 0.10, 0.05,         // atr, trend, sr
   437→    0.06, 0.06, 0.08,         // stochCross, macdCross, div
   438→  ];
   439→
   440→  let bullConsensus = 0;
   441→  let totalWeight = 0;
   442→  for (let j = 0; j < 16; j++) {
   443→    if (j === 1 && !hasVolume) continue;
   444→    bullConsensus += DEFAULT_WEIGHTS[j] * features[j];
   445→    totalWeight += DEFAULT_WEIGHTS[j];
   446→  }
   447→  if (totalWeight > 0) bullConsensus /= totalWeight;
   448→
   449→  return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: false };
   450→}

   512→// ─── Decision Graph Scenario Probabilities (3-Branch DAG) ──────────────────
   513→// Implements the exact structure from TechnicalAnalysisDssGraph.txt:
   514→//   Root → 3 branches (Trend Following A, Breakout B, Reversal C)
   515→//   Each branch → 9 edges → 9 leaf scenarios
   516→//   P(Scenario_S) = Σ_branch P(Branch) × P(Edge_S | Branch)
   517→//   Where P(Branch) is derived from EMV maximization,
   518→//   and P(Edge) is the conditional probability per edge formula.
   519→// ═══════════════════════════════════════════════════════════════════════════
   520→
   521→export interface DecisionGraphContext {
   522→  bullConsensus: number;
   523→  price: number;
   524→  R1_level: number;
   525→  S1_level: number;
   526→  MA100: number;
   527→  rsi: number;

   533→  macdSignal: number;
   534→  adx: number;
   535→  diPlus: number;
   536→  diMinus: number;
   537→  atr: number;
   538→  bbUpper: number;
   539→  bbLower: number;
   540→  bbMiddle: number;
   541→  hasVolume: boolean;
   542→  volume: number;
   543→  avgVolume20: number;
   544→  mlResult: AdaptiveModelResult | null;
   545→  mlPrediction: MLPredictionResult | null;
   546→  mlRegime: MLRegimeResult | null;
   547→}
   548→
   549→/**
   550→ * Calculate 9-scenario probabilities using the 3-branch decision graph.
   551→ * Each branch (Trend Following, Breakout, Reversal) emits conditional probabilities
   552→ * for each scenario. Final probability is the EMV-weighted combination.
   553→ *
   554→ * Per TechnicalAnalysisDssGraph.txt:
   555→ *   P_Bull_Final = α × P_Bull_Current + (1-α) × P_Bull_Forecast
   556→ *   P_Bear_Final = α × P_Bear_Current + (1-α) × P_Bear_Forecast
   557→ *   P_Neutral_Final = 1 - (P_Bull_Final + P_Bear_Final)
   558→ *
   559→ *   P(Scenario_S) = P_Bull_Final × CP_S|Bull + P_Bear_Final × CP_S|Bear + P_Neutral_Final × CP_S|Neutral
   560→ */
   561→export function calculateScenarioProbabilities(
   562→  bullConsensus: number,
   563→  price: number,
   564→  SC1: number,
   565→  S1: number,
   566→  MA100: number,
   567→  rsi: number,
   568→  mfi: number,
   569→  stochK: number,
   570→  hasVolume: boolean,
   571→  mlResult: AdaptiveModelResult | null,
   572→  adx: number = 25,
   573→  atr: number = 0,
   574→): { pSC1: number; pSC2: number; pSC3: number; pSC4: number; pSC5: number; pSC6: number; pSC7: number; pSC8: number; pSC9: number; factors: { momentum: number; volatility: number; trend: number } } {
   575→  // ── Adaptive ML factors ──
   576→  const mlParams = mlResult?.adaptiveParams;
   577→  const factors = {
   578→    momentum: mlParams?.momentumFactor ?? 0.7,
   579→    volatility: mlParams?.volatilityFactor ?? 0.5,
   580→    trend: mlParams?.trendFactor ?? 0.6,
   581→  };
   582→
   583→  const bull = bullConsensus;
   584→  const bear = 1 - bullConsensus;
   585→
   586→  // ═══════════════════════════════════════════════════════════════════════
   587→  // STEP 1: Compute P_Bull_Current, P_Bear_Current from current indicators
   588→  // ═══════════════════════════════════════════════════════════════════════
   589→  // Use Bayesian weighting: W_i = P(Success|Tool_i) × Prior_i / Σ
   590→  // For current signals, each indicator provides a directional score.
   591→
   592→  // RSI signal (0=bearish, 1=bullish)
   593→  const rsiBull = rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   594→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   595→
   596→  // MFI signal (volume-dependent)
   597→  const mfiBull = hasVolume
   598→    ? (mfi > 80 ? 0.85 : mfi > 60 ? 0.65 : mfi > 40 ? 0.45 : mfi > 20 ? 0.3 : 0.15)
   599→    : 0.5; // neutral if no volume
   600→
   601→  // Stochastic signal
   602→  const stochBull = stochK > 80 ? 0.85 : stochK > 60 ? 0.65 : stochK > 40 ? 0.45
   603→    : stochK > 20 ? 0.3 : 0.15;
   604→
   605→  // ADX trend strength normalized (0-1)
   606→  const adxNorm = Math.min(adx / 45, 1);
   607→
   608→  // Price vs MA100 position
   609→  const maPosBull = MA100 > 0 ? clamp((price - MA100) / (MA100 * 0.05), 0, 1) : 0.5;
   610→
   611→  // Distance to S/R (proximity = more likely to react)
   612→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0.5;
   613→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0.5;
   614→
   615→  // Priors per category (from instructions): trend=0.28, oscillator=0.25, volume=0.15, volatility=0.10, leading=0.12, patterns=0.10
   616→  // Adjusted: if no volume, redistribute 0.15 proportionally
   617→  let priorTrend = 0.28, priorOsc = 0.25, priorVol = hasVolume ? 0.15 : 0;
   618→  let priorVolatility = 0.10, priorLeading = 0.12, priorPattern = 0.10;
   619→  if (!hasVolume) {
   620→    const totalRemaining = priorTrend + priorOsc + priorVolatility + priorLeading + priorPattern;
   621→    const redistribute = 0.15;
   622→    priorTrend += redistribute * (priorTrend / totalRemaining);
   623→    priorOsc += redistribute * (priorOsc / totalRemaining);
   624→    priorVolatility += redistribute * (priorVolatility / totalRemaining);
   625→    priorLeading += redistribute * (priorLeading / totalRemaining);
   626→    priorPattern += redistribute * (priorPattern / totalRemaining);
   627→  }
   628→
   629→  // P_Bull_Current = Σ(W_i × Signal_Bull_i) / Σ(W_i)
   630→  const currentBullSignals = [
   631→    { signal: rsiBull, weight: priorOsc * 0.4 },           // RSI (oscillator)
   632→    { signal: mfiBull, weight: priorVol },                 // MFI (volume)
   633→    { signal: stochBull, weight: priorOsc * 0.35 },        // Stochastic (oscillator)
   634→    { signal: maPosBull, weight: priorTrend * 0.5 },       // MA position (trend)
   635→    { signal: adxNorm > 0.5 && bull > 0.5 ? bull : 0.5, weight: priorTrend * 0.3 }, // ADX trend (trend)
   636→    { signal: distS1, weight: priorPattern * 0.5 },        // Near support = bullish bounce
   637→  ];
   638→
   639→  let sumWeight = 0;
   640→  let sumWeightedBull = 0;
   641→  for (const { signal, weight } of currentBullSignals) {
   642→    if (weight <= 0) continue;
   643→    sumWeightedBull += weight * signal;
   644→    sumWeight += weight;
   645→  }
   646→  const P_Bull_Current = sumWeight > 0 ? clamp(sumWeightedBull / sumWeight, 0, 1) : bull;
   647→  const P_Bear_Current = 1 - P_Bull_Current;
   648→
   649→  // ═══════════════════════════════════════════════════════════════════════
   650→  // STEP 2: P_Bull_Forecast (from ML prediction, or decayed current)
   651→  // ═══════════════════════════════════════════════════════════════════════
   652→  // If ML prediction available, use predicted return to estimate future bull prob
   653→  // Otherwise use exponential decay of current state
   654→  const alpha = 0.5; // weight of current state (per instructions)
   655→
   656→  // Use ML predicted return if available, otherwise use consensus as proxy
   657→  const mlReturn = mlResult?.predictionProb;
   658→  let P_Bull_Forecast: number;
   659→  if (mlReturn !== null && mlReturn !== undefined) {
   660→    // ML gives direct P(bull) prediction — use it as forecast
   661→    P_Bull_Forecast = clamp(mlReturn, 0, 1);
   662→  } else {
   663→    // Decay: forecast = current × decay + 0.5 × (1-decay)  (mean-revert to 0.5)
   664→    const decayFactor = 0.85;
   665→    P_Bull_Forecast = P_Bull_Current * decayFactor + 0.5 * (1 - decayFactor);
   666→  }
   667→  const P_Bear_Forecast = 1 - P_Bull_Forecast;
   668→
   669→  // ═══════════════════════════════════════════════════════════════════════
   670→  // STEP 3: Final directional probabilities (combining current + forecast)
   671→  // ═══════════════════════════════════════════════════════════════════════
   672→  const P_Bull_Final = clamp(alpha * P_Bull_Current + (1 - alpha) * P_Bull_Forecast, 0.02, 0.98);
   673→  const P_Bear_Final = clamp(alpha * P_Bear_Current + (1 - alpha) * P_Bear_Forecast, 0.02, 0.98);
   674→  const P_Neutral_Final = Math.max(0.04, 1 - P_Bull_Final - P_Bear_Final);
   675→
   676→  // Normalize so they sum to 1
   677→  const dirSum = P_Bull_Final + P_Bear_Final + P_Neutral_Final;
   678→  const pBull = P_Bull_Final / dirSum;
   679→  const pBear = P_Bear_Final / dirSum;
   680→  const pNeutral = P_Neutral_Final / dirSum;
   681→
   682→  // ═══════════════════════════════════════════════════════════════════════
   683→
   684→  // Helper: Overbought/Oversold
   685→  const overbought = rsi > 70 ? (rsi - 70) / 30 : 0;
   686→  const oversold = rsi < 30 ? (30 - rsi) / 30 : 0;
   687→
   688→  // Helper: Volatility state
   689→  const atrPct = price > 0 ? (atr || price * 0.02) / price : 0.02;
   690→  const volHigh = clamp((atrPct - 0.02) / 0.03, 0, 1);
   691→  const volLow = clamp(1 - (atrPct - 0.01) / 0.02, 0, 1);
   692→
   693→  // Helper: Divergence detection
   694→  let divBull = 0;
   695→  let divBear = 0;
   696→  if (mlResult?.predictionProb !== undefined) {
   697→    const mlBull = mlResult.predictionProb ?? 0.5;
   698→    if (bull > 0.6 && mlBull < 0.4) divBear = (bull - mlBull) * 1.5;
   699→    if (bull < 0.4 && mlBull > 0.6) divBull = (mlBull - bull) * 1.5;
   700→  }
   701→  divBull = clamp(divBull, 0, 0.3);
   702→  divBear = clamp(divBear, 0, 0.3);
   703→
   704→  // Helper: Volume confirmation
   705→  const volConfirm = hasVolume ? 0.6 : 0.4;
   706→
   707→  // Helper: ADX > 40 = strong trend
   708→  const adxStrong = adx > 40 ? 1 : adx > 25 ? (adx - 25) / 15 : 0;
   709→
   710→  // ── Conditional probability distributions CP(S_i | State) ──
   711→  // Per attachment instructions: 3-branch decision graph
   712→  //   Branch A: Trend Following → scenarios A1-A9
   713→  //   Branch B: Breakout Strategy → scenarios B1-B9
   714→  //   Branch C: Reversal Strategy → scenarios C1-C9
   715→  // Final: P(S_i) = P(Bull)×CP_i|Bull + P(Bear)×CP_i|Bear + P(Neutral)×CP_i|Neutral
   716→  // Each CP distribution sums to 1 after normalization.
   717→  //
   718→  // ADX governs range/trend split (per attachment: ADX < 20 = weak trend)
   719→  // FIX: Range scenarios capped so they never dominate (>40% of any CP dist)
   720→
   721→  // ADX-based trend strength (per attachment: ADX_Normalized = min(ADX/60, 1))
   722→  const adxNormVal = Math.min(adx / 60, 1);
   723→  // Range weight: inversely proportional to ADX, but capped at 0.35 (was 0.6!)
   724→  const adxRange = adx < 20 ? 0.35 : adx < 30 ? 0.20 : 0.10;
   725→  const adxTrend = 1 - adxRange;
   726→
   727→  // ── Per attachment edge formulas (lines 657-716): ──
   728→  // A1: Strong Bull  = P_Bull × P_ADX>40 × P_Forecast_Bull_Strong
   729→  // A2: Accel Bull   = P_Bull × P_Volatility_High × P_Volume_High_Future
   730→  // A3: Cautious Bull= P_Bull × P_Divergence_Bearish_Future
   731→  // A4: Strong Bear  = P_Bear × P_ADX>40 × P_Forecast_Bear_Strong
   732→  // A5: Accel Bear   = P_Bear × P_Volatility_High × P_Volume_High_Future
   733→  // A6: Cautious Bear= P_Bear × P_Divergence_Bullish_Future
   734→  // A7: LowVol Range = P_Neutral × P_Volatility_Low_Future
   735→  // A8: HighVol Range= P_Neutral × P_Volatility_High_Future
   736→  // A9: Shock        = P_Shock × P_Forecast_Anomaly (0.02-0.05)
   737→
   738→  // CP_i|Bull: Bull scenarios dominate, range scenarios limited
   739→  const cpBull = [
   740→    // SC1: Strong Bull — ADX strength and trend quality
   741→    0.28 * adxTrend * (0.5 + 0.5 * adxNormVal),
   742→    // SC2: Accelerating Bull — momentum + volatility
   743→    0.15 * adxTrend * (0.3 + 0.7 * factors.momentum),
   744→    // SC3: Cautious Bull — reduced by overbought risk
   745→    0.12 * (1 - overbought * 0.6) * adxTrend,
   746→    // SC4: Strong Bear — small probability in bull market, driven by divergence
   747→    0.03 * (1 + divBear),
   748→    // SC5: Accelerating Bear — very small in bull market
   749→    0.02,
   750→    // SC6: Cautious Bear — driven by divergence warning
   751→    0.03 * (1 + divBear),
   752→    // SC7: Low-Vol Range — bounded, not dominant
   753→    adxRange * 0.45 * volLow,
   754→    // SC8: High-Vol Range — bounded
   755→    adxRange * 0.30 * volHigh,
   756→    // SC9: Shock — fixed small probability
   757→    0.04 * (1 + overbought * 0.5 + volHigh * 0.5),
   758→  ];
   759→
   760→  // CP_i|Bear: Mirror of bull (bear scenarios dominate)
   761→  const cpBear = [
   762→    // SC1: Strong Bull — small in bear market
   763→    0.03 * (1 + divBull),
   764→    // SC2: Accelerating Bull — very small
   765→    0.02,
   766→    // SC3: Cautious Bull — divergence-driven
   767→    0.03 * (1 + divBull),
   768→    // SC4: Strong Bear — ADX strength
   769→    0.28 * adxTrend * (0.5 + 0.5 * adxNormVal),
   770→    // SC5: Accelerating Bear — momentum
   771→    0.15 * adxTrend * (0.3 + 0.7 * factors.momentum),
   772→    // SC6: Cautious Bear — reduced by oversold risk
   773→    0.12 * (1 - oversold * 0.6) * adxTrend,
   774→    // SC7: Low-Vol Range — bounded
   775→    adxRange * 0.45 * volLow,
   776→    // SC8: High-Vol Range — bounded
   777→    adxRange * 0.30 * volHigh,
   778→    // SC9: Shock
   779→    0.04 * (1 + oversold * 0.5 + volHigh * 0.5),
   780→  ];
   781→
   782→  // CP_i|Neutral: More balanced distribution per attachment
   783→  // Per attachment: neutral market has trend signals too, just weaker
   784→  // Range gets adxRange but trend scenarios get reasonable weight
   785→  const cpNeutral = [
   786→    // SC1: Mild Bull — smaller in neutral
   787→    0.08 * adxTrend * (0.5 + 0.5 * adxNormVal),
   788→    // SC2: Bull Acceleration
   789→    0.04 * adxTrend * (0.3 + 0.7 * factors.momentum),
   790→    // SC3: Cautious Bull
   791→    0.07 * adxTrend,
   792→    // SC4: Mild Bear
   793→    0.08 * adxTrend * (0.5 + 0.5 * adxNormVal),
   794→    // SC5: Bear Acceleration
   795→    0.04 * adxTrend * (0.3 + 0.7 * factors.momentum),
   796→    // SC6: Cautious Bear
   797→    0.07 * adxTrend,
   798→    // SC7: Low-Vol Range — adxRange weighted, split between vol states
   799→    adxRange * 0.50 * volLow,
   800→    // SC8: High-Vol Range
   801→    adxRange * 0.35 * volHigh,
   802→    // SC9: Shock — small fixed
   803→    0.04,
   804→  ];
   805→
   806→  const normCP = (cp: number[]): number[] => {
   807→    const s = cp.reduce((a, b) => a + b, 0);
   808→    return s > 0 ? cp.map(v => v / s) : cp.map(() => 1 / 9);
   809→  };
   810→  const ncpBull = normCP(cpBull);
   811→  const ncpBear = normCP(cpBear);
   812→  const ncpNeutral = normCP(cpNeutral);
   813→
   814→  // ── Final probabilities per instructions formula ──
   815→  // P(S_i) = P(Bull)*CP_i|Bull + P(Bear)*CP_i|Bear + P(Neutral)*CP_i|Neutral
   816→  const rawS = [0, 0, 0, 0, 0, 0, 0, 0, 0];
   817→  for (let i = 0; i < 9; i++) {
   818→    rawS[i] = pBull * ncpBull[i] + pBear * ncpBear[i] + pNeutral * ncpNeutral[i];
   819→  }
   820→
   821→  // Normalize to sum = 100
   822→  const totalRaw = rawS.reduce((a, b) => a + b, 0);
   823→  if (totalRaw <= 0) {
   824→    return {
   825→      pSC1: 12, pSC2: 10, pSC3: 8, pSC4: 12, pSC5: 10, pSC6: 8,
   826→      pSC7: 15, pSC8: 15, pSC9: 10, factors,
   827→    };
   828→  }
   829→
   830→  const pSC1 = Math.round(rawS[0] / totalRaw * 100);
   831→  const pSC2 = Math.round(rawS[1] / totalRaw * 100);
   832→  const pSC3 = Math.round(rawS[2] / totalRaw * 100);
   833→  const pSC4 = Math.round(rawS[3] / totalRaw * 100);
   834→  const pSC5 = Math.round(rawS[4] / totalRaw * 100);
   835→  const pSC6 = Math.round(rawS[5] / totalRaw * 100);
   836→  const pSC7 = Math.round(rawS[6] / totalRaw * 100);
   837→  const pSC8 = Math.round(rawS[7] / totalRaw * 100);
   838→  const pSC9 = Math.max(1, 100 - (pSC1 + pSC2 + pSC3 + pSC4 + pSC5 + pSC6 + pSC7 + pSC8));
   839→  return { pSC1, pSC2, pSC3, pSC4, pSC5, pSC6, pSC7, pSC8, pSC9, factors };
   840→}
   841→
   842→export function calculateEdgeWeights(
   843→  bullConsensus: number,
   844→  adx: number,
   845→  mlResult: AdaptiveModelResult | null,
   846→): { up: number; down: number; pullback: number; risk: number } {
   847→  // Default coefficients
   848→  let trendCoef = 1 / 3;
   849→  let momentumCoef = 1 / 3;
   850→  let volatilityCoef = 1 / 3;
   851→
   852→  if (mlResult?.isTrained && mlResult.coefficients.length === 16) {
   853→    const ec = mlResult.coefficients;
   854→    const absTrend = Math.abs(ec[11] ?? 0);
   855→    const absMomentum = Math.abs(ec[0] ?? 0);
   856→    const absVolatility = Math.abs(ec[10] ?? 0);
   857→    const total = absTrend + absMomentum + absVolatility;
   858→    if (total > 0) {
   859→      trendCoef = absTrend / total;
   860→      momentumCoef = absMomentum / total;
   861→      volatilityCoef = absVolatility / total;
   862→    }
   863→  }
   864→
   865→  // Dynamic base calculations
   866→  const up_base = bullConsensus * 0.7 + 0.15;
   867→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
   868→  const pullback_base = 0.25;
   869→  const risk_base = 0.12 * (1.2 - Math.min(adx / 100, 1));
   870→
   871→  // Apply ML-driven adjustments
   872→  const up = clamp(up_base * (1 + (momentumCoef - 1 / 3) * 0.3), 0.05, 0.95);
   873→  const down = clamp(down_base * (1 + (trendCoef - 1 / 3) * 0.3), 0.05, 0.95);
   874→  const pullback = clamp(pullback_base * (1 + (volatilityCoef - 1 / 3) * 0.2), 0.05, 0.50);
   875→  const risk = clamp(risk_base * (1 - (trendCoef - 1 / 3) * 0.5), 0.02, 0.30);
   876→
   877→  return { up, down, pullback, risk };
   878→}

   944→export interface PricePrediction {
   945→  predictedPrices: number[];
   946→  predictedReturn5d: number;
   947→  confidence: number;
   948→  trendDirection: 'up' | 'down' | 'flat';
   949→}
   950→
   951→export function predictPrices(data: OHLCV[]): PricePrediction | null {
   952→  if (data.length < 40) return null;
   953→  const PREDICT_DAYS = 5;
   954→  const closes = data.map(d => d.close);
   955→  const price = closes[closes.length - 1];
   956→  if (price <= 0) return null;
   957→
   958→  const LOOKBACK = 20;
   959→  const returns: number[] = [];
   960→  for (let i = closes.length - LOOKBACK; i < closes.length; i++) {
   961→    returns.push(i > 0 && closes[i - 1] > 0 ? (closes[i] - closes[i - 1]) / closes[i - 1] : 0);
   962→  }
   963→
   964→  const mom3 = returns.slice(-3).reduce((a, b) => a + b, 0) / 3;
   965→  const mom10 = returns.slice(-10).reduce((a, b) => a + b, 0) / 10;
   966→  const ma20 = sma(closes, 20);
   967→  const deviation = price > 0 && ma20 > 0 ? (price - ma20) / ma20 : 0;
   968→
   969→  const trendSlice = closes.slice(-20);
   970→  const tn = trendSlice.length;
   971→  let sx = 0, sy = 0, sxy = 0, sx2 = 0;
   972→  for (let i = 0; i < tn; i++) { sx += i; sy += trendSlice[i]; sxy += i * trendSlice[i]; sx2 += i * i; }
   973→  const d = tn * sx2 - sx * sx;
   974→  let slope = 0, r2 = 0;
   975→  if (d !== 0) {
   976→    slope = (tn * sxy - sx * sy) / d;
   977→    const meanY = sy / tn;
   978→    let ssRes = 0, ssTot = 0;
   979→    for (let i = 0; i < tn; i++) { ssRes += (trendSlice[i] - (slope * i + meanY - slope * (tn - 1) / 2)) ** 2; ssTot += (trendSlice[i] - meanY) ** 2; }
   980→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
   981→  }
   982→
   983→  let atr = 0;
   984→  for (let i = Math.max(1, data.length - 14); i < data.length; i++) {
   985→    atr += Math.max(data[i].high - data[i].low, Math.abs(data[i].high - data[i - 1].close), Math.abs(data[i].low - data[i - 1].close));
   986→  }
   987→  atr /= Math.min(14, data.length - 1);
   988→  const dailyVol = price > 0 ? atr / price : 0.02;
   989→
   990→  const trendDaily = price > 0 ? slope / price : 0;
   991→  const meanRevDaily = -deviation * 0.05;
   992→  const momentumDaily = mom3 * 0.4 + mom10 * 0.1;
   993→  const trendW = r2 * 0.6, mrW = Math.min(Math.abs(deviation) * 5, 0.4), momW = 0.2;
   994→  const predDaily = trendDaily * trendW + meanRevDaily * mrW + momentumDaily * momW;
   995→
   996→  const predictedPrices: number[] = [];
   997→  let cp = price;
   998→  for (let dd = 1; dd <= PREDICT_DAYS; dd++) {
   999→    const decay = Math.exp(-0.1 * (dd - 1));
  1000→    cp = cp * (1 + predDaily * decay);
  1001→    predictedPrices.push(Math.round(cp));
  1002→  }
  1003→
  1004→  const predictedReturn5d = (predictedPrices[PREDICT_DAYS - 1] - price) / price;
  1005→  return {
  1006→    predictedPrices,
  1007→    predictedReturn5d,
  1008→    confidence: Math.min(0.9, Math.max(0.1, r2 * 0.7 + 0.2)),
  1009→    trendDirection: predictedReturn5d > 0.005 ? 'up' : predictedReturn5d < -0.005 ? 'down' : 'flat',
  1010→  };
  1011→}

  1013→export function evaluatePredictedIndicators(data: OHLCV[], prediction: PricePrediction): { mlScore: number; predictedBullConsensus: number } {
  1014→  const price = data[data.length - 1].close;
  1015→  if (price <= 0 || prediction.predictedPrices.length === 0) return { mlScore: 0.5, predictedBullConsensus: 0.5 };
  1016→  const avgP = prediction.predictedPrices.reduce((a, b) => a + b, 0) / prediction.predictedPrices.length;
  1017→  const closes = data.map(d => d.close);
  1018→  const ma21 = sma(closes, 21), ma100 = sma(closes, 100), rsi = calcRSI(closes);
  1019→  const rsiP = Math.max(0, Math.min(1, (rsi + (avgP > price ? 5 : -5)) / 100));
  1020→  const ma21S = ma21 > 0 ? (avgP > ma21 ? 0.7 : 0.3) : 0.5;
  1021→  const ma100S = ma100 > 0 ? (avgP > ma100 ? 0.65 : 0.35) : 0.5;
  1022→  const dirS = avgP > price ? 0.65 : 0.35;
  1023→  return {
  1024→    mlScore: Math.max(0, Math.min(1, 0.5 + prediction.predictedReturn5d * 10 * prediction.confidence)),
  1025→    predictedBullConsensus: Math.max(0, Math.min(1, rsiP * 0.3 + ma21S * 0.25 + ma100S * 0.2 + dirS * 0.25)),
  1026→  };
  1027→}
