    17→    17→function safeNum(v: number, fallback: number = 0): number {
    18→    18→  return (typeof v === 'number' && isFinite(v)) ? v : fallback;
    19→    19→}
    20→    20→
    21→    21→function clamp(v: number, min: number, max: number): number {
    22→    22→  return Math.max(min, Math.min(max, safeNum(v, (min + max) / 2)));
    23→    23→}
    24→    24→
    25→    25→function normalize(arr: number[]): number[] {
    26→    26→  const safe = arr.map(v => safeNum(v, 0));
    27→    27→  const sum = safe.reduce((a, b) => a + b, 0);
    28→    28→  if (sum <= 0) return arr.map(() => 1 / arr.length);
    29→    29→  return safe.map((v) => v / sum);
    30→    30→}
    31→    31→
    32→    32→function sigmoid(x: number): number {
    33→    33→  const sx = safeNum(x, 0);
    34→    34→  if (Math.abs(sx) > 500) return sx > 0 ? 1 : 0; // prevent Infinity
    35→    35→  return 1 / (1 + Math.exp(-sx));
    36→    36→}

  1024→  1024→function buildContext(input: GraphInput): ProbContext {
  1025→  1025→  const adxNorm = clamp(input.adx / 60, 0, 1);
  1026→  1026→  const srProximity = clamp(
  1027→  1027→    1 - Math.min(input.distToR1, input.distToS1) * 10,
  1028→  1028→    0,
  1029→  1029→    1
  1030→  1030→  );
  1031→  1031→  const oscillatorExtreme = clamp(
  1032→  1032→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
  1033→  1033→    0,
  1034→  1034→    1
  1035→  1035→  );
  1036→  1036→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1037→  1037→
  1038→  1038→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1039→  1039→  const overboughtRisk =
  1040→  1040→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1041→  1041→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1042→  1042→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1043→  1043→
  1044→  1044→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1045→  1045→  const oversoldBounce =
  1046→  1046→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1047→  1047→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1048→  1048→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1049→  1049→
  1050→  1050→  return {
  1051→  1051→    input,
  1052→  1052→    adxNorm,
  1053→  1053→    srProximity,
  1054→  1054→    oscillatorExtreme,
  1055→  1055→    divergenceProxy,
  1056→  1056→    overboughtRisk,
  1057→  1057→    oversoldBounce,
  1058→  1058→  };
  1059→  1059→}

  1061→  1061→// === Node Probability Computation =============================================
  1062→  1062→// For each non-terminal node, returns an array of probabilities for its
  1063→  1063→// outgoing edges. The order MUST match the order of edges in createEdges().
  1064→  1064→// =============================================================================
  1065→  1065→
  1066→  1066→function computeNodeProbabilities(
  1067→  1067→  nodeId: string,
  1068→  1068→  ctx: ProbContext,
  1069→  1069→  _numChildren: number
  1070→  1070→): number[] {
  1071→  1071→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1072→  1072→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1073→  1073→
  1074→  1074→  switch (nodeId) {
  1075→  1075→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1076→  1076→    case 'ROOT': {
  1077→  1077→      const trendW = adxNorm * 0.5 + input.mlTrend * 0.18 + 0.1;
  1078→  1078→      const breakW = srProximity * 0.3 + (1 - adxNorm) * 0.08 + 0.08;
  1079→  1079→      const revW = ctx.divergenceProxy * 0.35 + (1 - adxNorm) * 0.12 + 0.06;
  1080→  1080→      return normalize([trendW, breakW, revW]);
  1081→  1081→    }
  1082→  1082→
  1083→  1083→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1084→  1084→    case 'N_TREND': {
  1085→  1085→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1086→  1086→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1087→  1087→      const flatUncertainty = (1 - Math.abs(input.bullConsensus - 0.5) * 2);
  1088→  1088→      const flat = flatUncertainty * clamp(1 - adxNorm * 1.5, 0, 1) * 0.45;
  1089→  1089→      return normalize([bull, bear, flat]);
  1090→  1090→    }
  1091→  1091→
  1092→  1092→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1093→  1093→    case 'N_T_BULL': {
  1094→  1094→      const momHigh =
  1095→  1095→        clamp(
  1096→  1096→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1097→  1097→            (input.macdHist > 0
  1098→  1098→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1099→  1099→              : 0),
  1100→  1100→          0,
  1101→  1101→          1
  1102→  1102→        ) * input.mlMomentum;
  1103→  1103→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1104→  1104→      return normalize([momHigh, momMod, overboughtRisk]);
  1105→  1105→    }
  1106→  1106→
  1107→  1107→    // ── N_T_B_MOM_HIGH → [SC8, SC9, SC7] ────────────────────────────────────
  1108→  1108→    // Strong bullish momentum — all three outcomes are bullish (no range)
  1109→  1109→    case 'N_T_B_MOM_HIGH': {

  1111→  1111→      const pSC9 = 0.15 + input.strengthIndex * 0.2;
  1112→  1112→      const pSC7 = 0.35 * input.confidenceIndex + 0.15;

  1114→  1114→    }
  1115→  1115→
  1116→  1116→    // ── N_T_B_MOM_MOD → [SC7, SC6, SC5] ─────────────────────────────────────
  1117→  1117→    // Moderate bullish — small explicit SC5 based on uncertainty
  1118→  1118→    case 'N_T_B_MOM_MOD': {

  1120→  1120→      const pSC6 = 0.55 * (1 - input.confidenceIndex) + 0.15;
  1121→  1121→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.02, 0.15);
  1122→  1122→      return normalize([pSC7, pSC6, pSC5]);
  1123→  1123→    }
  1124→  1124→
  1125→  1125→    // ── N_T_B_OVERBOUGHT → [SC4, SC1, SC9] ──────────────────────────────────

  1130→  1130→      const pSC9 = 0.1 + input.mlMomentum * 0.2;

  1139→  1139→            (input.macdHist < 0
  1140→  1140→              ? (Math.abs(input.macdHist) / (atrSafe * 0.5)) * 0.3
  1141→  1141→              : 0),
  1142→  1142→          0,
  1143→  1143→          1
  1144→  1144→        ) * input.mlMomentum;

  1392→  1392→// === Graph Traversal ==========================================================
  1393→  1393→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1394→  1394→// =============================================================================
  1395→  1395→
  1396→  1396→function traverseGraph(
  1397→  1397→  nodeId: string,
  1398→  1398→  pathProb: number,
  1399→  1399→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1400→  1400→  ctx: ProbContext,
  1401→  1401→  adj: Map<string, number[]>,
  1402→  1402→  edges: GraphEdge[],
  1403→  1403→  edgeProbabilities: Record<number, number>,
  1404→  1404→  nodeAccum: Record<string, number>,
  1405→  1405→  scenarios: Record<string, number>,
  1406→  1406→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1407→  1407→): void {
  1408→  1408→  // Accumulate path probability at this node
  1409→  1409→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1410→  1410→
  1411→  1411→  const outEdgeIndices = adj.get(nodeId);
  1412→  1412→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1413→  1413→
  1414→  1414→  // Compute outgoing edge probabilities for this node
  1415→  1415→  const probs = computeNodeProbabilities(
  1416→  1416→    nodeId,
  1417→  1417→    ctx,
  1418→  1418→    outEdgeIndices.length
  1419→  1419→  );
  1420→  1420→
  1421→  1421→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1422→  1422→    const edgeIdx = outEdgeIndices[i];
  1423→  1423→    const edge = edges[edgeIdx];
  1424→  1424→    const prob = probs[i] || 0;
  1425→  1425→
  1426→  1426→    // Store edge probability
  1427→  1427→    edgeProbabilities[edgeIdx] = prob;
  1428→  1428→
  1429→  1429→    // Determine branch for child nodes
  1430→  1430→    let childBranch = currentBranch;
  1431→  1431→    if (nodeId === 'ROOT') {
  1432→  1432→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1433→  1433→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1434→  1434→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1435→  1435→    }
  1436→  1436→
  1437→  1437→    const childPathProb = pathProb * prob;
  1438→  1438→
  1439→  1439→    // Check if target is a terminal node (SC1–SC9)
  1440→  1440→    const isTerminal =
  1441→  1441→      edge.to.length === 3 &&
  1442→  1442→      edge.to[0] === 'S' &&
  1443→  1443→      edge.to[1] === 'C' &&
  1444→  1444→      edge.to[2] >= '1' &&
  1445→  1445→      edge.to[2] <= '9';
  1446→  1446→
  1447→  1447→    if (isTerminal) {
  1448→  1448→      // Accumulate scenario probability
  1449→  1449→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1450→  1450→      // Accumulate branch contribution
  1451→  1451→      if (childBranch) {
  1452→  1452→        contributions[edge.to][childBranch] += childPathProb;
  1453→  1453→      }
  1454→  1454→    }
  1455→  1455→
  1456→  1456→    // Recurse into child
  1457→  1457→    traverseGraph(
  1458→  1458→      edge.to,
  1459→  1459→      childPathProb,
  1460→  1460→      childBranch,
  1461→  1461→      ctx,
  1462→  1462→      adj,
  1463→  1463→      edges,
  1464→  1464→      edgeProbabilities,
  1465→  1465→      nodeAccum,
  1466→  1466→      scenarios,
  1467→  1467→      contributions
  1468→  1468→    );
  1469→  1469→  }
  1470→  1470→}

  1591→  1591→      .sort((a, b) => b.raw - a.raw);
  1592→  1592→
  1593→  1593→    for (let i = 0; i < entries.length; i++) {
  1594→  1594→      const isLast = i === entries.length - 1;
  1595→  1595→      if (isLast) {

  1599→  1599→        finalPcts[entries[i].key] = rounded;
  1600→  1600→        runningTotal += rounded;
  1601→  1601→      }
  1602→  1602→    }
  1603→  1603→  }
  1604→  1604→
  1605→  1605→  // Ensure no negative values due to rounding

  1608→  1608→  }

  1618→  1618→

  1623→  1623→  }
  1624→  1624→

  1626→  1626→  for (const [key, pct] of Object.entries(finalPcts)) {
  1627→  1627→    nodeValues[key] = `${pct}%`;
  1628→  1628→  }
  1629→  1629→
  1630→  1630→  // 9. Build path contributions (as percentages)
  1631→  1631→  const pathContributions: Record<
  1632→  1632→    string,
  1633→  1633→    { trend: number; breakout: number; reversal: number }
