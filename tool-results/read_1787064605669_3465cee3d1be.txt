   102→type ToolType = 'cursor' | 'trendline' | 'hline' | 'vline' | 'fibonacci' | 'rectangle' | 'text' | 'brush' | 'measure' | 'arrow';
   103→
   104→// — S/R strength-based line styling —————————————─
   105→function srLineStyle(strength: number, isTarget: boolean): { lineWidth: number; lineStyle: 0 | 1 | 2; color: string } {
   106→  if (isTarget) {
   107→    return { lineWidth: 3, lineStyle: 0, color: '' };
   108→  }
   109→  if (strength >= 9) return { lineWidth: 4, lineStyle: 0, color: '' };
   110→  if (strength >= 7) return { lineWidth: 3, lineStyle: 0, color: '' };
   111→  if (strength >= 5) return { lineWidth: 2, lineStyle: 1, color: '' };
   112→  if (strength >= 3) return { lineWidth: 2, lineStyle: 2, color: '' };
   113→  return { lineWidth: 1, lineStyle: 2, color: '' };
   114→}

   146→/* ----------------------------- COMPONENT --- */
   147→
   148→export default function CandlestickChart({ data, ta, height = 520 }: CandlestickChartProps) {
   149→  const chartContainerRef = useRef<HTMLDivElement>(null);
   150→  const svgRef = useRef<SVGSVGElement>(null);
   151→  const chartRef = useRef<IChartApi | null>(null);
   152→  const subscriberRef = useRef<{ dispose: () => void } | null>(null);
   153→
   154→  const [drawings, setDrawings] = useState<Drawing[]>([]);
   155→  const [activeTool, setActiveTool] = useState<ToolType>('cursor');
   156→  const [activeColor, setActiveColor] = useState(PALETTE[0]);
   157→  const [isDrawing, setIsDrawing] = useState(false);
   158→  const [currentDrawing, setCurrentDrawing] = useState<Drawing | null>(null);
   159→  const [textInput, setTextInput] = useState<{ x: number; y: number; drawingId: string } | null>(null);
   160→  const [textValue, setTextValue] = useState('');
   161→  const [renderKey, setRenderKey] = useState(0);
   162→  const [chartSize, setChartSize] = useState({ w: 0, h: 0 });
   163→
   164→  /* - Helper: coordinate conversion -------------- */
   165→  const getChartPoint = useCallback((clientX: number, clientY: number): DrawingPoint | null => {
   166→    const chart = chartRef.current;
   167→    const container = chartContainerRef.current;
   168→    if (!chart || !container) return null;
   169→    const rect = container.getBoundingClientRect();
   170→    const x = clientX - rect.left;
   171→    const y = clientY - rect.top;
   172→    const price = (chart.priceScale('right') as any).coordinateToPrice(y);
   173→    const time = chart.timeScale().coordinateToTime(x);
   174→    if (price === null || time === null) return null;
   175→    return { time: time as number, price };
   176→  }, []);
   177→
   178→  /* - SVG Rendering --------------------- */
   179→  const renderDrawings = useCallback(() => {
   180→    const chart = chartRef.current;
   181→    const svg = svgRef.current;
   182→    const container = chartContainerRef.current;
   183→    if (!chart || !svg || !container || drawings.length === 0) {
   184→      if (svg) svg.innerHTML = '';
   185→      return;
   186→    }
   187→
   188→    const rect = container.getBoundingClientRect();
   189→    const w = rect.width;
   190→    const h = rect.height;
   191→    if (w === 0 || h === 0) return;
   192→
   193→    let html = '';
   194→
   195→    // Defs for arrow markers
   196→    html += `<defs>`;
   197→    PALETTE.forEach((c) => {
   198→      html += `<marker id="arrow-${c.replace('#','')}" markerWidth="8" markerHeight="6" refX="7" refY="3" orient="auto"><polygon points="0 0, 8 3, 0 6" fill="${c}" /></marker>`;
   199→    });
   200→    html += `</defs>`;
   201→
   202→    const allDrawings = [...drawings];
   203→    if (currentDrawing) allDrawings.push(currentDrawing);
   204→
   205→    for (const d of allDrawings) {
   206→      if (!d.completed && d.points.length === 0) continue;
   207→      const col = d.color;
   208→      const lw = d.lineWidth;
   209→
   210→      switch (d.type) {
   211→        case 'trendline': {
   212→          if (d.points.length < 1) break;
   213→          const p0 = d.points[0];
   214→          const p1 = d.points.length > 1 ? d.points[d.points.length - 1] : p0;
   215→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   216→          const y0 = (chart.priceScale('right') as any).priceToCoordinate(p0.price);
   217→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   218→          const y1 = (chart.priceScale('right') as any).priceToCoordinate(p1.price);
   219→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   220→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" />`;
   221→          break;
   222→        }
   223→        case 'hline': {
   224→          if (d.points.length < 1) break;
   225→          const p = d.points[0];
   226→          const y = (chart.priceScale('right') as any).priceToCoordinate(p.price);
   227→          if (y == null) break;
   228→          html += `<line x1="0" y1="${y}" x2="${w}" y2="${y}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   229→          html += `<text x="4" y="${y - 4}" fill="${col}" font-size="11" font-family="Vazirmatn, sans-serif">${toPersianDigits(p.price.toFixed(0))}</text>`;
   230→          break;
   231→        }
   232→        case 'vline': {
   233→          if (d.points.length < 1) break;
   234→          const p = d.points[0];
   235→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   236→          if (x == null) break;
   237→          html += `<line x1="${x}" y1="0" x2="${x}" y2="${h}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   238→          break;
   239→        }
   240→        case 'fibonacci': {
   241→          if (d.points.length < 2) break;
   242→          const p0 = d.points[0];
   243→          const p1 = d.points[1];
   244→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   245→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   246→          if (x0 == null || x1 == null) break;
   247→          const priceHigh = Math.max(p0.price, p1.price);
   248→          const priceLow = Math.min(p0.price, p1.price);
   249→          const priceRange = priceHigh - priceLow;
   250→          if (priceRange === 0) break;
   251→
   252→          const yTop = (chart.priceScale('right') as any).priceToCoordinate(priceHigh);
   253→          const yBottom = (chart.priceScale('right') as any).priceToCoordinate(priceLow);
   254→          if (yTop == null || yBottom == null) break;
   255→          html += `<rect x="${Math.min(x0,x1)}" y="${yTop}" width="${Math.abs(x1-x0)}" height="${yBottom - yTop}" fill="${col}" opacity="0.04" />`;
   256→
   257→          for (const level of FIB_LEVELS) {
   258→            const price = priceHigh - level.pct * priceRange;
   259→            const y = (chart.priceScale('right') as any).priceToCoordinate(price);
   260→            if (y == null) continue;
   261→            html += `<line x1="${Math.min(x0,x1) - 10}" y1="${y}" x2="${Math.max(x0,x1) + 10}" y2="${y}" stroke="${col}" stroke-width="1" opacity="0.6" />`;
   262→            html += `<rect x="${Math.max(x0,x1) + 12}" y="${y - 8}" width="56" height="16" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="2" />`;
   263→            html += `<text x="${Math.max(x0,x1) + 14}" y="${y + 4}" fill="${col}" font-size="10" font-family="Vazirmatn, sans-serif">${toPersianDigits(level.label)} ${toPersianDigits(price.toFixed(0))}</text>`;
   264→          }
   265→          break;
   266→        }
   267→        case 'rectangle': {
   268→          if (d.points.length < 2) break;
   269→          const p0 = d.points[0];
   270→          const p1 = d.points[d.points.length - 1];
   271→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   272→          const y0 = (chart.priceScale('right') as any).priceToCoordinate(p0.price);
   273→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   274→          const y1 = (chart.priceScale('right') as any).priceToCoordinate(p1.price);
   275→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   276→          const rx = Math.min(x0, x1);
   277→          const ry = Math.min(y0, y1);
   278→          const rw = Math.abs(x1 - x0);
   279→          const rh = Math.abs(y1 - y0);
   280→          html += `<rect x="${rx}" y="${ry}" width="${rw}" height="${rh}" fill="${col}" fill-opacity="0.08" stroke="${col}" stroke-width="${lw}" rx="2" />`;
   281→          break;
   282→        }
   283→        case 'text': {
   284→          if (d.points.length < 1 || !d.text) break;
   285→          const p = d.points[0];
   286→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   287→          const y = (chart.priceScale('right') as any).priceToCoordinate(p.price);
   288→          if (x == null || y == null) break;
   289→          html += `<rect x="${x + 4}" y="${y - 16}" width="${d.text.length * 8 + 12}" height="22" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="3" />`;
   290→          html += `<text x="${x + 10}" y="${y + 1}" fill="${col}" font-size="12" font-family="Vazirmatn, sans-serif">${d.text}</text>`;
   291→          break;
   292→        }
   293→        case 'brush': {
   294→          if (d.points.length < 2) break;
   295→          let pts = '';
   296→          let valid = true;
   297→          for (const p of d.points) {
   298→            const x = chart.timeScale().timeToCoordinate(p.time as Time);
   299→            const y = (chart.priceScale('right') as any).priceToCoordinate(p.price);
   300→            if (x == null || y == null) { valid = false; break; }
   301→            pts += `${x},${y} `;
   302→          }
   303→          if (!valid || pts.length === 0) break;
   304→          html += `<polyline points="${pts.trim()}" fill="none" stroke="${col}" stroke-width="${lw}" stroke-linecap="round" stroke-linejoin="round" />`;
   305→          break;
   306→        }
   307→        case 'measure': {
   308→          if (d.points.length < 2) break;
   309→          const p0 = d.points[0];
   310→          const p1 = d.points[d.points.length - 1];
   311→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   312→          const y0 = (chart.priceScale('right') as any).priceToCoordinate(p0.price);
   313→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   314→          const y1 = (chart.priceScale('right') as any).priceToCoordinate(p1.price);
   315→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   316→
   317→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="4,2" />`;
   318→          const diff = p1.price - p0.price;
   319→          const pctChg = p0.price !== 0 ? ((diff / p0.price) * 100) : 0;
   320→          const bars = Math.abs(Math.round(p1.time - p0.time));
   321→          const midX = (x0 + x1) / 2;
   322→          const midY = (y0 + y1) / 2;
   323→          const label = `${diff >= 0 ? '+' : ''}${toPersianDigits(diff.toFixed(0))} (${pctChg >= 0 ? '+' : ''}${toPersianDigits(pctChg.toFixed(1))}%) ${toPersianDigits(String(bars))}bar`;
   324→          html += `<rect x="${midX - 40}" y="${midY - 24}" width="80" height="20" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="3" />`;
   325→          html += `<text x="${midX - 36}" y="${midY - 10}" fill="${col}" font-size="10" font-family="Vazirmatn, sans-serif">${label}</text>`;
   326→          break;
   327→        }
   328→        case 'arrow': {
   329→          if (d.points.length < 2) break;
   330→          const p0 = d.points[0];
   331→          const p1 = d.points[d.points.length - 1];
   332→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   333→          const y0 = (chart.priceScale('right') as any).priceToCoordinate(p0.price);
   334→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   335→          const y1 = (chart.priceScale('right') as any).priceToCoordinate(p1.price);
   336→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   337→          const markerId = `arrow-${col.replace('#','')}`;
   338→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" marker-end="url(#${markerId})" />`;
   339→          break;
   340→        }
   341→      }
   342→    }
   343→
   344→    svg.innerHTML = html;
   345→  }, [drawings, currentDrawing]);
   346→
   347→  /* - Chart Build ---------------------- */
   348→  const build = useCallback(() => {
   349→    const chartEl = chartContainerRef.current;
   350→    if (!chartEl || data.length === 0) return;
   351→
   352→    if (chartRef.current) {
   353→      chartRef.current.remove();
   354→      chartRef.current = null;
   355→    }
   356→
   357→    // — Build Jalali time map — detect Gregorian vs Jalali dates —
   358→    const jalaliMap = new Map<number, string>();
   359→    data.forEach((d, i) => {
   360→      jalaliMap.set(i, isGregorianDate(d.date) ? candleDateToJalali(d.date, 'compact') : formatJalaliString(d.date, 'compact'));
   361→    });
   362→
   363→    // — Persian numeral price formatter —
   364→    const persianPriceFormatter = (price: number) => {
   365→      return toPersianDigits(price.toLocaleString('en', { maximumFractionDigits: 0 }));
   366→    };
   367→
   368→    const chart = createChart(chartEl, {
   369→      layout: { background: { type: ColorType.Solid, color: BG }, textColor: TXT, fontSize: 11 },
   370→      grid: { vertLines: { color: GRID }, horzLines: { color: GRID } },
   371→      crosshair: { mode: CrosshairMode.Normal },
   372→      rightPriceScale: {
   373→        borderColor: '#e5e7eb',
   374→        scaleMargins: { top: 0.05, bottom: 0.3 },
   375→      },
   376→      timeScale: {
   377→        borderColor: '#e5e7eb',
   378→        rightOffset: 5,
   379→        barSpacing: 6,
   380→        tickMarkFormatter: (time: Time) => {
   381→          const idx = time as number;
   382→          return jalaliMap.get(idx) || String(idx);
   383→        },
   384→      },
   385→      localization: {
   386→        priceFormatter: persianPriceFormatter,
   387→      },
   388→      width: chartEl.clientWidth,
   389→      height,
   390→    });
   391→    chartRef.current = chart;
   392→
   393→    // Numeric time index
   394→    const candles = data.map((d, i) => ({
   395→      time: i as Time,
   396→      open: d.open,
   397→      high: d.high,
   398→      low: d.low,
   399→      close: d.close,
   400→    }));
   401→
   402→    const cs = chart.addSeries(CandlestickSeries, {
   403→      upColor: BULL,
   404→      downColor: BEAR,
   405→      borderUpColor: BULL,
   406→      borderDownColor: BEAR,
   407→      wickUpColor: BULL,
   408→      wickDownColor: BEAR,
   409→    });
   410→    cs.setData(candles);
   411→
   412→    // Volume
   413→    const vol = chart.addSeries(HistogramSeries, {
   414→      priceFormat: { type: 'volume' },
   415→      priceScaleId: 'vol',
   416→    });
   417→    vol.setData(
   418→      data.map((d, i) => ({
   419→        time: i as Time,
   420→        value: d.volume,
   421→        color: d.close >= d.open ? BULL : BEAR,
   422→      }))
   423→    );
   424→    chart.priceScale('vol').applyOptions({ scaleMargins: { top: 0.8, bottom: 0 } });
   425→
   426→    // TA overlays
   427→    if (ta) {
   428→      for (const cfg of SMA_CFG) {
   429→        const v = ta.sma[cfg.key];
   430→        if (v && v > 0)
   431→          cs.createPriceLine({
   432→            price: v,
   433→            color: cfg.color,
   434→            lineWidth: 1,
   435→            lineStyle: 0,
   436→            axisLabelVisible: true,
   437→            title: cfg.title,
   438→          });
   439→      }
   440→      const bb = ta.bollingerBands;
   441→      if (bb) {
   442→        if (bb.upper > 0) cs.createPriceLine({ price: bb.upper, color: 'rgba(124,58,237,0.6)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Upper' });
   443→        if (bb.middle > 0) cs.createPriceLine({ price: bb.middle, color: 'rgba(124,58,237,0.6)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Mid' });
   444→        if (bb.lower > 0) cs.createPriceLine({ price: bb.lower, color: 'rgba(124,58,237,0.6)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Lower' });
   445→      }
   446→      // SAR — use dark color instead of white (white invisible on white bg)
   447→      if (ta.sar > 0)
   448→        cs.createPriceLine({ price: ta.sar, color: 'rgba(217,119,6,0.7)', lineWidth: 1, lineStyle: 1, axisLabelVisible: true, title: 'SAR' });
   449→
   450→      // S/R with strength
   451→      const resistanceStrengths = ta.resistanceStrengths || [];
   452→      const supportStrengths = ta.supportStrengths || [];
   453→
   454→      resistanceStrengths.forEach((r, i) => {
   455→        if (r.price <= 0) return;
   456→        const style = srLineStyle(r.strength, r.isTarget);
   457→        const title = `R${toPersianDigits(String(i + 1))} (${toPersianDigits(String(r.strength))}/10)`;
   458→        cs.createPriceLine({
   459→          price: r.price,
   460→          color: r.isTarget ? '#dc2626' : BEAR,
   461→          lineWidth: style.lineWidth as 1|2|3|4,
   462→          lineStyle: style.lineStyle,
   463→          axisLabelVisible: i < 6,
   464→          title,
   465→        });
   466→      });
   467→
   468→      supportStrengths.forEach((s, i) => {
   469→        if (s.price <= 0) return;
   470→        const style = srLineStyle(s.strength, s.isTarget);
   471→        const title = `S${toPersianDigits(String(i + 1))} (${toPersianDigits(String(s.strength))}/10)`;
   472→        cs.createPriceLine({
   473→          price: s.price,
   474→          color: s.isTarget ? '#16a34a' : BULL,
   475→          lineWidth: style.lineWidth as 1|2|3|4,
   476→          lineStyle: style.lineStyle,
   477→          axisLabelVisible: i < 6,
   478→          title,
   479→        });
   480→      });
   481→    }
   482→
   483→    chart.timeScale().fitContent();
   484→
   485→    // Subscribe to visible range changes to re-render SVG
   486→    requestAnimationFrame(() => {
   487→      if (chartRef.current) {
   488→        const sub = chartRef.current.timeScale().subscribeVisibleLogicalRangeChange(() => {
   489→          setRenderKey((k) => k + 1);
   490→        });
   491→        subscriberRef.current = sub as unknown as { dispose: () => void };
   492→      }
   493→    });
   494→
   495→    // Set initial chart size (deferred to avoid React batching issue)
   496→    requestAnimationFrame(() => {
   497→      if (chartContainerRef.current) {
   498→        setChartSize({ w: chartContainerRef.current.clientWidth, h: height });
   499→      }
   500→    });
   501→  }, [data, ta, height]);
   502→
   503→  /* - Re-render SVG when renderKey changes --------- */
   504→  useEffect(() => {
   505→    renderDrawings();
   506→  }, [renderKey, renderDrawings]);
   507→
   508→  /* - Init chart ----------------------- */
   509→  useEffect(() => {
   510→    const el = chartContainerRef.current;
   511→    if (!el) return;
   512→    const ro = new ResizeObserver((entries) => {
   513→      for (const e of entries) {
   514→        const { width: w, height: h } = e.contentRect;
   515→        if (w > 0 && h > 0) {
   516→          chartRef.current?.applyOptions({ width: w, height: h });
   517→          setChartSize({ w, h });
   518→        }
   519→      }
   520→    });
   521→    ro.observe(el);
   522→    build();
   523→    return () => {
   524→      ro.disconnect();
   525→      subscriberRef.current?.dispose();
   526→      subscriberRef.current = null;
   527→      if (chartRef.current) {
   528→        chartRef.current.remove();
   529→        chartRef.current = null;
   530→      }
   531→    };
   532→  }, [build]);
   533→
   534→  /* - Mouse event handlers ----------------- */
   535→  const handleMouseDown = useCallback(
   536→    (e: React.MouseEvent) => {
   537→      if (activeTool === 'cursor') return;
   538→      if (textInput) return;
   539→
   540→      const point = getChartPoint(e.clientX, e.clientY);
   541→      if (!point) return;
   542→
   543→      if (activeTool === 'hline') {
   544→        const drawing: Drawing = {
   545→          id: Date.now().toString(),
   546→          type: 'hline',
   547→          points: [point],
   548→          color: activeColor,
   549→          lineWidth: 1,
   550→          completed: true,
   551→        };
   552→        setDrawings((prev) => [...prev, drawing]);
   553→        setIsDrawing(false);
   554→        return;
   555→      }
   556→
   557→      if (activeTool === 'text') {
   558→        const chart = chartRef.current;
   559→        const container = chartContainerRef.current;
   560→        if (!chart || !container) return;
   561→        const rect = container.getBoundingClientRect();
   562→        const x = e.clientX - rect.left;
   563→        const y = e.clientY - rect.top;
   564→
   565→        const tempDrawing: Drawing = {
   566→          id: Date.now().toString(),
   567→          type: 'text',
   568→          points: [point],
   569→          color: activeColor,
   570→          lineWidth: 1,
   571→          text: '',
   572→          completed: false,
   573→        };
   574→        setCurrentDrawing(tempDrawing);
   575→        setTextInput({ x, y, drawingId: tempDrawing.id });
   576→        setTextValue('');
   577→        setIsDrawing(true);
   578→        return;
   579→      }
   580→
   581→      const drawing: Drawing = {
   582→        id: Date.now().toString(),
   583→        type: activeTool,
   584→        points: [point],
   585→        color: activeColor,
   586→        lineWidth: activeTool === 'brush' ? 2 : 1,
   587→        completed: false,
   588→      };
   589→      setCurrentDrawing(drawing);
   590→      setIsDrawing(true);
   591→    },
   592→    [activeTool, activeColor, getChartPoint, textInput]
   593→  );
   594→
   595→  const handleMouseMove = useCallback(
   596→    (e: React.MouseEvent) => {
   597→      if (!isDrawing || !currentDrawing) return;
   598→
   599→      const point = getChartPoint(e.clientX, e.clientY);
   600→      if (!point) return;
   601→
   602→      if (currentDrawing.type === 'brush') {
   603→        setCurrentDrawing({
   604→          ...currentDrawing,
   605→          points: [...currentDrawing.points, point],
   606→        });
   607→      } else if (currentDrawing.type !== 'text' && currentDrawing.type !== 'hline') {
   608→        setCurrentDrawing({
   609→          ...currentDrawing,
   610→          points: [currentDrawing.points[0], point],
   611→        });
   612→      }
   613→    },
   614→    [isDrawing, currentDrawing, getChartPoint]
   615→  );
   616→
   617→  const handleMouseUp = useCallback(() => {
   618→    if (!isDrawing || !currentDrawing) return;
   619→
   620→    if (currentDrawing.type === 'text') return;
   621→
   622→    const completed: Drawing = {
   623→      ...currentDrawing,
   624→      completed: true,
   625→    };
   626→
   627→    const minPoints = currentDrawing.type === 'hline' ? 1 : 2;
   628→    if (completed.points.length >= minPoints) {
   629→      setDrawings((prev) => [...prev, completed]);
   630→    }
   631→
   632→    setCurrentDrawing(null);
   633→    setIsDrawing(false);
   634→  }, [isDrawing, currentDrawing]);
   635→
   636→  /* - Text input submit ------------------- */
   637→  const handleTextSubmit = useCallback(() => {
   638→    if (!textInput || !textValue.trim() || !currentDrawing) {
   639→      setTextInput(null);
   640→      setCurrentDrawing(null);
   641→      setIsDrawing(false);
   642→      return;
   643→    }
   644→
   645→    const completed: Drawing = {
   646→      ...currentDrawing,
   647→      text: textValue.trim(),
   648→      completed: true,
   649→    };
   650→    setDrawings((prev) => [...prev, completed]);
   651→    setTextInput(null);
   652→    setTextValue('');
   653→    setCurrentDrawing(null);
   654→    setIsDrawing(false);
   655→  }, [textInput, textValue, currentDrawing]);
   656→
   657→  const handleTextCancel = useCallback(() => {
   658→    setTextInput(null);
   659→    setTextValue('');
   660→    setCurrentDrawing(null);
   661→    setIsDrawing(false);
   662→  }, []);
   663→
   664→  /* - Delete / Clear -------------------- */
   665→  const handleDeleteLast = useCallback(() => {
   666→    setDrawings((prev) => prev.slice(0, -1));
   667→  }, []);
   668→
   669→  const handleClearAll = useCallback(() => {
   670→    setDrawings([]);
   671→  }, []);
   672→
   673→  /* - Render ------------------------- */
   674→  return (
   675→    <div className="w-full">
   676→      {/* - TOOLBAR --------------------- */}
   677→      <div
   678→        className="flex items-center gap-1 px-2 py-1.5 rounded-t-lg border border-b-0 overflow-x-auto flex-nowrap"
   679→        style={{ backgroundColor: TOOLBAR_BG, borderColor: TOOLBAR_BORDER }}
   680→      >
   681→        {TOOLS.map((tool) => (
   682→          <button
   683→            key={tool.id}
   684→            onClick={() => {
   685→              setActiveTool(tool.id);
   686→              if (isDrawing) {
   687→                setIsDrawing(false);
   688→                setCurrentDrawing(null);
   689→                setTextInput(null);
   690→              }
   691→            }}
   692→            title={tool.label}
   693→            className={`shrink-0 flex items-center justify-center w-8 h-8 rounded-md transition-all cursor-pointer ${
   694→              activeTool === tool.id
   695→                ? 'bg-amber-500/20 text-amber-800 ring-1 ring-amber-500/30'
   696→                : 'text-gray-500 hover:bg-gray-100 hover:text-gray-700'
   697→            }`}
   698→          >
   699→            {tool.icon}
   700→          </button>
   701→        ))}
   702→
   703→        <div className="w-px h-6 bg-gray-200 mx-1 shrink-0" />
   704→
   705→        {PALETTE.map((c) => (
   706→          <button
   707→            key={c}
   708→            onClick={() => setActiveColor(c)}
   709→            title={c}
   710→            className={`shrink-0 w-6 h-6 rounded-full border-2 transition-all cursor-pointer ${
   711→              activeColor === c ? 'border-gray-900 scale-110' : 'border-transparent hover:scale-105'
   712→            }`}
   713→            style={{ backgroundColor: c }}
   714→          />
   715→        ))}
   716→
   717→        <div className="w-px h-6 bg-gray-200 mx-1 shrink-0" />
   718→
   719→        <button
   720→          onClick={handleDeleteLast}
   721→          title="حذف آخر"
   722→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-500 hover:bg-red-50 hover:text-red-700 transition-all cursor-pointer"
   723→        >
   724→          <Trash2 className="w-3.5 h-3.5" />
   725→        </button>
   726→
   727→        <button
   728→          onClick={handleClearAll}
   729→          title="پاک کردن همه"
   730→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-500 hover:bg-red-50 hover:text-red-700 transition-all cursor-pointer"
   731→        >
   732→          <X className="w-3.5 h-3.5" />
   733→        </button>
   734→
   735→        <div className="mr-auto shrink-0 text-xs text-gray-500 font-medium" dir="rtl">
   736→          {TOOLS.find((t) => t.id === activeTool)?.label}
   737→        </div>
   738→      </div>
   739→
   740→      <div
   741→        className="relative rounded-b-lg overflow-hidden border border-gray-200"
   742→        style={{ height }}
   743→      >
   744→        <div
   745→          ref={chartContainerRef}
   746→          className="w-full h-full"
   747→          data-chart
   748→        />
   749→
   750→        <svg
   751→          ref={svgRef}
   752→          className="absolute top-0 left-0"
   753→          style={{
   754→            width: chartSize.w || '100%',
   755→            height: chartSize.h || '100%',
   756→            pointerEvents: activeTool !== 'cursor' ? 'auto' : 'none',
   757→            zIndex: 10,
   758→          }}
   759→          onMouseDown={handleMouseDown}
   760→          onMouseMove={handleMouseMove}
   761→          onMouseUp={handleMouseUp}
   762→          onMouseLeave={handleMouseUp}
   763→        />
   764→
   765→        {textInput && (
   766→          <div
   767→            className="absolute z-20"
   768→            style={{
   769→              left: textInput.x,
   770→              top: textInput.y - 8,
   771→            }}
   772→          >
   773→            <input
   774→              type="text"
   775→              value={textValue}
   776→              onChange={(e) => setTextValue(e.target.value)}
   777→              onKeyDown={(e) => {
   778→                if (e.key === 'Enter') handleTextSubmit();
   779→                if (e.key === 'Escape') handleTextCancel();
   780→              }}
   781→              placeholder="متن..."
   782→              autoFocus
   783→              dir="rtl"
   784→              className="bg-white border border-gray-200 rounded px-2 py-1 text-sm text-gray-900 outline-none focus:border-amber-500 w-32"
   785→              onBlur={handleTextSubmit}
   786→            />
   787→          </div>
   788→        )}
   789→      </div>
   790→    </div>
   791→  );
   792→}
