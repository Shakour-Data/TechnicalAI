   667→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   668→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   669→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   670→  grade?: string; overlapCount?: number; touchCount?: number;
   671→}) {
   672→  const isSupport = type === 'support';
   673→  const isTargetType = type === 'target';
   674→
   675→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   676→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   677→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   678→
   679→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   680→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   681→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   682→
   683→  return (
   684→    <div
   685→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   686→      style={{
   687→        background: bgAccent,
   688→        border: `1px solid ${borderAccent}`,
   689→        borderRight: `3px solid ${accentColor}`,
   690→      }}
   691→    >
   692→      {/* Label + Price */}
   693→      <div className="flex-1 min-w-0">
   694→        <div className="flex items-center gap-2">
   695→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   696→          {isTarget && <span className="text-[9px]">🎯</span>}
   697→          {overlapCount !== undefined && overlapCount > 0 && (
   698→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   699→              {toPersianDigits(overlapCount)} هم‌پوشانی
   700→            </span>
   701→          )}
   702→        </div>
   703→        <div className="flex items-baseline gap-1">
   704→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   705→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   706→        </div>
   707→      </div>
   708→
   709→      {/* Strength + Grade */}
   710→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   711→        <div className="flex items-center gap-1.5">
   712→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   713→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   714→          <span className="text-[9px] text-gray-500">/۱۰</span>
   715→        </div>
   716→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   717→          <div
   718→            className="h-full rounded-full transition-all duration-500"
   719→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   720→          />
   721→        </div>
   722→        {touchCount !== undefined && touchCount > 0 && (
   723→          <span className="text-[8px] text-gray-400">{toPersianDigits(touchCount)} برخورد</span>
   724→        )}
   725→      </div>
   726→    </div>
   727→  );
   728→}
