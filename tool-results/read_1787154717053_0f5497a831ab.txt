   230→function parsePersianNum(str: string | null | undefined): number {
   231→  if (!str) return 0;
   232→  return Number(str.replace(/[,۰-۹]/g, (c) => {
   233→    const persianDigits = '۰۱۲۳۴۵۶۷۸۹';
   234→    const idx = persianDigits.indexOf(c);
   235→    return idx >= 0 ? String(idx) : c === ',' ? '' : c;
   236→  }));
   237→}

   256→/**
   257→ * Fetch latest 3 candles from tgju.org chart API for given keys.
   258→ * Returns map of key → { close, high, low, prevClose }.
   259→ * Used for BOTH static instruments AND currency/gold items (to get accurate change).
   260→ */
   261→async function batchFetchChartPrices(keys: string[]): Promise<Map<string, StaticPriceInfo>> {
   262→  if (keys.length === 0) return new Map();
   263→
   264→  const now = Date.now();
   265→  if (staticPriceCache.data.size > 0 && now - staticPriceCache.ts < STATIC_PRICE_TTL) {
   266→    // Return cached data for requested keys
   267→    const result = new Map<string, StaticPriceInfo>();
   268→    for (const key of keys) {
   269→      const cached = staticPriceCache.data.get(key);
   270→      if (cached) result.set(key, cached);
   271→    }
   272→    if (result.size === keys.length) return result;
   273→  }
   274→
   275→  const result = new Map<string, StaticPriceInfo>();
   276→  let zai: Awaited<ReturnType<typeof import('z-ai-web-dev-sdk').default.create>> | null = null;
   277→
   278→  try {
   279→    const ZAI = (await import('z-ai-web-dev-sdk')).default;
   280→    zai = await ZAI.create();
   281→  } catch {
   282→    return result;
   283→  }
   284→
   285→  const deadline = Date.now() + BATCH_TIMEOUT_MS;
   286→  let idx = 0;
   287→
   288→  while (idx < keys.length && Date.now() < deadline) {
   289→    const batch = keys.slice(idx, idx + CONCURRENCY);
   290→    idx += CONCURRENCY;
   291→
   292→    const promises = batch.map(async (key) => {
   293→      try {
   294→        const url = `${TGJU_CHART_API}/${key}?lang=fa&order_dir=desc&start=0&length=3`;
   295→        const res = await zai!.functions.invoke('page_reader', { url });
   296→        const html = res.data.html || '';
   297→        const jsonStr = html.replace(/<[^>]+>/g, '').trim();
   298→        if (!jsonStr.startsWith('{')) return null;
   299→        const data = JSON.parse(jsonStr);
   300→        const rows: string[][] = data.data;
   301→        if (!rows || rows.length < 2) return null;
   302→        const newest = rows[0];
   303→        const prev = rows[1];
   304→        return {
   305→          key,
   306→          close: parsePersianNum(newest[3]),
   307→          high: parsePersianNum(newest[2]),
   308→          low: parsePersianNum(newest[1]),
   309→          prevClose: parsePersianNum(prev[3]),
   310→        };
   311→      } catch {
   312→        return null;
   313→      }
   314→    });
   315→
   316→    const results = await Promise.allSettled(promises);
   317→    for (const r of results) {
   318→      if (r.status === 'fulfilled' && r.value) {
   319→        result.set(r.value.key, {
   320→          close: r.value.close,
   321→          high: r.value.high,
   322→          low: r.value.low,
   323→          prevClose: r.value.prevClose,
   324→        });
   325→      }
   326→    }
   327→  }
   328→
   329→  // Update global cache with all fetched data
   330→  for (const [key, val] of result) {
   331→    staticPriceCache.data.set(key, val);
   332→  }
   333→  if (result.size > 0) {
   334→    staticPriceCache.ts = now;
   335→  }
   336→
   337→  return result;
   338→}

   340→/* ─── Fetch live instruments (currency + gold from tgju-api) + static list ── */
   341→
   342→export async function fetchTgjuInstruments(): Promise<TgjuInstrument[]> {
   343→  const now = Date.now();
   344→  if (liveInstrumentsCache.data && now - liveInstrumentsCache.ts < INSTRUMENTS_TTL) {
   345→    return liveInstrumentsCache.data;
   346→  }
   347→
   348→  const controller = new AbortController();
   349→  const timeout = setTimeout(() => controller.abort(), 10000);
   350→
   351→  try {
   352→    const [currencyRes, goldRes] = await Promise.all([
   353→      fetch(`${TGJU_API_BASE}/api/price/currency`, { signal: controller.signal }),
   354→      fetch(`${TGJU_API_BASE}/api/price/gold`, { signal: controller.signal }),
   355→    ]);
   356→
   357→    clearTimeout(timeout);
   358→
   359→    const instruments: TgjuInstrument[] = [];
   360→    const currencyGoldKeys: string[] = []; // keys to fetch chart data for accurate change
   361→
   362→    if (currencyRes.ok) {
   363→      const currencies: TgjuPriceItem[] = await currencyRes.json();
   364→      for (const c of currencies) {
   365→        const price = parsePersianNum(c.price);
   366→        const low = parsePersianNum(c.low_price);
   367→        const high = parsePersianNum(c.high_price);
   368→        instruments.push({
   369→          title: c.title,
   370→          key: c.key,
   371→          price,
   372→          highPrice: high,
   373→          lowPrice: low,
   374→          change: 0, // Will be updated from chart data
   375→          changePercent: 0, // Will be updated from chart data
   376→          category: 'currency',
   377→        });
   378→        currencyGoldKeys.push(c.key);
   379→      }
   380→    }
   381→
   382→    if (goldRes.ok) {
   383→      const goldCategories: TgjuGoldCategory[] = await goldRes.json();
   384→      for (const cat of goldCategories) {
   385→        for (const item of cat.prices) {
   386→          const price = parsePersianNum(item.price);
   387→          const low = parsePersianNum(item.low_price);
   388→          const high = parsePersianNum(item.high_price);
   389→
   390→          let category: TgjuCategoryType = 'gold';
   391→          if (item.key.startsWith('ime_fund_')) category = 'gold_etf';
   392→          else if (item.key.startsWith('silver_')) category = 'silver';
   393→
   394→          instruments.push({
   395→            title: item.title,
   396→            key: item.key,
   397→            price,
   398→            highPrice: high,
   399→            lowPrice: low,
   400→            change: 0, // Will be updated from chart data
   401→            changePercent: 0, // Will be updated from chart data
   402→            category,
   403→            groupTitle: cat.title,
   404→          });
   405→          currencyGoldKeys.push(item.key);
   406→        }
   407→      }
   408→    }
   409→
   410→    // Add static instruments (prices will be populated by background batch fetch)
   411→    for (const s of STATIC_INSTRUMENTS) {
   412→      const cached = staticPriceCache.data.get(s.key);
   413→      instruments.push({
   414→        title: s.title,
   415→        key: s.key,
   416→        price: cached?.close ?? 0,
   417→        highPrice: cached?.high ?? 0,
   418→        lowPrice: cached?.low ?? 0,
   419→        change: cached ? (cached.close - cached.prevClose) : 0,
   420→        changePercent: cached && cached.prevClose > 0 ? ((cached.close - cached.prevClose) / cached.prevClose) * 100 : 0,
   421→        category: s.category,
   422→        groupTitle: s.groupTitle,
   423→      });
   424→    }
   425→
   426→    liveInstrumentsCache.data = instruments;
   427→    liveInstrumentsCache.ts = now;
   428→
   429→    // ── Background: fetch chart data for ALL TGJU keys ──
   430→    // This gives us accurate prevClose for currency/gold (instead of using intraday low)
   431→    // and populates prices for static instruments
   432→    const allKeys = [
   433→      ...currencyGoldKeys,
   434→      ...STATIC_INSTRUMENTS.map(s => s.key),
   435→    ];
   436→    batchFetchChartPrices(allKeys).then((priceMap) => {
   437→      if (priceMap.size > 0 && liveInstrumentsCache.data) {
   438→        for (const inst of liveInstrumentsCache.data) {
   439→          const p = priceMap.get(inst.key);
   440→          if (!p) continue;
   441→
   442→          if (inst.category === 'currency' || inst.category === 'gold' ||
   443→              inst.category === 'silver' || inst.category === 'gold_etf') {
   444→            // Currency/gold: keep live price, but use chart prevClose for accurate change
   445→            if (p.prevClose > 0) {
   446→              inst.change = inst.price - p.prevClose;
   447→              inst.changePercent = (inst.change / p.prevClose) * 100;
   448→            }
   449→          } else if (inst.price === 0) {
   450→            // Static instruments: update all fields from chart data
   451→            inst.price = p.close;
   452→            inst.highPrice = p.high;
   453→            inst.lowPrice = p.low;
   454→            inst.change = p.close - p.prevClose;
   455→            inst.changePercent = p.prevClose > 0 ? ((p.close - p.prevClose) / p.prevClose) * 100 : 0;
   456→          }
   457→        }
   458→      }
   459→    }).catch(() => {});
   460→
   461→    return instruments;
   462→  } catch {
   463→    clearTimeout(timeout);
   464→    return liveInstrumentsCache.data || [];
   465→  }
   466→}

   468→/* ─── Fetch historical OHLC data via z-ai page_reader ── */
   469→
   470→export async function fetchTgjuHistory(tgjuKey: string): Promise<TgjuOHLC[]> {
   471→  const cached = historyCache.get(tgjuKey);
   472→  const now = Date.now();
   473→  if (cached && now - cached.ts < HISTORY_TTL) {
   474→    return cached.data;
   475→  }
   476→
   477→  try {
   478→    const ZAI = (await import('z-ai-web-dev-sdk')).default;
   479→    const zai = await ZAI.create();
   480→
   481→    const url = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=5000`;
   482→
   483→    const result = await zai.functions.invoke('page_reader', { url });
   484→    const html = result.data.html || '';
   485→
   486→    // Strip HTML tags (page_reader wraps in <pre> tags)
   487→    const jsonStr = html.replace(/<[^>]+>/g, '').trim();
   488→
   489→    if (!jsonStr.startsWith('{')) {
   490→      return [];
   491→    }
   492→
   493→    const data = JSON.parse(jsonStr);
   494→    const rows: string[][] = data.data;
   495→
   496→    if (!rows || rows.length === 0) {
   497→      return [];
   498→    }
   499→
   500→    // Parse: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   501→    const candles: TgjuOHLC[] = rows
   502→      .map((row) => ({
   503→        date: (row[6] || '').replace(/\//g, '-'),
   504→        open: parsePersianNum(row[0]),
   505→        high: parsePersianNum(row[2]),
   506→        low: parsePersianNum(row[1]),
   507→        close: parsePersianNum(row[3]),
   508→      }))
   509→      .filter((c) => c.open > 0 && c.high > 0 && c.low > 0 && c.close > 0)
   510→      .reverse(); // newest-last for charting
   511→
   512→    historyCache.set(tgjuKey, { data: candles, ts: now });
   513→    return candles;
   514→  } catch (err) {
   515→    console.error(`[TGJU] Failed to fetch history for ${tgjuKey}:`, err);
   516→    return cached?.data || [];
   517→  }
   518→}
