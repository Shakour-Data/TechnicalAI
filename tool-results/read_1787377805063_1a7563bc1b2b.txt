   813→   813→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   814→   814→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
   815→   815→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
   816→   816→  grade?: string; overlapCount?: number; touchCount?: number;
   817→   817→}) {
   818→   818→  const isSupport = type === 'support';
   819→   819→  const isTargetType = type === 'target';
   820→   820→
   821→   821→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   822→   822→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   823→   823→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   824→   824→
   825→   825→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   826→   826→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
   827→   827→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
   828→   828→
   829→   829→  return (
   830→   830→    <div
   831→   831→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   832→   832→      style={{
   833→   833→        background: bgAccent,
   834→   834→        border: `1px solid ${borderAccent}`,
   835→   835→        borderRight: `3px solid ${accentColor}`,
   836→   836→      }}
   837→   837→    >
   838→   838→      {/* Label + Price */}
   839→   839→      <div className="flex-1 min-w-0">
   840→   840→        <div className="flex items-center gap-2">
   841→   841→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   842→   842→          {isTarget && <span className="text-[9px]">🎯</span>}
   843→   843→          {overlapCount !== undefined && overlapCount > 0 && (
   844→   844→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
   845→   845→              {toPersianDigits(overlapCount)} هم‌پوشانی
   846→   846→            </span>
   847→   847→          )}
   848→   848→        </div>
   849→   849→        <div className="flex items-baseline gap-1">
   850→   850→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   851→   851→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
   852→   852→        </div>
   853→   853→      </div>
   854→   854→
   855→   855→      {/* Strength + Grade */}
   856→   856→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
   857→   857→        <div className="flex items-center gap-1.5">
   858→   858→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
   859→   859→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
   860→   860→          <span className="text-[9px] text-gray-500">/۱۰</span>
   861→   861→        </div>
   862→   862→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
   863→   863→          <div
