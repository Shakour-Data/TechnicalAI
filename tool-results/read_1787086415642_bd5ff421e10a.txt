   110→   110→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   111→   111→function sma(closes: number[], period: number): number {
   112→   112→  if (closes.length < period) return 0;
   113→   113→  const slice = closes.slice(-period);
   114→   114→  return slice.reduce((a, b) => a + b, 0) / period;
   115→   115→}

   117→   117→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   118→   118→function emaCalc(closes: number[], period: number): number {
   119→   119→  if (closes.length < period) return 0;
   120→   120→  const k = 2 / (period + 1);
   121→   121→  let emaVal = sma(closes.slice(0, period), period);
   122→   122→  for (let i = period; i < closes.length; i++) {
   123→   123→    emaVal = closes[i] * k + emaVal * (1 - k);
   124→   124→  }
   125→   125→  return emaVal;
   126→   126→}

   128→   128→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   129→   129→function calcRSI(closes: number[], period: number = 14): number {
   130→   130→  if (closes.length < period + 1) return 50;
   131→   131→  let avgGain = 0;
   132→   132→  let avgLoss = 0;
   133→   133→  for (let i = 1; i <= period; i++) {
   134→   134→    const diff = closes[i] - closes[i - 1];
   135→   135→    if (diff > 0) avgGain += diff;
   136→   136→    else avgLoss += Math.abs(diff);
   137→   137→  }
   138→   138→  avgGain /= period;
   139→   139→  avgLoss /= period;
   140→   140→  for (let i = period + 1; i < closes.length; i++) {
   141→   141→    const diff = closes[i] - closes[i - 1];
   142→   142→    const gain = diff > 0 ? diff : 0;
   143→   143→    const loss = diff < 0 ? Math.abs(diff) : 0;
   144→   144→    avgGain = (avgGain * (period - 1) + gain) / period;
   145→   145→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   146→   146→  }
   147→   147→  if (avgLoss === 0) return 100;
   148→   148→  const rs = avgGain / avgLoss;
   149→   149→  return 100 - 100 / (1 + rs);
   150→   150→}

   152→   152→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   153→   153→function calcMFI(data: OHLCV[], period: number = 14): number {
   154→   154→  if (data.length < period + 1) return 50;
   155→   155→  let posFlow = 0;
   156→   156→  let negFlow = 0;
   157→   157→  for (let i = data.length - period; i < data.length; i++) {
   158→   158→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   159→   159→    const mf = tp * data[i].volume;
   160→   160→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   161→   161→    if (tp > prevTp) posFlow += mf;
   162→   162→    else negFlow += mf;
   163→   163→  }
   164→   164→  if (negFlow === 0) return 100;
   165→   165→  const mfr = posFlow / negFlow;
   166→   166→  return 100 - 100 / (1 + mfr);
   167→   167→}

   170→   170→function calcCCI(data: OHLCV[], period: number = 20): number {
   171→   171→  if (data.length < period) return 0;
   172→   172→  const tps: number[] = [];
   173→   173→  for (let i = data.length - period; i < data.length; i++) {
   174→   174→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   175→   175→  }
   176→   176→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   177→   177→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   178→   178→  if (meanDev === 0) return 0;
   179→   179→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   180→   180→}

   182→   182→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   183→   183→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   184→   184→  if (data.length < kPeriod) return { k: 50, d: 50 };
   185→   185→  const rawKs: number[] = [];
   186→   186→  for (let i = kPeriod - 1; i < data.length; i++) {
   187→   187→    let lowest = Infinity;
   188→   188→    let highest = -Infinity;
   189→   189→    for (let j = i - kPeriod + 1; j <= i; j++) {
   190→   190→      if (data[j].low < lowest) lowest = data[j].low;
   191→   191→      if (data[j].high > highest) highest = data[j].high;
   192→   192→    }
   193→   193→    const range = highest - lowest;
   194→   194→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   195→   195→  }
   196→   196→  const smoothedK: number[] = [];
   197→   197→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   198→   198→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   199→   199→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   200→   200→  }
   201→   201→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   202→   202→  const dSlice = smoothedK.slice(-smoothD);
   203→   203→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   204→   204→  return { k, d };
   205→   205→}

   208→   208→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   209→   209→  if (data.length < period) return -50;
   210→   210→  const slice = data.slice(-period);
   211→   211→  let highest = -Infinity;
   212→   212→  let lowest = Infinity;
   213→   213→  for (const c of slice) {
   214→   214→    if (c.high > highest) highest = c.high;
   215→   215→    if (c.low < lowest) lowest = c.low;
   216→   216→  }
   217→   217→  const range = highest - lowest;
   218→   218→  if (range === 0) return -50;
   219→   219→  return ((highest - data[data.length - 1].close) / range) * -100;
   220→   220→}

   222→   222→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   223→   223→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   224→   224→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   225→   225→  const emaFast = emaCalc(closes, fast);
   226→   226→  const emaSlow = emaCalc(closes, slow);
   227→   227→  const macdLine = emaFast - emaSlow;
   228→   228→  // Calculate signal as EMA of MACD values
   229→   229→  const macdValues: number[] = [];
   230→   230→  const kFast = 2 / (fast + 1);
   231→   231→  const kSlow = 2 / (slow + 1);
   232→   232→  let eFast = sma(closes.slice(0, fast), fast);
   233→   233→  let eSlow = sma(closes.slice(0, slow), slow);
   234→   234→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   235→   235→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   236→   236→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   237→   237→    macdValues.push(eFast - eSlow);
   238→   238→  }
   239→   239→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   240→   240→  const kSig = 2 / (sig + 1);
   241→   241→  let signalVal = sma(macdValues.slice(0, sig), sig);
   242→   242→  for (let i = sig; i < macdValues.length; i++) {
   243→   243→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   244→   244→  }
   245→   245→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   246→   246→}

   248→   248→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   249→   249→function calcATR(data: OHLCV[], period: number = 14): number {
   250→   250→  if (data.length < 2) return 0;
   251→   251→  const trs: number[] = [];
   252→   252→  for (let i = 1; i < data.length; i++) {
   253→   253→    const tr = Math.max(
   254→   254→      data[i].high - data[i].low,
   255→   255→      Math.abs(data[i].high - data[i - 1].close),
   256→   256→      Math.abs(data[i].low - data[i - 1].close)
   257→   257→    );
   258→   258→    trs.push(tr);
   259→   259→  }
   260→   260→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   261→   261→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   262→   262→  for (let i = period; i < trs.length; i++) {
   263→   263→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   264→   264→  }
   265→   265→  return atrVal;
   266→   266→}

   268→   268→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   269→   269→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   270→   270→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   271→   271→  const plusDMs: number[] = [];
   272→   272→  const minusDMs: number[] = [];
   273→   273→  const trs: number[] = [];
   274→   274→  for (let i = 1; i < data.length; i++) {
   275→   275→    const upMove = data[i].high - data[i - 1].high;
   276→   276→    const downMove = data[i - 1].low - data[i].low;
   277→   277→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   278→   278→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   279→   279→    trs.push(Math.max(
   280→   280→      data[i].high - data[i].low,
   281→   281→      Math.abs(data[i].high - data[i - 1].close),
   282→   282→      Math.abs(data[i].low - data[i - 1].close)
   283→   283→    ));
   284→   284→  }
   285→   285→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   286→   286→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   287→   287→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   288→   288→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   289→   289→  const dxValues: number[] = [];
   290→   290→  for (let i = period; i < trs.length; i++) {
   291→   291→    smoothTR = smoothTR - smoothTR / period + trs[i];
   292→   292→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   293→   293→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   294→   294→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   295→   295→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   296→   296→    const diSum = diPlus + diMinus;
   297→   297→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   298→   298→    dxValues.push(dx);
   299→   299→  }
   300→   300→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   301→   301→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   302→   302→  for (let i = period; i < dxValues.length; i++) {
   303→   303→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   304→   304→  }
   305→   305→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   306→   306→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   307→   307→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   308→   308→}

   310→   310→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   311→   311→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   312→   312→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   313→   313→  let isLong = true;
   314→   314→  let af = step;
   315→   315→  let ep = data[0].high;
   316→   316→  let sar = data[0].low;
   317→   317→  for (let i = 1; i < data.length; i++) {
   318→   318→    const prevSar = sar;
   319→   319→    sar = prevSar + af * (ep - prevSar);
   320→   320→    // Ensure SAR is within bounds
   321→   321→    if (isLong) {
   322→   322→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   323→   323→      if (data[i].low < sar) {
   324→   324→        isLong = false;
   325→   325→        sar = ep;
   326→   326→        ep = data[i].low;
   327→   327→        af = step;
   328→   328→      } else {
   329→   329→        if (data[i].high > ep) {
   330→   330→          ep = data[i].high;
   331→   331→          af = Math.min(af + step, max);
   332→   332→        }
   333→   333→      }
   334→   334→    } else {
   335→   335→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   336→   336→      if (data[i].high > sar) {
   337→   337→        isLong = true;
   338→   338→        sar = ep;
   339→   339→        ep = data[i].high;
   340→   340→        af = step;
   341→   341→      } else {
   342→   342→        if (data[i].low < ep) {
   343→   343→          ep = data[i].low;
   344→   344→          af = Math.min(af + step, max);
   345→   345→        }
   346→   346→      }
   347→   347→    }
   348→   348→  }
   349→   349→  return sar;
   350→   350→}

   353→   353→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   354→   354→  if (closes.length < period) {
   355→   355→    const c = closes[closes.length - 1] ?? 0;
   356→   356→    return { upper: c, middle: c, lower: c };
   357→   357→  }
   358→   358→  const slice = closes.slice(-period);
   359→   359→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   360→   360→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   361→   361→  const sd = Math.sqrt(variance);
   362→   362→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   363→   363→}

   366→   366→function calcOBV(data: OHLCV[]): number {
   367→   367→  let obv = 0;
   368→   368→  for (let i = 1; i < data.length; i++) {
   369→   369→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   370→   370→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   371→   371→  }
   372→   372→  return obv;
   373→   373→}

   375→   375→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   376→   376→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   377→   377→  const n = values.length;
   378→   378→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   379→   379→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   380→   380→  for (let i = 0; i < n; i++) {
   381→   381→    sumX += i;
   382→   382→    sumY += values[i];
   383→   383→    sumXY += i * values[i];
   384→   384→    sumX2 += i * i;
   385→   385→    sumY2 += values[i] * values[i];
   386→   386→  }
   387→   387→  const denom = n * sumX2 - sumX * sumX;
   388→   388→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   389→   389→  const slope = (n * sumXY - sumX * sumY) / denom;
   390→   390→  const intercept = (sumY - slope * sumX) / n;
   391→   391→  const meanY = sumY / n;
   392→   392→  let ssRes = 0, ssTot = 0;
   393→   393→  for (let i = 0; i < n; i++) {
   394→   394→    const predicted = slope * i + intercept;
   395→   395→    ssRes += (values[i] - predicted) ** 2;
   396→   396→    ssTot += (values[i] - meanY) ** 2;
   397→   397→  }
   398→   398→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   399→   399→  return { slope, intercept, r2 };
   400→   400→}

   402→   402→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   403→   403→function roundToNice(price: number): number {
   404→   404→  const abs = Math.abs(price);
   405→   405→  let step: number;
   406→   406→  if (abs >= 100000) step = 100;
   407→   407→  else if (abs >= 10000) step = 50;
   408→   408→  else if (abs >= 1000) step = 10;
   409→   409→  else if (abs >= 100) step = 5;
   410→   410→  else step = 1;
   411→   411→  return Math.round(price / step) * step;
   412→   412→}

   416→   416→  const highs: { price: number; idx: number }[] = [];
   417→   417→  const lows: { price: number; idx: number }[] = [];
   418→   418→  for (let i = lookback; i < data.length - lookback; i++) {
   419→   419→    let isHigh = true;
   420→   420→    let isLow = true;
   421→   421→    for (let j = i - lookback; j <= i + lookback; j++) {
   422→   422→      if (j === i) continue;
   423→   423→      if (data[j].high >= data[i].high) isHigh = false;
   424→   424→      if (data[j].low <= data[i].low) isLow = false;
   425→   425→    }
   426→   426→    if (isHigh) highs.push({ price: data[i].high, idx: i });
   427→   427→    if (isLow) lows.push({ price: data[i].low, idx: i });
   428→   428→  }
   429→   429→  return { highs, lows };
   430→   430→}

   495→   495→// ─── Helper: Linear System Solver (Gaussian Elimination) ─────────────────
   496→   496→function solveLinearSystem(A: number[][], b: number[]): number[] | null {
   497→   497→  const n = b.length;
   498→   498→  const aug = A.map((row, i) => [...row, b[i]]);
   499→   499→
   500→   500→  for (let col = 0; col < n; col++) {
   501→   501→    let maxVal = Math.abs(aug[col][col]);
   502→   502→    let maxRow = col;
   503→   503→    for (let row = col + 1; row < n; row++) {
   504→   504→      if (Math.abs(aug[row][col]) > maxVal) {
   505→   505→        maxVal = Math.abs(aug[row][col]);
   506→   506→        maxRow = row;
   507→   507→      }
   508→   508→    }
   509→   509→    if (maxVal < 1e-10) return null;
   510→   510→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   511→   511→
   512→   512→    for (let row = col + 1; row < n; row++) {
   513→   513→      const factor = aug[row][col] / aug[col][col];
   514→   514→      for (let j = col; j <= n; j++) {
   515→   515→        aug[row][j] -= factor * aug[col][j];
   516→   516→      }
   517→   517→    }
   518→   518→  }
   519→   519→
   520→   520→  const x = new Array(n).fill(0);
   521→   521→  for (let i = n - 1; i >= 0; i--) {
   522→   522→    x[i] = aug[i][n];
   523→   523→    for (let j = i + 1; j < n; j++) {
   524→   524→      x[i] -= aug[i][j] * x[j];
   525→   525→    }
   526→   526→    if (Math.abs(aug[i][i]) < 1e-10) return null;
   527→   527→    x[i] /= aug[i][i];
   528→   528→  }
   529→   529→
   530→   530→  return x;
   531→   531→}

   533→   533→// ─── Helper: Generate All 7-Method S/R Levels ─────────────────────────────
   534→   534→function generateAllLevels(data: OHLCV[], endIdx: number): RawSRLevel[] {
   535→   535→  const levels: RawSRLevel[] = [];
   536→   536→  if (endIdx < 10) return levels;
   537→   537→
   538→   538→  const slice = data.slice(0, endIdx + 1);
   539→   539→  const currentPrice = slice[slice.length - 1].close;
   540→   540→  const closes = slice.map(d => d.close);
   541→   541→  const noFib: [string, string] = ['—', ''];
   542→   542→
   543→   543→  // ── Method 1: Swing High/Low (Price Action) ──────────────────────────
   544→   544→  for (const lookback of [3, 5, 7]) {
   545→   545→    const swings = findSwingLevels(slice, lookback);
   546→   546→    const tol = currentPrice * 0.005; // 0.5% tolerance
   547→   547→
   548→   548→    // Connect swing highs → resistance
   549→   549→    const usedHigh = new Set<number>();
   550→   550→    for (const h of swings.highs) {
   551→   551→      if (usedHigh.has(h.idx)) continue;
   552→   552→      let sum = h.price;
   553→   553→      let count = 1;
   554→   554→      for (const h2 of swings.highs) {
   555→   555→        if (h2 === h || usedHigh.has(h2.idx)) continue;
   556→   556→        if (Math.abs(h2.price - h.price) < tol) {
   557→   557→          sum += h2.price;
   558→   558→          count++;
   559→   559→          usedHigh.add(h2.idx);
   560→   560→        }
   561→   561→      }
   562→   562→      usedHigh.add(h.idx);
   563→   563→      if (sum / count > currentPrice * 1.005) {
   564→   564→        levels.push({ price: sum / count, method: 'Swing_High', type: 'resistance', ...noFib });
   565→   565→      }
   566→   566→    }
   567→   567→
   568→   568→    // Connect swing lows → support
   569→   569→    const usedLow = new Set<number>();
   570→   570→    for (const l of swings.lows) {
   571→   571→      if (usedLow.has(l.idx)) continue;
   572→   572→      let sum = l.price;
   573→   573→      let count = 1;
   574→   574→      for (const l2 of swings.lows) {
   575→   575→        if (l2 === l || usedLow.has(l2.idx)) continue;
   576→   576→        if (Math.abs(l2.price - l.price) < tol) {
   577→   577→          sum += l2.price;
   578→   578→          count++;
   579→   579→          usedLow.add(l2.idx);
   580→   580→        }
   581→   581→      }
   582→   582→      usedLow.add(l.idx);
   583→   583→      if (sum / count < currentPrice * 0.995) {
   584→   584→        levels.push({ price: sum / count, method: 'Swing_Low', type: 'support', ...noFib });
   585→   585→      }
   586→   586→    }
   587→   587→  }
   588→   588→
   589→   589→  // ── Method 2: SMA Dynamic S/R ──────────────────────────────────────
   590→   590→  for (const period of [20, 50, 100, 200]) {
   591→   591→    const smaVal = sma(closes, period);
   592→   592→    if (smaVal <= 0) continue;
   593→   593→    const type: 'support' | 'resistance' = currentPrice > smaVal ? 'support' : 'resistance';
   594→   594→    levels.push({ price: smaVal, method: `SMA_${period}`, type, ...noFib });
   595→   595→  }
   596→   596→
   597→   597→  // ── Method 3: Bollinger Bands ───────────────────────────────────────
   598→   598→  const bb = calcBollingerBands(closes);
   599→   599→  if (bb.upper > 0 && bb.lower > 0) {
   600→   600→    levels.push({ price: bb.upper, method: 'BB_Upper', type: 'resistance', ...noFib });
   601→   601→    levels.push({ price: bb.lower, method: 'BB_Lower', type: 'support', ...noFib });
   602→   602→  }
   603→   603→
   604→   604→  // ── Method 4: Fibonacci Retracement + Extension ─────────────────────
   605→   605→  interface SwingPair { high: number; highIdx: number; low: number; lowIdx: number; range: number }
   606→   606→  let bestSwing: SwingPair | null = null;
   607→   607→
   608→   608→  for (const lb of [3, 5, 7, 10, 15, 20]) {
   609→   609→    const sw = findSwingLevels(slice, lb);
   610→   610→    if (sw.highs.length === 0 || sw.lows.length === 0) continue;
   611→   611→    for (const h of sw.highs) {
   612→   612→      for (const l of sw.lows) {
   613→   613→        if (Math.abs(h.idx - l.idx) < 3) continue;
   614→   614→        const range = Math.abs(h.price - l.price);
   615→   615→        if (range / currentPrice < 0.03) continue;
   616→   616→        if (!bestSwing || range > bestSwing.range) {
   617→   617→          bestSwing = {
   618→   618→            high: Math.max(h.price, l.price),
   619→   619→            highIdx: h.price > l.price ? h.idx : l.idx,
   620→   620→            low: Math.min(h.price, l.price),
   621→   621→            lowIdx: h.price < l.price ? l.idx : h.idx,
   622→   622→            range,
   623→   623→          };
   624→   624→        }
   625→   625→      }
   626→   626→    }
   627→   627→  }
   628→   628→
   629→   629→  if (bestSwing) {
   630→   630→    const H = bestSwing.high;
   631→   631→    const L = bestSwing.low;
   632→   632→    const range = H - L;
   633→   633→
   634→   634→    for (const ratio of FIB_RETRACEMENTS) {
   635→   635→      const price = H - ratio * range;
   636→   636→      if (price <= 0) continue;
   637→   637→      const rounded = roundToNice(price);
   638→   638→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   639→   639→      levels.push({
   640→   640→        price: rounded, method: `Fibonacci_${ratio}`, type,
   641→   641→        fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   642→   642→      });
   643→   643→    }
   644→   644→
   645→   645→    for (const extRatio of FIB_EXTENSIONS) {
   646→   646→      const price = H + (extRatio - 1) * range;
   647→   647→      if (price <= 0) continue;
   648→   648→      const rounded = roundToNice(price);
   649→   649→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   650→   650→      levels.push({
   651→   651→        price: rounded, method: `Fibonacci_${extRatio}`, type,
   652→   652→        fibRatio: String(extRatio), fibLabel: FIB_LABELS[String(extRatio)] || '',
   653→   653→      });
   654→   654→    }
   655→   655→
   656→   656→    // Inverse extensions (below L)
   657→   657→    for (const extRatio of FIB_EXTENSIONS) {
   658→   658→      const price = L - (extRatio - 1) * range;
   659→   659→      if (price <= 0) continue;
   660→   660→      const rounded = roundToNice(price);
   661→   661→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   662→   662→      levels.push({
   663→   663→        price: rounded, method: `Fibonacci_${-extRatio}`, type,
   664→   664→        fibRatio: String(-extRatio), fibLabel: '',
   665→   665→      });
   666→   666→    }
   667→   667→
   668→   668→    // Secondary swing confluence (check overlap with secondary swings)
   669→   669→    const secondarySwings: SwingPair[] = [];
   670→   670→    for (const lb of [3, 5, 7]) {
   671→   671→      const sw = findSwingLevels(slice, lb);
   672→   672→      for (const h of sw.highs) {
   673→   673→        for (const l of sw.lows) {
   674→   674→          if (Math.abs(h.idx - l.idx) < 3) continue;
   675→   675→          const r = Math.abs(h.price - l.price);
   676→   676→          if (r / currentPrice < 0.05) continue;
   677→   677→          if (bestSwing && Math.abs(r - bestSwing.range) / bestSwing.range < 0.1) continue;
   678→   678→          secondarySwings.push({
   679→   679→            high: Math.max(h.price, l.price), highIdx: h.price > l.price ? h.idx : l.idx,
   680→   680→            low: Math.min(h.price, l.price), lowIdx: h.price < l.price ? l.idx : h.idx,
   681→   681→            range: r,
   682→   682→          });
   683→   683→        }
   684→   684→      }
   685→   685→    }
   686→   686→    // Secondary swings' fib levels are merged via confluence (1% rule in mergeLevels)
   687→   687→    const confluenceTol = currentPrice * 0.005;
   688→   688→    for (const sec of secondarySwings.slice(0, 3)) {
   689→   689→      const sH = sec.high, sL = sec.low, sR = sH - sL;
   690→   690→      for (const ratio of FIB_RETRACEMENTS) {
   691→   691→        const price = roundToNice(sH - ratio * sR);
   692→   692→        if (price <= 0) continue;
   693→   693→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   694→   694→        levels.push({
   695→   695→          price, method: `Fibonacci_${ratio}`, type,
   696→   696→          fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   697→   697→        });
   698→   698→      }
   699→   699→      for (const extR of FIB_EXTENSIONS) {
   700→   700→        const price = roundToNice(sH + (extR - 1) * sR);
   701→   701→        if (price <= 0) continue;
   702→   702→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   703→   703→        levels.push({
   704→   704→          price, method: `Fibonacci_${extR}`, type,
   705→   705→          fibRatio: String(extR), fibLabel: FIB_LABELS[String(extR)] || '',
   706→   706→        });
   707→   707→      }
   708→   708→    }
   709→   709→  }
   710→   710→
   711→   711→  // ── Method 5: Volume Profile (VAP) ──────────────────────────────────
   712→   712→  const dataHasVolume = slice.some(d => d.volume > 0);
   713→   713→  if (dataHasVolume) {
   714→   714→    let minP = Infinity, maxP = -Infinity;
   715→   715→    for (const d of slice) {
   716→   716→      if (d.low < minP) minP = d.low;
   717→   717→      if (d.high > maxP) maxP = d.high;
   718→   718→    }
   719→   719→    const numBins = 40;
   720→   720→    const binSize = Math.max((maxP - minP) / numBins, currentPrice * 0.001);
   721→   721→    const vap = new Array(numBins).fill(0);
   722→   722→    const binCenters: number[] = [];
   723→   723→
   724→   724→    for (let b = 0; b < numBins; b++) {
   725→   725→      binCenters.push(minP + (b + 0.5) * binSize);
   726→   726→    }
   727→   727→
   728→   728→    for (const d of slice) {
   729→   729→      for (let b = 0; b < numBins; b++) {
   730→   730→        if (d.low <= binCenters[b] && binCenters[b] <= d.high) {
   731→   731→          vap[b] += d.volume;
   732→   732→        }
   733→   733→      }
   734→   734→    }
   735→   735→
   736→   736→    const maxVAP = Math.max(...vap);
   737→   737→    if (maxVAP > 0) {
   738→   738→      // POC (Point of Control)
   739→   739→      let pocIdx = 0;
   740→   740→      for (let b = 1; b < numBins; b++) {
   741→   741→        if (vap[b] > vap[pocIdx]) pocIdx = b;
   742→   742→      }
   743→   743→      const pocPrice = binCenters[pocIdx];
   744→   744→      const pocType: 'support' | 'resistance' = pocPrice > currentPrice ? 'resistance' : 'support';
   745→   745→      levels.push({ price: pocPrice, method: 'VAP_POC', type: pocType, ...noFib });
   746→   746→
   747→   747→      // HVN (High Volume Nodes)
   748→   748→      for (let b = 0; b < numBins; b++) {
   749→   749→        if (b === pocIdx) continue;
   750→   750→        if (vap[b] > 0.7 * maxVAP) {
   751→   751→          const hvnPrice = binCenters[b];
   752→   752→          const hvnType: 'support' | 'resistance' = hvnPrice > currentPrice ? 'resistance' : 'support';
   753→   753→          levels.push({ price: hvnPrice, method: 'VAP_HVN', type: hvnType, ...noFib });
   754→   754→        }
   755→   755→      }
   756→   756→    }
   757→   757→  }
   758→   758→
   759→   759→  // ── Method 6: Pivot Points (4 algorithms) ───────────────────────────
   760→   760→  if (endIdx >= 1) {
   761→   761→    const prev = slice[endIdx - 1]; // Previous completed bar
   762→   762→    const pH = prev.high, pL = prev.low, pC = prev.close, pO = prev.open;
   763→   763→
   764→   764→    // Standard Pivot
   765→   765→    {
   766→   766→      const PP = (pH + pL + pC) / 3;
   767→   767→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   768→   768→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   769→   769→      const R3 = pH + 2 * (PP - pL), S3 = pL - 2 * (pH - PP);
   770→   770→ if (R1 > 0) levels.push({ price: R1, method: 'Pivot_R1', type: 'resistance', ...noFib });
   771→   771→      if (R2 > 0) levels.push({ price: R2, method: 'Pivot_R2', type: 'resistance', ...noFib });
   772→   772→      if (R3 > 0) levels.push({ price: R3, method: 'Pivot_R3', type: 'resistance', ...noFib });
   773→   773→      if (S1 > 0) levels.push({ price: S1, method: 'Pivot_S1', type: 'support', ...noFib });
   774→   774→      if (S2 > 0) levels.push({ price: S2, method: 'Pivot_S2', type: 'support', ...noFib });
   775→   775→      if (S3 > 0) levels.push({ price: S3, method: 'Pivot_S3', type: 'support', ...noFib });
   776→   776→    }
   777→   777→
   778→   778→    // Fibonacci Pivot
   779→   779→    {
   780→   780→      const PP = (pH + pL + pC) / 3;
   781→   781→      const range = pH - pL;
   782→   782→      for (const [mult, suffix] of [[0.382, '1'], [0.618, '2'], [1.0, '3']] as [number, string][]) {
   783→   783→        const R = PP + mult * range;
   784→   784→        const S = PP - mult * range;
   785→   785→        if (R > 0) levels.push({ price: R, method: `Fib_Pivot_R${suffix}`, type: 'resistance', ...noFib });
   786→   786→        if (S > 0) levels.push({ price: S, method: `Fib_Pivot_S${suffix}`, type: 'support', ...noFib });
   787→   787→      }
   788→   788→    }
   789→   789→
   790→   790→    // Woodie's Pivot
   791→   791→    {
   792→   792→      const PP = (pH + pL + 2 * pC) / 4;
   793→   793→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   794→   794→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   795→   795→      if (R1 > 0) levels.push({ price: R1, method: 'Woodie_R1', type: 'resistance', ...noFib });
   796→   796→      if (R2 > 0) levels.push({ price: R2, method: 'Woodie_R2', type: 'resistance', ...noFib });
   797→   797→      if (S1 > 0) levels.push({ price: S1, method: 'Woodie_S1', type: 'support', ...noFib });
   798→   798→      if (S2 > 0) levels.push({ price: S2, method: 'Woodie_S2', type: 'support', ...noFib });
   799→   799→    }
   800→   800→
   801→   801→    // DeMark Pivot
   802→   802→    {
   803→   803→      const X = pC < pO ? pH + 2 * pL + pC : pC > pO ? 2 * pH + pL + pC : pH + pL + 2 * pC;
   804→   804→      const PP = X / 4;
   805→   805→      const R1 = X / 2 - pL;
   806→   806→      const S1 = X / 2 - pH;
   807→   807→      if (R1 > 0) levels.push({ price: R1, method: 'DeMark_R1', type: 'resistance', ...noFib });
   808→   808→      if (S1 > 0) levels.push({ price: S1, method: 'DeMark_S1', type: 'support', ...noFib });
   809→   809→    }
   810→   810→  }
   811→   811→
   812→   812→  // ── Method 7: Psychological Levels ──────────────────────────────────
   813→   813→  {
   814→   814→    const step = psychStep(currentPrice);
   815→   815→    const base = psychRound(currentPrice);
   816→   816→    for (const mult of [-2, -1, 1, 2]) {
   817→   817→      const price = base + mult * step;
   818→   818→      if (price <= 0) continue;
   819→   819→      const type: 'support' | 'resistance' = price > currentPrice ? 'resistance' : 'support';
   820→   820→      levels.push({ price, method: 'Psychological', type, ...noFib });
   821→   821→    }
   822→   822→  }
   823→   823→
   824→   824→  return levels;
   825→   825→}

   827→   827→// ─── Helper: Merge Nearby Levels (1% Confluence) ──────────────────────────
   828→   828→function mergeLevels(rawLevels: RawSRLevel[], currentPrice: number): MergedSRLevel[] {
   829→   829→  if (rawLevels.length === 0) return [];
   830→   830→
   831→   831→  const tol = currentPrice * 0.01; // 1% tolerance
   832→   832→  const merged: MergedSRLevel[] = [];
   833→   833→  const used = new Set<number>();
   834→   834→
   835→   835→  for (let i = 0; i < rawLevels.length; i++) {
   836→   836→    if (used.has(i)) continue;
   837→   837→
   838→   838→    let sumPrice = rawLevels[i].price;
   839→   839→    let count = 1;
   840→   840→    const methods = new Set<string>();
   841→   841→    methods.add(rawLevels[i].method);
   842→   842→    let bestFibRatio = rawLevels[i].fibRatio;
   843→   843→    let bestFibLabel = rawLevels[i].fibLabel;
   844→   844→
   845→   845→    for (let j = i + 1; j < rawLevels.length; j++) {
   846→   846→      if (used.has(j)) continue;
   847→   847→      if (Math.abs(rawLevels[j].price - rawLevels[i].price) < tol) {
   848→   848→        sumPrice += rawLevels[j].price;
   849→   849→        count++;
   850→   850→        methods.add(rawLevels[j].method);
   851→   851→        used.add(j);
   852→   852→        // Prefer Fibonacci label if available
   853→   853→        if (rawLevels[j].fibRatio !== '—' && bestFibRatio === '—') {
   854→   854→          bestFibRatio = rawLevels[j].fibRatio;
   855→   855→          bestFibLabel = rawLevels[j].fibLabel;
   856→   856→        }
   857→   857→      }
   858→   858→    }
   859→   859→    used.add(i);
   860→   860→
   861→   861→    const avgPrice = sumPrice / count;
   862→   862→    const type: 'support' | 'resistance' = avgPrice >= currentPrice ? 'resistance' : 'support';
   863→   863→
   864→   864→    merged.push({
   865→   865→      price: avgPrice,
   866→   866→      type,
   867→   867→      methods: [...methods],
   868→   868→      overlapCount: methods.size,
   869→   869→      fibRatio: bestFibRatio,
   870→   870→      fibLabel: bestFibLabel,
   871→   871→      score: 0,
   872→   872→      strength: 1,
   873→   873→      touchCount: 0,
   874→   874→      volumeRatio: 0,
   875→   875→      freshness: 0,
   876→   876→      distancePercent: 0,
   877→   877→    });
   878→   878→  }
   879→   879→
   880→   880→  return merged;
   881→   881→}

   883→   883→// ─── Helper: Compute Level Features (5 ML features) ──────────────────────
   884→   884→function computeLevelFeatures(
   885→   885→  levelPrice: number,
   886→   886→  data: OHLCV[],
   887→   887→  endIdx: number,
   888→   888→  currentPrice: number,
   889→   889→  overlapCount: number,
   890→   890→): number[] {
   891→   891→  const lookback = Math.min(30, endIdx);
   892→   892→  const startIdx = endIdx - lookback;
   893→   893→  const touchTol = currentPrice * 0.002; // 0.2%
   894→   894→
   895→   895→  let touchCount = 0;
   896→   896→  let totalVolAtTouches = 0;
   897→   897→  let totalVol = 0;
   898→   898→  let barsSinceLastTouch = lookback;
   899→   899→
   900→   900→  for (let i = startIdx; i <= endIdx; i++) {
   901→   901→    const bar = data[i];
   902→   902→    totalVol += bar.volume;
   903→   903→
   904→   904→    if (bar.low <= levelPrice + touchTol && bar.high >= levelPrice - touchTol) {
   905→   905→      touchCount++;
   906→   906→      totalVolAtTouches += bar.volume;
   907→   907→      barsSinceLastTouch = endIdx - i;
   908→   908→    }
   909→   909→  }
   910→   910→
   911→   911→  const avgVol = lookback > 0 ? totalVol / lookback : 0;
   912→   912→  const volumeRatio = (touchCount > 0 && avgVol > 0) ? (totalVolAtTouches / touchCount) / avgVol : 1;
   913→   913→  const freshness = Math.max(0, Math.min(1, 1 - barsSinceLastTouch / 30));
   914→   914→  const distancePercent = Math.min(1, Math.abs(levelPrice - currentPrice) / currentPrice);
   915→   915→
   916→   916→  // Features: [touch_count, volume_ratio, overlap_count, freshness, distance_percent]
   917→   917→  return [touchCount, volumeRatio, overlapCount - 1, freshness, distancePercent];
   918→   918→}

   920→   920→// ─── Helper: Train S/R Scoring Weights (Ridge OLS) ───────────────────────
   921→   921→function trainSRWeights(data: OHLCV[]): number[] {
   922→   922→  const defaultWeights = [0.25, 0.20, 0.25, 0.15, 0.15];
   923→   923→
   924→   924→  if (data.length < 50) return defaultWeights;
   925→   925→
   926→   926→  const X: number[][] = [];
   927→   927→  const y: number[] = [];
   928→   928→
   929→   929→  for (let i = 40; i <= data.length - 11; i += 5) {
   930→   930→    const barPrice = data[i].close;
   931→   931→    const rawLevels = generateAllLevels(data, i);
   932→   932→    if (rawLevels.length === 0) continue;
   933→   933→    const merged = mergeLevels(rawLevels, barPrice);
   934→   934→
   935→   935→    for (const level of merged) {
   936→   936→      const features = computeLevelFeatures(level.price, data, i, barPrice, level.overlapCount);
   937→   937→      X.push(features);
   938→   938→
   939→   939→      // Label: did price return within 10 bars?
   940→   940→      let returned = false;
   941→   941→      const retTol = barPrice * 0.005;
   942→   942→      for (let j = i + 1; j <= Math.min(i + 10, data.length - 1); j++) {
   943→   943→        if (data[j].low <= level.price + retTol && data[j].high >= level.price - retTol) {
   944→   944→          returned = true;
   945→   945→          break;
   946→   946→        }
   947→   947→      }
   948→   948→      y.push(returned ? 1 : 0);
   949→   949→    }
   950→   950→  }
   951→   951→
   952→   952→  if (X.length < 20) return defaultWeights;
   953→   953→
   954→   954→  // Ridge OLS: w = (X^T X + λI)^(-1) X^T y
   955→   955→  const n = X.length;
   956→   956→  const p = 5;
   957→   957→  const lambda = 0.01;
   958→   958→
   959→   959→  const XtX: number[][] = Array.from({ length: p }, () => new Array(p).fill(0));
   960→   960→  for (let i = 0; i < n; i++) {
   961→   961→    for (let a = 0; a < p; a++) {
   962→   962→      for (let b = 0; b < p; b++) {
   963→   963→        XtX[a][b] += X[i][a] * X[i][b];
   964→   964→      }
   965→   965→    }
   966→   966→  }
   967→   967→  for (let a = 0; a < p; a++) XtX[a][a] += lambda;
   968→   968→
   969→   969→  const Xty: number[] = new Array(p).fill(0);
   970→   970→  for (let i = 0; i < n; i++) {
   971→   971→    for (let a = 0; a < p; a++) {
   972→   972→      Xty[a] += X[i][a] * y[i];
   973→   973→    }
   974→   974→  }
   975→   975→
   976→   976→  const weights = solveLinearSystem(XtX, Xty);
   977→   977→  if (!weights) return defaultWeights;
   978→   978→  if (weights.some(w => isNaN(w) || !isFinite(w))) return defaultWeights;
   979→   979→
   980→   980→  for (let i = 0; i < p; i++) {
   981→   981→    if (weights[i] < 0) weights[i] = 0;
   982→   982→  }
   983→   983→
   984→   984→  const sum = weights.reduce((a, b) => a + b, 0);
   985→   985→  if (sum <= 0) return defaultWeights;
   986→   986→
   987→   987→  return weights.map(w => w / sum);
   988→   988→}

   990→   990→// ─── Support / Resistance (7-Method ML-Based System) ─────────────────────
   991→   991→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   992→   992→  resistances: number[];
   993→   993→  supports: number[];
   994→   994→  supportStrengths: LevelStrength[];
   995→   995→  resistanceStrengths: LevelStrength[];
   996→   996→  priceTargets: LevelStrength[];
   997→   997→} {
   998→   998→  if (data.length < 10) {
   999→   999→    return {
  1000→  1000→      resistances: [], supports: [],
  1001→  1001→      supportStrengths: [], resistanceStrengths: [],
  1002→  1002→      priceTargets: [],
  1003→  1003→    };
  1004→  1004→  }
  1005→  1005→
  1006→  1006→  // ── Step 1: Generate all levels from 7 methods ────────────────────────
  1007→  1007→  const rawLevels = generateAllLevels(data, data.length - 1);
  1008→  1008→
  1009→  1009→  // ── Step 2: Merge nearby levels (confluence within 1%) ────────────────
  1010→  1010→  let merged = mergeLevels(rawLevels, currentPrice);
  1011→  1011→
  1012→  1012→  // ── Step 3: Train ML weights on historical data ───────────────────────
  1013→  1013→  const weights = trainSRWeights(data);
  1014→  1014→
  1015→  1015→  // ── Step 4: Compute features & score each merged level ────────────────
  1016→  1016→  for (const level of merged) {
  1017→  1017→    const features = computeLevelFeatures(level.price, data, data.length - 1, currentPrice, level.overlapCount);
  1018→  1018→
  1019→  1019→    const touchScore = Math.min(features[0] / 5, 1);
  1020→  1020→    const volumeScore = Math.min(features[1] / 2, 1);
  1021→  1021→    const overlapScore = Math.min(features[2] * 2, 10) / 10;
  1022→  1022→    const freshnessScore = features[3];
  1023→  1023→    const distanceScore = 1 - features[4];
  1024→  1024→
  1025→  1025→    const powerScore = (
  1026→  1026→      weights[0] * touchScore +
  1027→  1027→      weights[1] * volumeScore +
  1028→  1028→      weights[2] * overlapScore +
  1029→  1029→      weights[3] * freshnessScore +
  1030→  1030→      weights[4] * distanceScore
  1031→  1031→    ) * 10;
  1032→  1032→
  1033→  1033→    level.score = Math.max(0, Math.min(10, powerScore));
  1034→  1034→    level.strength = Math.max(1, Math.min(10, Math.round(level.score)));
  1035→  1035→    level.touchCount = features[0];
  1036→  1036→    level.volumeRatio = features[1];
  1037→  1037→    level.freshness = features[3];
  1038→  1038→    level.distancePercent = features[4];
  1039→  1039→  }
  1040→  1040→
  1041→  1041→  // ── Step 5: Psychological rounding of all level prices ─────────────────
  1042→  1042→  for (const level of merged) {
  1043→  1043→    level.price = psychRound(level.price);
  1044→  1044→  }
  1045→  1045→
  1046→  1046→  // Re-classify types after rounding
  1047→  1047→  for (const level of merged) {
  1048→  1048→    level.type = level.price >= currentPrice ? 'resistance' : 'support';
  1049→  1049→  }
  1050→  1050→
  1051→  1051→  // Deduplicate after rounding (keep strongest at same price)
  1052→  1052→  {
  1053→  1053→    const deduped: MergedSRLevel[] = [];
  1054→  1054→    for (const level of merged) {
  1055→  1055→      const existing = deduped.find(d => Math.abs(d.price - level.price) < currentPrice * 0.001);
  1056→  1056→      if (existing) {
  1057→  1057→        if (level.score > existing.score) {
  1058→  1058→          Object.assign(existing, level);
  1059→  1059→        }
  1060→  1060→      } else {
  1061→  1061→        deduped.push(level);
  1062→  1062→      }
  1063→  1063→    }
  1064→  1064→    merged = deduped;
  1065→  1065→  }
  1066→  1066→
  1067→  1067→  // ── Step 6: Separate into supports and resistances ─────────────────────
  1068→  1068→  let supports = merged.filter(l => l.type === 'support');
  1069→  1069→  let resistances = merged.filter(l => l.type === 'resistance');
  1070→  1070→
  1071→  1071→  // ── Step 7: Distance filtering (strict 5-10% gap rules) ─────────────
  1072→  1072→  // Rules:
  1073→  1073→  //   1. Consecutive lines of same type: 5% ≤ gap ≤ 10%
  1074→  1074→  //   2. Nearest support ≤ 10% below price (can be < 5%)
  1075→  1075→  //   3. Nearest resistance ≤ 10% above price (can be < 5%)
  1076→  1076→
  1077→  1077→  const filterAndSelectLevels = (
  1078→  1078→    levels: MergedSRLevel[],
  1079→  1079→    type: 'support' | 'resistance'
  1080→  1080→  ): MergedSRLevel[] => {
  1081→  1081→    if (levels.length === 0) return [];
  1082→  1082→
  1083→  1083→    // Sort: supports descending (nearest first), resistances ascending (nearest first)
  1084→  1084→    const sorted = [...levels].sort((a, b) =>
  1085→  1085→      type === 'support' ? b.price - a.price : a.price - b.price
  1086→  1086→    );
  1087→  1087→
  1088→  1088→    // ── Rule 2/3: Nearest level must be within 10% of price ──
  1089→  1089→    const nearest = sorted[0];
  1090→  1090→    const distFromPrice = Math.abs(nearest.price - currentPrice) / currentPrice;
  1091→  1091→    if (distFromPrice > 0.10) {
  1092→  1092→      // Nearest level is too far — generate a synthetic level at ~5% from price
  1093→  1093→      const syntheticPrice = type === 'support'
  1094→  1094→        ? Math.round(currentPrice * 0.95 / psychStep(currentPrice)) * psychStep(currentPrice)
  1095→  1095→        : Math.round(currentPrice * 1.05 / psychStep(currentPrice)) * psychStep(currentPrice);
  1096→  1096→      const synthetic: MergedSRLevel = {
  1097→  1097→        price: syntheticPrice,
  1098→  1098→        type,
  1099→  1099→        score: 3,
  1100→  1100→        strength: 3,
  1101→  1101→        overlapCount: 1,
  1102→  1102→        methods: ['Psychological'],
  1103→  1103→        fibRatio: '—',
  1104→  1104→        fibLabel: '',
  1105→  1105→      };
  1106→  1106→      sorted.unshift(synthetic);
  1107→  1107→    }
  1108→  1108→
  1109→  1109→    // ── Rule 1: Select levels with 5-10% gaps between consecutive ──
  1110→  1110→    const result: MergedSRLevel[] = [sorted[0]];
  1111→  1111→
  1112→  1112→    for (let i = 1; i < sorted.length && result.length < 6; i++) {
  1113→  1113→      const prevPrice = result[result.length - 1].price;
  1114→  1114→      const currPrice = sorted[i].price;
  1115→  1115→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
  1116→  1116→
  1117→  1117→      if (gap < 0.05) {
  1118→  1118→        // Too close — keep the stronger one
  1119→  1119→        if (sorted[i].score > result[result.length - 1].score) {
  1120→  1120→          result[result.length - 1] = sorted[i];
  1121→  1121→        }
  1122→  1122→      } else if (gap <= 0.10) {
  1123→  1123→        // Perfect gap (5-10%) — accept
  1124→  1124→        result.push(sorted[i]);
  1125→  1125→      } else {
  1126→  1126→        // Gap > 10% — try to find an intermediate level
  1127→  1127→        let found = false;
  1128→  1128→        for (let j = i + 1; j < sorted.length; j++) {
  1129→  1129→          const midPrice = sorted[j].price;
  1130→  1130→          const gapFromPrev = Math.abs(midPrice - prevPrice) / prevPrice;
  1131→  1131→          const gapToNext = Math.abs(currPrice - midPrice) / midPrice;
  1132→  1132→          if (gapFromPrev >= 0.05 && gapFromPrev <= 0.10) {
  1133→  1133→            result.push(sorted[j]);
  1134→  1134→            found = true;
  1135→  1135→            // Skip levels that are too close to the inserted one
  1136→  1136→            i = j;
  1137→  1137→            break;
  1138→  1138→          }
  1139→  1139→        }
  1140→  1140→        if (!found) {
  1141→  1141→          // No intermediate found — generate a synthetic level at ~7.5% from prev
  1142→  1142→          const synthGap = 0.075;
  1143→  1143→          const syntheticPrice = type === 'support'
  1144→  1144→            ? Math.round((prevPrice * (1 - synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice)
  1145→  1145→            : Math.round((prevPrice * (1 + synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice);
  1146→  1146→          // Make sure it's not too close to next level
  1147→  1147→          const gapToNext = Math.abs(currPrice - syntheticPrice) / syntheticPrice;
  1148→  1148→          if (gapToNext >= 0.05) {
  1149→  1149→            const synthetic: MergedSRLevel = {
  1150→  1150→              price: syntheticPrice,
  1151→  1151→              type,
  1152→  1152→              score: 2,
  1153→  1153→              strength: 2,
  1154→  1154→              overlapCount: 1,
  1155→  1155→              methods: ['Psychological'],
  1156→  1156→              fibRatio: '—',
  1157→  1157→              fibLabel: '',
  1158→  1158→            };
  1159→  1159→            result.push(synthetic);
  1160→  1160→          }
  1161→  1161→        }
  1162→  1162→      }
  1163→  1163→    }
  1164→  1164→
  1165→  1165→    return result;
  1166→  1166→  };
  1167→  1167→
  1168→  1168→  const finalSupports = filterAndSelectLevels(supports, 'support');
  1169→  1169→  const finalResistances = filterAndSelectLevels(resistances, 'resistance');
  1170→  1170→
  1171→  1171→  // ── Step 9: Build LevelStrength objects ────────────────────────────────
  1172→  1172→  const toLevelStrength = (m: MergedSRLevel): LevelStrength => {
  1173→  1173→    let grade: string;
  1174→  1174→    if (m.score >= 8.5) grade = 'Very Strong';
  1175→  1175→    else if (m.score >= 7) grade = 'Strong';
  1176→  1176→    else if (m.score >= 5) grade = 'Moderate';
  1177→  1177→    else grade = 'Weak';
  1178→  1178→
  1179→  1179→    return {
  1180→  1180→      price: m.price,
  1181→  1181→      strength: m.strength,
  1182→  1182→      score: m.score,
  1183→  1183→      grade,
  1184→  1184→      isTarget: m.score >= 7,
  1185→  1185→      overlapCount: m.overlapCount,
  1186→  1186→      methods: m.methods,
  1187→  1187→      fibRatio: m.fibRatio,
  1188→  1188→      fibLabel: m.fibLabel,
  1189→  1189→    };
  1190→  1190→  };
  1191→  1191→
  1192→  1192→  const supportStrengths = finalSupports.map(toLevelStrength);
  1193→  1193→  const resistanceStrengths = finalResistances.map(toLevelStrength);
  1194→  1194→
  1195→  1195→  // Price targets: all levels with isTarget=true
  1196→  1196→  const priceTargets = [...supportStrengths, ...resistanceStrengths].filter(l => l.isTarget);
  1197→  1197→
  1198→  1198→  return {
  1199→  1199→    resistances: resistanceStrengths.map(l => l.price),
  1200→  1200→    supports: supportStrengths.map(l => l.price),
  1201→  1201→    supportStrengths,
  1202→  1202→    resistanceStrengths,
  1203→  1203→    priceTargets,
  1204→  1204→  };
  1205→  1205→}

  1208→  1208→// ─── Trend Analysis ───────────────────────────────────────────────────────────
  1209→  1209→function calcTrend(closes: number[], period: number): TrendResult {
  1210→  1210→  const slice = closes.slice(-Math.min(period, closes.length));
  1211→  1211→  const { slope, r2 } = linearRegression(slice);
  1212→  1212→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
  1213→  1213→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
  1214→  1214→  const priceThreshold = avgPrice * 0.001; // 0.1%
  1215→  1215→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
  1216→  1216→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
  1217→  1217→}

  1221→  1221→// VDss 7-LAYER PROBABILITY ENGINE

  1223→  1223→
  1224→  1224→// ─── Helper: simplified S/R at a historical bar using swing levels ─────────────
  1225→  1225→function simplifiedSR(data: OHLCV[], barIdx: number): { R1: number; S1: number } {
  1226→  1226→  const slice = data.slice(0, barIdx + 1);
  1227→  1227→  const price = slice[slice.length - 1].close;
  1228→  1228→  let nearestR = price * 1.05;
  1229→  1229→  let nearestS = price * 0.95;
  1230→  1230→
  1231→  1231→  // Collect all swing highs (resistance) and swing lows (support)
  1232→  1232→  const resistances: number[] = [];
  1233→  1233→  const supports: number[] = [];
  1234→  1234→  for (const lb of [3, 5, 7]) {
  1235→  1235→    const swings = findSwingLevels(slice, lb);
  1236→  1236→    resistances.push(...swings.highs);
  1237→  1237→    supports.push(...swings.lows);
  1238→  1238→  }
  1239→  1239→
  1240→  1240→  // Find nearest resistance above price
  1241→  1241→  const abovePrice = resistances.filter(r => r > price).sort((a, b) => a - b);
  1242→  1242→  if (abovePrice.length > 0) nearestR = abovePrice[0];
  1243→  1243→
  1244→  1244→  // Find nearest support below price
  1245→  1245→  const belowPrice = supports.filter(s => s < price).sort((a, b) => b - a);
  1246→  1246→  if (belowPrice.length > 0) nearestS = belowPrice[0];
  1247→  1247→
  1248→  1248→  return { R1: nearestR, S1: nearestS };
  1249→  1249→}
  1250→  1250→
  1251→  1251→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
  1252→  1252→// Returns true if bullish divergence detected, false if bearish, null if none
  1253→  1253→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
  1254→  1254→  const lookback = 20;
  1255→  1255→  const start = Math.max(0, barIdx - lookback);
  1256→  1256→  const slice = data.slice(start, barIdx + 1);
  1257→  1257→  if (slice.length < 10) return null;
  1258→  1258→
  1259→  1259→  const mid = Math.floor(slice.length / 2);
  1260→  1260→  const firstHalf = slice.slice(0, mid);
  1261→  1261→  const secondHalf = slice.slice(mid);
  1262→  1262→
  1263→  1263→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
  1264→  1264→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
  1265→  1265→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
  1266→  1266→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
  1267→  1267→
  1268→  1268→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
  1269→  1269→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
  1270→  1270→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
  1271→  1271→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
  1272→  1272→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
  1273→  1273→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
  1274→  1274→
  1275→  1275→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
  1276→  1276→    return false; // bearish divergence
  1277→  1277→  }
  1278→  1278→
  1279→  1279→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
  1280→  1280→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
  1281→  1281→
  1282→  1282→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
  1283→  1283→    return true; // bullish divergence
  1284→  1284→  }
  1285→  1285→
  1286→  1286→  return null;
  1287→  1287→}
  1288→  1288→
  1289→  1289→// ═══════════════════════════════════════════════════════════════════════════════
  1290→  1290→// LAYER 1 & 2 — Feature Computation at a Historical Bar
  1291→  1291→// ═══════════════════════════════════════════════════════════════════════════════
  1292→  1292→
  1293→  1293→/**
  1294→  1294→ * Computes all 16 ML features at a specific historical bar.
  1295→  1295→ * Uses data[0..barIdx] (inclusive) for all indicator calculations.
  1296→  1296→ * Returns an object keyed by feature name matching FEATURES_ORDER.
  1297→  1297→ */
  1298→  1298→function computeFeaturesAtBar(data: OHLCV[], barIdx: number): Record<string, number> {
  1299→  1299→  // Need at least 30 bars for meaningful indicators
  1300→  1300→  if (barIdx < 30) {
  1301→  1301→    const empty: Record<string, number> = {};
  1302→  1302→    for (const key of FEATURES_ORDER) empty[key] = 0.5;
  1303→  1303→    return empty;
  1304→  1304→  }
  1305→  1305→
  1306→  1306→  const slice = data.slice(0, barIdx + 1);
  1307→  1307→  const closes = slice.map(d => d.close);
  1308→  1308→  const price = closes[closes.length - 1];
  1309→  1309→  const eps = 1e-10;
  1310→  1310→
  1311→  1311→  // ── Compute all base indicators ───────────────────────────────────────────
  1312→  1312→  const rsiVal = calcRSI(closes);
  1313→  1313→  const sliceHasVolume = slice.some(d => d.volume > 0);
  1314→  1314→  const mfiVal = sliceHasVolume ? calcMFI(slice) : 50; // neutral when no volume
  1315→  1315→  const cciVal = calcCCI(slice);
  1316→  1316→  const stoch = calcStochastic(slice);
  1317→  1317→  const macd = calcMACD(closes);
  1318→  1318→  const adxResult = calcADX(slice);
  1319→  1319→  const atrVal = calcATR(slice);
  1320→  1320→  const bb = calcBollingerBands(closes);
  1321→  1321→  const ma21 = sma(closes, 21);
  1322→  1322→  const ma100 = sma(closes, 100);
  1323→  1323→  const ema12 = emaCalc(closes, 12);
  1324→  1324→  const ema26 = emaCalc(closes, 26);
  1325→  1325→  const trendResult = calcTrend(closes, 21);
  1326→  1326→  const sr = simplifiedSR(data, barIdx);
  1327→  1327→
  1328→  1328→  // ── Layer 1: Raw Scores (s_*) ────────────────────────────────────────────
  1329→  1329→
  1330→  1330→  // s_rsi — RSI mapped to 0-1
  1331→  1331→  const s_rsi = rsiVal > 80 ? 0.95
  1332→  1332→    : rsiVal > 70 ? 0.85
  1333→  1333→    : rsiVal > 60 ? 0.70
  1334→  1334→    : rsiVal > 50 ? 0.55
  1335→  1335→    : rsiVal > 40 ? 0.45
  1336→  1336→    : rsiVal > 30 ? 0.30
  1337→  1337→    : rsiVal > 20 ? 0.15
  1338→  1338→    : 0.05;
  1339→  1339→
  1340→  1340→  // s_mfi — MFI mapped to 0-1 (same formula as RSI)
  1341→  1341→  const s_mfi = mfiVal > 80 ? 0.95
  1342→  1342→    : mfiVal > 70 ? 0.85
  1343→  1343→    : mfiVal > 60 ? 0.70
  1344→  1344→    : mfiVal > 50 ? 0.55
  1345→  1345→    : mfiVal > 40 ? 0.45
  1346→  1346→    : mfiVal > 30 ? 0.30
  1347→  1347→    : mfiVal > 20 ? 0.15
  1348→  1348→    : 0.05;
  1349→  1349→
  1350→  1350→  // s_cci — CCI mapped to 0-1
  1351→  1351→  const s_cci = cciVal > 200 ? 0.90
  1352→  1352→    : cciVal > 100 ? 0.75
  1353→  1353→    : cciVal > 0 ? 0.60
  1354→  1354→    : cciVal > -100 ? 0.40
  1355→  1355→    : cciVal > -200 ? 0.25
  1356→  1356→    : 0.10;
  1357→  1357→
  1358→  1358→  // s_adx — Directional bias
  1359→  1359→  const s_adx = (adxResult.diPlus - adxResult.diMinus + 100) / 200;
  1360→  1360→
  1361→  1361→  // s_macd — MACD histogram + signal combined
  1362→  1362→  const absMacdLine = Math.abs(macd.line) + 1;
  1363→  1363→  const s_macd = macd.line > macd.signal
  1364→  1364→    ? 0.5 + Math.min(0.5, macd.histogram / absMacdLine)
  1365→  1365→    : 0.5 - Math.min(0.5, Math.abs(macd.histogram) / absMacdLine);
  1366→  1366→
  1367→  1367→  // s_stoch — Stochastic %K mapped to 0-1
  1368→  1368→  const s_stoch = stoch.k / 100;
  1369→  1369→
  1370→  1370→  // s_bb — Bollinger Band position
  1371→  1371→  const bbRange = bb.upper - bb.lower;
  1372→  1372→  const s_bb = bbRange > eps
  1373→  1373→    ? clamp((price - bb.lower) / bbRange, 0, 1)
  1374→  1374→    : 0.5;
  1375→  1375→
  1376→  1376→  // s_ma21 — Price relative to 21-period SMA
  1377→  1377→  const s_ma21 = ma21 > eps
  1378→  1378→    ? clamp(0.5 + (price - ma21) / ma21, 0, 1)
  1379→  1379→    : 0.5;
  1380→  1380→
  1381→  1381→  // s_ma100 — Price relative to 100-period SMA
  1382→  1382→  const s_ma100 = ma100 > eps
  1383→  1383→    ? clamp(0.5 + (price - ma100) / ma100, 0, 1)
  1384→  1384→    : 0.5;
  1385→  1385→
  1386→  1386→  // s_ema — EMA12 vs EMA26 spread
  1387→  1387→  const s_ema = ema26 > eps
  1388→  1388→    ? clamp(0.5 + (ema12 - ema26) / ema26 * 5, 0, 1)
  1389→  1389→    : 0.5;
  1390→  1390→
  1391→  1391→  // s_atr — Inverse volatility (lower ATR = higher score)
  1392→  1392→  const s_atr = price > eps ? 1 - Math.min(atrVal / price * 10, 1) : 0.5;
  1393→  1393→
  1394→  1394→  // s_trend — R² + angle combination
  1395→  1395→  const s_trend = (trendResult.r2 * 0.7) +
  1396→  1396→    (Math.max(0, Math.min(1, (trendResult.angle + 45) / 90)) * 0.3);
  1397→  1397→
  1398→  1398→  // s_sr — Support/Resistance proximity
  1399→  1399→  const dR1 = (sr.R1 - price) / (price > eps ? price : 1);
  1400→  1400→  const dS1 = (price - sr.S1) / (price > eps ? price : 1);
  1401→  1401→  const s_sr = clamp(0.5 - dR1 * 1.5 + dS1 * 1.0, 0, 1);
  1402→  1402→
  1403→  1403→  // ── Layer 2: Corrections (f_*) ───────────────────────────────────────────
  1404→  1404→
  1405→  1405→  // f_rsi — RSI with momentum correction
  1406→  1406→  const rsiCurrent = rsiVal;
  1407→  1407→  const rsi5ago = barIdx >= 5
  1408→  1408→    ? calcRSI(closes.slice(0, barIdx - 5 + 1))
  1409→  1409→    : rsiCurrent;
  1410→  1410→  const f_rsi = clamp(s_rsi + (rsiCurrent - rsi5ago) / 100 * 0.08, 0, 1);
  1411→  1411→
  1412→  1412→  // f_mfi — MFI with momentum correction (skip when no volume)
  1413→  1413→  const mfiCurrent = mfiVal;
  1414→  1414→  const mfi5ago = (barIdx >= 5 && sliceHasVolume)
  1415→  1415→    ? calcMFI(data.slice(0, barIdx - 5 + 1))
  1416→  1416→    : mfiCurrent;
  1417→  1417→  const f_mfi = sliceHasVolume
  1418→  1418→    ? clamp(s_mfi + (mfiCurrent - mfi5ago) / 100 * 0.08, 0, 1)
  1419→  1419→    : 0.5; // neutral when no volume
  1420→  1420→
  1421→  1421→  // f_cci — CCI with momentum correction (scale: /200 instead of /100)
  1422→  1422→  const cciCurrent = cciVal;
  1423→  1423→  const cci5ago = barIdx >= 5
  1424→  1424→    ? calcCCI(data.slice(0, barIdx - 5 + 1))
  1425→  1425→    : cciCurrent;
  1426→  1426→  const f_cci = clamp(s_cci + (cciCurrent - cci5ago) / 200 * 0.08, 0, 1);
  1427→  1427→
  1428→  1428→  // f_macd — Same as s_macd (no additional correction)
  1429→  1429→  const f_macd = s_macd;
  1430→  1430→
  1431→  1431→  // f_stoch — Same as s_stoch
  1432→  1432→  const f_stoch = s_stoch;
  1433→  1433→
  1434→  1434→  // f_stochCross — Stochastic K/D crossover detection
  1435→  1435→  let f_stochCross = 0.1;
  1436→  1436→  if (barIdx >= 1) {
  1437→  1437→    const prevSlice = data.slice(0, barIdx);
  1438→  1438→    const prevStoch = calcStochastic(prevSlice);
  1439→  1439→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
  1440→  1440→      f_stochCross = 0.9; // K crossed above D (bullish)
  1441→  1441→    }
  1442→  1442→  }
  1443→  1443→
  1444→  1444→  // f_macdCross — MACD/Signal crossover detection
  1445→  1445→  let f_macdCross = 0.1;
  1446→  1446→  if (barIdx >= 1) {
  1447→  1447→    const prevCloses = closes.slice(0, -1);
  1448→  1448→    if (prevCloses.length >= 35) {
  1449→  1449→      const prevMacd = calcMACD(prevCloses);
  1450→  1450→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
  1451→  1451→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
  1452→  1452→      }
  1453→  1453→    }
  1454→  1454→  }
  1455→  1455→
  1456→  1456→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
  1457→  1457→  const divResult = detectDivergenceSimple(data, barIdx);
  1458→  1458→  const f_div = divResult === true ? 0.85 : 0.15;
  1459→  1459→
  1460→  1460→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
  1461→  1461→  return {
  1462→  1462→    rsi: f_rsi,
  1463→  1463→    mfi: f_mfi,
  1464→  1464→    cci: f_cci,
  1465→  1465→    adx: s_adx,
  1466→  1466→    macd: f_macd,
  1467→  1467→    stoch: f_stoch,
  1468→  1468→    bb: s_bb,
  1469→  1469→    ma21: s_ma21,
  1470→  1470→    ma100: s_ma100,
  1471→  1471→    ema: s_ema,
  1472→  1472→    atr: s_atr,
  1473→  1473→    trend: s_trend,
  1474→  1474→    sr: s_sr,
  1475→  1475→    stochCross: f_stochCross,
  1476→  1476→    macdCross: f_macdCross,
  1477→  1477→    div: f_div,
  1478→  1478→  };
  1479→  1479→}

  1482→  1482→// TRAINING DATA CREATION (Layer 0 — Data Preparation)
  1483→  1483→// ═══════════════════════════════════════════════════════════════════════════════
  1484→  1484→
  1485→  1485→/**
  1486→  1486→ * Creates supervised training data from historical OHLCV data.
  1487→  1487→ * For each bar, computes features and labels based on future price movement.
  1488→  1488→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
  1489→  1489→ * Samples every `sampleStep` bars to reduce computation.
  1490→  1490→ */
  1491→  1491→function createTrainingData(
  1492→  1492→  data: OHLCV[],
  1493→  1493→  horizon: number = 5,
  1494→  1494→  sampleStep: number = 5
  1495→  1495→): { X: number[][]; y: number[] } {
  1496→  1496→  const X: number[][] = [];
  1497→  1497→  const y: number[] = [];
  1498→  1498→
  1499→  1499→  // Start from horizon+30 to ensure enough data for indicators
  1500→  1500→  // End at data.length - horizon - 1 to have future bars for labeling
  1501→  1501→  // Sample every sampleStep bars for performance
  1502→  1502→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
  1503→  1503→    const features = computeFeaturesAtBar(data, i);
  1504→  1504→
  1505→  1505→    // Convert to array following FEATURES_ORDER
  1506→  1506→    const featureVec: number[] = [];
  1507→  1507→    let valid = true;
  1508→  1508→    for (const key of FEATURES_ORDER) {
  1509→  1509→      const val = features[key];
  1510→  1510→      if (val === undefined || val === null || Number.isNaN(val)) {
  1511→  1511→        valid = false;
  1512→  1512→        break;
  1513→  1513→      }
  1514→  1514→      featureVec.push(val);
  1515→  1515→    }
  1516→  1516→    if (!valid) continue;
  1517→  1517→
  1518→  1518→    // Label: 1 if future close > 1% above current close
  1519→  1519→    const currentClose = data[i].close;
  1520→  1520→    const futureClose = data[i + horizon].close;
  1521→  1521→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
  1522→  1522→
  1523→  1523→    X.push(featureVec);
  1524→  1524→    y.push(label);
  1525→  1525→  }
  1526→  1526→
  1527→  1527→  return { X, y };
  1528→  1528→}

  1530→  1530→// ═══════════════════════════════════════════════════════════════════════════════
  1531→  1531→// VDss GRAPH STRUCTURE (Layer 5 — State Machine)
  1532→  1532→// ═══════════════════════════════════════════════════════════════════════════════
  1533→  1533→
  1534→  1534→/**
  1535→  1535→ * VDss Graph: Each node represents a market state.
  1536→  1536→ * Edges represent transitions with types: 'up', 'down', 'pullback', 'risk', 'terminal'.
  1537→  1537→ * Terminal nodes map to scenario results (R1=strong bullish, R5=deep correction).
  1538→  1538→ */
  1539→  1539→const VDSS_GRAPH = {
  1540→  1540→  // outgoing edges from each node: [target, type]
  1541→  1541→  outgoing: {
  1542→  1542→    A: [['B', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1543→  1543→    B: [['C', 'up'], ['G', 'pullback'], ['A', 'risk']] as [string, string][],
  1544→  1544→    C: [['D', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1545→  1545→    D: [['E', 'up'], ['B', 'pullback'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1546→  1546→    E: [['F', 'up'], ['D', 'pullback'], ['G', 'pullback'], ['H', 'down'], ['R1', 'terminal']] as [string, string][],
  1547→  1547→    F: [['R1', 'terminal'], ['E', 'pullback'], ['D', 'down'], ['G', 'risk']] as [string, string][],
  1548→  1548→    G: [['B', 'up'], ['A', 'pullback'], ['H', 'down'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1549→  1549→    H: [['G', 'pullback'], ['I', 'down'], ['B', 'up'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1550→  1550→    I: [['H', 'pullback'], ['J', 'down'], ['G', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1551→  1551→    J: [['I', 'pullback'], ['K', 'down'], ['H', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1552→  1552→    K: [['J', 'pullback'], ['I', 'up'], ['L', 'risk'], ['R4', 'terminal'], ['R5', 'terminal']] as [string, string][],
  1553→  1553→    L: [['C', 'up'], ['H', 'pullback'], ['J', 'down'], ['K', 'risk'], ['R5', 'terminal']] as [string, string][],
  1554→  1554→  },
  1555→  1555→  // which result scenario each terminal node maps to
  1556→  1556→  terminalMap: { R1: 'R1', R2: 'R2', R3: 'R3', R4: 'R4', R5: 'R5' } as Record<string, string>,
  1557→  1557→  startNode: 'A' as string,
  1558→  1558→};
  1559→  1559→
  1560→  1560→// ═══════════════════════════════════════════════════════════════════════════════
  1561→  1561→// LAYER 5 — Edge Weight Calculation
  1562→  1562→// ═══════════════════════════════════════════════════════════════════════════════
  1563→  1563→
  1564→  1564→/**
  1565→  1565→ * Calculates transition edge weights for the VDss graph.
  1566→  1566→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
  1567→  1567→ * to determine the relative probability of up/down/pullback/risk transitions.
  1568→  1568→ */
  1569→  1569→function calculateEdgeWeights(
  1570→  1570→  bullConsensus: number,
  1571→  1571→  adx: number,
  1572→  1572→  mlModel: AdaptiveWeightModel
  1573→  1573→): { up: number; down: number; pullback: number; risk: number } {
  1574→  1574→  // Extract ML coefficient insights if model is trained
  1575→  1575→  let trend_coef = 1 / 3;
  1576→  1576→  let momentum_coef = 1 / 3;
  1577→  1577→  let volatility_coef = 1 / 3;
  1578→  1578→
  1579→  1579→  if (mlModel.isTrained && mlModel.weights) {
  1580→  1580→    const coefs = mlModel.getCoefficients();
  1581→  1581→    // Map feature indices to coefficient categories
  1582→  1582→    // FEATURES_ORDER: ['rsi','mfi','cci','adx','macd','stoch','bb','ma21','ma100','ema','atr','trend','sr','stochCross','macdCross','div']
  1583→  1583→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER
  1584→  1584→    const rsiIdx = 0;     // 'rsi' in FEATURES_ORDER
  1585→  1585→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
  1586→  1586→
  1587→  1587→    const absTrend = Math.abs(coefs[trendIdx] ?? 0);
  1588→  1588→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
  1589→  1589→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
  1590→  1590→
  1591→  1591→    const coefSum = absTrend + absRsi + absAtr;
  1592→  1592→    if (coefSum > 1e-10) {
  1593→  1593→      trend_coef = absTrend / coefSum;
  1594→  1594→      momentum_coef = absRsi / coefSum;
  1595→  1595→      volatility_coef = absAtr / coefSum;
  1596→  1596→    }
  1597→  1597→  }
  1598→  1598→
  1599→  1599→  // Base weights driven by bull consensus
  1600→  1600→  const up_base = bullConsensus * 0.7 + 0.15;
  1601→  1601→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
  1602→  1602→  const pullback_base = 0.25;
  1603→  1603→  const risk_base = 0.12 * (1.2 - adx / 100);
  1604→  1604→
  1605→  1605→  // Apply ML coefficient adjustments
  1606→  1606→  const up = clamp(
  1607→  1607→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
  1608→  1608→    0.05, 0.95
  1609→  1609→  );
  1610→  1610→  const down = clamp(
  1611→  1611→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
  1612→  1612→    0.05, 0.95
  1613→  1613→  );
  1614→  1614→  const pullback = clamp(
  1615→  1615→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
  1616→  1616→    0.05, 0.50
  1617→  1617→  );
  1618→  1618→  const risk = clamp(
  1619→  1619→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
  1620→  1620→    0.02, 0.30
  1621→  1621→  );
  1622→  1622→
  1623→  1623→  return { up, down, pullback, risk };
  1624→  1624→}
  1625→  1625→
  1626→  1626→// ═══════════════════════════════════════════════════════════════════════════════
  1627→  1627→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)
  1628→  1628→// ═══════════════════════════════════════════════════════════════════════════════
  1629→  1629→
  1630→  1630→/**
  1631→  1631→ * Performs iterative DFS from the start node through the VDss graph,
  1632→  1632→ * computing raw path probabilities, then calibrating them against
  1633→  1633→ * Layer 4 scenario probabilities.
  1634→  1634→ *
  1635→  1635→ * Returns:
  1636→  1636→ * - calibratedPaths: Map<pathString, probability>
  1637→  1637→ * - calibrationFactors: per-scenario multiplier used
  1638→  1638→ * - scenarioSums: raw probability sums per scenario before calibration
  1639→  1639→ */
  1640→  1640→function calculatePathProbabilities(
  1641→  1641→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
  1642→  1642→  scenarioProbs: Record<string, number>
  1643→  1643→): {
  1644→  1644→  calibratedPaths: Map<string, number>;
  1645→  1645→  calibrationFactors: Record<string, number>;
  1646→  1646→  scenarioSums: Record<string, number>;
  1647→  1647→} {
  1648→  1648→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
  1649→  1649→  const MAX_PATHS = 200;
  1650→  1650→
  1651→  1651→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
  1652→  1652→  type StackEntry = [string, string, number];
  1653→  1653→
  1654→  1654→  // Collect all terminal paths with raw probabilities
  1655→  1655→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
  1656→  1656→
  1657→  1657→  // Iterative DFS
  1658→  1658→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
  1659→  1659→
  1660→  1660→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
  1661→  1661→    const [node, pathStr, pathProb] = stack.pop()!;
  1662→  1662→
  1663→  1663→    const edges = outgoing[node as keyof typeof outgoing];
  1664→  1664→    if (!edges) continue;
  1665→  1665→
  1666→  1666→    // Normalize outgoing edge weights to probabilities for this node
  1667→  1667→    // Map edge types to their weights
  1668→  1668→    let totalWeight = 0;
  1669→  1669→    const edgeWeightsList: number[] = [];
  1670→  1670→    for (const [, edgeType] of edges) {
  1671→  1671→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
  1672→  1672→      edgeWeightsList.push(w);
  1673→  1673→      totalWeight += w;
  1674→  1674→    }
  1675→  1675→
  1676→  1676→    if (totalWeight < 1e-10) continue;
  1677→  1677→
  1678→  1678→    // Expand edges
  1679→  1679→    for (let i = 0; i < edges.length; i++) {
  1680→  1680→      const [target, edgeType] = edges[i];
  1681→  1681→      const edgeProb = edgeWeightsList[i] / totalWeight;
  1682→  1682→      const newProb = pathProb * edgeProb;
  1683→  1683→      const newPath = pathStr + '→' + target;
  1684→  1684→
  1685→  1685→      if (edgeType === 'terminal') {
  1686→  1686→        // Reached a terminal node
  1687→  1687→        const scenario = terminalMap[target] ?? 'R3';
  1688→  1688→        rawPaths.push({ path: newPath, scenario, prob: newProb });
  1689→  1689→      } else {
  1690→  1690→        // Continue DFS
  1691→  1691→        stack.push([target, newPath, newProb]);
  1692→  1692→      }
  1693→  1693→    }
  1694→  1694→  }
  1695→  1695→
  1696→  1696→  // Group raw probabilities by scenario
  1697→  1697→  const rawSums: Record<string, number> = { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0 };
  1698→  1698→  for (const rp of rawPaths) {
  1699→  1699→    rawSums[rp.scenario] = (rawSums[rp.scenario] ?? 0) + rp.prob;
  1700→  1700→  }
  1701→  1701→
  1702→  1702→  // Compute calibration factors: scenarioProbs[scenario] / rawSum[scenario]
  1703→  1703→  const calibrationFactors: Record<string, number> = {};
  1704→  1704→  for (const scenario of ['R1', 'R2', 'R3', 'R4', 'R5']) {
  1705→  1705→    const rawSum = rawSums[scenario] ?? 0;
  1706→  1706→    const targetProb = (scenarioProbs[scenario] ?? 20) / 100; // convert from percentage
  1707→  1707→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
  1708→  1708→  }
  1709→  1709→
  1710→  1710→  // Apply calibration to each path
  1711→  1711→  const calibratedPaths = new Map<string, number>();
  1712→  1712→  for (const rp of rawPaths) {
  1713→  1713→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
  1714→  1714→    calibratedPaths.set(rp.path, rp.prob * factor);
  1715→  1715→  }
  1716→  1716→
  1717→  1717→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
  1718→  1718→}
  1719→  1719→
  1720→  1720→// ═══════════════════════════════════════════════════════════════════════════════
  1721→  1721→// LAYER 4 — Scenario Probability Calculation (Adaptive with ML)
  1722→  1722→// ═══════════════════════════════════════════════════════════════════════════════
  1723→  1723→
  1724→  1724→/**
  1725→  1725→ * Calculates the 5 VDss scenario probabilities (R1-R5) that sum to 100.
  1726→  1726→ * Uses bull/bear consensus, key indicator values, S/R levels, and
  1727→  1727→ * ML-derived adaptive factors for fine-tuning.
  1728→  1728→ */
  1729→  1729→function calculateScenarioProbabilities(
  1730→  1730→  bullConsensus: number,
  1731→  1731→  price: number,
  1732→  1732→  R1: number,
  1733→  1733→  S1: number,
  1734→  1734→  MA100: number,
  1735→  1735→  rsi: number,
  1736→  1736→  mfi: number,
  1737→  1737→  stochK: number,
  1738→  1738→  mlModel: AdaptiveWeightModel
  1739→  1739→): { R1: number; R2: number; R3: number; R4: number; R5: number; factors: { momentum: number; volatility: number; trend: number } } {
  1740→  1740→  // Distance metrics to key levels (spec uses exp(-3 * ...))
  1741→  1741→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0;
  1742→  1742→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0;
  1743→  1743→
  1744→  1744→  // Binary risk flags (per spec) — MFI only used when volume is available
  1745→  1745→  const overboughtRisk = (rsi > 70 || (mfi > 80 && mfi !== 50) || stochK > 80) ? 1 : 0;
  1746→  1746→  const oversoldBounce = (rsi < 30 || (mfi < 20 && mfi !== 50) || stochK < 20) ? 1 : 0;
  1747→  1747→
  1748→  1748→  // Extract adaptive factors from ML model if trained
  1749→  1749→  let momentum = 0.7;
  1750→  1750→  let volatility = 0.5;
  1751→  1751→  let trend = 0.6;
  1752→  1752→
  1753→  1753→  if (mlModel.isTrained && mlModel.weights) {
  1754→  1754→    const coefs = mlModel.getCoefficients();
  1755→  1755→    momentum = sigmoid(coefs[0] ?? 0);   // rsi coefficient → momentum factor
  1756→  1756→    volatility = sigmoid(coefs[10] ?? 0); // atr coefficient → volatility factor
  1757→  1757→    trend = sigmoid(coefs[11] ?? 0);     // trend coefficient → trend factor
  1758→  1758→  }
  1759→  1759→
  1760→  1760→  // Below MA100 flag
  1761→  1761→  const belowMA100 = price < MA100 ? 1 : 0;
  1762→  1762→
  1763→  1763→  // ── Raw scenario probabilities (matching user spec) ─────────────────────
  1764→  1764→
  1765→  1765→  // R1: Strong Bullish — aggressive breakout
  1766→  1766→  const raw_R1 =
  1767→  1767→    bullConsensus * distR1 * (1 - overboughtRisk * momentum * 0.7) +
  1768→  1768→    oversoldBounce * 0.3 * distR1 * momentum;
  1769→  1769→
  1770→  1770→  // R2: Gradual Uptrend — steady rise
  1771→  1771→  const raw_R2 =
  1772→  1772→    ((1 - Math.abs(bullConsensus - 0.65)) * 0.8 +
  1773→  1773→    (bullConsensus > 0.4 ? 0.2 : 0)) * trend;
  1774→  1774→
  1775→  1775→  // R3: Range-bound
  1776→  1776→  const raw_R3 =
  1777→  1777→    (1 - Math.abs(bullConsensus - 0.5)) * 0.6 * (1 + (1 - volatility) * 0.2);
  1778→  1778→
  1779→  1779→  // R4: Moderate Correction
  1780→  1780→  const raw_R4 =
  1781→  1781→    ((1 - bullConsensus) * distS1 * 0.7 +
  1782→  1782→    overboughtRisk * 0.5 * momentum) *
  1783→  1783→    (0.8 + 0.2 * (1 - trend));
  1784→  1784→
  1785→  1785→  // R5: Deep Correction
  1786→  1786→  const raw_R5 =
  1787→  1787→    ((1 - bullConsensus) ** 2 * 0.5 + belowMA100 * 0.2) *
  1788→  1788→    (1 + (1 - trend) * 0.3);
  1789→  1789→
  1790→  1790→  // ── Normalize to sum = 100 ───────────────────────────────────────────
  1791→  1791→  const sum = raw_R1 + raw_R2 + raw_R3 + raw_R4 + raw_R5;
  1792→  1792→  const pR1 = sum > 0 ? Math.round(raw_R1 / sum * 100) : 20;
  1793→  1793→  const pR2 = sum > 0 ? Math.round(raw_R2 / sum * 100) : 20;
  1794→  1794→  const pR3 = sum > 0 ? Math.round(raw_R3 / sum * 100) : 20;
  1795→  1795→  const pR4 = sum > 0 ? Math.round(raw_R4 / sum * 100) : 20;
  1796→  1796→  const pR5 = 100 - (pR1 + pR2 + pR3 + pR4);
  1797→  1797→
  1798→  1798→  return {
  1799→  1799→    R1: pR1, R2: pR2, R3: pR3, R4: pR4, R5: pR5,
  1800→  1800→    factors: { momentum, volatility, trend },
  1801→  1801→  };
  1802→  1802→}

  1847→  1847→    ema26: emaCalc(closes, 26),
  1848→  1848→  };
  1849→  1849→
