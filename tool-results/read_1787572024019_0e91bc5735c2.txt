  1149→function scoreSchool(s: SchoolDef, inp: NarrativeInput): number {
  1150→  let s_ = 0;
  1151→  const { trendDirection, rsi, adx, bbPosition, diPlus, diMinus } = inp;
  1152→  switch (s.id) {
  1153→    case 'trend': s_ = adx > 25 ? 8 : adx > 15 ? 5 : 2; break;
  1154→    case 'classical': s_ = (inp.nearResistance || inp.nearSupport) ? 9 : 5; break;
  1155→    case 'candlestick': s_ = (inp.nearResistance || inp.nearSupport) ? 8 : 4; break;
  1156→    case 'oscillator': s_ = (rsi > 70 || rsi < 30) ? 9 : Math.abs(rsi - 50) > 15 ? 6 : 3; break;
  1157→    case 'harmonic': s_ = (inp.nearResistance || inp.nearSupport) ? 7 : 3; break;
  1158→    case 'elliott': s_ = inp.trendR2 > 0.7 ? 9 : inp.trendR2 > 0.4 ? 5 : 2; break;
  1159→    case 'volume': s_ = inp.hasVolume ? 8 : 2; break;
  1160→    case 'sr': s_ = (inp.nearResistance || inp.nearSupport) ? 9 : 4; break;
  1161→    case 'volatility': s_ = inp.atr / inp.price > 0.03 ? 8 : inp.atr / inp.price > 0.01 ? 5 : 2; break;
  1162→    case 'hybrid': s_ = 6; break;
  1163→    default: s_ = 4;
  1164→  }
  1165→  if (s.id === 'trend' && trendDirection === 'up') s_ += 1;
  1166→  if (s.id === 'trend' && diPlus > diMinus) s_ += 1;
  1167→  return s_;
  1168→}

  1264→/** Simple deterministic hash for date+instrument rotation */
  1265→function simpleHash(str: string): number {
  1266→  let hash = 0;
  1267→  for (let i = 0; i < str.length; i++) {
  1268→    const ch = str.charCodeAt(i);
  1269→    hash = ((hash << 5) - hash) + ch;
  1270→    hash |= 0;
  1271→  }
  1272→  return Math.abs(hash);
  1273→}

  1275→/**
  1276→ * Selects a V12 persona, template format, and variation seed
  1277→ * based on deterministic hash of date + instrument + school.
  1278→ * This ensures: different persona per day, different per instrument,
  1279→ * and school affinity matching.
  1280→ */
  1281→export function selectV12Persona(
  1282→  instrument: string,
  1283→  date: string,
  1284→  schoolId: string
  1285→): { persona: V12Persona; template: V12TemplateFormat; variationSeed: number } {
  1286→  const hashInput = `${date}:${instrument}:v10:2024`;
  1287→  const dayHash = simpleHash(hashInput);
  1288→
  1289→  // Step 1: Find personas matching school affinity
  1290→  const matching = PERSONAS.filter(p => p.schoolAffinity === schoolId);
  1291→  // Step 2: If no match, use all personas
  1292→  const pool = matching.length > 0 ? matching : PERSONAS;
  1293→  // Step 3: Select persona by hash
  1294→  const personaIndex = dayHash % pool.length;
  1295→  const persona = pool[personaIndex];
  1296→
  1297→  // Step 4: Select template — vary by hash % 7
  1298→  let templateIndex: number;
  1299→  const templateMod = dayHash % 7;
  1300→  if (templateMod < 2) templateIndex = 1; // technical (most common)
  1301→  else if (templateMod < 3) templateIndex = 0; // executive
  1302→  else if (templateMod < 4) templateIndex = 3; // forecast
  1303→  else if (templateMod < 5) templateIndex = 4; // educational
  1304→  else templateIndex = 2; // trading
  1305→  // Occasional variation: hash % 5 == 0 → use a different template
  1306→  if (dayHash % 5 === 0) {
  1307→    templateIndex = dayHash % TEMPLATE_FORMATS.length;
  1308→  }
  1309→  const template = TEMPLATE_FORMATS[templateIndex];
  1310→
  1311→  return {
  1312→    persona,
  1313→    template,
  1314→    variationSeed: dayHash,
  1315→  };
  1316→}

  1322→/**
  1323→ * Selects N phrases from a section of the phrase bank, rotated by seed.
  1324→ * Returns a comma-separated string of selected phrases.

  1326→function selectPhrases(section: string, subsection: string, count: number, seed: number): string {
  1327→  const sectionData = PHRASE_BANK[section];
  1328→  if (!sectionData) return '';
  1329→  const phrases = sectionData[subsection] || sectionData[Object.keys(sectionData)[0]] || [];
  1330→  if (phrases.length === 0) return '';
  1331→  const startIdx = seed % phrases.length;
  1332→  const selected: string[] = [];
  1333→  for (let i = 0; i < Math.min(count, phrases.length); i++) {
  1334→    selected.push(phrases[(startIdx + i) % phrases.length]);
  1335→  }
  1336→  return selected.join('\n- ');
  1337→}
