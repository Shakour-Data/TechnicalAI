    99→export default function VdesAnalysis(props: VdesAnalysisProps) {
   100→  const {
   101→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
   102→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
   103→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
   104→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
   105→    supportStrengths, resistanceStrengths, priceTargets,
   106→    isIndex, isTgju,
   107→  } = props;
   108→
   109→  // ── Instrument Context (for intelligent naming) ─────────────────
   110→  const instrumentLabel = useMemo(() => {
   111→    if (isIndex) return { name: `شاخص ${symbolName}`, type: 'index' as const, verb: 'قرار دارد' };
   112→    if (isTgju) return { name: symbolName, type: 'tgju' as const, verb: 'قرار دارد' };
   113→    return { name: `سهم ${symbolName}`, type: 'stock' as const, verb: 'قرار دارد' };
   114→  }, [symbolName, isIndex, isTgju]);
   115→
   116→  const IL = instrumentLabel;
   117→
   118→  const containerRef = useRef<HTMLDivElement>(null);
   119→  const textAnalysisRef = useRef<HTMLDivElement>(null);
   120→  const [exporting, setExporting] = useState(false);
   121→
   122→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   123→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   124→  const S1_level = supports[0] ?? currentPrice * 0.95;
   125→  const S2_level = supports[1] ?? currentPrice * 0.90;
   126→  const S3_level = supports[2] ?? currentPrice * 0.85;
   127→
   128→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   129→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   130→
   131→  // ── Dominant scenario ──────────────────────────────────────────
   132→  let highestKey = 'R3';
   133→  let highestProb = 0;
   134→  for (const key of SCENARIO_KEYS) {
   135→    if (scenarios[key].probability > highestProb) {
   136→      highestProb = scenarios[key].probability;
   137→      highestKey = key;
   138→    }
   139→  }
   140→  const strategy = STRATEGY_MAP[highestKey];
   141→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   142→
   143→  // ── Text variety: hash-based selector for varied language ──────────
   144→  const varietySeed = useMemo(() => {
   145→    const raw = `${rsi}${macdHist}${cci}${adx}${stochK}${trendAngle}${Math.round(currentPrice)}`;
   146→    let hash = 0;
   147→    for (let i = 0; i < raw.length; i++) hash = ((hash << 5) - hash + raw.charCodeAt(i)) | 0;
   148→    return Math.abs(hash);
   149→  }, [rsi, macdHist, cci, adx, stochK, trendAngle, currentPrice]);
   150→
   151→  // ── RSI signal ─────────────────────────────────────────────────
   152→  const rsiZone = rsi > 70 ? 'overbought' : rsi > 60 ? 'weak_ob' : rsi > 40 ? 'neutral' : rsi > 30 ? 'weak_os' : 'oversold';
   153→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   154→  const rsiColor = rsi > 70 ? 'text-red-400' : rsi > 60 ? 'text-amber-400' : rsi > 40 ? 'text-gray-300' : rsi > 30 ? 'text-amber-400' : 'text-emerald-400';
   155→
   156→  // ── Stochastic signal ──────────────────────────────────────────
   157→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   158→
   159→  // ── Trend text ─────────────────────────────────────────────────
   160→  const isUp = trendDirection === 'up';
   161→  const isDown = trendDirection === 'down';
   162→  const trendLabel = isUp ? 'صعودی' : isDown ? 'نزولی' : 'خنثی';
   163→  const trendColorCls = isUp ? 'text-emerald-400' : isDown ? 'text-red-400' : 'text-amber-400';
   164→  const adxStrength = adx > 40 ? 'بسیار قوی' : adx > 25 ? 'قوی' : adx > 15 ? 'متوسط' : 'ضعیف';
   165→  const adxColor = adx > 40 ? 'text-emerald-400' : adx > 25 ? 'text-amber-400' : 'text-gray-500';
   166→  const r2Pct = (trendR2 * 100).toFixed(1);
   167→
   168→  // ── Bollinger position ─────────────────────────────────────────
   169→  const bbRange = bollingerUpper - bollingerLower;
   170→  const bbPos = bbRange > 0 ? toPersianDigits(((currentPrice - bollingerLower) / bbRange * 100).toFixed(0)) : '۵۰';
   171→  const bbAbove = currentPrice > bollingerUpper;
   172→  const bbBelow = currentPrice < bollingerLower;
   173→  const bbSignal = bbAbove ? 'بالای باند بالایی' : bbBelow ? 'زیر باند پایینی' : `داخل باندها (${bbPos}٪)`;
   174→
   175→  // ── Distance to MA ─────────────────────────────────────────────
   176→  const distMa21 = ((currentPrice - ma21) / ma21 * 100).toFixed(1);
   177→  const distMa100 = ((currentPrice - ma100) / ma100 * 100).toFixed(1);
   178→
   179→  // ── TradingView scenario colors ────────────────────────────────
   180→  const chartCandles = useMemo(() => {
   181→    if (candles.length <= 2000) return candles;
   182→    return candles.slice(-2000);
   183→  }, [candles]);
   184→
   185→  const tvScenarios = useMemo(() => {
   186→    const colors: Record<string, string> = { R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62' };
   187→    return Object.fromEntries(
   188→      SCENARIO_KEYS.map(k => [k, { targetMin: scenarios[k].targetMin, targetMax: scenarios[k].targetMax, name: scenarios[k].name, probability: scenarios[k].probability, color: colors[k] }])
   189→    );
   190→  }, [scenarios]);
   191→
   192→  // ── Strategy recommendation text (intelligent) ────────────────────
   193→  const mfiWarning = hasVolume ? ` (مقدار ${toFa(mfi)})` : '';
   194→  const strategyText = useMemo(() => {
   195→    const v = varietySeed % 3; // 3 variants per signal type
   196→    if (rsi > 70 || (hasVolume && mfi > 80)) {
   197→      const variants = [
   198→        `هشدار اشباع خرید فعال است (RSI: ${toFa(rsi)}${mfiWarning}). فاصله تا مقاومت اول ${toFa(R1_level)} ریال است. بهترین استراتژی در این شرایط، صبر برای اصلاح قیمت تا محدوده حمایت ${toFa(S1_level)} تا ${toFa(S2_level)} و ورود در نقطه بهینه‌تر می‌باشد.`,
   199→        `با توجه به ورود به منطقه اشباع خرید (RSI=${toFa(rsi)}${mfiWarning}) و نزدیکی به مقاومت ${toFa(R1_level)}، ریسک اصلاح کوتاه‌مدت بالا ارزیابی می‌شود. پیشنهاد: انتظار برای بازگشت قیمت به ${toFa(S1_level)}-${toFa(S2_level)} و سپس بررسی مجدد سیگنال‌ها.`,
   200→        `${IL.name} در محدوده اشباع خرید (RSI: ${toFa(rsi)}${mfiWarning}) ${IL.verb}. با در نظر گرفتن مقاومت ${toFa(R1_level)} در بالا و حمایت ${toFa(S1_level)} در پایین، ورود جدید در این نقطه توصیه نمی‌شود. بهتر است منتظر عقبنشینی به ${toFa(S2_level)} بمانید.`,
   201→      ];
   202→      return variants[v];
   203→    }
   204→    if (rsi < 30 || (hasVolume && mfi < 20)) {
   205→      const variants = [
   206→        `اشباع فروش قابل توجه (RSI: ${toFa(rsi)}${mfiWarning}) در محدوده حمایت ${toFa(S1_level)} شناسایی شد. این شرایط می‌تواند فرصت مناسبی برای ورود باشد، با حد ضرر ${toFa(S2_level)}. هدف اول ${toFa(R1_level)} و هدف دوم ${toFa(R2_level)} ریال.`,
   207→        `${IL.name} به منطقه اشباع فروش رسیده (RSI=${toFa(rsi)}${mfiWarning}). فاصله تا حمایت کلیدی ${toFa(S1_level)} کم است. اگر واکنش مثبت به این حمایت مشاهده شود، بازگشت به ${toFa(R1_level)} محتمل خواهد بود.`,
   208→        `شاخص قدرت نسبی در محدوده اشباع فروش (مقدار ${toFa(rsi)}${mfiWarning}) ${IL.verb}. نزدیکی به حمایت ${toFa(S1_level)} و سیگنال فروش بیش از حد، احتمال بازگشت کوتاه‌مدت را تقویت می‌کند. حد ضرر پیشنهادی: ${toFa(S2_level)}.`,
   209→      ];
   210→      return variants[v];
   211→    }
   212→    const variants = [
   213→      `سیگنال‌ها در محدوده خنثی قرار دارند (RSI: ${toFa(rsi)}، قدرت روند: ${adxStrength}). محدوده رنج فعلی بین ${toFa(S1_level)} و ${toFa(R1_level)} است. تا خروج قیمت از این کانال، بهترین اقدام نظاره و عدم ورود است.`,
   214→      `وضعیت فعلی ${IL.name} نشان‌دهنده عدم قطعیت (RSI=${toFa(rsi)}، ADX=${toFa(adx)}). ${IL.name} در بازه ${toFa(S1_level)} تا ${toFa(R1_level)} در نوسان است. شکست معتبر یکی از این سطوح، مسیر آینده را مشخص خواهد کرد.`,
   215→      `شاخص‌ها هم‌جهت نیستند (RSI: ${toFa(rsi)}) و قدرت روند ${adxStrength} ارزیابی می‌شود. تا زمانی که ${IL.name} از محدوده ${toFa(S1_level)}-${toFa(R1_level)} خارج نشود، ورود با ریسک بالا همراه خواهد بود.`,
   216→    ];
   217→    return variants[v];
   218→  }, [rsi, mfi, hasVolume, R1_level, R2_level, S1_level, S2_level, varietySeed, IL.name, IL.verb]);
   219→
   220→  // ═══════════════════════════════════════════════════════════════════
   221→  // Export handlers
   222→  // ═══════════════════════════════════════════════════════════════════
   223→
   224→  // ── Helper: capture container as high-res image (hides toolbar) ──
   225→  const captureContainer = useCallback(async (): Promise<string | null> => {
   226→    const el = containerRef.current;
   227→    if (!el) return null;
   228→    // Hide export buttons before capturing
   229→    const toolbar = el.querySelector('[data-export-toolbar]') as HTMLElement | null;
   230→    if (toolbar) toolbar.style.visibility = 'hidden';
   231→    try {
   232→      const { toPng } = await import('html-to-image');
   233→      return await toPng(el, { backgroundColor: '#060a13', pixelRatio: 2, quality: 0.95 });
   234→    } finally {
   235→      if (toolbar) toolbar.style.visibility = '';
   236→    }
   237→  }, []);
   238→
   239→  // ── Export PDF (full page) ─────────────────────────────────────
   240→  const handleExportPDF = useCallback(async () => {
   241→    setExporting(true);
   242→    try {
   243→      const dataUrl = await captureContainer();
   244→      if (!dataUrl) return;
   245→      const { jsPDF } = await import('jspdf');
   246→      const img = new Image();
   247→      img.src = dataUrl;
   248→      await new Promise((resolve) => { img.onload = resolve; });
   249→      const pdfWidth = 595.28;
   250→      const pdfHeight = (img.height * pdfWidth) / img.width;
   251→      const pdf = new jsPDF({ orientation: pdfHeight > pdfWidth ? 'portrait' : 'landscape', unit: 'pt', format: [pdfWidth, Math.min(pdfHeight, 841.89)] });
   252→      pdf.addImage(dataUrl, 'PNG', 0, 0, pdfWidth, pdfHeight);
   253→      pdf.save(`${IL.name}-تحلیل-تکنیکال.pdf`);
   254→    } catch (err) { console.error('PDF export failed:', err); }
   255→    finally { setExporting(false); }
   256→  }, [IL.name, captureContainer]);
   257→

   259→  const handleExportHTML = useCallback(async () => {
   260→    setExporting(true);
   261→    try {
   262→      const dataUrl = await captureContainer();
   263→      if (!dataUrl) return;
   264→      const now = new Date();
   265→      const dateStr = toShamsi(now, 'long');
   266→      const base64 = dataUrl.replace(/^data:image\/png;base64,/, '');
   267→      const html = `<!DOCTYPE html>
   268→<html lang="fa" dir="rtl">
   269→<head>
   270→<meta charset="UTF-8">
   271→<meta name="viewport" content="width=device-width, initial-scale=1.0">
   272→<title>تحلیل تکنیکال ${IL.name}</title>
   273→<style>
   274→  * { margin: 0; padding: 0; box-sizing: border-box; }
   275→  body {
   276→    background: #060a13;
   277→    display: flex;
   278→    justify-content: center;
   279→    padding: 20px;
   280→    min-height: 100vh;
   281→  }
   282→  .page-wrapper {
   283→    width: 100%;
   284→    max-width: 960px;
   285→  }
   286→  .page-wrapper img {
   287→    width: 100%;
   288→    height: auto;
   289→    display: block;
   290→    border-radius: 16px;
   291→  }
   292→  .footer {
   293→    text-align: center;
   294→    font-family: Tahoma, Arial, sans-serif;
   295→    font-size: 11px;
   296→    color: #4b5563;
   297→    margin-top: 16px;
   298→    padding-top: 12px;
   299→    border-top: 1px solid rgba(255,255,255,0.05);
   300→  }
   301→  @media print {
   302→    body { padding: 0; background: white; }
   303→    .footer { color: #999; }
   304→  }
   305→</style>
   306→</head>
   307→<body>
   308→<div class="page-wrapper">
   309→  <img src="data:image/png;base64,${base64}" alt="تحلیل تکنیکال ${IL.name}" />
   310→  <div class="footer">تاریخ تهیه: ${dateStr} — صرفاً جنبه تحلیلی دارد و توصیه سرمایه‌گذاری نیست.</div>
   311→</div>
   312→</body>
   313→</html>`;
   314→      const blob = new Blob([html], { type: 'text/html;charset=utf-8' });
   315→      const url = URL.createObjectURL(blob);
   316→      const a = document.createElement('a'); a.href = url; a.download = `${IL.name}-تحلیل-تکنیکال.html`; a.click();
   317→      URL.revokeObjectURL(url);
   318→    } catch (err) { console.error('HTML export failed:', err); }
   319→    finally { setExporting(false); }
   320→  }, [IL.name, captureContainer]);
   321→
   322→  // ── Export Plain Text ──────────────────────────────────────────
   323→  const handleExportText = useCallback(() => {
   324→    const el = textAnalysisRef.current;
   325→    if (!el) return;
   326→    setExporting(true);
   327→    try {
   328→      const now = new Date();
   329→      const dateStr = toShamsi(now, 'long');
   330→      // Extract text content, clean up whitespace
   331→      const rawText = el.innerText || el.textContent || '';
   332→      const lines = rawText.split('\n').map(l => l.trim()).filter(Boolean);
   333→      const text = `تحلیل تکنیکال ${IL.name}
   334→${'═'.repeat(50)}
   335→تاریخ تهیه: ${dateStr}
   336→
   337→${lines.join('\n')}
   338→
   339→${'─'.repeat(50)}
   340→صرفاً جنبه تحلیلی دارد و توصیه سرمایه‌گذاری نیست.`;
   341→      const blob = new Blob([`\uFEFF${text}`], { type: 'text/plain;charset=utf-8' });
   342→      const url = URL.createObjectURL(blob);
   343→      const a = document.createElement('a'); a.href = url; a.download = `${IL.name}-تحلیل-متنی.txt`; a.click();
   344→      URL.revokeObjectURL(url);
   345→    } catch (err) { console.error('Text export failed:', err); }
   346→    finally { setExporting(false); }
   347→  }, [IL.name]);
   348→
   349→  return (
   350→    <div ref={containerRef} className="space-y-5" dir="rtl">
   351→      {/* ═══ HEADER ═══ */}
   352→      <div className="rounded-[32px] px-6 py-5 border border-amber-500/20"
   353→        style={{ background: 'rgba(18,28,46,0.7)', backdropFilter: 'blur(14px)', boxShadow: '0 25px 50px -12px rgba(0,0,0,0.6)' }}>

   700→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
   701→function LevelBox({ index, label, price, strength, isTarget, type }: {
   702→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target';
   703→}) {
   704→  const isSupport = type === 'support';
   705→  const isTargetType = type === 'target';
   706→
   707→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
   708→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
   709→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
   710→
   711→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
   712→  const strengthLabel = strength >= 8 ? 'بسیار قوی' : strength >= 6 ? 'قوی' : strength >= 4 ? 'متوسط' : strength >= 2 ? 'ضعیف' : 'بسیار ضعیف';
   713→  const strengthColor = strength >= 8 ? '#34c98b' : strength >= 6 ? '#3ad5db' : strength >= 4 ? '#ffb11b' : strength >= 2 ? '#ff7b32' : '#ef4d62';
   714→
   715→  // Distance from current price (if available from context)
   716→  return (
   717→    <div
   718→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
   719→      style={{
   720→        background: bgAccent,
   721→        border: `1px solid ${borderAccent}`,
   722→        borderRight: `3px solid ${accentColor}`,
   723→      }}
   724→    >
   725→      {/* Label + Price */}
   726→      <div className="flex-1 min-w-0">
   727→        <div className="flex items-center gap-2">
   728→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
   729→          {isTarget && <span className="text-[9px]">🎯</span>}
   730→        </div>
   731→        <span className="text-sm font-black text-gray-100 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
   732→        <span className="text-[10px] text-gray-500 mr-1">ریال</span>
   733→      </div>
   734→
   735→      {/* Strength bar */}
   736→      <div className="flex flex-col items-end gap-1 min-w-[80px]">
   737→        <div className="flex items-center gap-1.5">
   738→          <span className="text-[9px] font-medium" style={{ color: strengthColor }}>{strengthLabel}</span>
   739→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(strength)}</span>
   740→          <span className="text-[9px] text-gray-500">/۱۰</span>
   741→        </div>
   742→        <div className="w-full h-1.5 rounded-full bg-white/5 overflow-hidden">
   743→          <div
   744→            className="h-full rounded-full transition-all duration-500"
   745→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
   746→          />
   747→        </div>
   748→      </div>
   749→    </div>
   750→  );
   751→}
