   275→interface ScoredCandidate<T> { item: T; score: number; }

   281→// ─── School Scoring ─────────────────────────────────────────────────────
   282→function scoreSchool(input: NarrativeInput): ScoreFn {
   283→  return () => {
   284→    let s = 0;
   285→    const { adx, rsi, stochK, macdHist, obv, hasVolume, cci, bbPosition, trendDirection, nearResistance, nearSupport, trendR2, diPlus, diMinus } = input;
   286→
   287→    // Trend Following: strong trend + high ADX + good R2
   288→    if (adx > 25 && trendR2 > 0.7) s += (trendDirection === 'up' ? 8 : 6);
   289→    if (adx > 35) s += 3;
   290→    if (diPlus > diMinus && macdHist > 0) s += 2;
   291→
   292→    // Classical Patterns: near key levels + conflicting signals
   293→    if (nearResistance || nearSupport) s += 4;
   294→    if ((rsi > 60 && trendDirection === 'down') || (rsi < 40 && trendDirection === 'up')) s += 5; // divergence-like
   295→    if (cci > 100 || cci < -100) s += 3;
   296→
   297→    // Candlestick: near key levels (reversal zone)
   298→    if (nearResistance && stochK > 75) s += 4;
   299→    if (nearSupport && stochK < 25) s += 4;
   300→    if (nearResistance || nearSupport) s += 2;
   301→
   302→    // Fibonacci: after extended move (far from MA)
   303→    if (input.priceVsMa21 !== input.priceVsMa100) s += 3; // price between MAs = retracement zone
   304→    if (bbPosition > 80 || bbPosition < 20) s += 3; // extended from mean
   305→
   306→    // Volatility & Volume: BB squeeze or expansion + volume focus
   307→    if (bbPosition > 20 && bbPosition < 80 && adx < 20) s += 6; // squeeze
   308→    if (bbPosition > 90 || bbPosition < 10) s += 3; // expansion
   309→    if (hasVolume) s += 1;
   310→
   311→    // Oscillators: extreme readings + divergence
   312→    if (rsi > 70 || rsi < 30) s += 5;
   313→    if (stochK > 80 || stochK < 20) s += 4;
   314→    if (cci > 100 || cci < -100) s += 3;
   315→    if ((macdHist > 0 && rsi < 50) || (macdHist < 0 && rsi > 50)) s += 4; // divergence
   316→
   317→    // Support/Resistance: range market + at levels
   318→    if (adx < 20) s += 5; // range = S/R analysis
   319→    if (nearResistance) s += 3;
   320→    if (nearSupport) s += 3;
   321→
   322→    // Harmonic: at Fibonacci + near key levels
   323→    if ((bbPosition > 80 || bbPosition < 20) && (nearResistance || nearSupport)) s += 5;
   324→    if (input.priceVsMa21 !== input.priceVsMa100) s += 2;
   325→
   326→    // Elliott Wave: strong trend + good R2 + extended move
   327→    if (adx > 30 && trendR2 > 0.8) s += 4;
   328→    if (trendDirection === 'up' && bbPosition > 75) s += 3;
   329→    if (trendDirection === 'down' && bbPosition < 25) s += 3;
   330→
   331→    // Psychology: uncertain + extreme sentiment
   332→    if (adx < 15 || (rsi > 70 && trendDirection === 'down') || (rsi < 30 && trendDirection === 'up')) s += 5;
   333→    if (hasVolume && obv !== 0 && Math.sign(obv) !== (trendDirection === 'up' ? 1 : -1)) s += 4; // OBV divergence
   334→
   335→    return s;
   336→  };

   340→function scoreStyle(input: NarrativeInput): ScoreFn {
   341→  return () => {
   342→    let s = 0;
   343→    const { adx, rsi, stochK, macdHist, obv, hasVolume, bbPosition, trendDirection, nearResistance, nearSupport } = input;
   344→
   345→    // Conservative: range or uncertain
   346→    if (adx < 20) s += 6;
   347→    if (rsi > 40 && rsi < 60) s += 3;
   348→    if (bbPosition > 30 && bbPosition < 70) s += 2;
   349→
   350→    // Scalper: strong trend + momentum
   351→    if (adx > 25 && trendDirection === 'up' && macdHist > 0) s += 6;
   352→    if (stochK > 50 && stochK < 80) s += 2;
   353→    if (bbPosition > 50) s += 2;
   354→
   355→    // Trend Follower: clear trend
   356→    if (adx > 25) s += 4;
   357→    if (trendDirection === 'up' && macdHist > 0) s += 3;
   358→    if (trendDirection === 'down' && macdHist < 0) s += 3;
   359→
   360→    // Bearish: overbought near resistance
   361→    if ((rsi > 70 || stochK > 80) && nearResistance) s += 7;
   362→    if (macdHist < 0 && rsi > 60) s += 4; // divergence
   363→    if (cci > 100) s += 2;
   364→
   365→    // Narrative: uncertain + mixed signals
   366→    if (adx < 20) s += 4;
   367→    if ((macdHist > 0 && rsi < 50) || (macdHist < 0 && rsi > 50)) s += 3;
   368→
   369→    // Decision-Oriented: clear levels
   370→    if (nearResistance || nearSupport) s += 4;
   371→    if (adx > 25) s += 2;
   372→
   373→    // Volume Analyst: volume data available + divergence
   374→    if (hasVolume && obv !== 0) s += 4;
   375→    if (hasVolume && Math.sign(obv) !== (trendDirection === 'up' ? 1 : -1)) s += 3;
   376→
   377→    // Pattern Analyst: at key levels
   378→    if (nearResistance || nearSupport) s += 5;
   379→    if (bbPosition > 85 || bbPosition < 15) s += 3;
   380→
   381→    // Psychological: extreme sentiment
   382→    if (rsi > 75 || rsi < 25) s += 5;
   383→    if (stochK > 85 || stochK < 15) s += 3;
   384→    if (hasVolume && obv !== 0 && Math.sign(obv) !== (trendDirection === 'up' ? 1 : -1)) s += 2;
   385→
   386→    // Volatility Analyst: squeeze or extreme
   387→    if (bbPosition > 20 && bbPosition < 80 && adx < 20) s += 5;
   388→    if (bbPosition > 90 || bbPosition < 10) s += 4;
   389→
   390→    return s;
   391→  };

   396→  return () => {
   397→    let s = 0;
   398→    const { adx, rsi, stochK, macdHist, obv, hasVolume, bbPosition, trendDirection, nearResistance, nearSupport, dominantScenarioKey } = input;
   399→
   400→    // Formal Academic: clear signals + strong trend
   401→    if (adx > 25) s += 3;
   402→    if (trendR2 > 0.7) s += 2;
   403→
   404→    // Fast Operational: strong momentum
   405→    if (adx > 30 && macdHist > 0 && trendDirection === 'up') s += 5;
   406→    if (stochK > 60) s += 2;
   407→
   408→    // Philosophical Temporal: moderate trend
   409→    if (adx > 15 && adx < 35) s += 4;
   410→    if (rsi > 40 && rsi < 60) s += 2;
   411→
   412→    // Warning Risk-Focused: overbought/oversold + conflict
   413→    if ((rsi > 70 || stochK > 80) && nearResistance) s += 6;
   414→    if (macdHist > 0 && rsi < 50) s += 3;
   415→
   416→    // Storytelling Descriptive: uncertain + range
   417→    if (adx < 25) s += 4;
   418→    if (bbPosition > 30 && bbPosition < 70) s += 2;
   419→
   420→    // Step-by-Step Practical: clear levels + decisive
   421→    if (nearResistance || nearSupport) s += 4;
   422→    if (adx > 25) s += 2;
   423→
   424→    // Skeptical Questioning: conflicting signals
   425→    if ((macdHist > 0 && rsi < 50) || (macdHist < 0 && rsi > 50)) s += 6;
   426→    if (rsi > 60 && trendDirection === 'down') s += 3;
   427→    if (rsi < 40 && trendDirection === 'up') s += 3;
   428→
   429→    // Optimistic Trend-Following: strong bullish
   430→    if (trendDirection === 'up' && adx > 25 && rsi > 50 && rsi < 70) s += 6;
   431→    if (macdHist > 0 && obv > 0) s += 2;
   432→
   433→    // Simple Clear: clean setup
   434→    if (adx > 20 && adx < 40 && rsi > 40 && rsi < 60) s += 4;
   435→    if (!nearResistance && !nearSupport) s += 2;
   436→
   437→    // Quantitative Rigorous: data-heavy
   438→    if (hasVolume && adx > 25) s += 3;
   439→    if (rsi > 70 || rsi < 30) s += 2;
   440→
   441→    // Multi-Layer Mysterious: uncertain + complexity
   442→    if (adx < 20) s += 4;
   443→    if ((macdHist > 0 && rsi < 50) || (macdHist < 0 && rsi > 50)) s += 3;
   444→    if (dominantScenarioKey === 'R3') s += 2; // dominant = correction = complex
   445→
   446→    // Educational Guiding: good teaching moment
   447→    if (nearResistance || nearSupport) s += 3;
   448→    if (rsi > 65 || rsi < 35) s += 2;
   449→
   450→    // Comparative Multi-Scenario: multiple viable scenarios
   451→    const probs = Object.values(input.scenarioProbabilities);
   452→    const maxProb = Math.max(...probs);
   453→    if (maxProb < 35) s += 5; // no clear winner
   454→    if (probs.filter(p => p > 20).length >= 3) s += 3;
   455→
   456→    // Balanced Neutral: range market
   457→    if (adx < 20) s += 5;
   458→    if (rsi > 40 && rsi < 60) s += 3;
   459→
   460→    // Conclusive Result-Oriented: clear signal
   461→    if (adx > 25 && (rsi > 60 || rsi < 40)) s += 4;
   462→    if (dominantScenarioKey === 'R1' || dominantScenarioKey === 'R2') s += 2;
   463→
   464→    return s;
   465→  };
   466→}

   487→// Helper: build NarrativeInput from VdesRequest data
   488→export function buildNarrativeInput(body: {
   489→  currentPrice: number;
   490→  trendDirection: string;
   491→  rsi: number;
   492→  adx: number;
   493→  stochK: number;
   494→  stochD: number;
   495→  macdLine: number;
   496→  macdSignal: number;
   497→  macdHist: number;
   498→  diPlus: number;
   499→  diMinus: number;
   500→  obv: number;
   501→  hasVolume: boolean;
   502→  cci: number;
   503→  mfi: number;
   504→  atr: number;
   505→  bollingerUpper: number;
   506→  bollingerMiddle: number;
   507→  bollingerLower: number;
   508→  trendAngle: number;
   509→  trendR2: number;
   510→  overallSignal: string;
   511→  scenarios: Record<string, { probability: number }>;
   512→  resistances: number[];
   513→  supports: number[];
   514→  ma21: number;
   515→  ma100: number;
   516→}): NarrativeInput {
   517→  const bbRange = body.bollingerUpper - body.bollingerLower;
   518→  const bbPosition = bbRange > 0
   519→    ? Math.max(0, Math.min(100, ((body.currentPrice - body.bollingerLower) / bbRange) * 100))
   520→    : 50;
   521→
   522→  const R1 = body.resistances[0] || body.currentPrice * 1.05;
   523→  const S1 = body.supports[0] || body.currentPrice * 0.95;
   524→
   525→  // Find dominant scenario
   526→  let dominantKey = 'R3';
   527→  let dominantProb = 0;
   528→  for (const k of ['R1', 'R2', 'R3', 'R4', 'R5'] as const) {
   529→    const p = body.scenarios[k]?.probability ?? 0;
   530→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
   531→  }
   532→
   533→  return {
   534→    price: body.currentPrice,
   535→    trendDirection: body.trendDirection,
   536→    rsi: body.rsi,
   537→    adx: body.adx,
   538→    stochK: body.stochK,
   539→    stochD: body.stochD,
   540→    macdLine: body.macdLine,
   541→    macdSignal: body.macdSignal,
   542→    macdHist: body.macdHist,
   543→    diPlus: body.diPlus,
   544→    diMinus: body.diMinus,
   545→    obv: body.obv,
   546→    hasVolume: body.hasVolume,
   547→    cci: body.cci,
   548→    mfi: body.mfi,
   549→    atr: body.atr,
   550→    bbPosition,
   551→    bbUpper: body.bollingerUpper,
   552→    bbLower: body.bollingerLower,
   553→    dominantScenarioKey: dominantKey,
   554→    scenarioProbabilities: {
   555→      R1: body.scenarios.R1?.probability ?? 0,
   556→      R2: body.scenarios.R2?.probability ?? 0,
   557→      R3: body.scenarios.R3?.probability ?? 0,
   558→      R4: body.scenarios.R4?.probability ?? 0,
   559→      R5: body.scenarios.R5?.probability ?? 0,
   560→    },
   561→    nearResistance: R1 > 0 ? Math.abs(body.currentPrice - R1) / R1 < 0.03 : false,
   562→    nearSupport: S1 > 0 ? Math.abs(body.currentPrice - S1) / S1 < 0.03 : false,
   563→    priceVsMa21: body.currentPrice > body.ma21 ? 'above' : 'below',
   564→    priceVsMa100: body.currentPrice > body.ma100 ? 'above' : 'below',
   565→    trendR2: body.trendR2,
   566→  };
   567→}
