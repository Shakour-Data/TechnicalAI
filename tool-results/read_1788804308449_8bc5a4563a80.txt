    17→    17→function sma(closes: number[], period: number): number {
    18→    18→  if (closes.length < period) return 0;
    19→    19→  const slice = closes.slice(-period);
    20→    20→  return slice.reduce((a, b) => a + b, 0) / period;
    21→    21→}
    22→    22→
    23→    23→function emaCalc(closes: number[], period: number): number {
    24→    24→  if (closes.length < period) return 0;
    25→    25→  const k = 2 / (period + 1);
    26→    26→  let v = sma(closes.slice(0, period), period);
    27→    27→  for (let i = period; i < closes.length; i++) {
    28→    28→    v = closes[i] * k + v * (1 - k);
    29→    29→  }
    30→    30→  return v;
    31→    31→}

    33→    33→function calcRSI(closes: number[], period = 14): number {
    34→    34→  if (closes.length < period + 1) return 50;
    35→    35→  let avgGain = 0, avgLoss = 0;
    36→    36→  for (let i = 1; i <= period; i++) {
    37→    37→    const diff = closes[i] - closes[i - 1];
    38→    38→    if (diff > 0) avgGain += diff; else avgLoss += Math.abs(diff);
    39→    39→  }
    40→    40→  avgGain /= period; avgLoss /= period;
    41→    41→  for (let i = period + 1; i < closes.length; i++) {
    42→    42→    const diff = closes[i] - closes[i - 1];
    43→    43→    avgGain = (avgGain * (period - 1) + (diff > 0 ? diff : 0)) / period;
    44→    44→    avgLoss = (avgLoss * (period - 1) + (diff < 0 ? Math.abs(diff) : 0)) / period;
    45→    45→  }
    46→    46→  if (avgLoss === 0) return 100;
    47→    47→  return 100 - 100 / (1 + avgGain / avgLoss);
    48→    48→}

    50→    50→function calcCCI(data: OHLCV[], period = 20): number {
    51→    51→  if (data.length < period) return 0;
    52→    52→  const tps: number[] = [];
    53→    53→  for (let i = data.length - period; i < data.length; i++) {
    54→    54→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
    55→    55→  }
    56→    56→  const mean = tps.reduce((a, b) => a + b, 0) / tps.length;
    57→    57→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - mean), 0) / tps.length;
    58→    58→  return meanDev === 0 ? 0 : (tps[tps.length - 1] - mean) / (0.015 * meanDev);
    59→    59→}

    61→    61→function calcMFI(data: OHLCV[], period = 14): number {
    62→    62→  if (data.length < period + 1) return 50;
    63→    63→  let posFlow = 0, negFlow = 0;
    64→    64→  for (let i = data.length - period; i < data.length; i++) {
    65→    65→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
    66→    66→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
    67→    67→    const mf = tp * data[i].volume;
    68→    68→    if (tp > prevTp) posFlow += mf; else negFlow += mf;
    69→    69→  }
    70→    70→  return negFlow === 0 ? 100 : 100 - 100 / (1 + posFlow / negFlow);
    71→    71→}

    73→    73→function calcStochastic(data: OHLCV[], kPeriod = 14, smoothK = 3, smoothD = 3): { k: number; d: number } {
    74→    74→  const rawKs: number[] = [];
    75→    75→  for (let i = kPeriod - 1; i < data.length; i++) {
    76→    76→    let lowest = Infinity, highest = -Infinity;
    77→    77→    for (let j = i - kPeriod + 1; j <= i; j++) {
    78→    78→      if (data[j].low < lowest) lowest = data[j].low;
    79→    79→      if (data[j].high > highest) highest = data[j].high;
    80→    80→    }
    81→    81→    const range = highest - lowest;
    82→    82→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
    83→    83→  }
    84→    84→  const smoothedK: number[] = [];
    85→    85→  for (let i = smoothK - 1; i < rawKs.length; i++) {
    86→    86→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
    87→    87→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
    88→    88→  }
    89→    89→  const k = smoothedK[smoothedK.length - 1] ?? 50;
    90→    90→  const dSlice = smoothedK.slice(-smoothD);
    91→    91→  const d = dSlice.length > 0 ? dSlice.reduce((a, b) => a + b, 0) / dSlice.length : 50;
    92→    92→  return { k, d };
    93→    93→}

    95→    95→function calcMACD(closes: number[], fast = 12, slow = 26, sig = 9): { line: number; signal: number; histogram: number } {
    96→    96→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
    97→    97→  const kFast = 2 / (fast + 1), kSlow = 2 / (slow + 1), kSig = 2 / (sig + 1);
    98→    98→  let eFast = sma(closes.slice(0, fast), fast);
    99→    99→  let eSlow = sma(closes.slice(0, slow), slow);
   100→   100→  const macdVals: number[] = [];
   101→   101→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   102→   102→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   103→   103→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   104→   104→    macdVals.push(eFast - eSlow);
   105→   105→  }
   106→   106→  if (macdVals.length < sig) return { line: 0, signal: 0, histogram: 0 };
   107→   107→  let s = sma(macdVals.slice(0, sig), sig);
   108→   108→  for (let i = sig; i < macdVals.length; i++) {
   109→   109→    s = macdVals[i] * kSig + s * (1 - kSig);
   110→   110→  }
   111→   111→  const line = macdVals[macdVals.length - 1];
   112→   112→  return { line, signal: s, histogram: line - s };
   113→   113→}

   115→   115→function calcBollingerBands(closes: number[], period = 20, mult = 2): { upper: number; middle: number; lower: number } {
   116→   116→  if (closes.length < period) return { upper: 0, middle: 0, lower: 0 };
   117→   117→  const slice = closes.slice(-period);
   118→   118→  const mid = slice.reduce((a, b) => a + b, 0) / period;
   119→   119→  const variance = slice.reduce((a, b) => a + (b - mid) ** 2, 0) / period;
   120→   120→  const sd = Math.sqrt(variance);
   121→   121→  return { upper: mid + mult * sd, middle: mid, lower: mid - mult * sd };
   122→   122→}

   124→   124→function calcADX(data: OHLCV[], period = 14): { adx: number; diPlus: number; diMinus: number } {
   125→   125→  if (data.length < period * 2) return { adx: 0, diPlus: 0, diMinus: 0 };
   126→   126→  let sumPDM = 0, sumMDM = 0, sumTR = 0;
   127→   127→  for (let i = data.length - period; i < data.length; i++) {
   128→   128→    const upMove = data[i].high - data[i - 1].high;
   129→   129→    const downMove = data[i - 1].low - data[i].low;
   130→   130→    sumPDM += upMove > downMove && upMove > 0 ? upMove : 0;
   131→   131→    sumMDM += downMove > upMove && downMove > 0 ? downMove : 0;
   132→   132→    sumTR += Math.max(
   133→   133→      data[i].high - data[i].low,
   134→   134→      Math.abs(data[i].high - data[i - 1].close),
   135→   135→      Math.abs(data[i].low - data[i - 1].close)
   136→   136→    );
   137→   137→  }
   138→   138→  const diPlus = sumTR > 0 ? (sumPDM / sumTR) * 100 : 0;
   139→   139→  const diMinus = sumTR > 0 ? (sumMDM / sumTR) * 100 : 0;
   140→   140→  const dx = (diPlus + diMinus) > 0 ? Math.abs(diPlus - diMinus) / (diPlus + diMinus) * 100 : 0;
   141→   141→  return { adx: dx, diPlus, diMinus };
   142→   142→}

   144→   144→function calcATR(data: OHLCV[], period = 14): number {
   145→   145→  if (data.length < period + 1) return 0;
   146→   146→  let sum = 0;
   147→   147→  for (let i = data.length - period; i < data.length; i++) {
   148→   148→    sum += Math.max(
   149→   149→      data[i].high - data[i].low,
   150→   150→      Math.abs(data[i].high - data[i - 1].close),
   151→   151→      Math.abs(data[i].low - data[i - 1].close)
   152→   152→    );
   153→   153→  }
   154→   154→  return sum / period;
   155→   155→}

   165→   165→const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));
   166→   166→
   167→   167→/**
   168→   168→ * Extract the 16 VDSS (Volatility-Direction-Strength-Structure) features at a
   169→   169→ * given point in the OHLCV history. Feature order must match
   170→   170→ * {@link VDSS_FEATURE_NAMES} exactly.
   171→   171→ *
   172→   172→ * The extraction proceeds in two layers:
   173→   173→ * - **Layer 1** — Raw indicator scores (`s_*`) mapped into [0,1] via
   174→   174→ *   threshold tables or clamped normalisation.
   175→   175→ * - **Layer 2** — Momentum-corrected scores (`f_*`) that apply a small delta
   176→   176→ *   adjustment based on the 5-bar lookback change, plus binary cross-signal
   177→   177→ *   features (Stochastic cross, MACD cross, RSI divergence).
   178→   178→ *
   179→   179→ * ### The 16 Features (in order)
   180→   180→ * | Index | Name            | Source                       | Layer |
   181→   181→ * |-------|-----------------|------------------------------|-------|
   182→   182→ * |  0    | RSI             | `calcRSI(14)`                | 2     |
   183→   183→ * |  1    | MFI             | `calcMFI(14)` (skip if no vol)| 2    |
   184→   184→ * |  2    | CCI             | `calcCCI(20)`                | 2     |
   185→   185→ * |  3    | ADX             | `calcADX(14)` DI± diff       | 1     |
   186→   186→ * |  4    | MACD            | `calcMACD(12,26,9)` histogram| 2     |
   187→   187→ * |  5    | Stochastic      | `calcStochastic(14,3,3)`     | 2     |
   188→   188→ * |  6    | BB_position     | Bollinger %B (20,2)          | 1     |
   189→   189→ * |  7    | MA21            | Price vs SMA-21              | 1     |
   190→   190→ * |  8    | MA100           | Price vs SMA-100             | 1     |
   191→   191→ * |  9    | EMA_gap         | (EMA12-EMA26)/EMA26          | 1     |
   192→   192→ * | 10    | ATR_pct         | 1 − ATR/price×10, clamped    | 1     |
   193→   193→ * | 11    | Trend           | R² × |angle| regression      | 1     |
   194→   194→ * | 12    | SR_distance     | Nearest S/R proximity        | 1     |
   195→   195→ * | 13    | Stoch_cross     | K/D bullish cross (binary)   | 2     |
   196→   196→ * | 14    | MACD_cross      | Line/Signal bullish cross    | 2     |
   197→   197→ * | 15    | Divergence      | Price/RSI divergence (binary)| 2     |
   198→   198→ *
   199→   199→ * @param data     - Full OHLCV history (chronological, oldest first)
   200→   200→ * @param endIdx   - Index of the candle to compute features for
   201→   201→ * @param hasVolume - Whether the data contains valid volume bars
   202→   202→ * @returns Array of exactly 16 feature values, each in [0,1].
   203→   203→ *          Returns `new Array(16).fill(0.5)` if the price at `endIdx` is ≤ 0.
   204→   204→ */
   205→   205→export function extractVDSSFeatures(data: OHLCV[], endIdx: number, hasVolume: boolean): number[] {
   206→   206→  const slice = data.slice(0, endIdx + 1);
   207→   207→  const closes = slice.map(d => d.close);
   208→   208→  const price = closes[closes.length - 1];
   209→   209→  if (price <= 0) return new Array(16).fill(0.5);
   210→   210→
   211→   211→  // ── Compute indicators ──
   212→   212→  const rsi = calcRSI(closes);
   213→   213→  const mfi = hasVolume ? calcMFI(slice) : 50;
   214→   214→  const cci = calcCCI(slice);
   215→   215→  const adxResult = calcADX(slice);
   216→   216→  const macd = calcMACD(closes);
   217→   217→  const stoch = calcStochastic(slice);
   218→   218→  const bb = calcBollingerBands(closes);
   219→   219→  const atr = calcATR(slice);
   220→   220→  const ma21 = sma(closes, 21);
   221→   221→  const ma100 = sma(closes, 100);
   222→   222→  const ema12 = emaCalc(closes, 12);
   223→   223→  const ema26 = emaCalc(closes, 26);
   224→   224→
   225→   225→  // ── LAYER 1: Raw scores (s_*) ──
   226→   226→  const s_rsi = rsi > 80 ? 0.95 : rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   227→   227→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   228→   228→  const s_mfi = mfi > 80 ? 0.95 : mfi > 70 ? 0.85 : mfi > 60 ? 0.7 : mfi > 50 ? 0.55
   229→   229→    : mfi > 40 ? 0.45 : mfi > 30 ? 0.3 : mfi > 20 ? 0.15 : 0.05;
   230→   230→  const s_cci = cci > 200 ? 0.9 : cci > 100 ? 0.75 : cci > 0 ? 0.6
   231→   231→    : cci > -100 ? 0.4 : cci > -200 ? 0.25 : 0.1;
   232→   232→  const s_adx = adxResult.diPlus > adxResult.diMinus
   233→   233→    ? clamp(0.5 + (adxResult.diPlus - adxResult.diMinus) / 100, 0.5, 1)
   234→   234→    : clamp(0.5 - (adxResult.diMinus - adxResult.diPlus) / 100, 0, 0.5);
   235→   235→  const s_macd = macd.histogram > 0 && macd.line > macd.signal ? 0.9
   236→   236→    : macd.histogram > 0 ? 0.7 : macd.line > macd.signal ? 0.55
   237→   237→    : macd.histogram < 0 && macd.line < macd.signal ? 0.1 : 0.3;
   238→   238→  const s_stoch = clamp(
   239→   239→    (stoch.k > 80 ? 0.9 : stoch.k > 70 ? 0.8 : stoch.k > 50 ? 0.6
   240→   240→      : stoch.k > 30 ? 0.4 : stoch.k > 20 ? 0.2 : 0.1)
   241→   241→    + (stoch.k > stoch.d ? 0.1 : -0.1), 0, 1);
   242→   242→  const bbRange = bb.upper - bb.lower;
   243→   243→  const s_bb = bbRange === 0 ? 0.5 : clamp((price - bb.lower) / bbRange, 0, 1);
   244→   244→  const s_ma21 = clamp(price > ma21 ? 0.7 + ((price - ma21) / ma21) * 0.5 : 0.3 + ((price - ma21) / ma21) * 0.5, 0, 1);
   245→   245→  const s_ma100 = clamp(price > ma100 ? 0.65 + ((price - ma100) / ma100) * 0.3 : 0.35 + ((price - ma100) / ma100) * 0.3, 0, 1);
   246→   246→  const emaGap = ema12 > 0 && ema26 > 0 ? (ema12 - ema26) / ema26 : 0;
   247→   247→  const s_ema = clamp(0.5 + emaGap * 5, 0, 1);
   248→   248→  const s_atr = clamp(1 - atr / price * 10, 0.2, 1);
   249→   249→
   250→   250→  // Trend (R² + angle)
   251→   251→  const trendPeriod = Math.min(21, closes.length);
   252→   252→  const trendSlice = closes.slice(-trendPeriod);
   253→   253→  const n = trendSlice.length;
   254→   254→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   255→   255→  for (let i = 0; i < n; i++) { sumX += i; sumY += trendSlice[i]; sumXY += i * trendSlice[i]; sumX2 += i * i; sumY2 += trendSlice[i] ** 2; }
   256→   256→  const denom = n * sumX2 - sumX * sumX;
   257→   257→  let r2 = 0, angle = 0;
   258→   258→  if (denom !== 0) {
   259→   259→    const slope = (n * sumXY - sumX * sumY) / denom;
   260→   260→    const meanY = sumY / n;
   261→   261→    let ssRes = 0, ssTot = 0;
   262→   262→    for (let i = 0; i < n; i++) {
   263→   263→      ssRes += (trendSlice[i] - (slope * i + meanY - slope * (n - 1) / 2)) ** 2;
   264→   264→      ssTot += (trendSlice[i] - meanY) ** 2;
   265→   265→    }
   266→   266→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
   267→   267→    // Convert slope/meanY to percentage before atan (same fix as ta-engine.ts)
   268→   268→    // so that 1% daily rise → 45°, not ~0.57°
   269→   269→    angle = meanY > 0 ? Math.atan((slope / meanY) * 100) * (180 / Math.PI) : 0;
   270→   270→  }
   271→   271→  const tStr = r2 * Math.abs(angle) / 45;
   272→   272→  const direction = denom !== 0 ? (n * sumXY - sumX * sumY) / denom : 0;
   273→   273→  const s_trend = direction > 0 ? clamp(0.5 + tStr / 2, 0.5, 1)
   274→   274→    : direction < 0 ? clamp(0.5 - tStr / 2, 0, 0.5) : 0.5;
   275→   275→
   276→   276→  // Simplified S/R: distance to nearest local high/low
   277→   277→  let s_sr = 0.5;
   278→   278→  if (slice.length > 20) {
   279→   279→    const lookback = slice.slice(-30);
   280→   280→    let nearestHigh = Infinity, nearestLow = 0;
   281→   281→    for (const d of lookback) {
   282→   282→      if (d.high > price && d.high < nearestHigh) nearestHigh = d.high;
   283→   283→      if (d.low < price && d.low > nearestLow) nearestLow = d.low;
   284→   284→    }
   285→   285→    const dR = nearestHigh < Infinity ? Math.abs(price - nearestHigh) / nearestHigh : 1;
   286→   286→    const dS = nearestLow > 0 ? Math.abs(price - nearestLow) / nearestLow : 1;
   287→   287→    s_sr = clamp(0.5 - dR * 1.5 + dS * 1.0, 0, 1);
   288→   288→  }
   289→   289→
   290→   290→  // ── LAYER 2: Momentum correction ──
   291→   291→  const TB = 5;
   292→   292→  let f_rsi = s_rsi, f_mfi = s_mfi, f_cci = s_cci, f_macd = s_macd, f_stoch = s_stoch;
   293→   293→  if (slice.length > TB + 15) {
   294→   294→    const pastSlice = slice.slice(0, -TB);
   295→   295→    const pastCloses = pastSlice.map(d => d.close);
   296→   296→    const rsiPast = calcRSI(pastCloses);
   297→   297→    const mfiPast = hasVolume ? calcMFI(pastSlice) : 50;
   298→   298→    const cciPast = calcCCI(pastSlice);
   299→   299→    const macdPast = calcMACD(pastCloses);
   300→   300→    const stochPast = calcStochastic(pastSlice);
   301→   301→
   302→   302→    f_rsi = clamp(s_rsi + (rsi - rsiPast) / 100 * 0.08, 0, 1);
   303→   303→    f_mfi = clamp(s_mfi + (mfi - mfiPast) / 100 * 0.08, 0, 1);
   304→   304→    f_cci = clamp(s_cci + (cci - cciPast) / 300 * 0.06, 0, 1);
   305→   305→    f_macd = clamp(s_macd + (macd.histogram - macdPast.histogram) / (Math.abs(macd.histogram) + Math.abs(macdPast.histogram) + 1) * 3 * 0.08, 0, 1);
   306→   306→    f_stoch = clamp(s_stoch + (stoch.k - stochPast.k) / 50 * 0.06, 0, 1);
   307→   307→  }
   308→   308→
   309→   309→  // ── LAYER 2: Cross signals (binary per spec) ──
   310→   310→  let f_stochCross = 0.1;
   311→   311→  let f_macdCross = 0.1;
   312→   312→  let f_div = 0.15;
   313→   313→
   314→   314→  if (slice.length > 20) {
   315→   315→    const prevSlice = slice.slice(0, -1);
   316→   316→    const prevStoch = calcStochastic(prevSlice);
   317→   317→    const prevMacd = calcMACD(prevSlice.map(d => d.close));
   318→   318→
   319→   319→    // Stochastic bullish cross: K_prev < D_prev AND K_curr > D_curr
   320→   320→    f_stochCross = (prevStoch.k < prevStoch.d && stoch.k > stoch.d) ? 0.9 : 0.1;
   321→   321→
   322→   322→    // MACD bullish cross: line_prev < signal_prev AND line_curr > signal_curr
   323→   323→    f_macdCross = (prevMacd.line < prevMacd.signal && macd.line > macd.signal) ? 0.9 : 0.1;
   324→   324→
   325→   325→    // Divergence detection
   326→   326→    if (closes.length > 40) {
   327→   327→      const p20 = closes[closes.length - 21];
   328→   328→      const rsi20 = calcRSI(closes.slice(0, -20));
   329→   329→      const pChange = price - p20;
   330→   330→      const rsiChange = rsi - rsi20;
   331→   331→      if (pChange !== 0 && rsiChange !== 0) {
   332→   332→        f_div = (pChange > 0 && rsiChange < 0) ? 0.85 : (pChange < 0 && rsiChange > 0) ? 0.85 : 0.15;
   333→   333→      }
   334→   334→    }
   335→   335→  }
   336→   336→
   337→   337→  // ── Return 16 features in exact order ──
   338→   338→  return [

   380→   380→// ─── Main Training Function ─────────────────────────────────────────────────
   381→   381→
   382→   382→/**
   383→   383→ * Train the Adaptive Weight Model for a given symbol using logistic regression
   384→   384→ * with **TimeSeriesSplit** cross-validation.
   385→   385→ *
   386→   386→ * ### Training Pipeline
   387→   387→ * 1. **Feature extraction** — For each candle from index `START_IDX` (40) to
   388→   388→ *    `data.length − FORWARD_DAYS`, computes the 16 VDSS features via
   389→   389→ *    {@link extractVDSSFeatures}.
   390→   390→ * 2. **Binary labelling** — Labels each sample as `1` (bullish) if the
   391→   391→ *    5-day forward return exceeds +1%, otherwise `0` (bearish).
   392→   392→ * 3. **Model training** — Fits an {@link AdaptiveWeightModel} (L2-regularised
   393→   393→ *    logistic regression) with TimeSeriesSplit validation on the feature/label
   394→   394→ *    matrices.
   395→   395→ * 4. **Prediction** — Scores the most recent candle's features and caches the
   396→   396→ *    result (TTL = 1 hour).
   397→   397→ *
   398→   398→ * Returns `null` if there are insufficient data points, non-finite features,
   399→   399→ * or if the underlying model fails to converge.
   400→   400→ *
   401→   401→ * @param data       - Full OHLCV history (chronological, oldest first)
   402→   402→ * @param symbol     - Ticker/symbol identifier used as the model cache key
   403→   403→ * @param minSamples - Minimum number of valid training samples required (default 70)
   404→   404→ * @returns The trained {@link AdaptiveModelResult} on success, or `null` on failure
   405→   405→ */
   406→   406→export function trainAdaptiveModel(data: OHLCV[], symbol: string, minSamples = 70): AdaptiveModelResult | null {
   407→   407→  const FORWARD_DAYS = 5;
   408→   408→  const START_IDX = 40; // Need enough history for all indicators to stabilize
   409→   409→
   410→   410→  if (data.length < START_IDX + FORWARD_DAYS + minSamples) return null;
   411→   411→
   412→   412→  const hasVolume = data.some(d => d.volume > 0);
   413→   413→
   414→   414→  const X: number[][] = [];
   415→   415→  const y: number[] = [];
   416→   416→
   417→   417→  for (let i = START_IDX; i < data.length - FORWARD_DAYS; i++) {
   418→   418→    const features = extractVDSSFeatures(data, i, hasVolume);
   419→   419→    const currentPrice = data[i].close;
   420→   420→    const futurePrice = data[i + FORWARD_DAYS].close;
   421→   421→
   422→   422→    if (currentPrice <= 0) continue;
   423→   423→    if (features.some(f => !isFinite(f))) continue;
   424→   424→
   425→   425→    // Label: 1 if 5-day forward return > 1%, else 0
   426→   426→    const forwardReturn = (futurePrice - currentPrice) / currentPrice;
   427→   427→    const label = forwardReturn > 0.01 ? 1 : 0;
   428→   428→
   429→   429→    X.push(features);
   430→   430→    y.push(label);
   431→   431→  }
   432→   432→
   433→   433→  if (X.length < minSamples) return null;
   434→   434→
   435→   435→  const model = new AdaptiveWeightModel(minSamples);
   436→   436→  const success = model.train(X, y);
   437→   437→
   438→   438→  if (!success) return null;
   439→   439→
   440→   440→  // Predict current features
   441→   441→  const currentFeatures = extractVDSSFeatures(data, data.length - 1, hasVolume);
   442→   442→  model.predictScore(currentFeatures);
   443→   443→
   444→   444→  const result = model.getResult();
   445→   445→  setCachedAdaptiveModel(symbol, result);
   446→   446→
   447→   447→  return result;
   448→   448→}

   450→   450→// ─── Calculate bullConsensus with ML Weights (Layer 3) ──────────────────────
   451→   451→
   452→   452→/**
   453→   453→ * Calculate the ML-weighted bull consensus from the 16 VDSS features.
   454→   454→ *
   455→   455→ * ### ML Path (when `mlResult` is trained)
   456→   456→ * Computes a weighted average: `Σ(w_j × f_j) / Σ(w_j)`, skipping the MFI
   457→   457→ * feature (index 1) when `hasVolume` is false. The raw consensus is then
   458→   458→ * blended with the ML direct prediction probability, with the blend weight
   459→   459→ * proportional to `recentAccuracy × 0.30` (capped at 0.30).
   460→   460→ *
   461→   461→ * ### Fallback Heuristic (when ML is unavailable)
   462→   462→ * Uses fixed default weights and the same weighted-average formula:
   463→   463→ * ```
   464→   464→ * DEFAULT_WEIGHTS = [
   465→   465→ *   0.15, 0.12, 0.10, 0.08,  // rsi, mfi, cci, adx
   466→   466→ *   0.14, 0.10, 0.08,         // macd, stoch, bb
   467→   467→ *   0.10, 0.08, 0.10,         // ma21, ma100, ema
   468→   468→ *   0.05, 0.10, 0.05,         // atr, trend, sr
   469→   469→ *   0.06, 0.06, 0.08,         // stochCross, macdCross, div
   470→   470→ * ]
   471→   471→ * ```
   472→   472→ *
   473→   473→ * The fallback heuristic can also be expressed as a sigmoid of a raw bull
   474→   474→ * score derived from individual indicator contributions:
   475→   475→ * ```
   476→   476→ * bullScore = (rsi<40 ? 0.3 : rsi>60 ? -0.1 : 0)
   477→   477→ *           + (adx>25 && diPlus>diMinus ? 0.3 : 0)
   478→   478→ *           + (macdHist>0 ? 0.2 : -0.2)
   479→   479→ *           + (bbPos<0.3 ? 0.2 : bbPos>0.7 ? -0.2 : 0)
   480→   480→ * bullConsensus = sigmoid(bullScore)
   481→   481→ * ```
   482→   482→ *
   483→   483→ * @param features  - Array of 16 VDSS feature values for the current candle
   484→   484→ * @param mlResult  - Trained adaptive model result, or `null` to use fallback weights
   485→   485→ * @param hasVolume - Whether volume data is valid (disables MFI feature at index 1)
   486→   486→ * @returns Object with `bullConsensus` clamped to [0,1] and `usedML` flag
   487→   487→ */
   488→   488→export function calculateBullConsensus(
   489→   489→  features: number[],
   490→   490→  mlResult: AdaptiveModelResult | null,
   491→   491→  hasVolume: boolean,
   492→   492→): { bullConsensus: number; usedML: boolean } {
   493→   493→  if (mlResult && mlResult.isTrained && mlResult.weights.length === 16) {
   494→   494→    const weights = mlResult.weights;
   495→   495→    let bullConsensus = 0;
   496→   496→    let totalWeight = 0;
   497→   497→
   498→   498→    for (let j = 0; j < 16; j++) {
   499→   499→      // Skip MFI (index 1) if no volume
   500→   500→      if (j === 1 && !hasVolume) continue;
   501→   501→      bullConsensus += weights[j] * features[j];
   502→   502→      totalWeight += weights[j];
   503→   503→    }
   504→   504→
   505→   505→    if (totalWeight > 0) bullConsensus /= totalWeight;
   506→   506→
   507→   507→    // Blend with ML direct prediction (dynamic weight based on accuracy)
   508→   508→    if (mlResult.predictionProb !== null) {
   509→   509→      const mlWeight = Math.max(0, Math.min(mlResult.recentAccuracy * 0.30, 0.30));
   510→   510→      bullConsensus = bullConsensus * (1 - mlWeight) + mlResult.predictionProb * mlWeight;
   511→   511→    }
   512→   512→
   513→   513→    return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: true };
   514→   514→  }
   515→   515→
   516→   516→  // Fallback: default weights (per spec, only when ML is unavailable)
   517→   517→  const DEFAULT_WEIGHTS = [
   518→   518→    0.15, 0.12, 0.10, 0.08,  // rsi, mfi, cci, adx
   519→   519→    0.14, 0.10, 0.08,         // macd, stoch, bb
   520→   520→    0.10, 0.08, 0.10,         // ma21, ma100, ema
   521→   521→    0.05, 0.10, 0.05,         // atr, trend, sr
   522→   522→    0.06, 0.06, 0.08,         // stochCross, macdCross, div
   523→   523→  ];
   524→   524→
   525→   525→  let bullConsensus = 0;
   526→   526→  let totalWeight = 0;
   527→   527→  for (let j = 0; j < 16; j++) {
   528→   528→    if (j === 1 && !hasVolume) continue;
   529→   529→    bullConsensus += DEFAULT_WEIGHTS[j] * features[j];
   530→   530→    totalWeight += DEFAULT_WEIGHTS[j];
   531→   531→  }
   532→   532→  if (totalWeight > 0) bullConsensus /= totalWeight;
   533→   533→
   534→   534→  return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: false };
   535→   535→}

   599→   599→// ─── Decision Graph Scenario Probabilities (3-Branch DAG) ──────────────────
   600→   600→// Implements the exact structure from TechnicalAnalysisDssGraph.txt:
   601→   601→//   Root → 3 branches (Trend Following A, Breakout B, Reversal C)
   602→   602→//   Each branch → 9 edges → 9 leaf scenarios
   603→   603→//   P(Scenario_S) = Σ_branch P(Branch) × P(Edge_S | Branch)
   604→   604→//   Where P(Branch) is derived from EMV maximization,
   605→   605→//   and P(Edge) is the conditional probability per edge formula.
   606→   606→// ═══════════════════════════════════════════════════════════════════════════
   607→   607→
   608→   608→/**
   609→   609→ * Full market context for the decision-graph scenario probability calculation.
   610→   610→ * Aggregates all indicator values, ML outputs, and structural levels needed
   611→   611→ * by the 3-branch DAG.
   612→   612→ */
   613→   613→export interface DecisionGraphContext {
   614→   614→  bullConsensus: number;
   615→   615→  price: number;
   616→   616→  R1_level: number;
   617→   617→  S1_level: number;
   618→   618→  MA100: number;
   619→   619→  rsi: number;
   620→   620→  mfi: number;
   621→   621→  stochK: number;
   622→   622→  stochD: number;
   623→   623→  macdHist: number;
   624→   624→  macdLine: number;
   625→   625→  macdSignal: number;
   626→   626→  adx: number;
   627→   627→  diPlus: number;
   628→   628→  diMinus: number;
   629→   629→  atr: number;
   630→   630→  bbUpper: number;
   631→   631→  bbLower: number;
   632→   632→  bbMiddle: number;
   633→   633→  hasVolume: boolean;
   634→   634→  volume: number;
   635→   635→  avgVolume20: number;
   636→   636→  mlResult: AdaptiveModelResult | null;
   637→   637→  mlPrediction: MLPredictionResult | null;
   638→   638→  mlRegime: MLRegimeResult | null;
   639→   639→}
   640→   640→
   641→   641→/**
   642→   642→ * Calculate 9-scenario probabilities using the **3-branch decision DAG**.
   643→   643→ *
   644→   644→ * The DAG structure (per TechnicalAnalysisDssGraph.txt):
   645→   645→ * - **Root** → 3 branches: Trend Following (A), Breakout (B), Reversal (C)
   646→   646→ * - Each branch emits 9 conditional probabilities → 9 leaf scenarios
   647→   647→ * - `P(Scenario_S) = Σ_branch P(Branch) × P(Edge_S | Branch)`
   648→   648→ *
   649→   649→ * ### Computation Steps
   650→   650→ * 1. **P_Bull_Current / P_Bear_Current** — Bayesian-weighted aggregation of
   651→   651→ *    current indicator signals (RSI, MFI, StochK, MA100 position, ADX, S/R
   652→   652→ *    proximity) using category priors (trend=0.28, oscillator=0.25,
   653→   653→ *    volume=0.15, volatility=0.10, leading=0.12, patterns=0.10).
   654→   654→ * 2. **P_Bull_Forecast** — Uses ML `predictionProb` if available; otherwise
   655→   655→ *    applies exponential decay toward 0.5 (mean-reversion).
   656→   656→ * 3. **Final directional probs** — Combines current and forecast:
   657→   657→ *    ```
   658→   658→ *    P_Bull_Final  = α × P_Bull_Current  + (1-α) × P_Bull_Forecast
   659→   659→ *    P_Bear_Final  = α × P_Bear_Current  + (1-α) × P_Bear_Forecast
   660→   660→ *    P_Neutral_Final = max(0.04, 1 − P_Bull − P_Bear)
   661→   661→ *    ```
   662→   662→ * 4. **Conditional distributions** — CP(S_i | Bull/Bear/Neutral) computed
   663→   663→ *    from ADX trend strength, overbought/oversold, volatility, divergence,
   664→   664→ *    and ML-adaptive factors. Normalised to sum to 1 per state.
   665→   665→ * 5. **Final probabilities** — Weighted sum across states, normalised to 100.
   666→   666→ *
   667→   667→ * ### The 9 Scenarios
   668→   668→ * | i | Name              | Description                            |
   669→   669→ * |---|-------------------|----------------------------------------|
   670→   670→ * | 1 | Strong Bull       | ADX > 40, strong uptrend               |
   671→   671→ * | 2 | Accelerating Bull | High momentum + volatility expansion   |
   672→   672→ * | 3 | Cautious Bull     | Bullish but overbought risk            |
   673→   673→ * | 4 | Strong Bear       | ADX > 40, strong downtrend             |
   674→   674→ * | 5 | Accelerating Bear | High momentum + volatility expansion   |
   675→   675→ * | 6 | Cautious Bear     | Bearish but oversold risk              |
   676→   676→ * | 7 | Low-Vol Range     | ATR low, ADX < 20 → sideways           |
   677→   677→ * | 8 | High-Vol Range    | ATR high, ADX < 30 → choppy            |
   678→   678→ * | 9 | Shock             | Black-swan tail risk (2-5%)            |
   679→   679→ *
   680→   680→ * @param bullConsensus - Overall bull consensus ∈ [0,1]
   681→   681→ * @param price        - Current close price
   682→   682→ * @param SC1          - R1 resistance level
   683→   683→ * @param S1           - S1 support level
   684→   684→ * @param MA100        - 100-period simple moving average
   685→   685→ * @param rsi          - Current RSI value (0-100)
   686→   686→ * @param mfi          - Current MFI value (0-100)
   687→   687→ * @param stochK       - Current Stochastic %K value
   688→   688→ * @param hasVolume    - Whether volume data is valid
   689→   689→ * @param mlResult     - Trained adaptive model result (or null)
   690→   690→ * @param adx          - Current ADX value (default 25)
   691→   691→ * @param atr          - Current ATR value (default 0)
   692→   692→ * @returns Object with `pSC1`–`pSC9` (integers summing to 100) and adaptive `factors`
   693→   693→ */
   694→   694→export function calculateScenarioProbabilities(
   695→   695→  bullConsensus: number,
   696→   696→  price: number,
   697→   697→  SC1: number,
   698→   698→  S1: number,
   699→   699→  MA100: number,
   700→   700→  rsi: number,
   701→   701→  mfi: number,
   702→   702→  stochK: number,
   703→   703→  hasVolume: boolean,
   704→   704→  mlResult: AdaptiveModelResult | null,
   705→   705→  adx: number = 25,
   706→   706→  atr: number = 0,
   707→   707→): { pSC1: number; pSC2: number; pSC3: number; pSC4: number; pSC5: number; pSC6: number; pSC7: number; pSC8: number; pSC9: number; factors: { momentum: number; volatility: number; trend: number } } {
   708→   708→  // ── Adaptive ML factors ──
   709→   709→  const mlParams = mlResult?.adaptiveParams;
   710→   710→  const factors = {
   711→   711→    momentum: mlParams?.momentumFactor ?? 0.7,
   712→   712→    volatility: mlParams?.volatilityFactor ?? 0.5,
   713→   713→    trend: mlParams?.trendFactor ?? 0.6,
   714→   714→  };
   715→   715→
   716→   716→  const bull = bullConsensus;
   717→   717→  const bear = 1 - bullConsensus;
   718→   718→
   719→   719→  // ═══════════════════════════════════════════════════════════════════════
   720→   720→  // STEP 1: Compute P_Bull_Current, P_Bear_Current from current indicators
   721→   721→  // ═══════════════════════════════════════════════════════════════════════
   722→   722→  // Use Bayesian weighting: W_i = P(Success|Tool_i) × Prior_i / Σ
   723→   723→  // For current signals, each indicator provides a directional score.
   724→   724→
   725→   725→  // RSI signal (0=bearish, 1=bullish)
   726→   726→  const rsiBull = rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   727→   727→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   728→   728→
   729→   729→  // MFI signal (volume-dependent)
   730→   730→  const mfiBull = hasVolume
   731→   731→    ? (mfi > 80 ? 0.85 : mfi > 60 ? 0.65 : mfi > 40 ? 0.45 : mfi > 20 ? 0.3 : 0.15)
   732→   732→    : 0.5; // neutral if no volume
   733→   733→
   734→   734→  // Stochastic signal
   735→   735→  const stochBull = stochK > 80 ? 0.85 : stochK > 60 ? 0.65 : stochK > 40 ? 0.45
   736→   736→    : stochK > 20 ? 0.3 : 0.15;
   737→   737→
   738→   738→  // ADX trend strength normalized (0-1)
   739→   739→  const adxNorm = Math.min(adx / 45, 1);
   740→   740→
   741→   741→  // Price vs MA100 position
   742→   742→  const maPosBull = MA100 > 0 ? clamp((price - MA100) / (MA100 * 0.05), 0, 1) : 0.5;
   743→   743→
   744→   744→  // Distance to S/R (proximity = more likely to react)
   745→   745→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0.5;
   746→   746→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0.5;
   747→   747→
   748→   748→  // Priors per category (from instructions): trend=0.28, oscillator=0.25, volume=0.15, volatility=0.10, leading=0.12, patterns=0.10
   749→   749→  // Adjusted: if no volume, redistribute 0.15 proportionally
   750→   750→  let priorTrend = 0.28, priorOsc = 0.25, priorVol = hasVolume ? 0.15 : 0;
   751→   751→  let priorVolatility = 0.10, priorLeading = 0.12, priorPattern = 0.10;
   752→   752→  if (!hasVolume) {
   753→   753→    const totalRemaining = priorTrend + priorOsc + priorVolatility + priorLeading + priorPattern;
   754→   754→    const redistribute = 0.15;
   755→   755→    priorTrend += redistribute * (priorTrend / totalRemaining);
   756→   756→    priorOsc += redistribute * (priorOsc / totalRemaining);
   757→   757→    priorVolatility += redistribute * (priorVolatility / totalRemaining);
   758→   758→    priorLeading += redistribute * (priorLeading / totalRemaining);
   759→   759→    priorPattern += redistribute * (priorPattern / totalRemaining);
   760→   760→  }
   761→   761→
   762→   762→  // P_Bull_Current = Σ(W_i × Signal_Bull_i) / Σ(W_i)
   763→   763→  const currentBullSignals = [
   764→   764→    { signal: rsiBull, weight: priorOsc * 0.4 },           // RSI (oscillator)
   765→   765→    { signal: mfiBull, weight: priorVol },                 // MFI (volume)
   766→   766→    { signal: stochBull, weight: priorOsc * 0.35 },        // Stochastic (oscillator)
   767→   767→    { signal: maPosBull, weight: priorTrend * 0.5 },       // MA position (trend)
   768→   768→    { signal: adxNorm > 0.5 && bull > 0.5 ? bull : 0.5, weight: priorTrend * 0.3 }, // ADX trend (trend)
   769→   769→    { signal: distS1, weight: priorPattern * 0.5 },        // Near support = bullish bounce
   770→   770→  ];
   771→   771→
   772→   772→  let sumWeight = 0;
   773→   773→  let sumWeightedBull = 0;
   774→   774→  for (const { signal, weight } of currentBullSignals) {
   775→   775→    if (weight <= 0) continue;
   776→   776→    sumWeightedBull += weight * signal;
   777→   777→    sumWeight += weight;
   778→   778→  }
   779→   779→  const P_Bull_Current = sumWeight > 0 ? clamp(sumWeightedBull / sumWeight, 0, 1) : bull;
   780→   780→  const P_Bear_Current = 1 - P_Bull_Current;
   781→   781→
   782→   782→  // ═══════════════════════════════════════════════════════════════════════
   783→   783→  // STEP 2: P_Bull_Forecast (from ML prediction, or decayed current)
   784→   784→  // ═══════════════════════════════════════════════════════════════════════
   785→   785→  // If ML prediction available, use predicted return to estimate future bull prob
   786→   786→  // Otherwise use exponential decay of current state
   787→   787→  const alpha = 0.5; // weight of current state (per instructions)
   788→   788→
   789→   789→  // Use ML predicted return if available, otherwise use consensus as proxy
   790→   790→  const mlReturn = mlResult?.predictionProb;
   791→   791→  let P_Bull_Forecast: number;
   792→   792→  if (mlReturn !== null && mlReturn !== undefined) {
   793→   793→    // ML gives direct P(bull) prediction — use it as forecast
   794→   794→    P_Bull_Forecast = clamp(mlReturn, 0, 1);
   795→   795→  } else {
   796→   796→    // Decay: forecast = current × decay + 0.5 × (1-decay)  (mean-revert to 0.5)
   797→   797→    const decayFactor = 0.85;
   798→   798→    P_Bull_Forecast = P_Bull_Current * decayFactor + 0.5 * (1 - decayFactor);
   799→   799→  }
   800→   800→  const P_Bear_Forecast = 1 - P_Bull_Forecast;
   801→   801→
   802→   802→  // ═══════════════════════════════════════════════════════════════════════
   803→   803→  // STEP 3: Final directional probabilities (combining current + forecast)
   804→   804→  // ═══════════════════════════════════════════════════════════════════════
   805→   805→  const P_Bull_Final = clamp(alpha * P_Bull_Current + (1 - alpha) * P_Bull_Forecast, 0.02, 0.98);
   806→   806→  const P_Bear_Final = clamp(alpha * P_Bear_Current + (1 - alpha) * P_Bear_Forecast, 0.02, 0.98);
   807→   807→  const P_Neutral_Final = Math.max(0.04, 1 - P_Bull_Final - P_Bear_Final);
   808→   808→
   809→   809→  // Normalize so they sum to 1
   810→   810→  const dirSum = P_Bull_Final + P_Bear_Final + P_Neutral_Final;
   811→   811→  const pBull = P_Bull_Final / dirSum;
   812→   812→  const pBear = P_Bear_Final / dirSum;
   813→   813→  const pNeutral = P_Neutral_Final / dirSum;
   814→   814→
   815→   815→  // ═══════════════════════════════════════════════════════════════════════
   816→   816→
   817→   817→  // Helper: Overbought/Oversold
   818→   818→  const overbought = rsi > 70 ? (rsi - 70) / 30 : 0;
   819→   819→  const oversold = rsi < 30 ? (30 - rsi) / 30 : 0;
   820→   820→
   821→   821→  // Helper: Volatility state
   822→   822→  const atrPct = price > 0 ? (atr || price * 0.02) / price : 0.02;
   823→   823→  const volHigh = clamp((atrPct - 0.02) / 0.03, 0, 1);
   824→   824→  const volLow = clamp(1 - (atrPct - 0.01) / 0.02, 0, 1);
   825→   825→
   826→   826→  // Helper: Divergence detection
   827→   827→  let divBull = 0;
   828→   828→  let divBear = 0;
   829→   829→  if (mlResult?.predictionProb !== undefined) {
   830→   830→    const mlBull = mlResult.predictionProb ?? 0.5;
   831→   831→    if (bull > 0.6 && mlBull < 0.4) divBear = (bull - mlBull) * 1.5;
   832→   832→    if (bull < 0.4 && mlBull > 0.6) divBull = (mlBull - bull) * 1.5;
   833→   833→  }
   834→   834→  divBull = clamp(divBull, 0, 0.3);
   835→   835→  divBear = clamp(divBear, 0, 0.3);
   836→   836→
   837→   837→  // Helper: Volume confirmation
   838→   838→  const volConfirm = hasVolume ? 0.6 : 0.4;
   839→   839→
   840→   840→  // Helper: ADX > 40 = strong trend
   841→   841→  const adxStrong = adx > 40 ? 1 : adx > 25 ? (adx - 25) / 15 : 0;
   842→   842→
   843→   843→  // ── Conditional probability distributions CP(S_i | State) ──
   844→   844→  // Per attachment instructions: 3-branch decision graph
   845→   845→  //   Branch A: Trend Following → scenarios A1-A9
   846→   846→  //   Branch B: Breakout Strategy → scenarios B1-B9
   847→   847→  //   Branch C: Reversal Strategy → scenarios C1-C9
   848→   848→  // Final: P(S_i) = P(Bull)×CP_i|Bull + P(Bear)×CP_i|Bear + P(Neutral)×CP_i|Neutral
   849→   849→  // Each CP distribution sums to 1 after normalization.
   850→   850→  //
   851→   851→  // ADX governs range/trend split (per attachment: ADX < 20 = weak trend)
   852→   852→  // FIX: Range scenarios capped so they never dominate (>40% of any CP dist)
   853→   853→
   854→   854→  // ADX-based trend strength (per attachment: ADX_Normalized = min(ADX/60, 1))
   855→   855→  const adxNormVal = Math.min(adx / 60, 1);
   856→   856→  // Range weight: inversely proportional to ADX, but capped at 0.35 (was 0.6!)
   857→   857→  const adxRange = adx < 20 ? 0.35 : adx < 30 ? 0.20 : 0.10;
   858→   858→  const adxTrend = 1 - adxRange;
   859→   859→
   860→   860→  // ── Per attachment edge formulas (lines 657-716): ──
   861→   861→  // A1: Strong Bull  = P_Bull × P_ADX>40 × P_Forecast_Bull_Strong
   862→   862→  // A2: Accel Bull   = P_Bull × P_Volatility_High × P_Volume_High_Future
   863→   863→  // A3: Cautious Bull= P_Bull × P_Divergence_Bearish_Future
   864→   864→  // A4: Strong Bear  = P_Bear × P_ADX>40 × P_Forecast_Bear_Strong
   865→   865→  // A5: Accel Bear   = P_Bear × P_Volatility_High × P_Volume_High_Future
   866→   866→  // A6: Cautious Bear= P_Bear × P_Divergence_Bullish_Future
   867→   867→  // A7: LowVol Range = P_Neutral × P_Volatility_Low_Future
   868→   868→  // A8: HighVol Range= P_Neutral × P_Volatility_High_Future
   869→   869→  // A9: Shock        = P_Shock × P_Forecast_Anomaly (0.02-0.05)
   870→   870→
   871→   871→  // CP_i|Bull: Bull scenarios dominate, range scenarios limited
   872→   872→  const cpBull = [
   873→   873→    // SC1: Strong Bull — ADX strength and trend quality
   874→   874→    0.28 * adxTrend * (0.5 + 0.5 * adxNormVal),
   875→   875→    // SC2: Accelerating Bull — momentum + volatility
   876→   876→    0.15 * adxTrend * (0.3 + 0.7 * factors.momentum),
   877→   877→    // SC3: Cautious Bull — reduced by overbought risk
   878→   878→    0.12 * (1 - overbought * 0.6) * adxTrend,
   879→   879→    // SC4: Strong Bear — small probability in bull market, driven by divergence
   880→   880→    0.03 * (1 + divBear),
   881→   881→    // SC5: Accelerating Bear — very small in bull market
   882→   882→    0.02,
   883→   883→    // SC6: Cautious Bear — driven by divergence warning
   884→   884→    0.03 * (1 + divBear),
   885→   885→    // SC7: Low-Vol Range — bounded, not dominant
   886→   886→    adxRange * 0.45 * volLow,
   887→   887→    // SC8: High-Vol Range — bounded
   888→   888→    adxRange * 0.30 * volHigh,
   889→   889→    // SC9: Shock — fixed small probability
   890→   890→    0.04 * (1 + overbought * 0.5 + volHigh * 0.5),
   891→   891→  ];
   892→   892→
   893→   893→  // CP_i|Bear: Mirror of bull (bear scenarios dominate)
   894→   894→  const cpBear = [
   895→   895→    // SC1: Strong Bull — small in bear market
   896→   896→    0.03 * (1 + divBull),
   897→   897→    // SC2: Accelerating Bull — very small
   898→   898→    0.02,
   899→   899→    // SC3: Cautious Bull — divergence-driven
   900→   900→    0.03 * (1 + divBull),
   901→   901→    // SC4: Strong Bear — ADX strength
   902→   902→    0.28 * adxTrend * (0.5 + 0.5 * adxNormVal),
   903→   903→    // SC5: Accelerating Bear — momentum
   904→   904→    0.15 * adxTrend * (0.3 + 0.7 * factors.momentum),
   905→   905→    // SC6: Cautious Bear — reduced by oversold risk
   906→   906→    0.12 * (1 - oversold * 0.6) * adxTrend,
   907→   907→    // SC7: Low-Vol Range — bounded
   908→   908→    adxRange * 0.45 * volLow,
   909→   909→    // SC8: High-Vol Range — bounded
   910→   910→    adxRange * 0.30 * volHigh,
   911→   911→    // SC9: Shock
   912→   912→    0.04 * (1 + oversold * 0.5 + volHigh * 0.5),
   913→   913→  ];
   914→   914→
   915→   915→  // CP_i|Neutral: More balanced distribution per attachment
   916→   916→  // Per attachment: neutral market has trend signals too, just weaker
   917→   917→  // Range gets adxRange but trend scenarios get reasonable weight
   918→   918→  const cpNeutral = [
   919→   919→    // SC1: Mild Bull — smaller in neutral
   920→   920→    0.08 * adxTrend * (0.5 + 0.5 * adxNormVal),
   921→   921→    // SC2: Bull Acceleration
   922→   922→    0.04 * adxTrend * (0.3 + 0.7 * factors.momentum),
   923→   923→    // SC3: Cautious Bull
   924→   924→    0.07 * adxTrend,
   925→   925→    // SC4: Mild Bear
   926→   926→    0.08 * adxTrend * (0.5 + 0.5 * adxNormVal),
   927→   927→    // SC5: Bear Acceleration
   928→   928→    0.04 * adxTrend * (0.3 + 0.7 * factors.momentum),
   929→   929→    // SC6: Cautious Bear
   930→   930→    0.07 * adxTrend,
   931→   931→    // SC7: Low-Vol Range — adxRange weighted, split between vol states
   932→   932→    adxRange * 0.50 * volLow,
   933→   933→    // SC8: High-Vol Range
   934→   934→    adxRange * 0.35 * volHigh,
   935→   935→    // SC9: Shock — small fixed
   936→   936→    0.04,
   937→   937→  ];
   938→   938→
   939→   939→  const normCP = (cp: number[]): number[] => {
   940→   940→    const s = cp.reduce((a, b) => a + b, 0);
   941→   941→    return s > 0 ? cp.map(v => v / s) : cp.map(() => 1 / 9);
   942→   942→  };
   943→   943→  const ncpBull = normCP(cpBull);
   944→   944→  const ncpBear = normCP(cpBear);
   945→   945→  const ncpNeutral = normCP(cpNeutral);
   946→   946→
   947→   947→  // ── Final probabilities per instructions formula ──
   948→   948→  // P(S_i) = P(Bull)*CP_i|Bull + P(Bear)*CP_i|Bear + P(Neutral)*CP_i|Neutral
   949→   949→  const rawS = [0, 0, 0, 0, 0, 0, 0, 0, 0];
   950→   950→  for (let i = 0; i < 9; i++) {
   951→   951→    rawS[i] = pBull * ncpBull[i] + pBear * ncpBear[i] + pNeutral * ncpNeutral[i];
   952→   952→  }
   953→   953→
   954→   954→  // Normalize to sum = 100
   955→   955→  const totalRaw = rawS.reduce((a, b) => a + b, 0);
   956→   956→  if (totalRaw <= 0) {
   957→   957→    return {
   958→   958→      pSC1: 12, pSC2: 10, pSC3: 8, pSC4: 12, pSC5: 10, pSC6: 8,
   959→   959→      pSC7: 15, pSC8: 15, pSC9: 10, factors,
   960→   960→    };
   961→   961→  }
   962→   962→
   963→   963→  const pSC1 = Math.round(rawS[0] / totalRaw * 100);
   964→   964→  const pSC2 = Math.round(rawS[1] / totalRaw * 100);
   965→   965→  const pSC3 = Math.round(rawS[2] / totalRaw * 100);
   966→   966→  const pSC4 = Math.round(rawS[3] / totalRaw * 100);
   967→   967→  const pSC5 = Math.round(rawS[4] / totalRaw * 100);
   968→   968→  const pSC6 = Math.round(rawS[5] / totalRaw * 100);
   969→   969→  const pSC7 = Math.round(rawS[6] / totalRaw * 100);
   970→   970→  const pSC8 = Math.round(rawS[7] / totalRaw * 100);
   971→   971→  const pSC9 = Math.max(1, 100 - (pSC1 + pSC2 + pSC3 + pSC4 + pSC5 + pSC6 + pSC7 + pSC8));
   972→   972→  return { pSC1, pSC2, pSC3, pSC4, pSC5, pSC6, pSC7, pSC8, pSC9, factors };
   973→   973→}

   975→   975→/**
   976→   976→ * Compute edge weights for the DAG (Directed Acyclic Graph) used in the
   977→   977→ * decision-graph scoring model.
   978→   978→ *
   979→   979→ * The four edge weights represent the transition probabilities and risk
   980→   980→ * parameters that govern how the DAG propagates scenario outcomes:
   981→   981→ * - **up**        — Probability-weighted upside edge, driven by bull consensus
   982→   982→ *                    and ML momentum coefficient.
   983→   983→ * - **down**      — Probability-weighted downside edge, driven by bear consensus
   984→   984→ *                    and ML trend coefficient.
   985→   985→ * - **pullback**  — Mean-reversion pullback magnitude, modulated by the ML
   986→   986→ *                    volatility coefficient.
   987→   987→ * - **risk**      — Position-sizing risk parameter; inversely scaled by ADX
   988→   988→ *                    strength and adjusted by the ML trend coefficient.
   989→   989→ *
   990→   990→ * When `mlResult` is trained, the coefficients for features at indices
   991→   991→ * `[11]=Trend, [0]=RSI, [10]=ATR_pct` are used to derive ML-driven
   992→   992→ * `{trendCoef, momentumCoef, volatilityCoef}` (normalised to sum to 1).
   993→   993→ * Otherwise, all three coefficients default to 1/3 (equal weighting).
   994→   994→ *
   995→   995→ * @param bullConsensus - Overall bull consensus ∈ [0,1]
   996→   996→ * @param adx          - Current ADX value (higher = stronger trend)
   997→   997→ * @param mlResult     - Trained adaptive model result, or `null` for equal weights
   998→   998→ * @returns Object with `{ up, down, pullback, risk }` — each clamped to a safe range
   999→   999→ */
  1000→  1000→export function calculateEdgeWeights(
  1001→  1001→  bullConsensus: number,
  1002→  1002→  adx: number,
  1003→  1003→  mlResult: AdaptiveModelResult | null,
  1004→  1004→): { up: number; down: number; pullback: number; risk: number } {
  1005→  1005→  // Default coefficients
  1006→  1006→  let trendCoef = 1 / 3;
  1007→  1007→  let momentumCoef = 1 / 3;
  1008→  1008→  let volatilityCoef = 1 / 3;
  1009→  1009→
  1010→  1010→  if (mlResult?.isTrained && mlResult.coefficients.length === 16) {
  1011→  1011→    const ec = mlResult.coefficients;
  1012→  1012→    const absTrend = Math.abs(ec[11] ?? 0);
  1013→  1013→    const absMomentum = Math.abs(ec[0] ?? 0);
  1014→  1014→    const absVolatility = Math.abs(ec[10] ?? 0);
  1015→  1015→    const total = absTrend + absMomentum + absVolatility;
  1016→  1016→    if (total > 0) {
  1017→  1017→      trendCoef = absTrend / total;
  1018→  1018→      momentumCoef = absMomentum / total;
  1019→  1019→      volatilityCoef = absVolatility / total;
  1020→  1020→    }
  1021→  1021→  }
  1022→  1022→
  1023→  1023→  // Dynamic base calculations
  1024→  1024→  const up_base = bullConsensus * 0.7 + 0.15;
  1025→  1025→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
  1026→  1026→  const pullback_base = 0.25;
  1027→  1027→  const risk_base = 0.12 * (1.2 - Math.min(adx / 100, 1));
  1028→  1028→
  1029→  1029→  // Apply ML-driven adjustments
  1030→  1030→  const up = clamp(up_base * (1 + (momentumCoef - 1 / 3) * 0.3), 0.05, 0.95);
  1031→  1031→  const down = clamp(down_base * (1 + (trendCoef - 1 / 3) * 0.3), 0.05, 0.95);
  1032→  1032→  const pullback = clamp(pullback_base * (1 + (volatilityCoef - 1 / 3) * 0.2), 0.05, 0.50);
  1033→  1033→  const risk = clamp(risk_base * (1 - (trendCoef - 1 / 3) * 0.5), 0.02, 0.30);
  1034→  1034→
  1035→  1035→  return { up, down, pullback, risk };
  1036→  1036→}

  1128→  1128→/**
  1129→  1129→ * Backward-compatible price prediction using a simple linear-regression-based
  1130→  1130→ * trend + mean-reversion + momentum model.
  1131→  1131→ *
  1132→  1132→ * ### Prediction Method
  1133→  1133→ * 1. Computes 20-bar daily returns and derives short (3-bar) and medium
  1134→  1134→ *    (10-bar) momentum averages.
  1135→  1135→ * 2. Fits a linear regression on the last 20 closes to extract slope and R².
  1136→  1136→ * 3. Calculates ATR-based daily volatility.
  1137→  1137→ * 4. Combines three daily-return components with adaptive weights:
  1138→  1138→ *    - **Trend** (weight = R² × 0.6): regression slope / price
  1139→  1139→ *    - **Mean-reversion** (weight = min(|deviation|×5, 0.4)): −deviation × 0.05
  1140→  1140→ *    - **Momentum** (weight = 0.2): 0.4 × mom3 + 0.1 × mom10
  1141→  1141→ * 5. Projects 5 daily prices with exponential decay (λ = 0.1).
  1142→  1142→ *
  1143→  1143→ * @param data - Full OHLCV history (chronological, oldest first). Needs ≥ 40 bars.
  1144→  1144→ * @returns {@link PricePrediction} with 5 predicted prices, 5-day return,
  1145→  1145→ *          confidence ∈ [0.1, 0.9], and trend direction, or `null` if
  1146→  1146→ *          insufficient data or zero price.
  1147→  1147→ */
  1148→  1148→export function predictPrices(data: OHLCV[]): PricePrediction | null {
  1149→  1149→  if (data.length < 40) return null;
  1150→  1150→  const PREDICT_DAYS = 5;
  1151→  1151→  const closes = data.map(d => d.close);
  1152→  1152→  const price = closes[closes.length - 1];
  1153→  1153→  if (price <= 0) return null;
  1154→  1154→
  1155→  1155→  const LOOKBACK = 20;
  1156→  1156→  const returns: number[] = [];
  1157→  1157→  for (let i = closes.length - LOOKBACK; i < closes.length; i++) {
  1158→  1158→    returns.push(i > 0 && closes[i - 1] > 0 ? (closes[i] - closes[i - 1]) / closes[i - 1] : 0);
  1159→  1159→  }
  1160→  1160→
  1161→  1161→  const mom3 = returns.slice(-3).reduce((a, b) => a + b, 0) / 3;
  1162→  1162→  const mom10 = returns.slice(-10).reduce((a, b) => a + b, 0) / 10;
  1163→  1163→  const ma20 = sma(closes, 20);
  1164→  1164→  const deviation = price > 0 && ma20 > 0 ? (price - ma20) / ma20 : 0;
  1165→  1165→
  1166→  1166→  const trendSlice = closes.slice(-20);
  1167→  1167→  const tn = trendSlice.length;
  1168→  1168→  let sx = 0, sy = 0, sxy = 0, sx2 = 0;
  1169→  1169→  for (let i = 0; i < tn; i++) { sx += i; sy += trendSlice[i]; sxy += i * trendSlice[i]; sx2 += i * i; }
  1170→  1170→  const d = tn * sx2 - sx * sx;
  1171→  1171→  let slope = 0, r2 = 0;
  1172→  1172→  if (d !== 0) {
  1173→  1173→    slope = (tn * sxy - sx * sy) / d;
  1174→  1174→    const meanY = sy / tn;
  1175→  1175→    let ssRes = 0, ssTot = 0;
  1176→  1176→    for (let i = 0; i < tn; i++) { ssRes += (trendSlice[i] - (slope * i + meanY - slope * (tn - 1) / 2)) ** 2; ssTot += (trendSlice[i] - meanY) ** 2; }
  1177→  1177→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
  1178→  1178→  }
  1179→  1179→
  1180→  1180→  let atr = 0;
  1181→  1181→  for (let i = Math.max(1, data.length - 14); i < data.length; i++) {
  1182→  1182→    atr += Math.max(data[i].high - data[i].low, Math.abs(data[i].high - data[i - 1].close), Math.abs(data[i].low - data[i - 1].close));
  1183→  1183→  }
  1184→  1184→  atr /= Math.min(14, data.length - 1);
  1185→  1185→  const dailyVol = price > 0 ? atr / price : 0.02;
  1186→  1186→
  1187→  1187→  const trendDaily = price > 0 ? slope / price : 0;
  1188→  1188→  const meanRevDaily = -deviation * 0.05;
  1189→  1189→  const momentumDaily = mom3 * 0.4 + mom10 * 0.1;
  1190→  1190→  const trendW = r2 * 0.6, mrW = Math.min(Math.abs(deviation) * 5, 0.4), momW = 0.2;
  1191→  1191→  const predDaily = trendDaily * trendW + meanRevDaily * mrW + momentumDaily * momW;
  1192→  1192→
  1193→  1193→  const predictedPrices: number[] = [];
  1194→  1194→  let cp = price;
  1195→  1195→  for (let dd = 1; dd <= PREDICT_DAYS; dd++) {
  1196→  1196→    const decay = Math.exp(-0.1 * (dd - 1));
  1197→  1197→    cp = cp * (1 + predDaily * decay);
  1198→  1198→    predictedPrices.push(Math.round(cp));
  1199→  1199→  }
  1200→  1200→
  1201→  1201→  const predictedReturn5d = (predictedPrices[PREDICT_DAYS - 1] - price) / price;
  1202→  1202→  return {
  1203→  1203→    predictedPrices,
  1204→  1204→    predictedReturn5d,
  1205→  1205→    confidence: Math.min(0.9, Math.max(0.1, r2 * 0.7 + 0.2)),
  1206→  1206→    trendDirection: predictedReturn5d > 0.005 ? 'up' : predictedReturn5d < -0.005 ? 'down' : 'flat',
  1207→  1207→  };
  1208→  1208→}

  1210→  1210→export function evaluatePredictedIndicators(data: OHLCV[], prediction: PricePrediction): { mlScore: number; predictedBullConsensus: number } {
  1211→  1211→  const price = data[data.length - 1].close;
  1212→  1212→  if (price <= 0 || prediction.predictedPrices.length === 0) return { mlScore: 0.5, predictedBullConsensus: 0.5 };
  1213→  1213→  const avgP = prediction.predictedPrices.reduce((a, b) => a + b, 0) / prediction.predictedPrices.length;
  1214→  1214→  const closes = data.map(d => d.close);
  1215→  1215→  const ma21 = sma(closes, 21), ma100 = sma(closes, 100), rsi = calcRSI(closes);
  1216→  1216→  const rsiP = Math.max(0, Math.min(1, (rsi + (avgP > price ? 5 : -5)) / 100));
  1217→  1217→  const ma21S = ma21 > 0 ? (avgP > ma21 ? 0.7 : 0.3) : 0.5;
  1218→  1218→  const ma100S = ma100 > 0 ? (avgP > ma100 ? 0.65 : 0.35) : 0.5;
  1219→  1219→  const dirS = avgP > price ? 0.65 : 0.35;
  1220→  1220→  return {
  1221→  1221→    mlScore: Math.max(0, Math.min(1, 0.5 + prediction.predictedReturn5d * 10 * prediction.confidence)),
  1222→  1222→    predictedBullConsensus: Math.max(0, Math.min(1, rsiP * 0.3 + ma21S * 0.25 + ma100S * 0.2 + dirS * 0.25)),
  1223→  1223→  };
  1224→  1224→}
