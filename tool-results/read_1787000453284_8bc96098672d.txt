    68→
    69→const toFa = (n: number) => Math.round(n).toLocaleString('fa-IR');

    75→  R1: { label: 'تداوم صعود هیجانی', type: 'up', border: '#59e39b', badgeBg: 'rgba(89,227,155,0.2)', badgeColor: '#59e39b' },
    76→  R2: { label: 'پولبک سالم', type: 'pullback', border: '#ffb25f', badgeBg: 'rgba(255,178,95,0.2)', badgeColor: '#ffb25f' },
    77→  R3: { label: 'اصلاح کنترل‌شده', type: 'down', border: '#ff758a', badgeBg: 'rgba(255,117,138,0.2)', badgeColor: '#ff758a' },
    78→  R4: { label: 'اصلاح عمیق', type: 'down', border: '#ffb11b', badgeBg: 'rgba(255,177,27,0.2)', badgeColor: '#ffb11b' },

    81→
    82→const STRATEGY_MAP: Record<string, { text: string; tagCls: string }> = {

    84→  R2: { text: 'صعود تدریجی — ورود در اصلاح توصیه می‌شود', tagCls: 'bg-cyan-500/15 text-cyan-400 border-cyan-500/30' },
    85→  R3: { text: 'بازار رنج — منتظر خروج از محدوده بمانید', tagCls: 'bg-amber-500/15 text-amber-400 border-amber-500/30' },
    86→  R4: { text: 'اصلاحی — احتیاط و کاهش حجم معاملات', tagCls: 'bg-orange-500/15 text-orange-400 border-orange-500/30' },
    87→  R5: { text: 'نزولی قوی — خروج فوری توصیه می‌شود', tagCls: 'bg-red-500/15 text-red-400 border-red-500/30' },
    88→};

    91→// Component
    92→// ═══════════════════════════════════════════════════════════════════════════════
    93→
    94→export default function VdesAnalysis(props: VdesAnalysisProps) {
    95→  const {
    96→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
    97→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
    98→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
    99→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
   100→    supportStrengths, resistanceStrengths, priceTargets,
   101→  } = props;
   102→
   103→  const containerRef = useRef<HTMLDivElement>(null);
   104→  const [exporting, setExporting] = useState(false);
   105→
   106→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   107→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   108→  const S1_level = supports[0] ?? currentPrice * 0.95;
   109→  const S2_level = supports[1] ?? currentPrice * 0.90;
   110→  const S3_level = supports[2] ?? currentPrice * 0.85;
   111→
   112→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   113→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   114→
   115→  // ── Dominant scenario ──────────────────────────────────────────
   116→  let highestKey = 'R3';
   117→  let highestProb = 0;
   118→  for (const key of SCENARIO_KEYS) {
   119→    if (scenarios[key].probability > highestProb) {
   120→      highestProb = scenarios[key].probability;
   121→      highestKey = key;
   122→    }
   123→  }
   124→  const strategy = STRATEGY_MAP[highestKey];
   125→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   126→
   127→  // ── RSI signal ─────────────────────────────────────────────────
   128→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   129→  const rsiColor = rsi > 70 ? 'text-red-400' : rsi > 60 ? 'text-amber-400' : rsi > 40 ? 'text-gray-300' : rsi > 30 ? 'text-amber-400' : 'text-emerald-400';
   130→
   131→  // ── Stochastic signal ──────────────────────────────────────────
   132→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   133→
   134→  // ── MACD signal ────────────────────────────────────────────────
   135→  const macdSignalText = macdHist > 0 ? 'مومنتوم مثبت (هیستوگرام بالای صفر)' : 'مومنتوم منفی (هیستوگرام زیر صفر)';
   136→  const macdBullish = macdLine > macdSignal;
   137→
   138→  // ── Trend text ─────────────────────────────────────────────────
   139→  const trendText = trendDirection === 'up'
   140→    ? `صعودی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${toFaD(trendR2 * 100)}٪`
   141→    : trendDirection === 'down'
   142→    ? `نزولی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${toFaD(trendR2 * 100)}٪`
   143→    : 'خنثی و بدون جهت مشخص';
   144→
   145→
   146→
   147→  // ── Bollinger Band position (simplified) ────────────────────────
   148→  const bbRange = bollingerUpper - bollingerLower;
   149→  const bbPos = bbRange > 0 ? toPersianDigits(((currentPrice - bollingerLower) / bbRange * 100).toFixed(0)) : '۵۰';
   150→  const bbSignal = currentPrice > bollingerUpper ? 'بالای باند بالایی (اشباع خرید)'
   151→    : currentPrice < bollingerLower ? 'زیر باند پایینی (اشباع فروش)'
   152→    : `داخل باندها (${bbPos}٪ از بازه)`;
   153→
   154→
   155→  // ── TradingView scenario colors ────────────────────────────────
   156→  // Limit candles for chart rendering to prevent freeze on large datasets (e.g. derivatives with 10k+ candles)
   157→  const chartCandles = useMemo(() => {
   158→    if (candles.length <= 2000) return candles;
   159→    return candles.slice(-2000);
   160→  }, [candles]);
   161→
   162→  const tvScenarios = useMemo(() => {
   163→    const colors: Record<string, string> = {
   164→      R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62',
   165→    };
   166→    return Object.fromEntries(
   167→      SCENARIO_KEYS.map(k => [k, {
   168→        targetMin: scenarios[k].targetMin,
   169→        targetMax: scenarios[k].targetMax,
   170→        name: scenarios[k].name,
   171→        probability: scenarios[k].probability,
   172→        color: colors[k],
   173→      }])
   174→    );
   175→  }, [scenarios]);
   176→
   177→  // ── Strategy recommendation text ────────────────────────────────
   178→  const mfiWarning = hasVolume ? ` (مقدار ${toFa(mfi)})` : '';
   179→  const strategyText = rsi > 70 || (hasVolume && mfi > 80)
   180→    ? `با توجه به هشدار اشباع خرید (قدرت نسبی: ${toFa(rsi)}${mfiWarning}) و فاصله قیمت تا مقاومت ${toFa(R1_level)}، استراتژی محتاطانه، انتظار برای اصلاح قیمت و ورود در محدوده حمایت ${toFa(S1_level)} تا ${toFa(S2_level)} ریال می‌باشد. در این محدوده می‌توان با حد ضرر ${toFa(S2_level)} ریال وارد موقعیت خرید شد.`
   181→    : rsi < 30 || (hasVolume && mfi < 20)
   182→    ? `با توجه به اشباع فروش (قدرت نسبی: ${toFa(rsi)}${mfiWarning}) و نزدیکی به حمایت ${toFa(S1_level)}، فرصت خرید در محدوده فعلی با حد ضرر ${toFa(S2_level)} ریال قابل بررسی است. هدف اولیه ${toFa(R1_level)} و هدف ثانویه ${toFa(R2_level)} ریال تعیین می‌شود.`
   183→    : `با توجه به وضعیت خنثی شاخص‌ها (قدرت نسبی: ${toFa(rsi)}، قدرت روند: ${adx > 25 ? 'قوی' : 'ضعیف'})، انتظار برای خروج قیمت از محدوده ${toFa(S1_level)} تا ${toFa(R1_level)} ریال و سپس تصمیم‌گیری توصیه می‌شود.${hasVolume ? ' حجم معاملات و مومنتوم کلی را برای تأیید سیگنال پایش کنید.' : ' مومنتوم کلی و حرکات قیمتی را برای تأیید سیگنال پایش کنید.'}`;
   184→
   185→  // ── Export HTML ───────────────────────────────────────────────
   186→  const handleExportHTML = useCallback(async () => {
   187→    const el = containerRef.current;
   188→    if (!el) return;
   189→    setExporting(true);
   190→    try {
   191→      const now = new Date();
   192→      const dateStr = toShamsi(now, 'long');
   193→      const html = `<!DOCTYPE html>
   194→<html lang="fa" dir="rtl">
   195→<head>
   196→<meta charset="UTF-8">
   197→<title>تحلیل تکنیکال ${symbolName}</title>
   198→<style>
   199→  body { font-family: Tahoma, Arial, sans-serif; background: #060a13; color: #e5e7eb; direction: rtl; padding: 24px; line-height: 1.85; }
   200→  h1 { color: #f8e365; font-size: 20px; }
   201→  .badge { display: inline-block; padding: 4px 14px; border-radius: 20px; border: 1px solid rgba(255,255,255,0.05); background: rgba(255,255,255,0.04); font-size: 12px; color: #9ca3af; margin-left: 8px; }
   202→  .badge b { color: #e5e7eb; }
   203→  .section { background: rgba(14,22,40,0.7); border: 1px solid rgba(255,255,255,0.06); border-radius: 14px; padding: 20px; margin: 16px 0; }
   204→  .footer { text-align: center; font-size: 11px; color: #4b5563; margin-top: 16px; padding-top: 12px; border-top: 1px solid rgba(255,255,255,0.05); }
   205→</style>
   206→</head>
   207→<body>
   208→<h1>📈 تحلیل تکنیکال ${symbolName}</h1>
   209→<p style="margin-top:12px">
   210→  <span class="badge">📍 قیمت مرجع: <b>${toFa(currentPrice)} ریال</b></span>
   211→  <span class="badge">🎯 هدف: <b>${toFa(targetMin)} — ${toFa(targetMax)} ریال</b></span>
   212→  <span class="badge">📊 روند: <b>${trendText}</b></span>
   213→  <span class="badge">RSI: <b>${toFa(rsi)} (${rsiSignal})</b></span>
   214→</p>
   215→<div class="section">
   216→${el.querySelector('.space-y-4')?.innerHTML || '<p>محتوای تحلیل در اینجا قرار می‌گیرد.</p>'}
   217→</div>
   218→<div class="footer">تاریخ تهیه: ${dateStr} — صرفاً جنبه تحلیلی دارد و توصیه سرمایه‌گذاری نیست.</div>
   219→</body>
   220→</html>`;
   221→      const blob = new Blob([html], { type: 'text/html;charset=utf-8' });
   222→      const url = URL.createObjectURL(blob);
   223→      const a = document.createElement('a');
   224→      a.href = url;
   225→      a.download = `${symbolName}-تحلیل-تکنیکال.html`;
   226→      a.click();
   227→      URL.revokeObjectURL(url);
   228→    } catch (err) {
   229→      console.error('HTML export failed:', err);
   230→    } finally {
   231→      setExporting(false);
   232→    }
   233→  }, [symbolName, currentPrice, targetMin, targetMax, trendText, rsi, rsiSignal]);
   234→
   235→  // ── Export PDF ────────────────────────────────────────────────
   236→  const handleExportPDF = useCallback(async () => {
   237→    const el = containerRef.current;
   238→    if (!el) return;
   239→    setExporting(true);
   240→    try {
   241→      const { toPng } = await import('html-to-image');
   242→      const { jsPDF } = await import('jspdf');
   243→      const dataUrl = await toPng(el, { backgroundColor: '#060a13', pixelRatio: 2, quality: 0.95 });
   244→      const img = new Image();
   245→      img.src = dataUrl;
   246→      await new Promise((resolve) => { img.onload = resolve; });
   247→      const pdfWidth = 595.28;
   248→      const pdfHeight = (img.height * pdfWidth) / img.width;
   249→      const pdf = new jsPDF({
   250→        orientation: pdfHeight > pdfWidth ? 'portrait' : 'landscape',
   251→        unit: 'pt',
   252→        format: [pdfWidth, Math.min(pdfHeight, 841.89)],
   253→      });
   254→      pdf.addImage(dataUrl, 'PNG', 0, 0, pdfWidth, pdfHeight);
   255→      pdf.save(`${symbolName}-تحلیل-تکنیکال.pdf`);
   256→    } catch (err) {
   257→      console.error('PDF export failed:', err);
   258→    } finally {
   259→      setExporting(false);
   260→    }
   261→  }, [symbolName]);
   262→
   263→  return (
   264→    <div ref={containerRef} className="space-y-5" dir="rtl">
   265→      {/* ═══ HEADER ═══ */}
   266→      <div className="rounded-[32px] px-6 py-5 border border-amber-500/20"
   267→        style={{
   268→          background: 'rgba(18,28,46,0.7)',
   269→          backdropFilter: 'blur(14px)',
   270→          boxShadow: '0 25px 50px -12px rgba(0,0,0,0.6)',
   271→        }}>
   272→        <h1 className="text-xl font-bold mb-1" style={{ color: '#f8e365' }}>
   273→          📈 تحلیل تکنیکال {symbolName}
   274→        </h1>
   275→        <div className="flex flex-wrap gap-3 mt-2">
   276→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   277→            📍 قیمت مرجع: <b className="text-gray-200">{toFa(currentPrice)} ریال</b>
   278→          </span>
   279→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   280→            🎯 هدف کوتاه‌مدت: <b className="text-gray-200">{toFa(targetMin)} — {toFa(targetMax)} ریال</b>
   281→          </span>
   282→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   283→            📊 روند: <b className="text-gray-200">{trendText}</b>
   284→          </span>
   285→          <span className={`px-3.5 py-1 rounded-full border text-xs font-medium ${
   286→            rsi > 70 ? 'bg-red-500/15 text-red-400 border-red-500/30'
   287→            : rsi < 30 ? 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30'
   288→            : 'bg-amber-500/15 text-amber-400 border-amber-500/30'
   289→          }`}>
   290→            قدرت نسبی: {toFa(rsi)} ({rsiSignal})
   291→          </span>
   292→        </div>
   293→      </div>
   294→
   295→      {/* ═══ TRADINGVIEW CHART ═══ */}
   296→      <div className="rounded-[28px] p-4 border border-white/6"
   297→        style={{
   298→          background: 'rgba(14,22,40,0.7)',
   299→          backdropFilter: 'blur(8px)',
   300→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   301→        }}>
   302→        <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   303→          <h2 className="text-sm font-semibold text-gray-200 flex items-center gap-3">
   304→            <span style={{ color: '#f8e365' }}>📊</span>
   305→            نمودار قیمتی سهم (روزانه) — به همراه MA21، MA100، سطوح حمایت/مقاومت و اهداف قیمتی
   306→          </h2>
   307→          <div className="flex items-center gap-2">
   308→            <button
   309→              onClick={handleExportHTML}
   310→              disabled={exporting}
   311→              className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-white/5 border border-white/10 text-xs text-gray-300 hover:bg-white/10 hover:text-white transition-all disabled:opacity-50 cursor-pointer"
   312→            >
   313→              {exporting ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <FileText className="w-3.5 h-3.5" />}
   314→              خروجی HTML
   315→            </button>
   316→            <button
   317→              onClick={handleExportPDF}
   318→              disabled={exporting}
   319→              className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-white/5 border border-white/10 text-xs text-gray-300 hover:bg-white/10 hover:text-white transition-all disabled:opacity-50 cursor-pointer"
   320→            >
   321→              {exporting ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Download className="w-3.5 h-3.5" />}
   322→              خروجی PDF
   323→            </button>
   324→          </div>
   325→        </div>
   326→        <TradingViewChart
   327→          symbolName={symbolName}
   328→          candles={chartCandles}
   329→          supports={supports}
   330→          resistances={resistances}
   331→          supportStrengths={supportStrengths}
   332→          resistanceStrengths={resistanceStrengths}
   333→          ma21={ma21}
   334→          ma100={ma100}
   335→          scenarios={tvScenarios}
   336→        />
   337→        <div className="mt-3 flex flex-wrap gap-4 text-[10px] text-gray-500">
   338→          <span>🔴 خطوط قرمز: مقاومت‌ها ({resistances.map(toFa).join(' ، ')})</span>
   339→          <span>🟢 خطوط سبز: حمایت‌ها ({supports.map(toFa).join(' ، ')})</span>
   340→          <span>🟣 خط بنفش: میانگین ۱۰۰ روزه ({toFa(ma100)})</span>
   341→          <span>🔵 خط آبی: میانگین ۲۱ روزه ({toFa(ma21)})</span>
   342→          <span>🟡 خطوط زرد: اهداف قیمتی سناریوها (رند شده)</span>
   343→        </div>
   344→      </div>
   345→
   346→      {/* ═══ DUAL ANALYSIS LAYOUT ═══ */}
   347→      <div className="rounded-[28px] p-5 border border-white/6"
   348→        style={{
   349→          background: 'rgba(14,22,40,0.7)',
   350→          backdropFilter: 'blur(8px)',
   351→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   352→        }}>
   353→        <h2 className="text-sm font-semibold mb-4 text-gray-200 flex items-center gap-3">
   354→          <span style={{ color: '#f8e365' }}>🧠</span>
   355→          تحلیل جامع روند و سطوح کلیدی
   356→        </h2>
   357→
   358→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-5">
   359→          {/* ── Text Analysis (Right in RTL) ── */}
   360→          <div className="rounded-2xl p-4 border border-white/4 space-y-4 flex flex-col"
