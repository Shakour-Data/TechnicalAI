   136→// Component
   137→// ═══════════════════════════════════════════════════════════════════════════════
   138→
   139→export default function VdssGraph(props: VdssGraphProps) {
   140→  const { colors: C } = useTheme();
   141→  const {
   142→    symbolName,
   143→    currentPrice,
   144→    scenarios,
   145→    decisionGraph,
   146→  } = props;
   147→
   148→  const [activeFilter, setActiveFilter] = useState('all');
   149→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   150→  const graphRef = useRef<HTMLDivElement>(null);
   151→  const svgRef = useRef<SVGSVGElement>(null);
   152→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   153→  const shellRef = useRef<HTMLDivElement>(null);
   154→
   155→  // ── Derived data from decisionGraph ──────────────────────────────
   156→  const edges = decisionGraph?.edges ?? [];
   157→  const nodes = decisionGraph?.nodes ?? [];
   158→  const nodePositions = decisionGraph?.nodePositions ?? {};
   159→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   160→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   161→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   162→  const pathContributions = decisionGraph?.pathContributions ?? {};
   163→
   164→  // Build node lookup map
   165→  const nodeMap = useMemo(() => {
   166→    const map: Record<string, (typeof nodes)[number]> = {};
   167→    for (const n of nodes) map[n.id] = n;
   168→    return map;
   169→  }, [nodes]);
   170→
   171→  // ── Visible edges/nodes based on filter ──────────────────────────
   172→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   173→    if (activeFilter === 'all') {
   174→      const allEdgeIdx = edges.map((_, i) => i);
   175→      const allNodes = new Set(nodes.map(n => n.id));
   176→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   177→    }
   178→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   179→      const edgeSet = new Set<number>();
   180→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   181→      edges.forEach((e, i) => {
   182→        if (e.to === activeFilter) {
   183→          edgeSet.add(i);
   184→          nodeSet.add(e.from);
   185→        }
   186→      });
   187→      edges.forEach((e, i) => {
   188→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   189→          edgeSet.add(i);
   190→          nodeSet.add(e.from);
   191→        }
   192→      });
   193→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   194→    }
   195→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   196→      const branchType = `branch-${activeFilter}` as const;
   197→      const edgeSet = new Set<number>();
   198→      const nodeSet = new Set<string>(['ROOT']);
   199→      edges.forEach((e, i) => {
   200→        if (e.type === branchType) {
   201→          edgeSet.add(i);
   202→          nodeSet.add(e.from);
   203→          nodeSet.add(e.to);
   204→        }
   205→      });
   206→      const expanded = new Set<number>();
   207→      const expandedNodes = new Set<string>(nodeSet);
   208→      edges.forEach((e, i) => {
   209→        if (expandedNodes.has(e.from)) {
   210→          expanded.add(i);
   211→          expandedNodes.add(e.to);
   212→        }
   213→      });
   214→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   215→    }
   216→    // Type filter (up/pullback/down/risk)
   217→    const edgeSet = new Set<number>();
   218→    const nodeSet = new Set<string>(['ROOT']);
   219→    edges.forEach((e, i) => {
   220→      if (e.type === activeFilter) {
   221→        edgeSet.add(i);
   222→        nodeSet.add(e.from);
   223→        nodeSet.add(e.to);
   224→      }
   225→    });
   226→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   227→  }, [activeFilter, edges, nodes]);
   228→
   229→  // ── Draw SVG edges ───────────────────────────────────────────────
   230→  const drawEdges = useCallback(() => {
   231→    const svg = svgRef.current;
   232→    const graph = graphRef.current;
   233→    if (!svg || !graph || edges.length === 0) return;
   234→
   235→    let markersSvg = '';
   236→    const seenColors = new Set<string>();
   237→    for (const e of edges) {
   238→      const c = EDGE_COLORS[e.type];
   239→      if (c && !seenColors.has(e.type)) {
   240→        seenColors.add(e.type);
   241→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   242→      }
   243→    }
   244→
   245→    let pathsSvg = '';
   246→    const graphRect = graph.getBoundingClientRect();
   247→
   248→    for (let ei = 0; ei < edges.length; ei++) {
   249→      const { from: fromId, to: toId, label, type } = edges[ei];
   250→      const isVisible = visibleEdgeIndices.includes(ei);
   251→      const fromEl = nodeRefs.current[fromId];
   252→      const toEl = nodeRefs.current[toId];
   253→      if (!fromEl || !toEl) continue;
   254→
   255→      const fRect = fromEl.getBoundingClientRect();
   256→      const tRect = toEl.getBoundingClientRect();
   257→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   258→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   259→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   260→      const by = tRect.top - graphRect.top + tRect.height / 2;
   261→      const dx = bx - ax;
   262→      const dy = by - ay;
   263→      const dist = Math.sqrt(dx * dx + dy * dy);
   264→
   265→      if (dist === 0) continue;
   266→
   267→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   268→      const mx = (ax + bx) / 2;
   269→      const my = (ay + by) / 2;
   270→      const nx = -dy / dist;
   271→      const ny = dx / dist;
   272→      const cx = mx + nx * bend;
   273→      const cy = my + ny * bend;
   274→
   275→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   276→      const prob = edgeProbabilities[ei] ?? 0;
   277→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   278→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   279→
   280→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.72 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   281→
   282→      if (isVisible && ei >= 6) {
   283→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#ffffff" font-size="10" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="4" stroke-linejoin="round" opacity="0.95" data-type="${type}" class="edge-label">${label}</text>`;
   284→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   285→      }
   286→    }
   287→
   288→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   289→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   290→
   291→  useEffect(() => { drawEdges(); }, [drawEdges]);
   292→
   293→  useEffect(() => {
   294→    const el = shellRef.current;
   295→    if (!el) return;
   296→    let observer: ResizeObserver | null = null;
   297→    try {
   298→      observer = new ResizeObserver(() => { try { drawEdges(); } catch {} });
   299→      observer.observe(el);
   300→    } catch {}
   301→    return () => { try { observer?.disconnect(); } catch {} };
   302→  }, [drawEdges]);
   303→
   304→
   305→  // ── All filter buttons ────────────────────────────────────────────
   306→  const allFilters = [
   307→    ...TYPE_FILTERS,
   308→    { key: 'sep1', label: '│', isSep: true as const },
   309→    ...Object.entries(BRANCH_META).map(([k, v]) => ({ key: k, label: v.label, branchKey: k, branchColor: v.color })),
   310→    { key: 'sep2', label: '│', isSep: true as const },
   311→    ...SCENARIO_KEYS.map(k => ({ key: k, label: SCENARIO_META_LOCAL[k].label, scenarioKey: k })),
   312→  ];
   313→
   314→  // ── Corrected probabilityTrend: day 1 overridden with final scenario probabilities ──
   315→  const correctedTrend = useMemo(() => {
   316→    const raw = props.probabilityTrend;
   317→    if (!raw || !raw.scenarios || raw.scenarios.length === 0) return raw;
   318→
   319→    // Build final individual probabilities (matching boxes) for day 1
   320→    const day1Individuals: Record<string, number> = {};
   321→    for (const key of SCENARIO_KEYS) {
   322→      const s = scenarios[key as keyof typeof scenarios];
   323→      day1Individuals[key] = (s?.probability ?? 0) / 100;
   324→    }
   325→
   326→    // Corrected scenarios array
   327→    const correctedScenarios: ScenarioTrend[] = raw.scenarios.map(sc => {
   328→      const newInd = day1Individuals[sc.scenarioKey] ?? sc.trend[0]?.individualProb ?? 0;
   329→      const newCum = calculateCDF(sc.scenarioKey, day1Individuals);
   330→      const newTrend: DayPoint[] = sc.trend.map((dp, i) => {
   331→        if (i === 0) {
   332→          return { ...dp, individualProb: newInd, cumulativeProb: newCum };
   333→        }
   334→        return dp;
   335→      });
   336→      return {
   337→        ...sc,
   338→        currentProbability: newInd,
   339→        trend: newTrend,
   340→        peakProbability: Math.max(newInd, sc.peakProbability),
   341→      };
   342→    });
   343→
   344→    // Corrected groups
   345→    const correctedGroups = raw.groups.map(g => {
   346→      // Group cumulative for day 1
   347→      const day1AllInds: Record<string, number> = {};
   348→      for (const sc of raw.scenarios) {
   349→        day1AllInds[sc.scenarioKey] = day1Individuals[sc.scenarioKey] ?? sc.trend[0]?.individualProb ?? 0;
   350→      }
   351→      let groupCumDay1 = 0;
   352→      if (g.group === 'bullish') {
   353→        groupCumDay1 = calculateCDF('SC6', day1AllInds);
   354→      } else if (g.group === 'bearish') {
   355→        groupCumDay1 = calculateCDF('SC4', day1AllInds);
   356→      } else {
   357→        groupCumDay1 = day1AllInds['SC5'] ?? 0;
   358→      }
   359→
   360→      const newTrend: DayPoint[] = g.trend.map((dp, i) => {
   361→        if (i === 0) {
   362→          return { ...dp, individualProb: groupCumDay1, cumulativeProb: groupCumDay1 };
   363→        }
   364→        return dp;
   365→      });
   366→      return {
   367→        ...g,
   368→        trend: newTrend,
   369→        peakProbability: Math.max(groupCumDay1, g.peakProbability),
   370→      };
   371→    });
   372→
   373→    return { ...raw, scenarios: correctedScenarios, groups: correctedGroups };
   374→  }, [props.probabilityTrend, scenarios]);
   375→
   376→  // ── Meaningful node display values (Persian-formatted) ──────────
   377→  const nodeDisplayValues = useMemo(() => {
   378→    const display: Record<string, string> = {};
   379→    if (!decisionGraph) return display;
   380→
   381→    // Sub-branch parent mapping
   382→    const subParents: Record<string, string> = {
   383→      N_T_BULL: 'N_TREND', N_T_BEAR: 'N_TREND', N_T_FLAT: 'N_TREND',
   384→      N_B_UP: 'N_BREAK', N_B_DOWN: 'N_BREAK', N_B_NONE: 'N_BREAK',
   385→      N_R_BULL: 'N_REVERSAL', N_R_BEAR: 'N_REVERSAL', N_R_NONE: 'N_REVERSAL',
   386→    };
   387→
   388→    // Find edge probability from parent to child
   389→    const getEdgeProb = (from: string, to: string): number => {
   390→      for (let i = 0; i < edges.length; i++) {
   391→        if (edges[i].from === from && edges[i].to === to) {
   392→          return edgeProbabilities[i] ?? 0;
   393→        }
   394→      }
   395→      return 0;
   396→    };
   397→
   398→    // ROOT: no numeric value needed
   399→    display['ROOT'] = '';
   400→
   401→    // Main branches: absolute probability from root
   402→    display['N_TREND'] = toPersianDigits((branchProbs.trend * 100).toFixed(0)) + '٪';
   403→    display['N_BREAK'] = toPersianDigits((branchProbs.breakout * 100).toFixed(0)) + '٪';
   404→    display['N_REVERSAL'] = toPersianDigits((branchProbs.reversal * 100).toFixed(0)) + '٪';
   405→
   406→    // Sub-branches: conditional probability within parent (meaningful context)
   407→    for (const [sub, parent] of Object.entries(subParents)) {
   408→      const condProb = getEdgeProb(parent, sub);
   409→      display[sub] = toPersianDigits((condProb * 100).toFixed(0)) + '٪';
   410→    }
   411→
   412→    // Event/assessment nodes: no value (they represent conditions, not probability outcomes)
   413→    for (const node of nodes) {
   414→      if (node.type === 'event') {
   415→        display[node.id] = '';
   416→      }
   417→    }
   418→
   419→    // Terminal nodes (SC1-SC9): handled by the big badge, no extra text
   420→    for (const key of SCENARIO_KEYS) {
   421→      display[key] = '';
   422→    }
   423→
   424→    return display;
   425→  }, [decisionGraph, edges, edgeProbabilities, branchProbs, nodes]);
   426→
   427→  // ═══ Design dimensions ═══
   428→  const DESIGN_W = 1500;
   429→  const DESIGN_H = 1200;
   430→  const DISPLAY_W = 1200;
   431→  const DISPLAY_H = 1200;
   432→  const scaleX = DISPLAY_W / DESIGN_W;
   433→  const scaleY = DISPLAY_H / DESIGN_H;
   434→
   435→  // ── Loading state ────────────────────────────────────────────────
   436→  if (!decisionGraph) {
   437→    return <VdssGraphSkeleton />;
   438→  }
   439→
   440→  return (
   441→    <div className="space-y-3" dir="rtl">
   442→      {/* ═══ HEADER ═══ */}
   443→      <div className="flex items-center justify-between gap-4 px-5 py-4 rounded-2xl"
   444→        style={{
   445→          border: '1px solid rgba(58,213,219,.22)',
   446→          background: 'linear-gradient(105deg, rgba(14,35,53,.94), rgba(8,22,35,.77))',
   447→          boxShadow: D.shadow,
   448→        }}>
   449→        <div className="flex items-center gap-3.5">
   450→          <div style={{
   451→            width: 48, height: 48, display: 'grid', placeItems: 'center',
   452→            border: '1px solid rgba(255,177,27,.7)', borderRadius: 14,
   453→            color: D.gold, fontSize: 27,
   454→            boxShadow: 'inset 0 0 22px rgba(255,177,27,.12), 0 0 22px rgba(255,177,27,.08)',
   455→          }}>◈</div>
   456→          <div>
   457→            <h2 style={{ fontSize: 20, fontWeight: 700, margin: '0 0 6px', color: D.text }}>گراف تصمیم {symbolName}</h2>
   458→            <p style={{ fontSize: 12, color: D.muted, margin: 0 }}>مدل ۳‌شاخه‌ای | پیروی از روند، شکست، بازگشت | ۲۷ مسیر به ۹ سناریو</p>
   459→          </div>
   460→        </div>
   461→        <div className="text-left text-xs leading-loose pr-4"
   462→          style={{ color: '#ffffff', borderRight: '1px solid rgba(58,213,219,.24)' }}>
   463→          نقطهٔ مرجع: <b style={{ color: D.cyan }}>{toFa(currentPrice)}</b><br />
   464→          افق برآورد: ۱۰ تا ۲۵ جلسه معاملاتی
   465→        </div>
   466→      </div>
   467→
   468→      {/* ═══ Metric Cards ═══ */}
   469→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   470→        <DarkMetricCard label="مقدار مرجع" value={toFa(currentPrice) + ' ' + (props.currencyUnit ?? 'ریال')} color={D.cyan} />
   471→        <DarkMetricCard label="احتمال روند" value={toPersianDigits((branchProbs.trend * 100).toFixed(0)) + '٪'} color={D.cyan} />

   798→// Narrative Generation (deterministic, algorithmic Persian text)
   799→// ═══════════════════════════════════════════════════════════════════════════════
   800→
   801→export function generateDecisionGraphNarrative(p: {
   802→  symbolName: string;
   803→  currentPrice: number;
   804→  currencyUnit: string;
   805→  scenarios: VdssGraphProps['scenarios'];
   806→  decisionGraph: GraphData | null;
   807→  probabilityTrend: ProbabilityTrendResult | null;
   808→  branchProbs: { trend: number; breakout: number; reversal: number };
   809→  pathContributions: Record<string, { trend: number; breakout: number; reversal: number }>;
   810→  scenarioProbabilities: Record<string, number>;
   811→}): string {
   812→  const { symbolName, currentPrice, currencyUnit, scenarios, decisionGraph, probabilityTrend, branchProbs, pathContributions, scenarioProbabilities } = p;
   813→  const fa = (n: number) => toPersianDigits(Math.round(n).toLocaleString('en-US'));
   814→  const faPct = (n: number) => toPersianDigits((n * 100).toFixed(1));
   815→  const faPct0 = (n: number) => toPersianDigits((n * 100).toFixed(0));
   816→
   817→  const lines: string[] = [];
   818→
   819→  // ── 1. Graph explanation ──
   820→  lines.push(`گراف تصمیم: ${symbolName}`);
   821→  lines.push('');
   822→  lines.push('گراف تصمیم یک مدل محاسباتی بر پایه گراف جریانی مسیردار (DAG) است که برای تخمین احتمال 9 سناریوی آینده قیمت طراحی شده است. این گراف دارای 3 شاخه اصلی استراتژی شامل پیرویی از روند، شکست و بازگشت می باشد. هر شاخه 3 زیرشاخه دارد و هر زیرشاخه 3 یال شرطی به 3 سناریوی نهایی متصل می شود که در مجموع 27 مسیر مستقل را تشکیل می دهند. احتمال نهایی هر سناریو برابر با جمع احتمالات تمامی مسیرهایی است که به آن سناریو ختم می شوند.');
   823→  lines.push(`نقطه مرجع محاسبات: ${fa(currentPrice)} ${currencyUnit} | افق برآورد: 10 تا 25 جلسه معاملاتی.`);
   824→  lines.push('');
   825→
   826→  // ── 2. Exclusive probabilities ──
   827→  const ranked = SCENARIO_KEYS.map(k => ({
   828→    key: k,
   829→    prob: scenarios[k as keyof typeof scenarios]?.probability ?? 0,
   830→    label: SCENARIO_META_LOCAL[k].label,
   831→  })).sort((a, b) => b.prob - a.prob);
   832→
   833→  lines.push('احتمالات اختصاصی (Exclusive Probabilities)');
   834→  lines.push('');
   835→  lines.push('احتمال اختصاصی هر سناریو نشان دهنده احتمال وقوع دقیق آن سناریو در افق زمانی مورد نظر است. مجموع 9 احتمال باید برابر 100 درصد باشد:');
   836→  lines.push('');
   837→
   838→  for (const r of ranked) {
   839→    const s = scenarios[r.key as keyof typeof scenarios];
   840→    const targetRange = s ? `${fa(s.targetMin)} تا ${fa(s.targetMax)} ${currencyUnit}` : '';
   841→    lines.push(`• ${r.key} — ${r.label}: ${toPersianDigits(String(r.prob))}٪ (${faPct0(r.prob / 100)} درصد)${targetRange ? ' | بازه: ' + targetRange : ''}`);
   842→  }
   843→
   844→  const totalProb = ranked.reduce((s, r) => s + r.prob, 0);
   845→  lines.push(`
   846→مجموع: ${toPersianDigits(String(totalProb))}٪ — ${totalProb === 100 ? 'صحیح' : 'خطا در مجموع'}`);
   847→  lines.push('');
   848→
   849→  // Interpretation of top scenario
   850→  const topSc = ranked[0];
   851→  if (topSc) {
   852→    const group = SCENARIO_META[topSc.key]?.group;
   853→    const groupLabel = group === 'bullish' ? 'گاوی' : group === 'bearish' ? 'خرسی' : 'خنثی';

   870→  lines.push('');

   873→  if (bullishProb > bearishProb && bullishProb > neutralProb) {

   934→  lines.push(`احتمال شاخه‌ها: پیرویی از روند ${faPct0(branchProbs.trend)}٪، شکست ${faPct0(branchProbs.breakout)}٪، بازگشت ${faPct0(branchProbs.reversal)}٪`);
   935→  lines.push('');
   936→
   937→  for (const key of SCENARIO_KEYS) {
   938→    const contrib = pathContributions[key] ?? { trend: 0, breakout: 0, reversal: 0 };

   942→  }
   943→  lines.push('');
   944→
   945→  return lines.join('\n');
   946→}

  1012→
  1013→function generateAdvancedDGAnalysis(p: {
  1014→  symbolName: string;
  1015→  currentPrice: number;
  1016→  currencyUnit: string;
  1017→  scenarios: VdssGraphProps['scenarios'];
  1018→  decisionGraph: GraphData | null;
  1019→  probabilityTrend: ProbabilityTrendResult | null;
  1020→  rsi: number;
  1021→  adx: number;
  1022→  atr: number;
  1023→  trendDirection: string;
  1024→  bullScore: number;
  1025→  resistances: number[];
  1026→  supports: number[];
  1027→  instrumentType?: string;
  1028→  instrumentCategory?: string;
  1029→}): string {
  1030→  const { symbolName, currentPrice, currencyUnit, scenarios, decisionGraph, probabilityTrend,
  1031→    rsi, adx, atr, trendDirection, bullScore, resistances, supports, instrumentCategory } = p;
  1032→
  1033→  const unit = currencyUnit || 'ریال';
  1034→  const fa = (n: number) => toPersianDigits(Math.round(n).toLocaleString('en-US'));
  1035→  const pctW = (n: number) => `${toPersianDigits(String(Math.round(n * 100)))} درصد`;
  1036→
  1037→  // ── Compute derived data ──
  1038→  const ranked = SCENARIO_KEYS.map(k => ({
  1039→    key: k,
  1040→    prob: scenarios[k as keyof typeof scenarios]?.probability ?? 0,
  1041→    label: SCENARIO_META_LOCAL[k].label,
  1042→    group: SCENARIO_META[k]?.group ?? 'neutral',
  1043→    targetMin: scenarios[k as keyof typeof scenarios]?.targetMin ?? 0,
  1044→    targetMax: scenarios[k as keyof typeof scenarios]?.targetMax ?? 0,
  1045→  })).sort((a, b) => b.prob - a.prob);
  1046→
  1047→  const bearProb = ranked.filter(r => r.group === 'bearish').reduce((s, r) => s + r.prob, 0);
  1048→  const bullProb = ranked.filter(r => r.group === 'bullish').reduce((s, r) => s + r.prob, 0);
  1049→  const neutProb = ranked.filter(r => r.group === 'neutral').reduce((s, r) => s + r.prob, 0);
  1050→
  1051→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.333, breakout: 0.333, reversal: 0.334 };
  1052→  const pathContribs = decisionGraph?.pathContributions ?? {} as Record<string, { trend: number; breakout: number; reversal: number }>;
  1053→
  1054→  const topSc = ranked[0];
  1055→  const secondSc = ranked[1];
  1056→  const topGroup = topSc?.group ?? 'neutral';
  1057→  const topGroupLabel = topGroup === 'bullish' ? 'گاوی' : topGroup === 'bearish' ? 'خرسی' : 'خنثی';
  1058→
  1059→  const R1 = resistances[0] ?? 0;
  1060→  const S1 = supports[0] ?? 0;
  1061→  const rrRatio = (R1 > 0 && S1 > 0 && currentPrice > S1)
  1062→    ? ((R1 - currentPrice) / (currentPrice - S1)).toFixed(1) : null;
  1063→
  1064→  const rsiZone = rsi > 70 ? 'اشباع خرید' : rsi > 60 ? 'نزدیک اشباع خرید' : rsi < 30 ? 'اشباع فروش' : rsi < 40 ? 'نزدیک اشباع فروش' : 'منطقه نرمال';
  1065→  const adxStrength = adx > 40 ? 'روند قدرتمند' : adx > 25 ? 'روند متوسط' : 'روند ضعیف یا رنج';
  1066→  const atrPct = currentPrice > 0 ? (atr / currentPrice * 100).toFixed(1) : '0';
  1067→  const volLevel = parseFloat(atrPct) > 3 ? 'نوسان بالا' : parseFloat(atrPct) > 1.5 ? 'نوسان متوسط' : 'نوسان کم';
  1068→
  1069→  // Branch dominance
  1070→  const branches = [
  1071→    { key: 'trend', label: 'پیروی از روند', prob: branchProbs.trend, color: 'emerald' as const },
  1072→    { key: 'breakout', label: 'شکست', prob: branchProbs.breakout, color: 'amber' as const },
  1073→    { key: 'reversal', label: 'بازگشت', prob: branchProbs.reversal, color: 'purple' as const },
  1074→  ].sort((a, b) => b.prob - a.prob);
  1075→  const dominantBranch = branches[0];
  1076→  const weakBranch = branches[2];
  1077→
  1078→  // Probability trend analysis
  1079→  const risingScs: string[] = [];
  1080→  const fallingScs: string[] = [];
  1081→  const stableScs: string[] = [];
  1082→  if (probabilityTrend?.scenarios?.length === 9) {
  1083→    for (const sc of probabilityTrend.scenarios) {
  1084→      const meta = SCENARIO_META_LOCAL[sc.scenarioKey];
  1085→      const entry = `${meta?.label ?? sc.label}`;
  1086→      if (sc.trendDirection === 'rising') risingScs.push(entry);
  1087→      else if (sc.trendDirection === 'falling') fallingScs.push(entry);
  1088→      else stableScs.push(entry);
  1089→    }
  1090→  }
  1091→
  1092→  // Dominant branch contribution to top scenarios
  1093→  const topScContrib = pathContribs[topSc?.key ?? 'SC5'] ?? { trend: 0, breakout: 0, reversal: 0 };
  1094→  const topScDominantStrategy = Object.entries(topScContrib).sort(([,a],[,b]) => (b as number) - (a as number))[0];
  1095→  const topScDominantLabel = topScDominantStrategy?.[0] === 'trend' ? 'پیروی از روند'
  1096→    : topScDominantStrategy?.[0] === 'breakout' ? 'شکست' : 'بازگشت';
  1097→
  1098→  // ── Build analysis paragraphs ──
  1099→  const paragraphs: string[] = [];
  1100→
  1101→  // ── Title ──
  1102→  paragraphs.push(`**تحلیل پیشرفته گراف تصمیم ${symbolName}** ⚡`);
  1103→  paragraphs.push('');
  1104→
  1105→  // ── Paragraph 1: Graph structure & overall picture ──
  1106→  const p1Parts: string[] = [];
  1107→  p1Parts.push(`گراف تصمیم ${symbolName} در نقطه مرجع {color:green}${fa(currentPrice)} ${unit}{/color} یک مدل تحلیلی سه لایه با {color:blue}${toPersianDigits(String(27))} مسیر مستقل{/color} است که از ترکیب 3 استراتژی اصلی شامل پیروی از روند، شکست و بازگشت تشکیل شده است.`);
  1108→
  1109→  if (topSc) {
  1110→    p1Parts.push(`سناریوی غالب با احتمال {color:${topGroup === 'bullish' ? 'green' : topGroup === 'bearish' ? 'red' : 'amber'}}${pctW(topSc.prob)}{/color}، **${topSc.label}** است که در گروه {color:${topGroup === 'bullish' ? 'green' : topGroup === 'bearish' ? 'red' : 'amber'}}${topGroupLabel}{/color} قرار می گیرد.`);
  1111→  }
  1112→
  1113→  if (secondSc) {
  1114→    p1Parts.push(`سناریوی دوم با ${pctW(secondSc.prob)} احتمال، ${secondSc.label} می باشد و فاصله احتمالی بین دو سناریوی اول و دوم برابر ${toPersianDigits(String(Math.round((topSc.prob - secondSc.prob) * 100)))} درصد است${(topSc.prob - secondSc.prob) > 10 ? ' که نشان دهنده تمرکز بالای احتمالات است' : ''}.`);
  1115→  }
  1116→
  1117→  p1Parts.push(`مجموع احتمال گروه گاوی {color:green}${pctW(bullProb)}{/color}، گروه خنثی ${pctW(neutProb)} و گروه خرسی {color:red}${pctW(bearProb)}{/color} محاسبه شده است.`);
  1118→
  1119→  if (bullProb > bearProb + 0.2) {
  1120→    p1Parts.push(`اختلاف قابل توجه ${toPersianDigits(String(Math.round((bullProb - bearProb) * 100)))} درصدی بین گروه گاوی و خرسی، چشم انداز صعودی بازار را تایید می کند.`);
  1121→  } else if (bearProb > bullProb + 0.2) {
  1122→    p1Parts.push(`اختلاف ${toPersianDigits(String(Math.round((bearProb - bullProb) * 100)))} درصدی به نفع گروه خرسی، هشدار جدی برای معاملات خرید صادر می کند.`);
  1123→  } else {
  1124→    p1Parts.push(`فاصله کم بین گروه گاوی و خرسی نشان دهنده **عدم قطعیت بازار** و لزوم مدیریت ریسک دقیق است.`);
  1125→  }
  1126→
  1127→  paragraphs.push(p1Parts.join(' '));
  1128→  paragraphs.push('');
  1129→
  1130→  // ── Paragraph 2: Branch strategy analysis ──
  1131→  const p2Parts: string[] = [];
  1132→  p2Parts.push(`در ساختار درختی گراف تصمیم، استراتژی **${dominantBranch.label}** با سهم {color:${dominantBranch.color}}${pctW(dominantBranch.prob)}{/color} بیشترین تاثیر را در تعیین سناریوهای نهایی دارد.`);
  1133→
  1134→  if (weakBranch) {
  1135→    p2Parts.push(`در مقابل، استراتژی ${weakBranch.label} تنها ${pctW(weakBranch.prob)} سهم دارد که نشان می دهد بازار کمتر انتظار ${weakBranch.key === 'reversal' ? 'واگرایی و بازگشت' : weakBranch.key === 'breakout' ? 'شکست سطوح' : 'ادامه روند'} را دارد.`);
  1136→  }
  1137→
  1138→  // Which branch feeds the top scenario most
  1139→  p2Parts.push(`بررسی سهم استراتژی‌ها در سناریوی غالب (${topSc?.label}) نشان می دهد که ${topScDominantLabel} با سهم ${pctW(topScContrib[topScDominantStrategy?.[0] as 'trend' | 'breakout' | 'reversal'] ?? 0)} بیشترین نقش را ایفا می کند.`);
  1140→
  1141→  // Branch-specific insights
  1142→  if (branchProbs.trend > 0.45) {
  1143→    p2Parts.push(`غلبگی استراتژی پیروی از روند (${pctW(branchProbs.trend)}) تایید می کند که بازار در فاز **رونددار** قرار دارد و تحلیل بر اساس ادامه مسیر فعلی قابل اتکاتر است.`);
  1144→  } else if (branchProbs.breakout > 0.4) {

  1244→/** Parse {color:...}text{/color} and **bold** from analysis text */
  1245→function renderDGAIText(text: string) {
  1246→  const lines = text.split('\n');
  1247→  return lines.map((line, li) => {
  1248→    if (!line.trim()) return <br key={li} />;
  1249→    let html = line;
  1250→    html = html.replace(/\{color:([a-z]+)\}([^\{]*?)\{\/color\}/g, (_m: string, color: string, inner: string) => {
  1251→      const colorMap: Record<string, string> = {
  1252→        red: '#ef4444', green: '#22c55e', amber: '#f59e0b', blue: '#3b82f6',
  1253→        orange: '#f97316', purple: '#a855f7', emerald: '#10b981',
  1254→      };
  1255→      return '<span style="color:' + (colorMap[color] || color) + ';font-weight:700">' + inner + '</span>';
  1256→    });
  1257→    html = html.replace(/\*\*([^*]+?)\*\*/g, '<strong>$1</strong>');
  1258→    const isHeading = /^\*{0,2}[^*]{3,40}:$/.test(line.trim());
  1259→    if (isHeading) {
  1260→      return <h3 key={li} style={{ fontSize: 14, fontWeight: 700, color: D.text, marginTop: 12, marginBottom: 4 }} dangerouslySetInnerHTML={{ __html: html }} />;
  1261→    }
  1262→    return <p key={li} style={{ margin: '2px 0', lineHeight: 2.1, fontSize: 13, color: D.text }} dangerouslySetInnerHTML={{ __html: html }} />;
  1263→  });
  1264→}
