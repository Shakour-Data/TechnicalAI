   105→   105→function srLineStyle(strength: number, isTarget: boolean): { lineWidth: number; lineStyle: 0 | 1 | 2; color: string } {
   106→   106→  if (isTarget) {
   107→   107→    return { lineWidth: 3, lineStyle: 0, color: '' };
   108→   108→  }
   109→   109→  if (strength >= 9) return { lineWidth: 4, lineStyle: 0, color: '' };
   110→   110→  if (strength >= 7) return { lineWidth: 3, lineStyle: 0, color: '' };
   111→   111→  if (strength >= 5) return { lineWidth: 2, lineStyle: 1, color: '' };
   112→   112→  if (strength >= 3) return { lineWidth: 2, lineStyle: 2, color: '' };
   113→   113→  return { lineWidth: 1, lineStyle: 2, color: '' };
   114→   114→}
