   112→   112→function MiniSparkline({ predictions, color, width = 120, height = 40 }: { predictions: PredictionPoint[]; color: string; width?: number; height?: number }) {
   113→   113→  if (predictions.length < 2) return null;
   114→   114→
   115→   115→  const prices = predictions.map(p => p.price);
   116→   116→  const min = Math.min(...prices, ...predictions.map(p => p.lower));
   117→   117→  const max = Math.max(...prices, ...predictions.map(p => p.upper));
   118→   118→  const range = max - min || 1;
   119→   119→  const pad = 4;
   120→   120→
   121→   121→  const toX = (i: number) => pad + (i / (predictions.length - 1)) * (width - 2 * pad);
   122→   122→  const toY = (v: number) => pad + (1 - (v - min) / range) * (height - 2 * pad);
   123→   123→
   124→   124→  // Main line
   125→   125→  const linePoints = predictions.map((p, i) => `${toX(i)},${toY(p.price)}`).join(' ');
   126→   126→  // Upper band
   127→   127→  const upperPoints = predictions.map((p, i) => `${toX(i)},${toY(p.upper)}`).join(' ');
   128→   128→  // Lower band
   129→   129→  const lowerPoints = predictions.map((p, i) => `${toX(i)},${toY(p.lower)}`).reverse().join(' ');
   130→   130→  // Fill area
   131→   131→  const fillPath = `M${upperPoints} L${lowerPoints} Z`;
   132→   132→
   133→   133→  return (
   134→   134→    <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-10" style={{ minWidth: width }}>
   135→   135→      <defs>
   136→   136→        <linearGradient id={`fill-${color.replace('#', '')}`} x1="0" y1="0" x2="0" y2="1">
   137→   137→          <stop offset="0%" stopColor={color} stopOpacity="0.15" />
   138→   138→          <stop offset="100%" stopColor={color} stopOpacity="0.02" />
   139→   139→        </linearGradient>
   140→   140→      </defs>
   141→   141→      <path d={fillPath} fill={`url(#fill-${color.replace('#', '')})`} />
   142→   142→      <polyline points={upperPoints} fill="none" stroke={color} strokeOpacity="0.3" strokeWidth="0.5" strokeDasharray="2,2" />
   143→   143→      <polyline points={predictions.map((p, i) => `${toX(i)},${toY(p.lower)}`).join(' ')} fill="none" stroke={color} strokeOpacity="0.3" strokeWidth="0.5" strokeDasharray="2,2" />
   144→   144→      <polyline points={linePoints} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   145→   145→      {predictions.map((p, i) => (
   146→   146→        <circle key={i} cx={toX(i)} cy={toY(p.price)} r="1.5" fill={color} />
   147→   147→      ))}
   148→   148→    </svg>
   149→   149→  );
   150→   150→}
