   157→
   158→  const IL = instrumentLabel;
   159→
   160→  const containerRef = useRef<HTMLDivElement>(null);
   161→  const textAnalysisRef = useRef<HTMLDivElement>(null);

   163→  const [exporting, setExporting] = useState(false);

   166→

   168→  const [aiAnalysis, setAiAnalysis] = useState<string | null>(null);
   169→  const [aiLoading, setAiLoading] = useState(false);
   170→  const [aiError, setAiError] = useState<string | null>(null);
   171→  const aiFetchIdRef = useRef(0);
   172→
   173→  // ── Analysis version selector (V7 locked / V8 / V9 locked / V10 locked / V11 active) ─────────
   174→  const [analysisVersion, setAnalysisVersion] = useState<7 | 8 | 9 | 10 | 11>(11);

   181→    return buildV11Scenarios({
   182→      currentPrice,
   183→      atr,

   186→      supports,
   187→      resistances,
   188→      bollingerUpper,
   189→      bollingerLower,
   190→      bollingerMiddle,
   191→      ma21,
   192→      ma100,
   193→    });
   194→  }, [analysisVersion, currentPrice, atr, overallSignal, scenarios, supports, resistances,
   195→      bollingerUpper, bollingerLower, bollingerMiddle, ma21, ma100]);
   196→
   197→  // ── V10 Scenarios (LOCKED — backward compat) ───────────────
   198→  const v10Result = useMemo(() => {
   199→    if (analysisVersion !== 10) return null;
   200→    return buildV10Locked({
   201→      currentPrice,
   202→      atr,
   203→      overallSignal: overallSignal as 'bullish' | 'bearish' | 'neutral',
   204→      v9Scenarios: scenarios,
   205→      supports,
   206→      resistances,
   207→      bollingerUpper,
   208→      bollingerLower,
   209→      bollingerMiddle,
   210→      ma21,
   211→      ma100,
   212→    });
   213→  }, [analysisVersion, currentPrice, atr, overallSignal, scenarios, supports, resistances,
   214→      bollingerUpper, bollingerLower, bollingerMiddle, ma21, ma100]);
   215→
   216→  // ── Fetch AI analysis (robust with logging + auto-retry) ─────────────
   217→  const fetchAiAnalysis = useCallback(async (isRetry = false) => {
   218→    const fetchId = ++aiFetchIdRef.current;
   219→    setAiLoading(true);
   220→    setAiError(null);
   221→    try {
   222→      const apiUrl = analysisVersion === 11 ? '/api/v11-analysis' : analysisVersion === 10 ? '/api/v10-analysis' : analysisVersion === 9 ? '/api/v9-analysis' : analysisVersion === 8 ? '/api/v8-analysis' : '/api/vdes-analysis';
   223→      // V10/V11: use S-key scenarios, otherwise use V9 R-key scenarios
   224→      const sKeyResult = analysisVersion === 11 ? v11Result : analysisVersion === 10 ? v10Result : null;
   225→      const activeScenarios = sKeyResult ? sKeyResult.scenarios : scenarios;
   226→      const requestBody: any = {
   227→          symbolName, currentPrice, ma21, ma100,
   228→          rsi, mfi, cci, adx, stochK, stochD,
   229→          macdLine, macdSignal, macdHist,
   230→          diPlus, diMinus, sar, atr, obv, hasVolume,
   231→          bollingerUpper, bollingerMiddle, bollingerLower,
   232→          trendDirection, trendAngle, trendR2, overallSignal,
   233→          scenarios: activeScenarios, resistances, supports,
   234→          resistanceStrengths, supportStrengths,
   235→          isIndex, isTgju, currencyUnit,
   236→        };
   237→      // V10/V11: pass dominant + highest cumulative keys for narrative consistency
   238→      if (sKeyResult) {
   239→        if (analysisVersion === 11) {
   240→          requestBody.v11DominantKey = sKeyResult.dominantKey;
   241→          requestBody.v11HighestCumulativeKey = sKeyResult.highestCumulativeKey;
   242→        } else if (analysisVersion === 10) {
   243→          requestBody.v10DominantKey = sKeyResult.dominantKey;
   244→          requestBody.v10HighestCumulativeKey = sKeyResult.highestCumulativeKey;
   245→        }
   246→      }
   247→      // V9/V10/V11: include OHLCV history for ML training/prediction
   248→      if ((analysisVersion === 9 || analysisVersion === 10 || analysisVersion === 11) && candles.length > 0) {
   249→        requestBody.ohlcvHistory = candles.map(c => ({
   250→          date: c.date, open: c.open, high: c.high,
   251→          low: c.low, close: c.close, volume: c.volume,
   252→        }));
   253→      }
   254→
   255→      // AbortController for 90s client-side timeout
   256→      const controller = new AbortController();
   257→      const timeoutId = setTimeout(() => controller.abort(), 90000);
   258→
   259→      console.log(`[VdesAnalysis v${analysisVersion}] Fetching ${apiUrl} for ${symbolName} (retry=${isRetry})`);
   260→      const res = await fetch(apiUrl, {
   261→        method: 'POST',
   262→        headers: { 'Content-Type': 'application/json' },
   263→        body: JSON.stringify(requestBody),
   264→        signal: controller.signal,
   265→      });
   266→      clearTimeout(timeoutId);
   267→
   268→      if (!res.ok) {
   269→        let errorMsg = `خطای سرور (${res.status})`;
   270→        try {
   271→          const errData = await res.json();
   272→          errorMsg = errData.error || errorMsg;
   273→        } catch {
   274→          // Response body is not JSON (e.g. 502 from Caddy)
   275→          if (res.status === 502) errorMsg = 'سرور در حال بارگذاری مجدد است. لطفاً چند ثانیه صبر کنید و دوباره تلاش کنید.';
   276→        }
   277→        console.error(`[VdesAnalysis v${analysisVersion}] API error ${res.status}:`, errorMsg);
   278→        throw new Error(errorMsg);
   279→      }
   280→      const data = await res.json();
   281→      if (fetchId === aiFetchIdRef.current) {
   282→        if (data.analysis && data.analysis.trim().length > 0) {
   283→          setAiAnalysis(data.analysis);
   284→          console.log(`[VdesAnalysis v${analysisVersion}] Got analysis for ${symbolName}: ${data.analysis.length} chars`);
   285→        } else {
   286→          console.error(`[VdesAnalysis v${analysisVersion}] Empty analysis for ${symbolName}`, data);
   287→          setAiError('مدل پاسخ خالی تولید کرد. لطفاً دوباره تلاش کنید.');
   288→        }
   289→        if (data._debug?.v11) setV11Debug({ ...data._debug.v11, v11Persona: data._debug.v11Persona, v11Template: data._debug.v11Template });
   290→        else if (data._debug?.v10) setV11Debug({ ...data._debug.v10, v11Persona: data._debug.v10Persona, v10Template: data._debug.v10Template });
   291→        else setV11Debug(null);
   292→      }
   293→    } catch (err) {
   294→      console.error(`[VdesAnalysis v${analysisVersion}] Error for ${symbolName}:`, err);
   295→      if (fetchId === aiFetchIdRef.current) {
   296→        if (err instanceof Error && err.name === 'AbortError') {
   297→          setAiError('زمان پاسخگویی به پایان رسید. لطفاً دوباره تلاش کنید.');
   298→        } else {
   299→          setAiError(err instanceof Error ? err.message : 'خطای ناشناخته');
   300→        }
   301→        // Auto-retry once on transient errors (503/429/timeout) after 5s
   302→        if (!isRetry) {
   303→          const errMsg = err instanceof Error ? err.message : '';
   304→          const isTransient = errMsg.includes('503') || errMsg.includes('429') || errMsg.includes('پایان رسید') || errMsg.includes('بارگذاری مجدد');
   305→          if (isTransient) {
   306→            console.log(`[VdesAnalysis] Auto-retry in 5s for ${symbolName}`);
   307→            setTimeout(() => {
   308→              if (fetchId === aiFetchIdRef.current) fetchAiAnalysis(true).catch(() => {});
   309→            }, 5000);
   310→          }
   311→        }
   312→      }
   313→    } finally {
   314→      if (fetchId === aiFetchIdRef.current) {
   315→        setAiLoading(false);
   316→      }
   317→    }
   318→  }, [symbolName, currentPrice, ma21, ma100, rsi, mfi, cci, adx, stochK, stochD,
   319→    macdLine, macdSignal, macdHist, diPlus, diMinus, sar, atr, obv, hasVolume,
   320→    bollingerUpper, bollingerMiddle, bollingerLower, trendDirection, trendAngle, trendR2,
   321→    overallSignal, scenarios, resistances, supports, resistanceStrengths, supportStrengths,
   322→    isIndex, isTgju, currencyUnit, analysisVersion, v11Result, v10Result]);
   323→
   324→  // ── Auto-fetch AI analysis only on symbol or version change ──
   325→  // NOT on every data refresh (auto-refresh every 15min would re-trigger
   326→  // a 10-25s LLM call, making the text disappear each time)
   327→  const fetchTriggerKey = `${symbolName}:${analysisVersion}`;
   328→  const lastAutoFetchKeyRef = useRef('');
   329→  // Keep a ref to fetchAiAnalysis so the effect always calls the latest version
   330→  const fetchAiAnalysisRef = useRef(fetchAiAnalysis);
   331→  fetchAiAnalysisRef.current = fetchAiAnalysis;
   332→  useEffect(() => {
   333→    if (fetchTriggerKey === lastAutoFetchKeyRef.current) return;

   335→      console.log('[VdesAnalysis] Skipping fetch: no symbol or invalid price', { symbolName, currentPrice });

   337→    }
   338→    lastAutoFetchKeyRef.current = fetchTriggerKey;
   339→    // Reset old text when switching symbols/versions
   340→    setAiAnalysis(null);
   341→    setAiError(null);
   342→    // Small delay to ensure all useMemo values (v11Result etc.) are computed
   343→    const timer = setTimeout(() => {
   344→      fetchAiAnalysisRef.current().catch((e) => {
   345→        console.error('[VdesAnalysis] Auto-fetch failed:', e);
   346→      });
   347→    }, 150);
   348→    return () => clearTimeout(timer);
   349→  }, [fetchTriggerKey, symbolName, currentPrice]);
   350→
   351→  // ── Close export menu on outside click ───────────────────────
   352→  useEffect(() => {
   353→    if (!exportMenuOpen) return;
   354→    const handler = (e: MouseEvent) => {
   355→      if (exportMenuRef.current && !exportMenuRef.current.contains(e.target as Node)) {
   356→        setExportMenuOpen(false);
   357→      }
   358→    };
   359→    document.addEventListener('mousedown', handler);
   360→    return () => document.removeEventListener('mousedown', handler);
   361→  }, [exportMenuOpen]);
   362→
   363→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   364→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   365→  const S1_level = supports[0] ?? currentPrice * 0.95;
   366→  const S2_level = supports[1] ?? currentPrice * 0.90;
   367→  const S3_level = supports[2] ?? currentPrice * 0.85;
   368→
   369→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   370→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   371→
   372→  // ── Dominant scenario (V9) ──────────────────────────────────────
   373→  let highestKey = 'R3';
   374→  let highestProb = 0;
   375→  for (const key of SCENARIO_KEYS) {
   376→    if (scenarios[key].probability > highestProb) {
   377→      highestProb = scenarios[key].probability;
   378→      highestKey = key;
   379→    }
   380→  }
   381→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   382→
   383→  // ── V10/V11 dominant + highest cumulative + total ────────────
   384→  const sKeyResult = analysisVersion === 11 ? v11Result : analysisVersion === 10 ? v10Result : null;
   385→  const v11DominantKey = sKeyResult?.dominantKey || 'S5';
   386→  const v11HighestCumKey = sKeyResult?.highestCumulativeKey || 'S5';
   387→  const v11TotalProb = sKeyResult
   388→    ? V11_KEYS.reduce((sum, k) => sum + (sKeyResult.scenarios[k]?.probability || 0), 0)
   389→    : 0;
   390→
   391→  // ── Sorted scenarios by target price (ascending) ─────────────
   392→  const sortedScenarios = useMemo(() => {
   393→    return [...SCENARIO_KEYS]
   394→      .map(k => ({ key: k, ...scenarios[k] }))
   395→      .sort((a, b) => a.targetMin - b.targetMin);
   396→  }, [scenarios]);
   397→
   398→  // ── Dynamic scenario labels (context-aware based on overallSignal) ─
   399→  const dynamicMeta = useMemo(() => getDynamicScenarioMeta(overallSignal), [overallSignal]);
   400→  const strategy = dynamicMeta[highestKey];
   401→
   402→  // ── Text variety: hash-based selector for varied language ──────────
   403→  const varietySeed = useMemo(() => {
   404→    const raw = `${rsi}${macdHist}${cci}${adx}${stochK}${trendAngle}${Math.round(currentPrice)}`;
   405→    let hash = 0;
   406→    for (let i = 0; i < raw.length; i++) hash = ((hash << 5) - hash + raw.charCodeAt(i)) | 0;
   407→    return Math.abs(hash);
   408→  }, [rsi, macdHist, cci, adx, stochK, trendAngle, currentPrice]);
   409→

  1011→// ── Level Box (Support / Resistance / Target) ────────────────────────────────
  1012→function LevelBox({ index, label, price, strength, isTarget, type, unit, grade, overlapCount, touchCount }: {
  1013→  index: number; label: string; price: number; strength: number; isTarget: boolean; type: 'support' | 'resistance' | 'target'; unit?: string;
  1014→  grade?: string; overlapCount?: number; touchCount?: number;
  1015→}) {
  1016→  const isSupport = type === 'support';
  1017→  const isTargetType = type === 'target';
  1018→
  1019→  const accentColor = isTargetType ? '#ffb11b' : isSupport ? '#34c98b' : '#ef4d62';
  1020→  const bgAccent = isTargetType ? 'rgba(255,177,27,0.06)' : isSupport ? 'rgba(52,201,139,0.06)' : 'rgba(239,77,98,0.06)';
  1021→  const borderAccent = isTargetType ? 'rgba(255,177,27,0.3)' : isSupport ? 'rgba(52,201,139,0.3)' : 'rgba(239,77,98,0.3)';
  1022→
  1023→  const strengthPct = Math.min(100, Math.max(0, strength * 10));
  1024→  const displayGrade = grade || (strength >= 8.5 ? 'بسیار قوی' : strength >= 7 ? 'قوی' : strength >= 5 ? 'متوسط' : strength >= 3 ? 'ضعیف' : 'بسیار ضعیف');
  1025→  const strengthColor = strength >= 8.5 ? '#34c98b' : strength >= 7 ? '#3ad5db' : strength >= 5 ? '#ffb11b' : strength >= 3 ? '#ff7b32' : '#ef4d62';
  1026→
  1027→  return (
  1028→    <div
  1029→      className="flex items-center gap-3 px-3.5 py-2.5 rounded-xl border transition-all hover:scale-[1.01]"
  1030→      style={{
  1031→        background: bgAccent,
  1032→        border: `1px solid ${borderAccent}`,
  1033→        borderRight: `3px solid ${accentColor}`,
  1034→      }}
  1035→    >
  1036→      {/* Label + Price */}
  1037→      <div className="flex-1 min-w-0">
  1038→        <div className="flex items-center gap-2">
  1039→          <span className="text-[11px] font-bold" style={{ color: accentColor }}>{label}</span>
  1040→          {isTarget && <span className="text-[9px]">🎯</span>}
  1041→          {overlapCount !== undefined && overlapCount > 0 && (
  1042→            <span className="text-[8px] px-1.5 py-0.5 rounded-full bg-gray-100 text-gray-500 font-medium">
  1043→              {toPersianDigits(overlapCount)} هم‌پوشانی
  1044→            </span>
  1045→          )}
  1046→        </div>
  1047→        <div className="flex items-baseline gap-1">
  1048→          <span className="text-sm font-black text-gray-900 tabular-nums" dir="ltr">{toPersianDigits(Math.round(price).toLocaleString('en-US'))}</span>
  1049→          <span className="text-[10px] text-gray-500">{unit || 'ریال'}</span>
  1050→        </div>
  1051→      </div>
  1052→
  1053→      {/* Strength + Grade */}
  1054→      <div className="flex flex-col items-end gap-1 min-w-[100px]">
  1055→        <div className="flex items-center gap-1.5">
  1056→          <span className="text-[9px] font-bold" style={{ color: strengthColor }}>{displayGrade}</span>
  1057→          <span className="text-[10px] font-black tabular-nums" style={{ color: strengthColor }}>{toPersianDigits(Math.round(strength * 10) / 10)}</span>
  1058→          <span className="text-[9px] text-gray-500">/۱۰</span>
  1059→        </div>
  1060→        <div className="w-full h-1.5 rounded-full bg-gray-100 overflow-hidden">
  1061→          <div
  1062→            className="h-full rounded-full transition-all duration-500"
  1063→            style={{ width: `${strengthPct}%`, backgroundColor: strengthColor }}
