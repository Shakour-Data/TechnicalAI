   150→// Graph Algorithms
   151→// ═══════════════════════════════════════════════════════════════════════════════
   152→
   153→interface EdgeInfo {
   154→  from: string;
   155→  to: string;
   156→  label: string;
   157→  type: string;
   158→  prob: number;
   159→}
   160→
   161→interface PathInfo {
   162→  nodes: string[];
   163→  edges: number[];
   164→  prob: number;
   165→  target: string;
   166→}
   167→
   168→// Calculate edge probabilities based on market conditions
   169→function calcEdgeProbabilities(
   170→  bullScore: number, adxVal: number
   171→): Record<string, number> {
   172→  const adxW = Math.min(adxVal / 100, 1);
   173→  const typeWeights: Record<string, number> = {
   174→    up: bullScore * 0.6 + adxW * 0.4,
   175→    pullback: 0.3,
   176→    down: (1 - bullScore) * 0.6 + adxW * 0.4,
   177→    risk: 0.15 * (1.2 - adxW),
   178→  };
   179→
   180→  // Build adjacency: for each node, group outgoing edges by type
   181→  const adj: Record<string, { idx: number; type: string }[]> = {};
   182→  EDGES.forEach((e, i) => {
   183→    if (!adj[e[0]]) adj[e[0]] = [];
   184→    adj[e[0]].push({ idx: i, type: e[3] });
   185→  });
   186→
   187→  const edgeProbs: Record<string, number> = {};
   188→
   189→  for (const [node, outEdges] of Object.entries(adj)) {
   190→    // Sum weights of outgoing edges
   191→    const totalWeight = outEdges.reduce((sum, e) => sum + (typeWeights[e.type] ?? 0.1), 0);
   192→    if (totalWeight === 0) {
   193→      outEdges.forEach(e => { edgeProbs[String(e.idx)] = 1 / outEdges.length; });
   194→      continue;
   195→    }
   196→    outEdges.forEach(e => {
   197→      edgeProbs[String(e.idx)] = (typeWeights[e.type] ?? 0.1) / totalWeight;
   198→    });
   199→  }
   200→
   201→  return edgeProbs;
   202→}
   203→
   204→// Find all paths from 'A' to terminal nodes using DFS
   205→function findAllPaths(edgeProbs: Record<string, number>): PathInfo[] {
   206→  const adj: Record<string, { to: string; edgeIdx: number }[]> = {};
   207→  EDGES.forEach((e, i) => {
   208→    if (!adj[e[0]]) adj[e[0]] = [];
   209→    adj[e[0]].push({ to: e[1], edgeIdx: i });
   210→  });
   211→
   212→  const paths: PathInfo[] = [];
   213→  const MAX_PATHS = 200;
   214→
   215→  function dfs(node: string, visited: Set<string>, currentPath: string[], edgeIndices: number[], currentProb: number) {
   216→    if (SCENARIO_KEYS.includes(node as any)) {
   217→      paths.push({ nodes: [...currentPath, node], edges: [...edgeIndices], prob: currentProb, target: node });
   218→      return;
   219→    }
   220→    if (paths.length >= MAX_PATHS) return;
   221→    if (visited.has(node)) return;
   222→    visited.add(node);
   223→
   224→    const outs = adj[node] ?? [];
   225→    for (const { to, edgeIdx } of outs) {
   226→      const ep = edgeProbs[String(edgeIdx)] ?? 0.1;
   227→      dfs(to, visited, [...currentPath, node], [...edgeIndices, edgeIdx], currentProb * ep);
   228→    }
   229→    visited.delete(node);
   230→  }
   231→
   232→  dfs('A', new Set(), [], [], 1);
   233→  return paths.sort((a, b) => b.prob - a.prob);
   234→}
