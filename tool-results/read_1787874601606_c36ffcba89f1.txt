   347→   347→/**
   348→   348→ * Parse a number that might contain Persian digits, commas, or other formatting.
   349→   349→ * Handles null/undefined/empty strings gracefully.
   350→   350→ */
   351→   351→function parsePersianNum(str: string | null | undefined | number): number {
   352→   352→  if (str === null || str === undefined) return 0;
   353→   353→  const s = String(str);
   354→   354→  if (!s || s.trim() === '-' || s.trim() === '') return 0;
   355→   355→  return Number(s.replace(/[,۰-۹]/g, (c: string) => {
   356→   356→    const persianDigits = '۰۱۲۳۴۵۶۷۸۹';
   357→   357→    const idx = persianDigits.indexOf(c);
   358→   358→    return idx >= 0 ? String(idx) : c === ',' ? '' : c;
   359→   359→  }));
   360→   360→}

   362→   362→/**
   363→   363→ * Check if a value looks like a date string (contains / or - and is not a number).
   364→   364→ */
   365→   365→function looksLikeDate(val: string | null | undefined | number): boolean {
   366→   366→  if (!val) return false;
   367→   367→  const s = String(val).trim();
   368→   368→  return /[\/\-]/.test(s) && isNaN(Number(s.replace(/[,۰-۹\/\-]/g, '')));
   369→   369→}
   370→   370→
   371→   371→/**
   372→   372→ * Check if a value looks like a percentage (ends with % or is small like 0.5 to 50).
   373→   373→ */
   374→   374→function looksLikePercentage(val: string | null | undefined | number): boolean {
   375→   375→  if (!val) return false;
   376→   376→  const s = String(val).trim();
   377→   377→  if (s.endsWith('%')) return true;
   378→   378→  const n = parsePersianNum(val);
   379→   379→  return !isNaN(n) && Math.abs(n) < 100 && n !== 0;
   380→   380→}

   382→   382→/**
   383→   383→ * Auto-detect column mapping for TGJU API response.
   384→   384→ * Expected format: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   385→   385→ * But the API may change column order.
   386→   386→ */
   387→   387→function detectColumnMap(row: (string | number)[]): {
   388→   388→  openIdx: number; highIdx: number; lowIdx: number; closeIdx: number;
   389→   389→  jalaliDateIdx: number; gregorianDateIdx: number;
   390→   390→} | null {
   391→   391→  if (!row || row.length < 6) return null;
   392→   392→
   393→   393→  let jalaliDateIdx = -1;
   394→   394→  let gregorianDateIdx = -1;
   395→   395→  const numericIndices: number[] = [];
   396→   396→  const percentageIndices: number[] = [];
   397→   397→
   398→   398→  for (let i = 0; i < row.length; i++) {
   399→   399→    const val = row[i];
   400→   400→    if (looksLikeDate(val)) {
   401→   401→      // Jalali dates have Persian digits or format like 1404/05/25
   402→   402→      const s = String(val);
   403→   403→      if (/[۰-۹]/.test(s) || /^14\d{2}/.test(s.replace(/[۰-۹]/g, (d: string) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d))))) {
   404→   404→        if (jalaliDateIdx === -1) jalaliDateIdx = i;
   405→   405→      } else {
   406→   406→        if (gregorianDateIdx === -1) gregorianDateIdx = i;
   407→   407→      }
   408→   408→    } else if (looksLikePercentage(val)) {
   409→   409→      percentageIndices.push(i);
   410→   410→    } else {
   411→   411→      const n = parsePersianNum(val);
   412→   412→      if (isFinite(n) && n !== 0) {
   413→   413→        numericIndices.push(i);
   414→   414→      }
   415→   415→    }
   416→   416→  }
   417→   417→
   418→   418→  // If we couldn't auto-detect dates, fall back to assumed positions
   419→   419→  if (jalaliDateIdx === -1 && row.length >= 8) {
   420→   420→    // Default assumption: row[6] = gregorian, row[7] = jalali
   421→   421→    if (looksLikeDate(row[7])) jalaliDateIdx = 7;
   422→   422→    else if (looksLikeDate(row[6])) jalaliDateIdx = 6;
   423→   423→    if (gregorianDateIdx === -1 && looksLikeDate(row[6]) && jalaliDateIdx !== 6) gregorianDateIdx = 6;
   424→   424→    else if (gregorianDateIdx === -1 && looksLikeDate(row[7]) && jalaliDateIdx !== 7) gregorianDateIdx = 7;
   425→   425→  }
   426→   426→
   427→   427→  // Numeric columns (excluding percentages) should be OHLC + change
   428→   428→  // OHLC: typically 4 largest absolute values among non-percentage numerics
   429→   429→  // change: the remaining one
   430→   430→  const ohlcCandidates = numericIndices.filter(i => !percentageIndices.includes(i));
   431→   431→
   432→   432→  if (ohlcCandidates.length < 4) {
   433→   433→    // Fallback: assume first 4 numeric columns are OHLC
   434→   434→    if (numericIndices.length >= 4) {
   435→   435→      return {
   436→   436→        openIdx: numericIndices[0],
   437→   437→        highIdx: numericIndices[2],
   438→   438→        lowIdx: numericIndices[1],
   439→   439→        closeIdx: numericIndices[3],
   440→   440→        jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : 7,
   441→   441→        gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : 6,
   442→   442→      };
   443→   443→    }
   444→   444→    return null;
   445→   445→  }
   446→   446→
   447→   447→  // Sort by absolute value descending to identify OHLC (largest) vs change (smallest)
   448→   448→  const sorted = ohlcCandidates
   449→   449→    .map(i => ({ idx: i, val: Math.abs(parsePersianNum(row[i])) }))
   450→   450→    .sort((a, b) => b.val - a.val);
   451→   451→
   452→   452→  // The 4 largest should be OHLC. Among them, identify high (max) and low (min).
   453→   453→  // open and close are the other two.
   454→   454→  const top4 = sorted.slice(0, 4);
   455→   455→  const maxIdx = top4.reduce((a, b) => a.val > b.val ? a : b).idx;
   456→   456→  const minIdx = top4.reduce((a, b) => a.val < b.val ? a : b).idx;
   457→   457→  const others = top4.filter(x => x.idx !== maxIdx && x.idx !== minIdx);
   458→   458→
   459→   459→  // Among the two remaining, the one closer to high is open, the other is close
   460→   460→  // (heuristic: in an uptrend open < close, but we can't assume direction)
   461→   461→  // Default: first = open, second = close
   462→   462→  const openIdx = others[0]?.idx ?? ohlcCandidates[0];
   463→   463→  const closeIdx = others[1]?.idx ?? ohlcCandidates[3];
   464→   464→
   465→   465→  return {
   466→   466→    openIdx,
   467→   467→    highIdx: maxIdx,
   468→   468→    lowIdx: minIdx,
   469→   469→    closeIdx,
   470→   470→    jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : row.length - 1,
   471→   471→    gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : row.length - 2,
   472→   472→  };
   473→   473→}

   475→   475→/**
   476→   476→ * Validate a candle's OHLC values for sanity.
   477→   477→ * Returns true if the candle is valid.
   478→   478→ */
   479→   479→function isValidCandle(o: number, h: number, l: number, c: number): boolean {
   480→   480→  if (!isFinite(o) || !isFinite(h) || !isFinite(l) || !isFinite(c)) return false;
   481→   481→  if (o <= 0 || h <= 0 || l <= 0 || c <= 0) return false;
   482→   482→  // High must be >= Low, Open, Close
   483→   483→  if (h < l || h < o || h < c) return false;
   484→   484→  // Low must be <= High, Open, Close
   485→   485→  if (l > o || l > c) return false;
   486→   486→  // No infinite or extreme values
   487→   487→  if (o > 1e15 || h > 1e15 || l > 1e15 || c > 1e15) return false;
   488→   488→  return true;
   489→   489→}

   594→   594→    backgroundFetchYahooFallbackPrices(instruments).catch(() => {});

   603→   603→/**
   604→   604→ * Background task: fetch last 2 candles for currency/gold items to compute accurate daily change.
   605→   605→ * Uses direct fetch to avoid queue contention.
   606→   606→ * Only updates change/changePercent — keeps the live price from tgju-api.
   607→   607→ */
   608→   608→async function backgroundFetchCurrencyGoldChange(instruments: TgjuInstrument[]) {
   609→   609→  const currencyGoldItems = instruments.filter(
   610→   610→    (i) => i.category === 'currency' || i.category === 'gold' || i.category === 'silver' || i.category === 'gold_etf'
   611→   611→  );
   612→   612→
   613→   613→  // Limit to first 10 to avoid rate-limiting (most important: USD, EUR, gold coin, etc.)
   614→   614→  const priorityKeys = currencyGoldItems.slice(0, 10).map((i) => i.key);
   615→   615→
   616→   616→  for (const key of priorityKeys) {
   617→   617→    try {
   618→   618→      const url = `${TGJU_CHART_API}/${key}?lang=fa&order_dir=desc&start=0&length=2`;
   619→   619→      const res = await fetch(url, { signal: AbortSignal.timeout(10_000) });
   620→   620→      if (!res.ok) continue;
   621→   621→      const jsonStr = await res.text();
   622→   622→      if (!jsonStr.startsWith('{')) continue;
   623→   623→
   624→   624→      const data = JSON.parse(jsonStr);
   625→   625→      const rows: string[][] = data.data;
   626→   626→      if (!rows || rows.length < 2) continue;
   627→   627→
   628→   628→      const newest = rows[0];
   629→   629→      const prev = rows[1];
   630→   630→      const latestClose = parsePersianNum(newest[3]);
   631→   631→      const prevClose = parsePersianNum(prev[3]);
   632→   632→
   633→   633→      if (latestClose > 0 && prevClose > 0) {
   634→   634→        const inst = liveInstrumentsCache.data?.find((i) => i.key === key);
   635→   635→        if (inst) {
   636→   636→          inst.change = latestClose - prevClose;
   637→   637→          inst.changePercent = ((latestClose - prevClose) / prevClose) * 100;
   638→   638→        }
   639→   639→      }
   640→   640→    } catch {
   641→   641→      // skip individual errors
   642→   642→    }
   643→   643→  }
   644→   644→}

   654→   654→/**
   655→   655→ * Background task: fetch Yahoo Finance prices for static instruments that have price=0
   656→   656→ * (happens when api.tgju.org returns 403). Updates the live cache in-place.
   657→   657→ */
   658→   658→async function backgroundFetchYahooFallbackPrices(instruments: TgjuInstrument[]) {
   659→   659→  // Collect static instruments with price=0 that have a Yahoo mapping
   660→   660→  const zeroPriceItems = instruments.filter(
   661→   661→    (i) => i.price === 0 && i.key in TGJU_TO_YAHOO_MAP
   662→   662→  );
   663→   663→
   664→   664→  if (zeroPriceItems.length === 0) return;
   665→   665→
   666→   666→  console.log(`[TGJU] ${zeroPriceItems.length} static instruments have price=0, fetching from Yahoo Finance as fallback...`);
   667→   667→
   668→   668→  // Collect unique Yahoo symbols
   669→   669→  const keyToSymbol = new Map<string, string>();
   670→   670→  const symbolToKeys = new Map<string, string[]>();
   671→   671→  for (const item of zeroPriceItems) {
   672→   672→    const yahooSym = TGJU_TO_YAHOO_MAP[item.key];
   673→   673→    if (!yahooSym) continue;
   674→   674→    keyToSymbol.set(item.key, yahooSym);
   675→   675→    if (!symbolToKeys.has(yahooSym)) symbolToKeys.set(yahooSym, []);
   676→   676→    symbolToKeys.get(yahooSym)!.push(item.key);
   677→   677→  }
   678→   678→
   679→   679→  const uniqueSymbols = [...symbolToKeys.keys()];
   680→   680→
   681→   681→  try {
   682→   682→    // Dynamic import to avoid loading yahoo-finance2 for TSE-only users
   683→   683→    const YahooFinance = (await import('yahoo-finance2')).default;
   684→   684→    const yahooFinance = new YahooFinance({ suppressNotices: ['yahooSurvey'], validation: { logErrors: false } });
   685→   685→
   686→   686→    // Fetch in batches of 15 (smaller batches reduce Yahoo validation errors)
   687→   687→    const BATCH_SIZE = 15;
   688→   688→    for (let i = 0; i < uniqueSymbols.length; i += BATCH_SIZE) {
   689→   689→      const batch = uniqueSymbols.slice(i, i + BATCH_SIZE);
   690→   690→      try {
   691→   691→        const results = await yahooFinance.quote(batch, {
   692→   692→          fields: ['symbol', 'regularMarketPrice', 'regularMarketChange', 'regularMarketChangePercent', 'regularMarketPreviousClose', 'regularMarketDayHigh', 'regularMarketDayLow'],
   693→   693→        });
   694→   694→
   695→   695→        for (const r of results) {
   696→   696→          if (r.regularMarketPrice == null) continue;
   697→   697→          const tgjuKeys = symbolToKeys.get(r.symbol);
   698→   698→          if (!tgjuKeys) continue;
   699→   699→
   700→   700→          for (const tgjuKey of tgjuKeys) {
   701→   701→            const inst = liveInstrumentsCache.data?.find((x) => x.key === tgjuKey);
   702→   702→            if (inst && inst.price === 0) {
   703→   703→              inst.price = r.regularMarketPrice;
   704→   704→              inst.highPrice = r.regularMarketDayHigh ?? r.regularMarketPrice;
   705→   705→              inst.lowPrice = r.regularMarketDayLow ?? r.regularMarketPrice;
   706→   706→              inst.change = r.regularMarketChange ?? 0;
   707→   707→              inst.changePercent = r.regularMarketChangePercent ?? 0;
   708→   708→
   709→   709→              // Also update static price cache
   710→   710→              staticPriceCache.set(tgjuKey, {
   711→   711→                close: r.regularMarketPrice,
   712→   712→                high: r.regularMarketDayHigh ?? r.regularMarketPrice,
   713→   713→                low: r.regularMarketDayLow ?? r.regularMarketPrice,
   714→   714→                prevClose: r.regularMarketPreviousClose ?? r.regularMarketPrice,
   715→   715→                ts: Date.now(),
   716→   716→              });
   717→   717→            }
   718→   718→          }
   719→   719→        }
   720→   720→      } catch (err) {
   721→   721→        const errMsg = err instanceof Error ? err.message : String(err);
   722→   722→        if (errMsg.includes('validation') || errMsg.includes('FailedYahooValidationError')) {
   723→   723→          // Batch failed due to schema validation — retry individually
   724→   724→          console.warn(`[TGJU] Yahoo fallback batch ${Math.floor(i / BATCH_SIZE) + 1} validation error, retrying individually...`);
   725→   725→          for (const symbol of batch) {
   726→   726→            try {
   727→   727→              const singleResults = await yahooFinance.quote([symbol], {
   728→   728→                fields: ['symbol', 'regularMarketPrice', 'regularMarketChange', 'regularMarketChangePercent', 'regularMarketPreviousClose', 'regularMarketDayHigh', 'regularMarketDayLow'],
   729→   729→              });
   730→   730→              for (const r of singleResults) {
   731→   731→                if (r.regularMarketPrice == null) continue;
   732→   732→                const tgjuKeys = symbolToKeys.get(r.symbol);
   733→   733→                if (!tgjuKeys) continue;
   734→   734→                for (const tgjuKey of tgjuKeys) {
   735→   735→                  const inst = liveInstrumentsCache.data?.find((x) => x.key === tgjuKey);
   736→   736→                  if (inst && inst.price === 0) {
   737→   737→                    inst.price = r.regularMarketPrice;
   738→   738→                    inst.highPrice = r.regularMarketDayHigh ?? r.regularMarketPrice;
   739→   739→                    inst.lowPrice = r.regularMarketDayLow ?? r.regularMarketPrice;
   740→   740→                    inst.change = r.regularMarketChange ?? 0;
   741→   741→                    inst.changePercent = r.regularMarketChangePercent ?? 0;
   742→   742→                    staticPriceCache.set(tgjuKey, {
   743→   743→                      close: r.regularMarketPrice,
   744→   744→                      high: r.regularMarketDayHigh ?? r.regularMarketPrice,
   745→   745→                      low: r.regularMarketDayLow ?? r.regularMarketPrice,
   746→   746→                      prevClose: r.regularMarketPreviousClose ?? r.regularMarketPrice,
   747→   747→                      ts: Date.now(),
   748→   748→                    });
   749→   749→                  }
   750→   750→                }
   751→   751→              }
   752→   752→            } catch {
   753→   753→              // skip individual symbol errors
   754→   754→            }
   755→   755→          }
   756→   756→        } else {
   757→   757→          console.warn(`[TGJU] Yahoo fallback batch ${Math.floor(i / BATCH_SIZE) + 1} failed:`, errMsg);
   758→   758→        }
   759→   759→      }
   760→   760→    }
   761→   761→
   762→   762→    const filledCount = zeroPriceItems.filter((i) => i.price > 0).length;
   763→   763→    console.log(`[TGJU] Yahoo fallback filled ${filledCount}/${zeroPriceItems.length} instrument prices`);
   764→   764→  } catch (err) {
   765→   765→    console.error(`[TGJU] Yahoo fallback fetch failed:`, err);
   766→   766→  }
   767→   767→}

   769→   769→/* ─── Fetch historical OHLC data ── */
   770→   770→
   771→   771→export async function fetchTgjuHistory(tgjuKey: string): Promise<TgjuOHLC[]> {
   772→   772→  const cached = historyCache.get(tgjuKey);
   773→   773→  const now = Date.now();
   774→   774→  if (cached && now - cached.ts < HISTORY_TTL) {
   775→   775→    return cached.data;
   776→   776→  }
   777→   777→
   778→   778→  try {
   779→   779→    const url = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=5000`;
   780→   780→
   781→   781→    // Server-side fetch — no CORS issues, no ZAI queue needed
   782→   782→    const controller = new AbortController();
   783→   783→    const timer = setTimeout(() => controller.abort(), 25_000);
   784→   784→    let jsonStr: string;
   785→   785→    try {
   786→   786→      const res = await fetch(url, { signal: controller.signal, headers: { 'Accept': 'application/json' } });
   787→   787→      clearTimeout(timer);
   788→   788→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   789→   789→      jsonStr = await res.text();
   790→   790→    } catch (fetchErr) {
   791→   791→      clearTimeout(timer);
   792→   792→      // This is a server-side route — no CORS. If direct fetch fails, the API is down.
   793→   793→      console.warn(`[TGJU] Direct fetch failed for ${tgjuKey}:`, fetchErr instanceof Error ? fetchErr.message : fetchErr);
   794→   794→      return cached?.data || [];
   795→   795→    }
   796→   796→
   797→   797→    if (!jsonStr.startsWith('{')) {
   798→   798→      console.warn(`[TGJU] Non-JSON response for ${tgjuKey}: ${jsonStr.slice(0, 100)}`);
   799→   799→      return cached?.data || [];
   800→   800→    }
   801→   801→
   802→   802→    const parsed = JSON.parse(jsonStr);
   803→   803→    const rows: (string | number)[][] = parsed.data;
   804→   804→
   805→   805→    if (!rows || rows.length === 0) {
   806→   806→      console.warn(`[TGJU] Empty data array for ${tgjuKey}`);
   807→   807→      return cached?.data || [];
   808→   808→    }
   809→   809→
   810→   810→    // ── Auto-detect column mapping from first valid row ──
   811→   811→    let colMap = detectColumnMap(rows[0]);
   812→   812→
   813→   813→    // If auto-detect failed, try a few more rows
   814→   814→    if (!colMap) {
   815→   815→      for (let r = 1; r < Math.min(5, rows.length); r++) {
   816→   816→        colMap = detectColumnMap(rows[r]);
   817→   817→        if (colMap) break;
   818→   818→      }
   819→   819→    }
   820→   820→
   821→   821→    // If still no auto-detect, fall back to assumed column order
   822→   822→    if (!colMap) {
   823→   823→      console.warn(`[TGJU] Could not auto-detect columns for ${tgjuKey}, using default order`);
   824→   824→      colMap = { openIdx: 0, highIdx: 2, lowIdx: 1, closeIdx: 3, jalaliDateIdx: 7, gregorianDateIdx: 6 };
   825→   825→    } else {
   826→   826→      console.log(`[TGJU] Column map for ${tgjuKey}: O=${colMap.openIdx} H=${colMap.highIdx} L=${colMap.lowIdx} C=${colMap.closeIdx} Date=${colMap.jalaliDateIdx}`);
   827→   827→    }
   828→   828→
   829→   829→    const { openIdx, highIdx, lowIdx, closeIdx, jalaliDateIdx, gregorianDateIdx } = colMap;
   830→   830→
   831→   831→    // ── Parse rows ──
   832→   832→    const candles: TgjuOHLC[] = [];
   833→   833→    let invalidCount = 0;
   834→   834→
   835→   835→    for (const row of rows) {
   836→   836→      const rawDate = String(row[jalaliDateIdx] || row[gregorianDateIdx] || '');
   837→   837→      const date = rawDate.replace(/[\/]/g, '-');
   838→   838→      const o = parsePersianNum(row[openIdx]);
   839→   839→      const h = parsePersianNum(row[highIdx]);
   840→   840→      const l = parsePersianNum(row[lowIdx]);
   841→   841→      const c = parsePersianNum(row[closeIdx]);
   842→   842→
   843→   843→      if (!isValidCandle(o, h, l, c) || date.length === 0) {
   844→   844→        invalidCount++;
   845→   845→        continue;
   846→   846→      }
   847→   847→
   848→   848→      candles.push({ date, open: o, high: h, low: l, close: c });
   849→   849→    }
   850→   850→
   851→   851→    if (invalidCount > 0) {
   852→   852→      console.warn(`[TGJU] ${tgjuKey}: ${invalidCount}/${rows.length} invalid candles skipped`);
   853→   853→    }
   854→   854→
   855→   855→    if (candles.length === 0) {
   856→   856→      return cached?.data || [];
   857→   857→    }
   858→   858→
   859→   859→    // ── Detect and fix data drift ──
   860→   860→    // Check if prices suddenly jump > 50% between consecutive candles
   861→   861→    // (could indicate unit change, decimal shift, or API error)
   862→   862→    let driftCount = 0;
   863→   863→    for (let i = 1; i < candles.length; i++) {
   864→   864→      const prevClose = candles[i - 1].close;
   865→   865→      const currClose = candles[i].close;
   866→   866→      if (prevClose > 0) {
   867→   867→        const pctChange = Math.abs(currClose - prevClose) / prevClose;
   868→   868→        if (pctChange > 0.5) { // > 50% single-day change is suspicious for most instruments
   869→   869→          driftCount++;
   870→   870→        }
   871→   871→      }
   872→   872→    }
   873→   873→    if (driftCount > candles.length * 0.1) { // > 10% of candles have suspicious jumps
   874→   874→      console.warn(`[TGJU] ${tgjuKey}: Possible data drift detected (${driftCount} suspicious jumps out of ${candles.length} candles)`);
   875→   875→    }
   876→   876→
   877→   877→    // Reverse: API returns oldest-first (asc), we want newest-last for charting
   878→   878→    // But check: if the first date is newer than the last, data is already desc
   879→   879→    const firstDate = candles[0].date;
   880→   880→    const lastDate = candles[candles.length - 1].date;
   881→   881→    if (firstDate > lastDate) {
   882→   882→      candles.reverse();
   883→   883→    }
   884→   884→
   885→   885→    console.log(`[TGJU] ${tgjuKey}: Parsed ${candles.length} candles (${candles[0]?.date} to ${candles[candles.length - 1]?.date})`);
   886→   886→
   887→   887→    // Cache the result
   888→   888→    historyCache.set(tgjuKey, { data: candles, ts: now });
   889→   889→
   890→   890→    // Also update static price cache for this instrument
   891→   891→    if (candles.length >= 2) {
   892→   892→      const last = candles[candles.length - 1];
   893→   893→      const prev = candles[candles.length - 2];
   894→   894→      staticPriceCache.set(tgjuKey, {
   895→   895→        close: last.close,
   896→   896→        high: last.high,
   897→   897→        low: last.low,
   898→   898→        prevClose: prev.close,
   899→   899→        ts: now,
   900→   900→      });
   901→   901→
   902→   902→      // Update live instruments cache if this is a static instrument
   903→   903→      if (liveInstrumentsCache.data) {
   904→   904→        const inst = liveInstrumentsCache.data.find((i) => i.key === tgjuKey);
   905→   905→        if (inst && inst.price === 0) {
   906→   906→          inst.price = last.close;
   907→   907→          inst.highPrice = last.high;
   908→   908→          inst.lowPrice = last.low;
   909→   909→          inst.change = last.close - prev.close;
   910→   910→          inst.changePercent = prev.close > 0 ? ((last.close - prev.close) / prev.close) * 100 : 0;
   911→   911→        }
   912→   912→      }
   913→   913→    }
   914→   914→
   915→   915→    return candles;
   916→   916→  } catch (err) {
   917→   917→    console.error(`[TGJU] Failed to fetch history for ${tgjuKey}:`, err);
   918→   918→    return cached?.data || [];
   919→   919→  }
   920→   920→}
