   217→   217→// Branch 1: Trend Following — conditional probabilities for each scenario
   218→   218→function calcTrendConditionals(bullScore: number, adxVal: number, rsiVal: number, mfiVal: number, cciVal: number): number[] {
   219→   219→  const bull = bullScore;
   220→   220→  const bear = 1 - bullScore;
   221→   221→  const neutral = 1 - Math.abs(bullScore - 0.5) * 2; // max when bullScore=0.5
   222→   222→  const adxHigh = Math.min(adxVal / 40, 1);
   223→   223→  const volHigh = (rsiVal > 70 || rsiVal < 30) ? 0.7 : (rsiVal > 60 || rsiVal < 40) ? 0.4 : 0.2;
   224→   224→  const volLow = 1 - volHigh;
   225→   225→  const volHighFuture = mfiVal > 60 ? 0.65 : 0.35;
   226→   226→  const volLowFuture = 1 - volHighFuture;
   227→   227→  const divergence = Math.min(Math.abs(cciVal) / 150, 1);
   228→   228→  const forecastBullStrong = bull * adxHigh;
   229→   229→  const forecastBearStrong = bear * adxHigh;
   230→   230→  const shockDetection = 0.03;
   231→   231→
   232→   232→  // Raw conditional probabilities per spec formulas (A1-A9)
   233→   233→  const raw = [
   234→   234→    bull * adxHigh * forecastBullStrong,                           // A1: روند صعودی قوی
   235→   235→    bull * volHigh * volHighFuture,                                 // A2: روند صعودی شتاب‌دار
   236→   236→    bull * divergence * 0.6,                                        // A3: روند صعودی با احتیاط
   237→   237→    bear * adxHigh * forecastBearStrong,                           // A4: روند نزولی قوی
   238→   238→    bear * volHigh * volHighFuture,                                 // A5: روند نزولی شتاب‌دار
   239→   239→    bear * divergence * 0.6,                                        // A6: روند نزولی با احتیاط
   240→   240→    neutral * volLowFuture * 0.8,                                   // A7: رنج کم‌نوسان
   241→   241→    neutral * volHigh * 0.7,                                        // A8: رنج پرنوسان
   242→   242→    shockDetection * 0.04,                                          // A9: حالت شوک (0.02-0.05)
   243→   243→  ];
   244→   244→
   245→   245→  // Normalize to sum to 1
   246→   246→  const sum = raw.reduce((a, b) => a + b, 0) || 1;
   247→   247→  return raw.map(v => v / sum);
   248→   248→}

   250→   250→// Branch 2: Breakout — simplified spec formulas
   251→   251→function calcBreakoutConditionals(bullScore: number, adxVal: number, rsiVal: number, mfiVal: number): number[] {
   252→   252→  const bull = bullScore;
   253→   253→  const bear = 1 - bullScore;
   254→   254→  const volConfirm = mfiVal > 55 ? 0.7 : 0.3;
   255→   255→  const patternValid = Math.min(adxVal / 35, 1);
   256→   256→  const strongBreakout = (rsiVal > 65 || rsiVal < 35) ? 0.7 : 0.3;
   257→   257→  const adxRising = Math.min(adxVal / 45, 1);
   258→   258→  const multiLevel = Math.min((adxVal * mfiVal) / 5000, 1);
   259→   259→  const weakVolume = mfiVal < 40 ? 0.65 : 0.35;
   260→   260→  const anomaly = 0.025;
   261→   261→
   262→   262→  const raw = [
   263→   263→    bull * volConfirm * patternValid,             // B1: شکست صعودی معتبر
   264→   264→    bull * strongBreakout * adxRising,             // B2: شکست صعودی قوی
   265→   265→    bull * multiLevel * 0.5,                        // B3: شکست با تأیید حجم
   266→   266→    bear * volConfirm * patternValid,             // B4: شکست نزولی معتبر
   267→   267→    bear * strongBreakout * adxRising,             // B5: شکست نزولی قوی
   268→   268→    bear * weakVolume * 0.5,                        // B6: شکست با حجم ضعیف
   269→   269→    (1 - volConfirm) * bull * 0.4,                // B7: شکست کاذب صعودی
   270→   270→    (1 - volConfirm) * bear * 0.4,                // B8: شکست کاذب نزولی
   271→   271→    anomaly * 0.04,                                 // B9: شکست با شوک
   272→   272→  ];
   273→   273→
   274→   274→  const sum = raw.reduce((a, b) => a + b, 0) || 1;
   275→   275→  return raw.map(v => v / sum);
   276→   276→}

   278→   278→// Branch 3: Reversal — simplified spec formulas
   279→   279→function calcReversalConditionals(bullScore: number, rsiVal: number, cciVal: number, mfiVal: number): number[] {
   280→   280→  const bull = bullScore;
   281→   281→  const bear = 1 - bullScore;
   282→   282→  const rsiExtreme = (rsiVal > 70 || rsiVal < 30) ? 0.75 : (rsiVal > 60 || rsiVal < 40) ? 0.4 : 0.15;
   283→   283→  const cciExtreme = Math.min(Math.abs(cciVal) / 120, 1);
   284→   284→  const volConfirm = mfiVal > 50 ? 0.65 : 0.35;
   285→   285→  const strongSignal = rsiExtreme * cciExtreme;
   286→   286→  const srHold = Math.min(Math.abs(cciVal) / 100, 1);
   287→   287→  const bounceStrength = mfiVal > 55 ? 0.6 : 0.4;
   288→   288→  const rejectionStrength = mfiVal < 45 ? 0.6 : 0.4;
   289→   289→  const anomaly = 0.02;
   290→   290→
   291→   291→  const raw = [
   292→   292→    bull * rsiExtreme * volConfirm,               // C1: واگرایی صعودی تأییدشده
   293→   293→    bull * cciExtreme * volConfirm * 0.6,          // C2: الگوی بازگشتی صعودی
   294→   294→    bull * strongSignal * 0.8,                      // C3: واگرایی صعودی قوی
   295→   295→    bear * rsiExtreme * volConfirm,               // C4: واگرایی نزولی تأییدشده
   296→   296→    bear * cciExtreme * volConfirm * 0.6,          // C5: الگوی بازگشتی نزولی
   297→   297→    bear * strongSignal * 0.8,                      // C6: واگرایی نزولی قوی
   298→   298→    (1 - rsiExtreme) * bounceStrength * 0.5,       // C7: بازگشت از حمایت
   299→   299→    (1 - rsiExtreme) * rejectionStrength * 0.5,    // C8: بازگشت از مقاومت
   300→   300→    anomaly * 0.04,                                 // C9: بازگشت با شوک
   301→   301→  ];
   302→   302→
   303→   303→  const sum = raw.reduce((a, b) => a + b, 0) || 1;
   304→   304→  return raw.map(v => v / sum);
   305→   305→}

   307→   307→// Calculate branch selection probabilities
   308→   308→function calcBranchProbs(adxVal: number, bullScore: number, rsiVal: number, mfiVal: number): { trend: number; breakout: number; reversal: number } {
   309→   309→  const adxNorm = Math.min(adxVal / 60, 1);
   310→   310→  const volProxy = (rsiVal > 65 || rsiVal < 35) ? 0.6 : (rsiVal > 55 || rsiVal < 45) ? 0.4 : 0.2;
   311→   311→  const mfiProxy = mfiVal > 60 ? 0.6 : mfiVal < 40 ? 0.6 : 0.3;
   312→   312→  const divergenceProxy = mfiProxy * (1 - adxNorm);
   313→   313→
   314→   314→  let trend = adxNorm * 0.45 + bullScore * 0.15 + 0.12;
   315→   315→  let breakout = volProxy * 0.35 + (1 - adxNorm) * 0.1 + 0.1;
   316→   316→  let reversal = divergenceProxy * 0.3 + (1 - adxNorm) * 0.15 + 0.08;
   317→   317→
   318→   318→  // Normalize
   319→   319→  const sum = trend + breakout + reversal || 1;
   320→   320→  return { trend: trend / sum, breakout: breakout / sum, reversal: reversal / sum };
   321→   321→}

   335→   335→// Calculate all edge probabilities for the 3-branch graph
   336→   336→function calcAllEdgeProbs(
   337→   337→  bullScore: number, adxVal: number, rsiVal: number, mfiVal: number, cciVal: number
   338→   338→): Record<string, number> {
   339→   339→  const branchProbs = calcBranchProbs(adxVal, bullScore, rsiVal, mfiVal);
   340→   340→  const trendCond = calcTrendConditionals(bullScore, adxVal, rsiVal, mfiVal, cciVal);
   341→   341→  const breakoutCond = calcBreakoutConditionals(bullScore, adxVal, rsiVal, mfiVal);
   342→   342→  const reversalCond = calcReversalConditionals(bullScore, rsiVal, cciVal, mfiVal);
   343→   343→
   344→   344→  const probs: Record<string, number> = {};
   345→   345→
   346→   346→  // Root → Branch edges (indices 0-2)
   347→   347→  probs['0'] = branchProbs.trend;
   348→   348→  probs['1'] = branchProbs.breakout;
   349→   349→  probs['2'] = branchProbs.reversal;
   350→   350→
   351→   351→  // Branch → Event edges (indices 3-5) — deterministic
   352→   352→  probs['3'] = 1.0;
   353→   353→  probs['4'] = 1.0;
   354→   354→  probs['5'] = 1.0;
   355→   355→
   356→   356→  // Trend Following: Event A → R1-R9 (indices 6-14)
   357→   357→  for (let i = 0; i < 9; i++) {
   358→   358→    probs[String(6 + i)] = trendCond[i];
   359→   359→  }
   360→   360→
   361→   361→  // Breakout: Event B → R1-R9 (indices 15-23)
   362→   362→  for (let i = 0; i < 9; i++) {
   363→   363→    probs[String(15 + i)] = breakoutCond[i];
   364→   364→  }
   365→   365→
   366→   366→  // Reversal: Event C → R1-R9 (indices 24-32)
   367→   367→  for (let i = 0; i < 9; i++) {
   368→   368→    probs[String(24 + i)] = reversalCond[i];
   369→   369→  }
   370→   370→
   371→   371→  return probs;
   372→   372→}

   374→   374→// Find all 27 paths from ROOT to terminal nodes
   375→   375→function findAllPaths(edgeProbs: Record<string, number>): PathInfo[] {
   376→   376→  const paths: PathInfo[] = [];
   377→   377→  const branches = ['BR1', 'BR2', 'BR3'];
   378→   378→  const events = ['EA', 'EB', 'EC'];
   379→   379→  const branchNames = ['trend', 'breakout', 'reversal'];
   380→   380→
   381→   381→  for (let b = 0; b < 3; b++) {
   382→   382→    const branchEdgeIdx = b;       // 0, 1, 2
   383→   383→    const eventEdgeIdx = 3 + b;    // 3, 4, 5
   384→   384→    const baseEdgeIdx = 6 + b * 9; // 6, 15, 24
   385→   385→
   386→   386→    const branchProb = edgeProbs[String(branchEdgeIdx)] ?? 0.2;
   387→   387→    const eventProb = edgeProbs[String(eventEdgeIdx)] ?? 1;
   388→   388→
   389→   389→    for (let s = 0; s < 9; s++) {
   390→   390→      const scenarioEdgeIdx = baseEdgeIdx + s;
   391→   391→      const scenarioProb = edgeProbs[String(scenarioEdgeIdx)] ?? 0.1;
   392→   392→      const scenarioKey = SCENARIO_KEYS[s];
   393→   393→
   394→   394→      paths.push({
   395→   395→        nodes: ['ROOT', branches[b], events[b], scenarioKey],
   396→   396→        edges: [branchEdgeIdx, eventEdgeIdx, scenarioEdgeIdx],
   397→   397→        prob: branchProb * eventProb * scenarioProb,
   398→   398→        target: scenarioKey,
   399→   399→        branch: branchNames[b],
   400→   400→      });
   401→   401→    }
   402→   402→  }
   403→   403→
   404→   404→  return paths.sort((a, b) => b.prob - a.prob);
   405→   405→}
