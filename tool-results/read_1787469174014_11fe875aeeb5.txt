   167→// Graph Algorithms
   168→// ═══════════════════════════════════════════════════════════════════════════════
   169→
   170→interface EdgeInfo {
   171→  from: string;
   172→  to: string;
   173→  label: string;
   174→  type: string;
   175→  prob: number;
   176→}
   177→
   178→interface PathInfo {
   179→  nodes: string[];
   180→  edges: number[];
   181→  prob: number;
   182→  target: string;
   183→}
   184→
   185→// Calculate edge probabilities based on market conditions
   186→function calcEdgeProbabilities(
   187→  bullScore: number, adxVal: number
   188→): Record<string, number> {
   189→  const adxW = Math.min(adxVal / 100, 1);
   190→  const typeWeights: Record<string, number> = {
   191→    up: bullScore * 0.6 + adxW * 0.4,
   192→    pullback: 0.3,
   193→    down: (1 - bullScore) * 0.6 + adxW * 0.4,
   194→    risk: 0.15 * (1.2 - adxW),
   195→  };
   196→
   197→  // Build adjacency: for each node, group outgoing edges by type
   198→  const adj: Record<string, { idx: number; type: string }[]> = {};
   199→  EDGES.forEach((e, i) => {
   200→    if (!adj[e[0]]) adj[e[0]] = [];
   201→    adj[e[0]].push({ idx: i, type: e[3] });
   202→  });
   203→
   204→  const edgeProbs: Record<string, number> = {};
   205→
   206→  for (const [node, outEdges] of Object.entries(adj)) {
   207→    // Sum weights of outgoing edges
   208→    const totalWeight = outEdges.reduce((sum, e) => sum + (typeWeights[e.type] ?? 0.1), 0);
   209→    if (totalWeight === 0) {
   210→      outEdges.forEach(e => { edgeProbs[String(e.idx)] = 1 / outEdges.length; });
   211→      continue;
   212→    }
   213→    outEdges.forEach(e => {
   214→      edgeProbs[String(e.idx)] = (typeWeights[e.type] ?? 0.1) / totalWeight;
   215→    });
   216→  }
   217→
   218→  return edgeProbs;
   219→}
   220→
   221→// Find all paths from 'A' to terminal nodes using DFS
   222→function findAllPaths(edgeProbs: Record<string, number>): PathInfo[] {
   223→  const adj: Record<string, { to: string; edgeIdx: number }[]> = {};
   224→  EDGES.forEach((e, i) => {
   225→    if (!adj[e[0]]) adj[e[0]] = [];
   226→    adj[e[0]].push({ to: e[1], edgeIdx: i });
   227→  });
   228→
   229→  const paths: PathInfo[] = [];
   230→  const MAX_PATHS = 200;
   231→
   232→  function dfs(node: string, visited: Set<string>, currentPath: string[], edgeIndices: number[], currentProb: number) {
   233→    if (SCENARIO_KEYS.includes(node as any)) {
   234→      paths.push({ nodes: [...currentPath, node], edges: [...edgeIndices], prob: currentProb, target: node });
   235→      return;
   236→    }
   237→    if (paths.length >= MAX_PATHS) return;
   238→    if (visited.has(node)) return;
   239→    visited.add(node);
   240→
   241→    const outs = adj[node] ?? [];
   242→    for (const { to, edgeIdx } of outs) {
   243→      const ep = edgeProbs[String(edgeIdx)] ?? 0.1;
   244→      dfs(to, visited, [...currentPath, node], [...edgeIndices, edgeIdx], currentProb * ep);
   245→    }
   246→    visited.delete(node);
   247→  }
   248→
   249→  dfs('A', new Set(), [], [], 1);
   250→  return paths.sort((a, b) => b.prob - a.prob);
   251→}
