    29→/* ────────────────────────────────────────────────────────── TYPES ────────── */
    30→
    31→interface DrawingPoint {
    32→  time: number;
    33→  price: number;
    34→}
    35→
    36→interface Drawing {
    37→  id: string;
    38→  type: string;
    39→  points: DrawingPoint[];
    40→  color: string;
    41→  lineWidth: number;
    42→  text?: string;
    43→  completed: boolean;
    44→}
    45→
    46→interface LevelStrength {
    47→  price: number;
    48→  strength: number;
    49→  isTarget: boolean;
    50→}
    51→
    52→interface CandlestickChartProps {
    53→  data: Array<{
    54→    date: string;
    55→    open: number;
    56→    high: number;

    58→    close: number;
    59→    volume: number;
    60→  }>;
    61→  ta: {
    62→    sma: Record<string, number>;
    63→    bollingerBands: { upper: number; middle: number; lower: number };
    64→    resistances: number[];
    65→    supports: number[];
    66→    supportStrengths: LevelStrength[];
    67→    resistanceStrengths: LevelStrength[];
    68→    sar: number;
    69→  } | null;
    70→  height?: number;
    71→}

    89→];
    90→
    91→const FIB_LEVELS = [
    92→  { pct: 0, label: '0%' },
    93→  { pct: 0.236, label: '23.6%' },
    94→  { pct: 0.382, label: '38.2%' },
    95→  { pct: 0.5, label: '50%' },
    96→  { pct: 0.618, label: '61.8%' },
    97→  { pct: 0.786, label: '78.6%' },
    98→  { pct: 1, label: '100%' },
    99→];

   102→

   108→
   109→const TOOLS: ToolDef[] = [
   110→  { id: 'cursor', label: 'مؤشر', icon: <MousePointer2 className="w-3.5 h-3.5" /> },
   111→  { id: 'trendline', label: 'خط روند', icon: <Minus className="w-3.5 h-3.5" style={{ transform: 'rotate(-30deg)' }} /> },
   112→  { id: 'hline', label: 'خط افقی', icon: <ArrowLeftRight className="w-3.5 h-3.5" /> },
   113→  { id: 'vline', label: 'خط عمودی', icon: <MoveVertical className="w-3.5 h-3.5" /> },

   115→  { id: 'rectangle', label: 'مستطیل', icon: <RectangleHorizontal className="w-3.5 h-3.5" /> },
   116→  { id: 'text', label: 'متن', icon: <Type className="w-3.5 h-3.5" /> },
   117→  { id: 'brush', label: 'قلم', icon: <Pencil className="w-3.5 h-3.5" /> },
   118→  { id: 'measure', label: 'اندازه‌گیری', icon: <Ruler className="w-3.5 h-3.5" /> },
   119→  { id: 'arrow', label: 'پیکان', icon: <ArrowUpRight className="w-3.5 h-3.5" /> },
   120→];

   122→/* ────────────────────────────────────────────────────────── SKELETON ─────── */
   123→
   124→export function CandlestickChartSkeleton() {
   125→  return (
   126→    <div className="w-full">
   127→      <div className="h-10 bg-white/5 rounded-t-lg" />
   128→      <div className="w-full bg-white/5 rounded-b-lg" style={{ height: 520 }} />
   129→    </div>
   130→  );
   131→}
   132→
   133→/* ────────────────────────────────────────────────────────── COMPONENT ───── */
   134→
   135→export default function CandlestickChart({ data, ta, height = 520 }: CandlestickChartProps) {
   136→  const chartContainerRef = useRef<HTMLDivElement>(null);
   137→  const svgRef = useRef<SVGSVGElement>(null);
   138→  const chartRef = useRef<IChartApi | null>(null);
   139→  const subscriberRef = useRef<{ dispose: () => void } | null>(null);
   140→
   141→  const [drawings, setDrawings] = useState<Drawing[]>([]);
   142→  const [activeTool, setActiveTool] = useState<ToolType>('cursor');
   143→  const [activeColor, setActiveColor] = useState(PALETTE[0]);
   144→  const [isDrawing, setIsDrawing] = useState(false);
   145→  const [currentDrawing, setCurrentDrawing] = useState<Drawing | null>(null);
   146→  const [textInput, setTextInput] = useState<{ x: number; y: number; drawingId: string } | null>(null);
   147→  const [textValue, setTextValue] = useState('');
   148→  const [renderKey, setRenderKey] = useState(0);
   149→  const [chartSize, setChartSize] = useState({ w: 0, h: 0 });
   150→
   151→  /* ── Helper: coordinate conversion ─────────────────────────── */
   152→  const getChartPoint = useCallback((clientX: number, clientY: number): DrawingPoint | null => {
   153→    const chart = chartRef.current;
   154→    const container = chartContainerRef.current;
   155→    if (!chart || !container) return null;
   156→    const rect = container.getBoundingClientRect();
   157→    const x = clientX - rect.left;
   158→    const y = clientY - rect.top;
   159→    const price = chart.coordinateToPrice(y);
   160→    const time = chart.timeScale().coordinateToTime(x);
   161→    if (price === null || time === null) return null;
   162→    return { time: time as number, price };
   163→  }, []);
   164→
   165→  /* ── SVG Rendering ────────────────────────────────────────── */
   166→  const renderDrawings = useCallback(() => {
   167→    const chart = chartRef.current;
   168→    const svg = svgRef.current;
   169→    const container = chartContainerRef.current;
   170→    if (!chart || !svg || !container || drawings.length === 0) {
   171→      if (svg) svg.innerHTML = '';
   172→      return;
   173→    }
   174→
   175→    const rect = container.getBoundingClientRect();
   176→    const w = rect.width;
   177→    const h = rect.height;
   178→    if (w === 0 || h === 0) return;
   179→
   180→    let html = '';
   181→
   182→    // Defs for arrow markers
   183→    html += `<defs>`;
   184→    PALETTE.forEach((c, i) => {
   185→      html += `<marker id="arrow-${c.replace('#','')}" markerWidth="8" markerHeight="6" refX="7" refY="3" orient="auto"><polygon points="0 0, 8 3, 0 6" fill="${c}" /></marker>`;
   186→    });
   187→    html += `</defs>`;
   188→
   189→    const allDrawings = [...drawings];
   190→    if (currentDrawing) allDrawings.push(currentDrawing);
   191→
   192→    for (const d of allDrawings) {
   193→      if (!d.completed && d.points.length === 0) continue;
   194→      const col = d.color;
   195→      const lw = d.lineWidth;
   196→
   197→      switch (d.type) {
   198→        case 'trendline': {
   199→          if (d.points.length < 1) break;
   200→          const p0 = d.points[0];
   201→          const p1 = d.points.length > 1 ? d.points[d.points.length - 1] : p0;
   202→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   203→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   204→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   205→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   206→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   207→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" />`;
   208→          break;
   209→        }
   210→        case 'hline': {
   211→          if (d.points.length < 1) break;
   212→          const p = d.points[0];
   213→          const y = chart.priceScale('right').priceToCoordinate(p.price);
   214→          if (y == null) break;
   215→          html += `<line x1="0" y1="${y}" x2="${w}" y2="${y}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   216→          html += `<text x="4" y="${y - 4}" fill="${col}" font-size="11" font-family="sans-serif">${p.price.toFixed(0)}</text>`;
   217→          break;
   218→        }
   219→        case 'vline': {
   220→          if (d.points.length < 1) break;
   221→          const p = d.points[0];
   222→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   223→          if (x == null) break;
   224→          html += `<line x1="${x}" y1="0" x2="${x}" y2="${h}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   225→          break;
   226→        }
   227→        case 'fibonacci': {
   228→          if (d.points.length < 2) break;
   229→          const p0 = d.points[0];
   230→          const p1 = d.points[1];
   231→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   232→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   233→          if (x0 == null || x1 == null) break;
   234→          const priceHigh = Math.max(p0.price, p1.price);
   235→          const priceLow = Math.min(p0.price, p1.price);
   236→          const priceRange = priceHigh - priceLow;
   237→          if (priceRange === 0) break;
   238→
   239→          // Filled background
   240→          const yTop = chart.priceScale('right').priceToCoordinate(priceHigh);
   241→          const yBottom = chart.priceScale('right').priceToCoordinate(priceLow);
   242→          if (yTop == null || yBottom == null) break;
   243→          html += `<rect x="${Math.min(x0,x1)}" y="${yTop}" width="${Math.abs(x1-x0)}" height="${yBottom - yTop}" fill="${col}" opacity="0.04" />`;
   244→
   245→          for (const level of FIB_LEVELS) {
   246→            const price = priceHigh - level.pct * priceRange;
   247→            const y = chart.priceScale('right').priceToCoordinate(price);
   248→            if (y == null) continue;
   249→            html += `<line x1="${Math.min(x0,x1) - 10}" y1="${y}" x2="${Math.max(x0,x1) + 10}" y2="${y}" stroke="${col}" stroke-width="1" opacity="0.6" />`;
   250→            html += `<rect x="${Math.max(x0,x1) + 12}" y="${y - 8}" width="56" height="16" fill="#0b0f1a" rx="2" />`;
   251→            html += `<text x="${Math.max(x0,x1) + 14}" y="${y + 4}" fill="${col}" font-size="10" font-family="sans-serif">${level.label} ${price.toFixed(0)}</text>`;
   252→          }
   253→          break;
   254→        }
   255→        case 'rectangle': {
   256→          if (d.points.length < 2) break;
   257→          const p0 = d.points[0];
   258→          const p1 = d.points[d.points.length - 1];
   259→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   260→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   261→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   262→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   263→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   264→          const rx = Math.min(x0, x1);
   265→          const ry = Math.min(y0, y1);
   266→          const rw = Math.abs(x1 - x0);
   267→          const rh = Math.abs(y1 - y0);
   268→          html += `<rect x="${rx}" y="${ry}" width="${rw}" height="${rh}" fill="${col}" fill-opacity="0.08" stroke="${col}" stroke-width="${lw}" rx="2" />`;
   269→          break;
   270→        }
   271→        case 'text': {
   272→          if (d.points.length < 1 || !d.text) break;
   273→          const p = d.points[0];
   274→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   275→          const y = chart.priceScale('right').priceToCoordinate(p.price);
   276→          if (x == null || y == null) break;
   277→          html += `<rect x="${x + 4}" y="${y - 16}" width="${d.text.length * 8 + 12}" height="22" fill="#0b0f1a" stroke="${col}" stroke-width="1" rx="3" />`;
   278→          html += `<text x="${x + 10}" y="${y + 1}" fill="${col}" font-size="12" font-family="sans-serif">${d.text}</text>`;
   279→          break;
   280→        }
   281→        case 'brush': {
   282→          if (d.points.length < 2) break;
   283→          let pts = '';
   284→          let valid = true;
   285→          for (const p of d.points) {
   286→            const x = chart.timeScale().timeToCoordinate(p.time as Time);
   287→            const y = chart.priceScale('right').priceToCoordinate(p.price);
   288→            if (x == null || y == null) { valid = false; break; }
   289→            pts += `${x},${y} `;
   290→          }
   291→          if (!valid || pts.length === 0) break;
   292→          html += `<polyline points="${pts.trim()}" fill="none" stroke="${col}" stroke-width="${lw}" stroke-linecap="round" stroke-linejoin="round" />`;
   293→          break;
   294→        }
   295→        case 'measure': {
   296→          if (d.points.length < 2) break;
   297→          const p0 = d.points[0];
   298→          const p1 = d.points[d.points.length - 1];
   299→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   300→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   301→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   302→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   303→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   304→
   305→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="4,2" />`;
   306→          // Price difference
   307→          const diff = p1.price - p0.price;
   308→          const pctChg = p0.price !== 0 ? ((diff / p0.price) * 100) : 0;
   309→          const bars = Math.abs(Math.round(p1.time - p0.time));
   310→          const midX = (x0 + x1) / 2;
   311→          const midY = (y0 + y1) / 2;
   312→          const label = `${diff >= 0 ? '+' : ''}${diff.toFixed(0)} (${pctChg >= 0 ? '+' : ''}${pctChg.toFixed(1)}%) ${bars}bar`;
   313→          html += `<rect x="${midX - 40}" y="${midY - 24}" width="80" height="20" fill="#0b0f1a" stroke="${col}" stroke-width="1" rx="3" />`;
   314→          html += `<text x="${midX - 36}" y="${midY - 10}" fill="${col}" font-size="10" font-family="sans-serif">${label}</text>`;
   315→          break;
   316→        }
   317→        case 'arrow': {
   318→          if (d.points.length < 2) break;
   319→          const p0 = d.points[0];
   320→          const p1 = d.points[d.points.length - 1];
   321→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   322→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   323→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   324→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   325→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   326→          const markerId = `arrow-${col.replace('#','')}`;
   327→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" marker-end="url(#${markerId})" />`;
   328→          break;
   329→        }
   330→      }
   331→    }
   332→
   333→    svg.innerHTML = html;
   334→  }, [drawings, currentDrawing]);
   335→
   336→  /* ── Chart Build ──────────────────────────────────────────── */
   337→  const build = useCallback(() => {
   338→    const chartEl = chartContainerRef.current;
   339→    if (!chartEl || data.length === 0) return;
   340→
   341→    if (chartRef.current) {
   342→      chartRef.current.remove();
   343→      chartRef.current = null;
   344→    }
   345→
   346→    const chart = createChart(chartEl, {
   347→      layout: { background: { type: ColorType.Solid, color: BG }, textColor: TXT, fontSize: 11 },
   348→      grid: { vertLines: { color: GRID }, horzLines: { color: GRID } },
   349→      crosshair: { mode: CrosshairMode.Normal },
   350→      rightPriceScale: {
   351→        borderColor: 'rgba(255,255,255,0.08)',
   352→        scaleMargins: { top: 0.05, bottom: 0.3 },
   353→      },
   354→      timeScale: { borderColor: 'rgba(255,255,255,0.08)', rightOffset: 5, barSpacing: 6 },
   355→      width: chartEl.clientWidth,
   356→      height,
   357→    });
   358→    chartRef.current = chart;
   359→
   360→    // Numeric time index
   361→    const candles = data.map((d, i) => ({
   362→      time: i as Time,
   363→      open: d.open,
   364→      high: d.high,
   365→      low: d.low,
   366→      close: d.close,
   367→    }));
   368→
   369→    const cs = chart.addSeries(CandlestickSeries, {
   370→      upColor: BULL,
   371→      downColor: BEAR,
   372→      borderUpColor: BULL,
   373→      borderDownColor: BEAR,
   374→      wickUpColor: BULL,
   375→      wickDownColor: BEAR,
   376→    });
   377→    cs.setData(candles);
   378→
   379→    // Volume
   380→    const vol = chart.addSeries(HistogramSeries, {
   381→      priceFormat: { type: 'volume' },
   382→      priceScaleId: 'vol',
   383→    });
   384→    vol.setData(
   385→      data.map((d, i) => ({
   386→        time: i as Time,
   387→        value: d.volume,
   388→        color: d.close >= d.open ? BULL : BEAR,
   389→      }))
   390→    );
   391→    chart.priceScale('vol').applyOptions({ scaleMargins: { top: 0.8, bottom: 0 } });
   392→
   393→    // TA overlays
   394→    if (ta) {
   395→      for (const cfg of SMA_CFG) {
   396→        const v = ta.sma[cfg.key];
   397→        if (v && v > 0)
   398→          cs.createPriceLine({
   399→            price: v,
   400→            color: cfg.color,
   401→            lineWidth: 1,
   402→            lineStyle: 0,
   403→            axisLabelVisible: true,
   404→            title: cfg.title,
   405→          });
   406→      }
   407→      const bb = ta.bollingerBands;
   408→      if (bb) {
   409→        if (bb.upper > 0)
   410→          cs.createPriceLine({ price: bb.upper, color: 'rgba(161,98,255,0.5)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Upper' });
   411→        if (bb.middle > 0)
   412→          cs.createPriceLine({ price: bb.middle, color: 'rgba(161,98,255,0.5)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Mid' });
   413→        if (bb.lower > 0)
   414→          cs.createPriceLine({ price: bb.lower, color: 'rgba(161,98,255,0.5)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Lower' });
   415→      }
   416→      if (ta.sar > 0)
   417→        cs.createPriceLine({ price: ta.sar, color: 'rgba(255,255,255,0.35)', lineWidth: 1, lineStyle: 1, axisLabelVisible: true, title: 'SAR' });
   418→
   419→      // S/R with strength
   420→      const resistanceStrengths = ta.resistanceStrengths || [];
   421→      const supportStrengths = ta.supportStrengths || [];
   422→
   423→      resistanceStrengths.forEach((r, i) => {
   424→        if (r.price <= 0) return;
   425→        const lw = r.isTarget ? 3 : Math.max(1, Math.round(r.strength / 3));
   426→        const ls = r.isTarget ? 0 : 2;
   427→        const title = `R${i + 1} (${r.strength}/10)`;
   428→        cs.createPriceLine({
   429→          price: r.price,
   430→          color: r.isTarget ? '#ff6b6b' : BEAR,
   431→          lineWidth: lw,
   432→          lineStyle: ls as 0 | 1 | 2,
   433→          axisLabelVisible: i < 6,
   434→          title,
   435→        });
   436→      });
   437→
   438→      supportStrengths.forEach((s, i) => {
   439→        if (s.price <= 0) return;
   440→        const lw = s.isTarget ? 3 : Math.max(1, Math.round(s.strength / 3));
   441→        const ls = s.isTarget ? 0 : 2;
   442→        const title = `S${i + 1} (${s.strength}/10)`;
   443→        cs.createPriceLine({
   444→          price: s.price,
   445→          color: s.isTarget ? '#6bffb8' : BULL,
   446→          lineWidth: lw,
   447→          lineStyle: ls as 0 | 1 | 2,
   448→          axisLabelVisible: i < 6,
   449→          title,
   450→        });
   451→      });
   452→    }
   453→
   454→    chart.timeScale().fitContent();
   455→
   456→    // Subscribe to visible range changes to re-render SVG
   457→    requestAnimationFrame(() => {
   458→      if (chartRef.current) {
   459→        subscriberRef.current = chartRef.current.timeScale().subscribeVisibleLogicalRangeChange(() => {
   460→          setRenderKey((k) => k + 1);
   461→        });
   462→      }
   463→    });
   464→
   465→    // Set initial chart size (deferred to avoid React batching issue)
   466→    requestAnimationFrame(() => {
   467→      if (chartContainerRef.current) {
   468→        setChartSize({ w: chartContainerRef.current.clientWidth, h });
   469→      }
   470→    });
   471→  }, [data, ta, height]);
   472→
   473→  /* ── Re-render SVG when renderKey changes ────────────────── */
   474→  useEffect(() => {
   475→    renderDrawings();
   476→  }, [renderKey, renderDrawings]);
   477→
   478→  /* ── Init chart ───────────────────────────────────────────── */
   479→  useEffect(() => {
   480→    const el = chartContainerRef.current;
   481→    if (!el) return;
   482→    const ro = new ResizeObserver((entries) => {
   483→      for (const e of entries) {
   484→        const { width: w, height: h } = e.contentRect;
   485→        if (w > 0 && h > 0) {
   486→          chartRef.current?.applyOptions({ width: w, height: h });
   487→          setChartSize({ w, h });
   488→        }
   489→      }
   490→    });
   491→    ro.observe(el);
   492→    build();
   493→    return () => {
   494→      ro.disconnect();
   495→      subscriberRef.current?.dispose();
   496→      subscriberRef.current = null;
   497→      if (chartRef.current) {
   498→        chartRef.current.remove();
   499→        chartRef.current = null;
   500→      }
   501→    };
   502→  }, [build]);
   503→
   504→  /* ── Mouse event handlers ────────────────────────────────── */
   505→  const handleMouseDown = useCallback(
   506→    (e: React.MouseEvent) => {
   507→      if (activeTool === 'cursor') return;
   508→      if (textInput) return; // don't draw while text input is open
   509→
   510→      const point = getChartPoint(e.clientX, e.clientY);
   511→      if (!point) return;
   512→
   513→      // Horizontal line: complete on single click
   514→      if (activeTool === 'hline') {
   515→        const drawing: Drawing = {
   516→          id: Date.now().toString(),
   517→          type: 'hline',
   518→          points: [point],
   519→          color: activeColor,
   520→          lineWidth: 1,
   521→          completed: true,
   522→        };
   523→        setDrawings((prev) => [...prev, drawing]);
   524→        setIsDrawing(false);
   525→        return;
   526→      }
   527→
   528→      // Text tool: show inline input
   529→      if (activeTool === 'text') {
   530→        const chart = chartRef.current;
   531→        const container = chartContainerRef.current;
   532→        if (!chart || !container) return;
   533→        const rect = container.getBoundingClientRect();
   534→        const x = e.clientX - rect.left;
   535→        const y = e.clientY - rect.top;
   536→
   537→        const tempDrawing: Drawing = {
   538→          id: Date.now().toString(),
   539→          type: 'text',
   540→          points: [point],
   541→          color: activeColor,
   542→          lineWidth: 1,
   543→          text: '',
   544→          completed: false,
   545→        };
   546→        setCurrentDrawing(tempDrawing);
   547→        setTextInput({ x, y, drawingId: tempDrawing.id });
   548→        setTextValue('');
   549→        setIsDrawing(true);
   550→        return;
   551→      }
   552→
   553→      // Other tools: start drawing
   554→      const drawing: Drawing = {
   555→        id: Date.now().toString(),
   556→        type: activeTool,
   557→        points: [point],
   558→        color: activeColor,
   559→        lineWidth: activeTool === 'brush' ? 2 : 1,
   560→        completed: false,
   561→      };
   562→      setCurrentDrawing(drawing);
   563→      setIsDrawing(true);
   564→    },
   565→    [activeTool, activeColor, getChartPoint, textInput]
   566→  );
   567→
   568→  const handleMouseMove = useCallback(
   569→    (e: React.MouseEvent) => {
   570→      if (!isDrawing || !currentDrawing) return;
   571→
   572→      const point = getChartPoint(e.clientX, e.clientY);
   573→      if (!point) return;
   574→
   575→      if (currentDrawing.type === 'brush') {
   576→        // Add points continuously for brush
   577→        setCurrentDrawing({
   578→          ...currentDrawing,
   579→          points: [...currentDrawing.points, point],
   580→        });
   581→      } else if (currentDrawing.type !== 'text' && currentDrawing.type !== 'hline') {
   582→        // Update last point for other tools
   583→        setCurrentDrawing({
   584→          ...currentDrawing,
   585→          points: [currentDrawing.points[0], point],
   586→        });
   587→      }
   588→    },
   589→    [isDrawing, currentDrawing, getChartPoint]
   590→  );
   591→
   592→  const handleMouseUp = useCallback(() => {
   593→    if (!isDrawing || !currentDrawing) return;
   594→
   595→    if (currentDrawing.type === 'text') {
   596→      // Don't complete text yet - wait for text input submission
   597→      return;
   598→    }
   599→
   600→    const completed: Drawing = {
   601→      ...currentDrawing,
   602→      completed: true,
   603→    };
   604→
   605→    // Only save if it has enough points
   606→    const minPoints = currentDrawing.type === 'hline' ? 1 : 2;
   607→    if (completed.points.length >= minPoints) {
   608→      setDrawings((prev) => [...prev, completed]);
   609→    }
   610→
   611→    setCurrentDrawing(null);
   612→    setIsDrawing(false);
   613→  }, [isDrawing, currentDrawing]);
   614→
   615→  /* ── Text input submit ────────────────────────────────────── */
   616→  const handleTextSubmit = useCallback(() => {
   617→    if (!textInput || !textValue.trim() || !currentDrawing) {
   618→      setTextInput(null);
   619→      setCurrentDrawing(null);
   620→      setIsDrawing(false);
   621→      return;
   622→    }
   623→
   624→    const completed: Drawing = {
   625→      ...currentDrawing,
   626→      text: textValue.trim(),
   627→      completed: true,
   628→    };
   629→    setDrawings((prev) => [...prev, completed]);
   630→    setTextInput(null);
   631→    setTextValue('');
   632→    setCurrentDrawing(null);
   633→    setIsDrawing(false);
   634→  }, [textInput, textValue, currentDrawing]);
   635→
   636→  const handleTextCancel = useCallback(() => {
   637→    setTextInput(null);
   638→    setTextValue('');
   639→    setCurrentDrawing(null);
   640→    setIsDrawing(false);
   641→  }, []);
   642→
   643→  /* ── Delete / Clear ──────────────────────────────────────── */
   644→  const handleDeleteLast = useCallback(() => {
   645→    setDrawings((prev) => prev.slice(0, -1));
   646→  }, []);
   647→
   648→  const handleClearAll = useCallback(() => {
   649→    setDrawings([]);
   650→  }, []);
   651→
   652→  /* ── Render ───────────────────────────────────────────────── */
   653→  return (
   654→    <div className="w-full">
   655→      {/* ── TOOLBAR ────────────────────────────────────────── */}
   656→      <div
   657→        className="flex items-center gap-1 px-2 py-1.5 rounded-t-lg border border-b-0 overflow-x-auto flex-nowrap"
   658→        style={{ backgroundColor: TOOLBAR_BG, borderColor: TOOLBAR_BORDER }}
   659→      >
   660→        {/* Tool buttons */}
   661→        {TOOLS.map((tool) => (
   662→          <button
   663→            key={tool.id}
   664→            onClick={() => {
   665→              setActiveTool(tool.id);
   666→              // Cancel ongoing drawing when switching tools
   667→              if (isDrawing) {
   668→                setIsDrawing(false);
   669→                setCurrentDrawing(null);
   670→                setTextInput(null);
   671→              }
   672→            }}
   673→            title={tool.label}
   674→            className={`shrink-0 flex items-center justify-center w-8 h-8 rounded-md transition-all cursor-pointer ${
   675→              activeTool === tool.id
   676→                ? 'bg-amber-500/20 text-amber-400 ring-1 ring-amber-500/30'
   677→                : 'text-gray-400 hover:bg-white/5 hover:text-gray-200'
   678→            }`}
   679→          >
   680→            {tool.icon}
   681→          </button>
   682→        ))}
   683→
   684→        {/* Separator */}
   685→        <div className="w-px h-6 bg-white/10 mx-1 shrink-0" />
   686→
   687→        {/* Color picker */}
   688→        {PALETTE.map((c) => (
   689→          <button
   690→            key={c}
   691→            onClick={() => setActiveColor(c)}
   692→            title={c}
   693→            className={`shrink-0 w-6 h-6 rounded-full border-2 transition-all cursor-pointer ${
   694→              activeColor === c ? 'border-white scale-110' : 'border-transparent hover:scale-105'
   695→            }`}
   696→            style={{ backgroundColor: c }}
   697→          />
   698→        ))}
   699→
   700→        {/* Separator */}
   701→        <div className="w-px h-6 bg-white/10 mx-1 shrink-0" />
   702→
   703→        {/* Delete last */}
   704→        <button
   705→          onClick={handleDeleteLast}
   706→          title="حذف آخر"
   707→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-400 hover:bg-red-500/15 hover:text-red-400 transition-all cursor-pointer"
   708→        >
   709→          <Trash2 className="w-3.5 h-3.5" />
   710→        </button>
   711→
   712→        {/* Clear all */}
   713→        <button
   714→          onClick={handleClearAll}
   715→          title="پاک کردن همه"
   716→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-400 hover:bg-red-500/15 hover:text-red-400 transition-all cursor-pointer"
   717→        >
   718→          <X className="w-3.5 h-3.5" />
   719→        </button>
   720→
   721→        {/* Active tool label */}
   722→        <div className="mr-auto shrink-0 text-xs text-gray-500 font-medium" dir="rtl">
   723→          {TOOLS.find((t) => t.id === activeTool)?.label}
   724→        </div>
   725→      </div>
   726→
   727→      {/* ── CHART + SVG OVERLAY ─────────────────────────────── */}
   728→      <div
   729→        className="relative rounded-b-lg overflow-hidden border"
   730→        style={{ borderColor: TOOLBAR_BORDER, height }}
   731→      >
   732→        {/* Chart container */}
   733→        <div
   734→          ref={chartContainerRef}
   735→          className="w-full h-full"
   736→          data-chart
   737→        />
   738→
   739→        {/* SVG overlay for drawings */}
   740→        <svg
   741→          ref={svgRef}
   742→          className="absolute top-0 left-0"
   743→          style={{
   744→            width: chartSize.w || '100%',
   745→            height: chartSize.h || '100%',
   746→            pointerEvents: activeTool !== 'cursor' ? 'auto' : 'none',
   747→            zIndex: 10,
   748→          }}
   749→          onMouseDown={handleMouseDown}
   750→          onMouseMove={handleMouseMove}
   751→          onMouseUp={handleMouseUp}
   752→          onMouseLeave={handleMouseUp}
   753→        />
   754→
   755→        {/* Text input overlay */}
   756→        {textInput && (
   757→          <div
   758→            className="absolute z-20"
   759→            style={{
   760→              left: textInput.x,
   761→              top: textInput.y - 8,
   762→            }}
   763→          >
   764→            <input
   765→              type="text"
   766→              value={textValue}
   767→              onChange={(e) => setTextValue(e.target.value)}
   768→              onKeyDown={(e) => {
   769→                if (e.key === 'Enter') handleTextSubmit();
   770→                if (e.key === 'Escape') handleTextCancel();
   771→              }}
   772→              placeholder="متن..."
   773→              autoFocus
   774→              dir="rtl"
   775→              className="bg-[#0b0f1a] border border-gray-600 rounded px-2 py-1 text-sm text-white outline-none focus:border-amber-500 w-32"
   776→              onBlur={handleTextSubmit}
   777→            />
   778→          </div>
   779→        )}
   780→      </div>
   781→    </div>
   782→  );
   783→}
