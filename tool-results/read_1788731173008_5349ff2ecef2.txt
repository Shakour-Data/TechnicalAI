    17→    17→function safeNum(v: number, fallback: number = 0): number {
    18→    18→  return (typeof v === 'number' && isFinite(v)) ? v : fallback;
    19→    19→}
    20→    20→
    21→    21→function clamp(v: number, min: number, max: number): number {
    22→    22→  return Math.max(min, Math.min(max, safeNum(v, (min + max) / 2)));
    23→    23→}
    24→    24→
    25→    25→function normalize(arr: number[]): number[] {
    26→    26→  const safe = arr.map(v => safeNum(v, 0));
    27→    27→  const sum = safe.reduce((a, b) => a + b, 0);
    28→    28→  if (sum <= 0) return arr.map(() => 1 / arr.length);
    29→    29→  return safe.map((v) => v / sum);
    30→    30→}
    31→    31→
    32→    32→function sigmoid(x: number): number {
    33→    33→  const sx = safeNum(x, 0);
    34→    34→  if (Math.abs(sx) > 500) return sx > 0 ? 1 : 0; // prevent Infinity
    35→    35→  return 1 / (1 + Math.exp(-sx));
    36→    36→}

    38→    38→/**
    39→    39→ * IRON LAW: Round proportional floats to integers that sum to EXACTLY 100,
    40→    40→ * with each value constrained to [minVal, maxVal].
    41→    41→ *
    42→    42→ * Algorithm: Largest Remainder Method (LRM) + iterative bound enforcement.
    43→    43→ * This function GUARANTEES sum(result) === 100 and minVal <= result[i] <= maxVal
    44→    44→ * for all i, as long as n*minVal <= 100 <= n*maxVal.
    45→    45→ *
    46→    46→ * ⚠️  NEVER bypass this function when producing integer scenario percentages.
    47→    47→ */
    48→    48→function enforceSumTo100(
    49→    49→  rawFloats: number[],
    50→    50→  minVal: number,
    51→    51→  maxVal: number
    52→    52→): number[] {
    53→    53→  const n = rawFloats.length;
    54→    54→
    55→    55→  // Normalize floats to sum exactly to 100 first
    56→    56→  const floatSum = rawFloats.reduce((a, b) => a + b, 0);
    57→    57→  const normalized =
    58→    58→    floatSum > 0
    59→    59→      ? rawFloats.map((v) => (v / floatSum) * 100)
    60→    60→      : rawFloats.map(() => 100 / n);
    61→    61→
    62→    62→  // Step 1: LRM — floor all, distribute deficit by largest fractional part
    63→    63→  const result = normalized.map((v) => Math.max(Math.floor(v), 0));
    64→    64→  let deficit = 100 - result.reduce((a, b) => a + b, 0);
    65→    65→
    66→    66→  const byFrac = Array.from({ length: n }, (_, i) => ({
    67→    67→    idx: i,
    68→    68→    frac: normalized[i] - Math.floor(normalized[i]),
    69→    69→  })).sort((a, b) => b.frac - a.frac);
    70→    70→
    71→    71→  // Give +1 to top `deficit` entries (LRM)
    72→    72→  for (let d = 0; d < deficit && d < n; d++) {
    73→    73→    result[byFrac[d].idx]++;
    74→    74→  }
    75→    75→  // Now result sums to exactly 100
    76→    76→
    77→    77→  // Step 2: Iteratively enforce [minVal, maxVal] bounds while preserving sum = 100
    78→    78→  // Each iteration: move 1 from a violator to a suitable partner.
    79→    79→  // Converges because each move strictly reduces the total violation magnitude.
    80→    80→  let changed = true;
    81→    81→  let safety = 0;
    82→    82→  while (changed && safety < 100) {
    83→    83→    changed = false;
    84→    84→    safety++;
    85→    85→
    86→    86→    // Enforce minimum: raise any value below minVal by taking from the largest above minVal
    87→    87→    for (let i = 0; i < n; i++) {
    88→    88→      while (result[i] < minVal) {
    89→    89→        let donor = -1;
    90→    90→        for (let j = 0; j < n; j++) {
    91→    91→          if (j !== i && result[j] > minVal && (donor === -1 || result[j] > result[donor])) {
    92→    92→            donor = j;
    93→    93→          }
    94→    94→        }
    95→    95→        if (donor === -1) break; // no donor available — shouldn't happen if n*minVal <= 100
    96→    96→        result[donor]--;
    97→    97→        result[i]++;
    98→    98→        changed = true;
    99→    99→      }
   100→   100→    }
   101→   101→
   102→   102→    // Enforce maximum: lower any value above maxVal by giving to the smallest below maxVal
   103→   103→    for (let i = 0; i < n; i++) {
   104→   104→      while (result[i] > maxVal) {
   105→   105→        let receiver = -1;
   106→   106→        for (let j = 0; j < n; j++) {
   107→   107→          if (j !== i && result[j] < maxVal && (receiver === -1 || result[j] < result[receiver])) {
   108→   108→            receiver = j;
   109→   109→          }
   110→   110→        }
   111→   111→        if (receiver === -1) break; // no receiver — shouldn't happen if 100 <= n*maxVal
   112→   112→        result[i]--;
   113→   113→        result[receiver]++;
   114→   114→        changed = true;
   115→   115→      }
   116→   116→    }
   117→   117→  }
   118→   118→
   119→   119→  // Step 3: Final assertion — sum MUST be 100 (safety net, should never trigger)
   120→   120→  const finalSum = result.reduce((a, b) => a + b, 0);
   121→   121→  if (finalSum !== 100 && n > 0) {
   122→   122→    // Adjust the entry farthest from its bound
   123→   123→    const diff = 100 - finalSum;
   124→   124→    let bestIdx = 0;
   125→   125→    let bestRoom = -1;
   126→   126→    for (let i = 0; i < n; i++) {
   127→   127→      const room = diff > 0 ? maxVal - result[i] : result[i] - minVal;
   128→   128→      if (room > bestRoom) {
   129→   129→        bestRoom = room;
   130→   130→        bestIdx = i;
   131→   131→      }
   132→   132→    }
   133→   133→    result[bestIdx] += diff;
   134→   134→  }
   135→   135→
   136→   136→  return result;
   137→   137→}

  1125→  1125→function buildContext(input: GraphInput): ProbContext {
  1126→  1126→  const adxNorm = clamp(input.adx / 60, 0, 1);
  1127→  1127→  const srProximity = clamp(
  1128→  1128→    1 - Math.min(input.distToR1, input.distToS1) * 10,
  1129→  1129→    0,
  1130→  1130→    1
  1131→  1131→  );
  1132→  1132→  const oscillatorExtreme = clamp(
  1133→  1133→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
  1134→  1134→    0,
  1135→  1135→    1
  1136→  1136→  );
  1137→  1137→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1138→  1138→
  1139→  1139→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1140→  1140→  const overboughtRisk =
  1141→  1141→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1142→  1142→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1143→  1143→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1144→  1144→
  1145→  1145→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1146→  1146→  const oversoldBounce =
  1147→  1147→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1148→  1148→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1149→  1149→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1150→  1150→
  1151→  1151→  return {
  1152→  1152→    input,
  1153→  1153→    adxNorm,
  1154→  1154→    srProximity,
  1155→  1155→    oscillatorExtreme,
  1156→  1156→    divergenceProxy,
  1157→  1157→    overboughtRisk,
  1158→  1158→    oversoldBounce,
  1159→  1159→  };
  1160→  1160→}
  1161→  1161→
  1162→  1162→// === Node Probability Computation =============================================
  1163→  1163→// For each non-terminal node, returns an array of probabilities for its
  1164→  1164→// outgoing edges. The order MUST match the order of edges in createEdges().
  1165→  1165→// =============================================================================
  1166→  1166→
  1167→  1167→function computeNodeProbabilities(
  1168→  1168→  nodeId: string,
  1169→  1169→  ctx: ProbContext,
  1170→  1170→  _numChildren: number
  1171→  1171→): number[] {
  1172→  1172→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1173→  1173→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1174→  1174→
  1175→  1175→  switch (nodeId) {
  1176→  1176→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1177→  1177→    case 'ROOT': {
  1178→  1178→      const trendW = adxNorm * 0.5 + input.mlTrend * 0.18 + 0.1;
  1179→  1179→      const breakW = srProximity * 0.3 + (1 - adxNorm) * 0.08 + 0.08;
  1180→  1180→      const revW = ctx.divergenceProxy * 0.35 + (1 - adxNorm) * 0.12 + 0.06;
  1181→  1181→      return normalize([trendW, breakW, revW]);
  1182→  1182→    }
  1183→  1183→
  1184→  1184→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1185→  1185→    case 'N_TREND': {
  1186→  1186→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1187→  1187→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1188→  1188→      const flatUncertainty = (1 - Math.abs(input.bullConsensus - 0.5) * 2);
  1189→  1189→      const flat = flatUncertainty * clamp(1 - adxNorm * 1.5, 0, 1) * 0.45;
  1190→  1190→      return normalize([bull, bear, flat]);
  1191→  1191→    }
  1192→  1192→
  1193→  1193→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1194→  1194→    case 'N_T_BULL': {
  1195→  1195→      const momHigh =
  1196→  1196→        clamp(
  1197→  1197→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1198→  1198→            (input.macdHist > 0
  1199→  1199→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1200→  1200→              : 0),
  1201→  1201→          0,
  1202→  1202→          1
  1203→  1203→        ) * input.mlMomentum;
  1204→  1204→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1205→  1205→      return normalize([momHigh, momMod, overboughtRisk]);
  1206→  1206→    }
  1207→  1207→
  1208→  1208→    // ── N_T_B_MOM_HIGH → [SC8, SC9, SC7] ────────────────────────────────────
  1209→  1209→    // Strong bullish momentum — all three outcomes are bullish (no range)
  1210→  1210→    case 'N_T_B_MOM_HIGH': {
  1211→  1211→      const pSC8 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1212→  1212→      const pSC9 = 0.15 + input.strengthIndex * 0.2;
  1213→  1213→      const pSC7 = 0.35 * input.confidenceIndex + 0.15;
  1214→  1214→      return normalize([pSC8, pSC9, pSC7]);
  1215→  1215→    }
  1216→  1216→
  1217→  1217→    // ── N_T_B_MOM_MOD → [SC7, SC6, SC5] ─────────────────────────────────────
  1218→  1218→    // Moderate bullish — small explicit SC5 based on uncertainty
  1219→  1219→    case 'N_T_B_MOM_MOD': {
  1220→  1220→      const pSC7 = 0.55 * input.confidenceIndex + 0.2;
  1221→  1221→      const pSC6 = 0.55 * (1 - input.confidenceIndex) + 0.15;
  1222→  1222→      const pSC5 = clamp((1 - adxNorm) * 0.15, 0.02, 0.15);
  1223→  1223→      return normalize([pSC7, pSC6, pSC5]);
  1224→  1224→    }
  1225→  1225→
  1226→  1226→    // ── N_T_B_OVERBOUGHT → [SC4, SC1, SC9] ──────────────────────────────────
  1227→  1227→    // Overbought in uptrend — correction or last-gasp surge (no range)
  1228→  1228→    case 'N_T_B_OVERBOUGHT': {
  1229→  1229→      const pSC4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1230→  1230→      const pSC1 = 0.15 + input.strengthIndex * 0.15;
  1231→  1231→      const pSC9 = 0.1 + input.mlMomentum * 0.2;
  1232→  1232→      return normalize([pSC4, pSC1, pSC9]);
  1233→  1233→    }
  1234→  1234→
  1235→  1235→    // ── N_T_BEAR → [N_T_BE_MOM_LOW, N_T_BE_MOM_MOD, N_T_BE_OVERSOLD] ────
  1236→  1236→    case 'N_T_BEAR': {
  1237→  1237→      const momLow =
  1238→  1238→        clamp(
  1239→  1239→          (input.momentum < 0 ? Math.abs(input.momentum) / 50 : 0) +
  1240→  1240→            (input.macdHist < 0
  1241→  1241→              ? (Math.abs(input.macdHist) / (atrSafe * 0.5)) * 0.3
  1242→  1242→              : 0),
  1243→  1243→          0,
  1244→  1244→          1
  1245→  1245→        ) * input.mlMomentum;
  1246→  1246→      const momMod = (1 - momLow) * 0.6 * input.mlMomentum;
  1247→  1247→      return normalize([momLow, momMod, oversoldBounce]);
  1248→  1248→    }
  1249→  1249→
  1250→  1250→    // ── N_T_BE_MOM_LOW → [SC2, SC1, SC3] ────────────────────────────────────

  1493→  1493→// === Graph Traversal ==========================================================
  1494→  1494→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1495→  1495→// =============================================================================
  1496→  1496→
  1497→  1497→function traverseGraph(
  1498→  1498→  nodeId: string,
  1499→  1499→  pathProb: number,
  1500→  1500→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1501→  1501→  ctx: ProbContext,
  1502→  1502→  adj: Map<string, number[]>,
  1503→  1503→  edges: GraphEdge[],
  1504→  1504→  edgeProbabilities: Record<number, number>,
  1505→  1505→  nodeAccum: Record<string, number>,
  1506→  1506→  scenarios: Record<string, number>,
  1507→  1507→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1508→  1508→): void {
  1509→  1509→  // Accumulate path probability at this node
  1510→  1510→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1511→  1511→
  1512→  1512→  const outEdgeIndices = adj.get(nodeId);
  1513→  1513→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1514→  1514→
  1515→  1515→  // Compute outgoing edge probabilities for this node
  1516→  1516→  const probs = computeNodeProbabilities(
  1517→  1517→    nodeId,
  1518→  1518→    ctx,
  1519→  1519→    outEdgeIndices.length
  1520→  1520→  );
  1521→  1521→
  1522→  1522→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1523→  1523→    const edgeIdx = outEdgeIndices[i];
  1524→  1524→    const edge = edges[edgeIdx];
  1525→  1525→    const prob = probs[i] || 0;
  1526→  1526→
  1527→  1527→    // Store edge probability
  1528→  1528→    edgeProbabilities[edgeIdx] = prob;
  1529→  1529→
  1530→  1530→    // Determine branch for child nodes
  1531→  1531→    let childBranch = currentBranch;
  1532→  1532→    if (nodeId === 'ROOT') {
  1533→  1533→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1534→  1534→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1535→  1535→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1536→  1536→    }
  1537→  1537→
  1538→  1538→    const childPathProb = pathProb * prob;
  1539→  1539→
  1540→  1540→    // Check if target is a terminal node (SC1–SC9)
  1541→  1541→    const isTerminal =
  1542→  1542→      edge.to.length === 3 &&
  1543→  1543→      edge.to[0] === 'S' &&
  1544→  1544→      edge.to[1] === 'C' &&
  1545→  1545→      edge.to[2] >= '1' &&
  1546→  1546→      edge.to[2] <= '9';
  1547→  1547→
  1548→  1548→    if (isTerminal) {
  1549→  1549→      // Accumulate scenario probability
  1550→  1550→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1551→  1551→      // Accumulate branch contribution
  1552→  1552→      if (childBranch) {
  1553→  1553→        contributions[edge.to][childBranch] += childPathProb;
  1554→  1554→      }
  1555→  1555→    }
  1556→  1556→
  1557→  1557→    // Recurse into child
  1558→  1558→    traverseGraph(
  1559→  1559→      edge.to,
  1560→  1560→      childPathProb,
  1561→  1561→      childBranch,
  1562→  1562→      ctx,
  1563→  1563→      adj,
  1564→  1564→      edges,
  1565→  1565→      edgeProbabilities,
  1566→  1566→      nodeAccum,
  1567→  1567→      scenarios,
  1568→  1568→      contributions
  1569→  1569→    );
  1570→  1570→  }
  1571→  1571→}
