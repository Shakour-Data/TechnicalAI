   132→// Component
   133→// ═══════════════════════════════════════════════════════════════════════════════
   134→
   135→export default function VdssGraph(props: VdssGraphProps) {
   136→  const { colors: C } = useTheme();
   137→  const {
   138→    symbolName,
   139→    currentPrice,
   140→    scenarios,
   141→    decisionGraph,
   142→  } = props;
   143→
   144→  const [activeFilter, setActiveFilter] = useState('all');
   145→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   146→  const graphRef = useRef<HTMLDivElement>(null);
   147→  const svgRef = useRef<SVGSVGElement>(null);
   148→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   149→  const shellRef = useRef<HTMLDivElement>(null);
   150→
   151→  // ── Derived data from decisionGraph ──────────────────────────────
   152→  const edges = decisionGraph?.edges ?? [];
   153→  const nodes = decisionGraph?.nodes ?? [];
   154→  const nodePositions = decisionGraph?.nodePositions ?? {};
   155→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   156→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   157→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   158→  const pathContributions = decisionGraph?.pathContributions ?? {};
   159→
   160→  // Build node lookup map
   161→  const nodeMap = useMemo(() => {
   162→    const map: Record<string, (typeof nodes)[number]> = {};
   163→    for (const n of nodes) map[n.id] = n;
   164→    return map;
   165→  }, [nodes]);
   166→
   167→  // ── Visible edges/nodes based on filter ──────────────────────────
   168→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   169→    if (activeFilter === 'all') {
   170→      const allEdgeIdx = edges.map((_, i) => i);
   171→      const allNodes = new Set(nodes.map(n => n.id));
   172→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   173→    }
   174→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   175→      const edgeSet = new Set<number>();
   176→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   177→      edges.forEach((e, i) => {
   178→        if (e.to === activeFilter) {
   179→          edgeSet.add(i);
   180→          nodeSet.add(e.from);
   181→        }
   182→      });
   183→      edges.forEach((e, i) => {
   184→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   185→          edgeSet.add(i);
   186→          nodeSet.add(e.from);
   187→        }
   188→      });
   189→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   190→    }
   191→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   192→      const branchType = `branch-${activeFilter}` as const;
   193→      const edgeSet = new Set<number>();
   194→      const nodeSet = new Set<string>(['ROOT']);
   195→      edges.forEach((e, i) => {
   196→        if (e.type === branchType) {
   197→          edgeSet.add(i);
   198→          nodeSet.add(e.from);
   199→          nodeSet.add(e.to);
   200→        }
   201→      });
   202→      const expanded = new Set<number>();
   203→      const expandedNodes = new Set<string>(nodeSet);
   204→      edges.forEach((e, i) => {
   205→        if (expandedNodes.has(e.from)) {
   206→          expanded.add(i);
   207→          expandedNodes.add(e.to);
   208→        }
   209→      });
   210→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   211→    }
   212→    // Type filter (up/pullback/down/risk)
   213→    const edgeSet = new Set<number>();
   214→    const nodeSet = new Set<string>(['ROOT']);
   215→    edges.forEach((e, i) => {
   216→      if (e.type === activeFilter) {
   217→        edgeSet.add(i);
   218→        nodeSet.add(e.from);
   219→        nodeSet.add(e.to);
   220→      }
   221→    });
   222→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   223→  }, [activeFilter, edges, nodes]);
   224→
   225→  // ── Draw SVG edges ───────────────────────────────────────────────
   226→  const drawEdges = useCallback(() => {
   227→    const svg = svgRef.current;
   228→    const graph = graphRef.current;
   229→    if (!svg || !graph || edges.length === 0) return;
   230→
   231→    let markersSvg = '';
   232→    const seenColors = new Set<string>();
   233→    for (const e of edges) {
   234→      const c = EDGE_COLORS[e.type];
   235→      if (c && !seenColors.has(e.type)) {
   236→        seenColors.add(e.type);
   237→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="9" markerHeight="9" refX="8" refY="4.5" orient="auto"><path d="M0,0 L9,4.5 L0,9 z" fill="${c}"/></marker>`;
   238→      }
   239→    }
   240→
   241→    let pathsSvg = '';
   242→    const graphRect = graph.getBoundingClientRect();
   243→
   244→    for (let ei = 0; ei < edges.length; ei++) {
   245→      const { from: fromId, to: toId, label, type } = edges[ei];
   246→      const isVisible = visibleEdgeIndices.includes(ei);
   247→      const fromEl = nodeRefs.current[fromId];
   248→      const toEl = nodeRefs.current[toId];
   249→      if (!fromEl || !toEl) continue;
   250→
   251→      const fRect = fromEl.getBoundingClientRect();
   252→      const tRect = toEl.getBoundingClientRect();
   253→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   254→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   255→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   256→      const by = tRect.top - graphRect.top + tRect.height / 2;
   257→      const dx = bx - ax;
   258→      const dy = by - ay;
   259→      const dist = Math.sqrt(dx * dx + dy * dy);
   260→
   261→      if (dist === 0) continue;
   262→
   263→      const bend = Math.min(52, Math.max(18, dist * 0.11));
   264→      const mx = (ax + bx) / 2;
   265→      const my = (ay + by) / 2;
   266→      const nx = -dy / dist;
   267→      const ny = dx / dist;
   268→      const cx = mx + nx * bend;
   269→      const cy = my + ny * bend;
   270→
   271→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   272→      const prob = edgeProbabilities[ei] ?? 0;
   273→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   274→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   275→
   276→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.72 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   277→
   278→      if (isVisible && ei >= 6) {
   279→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#ffffff" font-size="10" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="4" stroke-linejoin="round" opacity="0.95" data-type="${type}" class="edge-label">${label}</text>`;
   280→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#07111b" stroke-width="3" stroke-linejoin="round" opacity="0.9" class="edge-prob">${probLabel}</text>`;
   281→      }
   282→    }
   283→
   284→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   285→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   286→
   287→  useEffect(() => { drawEdges(); }, [drawEdges]);
   288→
   289→  useEffect(() => {
   290→    const el = shellRef.current;
   291→    if (!el) return;
   292→    const observer = new ResizeObserver(() => drawEdges());
   293→    observer.observe(el);
   294→    return () => observer.disconnect();
   295→  }, [drawEdges]);
   296→
   297→
   298→  // ── Detail panel content ─────────────────────────────────────────
   299→  const detailContent = useMemo(() => {
   300→    if (!selectedNode) return null;
   301→    const node = nodeMap[selectedNode];
   302→    if (!node) return null;
   303→
   304→    if (node.isTerminal && SCENARIO_KEYS.includes(selectedNode as typeof SCENARIO_KEYS[number])) {
   305→      const contrib = pathContributions[selectedNode] ?? { trend: 0, breakout: 0, reversal: 0 };
   306→      const meta = SCENARIO_META_LOCAL[selectedNode];
   307→      const s = scenarios[selectedNode as keyof typeof scenarios];
   308→      return (
   309→        <div style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}>
   310→          <h3 style={{ fontSize: '14px', fontWeight: 700, margin: 0, color: meta.color }}>{node.title}</h3>
   311→          <p style={{ fontSize: '12px', color: '#ffffff', lineHeight: 1.95, margin: 0 }}>{node.desc}</p>
   312→          <span style={{
   313→            display: 'inline-block', padding: '4px 8px', margin: '3px 2px', borderRadius: 8,
   314→            color: '#ffffff', fontSize: '10px',
   315→            background: 'rgba(255,255,255,.06)', border: '1px solid rgba(255,255,255,.09)',
   316→          }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   317→          <div style={{ fontSize: '12px', color: meta.color, fontWeight: 700 }}>{toFa(scenarioProb)}٪</div>
   318→
   319→          <div style={{ marginTop: 12, paddingTop: 12, borderTop: `1px solid ${D.line}` }}>
   320→            <p style={{ fontSize: '12px', fontWeight: 700, marginBottom: 8, color: '#ffffff' }}>سهم هر استراتژی:</p>
   321→            {Object.entries(BRANCH_META).map(([bKey, bMeta]) => {
   322→              const val = contrib[bKey as 'trend' | 'breakout' | 'reversal'];
   323→              const pct = (val * 100).toFixed(1);
   324→              return (
   325→                <div key={bKey} style={{
   326→                  display: 'flex', alignItems: 'center', justifyContent: 'space-between',
   327→                  padding: '9px 0', borderBottom: '1px dashed rgba(255,255,255,.12)',
   328→                }}>
   329→                  <div style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
   330→                    <span style={{ display: 'inline-block', width: 8, height: 8, borderRadius: '50%', background: bMeta.color }} />
   331→                    <span style={{ fontSize: '11px', color: '#ffffff' }}>{bMeta.label}</span>
   332→                  </div>
   333→                  <span style={{ fontSize: '11px', fontWeight: 700, color: bMeta.color }}>{toPersianDigits(pct)}٪</span>
   334→                </div>
   335→              );
   336→            })}
   337→            <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '9px 0' }}>
   338→              <span style={{ fontSize: '11px', fontWeight: 700, color: '#ffffff' }}>مجموع</span>
   339→              <span style={{ fontSize: '11px', fontWeight: 900, color: meta.color }}>{toFa(s?.probability ?? 0)}٪</span>
   340→            </div>
   341→          </div>
   342→        </div>
   343→      );
   344→    }
   345→
   346→    const inputs = edges.filter(e => e.to === selectedNode);
   347→    const outputs = edges.filter(e => e.from === selectedNode);
   348→
   349→    return (
   350→      <div style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}>
   351→        <h3 style={{ fontSize: '14px', fontWeight: 700, margin: 0, color: '#ffffff' }}>{node.title}</h3>
   352→        <p style={{ fontSize: '12px', color: '#ffffff', lineHeight: 1.95, margin: 0 }}>{node.desc}</p>
   353→        <span style={{
   354→          display: 'inline-block', padding: '4px 8px', margin: '3px 2px', borderRadius: 8,
   355→          color: '#ffffff', fontSize: '10px',
   356→          background: 'rgba(255,255,255,.06)', border: '1px solid rgba(255,255,255,.09)',
   357→        }}>{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   358→        {inputs.length > 0 && (
   359→          <div>
   360→            <p style={{ fontSize: '12px', fontWeight: 600, marginBottom: 4, color: '#ffffff' }}>مسیرهای ورودی ({toFa(inputs.length)}):</p>
   361→            <ul style={{ margin: '12px 0 0', padding: 0, listStyle: 'none' }}>
   362→              {inputs.map((e, i) => {
   363→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   364→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   365→                return (
   366→                  <li key={i} style={{
   367→                    padding: '9px 0', borderTop: '1px dashed rgba(255,255,255,.12)',
   368→                    fontSize: '11px', lineHeight: 1.8, color: '#e0eaf0',
   369→                  }}>
   370→                    <b style={{ color: '#ffffff' }}>{SCENARIO_DISPLAY[e.from] || e.from} ← {SCENARIO_DISPLAY[e.to] || e.to}</b>
   371→                    <span style={{
   372→                      marginRight: 8, padding: '2px 6px', borderRadius: 4,
   373→                      fontSize: '9px', fontWeight: 700,
   374→                      background: `${edgeColor}22`, color: edgeColor,
   375→                    }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   376→                    <br />{e.label}
   377→                  </li>
   378→                );
   379→              })}
   380→            </ul>
   381→          </div>
   382→        )}
   383→        {outputs.length > 0 && (
   384→          <div>
   385→            <p style={{ fontSize: '12px', fontWeight: 600, marginBottom: 4, color: '#ffffff' }}>مسیرهای خروجی ({toFa(outputs.length)}):</p>
   386→            <ul style={{ margin: '12px 0 0', padding: 0, listStyle: 'none' }}>
   387→              {outputs.map((e, i) => {
   388→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   389→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   390→                return (
   391→                  <li key={i} style={{
   392→                    padding: '9px 0', borderTop: '1px dashed rgba(255,255,255,.12)',
   393→                    fontSize: '11px', lineHeight: 1.8, color: '#e0eaf0',
   394→                  }}>
   395→                    <b style={{ color: '#ffffff' }}>{SCENARIO_DISPLAY[e.from] || e.from} → {SCENARIO_DISPLAY[e.to] || e.to}</b>
   396→                    <span style={{
   397→                      marginRight: 8, padding: '2px 6px', borderRadius: 4,
   398→                      fontSize: '9px', fontWeight: 700,
   399→                      background: `${edgeColor}22`, color: edgeColor,
   400→                    }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   401→                    <br />{e.label}
   402→                  </li>
   403→                );
   404→              })}
   405→            </ul>
   406→          </div>
   407→        )}
   408→      </div>
   409→    );
   410→  }, [selectedNode, nodeMap, edges, edgeProbabilities, pathContributions, scenarios]);
   411→
   412→  // ── All filter buttons ────────────────────────────────────────────
   413→  const allFilters = [
   414→    ...TYPE_FILTERS,
   415→    { key: 'sep1', label: '│', isSep: true as const },
   416→    ...Object.entries(BRANCH_META).map(([k, v]) => ({ key: k, label: v.label, branchKey: k, branchColor: v.color })),
   417→    { key: 'sep2', label: '│', isSep: true as const },
   418→    ...SCENARIO_KEYS.map(k => ({ key: k, label: SCENARIO_META_LOCAL[k].label, scenarioKey: k })),
   419→  ];
   420→
   421→  // ── Corrected probabilityTrend: day 1 overridden with final scenario probabilities ──
   422→  const correctedTrend = useMemo(() => {
   423→    const raw = props.probabilityTrend;
   424→    if (!raw || !raw.scenarios || raw.scenarios.length === 0) return raw;
   425→
   426→    // Build final individual probabilities (matching boxes) for day 1
   427→    const day1Individuals: Record<string, number> = {};
   428→    for (const key of SCENARIO_KEYS) {
   429→      const s = scenarios[key as keyof typeof scenarios];
   430→      day1Individuals[key] = (s?.probability ?? 0) / 100;
   431→    }
   432→
   433→    // Corrected scenarios array
   434→    const correctedScenarios: ScenarioTrend[] = raw.scenarios.map(sc => {
   435→      const newInd = day1Individuals[sc.scenarioKey] ?? sc.trend[0]?.individualProb ?? 0;
   436→      const newCum = calculateCDF(sc.scenarioKey, day1Individuals);
   437→      const newTrend: DayPoint[] = sc.trend.map((dp, i) => {
   438→        if (i === 0) {
   439→          return { ...dp, individualProb: newInd, cumulativeProb: newCum };
   440→        }
   441→        return dp;
   442→      });
   443→      return {
   444→        ...sc,
   445→        currentProbability: newInd,
   446→        trend: newTrend,
   447→        peakProbability: Math.max(newInd, sc.peakProbability),

   983→function MultiSelectTrendChart({ data, selectedKeys, onToggleKey, mode, title }: {
   984→  data: ProbabilityTrendResult;
   985→  selectedKeys: Set<string>;
   986→  onToggleKey: (k: string) => void;
   987→  mode: 'individual' | 'cumulative';
   988→  title: string;
   989→}) {
   990→  const svgRef = useRef<SVGSVGElement>(null);
   991→  const tooltipRef = useRef<HTMLDivElement>(null);
   992→  const [hoveredDay, setHoveredDay] = useState<number | null>(null);
   993→
   994→  const scenarios = data.scenarios.filter(s => selectedKeys.has(s.scenarioKey));
   995→  const maxDays = data.scenarios[0]?.trend?.length ?? 0;
   996→
   997→  // Compute Y range — dynamic based on selected scenarios
   998→  let yMin = 0;
   999→  let yMax = 0;
  1000→  for (const sc of scenarios) {
  1001→    for (const d of sc.trend) {
  1002→      const v = mode === 'individual' ? d.individualProb : d.cumulativeProb;
  1003→      if (v > yMax) yMax = v;
  1004→    }
  1005→  }
  1006→  // For cumulative mode, ensure minimum range
  1007→  if (yMax <= 0) yMax = 0.1;
  1008→
  1009→  // Compute nice tick interval for Y-axis
  1010→  const range = yMax - yMin;
  1011→  const padding = range * 0.15;
  1012→  const paddedMax = yMax + padding;
  1013→  const paddedMin = Math.max(0, yMin - padding * 0.3);
  1014→
  1015→  // Find a nice step: aim for ~5 grid lines
  1016→  const niceSteps = [0.01, 0.02, 0.025, 0.05, 0.1, 0.15, 0.2, 0.25, 0.5, 1.0];
  1017→  const rawStep = (paddedMax - paddedMin) / 5;
  1018→  let step = niceSteps[0];
  1019→  for (const ns of niceSteps) {
  1020→    if (ns >= rawStep) { step = ns; break; }
  1021→  }
  1022→  // Round yMin down and yMax up to step boundaries
  1023→  const gridYMin = Math.floor(paddedMin / step) * step;
  1024→  const gridYMax = Math.ceil(paddedMax / step) * step;
  1025→  const gridRange = gridYMax - gridYMin;
  1026→
  1027→  const chartW = 1100;
  1028→  const chartH = 300;
  1029→  const padL = 55;
  1030→  const padR = 20;
  1031→  const padT = 20;
  1032→  const padB = 40;
  1033→  const plotW = chartW - padL - padR;
  1034→  const plotH = chartH - padT - padB;
  1035→
  1036→  const xOf = (day: number) => padL + plotW - ((day - 1) / Math.max(1, maxDays - 1)) * plotW;
  1037→  const yOf = (v: number) => padT + plotH - ((v - gridYMin) / Math.max(0.001, gridRange)) * plotH;
  1038→
  1039→  // Dynamic grid lines based on computed range
  1040→  const gridEls: React.JSX.Element[] = [];
  1041→  for (let v = gridYMin; v <= gridYMax + step * 0.01; v += step) {
  1042→    const y = yOf(v);
  1043→    if (y < padT - 2 || y > chartH - padB + 2) continue;
  1044→    const pctVal = v * 100;
  1045→    const pctLabel = Number.isInteger(pctVal) ? String(pctVal) : pctVal.toFixed(step < 0.05 ? 2 : 1);
  1046→    gridEls.push(
  1047→      <line key={`g${v}`} x1={padL} y1={y} x2={chartW - padR} y2={y} stroke="#f3f4f6" strokeWidth={0.7} />,
  1048→      <text key={`gl${v}`} x={padL - 6} y={y + 3.5} textAnchor="end" fill="#9ca3af" fontSize={9} fontFamily="inherit">
  1049→        {toPersianDigits(pctLabel)}%
  1050→      </text>,
  1051→    );
  1052→  }
  1053→  for (let d = 1; d <= maxDays; d += 5) {
  1054→    gridEls.push(
  1055→      <text key={`xl${d}`} x={xOf(d)} y={chartH - 8} textAnchor="middle" fill="#9ca3af" fontSize={8} fontFamily="inherit">
  1056→        {toPersianDigits(String(d))}
  1057→      </text>,
  1058→    );
  1059→  }
  1060→
  1061→  // Vertical hover line
  1062→  const hoverEl = hoveredDay ? (
  1063→    <line x1={xOf(hoveredDay)} y1={padT} x2={xOf(hoveredDay)} y2={chartH - padB} stroke="#94a3b8" strokeWidth={0.8} strokeDasharray="3,3" opacity={0.6} />
  1064→  ) : null;
  1065→
  1066→  // Tooltip
  1067→  const tooltipEl = hoveredDay ? (
  1068→    <div
  1069→      ref={tooltipRef}
  1070→      className="absolute z-10 bg-white border border-gray-200 rounded-lg shadow-lg p-3 text-[10px] pointer-events-none"
  1071→      style={{ left: xOf(hoveredDay) + 10, top: padT }}
  1072→      dir="rtl"
  1073→    >
  1074→      <div className="font-bold text-[#374151] mb-1">روز {toPersianDigits(String(hoveredDay))}</div>
  1075→      {scenarios.map(sc => {
  1076→        const dp = sc.trend.find(d => d.day === hoveredDay);
  1077→        if (!dp) return null;
  1078→        const val = mode === 'individual' ? dp.individualProb : dp.cumulativeProb;
  1079→        const color = SCENARIO_LINE_COLORS[sc.scenarioKey] || '#9ca3af';
  1080→        return (
  1081→          <div key={sc.scenarioKey} className="flex items-center justify-between gap-3">
  1082→            <span className="flex items-center gap-1">
  1083→              <span className="inline-block w-2 h-2 rounded-full" style={{ background: color }} />
  1084→              <span>{sc.label}</span>
  1085→            </span>
  1086→            <span className="font-bold" style={{ color }}>{toPersianDigits((val * 100).toFixed(1))}%</span>
  1087→          </div>
  1088→        );
  1089→      })}
  1090→    </div>
  1091→  ) : null;
  1092→
  1093→  return (
  1094→    <div className="rounded-2xl border border-[#e5e7eb] bg-[#ffffff] overflow-hidden"
  1095→      style={{ boxShadow: '0 2px 12px rgba(0,0,0,.05)' }}>
  1096→      {/* Header + Legend row */}
  1097→      <div className="px-4 pt-3 pb-2">
  1098→        <h3 className="text-xs font-bold text-[#111827] mb-2">{title}</h3>
  1099→        <div className="flex flex-wrap gap-x-4 gap-y-1.5">
  1100→          {data.scenarios.map(sc => {
  1101→            const color = SCENARIO_LINE_COLORS[sc.scenarioKey] || '#9ca3af';
  1102→            const isActive = selectedKeys.has(sc.scenarioKey);
  1103→            const trendIcon = sc.trendDirection === 'rising' ? '↑' : sc.trendDirection === 'falling' ? '↓' : '→';
  1104→            const curVal = sc.trend[0] ? (mode === 'individual' ? sc.trend[0].individualProb : sc.trend[0].cumulativeProb) : 0;
  1105→            return (
  1106→              <button
  1107→                key={sc.scenarioKey}
  1108→                onClick={() => onToggleKey(sc.scenarioKey)}
  1109→                className={`flex items-center gap-1.5 px-2 py-1 rounded-lg border text-[10px] font-medium transition-all cursor-pointer ${
  1110→                  isActive ? 'border-gray-300 bg-gray-50' : 'border-transparent bg-gray-100 opacity-40'
  1111→                }`}
  1112→              >
  1113→                <span className="inline-block w-2.5 h-2.5 rounded-sm" style={{ background: color }} />
  1114→                <span className="text-[#374151]">{sc.scenarioKey}</span>
  1115→                <span style={{ color }}>{trendIcon}</span>
  1116→                {isActive && <span className="text-[#6b7280]">{toPersianDigits((curVal * 100).toFixed(1))}%</span>}
  1117→              </button>
  1118→            );
  1119→          })}
  1120→        </div>
  1121→      </div>
  1122→
  1123→      {/* Chart area */}
  1124→      <div className="relative" style={{ overflowX: 'auto' }}>
  1125→        <svg
  1126→          ref={svgRef}
  1127→          width="100%"
  1128→          viewBox={`0 0 ${chartW} ${chartH}`}
  1129→          style={{ minWidth: 500, display: 'block' }}
  1130→          onMouseMove={(e) => {
  1131→            if (!svgRef.current) return;
  1132→            const rect = svgRef.current.getBoundingClientRect();
  1133→            const scaleX = chartW / rect.width;
  1134→            const svgX = (e.clientX - rect.left) * scaleX;
  1135→            // Find closest day
  1136→            let closestDay = 1;
  1137→            let closestDist = Infinity;
  1138→            for (let d = 1; d <= maxDays; d++) {
  1139→              const dist = Math.abs(xOf(d) - svgX);
  1140→              if (dist < closestDist) { closestDist = dist; closestDay = d; }
  1141→            }
  1142→            if (closestDist < 30) setHoveredDay(closestDay); else setHoveredDay(null);
  1143→          }}
  1144→          onMouseLeave={() => setHoveredDay(null)}
  1145→        >
  1146→          <rect x={0} y={0} width={chartW} height={chartH} fill="#fafafa" />
  1147→          {gridEls}
  1148→          <line x1={padL} y1={padT} x2={padL} y2={chartH - padB} stroke="#d1d5db" strokeWidth={0.7} />
  1149→          <line x1={padL} y1={chartH - padB} x2={chartW - padR} y2={chartH - padB} stroke="#d1d5db" strokeWidth={0.7} />
