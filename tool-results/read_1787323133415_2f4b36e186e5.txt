    99→def get_cached_model(symbol: str, model_name: str) -> Optional[Any]:
   100→    with _cache_lock:
   101→        entry = _model_cache.get(symbol, {}).get(model_name)
   102→        return entry["model"] if entry else None

   104→
   105→def get_cached_meta(symbol: str, model_name: str) -> Optional[Dict]:
   106→    with _cache_lock:
   107→        entry = _model_cache.get(symbol, {}).get(model_name)
   108→        return entry["meta"] if entry else None

   110→
   111→def get_cached_scaler(symbol: str) -> Optional[StandardScaler]:
   112→    with _cache_lock:
   113→        return _scaler_cache.get(symbol)

   115→
   116→def cache_model(symbol: str, model_name: str, model: Any, meta: Dict) -> None:
   117→    with _cache_lock:
   118→        _model_cache.setdefault(symbol, {})[model_name] = {"model": model, "meta": meta}
   119→
   120→
   121→def cache_scaler(symbol: str, scaler: StandardScaler) -> None:
   122→    with _cache_lock:
   123→        _scaler_cache[symbol] = scaler

   125→
   126→def flush_cache(symbol: Optional[str] = None) -> None:
   127→    """Flush model cache for a symbol or all."""
   128→    with _cache_lock:
   129→        if symbol:
   130→            _model_cache.pop(symbol, None)
   131→            _scaler_cache.pop(symbol, None)
   132→        else:
   133→            _model_cache.clear()
   134→            _scaler_cache.clear()

   142→def _safe_div(a, b, fill: float = 0.0):
   143→    """Divide with zero-safety for scalars and Series."""
   144→    if isinstance(a, pd.Series) or isinstance(b, pd.Series):
   145→        result = np.where(np.abs(b) > 1e-12, a / b, fill)
   146→        idx = a.index if isinstance(a, pd.Series) else b.index
   147→        return pd.Series(result, index=idx)
   148→    return a / b if abs(b) > 1e-12 else fill

   151→def extract_features(df: pd.DataFrame) -> pd.DataFrame:
   152→    """
   153→    Build 40+ technical features from OHLCV data.
   154→
   155→    Parameters
   156→    ----------
   157→    df : pd.DataFrame
   158→        Must contain columns: open, high, low, close, volume (lowercase).
   159→        Index should be sequential integers.
   160→
   161→    Returns
   162→    -------
   163→    pd.DataFrame  —  same row count, NaNs at the head where windows can't fill.
   164→    """
   165→    out = pd.DataFrame(index=df.index)
   166→    c = df["close"].astype(float)
   167→    h = df["high"].astype(float)
   168→    l = df["low"].astype(float)
   169→    o = df["open"].astype(float)
   170→    v = df["volume"].astype(float)
   171→
   172→    # ── Price returns ──
   173→    for period in (1, 3, 5, 10, 20):
   174→        out[f"return_{period}d"] = c.pct_change(period)
   175→
   176→    # ── Rolling statistics ──
   177→    for w in (5, 10, 20, 60):
   178→        roll = c.rolling(w, min_periods=1)
   179→        out[f"close_mean_{w}"] = roll.mean()
   180→        out[f"close_std_{w}"] = roll.std()
   181→        out[f"close_min_{w}"] = roll.min()
   182→        out[f"close_max_{w}"] = roll.max()
   183→
   184→    # ── RSI(14) ──
   185→    delta = c.diff()
   186→    gain = delta.clip(lower=0).rolling(14, min_periods=1).mean()
   187→    loss = (-delta.clip(upper=0)).rolling(14, min_periods=1).mean()
   188→    rs = _safe_div(gain, loss, fill=100.0)
   189→    out["rsi_14"] = 100.0 - (100.0 / (1.0 + rs))
   190→
   191→    # ── MACD(12,26,9) ──
   192→    ema12 = c.ewm(span=12, adjust=False).mean()
   193→    ema26 = c.ewm(span=26, adjust=False).mean()
   194→    out["macd"] = ema12 - ema26
   195→    out["macd_signal"] = out["macd"].ewm(span=9, adjust=False).mean()
   196→    out["macd_hist"] = out["macd"] - out["macd_signal"]
   197→
   198→    # ── Stochastic(14,3,3) ──
   199→    low14 = l.rolling(14, min_periods=1).min()
   200→    high14 = h.rolling(14, min_periods=1).max()
   201→    out["stoch_k"] = 100.0 * _safe_div(c - low14, high14 - low14)
   202→    out["stoch_d"] = out["stoch_k"].rolling(3, min_periods=1).mean()
   203→
   204→    # ── CCI(20) ──
   205→    typical = (h + l + c) / 3.0
   206→    sma20 = typical.rolling(20, min_periods=1).mean()
   207→    mad20 = typical.rolling(20, min_periods=1).apply(lambda x: np.mean(np.abs(x - x.mean())), raw=True)
   208→    out["cci_20"] = _safe_div(typical - sma20, 0.015 * mad20)
   209→
   210→    # ── ATR(14) ──
   211→    tr = pd.concat([h - l, (h - c.shift(1)).abs(), (l - c.shift(1)).abs()], axis=1).max(axis=1)
   212→    out["atr_14"] = tr.rolling(14, min_periods=1).mean()
   213→
   214→    # ── ADX(14) ──
   215→    plus_dm = h.diff()
   216→    minus_dm = -l.diff()
   217→    plus_dm = plus_dm.where((plus_dm > minus_dm) & (plus_dm > 0), 0.0)
   218→    minus_dm = minus_dm.where((minus_dm > plus_dm) & (minus_dm > 0), 0.0)
   219→    tr_smooth = tr.ewm(span=14, adjust=False).mean()
   220→    plus_di = 100.0 * _safe_div(plus_dm.ewm(span=14, adjust=False).mean(), tr_smooth)
   221→    minus_di = 100.0 * _safe_div(minus_dm.ewm(span=14, adjust=False).mean(), tr_smooth)
   222→    dx_sum = _safe_div(abs(plus_di - minus_di), plus_di + minus_di) * 100.0
   223→    out["adx_14"] = dx_sum.ewm(span=14, adjust=False).mean()
   224→
   225→    # ── Bollinger Bands position ──
   226→    bb_mid = c.rolling(20, min_periods=1).mean()
   227→    bb_std = c.rolling(20, min_periods=1).std()
   228→    out["bb_position"] = _safe_div(c - bb_mid, 2.0 * bb_std)
   229→
   230→    # ── Volume features ──
   231→    out["volume_sma_20"] = v.rolling(20, min_periods=1).mean()
   232→    out["volume_ratio"] = _safe_div(v, out["volume_sma_20"])
   233→    out["volume_zscore"] = _safe_div(v - out["volume_sma_20"], v.rolling(20, min_periods=1).std())
   234→
   235→    # OBV (vectorized)
   236→    direction = pd.Series(np.sign(c.diff()), index=df.index).fillna(0)
   237→    obv = (direction * v).cumsum()
   238→    out["obv"] = obv
   239→
   240→    # ── Price patterns (rolling 10-bar) ──
   241→    is_hh = (h > h.shift(1)).astype(float)
   242→    is_ll = (l < l.shift(1)).astype(float)
   243→    is_inside = ((h <= h.shift(1)) & (l >= l.shift(1))).astype(float)
   244→    pw = 10
   245→    out["higher_highs"] = is_hh.rolling(pw, min_periods=1).mean()
   246→    out["lower_lows"] = is_ll.rolling(pw, min_periods=1).mean()
   247→    out["inside_bar_ratio"] = is_inside.rolling(pw, min_periods=1).mean()
   248→
   249→    # ── Momentum ──
   250→    out["roc_10"] = _safe_div(c - c.shift(10), c.shift(10)) * 100.0
   251→    momentum_score = (
   252→        out.get("rsi_14", pd.Series(50, index=df.index)) - 50.0
   253→        + out.get("macd_hist", pd.Series(0, index=df.index)) * 10.0
   254→        + out.get("stoch_k", pd.Series(50, index=df.index)) - 50.0
   255→    ) / 3.0
   256→    out["momentum_score"] = momentum_score
   257→
   258→    # ── Lag features (returns & RSI for last 5 periods) ──
   259→    for lag in range(1, 6):
   260→        out[f"return_1d_lag{lag}"] = out["return_1d"].shift(lag)
   261→        out[f"rsi_14_lag{lag}"] = out["rsi_14"].shift(lag)
   262→
   263→    return out
   264→

   270→
   271→def build_ohlcv_df(ohlcv: List[List]) -> pd.DataFrame:
   272→    """Convert raw OHLCV list to DataFrame with lowercase columns."""
   273→    df = pd.DataFrame(ohlcv, columns=["date", "open", "high", "low", "close", "volume"])

   275→        df[col] = pd.to_numeric(df[col], errors="coerce")
   276→    df["date"] = df["date"].astype(str)
   277→    return df

   282→# ─────────────────────────────────────────────────────────────────────────────

   284→
   285→def build_direction_target(df: pd.DataFrame) -> pd.Series:
   286→    """Binary: 1 if 5-day forward return > 0.5%, else 0."""
   287→    fwd = df["close"].shift(-FORWARD_DAYS) / df["close"] - 1.0
   288→    return (fwd > DIRECTION_THRESHOLD).astype(int)
   289→
   290→
   291→def build_regression_target(df: pd.DataFrame) -> pd.Series:
   292→    """5-day forward return as percentage."""
   293→    return (df["close"].shift(-FORWARD_DAYS) / df["close"] - 1.0) * 100.0
   294→
   295→
   296→def build_volatility_target(df: pd.DataFrame) -> pd.Series:
   297→    """Next-10-day realized volatility (std of daily returns * sqrt(252) * 100)."""
   298→    ret = df["close"].pct_change()
   299→    return ret.shift(-VOLATILITY_WINDOW).rolling(VOLATILITY_WINDOW).std() * np.sqrt(252) * 100.0

   306→
   307→def _save_model_artifact(model: Any, path: str, input_dim: int) -> None:
   308→    """Save model as ONNX (preferred) or pickle fallback."""
   309→    import onnx
   310→    os.makedirs(os.path.dirname(path), exist_ok=True)
   311→    if ONNX_EXPORT_AVAILABLE:
   312→        try:
   313→            initial_type = [('float_input', FloatTensorType([None, input_dim]))]
   314→            if isinstance(model, xgb.XGBModel):
   315→                # XGBoost via onnxmltools
   316→                onnx_model = onnxmltools.convert_xgboost(model, initial_types=initial_type)
   317→            elif hasattr(model, 'predict_proba') or hasattr(model, 'predict'):
   318→                # sklearn model
   319→                onnx_model = skl2onnx.convert_sklearn(model, initial_types=initial_type)
   320→            else:
   321→                raise ValueError("Unknown model type for ONNX export")
   322→            onnx.save(onnx_model, path)
   323→            return
   324→        except Exception as e:
   325→            log("WARN", f"ONNX export failed for {path}: {e}, falling back to pickle")
   326→    # Pickle fallback
   327→    pkl_path = path.replace(".onnx", ".pkl")
   328→    with open(pkl_path, "wb") as f:
   329→        pickle.dump(model, f)

   331→
   332→def _load_model_artifact(path: str) -> Tuple[Any, bool]:
   333→    """Load model from ONNX or pickle. Returns (model_or_session, is_onnx)."""
   334→    if os.path.exists(path):
   335→        try:
   336→            session = ort.InferenceSession(path)
   337→            return session, True
   338→        except Exception:
   339→            pass
   340→    pkl_path = path.replace(".onnx", ".pkl")
   341→    if os.path.exists(pkl_path):
   342→        with open(pkl_path, "rb") as f:
   343→            return pickle.load(f), False
   344→    return None, False

   367→def _load_scaler(symbol: str) -> Optional[StandardScaler]:
   368→    cached = get_cached_scaler(symbol)
   369→    if cached is not None:
   370→        return cached
   371→    path = os.path.join(MODELS_DIR, symbol, "scaler.pkl")
   372→    if os.path.exists(path):
   373→        with open(path, "rb") as f:
   374→            scaler = pickle.load(f)
   375→        cache_scaler(symbol, scaler)
   376→        return scaler
   377→    return None

   384→
   385→def train_models(
   386→    symbol: str,
   387→    ohlcv: List[List],
   388→    model_names: Optional[List[str]] = None,
   389→) -> Dict[str, Any]:
   390→    """
   391→    Train requested models for a symbol.
   392→
   393→    Returns a dict with metrics and metadata.
   394→    """
   395→    if model_names is None:
   396→        model_names = MODEL_TYPES
   397→
   398→    # Validate
   399→    for mn in model_names:
   400→        if mn not in MODEL_TYPES:
   401→            raise ValueError(f"Unknown model type: {mn}")
   402→
   403→    df = build_ohlcv_df(ohlcv)
   404→    if len(df) < MIN_CANDLES:
   405→        raise ValueError(f"Need at least {MIN_CANDLES} candles, got {len(df)}")
   406→
   407→    log("INFO", f"Training {model_names} for {symbol} with {len(df)} candles")
   408→
   409→    # Feature engineering
   410→    features_df = extract_features(df)
   411→    feature_cols = [c for c in features_df.columns]
   412→
   413→    # Drop rows with NaN
   414→    valid_mask = features_df[feature_cols].notna().all(axis=1)
   415→    X = features_df.loc[valid_mask, feature_cols].values
   416→    feature_index = features_df.index[valid_mask]
   417→
   418→    # Build targets
   419→    dir_target = build_direction_target(df).loc[feature_index].values
   420→    reg_target = build_regression_target(df).loc[feature_index].values
   421→    vol_target = build_volatility_target(df).loc[feature_index].values
   422→
   423→    # Drop rows where targets are NaN
   424→    valid2 = ~(np.isnan(dir_target) | np.isnan(reg_target) | np.isnan(vol_target))
   425→    X = X[valid2]
   426→    dir_target = dir_target[valid2]
   427→    reg_target = reg_target[valid2]
   428→    vol_target = vol_target[valid2]
   429→
   430→    if len(X) < MIN_CANDLES:
   431→        raise ValueError(f"After cleaning, only {len(X)} valid samples (need {MIN_CANDLES})")
   432→
   433→    # Scale features
   434→    scaler = StandardScaler()
   435→    X_scaled = scaler.fit_transform(X)
   436→    input_dim = X_scaled.shape[1]
   437→
   438→    # Save scaler
   439→    _save_scaler(scaler, symbol)
   440→    cache_scaler(symbol, scaler)
   441→
   442→    # TimeSeriesSplit
   443→    tscv = TimeSeriesSplit(n_splits=5)
   444→    metrics: Dict[str, Dict] = {}
   445→    all_feature_importance: Dict[str, float] = {}
   446→
   447→    sym_dir = os.path.join(MODELS_DIR, symbol)
   448→    os.makedirs(sym_dir, exist_ok=True)
   449→
   450→    for mn in model_names:
   451→        log("INFO", f"  Training model: {mn}")
   452→
   453→        if mn == "xgboost_direction":
   454→            model = xgb.XGBClassifier(
   455→                n_estimators=200,
   456→                max_depth=5,
   457→                learning_rate=0.05,
   458→                subsample=0.8,
   459→                colsample_bytree=0.8,
   460→                use_label_encoder=False,
   461→                eval_metric="logloss",
   462→                verbosity=0,
   463→                n_jobs=-1,
   464→            )
   465→            target = dir_target
   466→            fold_metrics = {"accuracy": [], "f1": [], "auc": []}
   467→
   468→            for train_idx, test_idx in tscv.split(X_scaled):
   469→                model.fit(X_scaled[train_idx], target[train_idx])
   470→                preds = model.predict(X_scaled[test_idx])
   471→                proba = model.predict_proba(X_scaled[test_idx])[:, 1]
   472→                fold_metrics["accuracy"].append(accuracy_score(target[test_idx], preds))
   473→                fold_metrics["f1"].append(f1_score(target[test_idx], preds, zero_division=0))
   474→                try:
   475→                    fold_metrics["auc"].append(roc_auc_score(target[test_idx], proba))
   476→                except Exception:
   477→                    fold_metrics["auc"].append(0.5)
   478→
   479→            # Final fit on all data
   480→            model.fit(X_scaled, target)
   481→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   482→            imp = model.feature_importances_
   483→
   484→        elif mn == "xgboost_regression":
   485→            model = xgb.XGBRegressor(
   486→                n_estimators=200,
   487→                max_depth=5,
   488→                learning_rate=0.05,
   489→                subsample=0.8,
   490→                colsample_bytree=0.8,
   491→                verbosity=0,
   492→                n_jobs=-1,
   493→            )
   494→            target = reg_target
   495→            fold_metrics = {"mae": [], "rmse": [], "r2": []}
   496→
   497→            for train_idx, test_idx in tscv.split(X_scaled):
   498→                model.fit(X_scaled[train_idx], target[train_idx])
   499→                preds = model.predict(X_scaled[test_idx])
   500→                fold_metrics["mae"].append(mean_absolute_error(target[test_idx], preds))
   501→                fold_metrics["rmse"].append(np.sqrt(mean_squared_error(target[test_idx], preds)))
   502→                fold_metrics["r2"].append(r2_score(target[test_idx], preds))
   503→
   504→            model.fit(X_scaled, target)
   505→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   506→            imp = model.feature_importances_
   507→
   508→        elif mn == "ensemble_direction":
   509→            model = GradientBoostingClassifier(
   510→                n_estimators=200,
   511→                max_depth=4,
   512→                learning_rate=0.05,
   513→                subsample=0.8,
   514→                max_features="sqrt",
   515→                random_state=42,
   516→            )
   517→            target = dir_target
   518→            fold_metrics = {"accuracy": [], "f1": [], "auc": []}
   519→
   520→            for train_idx, test_idx in tscv.split(X_scaled):
   521→                model.fit(X_scaled[train_idx], target[train_idx])
   522→                preds = model.predict(X_scaled[test_idx])
   523→                proba = model.predict_proba(X_scaled[test_idx])[:, 1]
   524→                fold_metrics["accuracy"].append(accuracy_score(target[test_idx], preds))
   525→                fold_metrics["f1"].append(f1_score(target[test_idx], preds, zero_division=0))
   526→                try:
   527→                    fold_metrics["auc"].append(roc_auc_score(target[test_idx], proba))
   528→                except Exception:
   529→                    fold_metrics["auc"].append(0.5)
   530→
   531→            model.fit(X_scaled, target)
   532→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   533→            imp = model.feature_importances_
   534→
   535→        elif mn == "ensemble_regression":
   536→            model = GradientBoostingRegressor(
   537→                n_estimators=200,
   538→                max_depth=4,
   539→                learning_rate=0.05,
   540→                subsample=0.8,
   541→                max_features="sqrt",
   542→                random_state=42,
   543→            )
   544→            target = reg_target
   545→            fold_metrics = {"mae": [], "rmse": [], "r2": []}
   546→
   547→            for train_idx, test_idx in tscv.split(X_scaled):
   548→                model.fit(X_scaled[train_idx], target[train_idx])
   549→                preds = model.predict(X_scaled[test_idx])
   550→                fold_metrics["mae"].append(mean_absolute_error(target[test_idx], preds))
   551→                fold_metrics["rmse"].append(np.sqrt(mean_squared_error(target[test_idx], preds)))
   552→                fold_metrics["r2"].append(r2_score(target[test_idx], preds))
   553→
   554→            model.fit(X_scaled, target)
   555→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   556→            imp = model.feature_importances_
   557→
   558→        elif mn == "volatility_model":
   559→            model = RandomForestRegressor(
   560→                n_estimators=200,
   561→                max_depth=6,
   562→                min_samples_leaf=5,
   563→                random_state=42,
   564→                n_jobs=-1,
   565→            )
   566→            target = vol_target
   567→            fold_metrics = {"mae": [], "rmse": [], "r2": []}
   568→
   569→            for train_idx, test_idx in tscv.split(X_scaled):
   570→                model.fit(X_scaled[train_idx], target[train_idx])
   571→                preds = model.predict(X_scaled[test_idx])
   572→                fold_metrics["mae"].append(mean_absolute_error(target[test_idx], preds))
   573→                fold_metrics["rmse"].append(np.sqrt(mean_squared_error(target[test_idx], preds)))
   574→                fold_metrics["r2"].append(r2_score(target[test_idx], preds))
   575→
   576→            model.fit(X_scaled, target)
   577→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   578→            imp = model.feature_importances_
   579→
   580→        else:
   581→            continue
   582→
   583→        # Save model artifact
   584→        onnx_path = os.path.join(sym_dir, f"{mn}.onnx")
   585→        _save_model_artifact(model, onnx_path, input_dim)
   586→
   587→        # Cache in memory
   588→        cache_model(symbol, mn, model, metrics.get(mn, {}))
   589→
   590→        # Save metadata
   591→        meta = {
   592→            "model_type": mn,
   593→            "symbol": symbol,
   594→            "metrics": metrics.get(mn, {}),
   595→            "feature_cols": feature_cols,
   596→            "input_dim": input_dim,
   597→            "training_samples": int(len(X)),
   598→            "date_range": [str(df["date"].iloc[0]), str(df["date"].iloc[-1])],
   599→            "trained_at": datetime.utcnow().isoformat() + "Z",
   600→        }
   601→        _save_meta(meta, os.path.join(sym_dir, f"{mn}_meta.json"))
   602→
   603→        # Accumulate feature importance (average across models)
   604→        for i, col in enumerate(feature_cols):
   605→            all_feature_importance[col] = all_feature_importance.get(col, 0.0) + float(imp[i])
   606→
   607→    # Normalize feature importance
   608→    total_imp = sum(all_feature_importance.values())
   609→    if total_imp > 0:
   610→        all_feature_importance = {k: round(v / total_imp, 4) for k, v in all_feature_importance.items()}
   611→
   612→    # Top 10
   613→    top_features = dict(sorted(all_feature_importance.items(), key=lambda x: x[1], reverse=True)[:10])
   614→
   615→    return {
   616→        "status": "ok",
   617→        "symbol": symbol,
   618→        "models_trained": model_names,
   619→        "metrics": metrics,
   620→        "feature_importance": top_features,
   621→        "training_samples": len(X),
   622→        "date_range": [str(df["date"].iloc[0]), str(df["date"].iloc[-1])],
   623→    }

   630→
   631→def _predict_with_model(model: Any, features: np.ndarray, is_onnx: bool) -> np.ndarray:
   632→    """Run prediction on a single row of features."""
   633→    if is_onnx:
   634→        session = model  # ONNX Runtime session
   635→        input_name = session.get_inputs()[0].name
   636→        result = session.run(None, {input_name: features.astype(np.float32)})
   637→        # Classification: result[1] is probabilities; Regression: result[0] is values
   638→        if len(result) == 2 and result[1].shape[1] >= 2:
   639→            return result[1]  # probabilities
   640→        return result[0]  # regression
   641→    else:
   642→        # sklearn or xgboost native
   643→        if hasattr(model, "predict_proba"):
   644→            return model.predict_proba(features)  # type: ignore
   645→        return model.predict(features)  # type: ignore

   647→
   648→def predict_sessions(
   649→    symbol: str,
   650→    ohlcv: List[List],
   651→    horizon: int = 30,
   652→) -> Dict[str, Any]:
   653→    """
   654→    Generate multi-session prediction for a symbol.
   655→
   656→    Uses iterative prediction: predict 5 sessions, append, re-extract, repeat.
   657→    """
   658→    df = build_ohlcv_df(ohlcv)
   659→    if len(df) < MIN_CANDLES:
   660→        raise ValueError(f"Need at least {MIN_CANDLES} candles for prediction, got {len(df)}")
   661→
   662→    # Load scaler
   663→    scaler = _load_scaler(symbol)
   664→    if scaler is None:
   665→        raise ValueError(f"No trained scaler for symbol {symbol}")
   666→
   667→    # Determine which models are available
   668→    sym_dir = os.path.join(MODELS_DIR, symbol)
   669→    available_models = []
   670→    dir_model = None
   671→    dir_is_onnx = False
   672→    reg_model = None
   673→    reg_is_onnx = False
   674→
   675→    for mn in ("xgboost_direction", "ensemble_direction"):
   676→        model, is_onnx = _load_model_artifact(os.path.join(sym_dir, f"{mn}.onnx"))
   677→        if model is not None:
   678→            dir_model = model
   679→            dir_is_onnx = is_onnx
   680→            available_models.append(mn)
   681→            break
   682→
   683→    for mn in ("xgboost_regression", "ensemble_regression"):
   684→        model, is_onnx = _load_model_artifact(os.path.join(sym_dir, f"{mn}.onnx"))
   685→        if model is not None:
   686→            reg_model = model
   687→            reg_is_onnx = is_onnx
   688→            available_models.append(mn)
   689→            break
   690→
   691→    if dir_model is None and reg_model is None:
   692→        raise ValueError(f"No trained model for symbol {symbol}")
   693→
   694→    # Extract current features for the last candle
   695→    features_df = extract_features(df)
   696→    feature_cols = list(features_df.columns)
   697→    last_row = features_df.iloc[-1:].values
   698→    if np.any(np.isnan(last_row)):
   699→        # Forward-fill NaNs with column means
   700→        col_means = features_df.mean().values
   701→        nan_mask = np.isnan(last_row)
   702→        last_row[nan_mask] = np.take(col_means, np.where(nan_mask)[1])
   703→
   704→    current_features = {}
   705→    for i, col in enumerate(feature_cols):
   706→        current_features[col] = round(float(last_row[0, i]), 4)
   707→
   708→    # Iterative prediction
   709→    working_df = df.copy()
   710→    sessions: List[Dict] = []
   711→    last_close = float(df["close"].iloc[-1])
   712→    iterations = (horizon + PREDICTION_STEPS - 1) // PREDICTION_STEPS
   713→
   714→    for it in range(iterations):
   715→        remaining = horizon - len(sessions)
   716→        steps = min(PREDICTION_STEPS, remaining)
   717→
   718→        # Extract features from current working dataframe
   719→        feat_df = extract_features(working_df)
   720→        feat_cols = list(feat_df.columns)
   721→        last_features = feat_df.iloc[-1:].values
   722→
   723→        # Handle NaNs
   724→        if np.any(np.isnan(last_features)):
   725→            col_means = feat_df.mean().values
   726→            for j in range(last_features.shape[1]):
   727→                if np.isnan(last_features[0, j]):
   728→                    last_features[0, j] = col_means[j]
   729→
   730→        scaled = scaler.transform(last_features)
   731→
   732→        predicted_change = 0.0
   733→        direction = "flat"
   734→        confidence = 0.5
   735→
   736→        if reg_model is not None:
   737→            reg_result = _predict_with_model(reg_model, scaled, reg_is_onnx)
   738→            predicted_change = float(reg_result[0])
   739→
   740→        if dir_model is not None:
   741→            dir_result = _predict_with_model(dir_model, scaled, dir_is_onnx)
   742→            # dir_result is probability array [p_down, p_up]
   743→            if dir_result.ndim == 2 and dir_result.shape[1] >= 2:
   744→                p_up = float(dir_result[0, 1])
   745→            else:
   746→                p_up = float(dir_result[0]) if float(dir_result[0]) > 0.5 else 0.5
   747→            confidence = p_up if p_up > 0.5 else (1.0 - p_up)
   748→            direction = "up" if p_up > 0.5 else "down"
   749→
   750→        # If only regression model, derive direction from predicted change
   751→        if dir_model is None and reg_model is not None:
   752→            if predicted_change > DIRECTION_THRESHOLD * 100:
   753→                direction = "up"
   754→                confidence = min(0.9, 0.5 + abs(predicted_change) / 5.0)
   755→            elif predicted_change < -DIRECTION_THRESHOLD * 100:
   756→                direction = "down"
   757→                confidence = min(0.9, 0.5 + abs(predicted_change) / 5.0)
   758→
   759→        # Generate steps sessions
   760→        for s in range(steps):
   761→            session_num = len(sessions) + 1
   762→            step_change = predicted_change / steps
   763→            new_close = last_close * (1.0 + step_change / 100.0)
   764→            change_pct = round((new_close - last_close) / last_close * 100, 4)
   765→
   766→            sessions.append({
   767→                "session": session_num,
   768→                "predicted_close": round(new_close, 2),
   769→                "confidence": round(confidence, 4),
   770→                "direction": direction,
   771→                "change_pct": change_pct,
   772→            })
   773→            last_close = new_close
   774→
   775→        # Append predicted candle to working_df for next iteration
   776→        avg_change = predicted_change / steps
   777→        new_close_row = last_close * (1.0 + avg_change / 100.0)
   778→        new_row = {
   779→            "date": f"pred_{len(sessions)}",
   780→            "open": last_close,
   781→            "high": max(last_close, new_close_row) * (1.0 + abs(avg_change) / 200.0),
   782→            "low": min(last_close, new_close_row) * (1.0 - abs(avg_change) / 200.0),
   783→            "close": new_close_row,
   784→            "volume": 0.0,
   785→        }
   786→        working_df = pd.concat([working_df, pd.DataFrame([new_row])], ignore_index=True)
   787→
   788→    # Aggregate results
   789→    closes = [s["predicted_close"] for s in sessions]
   790→    min_price = min(closes)
   791→    max_price = max(closes)
   792→    final_close = closes[-1]
   793→    initial_close = float(df["close"].iloc[-1])
   794→    overall_change = (final_close - initial_close) / initial_close * 100
   795→
   796→    if overall_change > 1.0:
   797→        overall_direction = "up"
   798→    elif overall_change < -1.0:
   799→        overall_direction = "down"
   800→    else:
   801→        overall_direction = "flat"
   802→
   803→    avg_confidence = np.mean([s["confidence"] for s in sessions])
   804→    price_range = max_price - min_price
   805→    price_range_pct = price_range / initial_close * 100
   806→
   807→    if price_range_pct > 5.0:
   808→        risk_level = "high"
   809→    elif price_range_pct > 2.5:
   810→        risk_level = "medium"
   811→    else:
   812→        risk_level = "low"
   813→
   814→    return {
   815→        "status": "ok",
   816→        "symbol": symbol,
   817→        "prediction": {
   818→            "sessions": sessions,
   819→            "overall_direction": overall_direction,
   820→            "overall_confidence": round(float(avg_confidence), 4),
   821→            "target_price_min": round(min_price, 2),
   822→            "target_price_max": round(max_price, 2),
   823→            "risk_level": risk_level,
   824→        },
   825→        "models_used": available_models,
   826→        "current_features": current_features,
   827→    }
   828→

   835→def list_all_models() -> Dict[str, Any]:
   836→    """List all trained models across all symbols."""
   837→    result: Dict[str, List[Dict]] = {}
   838→    if not os.path.exists(MODELS_DIR):
   839→        return {"status": "ok", "models": {}}
   840→
   841→    for sym in sorted(os.listdir(MODELS_DIR)):
   842→        sym_path = os.path.join(MODELS_DIR, sym)
   843→        if not os.path.isdir(sym_path):
   844→            continue
   845→        models = []
   846→        for mn in MODEL_TYPES:
   847→            meta_path = os.path.join(sym_path, f"{mn}_meta.json")
   848→            if os.path.exists(meta_path):
   849→                meta = _load_meta(meta_path)
   850→                if meta:
   851→                    models.append(meta)
   852→        if models:
   853→            result[sym] = models
   854→
   855→    return {"status": "ok", "models": result}

   858→def get_symbol_models(symbol: str) -> Dict[str, Any]:
   859→    """Get models for a specific symbol."""
   860→    sym_path = os.path.join(MODELS_DIR, symbol)
   861→    if not os.path.isdir(sym_path):
   862→        return {"status": "error", "message": f"No trained model for symbol {symbol}"}
   863→
   864→    models = []
   865→    for mn in MODEL_TYPES:
   866→        meta_path = os.path.join(sym_path, f"{mn}_meta.json")
   867→        if os.path.exists(meta_path):
   868→            meta = _load_meta(meta_path)
   869→            if meta:
   870→                models.append(meta)
   871→
   872→    if not models:
   873→        return {"status": "error", "message": f"No trained model for symbol {symbol}"}
   874→
   875→    return {"status": "ok", "symbol": symbol, "models": models}

   883→def _json_response(handler: BaseHTTPRequestHandler, data: Dict, status: int = 200) -> None:
   884→    """Send a JSON response."""
   885→    body = json.dumps(data, ensure_ascii=False, default=str).encode("utf-8")
   886→    handler.send_response(status)
   887→    handler.send_header("Content-Type", "application/json; charset=utf-8")
   888→    handler.send_header("Content-Length", str(len(body)))
   889→    handler.end_headers()
   890→    handler.wfile.write(body)
   891→
   892→
   893→def _read_body(handler: BaseHTTPRequestHandler) -> bytes:
   894→    """Read request body."""
   895→    length = int(handler.headers.get("Content-Length", 0))
   896→    return handler.rfile.read(length) if length > 0 else b""

   899→class MLHandler(BaseHTTPRequestHandler):
   900→    """HTTP request handler for ML Training Service."""
   901→
   902→    # Suppress default stderr logging
   903→    def log_message(self, format: str, *args: Any) -> None:
   904→        pass

   906→    def _route(self, method: str) -> None:
   907→        """Route request to appropriate handler."""
   908→        parsed = urlparse(self.path)
   909→        path = parsed.path.rstrip("/")

   911→        try:
   912→            # ── Health ──

   915→                    "status": "ok",
   916→                    "service": "ml-trainer",
   917→                    "port": PORT,
   918→                    "version": "1.0.0",
   919→                    "onnx_export": ONNX_EXPORT_AVAILABLE,
   920→                    "cached_symbols": list(_model_cache.keys()),
   921→                })
   922→                return
   923→
   924→            # ── Train ──
   925→            if path == "/train" and method == "POST":
   926→                body = json.loads(_read_body(self))
   927→                symbol = body.get("symbol", "").strip().upper()
   928→                ohlcv = body.get("ohlcv", [])
   929→                models = body.get("models", MODEL_TYPES)
   930→
   931→                if not symbol:
   932→                    _json_response(self, {"status": "error", "message": "Missing 'symbol' field"}, 400)
   933→                    return
   934→                if not ohlcv or len(ohlcv) < MIN_CANDLES:
   935→                    _json_response(self, {"status": "error", "message": f"Need at least {MIN_CANDLES} OHLCV candles"}, 400)
   936→                    return
   937→
   938→                result = train_models(symbol, ohlcv, models)
   939→                _json_response(self, result)
   940→                return
   941→
   942→            # ── Predict ──
   943→            if path == "/predict" and method == "POST":
   944→                body = json.loads(_read_body(self))
   945→                symbol = body.get("symbol", "").strip().upper()
   946→                ohlcv = body.get("ohlcv", [])
   947→                horizon = min(body.get("horizon", 30), 90)  # cap at 90
   948→
   949→                if not symbol:
   950→                    _json_response(self, {"status": "error", "message": "Missing 'symbol' field"}, 400)
   951→                    return
   952→                if not ohlcv or len(ohlcv) < MIN_CANDLES:
   953→                    _json_response(self, {"status": "error", "message": f"Need at least {MIN_CANDLES} OHLCV candles"}, 400)
   954→                    return
   955→
   956→                result = predict_sessions(symbol, ohlcv, horizon)
   957→                _json_response(self, result)
   958→                return
   959→
   960→            # ── List all models ──
   961→            if path == "/models" and method == "GET":
   962→                _json_response(self, list_all_models())
   963→                return
   964→
   965→            # ── Get symbol models ──
   966→            if path.startswith("/models/") and method == "GET":
   967→                symbol = path[len("/models/"):].strip().upper()
   968→                if not symbol:
   969→                    _json_response(self, {"status": "error", "message": "Missing symbol in path"}, 400)
   970→                    return
   971→                _json_response(self, get_symbol_models(symbol))
   972→                return
   973→
   974→            # ── Retrain all ──
   975→            if path == "/retrain-all" and method == "POST":
   976→                body = json.loads(_read_body(self))
   977→                ohlcv_data = body.get("ohlcv_data", {})
   978→
   979→                if not ohlcv_data:

   981→                    return

   990→                    except Exception as e:
   991→                        errors.append({"symbol": sym, "error": str(e)})
   992→
   993→                _json_response(self, {
   994→                    "status": "ok",
   995→                    "retrained": list(results.keys()),
   996→                    "results": results,
   997→                    "errors": errors,
   998→                })
   999→                return
  1000→
  1001→            # ── 404 ──
  1002→            _json_response(self, {"status": "error", "message": f"Not found: {method} {path}"}, 404)
  1003→
  1004→        except json.JSONDecodeError:
  1005→            _json_response(self, {"status": "error", "message": "Invalid JSON body"}, 400)
  1006→        except ValueError as e:
  1007→            _json_response(self, {"status": "error", "message": str(e)}, 400)
  1008→        except Exception as e:
  1009→            log("ERROR", f"Unhandled {method} {path}: {traceback.format_exc()}")
