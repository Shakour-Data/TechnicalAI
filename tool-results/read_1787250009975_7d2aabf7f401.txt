    26→    26→// --- Global rate limiter (serial queue) ---

    28→    28→const AI_MIN_INTERVAL = 20_000;
    29→    29→let cooldownUntil = 0;
    30→    30→let activeCall = false;
    31→    31→const pendingQueue: Array<{
    32→    32→  resolve: (text: string) => void;

    37→    37→function processAIQueue() {
    38→    38→  if (activeCall || pendingQueue.length === 0) return;
    39→    39→  const now = Date.now();
    40→    40→
    41→    41→  if (cooldownUntil > now) {
    42→    42→    setTimeout(processAIQueue, Math.min(cooldownUntil - now, 5000));
    43→    43→    return;
    44→    44→  }
    45→    45→
    46→    46→  const elapsed = now - lastAITime;
    47→    47→  if (elapsed < AI_MIN_INTERVAL) {
    48→    48→    setTimeout(processAIQueue, AI_MIN_INTERVAL - elapsed);
    49→    49→    return;
    50→    50→  }
    51→    51→
    52→    52→  const item = pendingQueue.shift()!;
    53→    53→  activeCall = true;
    54→    54→
    55→    55→  callZaiSDK(item.messages)
    56→    56→    .then(text => {
    57→    57→      lastAITime = Date.now();
    58→    58→      cooldownUntil = 0;
    59→    59→      item.resolve(text);
    60→    60→    })
    61→    61→    .catch(err => {
    62→    62→      const msg = err.message || '';
    63→    63→      if (msg.includes('429')) {
    64→    64→        const cd = Math.max(cooldownUntil - Date.now(), 0) + 90_000;
    65→    65→        cooldownUntil = Date.now() + Math.min(cd, 300_000);
    66→    66→        console.warn(`[AI v5.1] 429, cooldown ${Math.min(cd, 300_000) / 1000}s`);
    67→    67→      }
    68→    68→      item.reject(err);
    69→    69→    })
    70→    70→    .finally(() => {
    71→    71→      activeCall = false;
    72→    72→      setTimeout(processAIQueue, 2000);
    73→    73→    });
    74→    74→}

    76→    76→function queueAI(messages: { role: string; content: string }[], timeoutMs = 90_000): Promise<string> {
    77→    77→  return new Promise((resolve, reject) => {
    78→    78→    const timer = setTimeout(() => {
    79→    79→      const idx = pendingQueue.findIndex(q => q.messages === messages);
    80→    80→      if (idx >= 0) pendingQueue.splice(idx, 1);
    81→    81→      reject(new Error('AI request timeout'));
    82→    82→    }, timeoutMs);
    83→    83→
    84→    84→    pendingQueue.push({
    85→    85→      messages,
    86→    86→      resolve: (text) => { clearTimeout(timer); resolve(text); },
    87→    87→      reject: (err) => { clearTimeout(timer); reject(err); },
    88→    88→    });
    89→    89→
    90→    90→    processAIQueue();
    91→    91→  });
    92→    92→}

    95→    95→function sleep(ms: number) { return new Promise(r => setTimeout(r, ms)); }
    96→    96→function jitter(base: number): number { return base + Math.random() * base * 0.5; }

    98→    98→// Common emoji ranges to strip from AI output
    99→    99→const EMOJI_PATTERNS = [
   100→   100→  /\p{Emoji_Presentation}/gu,
   101→   101→  /\p{Extended_Pictographic}/gu,
   102→   102→];

   104→   104→function stripEmojis(text: string): string {
   105→   105→  let result = text;
   106→   106→  for (const pattern of EMOJI_PATTERNS) {
   107→   107→    result = result.replace(pattern, '').trim();
   108→   108→  }
   109→   109→  return result;
   110→   110→}

   112→   112→async function callZaiSDK(messages: { role: string; content: string }[], maxRetries = 4): Promise<string> {
   113→   113→  for (let attempt = 1; attempt <= maxRetries; attempt++) {
   114→   114→    try {
   115→   115→      const zai = await getZai();
   116→   116→      const completion = await zai.chat.completions.create({
   117→   117→        messages,
   118→   118→        thinking: { type: 'disabled' },
   119→   119→      });
   120→   120→      const raw = completion.choices[0]?.message?.content;
   121→   121→      if (!raw || raw.trim().length === 0) {
   122→   122→        throw new Error('Empty AI response');
   123→   123→      }
   124→   124→      const text = stripEmojis(raw).trim();
   125→   125→      if (text.length < 10) {
   126→   126→        throw new Error('AI response too short: ' + text.slice(0, 100));
   127→   127→      }
   128→   128→      return text;
   129→   129→    } catch (err: unknown) {
   130→   130→      const msg = err instanceof Error ? err.message : String(err);
   131→   131→      if (msg.includes('429') && attempt < maxRetries) {
   132→   132→        const waitMs = Math.round(jitter(20_000 * Math.pow(2, attempt - 1))); // 20s, 40s, 80s, 160s + jitter
   133→   133→        console.warn(`[AI v5.1] 429 retry ${attempt}/${maxRetries}, waiting ${Math.round(waitMs / 1000)}s...`);
   134→   134→        await sleep(waitMs);
   135→   135→        continue;
   136→   136→      }
   137→   137→      throw err;
   138→   138→    }
   139→   139→  }
   140→   140→  throw new Error('Max retries exceeded');
   141→   141→}

   149→   149→function srGrade(strength: number): string {
   150→   150→  if (strength >= 8.5) return '\u0628\u0633\u06cc\u0627\u0631 \u0642\u0648\u06cc';
   151→   151→  if (strength >= 7) return '\u0642\u0648\u06cc';
   152→   152→  if (strength >= 5) return '\u0645\u062a\u0648\u0633\u0637';
   153→   153→  if (strength >= 3) return '\u0636\u0639\u06cc\u0641';
   154→   154→  return '\u0628\u0633\u06cc\u0627\u0631 \u0636\u0639\u06cc\u0641';
