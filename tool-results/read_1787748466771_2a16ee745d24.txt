    31→function renderAIText(text: string): string {
    32→  // Normalize zero-width and look-alike characters that may interfere with regex
    33→  const normalized = text.replace(/[\u200c\u200d\u200b\ufeff]/g, '');
    34→
    35→  // Color syntax: {color:hex}text{/color} or {color:named}text{/color} → <span>
    36→  const colorMap: Record<string, string> = {
    37→    'red': '#dc2626', 'red-600': '#dc2626', 'red-700': '#b91c1c',
    38→    'green': '#16a34a', 'emerald': '#059669', 'emerald-600': '#059669', 'emerald-700': '#047857',
    39→    'amber': '#d97706', 'amber-600': '#d97706', 'amber-700': '#b45309', 'amber-800': '#92400e',
    40→    'blue': '#2563eb', 'blue-600': '#2563eb', 'blue-700': '#1d4ed8',
    41→    'orange': '#ea580c',
    42→    'purple': '#9333ea', 'purple-600': '#9333ea', 'purple-700': '#7e22ce',
    43→  };
    44→
    45→  const withColors = normalized.replace(/\{color:([^}]+)\}([\s\S]*?)\{\/color\}/g, (_match, colorName: string, inner: string) => {
    46→    const hex = colorMap[colorName.trim()] || (colorName.startsWith('#') ? colorName : null);
    47→    if (!hex) return inner; // Unknown color: just return text without color
    48→    return `<span style="color:${hex}">${inner}</span>`;
    49→  });
    50→
    51→  return withColors
    52→    .split('\n\n')
    53→    .map(p => {
    54→      // Convert **bold** to <strong>
    55→      const html = p.replace(/\*\*(.+?)\*\*/g, '<strong class="font-bold text-gray-900">$1</strong>');
    56→      // Convert newlines within paragraph to <br>
    57→      return `<p class="mb-4">${html.replace(/\n/g, '<br>')}</p>`;
    58→    })
    59→    .join('');
    60→}
