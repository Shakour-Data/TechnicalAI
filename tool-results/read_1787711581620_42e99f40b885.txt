   254→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number, yahooSymbol?: string) => {
   255→    // Abort any previous in-flight fetch
   256→    if (fetchControllerRef.current) {
   257→      fetchControllerRef.current.abort();
   258→    }
   259→    const controller = new AbortController();
   260→    fetchControllerRef.current = controller;

   270→        const res = await fetch('/api/yahoo-analysis?symbol=' + encodeURIComponent(yahooSymbol), { signal: controller.signal });
   271→        if (controller.signal.aborted) return;
   272→        if (!res.ok) {
   273→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   274→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   275→          throw new Error(errMsg);
   276→        }
   277→        const json = await res.json();
   278→        if (controller.signal.aborted) return;
   279→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   280→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   281→        }
   282→        setData({ ...json, isYahoo: true });
   283→        setActivePanel('visual');
   284→      } catch (err) {
   285→        if (controller.signal.aborted) return;
   286→        setError(String(err));
   287→      } finally {
   288→        if (!controller.signal.aborted) {
   289→          setLoading(false);
   290→          setLoadingMessage(null);
   291→        }
   292→      }
   293→      return;
   294→    }
   295→
   296→    // TGJU instrument: fetch historical data via tgju.org chart API
   297→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   298→      setLoading(true);
   299→      setLoadingMessage('در حال دریافت داده‌های تاریخی ... (حدود ۱۵ ثانیه)');
   300→      setError(null);
   301→      setData(null);
   302→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId, yahooSymbol };
   303→      try {
   304→        const res = await fetch('/api/tgju-analysis?key=' + encodeURIComponent(tgjuKey), { signal: controller.signal });
   305→        if (controller.signal.aborted) return;
   306→        if (!res.ok) {
   307→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   308→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   309→          throw new Error(errMsg);
   310→        }
   311→        const json = await res.json();
   312→        if (controller.signal.aborted) return;
   313→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   314→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   315→        }
   316→        setData({ ...json, isTgju: true });
   317→        setActivePanel('visual');
   318→      } catch (err) {
   319→        if (controller.signal.aborted) return;
   320→        setError(String(err));
   321→      } finally {
   322→        if (!controller.signal.aborted) {
   323→          setLoading(false);
   324→          setLoadingMessage(null);
