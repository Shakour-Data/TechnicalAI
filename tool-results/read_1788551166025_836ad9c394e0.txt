    21→    21→function toPersianNum(n: number): string {
    22→    22→  if (!isFinite(n) || isNaN(n)) return '\u06f0';
    23→    23→  return Math.round(n).toLocaleString('fa-IR');

    25→    25→function srGrade(strength: number): string {
    26→    26→  if (strength >= 8.5) return '\u0628\u0633\u06cc\u0627\u0631 \u0642\u0648\u06cc';
    27→    27→  if (strength >= 7) return '\u0642\u0648\u06cc';
    28→    28→  if (strength >= 5) return '\u0645\u062a\u0648\u0633\u0637';
    29→    29→  if (strength >= 3) return '\u0636\u0639\u06cc\u0641';
    30→    30→  return '\u0628\u0633\u06cc\u0627\u0631 \u0636\u0639\u06cc\u0641';
    31→    31→}

    41→    41→function buildMLInput(body: Record<string, unknown>): MLSelectorInput {
    42→    42→  const price = (body.currentPrice as number) || 0;
    43→    43→  const trendDir = (body.trendDirection as string) || 'range';
    44→    44→  const adx = (body.adx as number) || 0;
    45→    45→  const rsi = (body.rsi as number) || 50;
    46→    46→  const stochK = (body.stochK as number) || 50;
    47→    47→  const macdHist = (body.macdHist as number) || 0;
    48→    48→  const obv = (body.obv as number) || 0;
    49→    49→  const bollingerUpper = (body.bollingerUpper as number) || 0;
    50→    50→  const bollingerLower = (body.bollingerLower as number) || 0;
    51→    51→  const resistance = (body.resistanceStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
    52→    52→  const support = (body.supportStrengths as Array<{ price: number }> | undefined)?.[0]?.price ?? 0;
    53→    53→  const atr = (body.atr as number) || 0;
    54→    54→  const bbRange = bollingerUpper - bollingerLower;
    55→    55→  const bbPosition = bbRange > 0 ? Math.min(100, Math.max(0, Math.round((price - bollingerLower) / bbRange * 100))) : 50;
    56→    56→
    57→    57→  // Find dominant scenario across ALL SC1-SC9
    58→    58→  const scenarios = body.scenarios as Record<string, { probability: number }> | undefined;
    59→    59→  let dominantKey = 'SC5';
    60→    60→  let dominantProb = 0;
    61→    61→  for (const k of ['SC1', 'SC2', 'SC3', 'SC4', 'SC5', 'SC6', 'SC7', 'SC8', 'SC9'] as const) {
    62→    62→    const p = scenarios?.[k]?.probability ?? 0;
    63→    63→    if (p > dominantProb) { dominantProb = p; dominantKey = k; }
    64→    64→  }
    65→    65→
    66→    66→  return {
    67→    67→    price,
    68→    68→    trend: trendDir === 'up' ? 'up' : trendDir === 'down' ? 'down' : 'range',
    69→    69→    adx, diPlus: (body.diPlus as number) || 0, diMinus: (body.diMinus as number) || 0,
    70→    70→    rsi, stochK, macdHist, obv, bbPosition, resistance, support, atr,
    71→    71→    scenarioDominant: dominantKey,
    72→    72→    hasVolume: (body.hasVolume as boolean) ?? false,
    73→    73→  };
    74→    74→}

    77→    77→function buildMSLV4Context(body: Record<string, unknown>): MSLV4Context {
    78→    78→  const price = (body.currentPrice as number) || 0;
    79→    79→  const atr = (body.atr as number) || 0;
    80→    80→  const atrRatio = price > 0 ? atr / price : 0;
    81→    81→
    82→    82→  const v11Probs = body.v11Probabilities as {
    83→    83→    scenarios: Array<{ key: string; name: string; rawProbability: number }>;
    84→    84→  } | undefined;
    85→    85→  let dominantScenario = '';
    86→    86→  if (v11Probs?.scenarios) {
    87→    87→    const sorted = [...v11Probs.scenarios].sort((a, b) => b.rawProbability - a.rawProbability);
    88→    88→    dominantScenario = sorted[0]?.name ?? '';
    89→    89→  } else {
    90→    90→    const scenarios = body.scenarios as Record<string, { name?: string; probability: number }> | undefined;
    91→    91→    let maxProb = 0;
    92→    92→    for (const k of ['SC1', 'SC2', 'SC3', 'SC4', 'SC5', 'SC6', 'SC7', 'SC8', 'SC9']) {
    93→    93→      const p = scenarios?.[k]?.probability ?? 0;
    94→    94→      if (p > maxProb) { maxProb = p; dominantScenario = scenarios?.[k]?.name ?? k; }
    95→    95→    }
    96→    96→  }
    97→    97→
    98→    98→  const trendDir = (body.trendDirection as string) || 'range';
    99→    99→  const isBull = trendDir === 'up';
   100→   100→  const isBear = trendDir === 'down';
   101→   101→  const adx = (body.adx as number) || 0;
   102→   102→  const trendStrength = adx > 40 ? 0.9 : adx > 25 ? 0.7 : adx > 15 ? 0.4 : 0.15;
   103→   103→
   104→   104→  let regime: MSLV4Context['regime'] = 'Range';
   105→   105→  if (isBull && trendStrength > 0.6) regime = 'Strong Bull';
   106→   106→  else if (isBull) regime = 'Weak Bull';
   107→   107→  else if (isBear && trendStrength > 0.6) regime = 'Strong Bear';
   108→   108→  else if (isBear) regime = 'Weak Bear';
   109→   109→
   110→   110→  const instrumentType = (body.instrumentType as string) || (body.symbolName as string) || 'stock';
   111→   111→  const ta = body.technicalAnalysis as {
   112→   112→    classicPatternCount?: number; harmonicPatternCount?: number;
   113→   113→    elliottWaveCount?: number; divergenceCount?: number;
   114→   114→  } | undefined;
   115→   115→  const overallConfidence = (body.overallConfidence as number) ?? (body.confidence as number) ?? 0.6;
   116→   116→
   117→   117→  // Extract probability trend signals for MSL selection
   118→   118→  const probTrend = extractProbTrendSignals(body);
   119→   119→
   120→   120→  return {
   121→   121→    asset: instrumentType, timeframe: (body.timeframe as string) || 'daily',
   122→   122→    regime, confidence: Math.max(0, Math.min(1, overallConfidence)),
   123→   123→    volatility: atrRatio, trend_strength: trendStrength, dominantScenario,
   124→   124→    pattern_counts: { classical: ta?.classicPatternCount ?? 0, harmonic: ta?.harmonicPatternCount ?? 0, elliott: ta?.elliottWaveCount ?? 0 },
   125→   125→    divergence_present: (ta?.divergenceCount ?? 0) > 0,
   126→   126→    hasVolume: (body.hasVolume as boolean) ?? false,
   127→   127→    rsi: (body.rsi as number) || 50, adx,
   128→   128→    audience_level: (body.audienceLevel as string) || 'pro',
   129→   129→    probTrend,
   130→   130→  };
   131→   131→}

   133→   133→// ─── Extract probability trend signals from body ──────────────────
   134→   134→function extractProbTrendSignals(body: Record<string, unknown>): MSLV4Context['probTrend'] {
   135→   135→  const pt = body.probabilityTrend as {
   136→   136→    scenarios?: { scenarioKey: string; group: string; trendDirection: string; trend: { individualProb: number; cumulativeProb: number }[] }[];
   137→   137→    groups?: { group: string; trendDirection: string; trend: { cumulativeProb: number }[] }[];
   138→   138→  } | undefined;
   139→   139→  if (!pt?.groups || pt.groups.length === 0) return undefined;
   140→   140→
   141→   141→  const findGroup = (g: string) => pt.groups!.find(x => x.group === g);
   142→   142→  const bullG = findGroup('bullish');
   143→   143→  const bearG = findGroup('bearish');
   144→   144→  const neutralG = findGroup('neutral');
   145→   145→
   146→   146→  // Find dominant scenario trend
   147→   147→  const scenarios = body.scenarios as Record<string, { probability: number }> | undefined;
   148→   148→  let domKey = 'SC5'; let domProb = 0;
   149→   149→  for (const k of ['SC1','SC2','SC3','SC4','SC5','SC6','SC7','SC8','SC9']) {
   150→   150→    const p = scenarios?.[k]?.probability ?? 0;
   151→   151→    if (p > domProb) { domProb = p; domKey = k; }
   152→   152→  }
   153→   153→  const domScenario = pt.scenarios?.find(s => s.scenarioKey === domKey);
   154→   154→
   155→   155→  // 7-day change (day 1 = today, day 7 = 7 days ago)
   156→   156→  const get7dChange = (trend: { cumulativeProb: number }[]) => {
   157→   157→    if (!trend || trend.length < 7) return 0;
   158→   158→    return trend[0].cumulativeProb - trend[6].cumulativeProb;
   159→   159→  };
   160→   160→
   161→   161→  return {
   162→   162→    bullGroupDir: bullG?.trendDirection as MSLV4Context['probTrend']['bullGroupDir'],
   163→   163→    bearGroupDir: bearG?.trendDirection as MSLV4Context['probTrend']['bearGroupDir'],
   164→   164→    neutralGroupDir: neutralG?.trendDirection as MSLV4Context['probTrend']['neutralGroupDir'],
   165→   165→    bullGroupCum: bullG?.trend?.[0]?.cumulativeProb,
   166→   166→    bearGroupCum: bearG?.trend?.[0]?.cumulativeProb,
   167→   167→    neutralGroupCum: neutralG?.trend?.[0]?.cumulativeProb,
   168→   168→    dominantCumDir: domScenario?.trendDirection as MSLV4Context['probTrend']['dominantCumDir'],
   169→   169→    dominantIndivDir: domScenario?.trendDirection as MSLV4Context['probTrend']['dominantIndivDir'],
   170→   170→    bullGroupChange7d: get7dChange(bullG?.trend ?? []),
   171→   171→    bearGroupChange7d: get7dChange(bearG?.trend ?? []),
   172→   172→    neutralGroupChange7d: get7dChange(neutralG?.trend ?? []),
   173→   173→  };
   174→   174→}

   417→   417→// ─── POST Handler ────────────────────────────────────────────────
   418→   418→export async function POST(req: NextRequest) {
   419→   419→  const startTime = Date.now();
   420→   420→  try {
   421→   421→    const body = await req.json();
   422→   422→    if (!body.currentPrice) {
   423→   423→      return NextResponse.json({ error: 'currentPrice is required' }, { status: 400 });
   424→   424→    }
   425→   425→
   426→   426→    const symbolName = String(body.symbolName || 'unknown');
   427→   427→    const today = getTodayDateStr();
   428→   428→
   429→   429→    // 1. Check daily persistent cache (Prisma/SQLite)
   430→   430→    // Also validate price match — if cached price is wildly different, re-generate
   431→   431→    try {
   432→   432→      const currentPrice = Number(body.currentPrice) || 0;
   433→   433→      const cached = await db.aiAnalysisCache.findUnique({
   434→   434→        where: { symbol_date: { symbol: symbolName, date: today } },
   435→   435→      });
   436→   436→      if (cached && cached.text) {
   437→   437→        // Price sanity check: if cached price differs by >5%, the cached analysis
   438→   438→        // was likely generated with wrong/stale data (e.g. 1392-era chart data)
   439→   439→        const cachedPrice = Number(cached.price) || 0;
   440→   440→        if (cachedPrice > 0 && currentPrice > 0) {
   441→   441→          const priceDiff = Math.abs(cachedPrice - currentPrice) / currentPrice;
   442→   442→          if (priceDiff > 0.05) {
   443→   443→            console.log(`[AI] Daily cache INVALIDATED for ${symbolName} (cached price diff ${(priceDiff * 100).toFixed(1)}% > 5%), re-generating`);
   444→   444→          } else {
   445→   445→            console.log(`[AI] Daily cache HIT for ${symbolName} (${today}) [${Date.now() - startTime}ms]`);
   446→   446→            return NextResponse.json({
   447→   447→              text: cached.text,
   448→   448→              ml: cached.ml ? JSON.parse(cached.ml) : undefined,
   449→   449→              cached: true,
   450→   450→              cachedDate: cached.date,
   451→   451→            });
   452→   452→          }
   453→   453→        } else {
   454→   454→          console.log(`[AI] Daily cache HIT for ${symbolName} (${today}) [${Date.now() - startTime}ms]`);
   455→   455→          return NextResponse.json({
   456→   456→            text: cached.text,
   457→   457→            ml: cached.ml ? JSON.parse(cached.ml) : undefined,
   458→   458→            cached: true,
   459→   459→            cachedDate: cached.date,
   460→   460→          });
   461→   461→        }
   462→   462→      }
   463→   463→    } catch (dbErr) {
   464→   464→      console.warn('[AI] DB cache read failed, continuing:', dbErr instanceof Error ? dbErr.message : dbErr);
   465→   465→    }
   466→   466→
   467→   467→    // 1b. Check for previous day's cache (fallback only if price is within 2%)
   468→   468→    try {
   469→   469→      const currentPrice = Number(body.currentPrice) || 0;
   470→   470→      const prevCached = await db.aiAnalysisCache.findFirst({
   471→   471→        where: { symbol: symbolName, date: { not: today } },
   472→   472→        orderBy: { createdAt: 'desc' },
   473→   473→      });
   474→   474→      if (prevCached && prevCached.text && prevCached.price > 0 && currentPrice > 0) {
   475→   475→        const priceDiff = Math.abs(prevCached.price - currentPrice) / currentPrice;
   476→   476→        if (priceDiff < 0.02) { // Less than 2% price difference
   477→   477→          console.log(`[AI] Previous cache fallback HIT for ${symbolName} (${prevCached.date}, price diff ${((priceDiff)*100).toFixed(1)}%) [${Date.now() - startTime}ms]`);
   478→   478→          return NextResponse.json({
   479→   479→            text: prevCached.text,
   480→   480→            ml: prevCached.ml ? JSON.parse(prevCached.ml) : undefined,
   481→   481→            cached: true,
   482→   482→            cachedDate: prevCached.date,
   483→   483→            isFallback: true,
   484→   484→          });
   485→   485→        } else {
   486→   486→          console.log(`[AI] Previous cache SKIP for ${symbolName} (price diff ${((priceDiff)*100).toFixed(1)}% > 2%)`);
   487→   487→        }
   488→   488→      }
   489→   489→    } catch (dbErr) {
   490→   490→      console.warn('[AI] Previous cache fallback read failed, continuing:', dbErr instanceof Error ? dbErr.message : dbErr);
   491→   491→    }
   492→   492→
   493→   493→    // 2. ML selection
   494→   494→    const mlInput = buildMLInput(body);
   495→   495→    const mlSelection = selectMLCombination(mlInput);
   496→   496→    const methods = selectMethods(mlInput);
   497→   497→

   499→   499→    const mslCtx = buildMSLV4Context(body);
   500→   500→    const mslResult = selectMSLV4(mslCtx);
   501→   501→    mlSelection.school = mslResult.school_of_analysis.primary.id as typeof mlSelection.school;
   502→   502→    mlSelection.style = mslResult.analysis_style.primary.id as typeof mlSelection.style;
   503→   503→    mlSelection.tone = mslResult.analysis_tone.primary.id as typeof mlSelection.tone;
   504→   504→    mlSelection.reasoning = `MSLv4: ${mslResult.school_of_analysis.primary.id}/${mslResult.analysis_style.primary.id}/${mslResult.analysis_tone.primary.id}`;
   505→   505→
   506→   506→    // 4. Build prompt
   507→   507→    const mslSystemPrompt = buildMSLV4PromptSection(mslResult);
   508→   508→    const userMessage = buildPrompt(body, mlSelection, methods);
   509→   509→    const dynamicSystemPrompt = `${SYSTEM_PROMPT}\n\n${mslSystemPrompt}`;
   510→   510→
   511→   511→    console.log(`[AI] Generating for ${symbolName} (cache miss) [${Date.now() - startTime}ms]`);
   512→   512→
   513→   513→    // 5. Call AI through dedicated channel — fail fast on 429
   514→   514→    //    Only retry 2 times max (3 total attempts) to avoid wasting time.
   515→   515→    //    If rate-limited, the previous-day fallback above already returned.
   516→   516→    const content = await dedicatedAIChatCompletion(
   517→   517→      [

   519→   519→        { role: 'user', content: userMessage },

   521→   521→      {
   522→   522→        timeoutMs: 90_000,  // 90s max (1 initial + 2 retries)
   523→   523→        maxRetries: 2,
   524→   524→      }

   526→   526→
   527→   527→    // Post-process: validate prices, fix Persian text, strip technical codes
   528→   528→    const priceRefs = buildPriceReferences(body);

   530→   530→
   531→   531→    // If prices are severely hallucinated, log but still return (don't crash)
   532→   532→    if (!priceValid) {
   533→   533→      console.warn(`[AI] Price hallucination detected for ${symbolName}: ${hallucinationCount} bad prices found. NOT caching.`);
   534→   534→    }
   535→   535→    const mlData = { school: mlSelection.school, style: mlSelection.style, tone: mlSelection.tone, reasoning: mlSelection.reasoning, methods };
   536→   536→
   537→   537→    // 7. Save to daily persistent cache — ONLY if prices are valid
   538→   538→    if (priceValid) {
   539→   539→      try {
   540→   540→        await db.aiAnalysisCache.upsert({
   541→   541→          where: { symbol_date: { symbol: symbolName, date: today } },
   542→   542→          create: {
   543→   543→            symbol: symbolName,
   544→   544→            date: today,
   545→   545→            text: cleaned,
   546→   546→            ml: JSON.stringify(mlData),
   547→   547→            price: Number(body.currentPrice) || 0,
   548→   548→          },
   549→   549→          update: {
   550→   550→            text: cleaned,
   551→   551→            ml: JSON.stringify(mlData),
   552→   552→            price: Number(body.currentPrice) || 0,
   553→   553→          },
   554→   554→        });
   555→   555→        console.log(`[AI] Saved to daily cache: ${symbolName} (${today}) [${Date.now() - startTime}ms]`);
   556→   556→      } catch (dbErr) {
   557→   557→        console.warn('[AI] DB cache write failed:', dbErr instanceof Error ? dbErr.message : dbErr);
   558→   558→      }
   559→   559→    } else {
   560→   560→      console.log(`[AI] Skipped cache save for ${symbolName} due to price hallucination`);
   561→   561→    }
   562→   562→
   563→   563→    console.log(`[AI] Complete for ${symbolName} [${Date.now() - startTime}ms]`);
   564→   564→    return NextResponse.json({ text: cleaned, ml: mlData });
   565→   565→
   566→   566→  } catch (err) {
   567→   567→    console.error(`[AI] Error [${Date.now() - startTime}ms]:`, err);
   568→   568→    return NextResponse.json({ error: userFriendlyError(err), text: '' }, { status: 500 });
   569→   569→  }
   570→   570→}

   572→   572→function userFriendlyError(err: unknown): string {
   573→   573→  const msg = err instanceof Error ? err.message : String(err);
   574→   574→  if (msg.includes('Rate limited') || msg.includes('429')) {
   575→   575→    return 'سرور هوشمند محدودیت سرعت دارد.';
   576→   576→  }
   577→   577→  if (msg.includes('Timed out') || msg.includes('timeout') || msg.includes('Timeout')) {
   578→   578→    return 'زمان پاسخدهی هوشمند به پایان رسید.';
   579→   579→  }
   580→   580→  if (msg.includes('concurrent')) {
   581→   581→    return 'درخواست تحلیل قبلی هنوز در حال اجراست.';
   582→   582→  }
   583→   583→  return 'خطایی در تولید تحلیل رخ داد.';
   584→   584→}
