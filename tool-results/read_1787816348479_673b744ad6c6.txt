   269→/* Advanced query parser — supports operators like country:US, exchange:NASD, sector:Tech */
   270→interface ParsedQuery {
   271→  country?: string;
   272→  exchange?: string;
   273→  sector?: string;
   274→  terms: string[];
   275→}
   276→
   277→function parseQuery(q: string): ParsedQuery {
   278→  const country: string[] = [];
   279→  const exchange: string[] = [];
   280→  const sector: string[] = [];
   281→  const terms: string[] = [];
   282→
   283→  for (const part of q.split(/\s+/)) {
   284→    if (!part) continue;
   285→    const lower = part.toLowerCase();
   286→    if (lower.startsWith('country:') || lower.startsWith('کشور:')) {
   287→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   288→      if (val) country.push(val.toLowerCase());
   289→    } else if (lower.startsWith('exchange:') || lower.startsWith('صرافی:')) {
   290→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   291→      if (val) exchange.push(val.toLowerCase());
   292→    } else if (lower.startsWith('sector:') || lower.startsWith('صنعت:')) {
   293→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   294→      if (val) sector.push(val.toLowerCase());
   295→    } else {
   296→      terms.push(lower);
   297→    }
   298→  }
   299→
   300→  return {
   301→    country: country.length > 0 ? country.join(' ') : undefined,
   302→    exchange: exchange.length > 0 ? exchange.join(' ') : undefined,
   303→    sector: sector.length > 0 ? sector.join(' ') : undefined,
   304→    terms,
   305→  };
   306→}

   312→const RECENT_KEY = 'fin-search-recent';
   313→const MAX_RECENT = 8;
   314→
   315→function getRecentSearches(): string[] {
   316→  if (typeof window === 'undefined') return [];
   317→  try { const raw = localStorage.getItem(RECENT_KEY); return raw ? JSON.parse(raw) : []; }
   318→  catch { return []; }
   319→}
   320→
   321→function addRecentSearch(name: string) {
   322→  if (typeof window === 'undefined' || !name) return;
   323→  try {
   324→    let list = getRecentSearches();
   325→    list = list.filter((r) => r !== name);
   326→    list.unshift(name);
   327→    if (list.length > MAX_RECENT) list = list.slice(0, MAX_RECENT);
   328→    localStorage.setItem(RECENT_KEY, JSON.stringify(list));
   329→  } catch { /* ignore */ }
   330→}

   536→      }

   539→          item.cs?.toLowerCase().includes(opSector);
   540→        if (!match) return null;
   541→      }
   542→
   543→      const fields = [item.l18, item.l30, item.cs, item.yahooSymbol || '', item.nameEn || '', item.exchange || '', item.country || '', item.countryEn || '', item.sector || ''];
   544→      const fullText = fields.join(' ').toLowerCase();
   545→      let score = 0;
   546→
   547→      for (const term of terms) {
   548→        if (!term) continue;
   549→        if (item.l18.toLowerCase().includes(term)) score += 10;
   550→        if (item.nameEn?.toLowerCase().includes(term)) score += 8;
   551→        if (item.yahooSymbol?.toLowerCase() === term) score += 20;
   552→        else if (item.yahooSymbol?.toLowerCase().includes(term)) score += 15;
   553→        if (item.l30.toLowerCase().includes(term)) score += 7;
   554→        if (item.exchange?.toLowerCase().includes(term)) score += 5;
   555→        if (item.country?.toLowerCase().includes(term)) score += 6;
   556→        if (item.countryEn?.toLowerCase() === term) score += 15;
   557→        else if (item.countryEn?.toLowerCase().includes(term)) score += 4;
   558→        if (item.sector?.toLowerCase().includes(term)) score += 5;
   559→        if (item.cs?.toLowerCase().includes(term)) score += 6;
   560→
   561→        if (!item.l18.toLowerCase().includes(term) && !item.l30.toLowerCase().includes(term) &&
   562→            !item.cs?.toLowerCase().includes(term) && !(item.yahooSymbol?.toLowerCase().includes(term)) &&
   563→            !(item.nameEn?.toLowerCase().includes(term)) && !(item.exchange?.toLowerCase().includes(term)) &&
   564→            !(item.country?.toLowerCase().includes(term)) && !(item.countryEn?.toLowerCase().includes(term)) &&
   565→            !(item.sector?.toLowerCase().includes(term))) {
   566→          return null;
   567→        }
   568→      }
   569→      /* If only operators (no terms), accept all filtered items */
   570→      if (terms.length === 0) score = 1;
   571→      return score;
   572→    };
   573→
   574→    const scored = items.map((item) => ({ item, score: scoreAndFilter(item) }))
   575→      .filter((r): r is { item: InstrumentItem; score: number } => r.score !== null && r.score > 0)
   576→      .sort((a, b) => b.score - a.score);

   578→    setTotalMatched(scored.length);
