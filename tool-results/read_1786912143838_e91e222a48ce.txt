   123→
   124→export function CandlestickChartSkeleton() {
   125→  return (
   126→    <div className="w-full">

   129→    </div>
   130→  );

   132→
   133→/* ────────────────────────────────────────────────────────── COMPONENT ───── */
   134→
   135→export default function CandlestickChart({ data, ta, height = 520 }: CandlestickChartProps) {
   136→  const containerRef = useRef<HTMLDivElement>(null);
   137→  const chartContainerRef = useRef<HTMLDivElement>(null);
   138→  const svgRef = useRef<SVGSVGElement>(null);
   139→  const chartRef = useRef<IChartApi | null>(null);
   140→  const csRef = useRef<ReturnType<typeof chartRef.current.addSeries> | null>(null);
   141→
   142→  const [drawings, setDrawings] = useState<Drawing[]>([]);
   143→  const [activeTool, setActiveTool] = useState<ToolType>('cursor');
   144→  const [activeColor, setActiveColor] = useState(PALETTE[0]);
   145→  const [isDrawing, setIsDrawing] = useState(false);
   146→  const [currentDrawing, setCurrentDrawing] = useState<Drawing | null>(null);
   147→  const [textInput, setTextInput] = useState<{ x: number; y: number; drawingId: string } | null>(null);
   148→  const [textValue, setTextValue] = useState('');
   149→  const [renderKey, setRenderKey] = useState(0);
   150→  const [chartSize, setChartSize] = useState({ w: 0, h: 0 });
   151→
   152→  /* ── Helper: coordinate conversion ─────────────────────────── */
   153→  const getChartPoint = useCallback((clientX: number, clientY: number): DrawingPoint | null => {
   154→    const chart = chartRef.current;
   155→    const container = chartContainerRef.current;
   156→    if (!chart || !container) return null;
   157→    const rect = container.getBoundingClientRect();
   158→    const x = clientX - rect.left;
   159→    const y = clientY - rect.top;
   160→    const price = chart.coordinateToPrice(y);
   161→    const time = chart.timeScale().coordinateToTime(x);
   162→    if (price === null || time === null) return null;
   163→    return { time: time as number, price };
   164→  }, []);
   165→
   166→  /* ── SVG Rendering ────────────────────────────────────────── */
   167→  const renderDrawings = useCallback(() => {
   168→    const chart = chartRef.current;
   169→    const svg = svgRef.current;
   170→    const container = chartContainerRef.current;
   171→    if (!chart || !svg || !container || drawings.length === 0) {
   172→      if (svg) svg.innerHTML = '';
   173→      return;
   174→    }
   175→
   176→    const rect = container.getBoundingClientRect();
   177→    const w = rect.width;
   178→    const h = rect.height;
   179→    if (w === 0 || h === 0) return;
   180→
   181→    let html = '';
   182→
   183→    // Defs for arrow markers
   184→    html += `<defs>`;
   185→    PALETTE.forEach((c, i) => {
   186→      html += `<marker id="arrow-${c.replace('#','')}" markerWidth="8" markerHeight="6" refX="7" refY="3" orient="auto"><polygon points="0 0, 8 3, 0 6" fill="${c}" /></marker>`;
   187→    });
   188→    html += `</defs>`;
   189→
   190→    const allDrawings = [...drawings];
   191→    if (currentDrawing) allDrawings.push(currentDrawing);
   192→
   193→    for (const d of allDrawings) {
   194→      if (!d.completed && d.points.length === 0) continue;
   195→      const col = d.color;
   196→      const lw = d.lineWidth;
   197→
   198→      switch (d.type) {
   199→        case 'trendline': {
   200→          if (d.points.length < 1) break;
   201→          const p0 = d.points[0];
   202→          const p1 = d.points.length > 1 ? d.points[d.points.length - 1] : p0;
   203→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   204→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   205→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   206→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   207→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   208→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" />`;
   209→          break;
   210→        }
   211→        case 'hline': {
   212→          if (d.points.length < 1) break;
   213→          const p = d.points[0];
   214→          const y = chart.priceScale('right').priceToCoordinate(p.price);
   215→          if (y == null) break;
   216→          html += `<line x1="0" y1="${y}" x2="${w}" y2="${y}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   217→          html += `<text x="4" y="${y - 4}" fill="${col}" font-size="11" font-family="sans-serif">${p.price.toFixed(0)}</text>`;
   218→          break;
   219→        }
   220→        case 'vline': {
   221→          if (d.points.length < 1) break;
   222→          const p = d.points[0];
   223→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   224→          if (x == null) break;
   225→          html += `<line x1="${x}" y1="0" x2="${x}" y2="${h}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   226→          break;
   227→        }
   228→        case 'fibonacci': {
   229→          if (d.points.length < 2) break;
   230→          const p0 = d.points[0];
   231→          const p1 = d.points[1];
   232→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   233→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   234→          if (x0 == null || x1 == null) break;
   235→          const priceHigh = Math.max(p0.price, p1.price);
   236→          const priceLow = Math.min(p0.price, p1.price);
   237→          const priceRange = priceHigh - priceLow;
   238→          if (priceRange === 0) break;
   239→
   240→          // Filled background
   241→          const yTop = chart.priceScale('right').priceToCoordinate(priceHigh);
   242→          const yBottom = chart.priceScale('right').priceToCoordinate(priceLow);
   243→          if (yTop == null || yBottom == null) break;
   244→          html += `<rect x="${Math.min(x0,x1)}" y="${yTop}" width="${Math.abs(x1-x0)}" height="${yBottom - yTop}" fill="${col}" opacity="0.04" />`;
   245→
   246→          for (const level of FIB_LEVELS) {
   247→            const price = priceHigh - level.pct * priceRange;
   248→            const y = chart.priceScale('right').priceToCoordinate(price);
   249→            if (y == null) continue;
   250→            html += `<line x1="${Math.min(x0,x1) - 10}" y1="${y}" x2="${Math.max(x0,x1) + 10}" y2="${y}" stroke="${col}" stroke-width="1" opacity="0.6" />`;
   251→            html += `<rect x="${Math.max(x0,x1) + 12}" y="${y - 8}" width="56" height="16" fill="#0b0f1a" rx="2" />`;
   252→            html += `<text x="${Math.max(x0,x1) + 14}" y="${y + 4}" fill="${col}" font-size="10" font-family="sans-serif">${level.label} ${price.toFixed(0)}</text>`;
   253→          }
   254→          break;
   255→        }
   256→        case 'rectangle': {
   257→          if (d.points.length < 2) break;
   258→          const p0 = d.points[0];
   259→          const p1 = d.points[d.points.length - 1];
   260→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   261→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   262→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   263→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   264→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   265→          const rx = Math.min(x0, x1);
   266→          const ry = Math.min(y0, y1);
   267→          const rw = Math.abs(x1 - x0);
   268→          const rh = Math.abs(y1 - y0);
   269→          html += `<rect x="${rx}" y="${ry}" width="${rw}" height="${rh}" fill="${col}" fill-opacity="0.08" stroke="${col}" stroke-width="${lw}" rx="2" />`;
   270→          break;
   271→        }
   272→        case 'text': {
   273→          if (d.points.length < 1 || !d.text) break;
   274→          const p = d.points[0];
   275→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   276→          const y = chart.priceScale('right').priceToCoordinate(p.price);
   277→          if (x == null || y == null) break;
   278→          html += `<rect x="${x + 4}" y="${y - 16}" width="${d.text.length * 8 + 12}" height="22" fill="#0b0f1a" stroke="${col}" stroke-width="1" rx="3" />`;
   279→          html += `<text x="${x + 10}" y="${y + 1}" fill="${col}" font-size="12" font-family="sans-serif">${d.text}</text>`;
   280→          break;
   281→        }
   282→        case 'brush': {
   283→          if (d.points.length < 2) break;
   284→          let pts = '';
   285→          let valid = true;
   286→          for (const p of d.points) {
   287→            const x = chart.timeScale().timeToCoordinate(p.time as Time);
   288→            const y = chart.priceScale('right').priceToCoordinate(p.price);
   289→            if (x == null || y == null) { valid = false; break; }
   290→            pts += `${x},${y} `;
   291→          }
   292→          if (!valid || pts.length === 0) break;
   293→          html += `<polyline points="${pts.trim()}" fill="none" stroke="${col}" stroke-width="${lw}" stroke-linecap="round" stroke-linejoin="round" />`;
   294→          break;
   295→        }
   296→        case 'measure': {
   297→          if (d.points.length < 2) break;
   298→          const p0 = d.points[0];
   299→          const p1 = d.points[d.points.length - 1];
   300→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   301→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   302→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   303→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   304→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   305→
   306→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="4,2" />`;
   307→          // Price difference
   308→          const diff = p1.price - p0.price;
   309→          const pctChg = p0.price !== 0 ? ((diff / p0.price) * 100) : 0;
   310→          const bars = Math.abs(Math.round(p1.time - p0.time));
   311→          const midX = (x0 + x1) / 2;
   312→          const midY = (y0 + y1) / 2;
   313→          const label = `${diff >= 0 ? '+' : ''}${diff.toFixed(0)} (${pctChg >= 0 ? '+' : ''}${pctChg.toFixed(1)}%) ${bars}bar`;
   314→          html += `<rect x="${midX - 40}" y="${midY - 24}" width="80" height="20" fill="#0b0f1a" stroke="${col}" stroke-width="1" rx="3" />`;
   315→          html += `<text x="${midX - 36}" y="${midY - 10}" fill="${col}" font-size="10" font-family="sans-serif">${label}</text>`;
   316→          break;
   317→        }
   318→        case 'arrow': {
   319→          if (d.points.length < 2) break;
   320→          const p0 = d.points[0];
   321→          const p1 = d.points[d.points.length - 1];
   322→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   323→          const y0 = chart.priceScale('right').priceToCoordinate(p0.price);
   324→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   325→          const y1 = chart.priceScale('right').priceToCoordinate(p1.price);
   326→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   327→          const markerId = `arrow-${col.replace('#','')}`;
   328→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" marker-end="url(#${markerId})" />`;
   329→          break;
   330→        }
   331→      }
   332→    }
   333→
   334→    svg.innerHTML = html;
   335→  }, [drawings, currentDrawing]);
   336→
   337→  /* ── Chart Build ──────────────────────────────────────────── */
   338→  const build = useCallback(() => {
   339→    const el = containerRef.current;
   340→    const chartEl = chartContainerRef.current;
   341→    if (!el || chartEl || data.length === 0) return;
   342→
   343→    if (chartRef.current) {
   344→      chartRef.current.remove();
   345→      chartRef.current = null;
   346→    }
   347→
   348→    const chart = createChart(chartEl, {
   349→      layout: { background: { type: ColorType.Solid, color: BG }, textColor: TXT, fontSize: 11 },
   350→      grid: { vertLines: { color: GRID }, horzLines: { color: GRID } },
   351→      crosshair: { mode: CrosshairMode.Normal },
   352→      rightPriceScale: {
   353→        borderColor: 'rgba(255,255,255,0.08)',
   354→        scaleMargins: { top: 0.05, bottom: 0.3 },
   355→      },
   356→      timeScale: { borderColor: 'rgba(255,255,255,0.08)', rightOffset: 5, barSpacing: 6 },
   357→      width: chartEl.clientWidth,
   358→      height,
   359→    });
   360→    chartRef.current = chart;
   361→
   362→    // Numeric time index
   363→    const candles = data.map((d, i) => ({
   364→      time: i as Time,
   365→      open: d.open,
   366→      high: d.high,
   367→      low: d.low,
   368→      close: d.close,
   369→    }));
   370→
   371→    const cs = chart.addSeries(CandlestickSeries, {
   372→      upColor: BULL,
   373→      downColor: BEAR,
   374→      borderUpColor: BULL,
   375→      borderDownColor: BEAR,
   376→      wickUpColor: BULL,
   377→      wickDownColor: BEAR,
   378→    });
   379→    csRef.current = cs;
   380→    cs.setData(candles);
   381→
   382→    // Volume
   383→    const vol = chart.addSeries(HistogramSeries, {
   384→      priceFormat: { type: 'volume' },
   385→      priceScaleId: 'vol',
   386→    });
   387→    vol.setData(
   388→      data.map((d, i) => ({
   389→        time: i as Time,
   390→        value: d.volume,
   391→        color: d.close >= d.open ? BULL : BEAR,
   392→      }))
   393→    );
   394→    chart.priceScale('vol').applyOptions({ scaleMargins: { top: 0.8, bottom: 0 } });
   395→
   396→    // TA overlays
   397→    if (ta) {
   398→      for (const cfg of SMA_CFG) {
   399→        const v = ta.sma[cfg.key];
   400→        if (v && v > 0)
   401→          cs.createPriceLine({
   402→            price: v,
   403→            color: cfg.color,
   404→            lineWidth: 1,
   405→            lineStyle: 0,
   406→            axisLabelVisible: true,
   407→            title: cfg.title,
   408→          });
   409→      }
   410→      const bb = ta.bollingerBands;
   411→      if (bb) {
   412→        if (bb.upper > 0)
   413→          cs.createPriceLine({ price: bb.upper, color: 'rgba(161,98,255,0.5)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Upper' });
   414→        if (bb.middle > 0)
   415→          cs.createPriceLine({ price: bb.middle, color: 'rgba(161,98,255,0.5)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Mid' });
   416→        if (bb.lower > 0)
   417→          cs.createPriceLine({ price: bb.lower, color: 'rgba(161,98,255,0.5)', lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title: 'BB Lower' });
   418→      }
   419→      if (ta.sar > 0)
   420→        cs.createPriceLine({ price: ta.sar, color: 'rgba(255,255,255,0.35)', lineWidth: 1, lineStyle: 1, axisLabelVisible: true, title: 'SAR' });
   421→
   422→      // S/R with strength
   423→      const resistanceStrengths = ta.resistanceStrengths || [];
   424→      const supportStrengths = ta.supportStrengths || [];
   425→
   426→      resistanceStrengths.forEach((r, i) => {
   427→        if (r.price <= 0) return;
   428→        const lw = r.isTarget ? 3 : Math.max(1, Math.round(r.strength / 3));
   429→        const ls = r.isTarget ? 0 : 2;
   430→        const title = `R${i + 1} (${r.strength}/10)`;
   431→        cs.createPriceLine({
   432→          price: r.price,
   433→          color: r.isTarget ? '#ff6b6b' : BEAR,
   434→          lineWidth: lw,
   435→          lineStyle: ls as 0 | 1 | 2,
   436→          axisLabelVisible: i < 6,
   437→          title,
   438→        });
   439→      });
   440→
   441→      supportStrengths.forEach((s, i) => {
   442→        if (s.price <= 0) return;
   443→        const lw = s.isTarget ? 3 : Math.max(1, Math.round(s.strength / 3));
   444→        const ls = s.isTarget ? 0 : 2;
   445→        const title = `S${i + 1} (${s.strength}/10)`;
   446→        cs.createPriceLine({
   447→          price: s.price,
   448→          color: s.isTarget ? '#6bffb8' : BULL,
   449→          lineWidth: lw,
   450→          lineStyle: ls as 0 | 1 | 2,
   451→          axisLabelVisible: i < 6,
   452→          title,
   453→        });
   454→      });
   455→    }
   456→
   457→    chart.timeScale().fitContent();
   458→
   459→    // Subscribe to visible range changes to re-render SVG
   460→    chart.timeScale().subscribeVisibleLogicalRangeChange(() => {
   461→      setRenderKey((k) => k + 1);
   462→    });
   463→
   464→    // Set initial chart size
   465→    setChartSize({ w: chartEl.clientWidth, h });
   466→  }, [data, ta, height]);
   467→
   468→  /* ── Re-render SVG when renderKey changes ────────────────── */
   469→  useEffect(() => {
   470→    renderDrawings();
   471→  }, [renderKey, renderDrawings]);
   472→
   473→  /* ── Init chart ───────────────────────────────────────────── */
   474→  useEffect(() => {
   475→    build();
   476→    const el = chartContainerRef.current;
   477→    if (!el) return;
   478→    const ro = new ResizeObserver((entries) => {
   479→      for (const e of entries) {
   480→        const { width: w, height: h } = e.contentRect;
   481→        if (w > 0 && h > 0) {
   482→          chartRef.current?.applyOptions({ width: w, height: h });
   483→          setChartSize({ w, h });
   484→        }
   485→      }
   486→    });
   487→    ro.observe(el);
   488→    return () => {
   489→      ro.disconnect();
   490→      if (chartRef.current) {
   491→        chartRef.current.remove();
   492→        chartRef.current = null;
   493→      }
   494→    };
   495→  }, [build]);
   496→
   497→  /* ── Mouse event handlers ────────────────────────────────── */
   498→  const handleMouseDown = useCallback(
   499→    (e: React.MouseEvent) => {
   500→      if (activeTool === 'cursor') return;
   501→      if (textInput) return; // don't draw while text input is open
   502→
   503→      const point = getChartPoint(e.clientX, e.clientY);
   504→      if (!point) return;
   505→
   506→      // Horizontal line: complete on single click
   507→      if (activeTool === 'hline') {
   508→        const drawing: Drawing = {
   509→          id: Date.now().toString(),
   510→          type: 'hline',
   511→          points: [point],
   512→          color: activeColor,
   513→          lineWidth: 1,
   514→          completed: true,
   515→        };
   516→        setDrawings((prev) => [...prev, drawing]);
   517→        setIsDrawing(false);
   518→        return;
   519→      }
   520→
   521→      // Text tool: show inline input
   522→      if (activeTool === 'text') {
   523→        const chart = chartRef.current;
   524→        const container = chartContainerRef.current;
   525→        if (!chart || !container) return;
   526→        const rect = container.getBoundingClientRect();
   527→        const x = e.clientX - rect.left;
   528→        const y = e.clientY - rect.top;
   529→
   530→        const tempDrawing: Drawing = {
   531→          id: Date.now().toString(),
   532→          type: 'text',
   533→          points: [point],
   534→          color: activeColor,
   535→          lineWidth: 1,
   536→          text: '',
   537→          completed: false,
   538→        };
   539→        setCurrentDrawing(tempDrawing);
   540→        setTextInput({ x, y, drawingId: tempDrawing.id });
   541→        setTextValue('');
   542→        setIsDrawing(true);
   543→        return;
   544→      }
   545→
   546→      // Other tools: start drawing
   547→      const drawing: Drawing = {
   548→        id: Date.now().toString(),
   549→        type: activeTool,
   550→        points: [point],
   551→        color: activeColor,
   552→        lineWidth: activeTool === 'brush' ? 2 : 1,
   553→        completed: false,
   554→      };
   555→      setCurrentDrawing(drawing);
   556→      setIsDrawing(true);
   557→    },
   558→    [activeTool, activeColor, getChartPoint, textInput]
   559→  );
   560→
   561→  const handleMouseMove = useCallback(
   562→    (e: React.MouseEvent) => {
   563→      if (!isDrawing || !currentDrawing) return;
   564→
   565→      const point = getChartPoint(e.clientX, e.clientY);
   566→      if (!point) return;
   567→
   568→      if (currentDrawing.type === 'brush') {
   569→        // Add points continuously for brush
   570→        setCurrentDrawing({
   571→          ...currentDrawing,
   572→          points: [...currentDrawing.points, point],
   573→        });
   574→      } else if (currentDrawing.type !== 'text' && currentDrawing.type !== 'hline') {
   575→        // Update last point for other tools
   576→        setCurrentDrawing({
   577→          ...currentDrawing,
   578→          points: [currentDrawing.points[0], point],
   579→        });
   580→      }
   581→    },
   582→    [isDrawing, currentDrawing, getChartPoint]
   583→  );
   584→
   585→  const handleMouseUp = useCallback(() => {
   586→    if (!isDrawing || !currentDrawing) return;
   587→
   588→    if (currentDrawing.type === 'text') {
   589→      // Don't complete text yet - wait for text input submission
   590→      return;
   591→    }
   592→
   593→    const completed: Drawing = {
   594→      ...currentDrawing,
   595→      completed: true,
   596→    };
   597→
   598→    // Only save if it has enough points
   599→    const minPoints = currentDrawing.type === 'hline' ? 1 : 2;
   600→    if (completed.points.length >= minPoints) {
   601→      setDrawings((prev) => [...prev, completed]);
   602→    }
   603→
   604→    setCurrentDrawing(null);
   605→    setIsDrawing(false);
   606→  }, [isDrawing, currentDrawing]);
   607→
   608→  /* ── Text input submit ────────────────────────────────────── */
   609→  const handleTextSubmit = useCallback(() => {
   610→    if (!textInput || !textValue.trim() || !currentDrawing) {
   611→      setTextInput(null);
   612→      setCurrentDrawing(null);
   613→      setIsDrawing(false);
   614→      return;
   615→    }
   616→
   617→    const completed: Drawing = {
   618→      ...currentDrawing,
   619→      text: textValue.trim(),
   620→      completed: true,
   621→    };
   622→    setDrawings((prev) => [...prev, completed]);
   623→    setTextInput(null);
   624→    setTextValue('');
   625→    setCurrentDrawing(null);
   626→    setIsDrawing(false);
   627→  }, [textInput, textValue, currentDrawing]);
   628→
   629→  const handleTextCancel = useCallback(() => {
   630→    setTextInput(null);
   631→    setTextValue('');
   632→    setCurrentDrawing(null);
   633→    setIsDrawing(false);
   634→  }, []);
   635→
   636→  /* ── Delete / Clear ──────────────────────────────────────── */
   637→  const handleDeleteLast = useCallback(() => {
   638→    setDrawings((prev) => prev.slice(0, -1));
   639→  }, []);
   640→
   641→  const handleClearAll = useCallback(() => {
   642→    setDrawings([]);
   643→  }, []);
   644→
   645→  /* ── Render ───────────────────────────────────────────────── */
   646→  return (
   647→    <div className="w-full">
   648→      {/* ── TOOLBAR ────────────────────────────────────────── */}
   649→      <div
   650→        className="flex items-center gap-1 px-2 py-1.5 rounded-t-lg border border-b-0 overflow-x-auto flex-nowrap"
   651→        style={{ backgroundColor: TOOLBAR_BG, borderColor: TOOLBAR_BORDER }}
   652→      >
   653→        {/* Tool buttons */}
   654→        {TOOLS.map((tool) => (
   655→          <button
   656→            key={tool.id}
   657→            onClick={() => {
   658→              setActiveTool(tool.id);
   659→              // Cancel ongoing drawing when switching tools
   660→              if (isDrawing) {
   661→                setIsDrawing(false);
   662→                setCurrentDrawing(null);
   663→                setTextInput(null);
   664→              }
   665→            }}
   666→            title={tool.label}
   667→            className={`shrink-0 flex items-center justify-center w-8 h-8 rounded-md transition-all cursor-pointer ${
   668→              activeTool === tool.id
   669→                ? 'bg-amber-500/20 text-amber-400 ring-1 ring-amber-500/30'
   670→                : 'text-gray-400 hover:bg-white/5 hover:text-gray-200'
   671→            }`}
   672→          >
   673→            {tool.icon}
   674→          </button>
   675→        ))}
   676→
   677→        {/* Separator */}
   678→        <div className="w-px h-6 bg-white/10 mx-1 shrink-0" />
   679→
   680→        {/* Color picker */}
   681→        {PALETTE.map((c) => (
   682→          <button
   683→            key={c}
   684→            onClick={() => setActiveColor(c)}
   685→            title={c}
   686→            className={`shrink-0 w-6 h-6 rounded-full border-2 transition-all cursor-pointer ${
   687→              activeColor === c ? 'border-white scale-110' : 'border-transparent hover:scale-105'
   688→            }`}
   689→            style={{ backgroundColor: c }}
   690→          />
   691→        ))}
   692→
   693→        {/* Separator */}
   694→        <div className="w-px h-6 bg-white/10 mx-1 shrink-0" />
   695→
   696→        {/* Delete last */}
   697→        <button
   698→          onClick={handleDeleteLast}
   699→          title="حذف آخر"
   700→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-400 hover:bg-red-500/15 hover:text-red-400 transition-all cursor-pointer"
   701→        >
   702→          <Trash2 className="w-3.5 h-3.5" />
   703→        </button>
   704→
   705→        {/* Clear all */}
   706→        <button
   707→          onClick={handleClearAll}
   708→          title="پاک کردن همه"
   709→          className="shrink-0 flex items-center justify-center w-8 h-8 rounded-md text-gray-400 hover:bg-red-500/15 hover:text-red-400 transition-all cursor-pointer"
   710→        >
   711→          <X className="w-3.5 h-3.5" />
   712→        </button>
   713→
   714→        {/* Active tool label */}
   715→        <div className="mr-auto shrink-0 text-xs text-gray-500 font-medium" dir="rtl">
   716→          {TOOLS.find((t) => t.id === activeTool)?.label}
   717→        </div>
   718→      </div>
   719→
   720→      {/* ── CHART + SVG OVERLAY ─────────────────────────────── */}
   721→      <div
   722→        className="relative rounded-b-lg overflow-hidden border"
   723→        style={{ borderColor: TOOLBAR_BORDER, height }}
   724→      >
   725→        {/* Chart container */}
   726→        <div
   727→          ref={chartContainerRef}
   728→          className="w-full h-full"
   729→          data-chart
   730→        />
   731→
   732→        {/* SVG overlay for drawings */}
   733→        <svg
   734→          ref={svgRef}
   735→          className="absolute top-0 left-0"
   736→          style={{
   737→            width: chartSize.w || '100%',
   738→            height: chartSize.h || '100%',
   739→            pointerEvents: activeTool !== 'cursor' ? 'auto' : 'none',
   740→            zIndex: 10,
   741→          }}
   742→          onMouseDown={handleMouseDown}
   743→          onMouseMove={handleMouseMove}
   744→          onMouseUp={handleMouseUp}
   745→          onMouseLeave={handleMouseUp}
   746→        />
   747→
   748→        {/* Text input overlay */}
   749→        {textInput && (
   750→          <div
   751→            className="absolute z-20"
   752→            style={{
   753→              left: textInput.x,
   754→              top: textInput.y - 8,
   755→            }}
   756→          >
   757→            <input
   758→              type="text"
   759→              value={textValue}
   760→              onChange={(e) => setTextValue(e.target.value)}
   761→              onKeyDown={(e) => {
   762→                if (e.key === 'Enter') handleTextSubmit();
   763→                if (e.key === 'Escape') handleTextCancel();
   764→              }}
   765→              placeholder="متن..."
   766→              autoFocus
   767→              dir="rtl"
   768→              className="bg-[#0b0f1a] border border-gray-600 rounded px-2 py-1 text-sm text-white outline-none focus:border-amber-500 w-32"
   769→              onBlur={handleTextSubmit}
   770→            />
   771→          </div>
   772→        )}
   773→      </div>
   774→    </div>
   775→  );
   776→}
