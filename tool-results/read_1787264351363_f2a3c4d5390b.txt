   227→  // Store last fetch params for auto-refresh

   234→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number) => {
   235→    // TGJU instrument: fetch historical data via tgju.org chart API
   236→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   237→      setLoading(true);
   238→      setLoadingMessage('در حال دریافت داده‌های تاریخی ... (حدود ۱۵ ثانیه)');
   239→      setError(null);
   240→      setData(null);
   241→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId };
   242→      try {
   243→        const res = await fetch(`/api/tgju-analysis?key=${encodeURIComponent(tgjuKey)}`);
   244→        if (!res.ok) {
   245→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   246→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch { /* non-JSON error */ }
   247→          throw new Error(errMsg);
   248→        }
   249→        const json = await res.json();
   250→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   251→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   252→        }
   253→        setData({ ...json, isTgju: true });
   254→        setActivePanel('visual');
   255→      } catch (err) {
   256→        setError(String(err));
   257→      } finally {
   258→        setLoading(false);
   259→        setLoadingMessage(null);
   260→      }
   261→      return;
   262→    }
   263→
   264→    // Index: fetch historical data via TSETMC CDN (webId), then finpy-tse (finpyIndex/finpySector)
   265→    if (category === 'index') {
   266→      setLoading(true);
   267→      setLoadingMessage('در حال دریافت داده‌های تاریخی شاخص ...');
   268→      setError(null);
   269→      setData(null);
   270→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId };
   271→      let gotError = false;
   272→      try {
   273→        // 1. Try webId (TSETMC CDN)
   274→        if (webId) {
   275→          setLoadingMessage('در حال دریافت داده‌های شاخص از TSETMC ...');
   276→          const res = await fetch(`/api/finpy-sector?webId=${encodeURIComponent(String(webId))}`);
   277→          if (res.ok) {
   278→            try {
   279→              const json = await res.json();
   280→              if (json.candles && json.candles.length > 0 && json.ta) {
   281→                setData(json);
   282→                setActivePanel('visual');
   283→                setLoading(false);
   284→                setLoadingMessage(null);
   285→                return;
   286→              }
   287→              if (json.error) { setError(json.error); gotError = true; }
   288→            } catch { /* invalid JSON */ }
   289→          } else { gotError = true; }
   290→        }
   291→
   292→        // 2. Main indices via finpy-tse
   293→        if (finpyIndex) {
   294→          setLoadingMessage('در حال دریافت داده‌های شاخص اصلی ...');
   295→          const res = await fetch(`/api/finpy-sector?indexKey=${encodeURIComponent(finpyIndex)}`);
   296→          if (res.ok) {
   297→            try {
   298→              const json = await res.json();
   299→              if (json.candles && json.candles.length > 0 && json.ta) {
   300→                setData(json);
   301→                setActivePanel('visual');
   302→                setLoading(false);
   303→                setLoadingMessage(null);
   304→                return;
   305→              }
   306→              if (json.error) { setError(json.error); gotError = true; }
   307→            } catch { /* invalid JSON */ }
   308→          } else { gotError = true; }
   309→        }
   310→
   311→        // 3. Sector indices via finpy-tse
   312→        if (finpySector) {
   313→          setLoadingMessage('در حال دریافت داده‌های شاخص گروه ...');
   314→          const res = await fetch(`/api/finpy-sector?sector=${encodeURIComponent(finpySector)}`);
   315→          if (res.ok) {
   316→            try {
   317→              const json = await res.json();
   318→              if (json.candles && json.candles.length > 0 && json.ta) {
   319→                setData(json);
   320→                setActivePanel('visual');
   321→                setLoading(false);
   322→                setLoadingMessage(null);
   323→                return;
   324→              }
   325→              if (json.error) { setError(json.error); gotError = true; }
   326→            } catch { /* invalid JSON */ }
   327→          } else { gotError = true; }
   328→        }
   329→
   330→        if (!gotError) {
   331→          setError('داده‌های تاریخی این شاخص در حال حاضر قابل دسترسی نیست.');
   332→        }
   333→      } catch (err) {
   334→        setError(String(err));
   335→      }
   336→      setLoading(false);
   337→      setLoadingMessage(null);
   338→      return;
   339→    }
   340→
   341→    // Regular TSE instrument: full TA analysis (تعدیل شده / adjusted prices only)
   342→    setLoading(true);
   343→    setError(null);
