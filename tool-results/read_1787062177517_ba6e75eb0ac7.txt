   200→  EDGES.forEach((e, i) => {
   201→    if (!adj[e[0]]) adj[e[0]] = [];
   202→    adj[e[0]].push({ to: e[1], edgeIdx: i });
   203→  });
   204→
   205→  const paths: PathInfo[] = [];
   206→  const MAX_PATHS = 200;
   207→
   208→  function dfs(node: string, visited: Set<string>, currentPath: string[], edgeIndices: number[], currentProb: number) {
   209→    if (SCENARIO_KEYS.includes(node as any)) {
   210→      paths.push({ nodes: [...currentPath, node], edges: [...edgeIndices], prob: currentProb, target: node });
   211→      return;
   212→    }
   213→    if (paths.length >= MAX_PATHS) return;
   214→    if (visited.has(node)) return;
   215→    visited.add(node);
   216→
   217→    const outs = adj[node] ?? [];
   218→    for (const { to, edgeIdx } of outs) {
   219→      const ep = edgeProbs[String(edgeIdx)] ?? 0.1;
   220→      dfs(to, visited, [...currentPath, node], [...edgeIndices, edgeIdx], currentProb * ep);
   221→    }
   222→    visited.delete(node);
   223→  }
   224→
   225→  dfs('A', new Set(), [], [], 1);
   226→  return paths.sort((a, b) => b.prob - a.prob);
   227→}

   599→                      className="inline-block mt-1.5 px-2.5 py-0.5 rounded-full text-sm font-bold"
   600→                      style={{
   601→                        background: `color-mix(in srgb, ${scenarioColor} 17%, transparent)`,
   602→                        color: scenarioColor,
   603→                      }}
   604→                    >{toFa(scenarioProb)}٪</span>
   605→                  )}
   606→                </div>
   607→              );
   608→            })}
   609→
   610→            {/* Legend */}
   611→            <div className="absolute bottom-3 right-3 p-2.5 rounded-lg border border-[#e5e7eb] bg-[#ffffff]/90 text-[10px] text-[#374151] leading-7 z-10">
   612→              <div className="flex items-center gap-1.5"><span className="inline-block w-2 h-2 rounded-full bg-[#34c98b]" />صعود و تأیید</div>
   613→              <div className="flex items-center gap-1.5"><span className="inline-block w-2 h-2 rounded-full bg-[#4186ff]" />پولبک و بازگشت</div>
   614→              <div className="flex items-center gap-1.5"><span className="inline-block w-2 h-2 rounded-full bg-[#ff7b32]" />اصلاح</div>
   615→              <div className="flex items-center gap-1.5"><span className="inline-block w-2 h-2 rounded-full bg-[#ef4d62]" />ریسک / ابطال</div>
   616→            </div>
   617→          </div>
   618→        </div>
   619→
   620→        {/* ═══ Right Panel: Path Probabilities ═══ */}
   621→        <div className="rounded-2xl border border-[#e5e7eb] p-4 flex flex-col"
   622→          style={{ background: 'linear-gradient(160deg, #ffffff, #f3f4f6)', boxShadow: '0 4px 16px rgba(0,0,0,.06)' }}>
   623→          <h2 className="text-sm font-bold text-[#111827] mb-3">📋 احتمال مسیرها</h2>
   624→
   625→          {selectedNode && detailContent ? (
   626→            <div className="border-t border-[#e5e7eb] pt-3 flex-1 overflow-y-auto max-h-[580px] custom-scrollbar">
   627→              {detailContent}
   628→            </div>
   629→          ) : (
   630→            <div className="flex-1 overflow-y-auto max-h-[580px] space-y-3 custom-scrollbar">
   631→              {/* Path probabilities per target */}
   632→              <div className="space-y-2">
   633→                {SCENARIO_KEYS.map(key => {
   634→                  const meta = SCENARIO_META[key];
   635→                  const s = scenarios[key];
   636→                  const pathProb = pathProbsByTarget[key] ?? 0;
   637→                  const isActive = activeFilter === key;
   638→                  return (
   639→                    <div
   640→                      key={key}
   641→                      onClick={() => setActiveFilter(key)}
   642→                      className={`rounded-xl p-3 cursor-pointer transition-all border ${
   643→                        isActive
   644→                          ? 'border-opacity-60'
   645→                          : 'border-[#e5e7eb]/60 hover:border-[#e5e7eb]'
   646→                      }`}
   647→                      style={{
   648→                        background: isActive ? `${meta.color}0a` : 'rgba(243,244,246,0.5)',
   649→                        borderColor: isActive ? meta.color + '80' : undefined,
   650→                      }}
   651→                    >
   652→                      <div className="flex items-center justify-between mb-1">
   653→                        <span className="text-xs font-bold" style={{ color: meta.color }}>{key} | {meta.label}</span>
   654→                        <span className="text-lg font-black" style={{ color: meta.color }}>{(pathProb * 100).toFixed(1)}٪</span>
   655→                      </div>
   656→                      {/* Probability bar */}
   657→                      <div className="w-full h-1.5 rounded-full bg-[#e5e7eb] mb-2">
   658→                        <div
   659→                          className="h-full rounded-full transition-all duration-300"
   660→                          style={{ width: `${Math.min(100, pathProb * 100)}%`, background: meta.color }}
   661→                        />
   662→                      </div>
   663→                      <div className="flex items-center justify-between text-[10px] text-[#6b7280]">
   664→                        <span>تعداد مسیرها: {filteredPaths.filter(p => p.target === key).length}</span>
   665→                        <span>احتمال سناریو: <b className="text-[#374151]">{toFa(s.probability)}٪</b></span>
   666→                      </div>
   667→                    </div>
   668→                  );
   669→                })}
   670→              </div>
   671→
   672→              {/* Top paths list */}
   673→              <div className="border-t border-[#e5e7eb] pt-3 mt-3">
   674→                <h3 className="text-xs font-bold text-[#111827] mb-2">مسیرهای اصلی (تا ۱۰ مسیر)</h3>
   675→                <div className="space-y-1.5">
   676→                  {filteredPaths.slice(0, 10).map((p, i) => {
   677→                    const meta = SCENARIO_META[p.target];
   678→                    return (
   679→                      <div key={i} className="flex items-center gap-2 text-[10px]">
   680→                        <span className="w-4 text-center font-bold" style={{ color: meta.color }}>{i + 1}</span>
   681→                        <span className="flex-1 text-[#374151]" dir="ltr">
   682→                          {p.nodes.join(' → ')}
   683→                        </span>
   684→                        <span className="font-bold min-w-[40px] text-left" style={{ color: meta.color }}>
   685→                          {(p.prob * 100).toFixed(1)}٪
   686→                        </span>
   687→                      </div>
   688→                    );
   689→                  })}
   690→                  {filteredPaths.length > 10 && (
   691→                    <div className="text-[10px] text-[#B0A89E] text-center pt-1">
   692→                      و {filteredPaths.length - 10} مسیر دیگر...
   693→                    </div>
   694→                  )}
   695→                </div>
   696→              </div>
   697→
   698→              <div className="border-t border-[#e5e7eb] pt-3 mt-3">
   699→                <p className="text-[10px] text-[#6b7280] leading-relaxed">
   700→                  <b>قاعده:</b> عبور یا شکست هر سطح فقط با تثبیت معتبر تلقی می‌شود.<br />
   701→                  <b>احتمال یال:</b> حاصل توزیع شرطی بر اساس سیگنال‌های تکنیکال.<br />
   702→                  <b>احتمال مسیر:</b> حاصل‌ضرب احتمال یال‌های مسیر.
   703→                </p>
   704→              </div>
   705→            </div>
   706→          )}
   707→        </div>
   708→      </div>
   709→
   710→      {/* ═══ Scenario Result Cards ═══ */}
   711→      <div className="mt-4 p-4 rounded-2xl border border-[#e5e7eb] bg-[#ffffff]">
   712→        <h2 className="text-sm font-bold text-[#111827] mb-3">گره‌های نتیجه و احتمال تجمیعی مسیرهای ورودی</h2>
   713→        <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-3">
   714→          {SCENARIO_KEYS.map(key => {
   715→            const s = scenarios[key];
   716→            const meta = SCENARIO_META[key];
   717→            return (
   718→              <div
   719→                key={key}
   720→                className="rounded-xl p-3"
   721→                style={{
   722→                  '--scolor': meta.color,
   723→                  border: `1px solid color-mix(in srgb, ${meta.color} 40%, transparent)`,
   724→                  background: `linear-gradient(160deg, color-mix(in srgb, ${meta.color} 8%, #ffffff), #ffffff)`,
   725→                } as React.CSSProperties}
   726→              >
   727→                <strong className="block text-xl font-black" style={{ color: meta.color }}>{toFa(s.probability)}٪</strong>
   728→                <span className="text-xs font-bold text-[#374151]">{key} | {meta.label}</span>
   729→                <small className="block text-[10px] text-[#6b7280] leading-relaxed mt-2" dir="ltr">
   730→                  {toFa(s.targetMin)} — {toFa(s.targetMax)} ریال
   731→                </small>
   732→              </div>
   733→            );
   734→          })}
   735→        </div>
   736→        <div className="mt-3 px-4 py-2.5 rounded-lg border-r-3 border-amber-700/60 bg-amber-50 text-[11px] text-[#374151] leading-relaxed">
   737→          <b>محدودیت مدل:</b> احتمال‌های سناریو توسط موتور محاسباتی بر اساس فرمول‌های تعریف‌شده (ADX, RSI, MFI, CCI و فاصله از سطوح) محاسبه شده‌اند. احتمال مسیرها از حاصل‌ضرب احتمال شرطی یال‌ها به دست می‌آید.
   738→        </div>
   739→      </div>
   740→    </div>
   741→  );
   742→}
