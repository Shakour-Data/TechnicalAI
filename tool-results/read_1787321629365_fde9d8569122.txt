   135→   135→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   136→   136→function sma(closes: number[], period: number): number {
   137→   137→  if (closes.length < period) return 0;
   138→   138→  const slice = closes.slice(-period);
   139→   139→  return slice.reduce((a, b) => a + b, 0) / period;
   140→   140→}

   142→   142→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   143→   143→function emaCalc(closes: number[], period: number): number {
   144→   144→  if (closes.length < period) return 0;
   145→   145→  const k = 2 / (period + 1);
   146→   146→  let emaVal = sma(closes.slice(0, period), period);
   147→   147→  for (let i = period; i < closes.length; i++) {
   148→   148→    emaVal = closes[i] * k + emaVal * (1 - k);
   149→   149→  }
   150→   150→  return emaVal;
   151→   151→}

   153→   153→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   154→   154→function calcRSI(closes: number[], period: number = 14): number {
   155→   155→  if (closes.length < period + 1) return 50;
   156→   156→  let avgGain = 0;
   157→   157→  let avgLoss = 0;
   158→   158→  for (let i = 1; i <= period; i++) {
   159→   159→    const diff = closes[i] - closes[i - 1];
   160→   160→    if (diff > 0) avgGain += diff;
   161→   161→    else avgLoss += Math.abs(diff);
   162→   162→  }
   163→   163→  avgGain /= period;
   164→   164→  avgLoss /= period;
   165→   165→  for (let i = period + 1; i < closes.length; i++) {
   166→   166→    const diff = closes[i] - closes[i - 1];
   167→   167→    const gain = diff > 0 ? diff : 0;
   168→   168→    const loss = diff < 0 ? Math.abs(diff) : 0;
   169→   169→    avgGain = (avgGain * (period - 1) + gain) / period;
   170→   170→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   171→   171→  }
   172→   172→  if (avgLoss === 0) return 100;
   173→   173→  const rs = avgGain / avgLoss;
   174→   174→  return 100 - 100 / (1 + rs);
   175→   175→}

   177→   177→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   178→   178→function calcMFI(data: OHLCV[], period: number = 14): number {
   179→   179→  if (data.length < period + 1) return 50;
   180→   180→  let posFlow = 0;
   181→   181→  let negFlow = 0;
   182→   182→  for (let i = data.length - period; i < data.length; i++) {
   183→   183→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   184→   184→    const mf = tp * data[i].volume;
   185→   185→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   186→   186→    if (tp > prevTp) posFlow += mf;
   187→   187→    else negFlow += mf;
   188→   188→  }
   189→   189→  if (negFlow === 0) return 100;
   190→   190→  const mfr = posFlow / negFlow;
   191→   191→  return 100 - 100 / (1 + mfr);
   192→   192→}

   195→   195→function calcCCI(data: OHLCV[], period: number = 20): number {
   196→   196→  if (data.length < period) return 0;
   197→   197→  const tps: number[] = [];
   198→   198→  for (let i = data.length - period; i < data.length; i++) {
   199→   199→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   200→   200→  }
   201→   201→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   202→   202→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   203→   203→  if (meanDev === 0) return 0;
   204→   204→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   205→   205→}

   208→   208→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   209→   209→  if (data.length < kPeriod) return { k: 50, d: 50 };
   210→   210→  const rawKs: number[] = [];
   211→   211→  for (let i = kPeriod - 1; i < data.length; i++) {
   212→   212→    let lowest = Infinity;
   213→   213→    let highest = -Infinity;
   214→   214→    for (let j = i - kPeriod + 1; j <= i; j++) {
   215→   215→      if (data[j].low < lowest) lowest = data[j].low;
   216→   216→      if (data[j].high > highest) highest = data[j].high;
   217→   217→    }
   218→   218→    const range = highest - lowest;
   219→   219→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   220→   220→  }
   221→   221→  const smoothedK: number[] = [];
   222→   222→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   223→   223→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   224→   224→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   225→   225→  }
   226→   226→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   227→   227→  const dSlice = smoothedK.slice(-smoothD);
   228→   228→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   229→   229→  return { k, d };
   230→   230→}

   233→   233→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   234→   234→  if (data.length < period) return -50;
   235→   235→  const slice = data.slice(-period);
   236→   236→  let highest = -Infinity;
   237→   237→  let lowest = Infinity;
   238→   238→  for (const c of slice) {
   239→   239→    if (c.high > highest) highest = c.high;
   240→   240→    if (c.low < lowest) lowest = c.low;
   241→   241→  }
   242→   242→  const range = highest - lowest;
   243→   243→  if (range === 0) return -50;
   244→   244→  return ((highest - data[data.length - 1].close) / range) * -100;
   245→   245→}

   247→   247→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   248→   248→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   249→   249→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   250→   250→  const emaFast = emaCalc(closes, fast);
   251→   251→  const emaSlow = emaCalc(closes, slow);
   252→   252→  const macdLine = emaFast - emaSlow;
   253→   253→  // Calculate signal as EMA of MACD values
   254→   254→  const macdValues: number[] = [];
   255→   255→  const kFast = 2 / (fast + 1);
   256→   256→  const kSlow = 2 / (slow + 1);
   257→   257→  let eFast = sma(closes.slice(0, fast), fast);
   258→   258→  let eSlow = sma(closes.slice(0, slow), slow);
   259→   259→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   260→   260→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   261→   261→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   262→   262→    macdValues.push(eFast - eSlow);
   263→   263→  }
   264→   264→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   265→   265→  const kSig = 2 / (sig + 1);
   266→   266→  let signalVal = sma(macdValues.slice(0, sig), sig);
   267→   267→  for (let i = sig; i < macdValues.length; i++) {
   268→   268→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   269→   269→  }
   270→   270→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   271→   271→}

   273→   273→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   274→   274→function calcATR(data: OHLCV[], period: number = 14): number {
   275→   275→  if (data.length < 2) return 0;
   276→   276→  const trs: number[] = [];
   277→   277→  for (let i = 1; i < data.length; i++) {
   278→   278→    const tr = Math.max(
   279→   279→      data[i].high - data[i].low,
   280→   280→      Math.abs(data[i].high - data[i - 1].close),
   281→   281→      Math.abs(data[i].low - data[i - 1].close)
   282→   282→    );
   283→   283→    trs.push(tr);
   284→   284→  }
   285→   285→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   286→   286→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   287→   287→  for (let i = period; i < trs.length; i++) {
   288→   288→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   289→   289→  }
   290→   290→  return atrVal;
   291→   291→}

   293→   293→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   294→   294→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   295→   295→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   296→   296→  const plusDMs: number[] = [];
   297→   297→  const minusDMs: number[] = [];
   298→   298→  const trs: number[] = [];
   299→   299→  for (let i = 1; i < data.length; i++) {
   300→   300→    const upMove = data[i].high - data[i - 1].high;
   301→   301→    const downMove = data[i - 1].low - data[i].low;
   302→   302→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   303→   303→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   304→   304→    trs.push(Math.max(
   305→   305→      data[i].high - data[i].low,
   306→   306→      Math.abs(data[i].high - data[i - 1].close),
   307→   307→      Math.abs(data[i].low - data[i - 1].close)
   308→   308→    ));
   309→   309→  }
   310→   310→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   311→   311→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   312→   312→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   313→   313→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   314→   314→  const dxValues: number[] = [];
   315→   315→  for (let i = period; i < trs.length; i++) {
   316→   316→    smoothTR = smoothTR - smoothTR / period + trs[i];
   317→   317→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   318→   318→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   319→   319→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   320→   320→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   321→   321→    const diSum = diPlus + diMinus;
   322→   322→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   323→   323→    dxValues.push(dx);
   324→   324→  }
   325→   325→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   326→   326→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   327→   327→  for (let i = period; i < dxValues.length; i++) {
   328→   328→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   329→   329→  }
   330→   330→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   331→   331→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   332→   332→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   333→   333→}

   335→   335→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   336→   336→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   337→   337→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   338→   338→  let isLong = true;
   339→   339→  let af = step;
   340→   340→  let ep = data[0].high;
   341→   341→  let sar = data[0].low;
   342→   342→  for (let i = 1; i < data.length; i++) {
   343→   343→    const prevSar = sar;
   344→   344→    sar = prevSar + af * (ep - prevSar);
   345→   345→    // Ensure SAR is within bounds
   346→   346→    if (isLong) {
   347→   347→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   348→   348→      if (data[i].low < sar) {
   349→   349→        isLong = false;
   350→   350→        sar = ep;
   351→   351→        ep = data[i].low;
   352→   352→        af = step;
   353→   353→      } else {
   354→   354→        if (data[i].high > ep) {
   355→   355→          ep = data[i].high;
   356→   356→          af = Math.min(af + step, max);
   357→   357→        }
   358→   358→      }
   359→   359→    } else {
   360→   360→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   361→   361→      if (data[i].high > sar) {
   362→   362→        isLong = true;
   363→   363→        sar = ep;
   364→   364→        ep = data[i].high;
   365→   365→        af = step;
   366→   366→      } else {
   367→   367→        if (data[i].low < ep) {
   368→   368→          ep = data[i].low;
   369→   369→          af = Math.min(af + step, max);
   370→   370→        }
   371→   371→      }
   372→   372→    }
   373→   373→  }
   374→   374→  return sar;
   375→   375→}

   378→   378→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   379→   379→  if (closes.length < period) {
   380→   380→    const c = closes[closes.length - 1] ?? 0;
   381→   381→    return { upper: c, middle: c, lower: c };
   382→   382→  }
   383→   383→  const slice = closes.slice(-period);
   384→   384→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   385→   385→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   386→   386→  const sd = Math.sqrt(variance);
   387→   387→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   388→   388→}

   391→   391→function calcOBV(data: OHLCV[]): number {
   392→   392→  let obv = 0;
   393→   393→  for (let i = 1; i < data.length; i++) {
   394→   394→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   395→   395→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   396→   396→  }
   397→   397→  return obv;
   398→   398→}

   400→   400→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   401→   401→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   402→   402→  const n = values.length;
   403→   403→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   404→   404→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   405→   405→  for (let i = 0; i < n; i++) {
   406→   406→    sumX += i;
   407→   407→    sumY += values[i];
   408→   408→    sumXY += i * values[i];
   409→   409→    sumX2 += i * i;
   410→   410→    sumY2 += values[i] * values[i];
   411→   411→  }
   412→   412→  const denom = n * sumX2 - sumX * sumX;
   413→   413→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   414→   414→  const slope = (n * sumXY - sumX * sumY) / denom;
   415→   415→  const intercept = (sumY - slope * sumX) / n;
   416→   416→  const meanY = sumY / n;
   417→   417→  let ssRes = 0, ssTot = 0;
   418→   418→  for (let i = 0; i < n; i++) {
   419→   419→    const predicted = slope * i + intercept;
   420→   420→    ssRes += (values[i] - predicted) ** 2;
   421→   421→    ssTot += (values[i] - meanY) ** 2;
   422→   422→  }
   423→   423→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   424→   424→  return { slope, intercept, r2 };
   425→   425→}

   427→   427→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   428→   428→function roundToNice(price: number): number {
   429→   429→  const abs = Math.abs(price);
   430→   430→  let step: number;
   431→   431→  if (abs >= 100000) step = 100;
   432→   432→  else if (abs >= 10000) step = 50;
   433→   433→  else if (abs >= 1000) step = 10;
   434→   434→  else if (abs >= 100) step = 5;
   435→   435→  else step = 1;
   436→   436→  return Math.round(price / step) * step;
   437→   437→}

   481→   481→function calcTrend(closes: number[], period: number): TrendResult {
   482→   482→  const slice = closes.slice(-Math.min(period, closes.length));
   483→   483→  const { slope, r2 } = linearRegression(slice);
   484→   484→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   485→   485→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   486→   486→  const priceThreshold = avgPrice * 0.001; // 0.1%
   487→   487→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   488→   488→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   489→   489→}
