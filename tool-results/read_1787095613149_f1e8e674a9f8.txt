   144→   144→// — S/R strength-based line styling (solid only, thickness = strength) ————
   145→   145→function srLineStyle(strength: number, isTarget: boolean): { lineWidth: number; lineStyle: 0; color: string } {
   146→   146→  if (isTarget) {
   147→   147→    return { lineWidth: 4, lineStyle: 0, color: '' };
   148→   148→  }
   149→   149→  if (strength >= 9) return { lineWidth: 5, lineStyle: 0, color: '' };
   150→   150→  if (strength >= 7) return { lineWidth: 4, lineStyle: 0, color: '' };
   151→   151→  if (strength >= 5) return { lineWidth: 3, lineStyle: 0, color: '' };
   152→   152→  if (strength >= 3) return { lineWidth: 2, lineStyle: 0, color: '' };
   153→   153→  return { lineWidth: 1, lineStyle: 0, color: '' };
   154→   154→}

   196→   196→export default function CandlestickChart({ data, ta, height = 520 }: CandlestickChartProps) {
   197→   197→  const chartContainerRef = useRef<HTMLDivElement>(null);
   198→   198→  const svgRef = useRef<SVGSVGElement>(null);
   199→   199→  const chartRef = useRef<IChartApi | null>(null);
   200→   200→  const csRef = useRef<any>(null);
   201→   201→  const subscriberRef = useRef<{ dispose: () => void } | null>(null);
   202→   202→  const crosshairSubRef = useRef<{ dispose: () => void } | null>(null);
   203→   203→
   204→   204→  // Crosshair legend state
   205→   205→  const [hoverInfo, setHoverInfo] = useState<{ date: string; o: number; h: number; l: number; c: number; v: number; chg: number } | null>(null);
   206→   206→
   207→   207→  // Build Jalali date strings for all candles (memoized)
   208→   208→  const jalaliDates = useMemo(() => {
   209→   209→    return data.map((d) =>
   210→   210→      isGregorianDate(d.date) ? candleDateToJalali(d.date, 'full') : formatJalaliString(d.date, 'full')
   211→   211→    );
   212→   212→  }, [data]);
   213→   213→
   214→   214→  const [drawings, setDrawings] = useState<Drawing[]>([]);
   215→   215→  const [activeTool, setActiveTool] = useState<ToolType>('cursor');
   216→   216→  const [activeColor, setActiveColor] = useState(PALETTE[0]);
   217→   217→  const [isDrawing, setIsDrawing] = useState(false);
   218→   218→  const [currentDrawing, setCurrentDrawing] = useState<Drawing | null>(null);
   219→   219→  const [textInput, setTextInput] = useState<{ x: number; y: number; drawingId: string } | null>(null);
   220→   220→  const [textValue, setTextValue] = useState('');
   221→   221→  const [renderKey, setRenderKey] = useState(0);
   222→   222→  const [chartSize, setChartSize] = useState({ w: 0, h: 0 });
   223→   223→
   224→   224→  /* - Helper: coordinate conversion -------------- */
   225→   225→  const getChartPoint = useCallback((clientX: number, clientY: number): DrawingPoint | null => {
   226→   226→    const chart = chartRef.current;
   227→   227→    const container = chartContainerRef.current;
   228→   228→    if (!chart || !container) return null;
   229→   229→    const rect = container.getBoundingClientRect();
   230→   230→    const x = clientX - rect.left;
   231→   231→    const y = clientY - rect.top;
   232→   232→    const price = csRef.current?.coordinateToPrice(y) ?? null;
   233→   233→    const time = chart.timeScale().coordinateToTime(x);
   234→   234→    if (price === null || time === null) return null;
   235→   235→    return { time: time as number, price };
   236→   236→  }, []);
   237→   237→
   238→   238→  /* - SVG Rendering --------------------- */
   239→   239→  const renderDrawings = useCallback(() => {
   240→   240→    const chart = chartRef.current;
   241→   241→    const svg = svgRef.current;
   242→   242→    const container = chartContainerRef.current;
   243→   243→    if (!chart || !svg || !container || drawings.length === 0) {
   244→   244→      if (svg) svg.innerHTML = '';
   245→   245→      return;
   246→   246→    }
   247→   247→
   248→   248→    const rect = container.getBoundingClientRect();
   249→   249→    const w = rect.width;
   250→   250→    const h = rect.height;
   251→   251→    if (w === 0 || h === 0) return;
   252→   252→
   253→   253→    let html = '';
   254→   254→
   255→   255→    // Defs for arrow markers
   256→   256→    html += `<defs>`;
   257→   257→    PALETTE.forEach((c) => {
   258→   258→      html += `<marker id="arrow-${c.replace('#','')}" markerWidth="8" markerHeight="6" refX="7" refY="3" orient="auto"><polygon points="0 0, 8 3, 0 6" fill="${c}" /></marker>`;
   259→   259→    });
   260→   260→    html += `</defs>`;
   261→   261→
   262→   262→    const allDrawings = [...drawings];
   263→   263→    if (currentDrawing) allDrawings.push(currentDrawing);
   264→   264→
   265→   265→    for (const d of allDrawings) {
   266→   266→      if (!d.completed && d.points.length === 0) continue;
   267→   267→      const col = d.color;
   268→   268→      const lw = d.lineWidth;
   269→   269→
   270→   270→      switch (d.type) {
   271→   271→        case 'trendline': {
   272→   272→          if (d.points.length < 1) break;
   273→   273→          const p0 = d.points[0];
   274→   274→          const p1 = d.points.length > 1 ? d.points[d.points.length - 1] : p0;
   275→   275→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   276→   276→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   277→   277→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   278→   278→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   279→   279→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   280→   280→          html += `<line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" stroke="${col}" stroke-width="${lw}" />`;
   281→   281→          break;
   282→   282→        }
   283→   283→        case 'hline': {
   284→   284→          if (d.points.length < 1) break;
   285→   285→          const p = d.points[0];
   286→   286→          const y = csRef.current?.priceToCoordinate(p.price);
   287→   287→          if (y == null) break;
   288→   288→          html += `<line x1="0" y1="${y}" x2="${w}" y2="${y}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   289→   289→          html += `<text x="4" y="${y - 4}" fill="${col}" font-size="11" font-family="Vazirmatn, sans-serif">${toPersianDigits(p.price.toFixed(0))}</text>`;
   290→   290→          break;
   291→   291→        }
   292→   292→        case 'vline': {
   293→   293→          if (d.points.length < 1) break;
   294→   294→          const p = d.points[0];
   295→   295→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   296→   296→          if (x == null) break;
   297→   297→          html += `<line x1="${x}" y1="0" x2="${x}" y2="${h}" stroke="${col}" stroke-width="${lw}" stroke-dasharray="6,3" />`;
   298→   298→          break;
   299→   299→        }
   300→   300→        case 'fibonacci': {
   301→   301→          if (d.points.length < 2) break;
   302→   302→          const p0 = d.points[0];
   303→   303→          const p1 = d.points[1];
   304→   304→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   305→   305→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   306→   306→          if (x0 == null || x1 == null) break;
   307→   307→          const priceHigh = Math.max(p0.price, p1.price);
   308→   308→          const priceLow = Math.min(p0.price, p1.price);
   309→   309→          const priceRange = priceHigh - priceLow;
   310→   310→          if (priceRange === 0) break;
   311→   311→
   312→   312→          const yTop = csRef.current?.priceToCoordinate(priceHigh);
   313→   313→          const yBottom = csRef.current?.priceToCoordinate(priceLow);
   314→   314→          if (yTop == null || yBottom == null) break;
   315→   315→          html += `<rect x="${Math.min(x0,x1)}" y="${yTop}" width="${Math.abs(x1-x0)}" height="${yBottom - yTop}" fill="${col}" opacity="0.04" />`;
   316→   316→
   317→   317→          for (const level of FIB_LEVELS) {
   318→   318→            const price = priceHigh - level.pct * priceRange;
   319→   319→            const y = csRef.current?.priceToCoordinate(price);
   320→   320→            if (y == null) continue;
   321→   321→            html += `<line x1="${Math.min(x0,x1) - 10}" y1="${y}" x2="${Math.max(x0,x1) + 10}" y2="${y}" stroke="${col}" stroke-width="1" opacity="0.6" />`;
   322→   322→            html += `<rect x="${Math.max(x0,x1) + 12}" y="${y - 8}" width="56" height="16" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="2" />`;
   323→   323→            html += `<text x="${Math.max(x0,x1) + 14}" y="${y + 4}" fill="${col}" font-size="10" font-family="Vazirmatn, sans-serif">${toPersianDigits(level.label)} ${toPersianDigits(price.toFixed(0))}</text>`;
   324→   324→          }
   325→   325→          break;
   326→   326→        }
   327→   327→        case 'rectangle': {
   328→   328→          if (d.points.length < 2) break;
   329→   329→          const p0 = d.points[0];
   330→   330→          const p1 = d.points[d.points.length - 1];
   331→   331→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   332→   332→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   333→   333→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   334→   334→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   335→   335→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
   336→   336→          const rx = Math.min(x0, x1);
   337→   337→          const ry = Math.min(y0, y1);
   338→   338→          const rw = Math.abs(x1 - x0);
   339→   339→          const rh = Math.abs(y1 - y0);
   340→   340→          html += `<rect x="${rx}" y="${ry}" width="${rw}" height="${rh}" fill="${col}" fill-opacity="0.08" stroke="${col}" stroke-width="${lw}" rx="2" />`;
   341→   341→          break;
   342→   342→        }
   343→   343→        case 'text': {
   344→   344→          if (d.points.length < 1 || !d.text) break;
   345→   345→          const p = d.points[0];
   346→   346→          const x = chart.timeScale().timeToCoordinate(p.time as Time);
   347→   347→          const y = csRef.current?.priceToCoordinate(p.price);
   348→   348→          if (x == null || y == null) break;
   349→   349→          html += `<rect x="${x + 4}" y="${y - 16}" width="${d.text.length * 8 + 12}" height="22" fill="#ffffff" stroke="#e5e7eb" stroke-width="1" rx="3" />`;
   350→   350→          html += `<text x="${x + 10}" y="${y + 1}" fill="${col}" font-size="12" font-family="Vazirmatn, sans-serif">${d.text}</text>`;
   351→   351→          break;
   352→   352→        }
   353→   353→        case 'brush': {
   354→   354→          if (d.points.length < 2) break;
   355→   355→          let pts = '';
   356→   356→          let valid = true;
   357→   357→          for (const p of d.points) {
   358→   358→            const x = chart.timeScale().timeToCoordinate(p.time as Time);
   359→   359→            const y = csRef.current?.priceToCoordinate(p.price);
   360→   360→            if (x == null || y == null) { valid = false; break; }
   361→   361→            pts += `${x},${y} `;
   362→   362→          }
   363→   363→          if (!valid || pts.length === 0) break;
   364→   364→          html += `<polyline points="${pts.trim()}" fill="none" stroke="${col}" stroke-width="${lw}" stroke-linecap="round" stroke-linejoin="round" />`;
   365→   365→          break;
   366→   366→        }
   367→   367→        case 'measure': {
   368→   368→          if (d.points.length < 2) break;
   369→   369→          const p0 = d.points[0];
   370→   370→          const p1 = d.points[d.points.length - 1];
   371→   371→          const x0 = chart.timeScale().timeToCoordinate(p0.time as Time);
   372→   372→          const y0 = csRef.current?.priceToCoordinate(p0.price);
   373→   373→          const x1 = chart.timeScale().timeToCoordinate(p1.time as Time);
   374→   374→          const y1 = csRef.current?.priceToCoordinate(p1.price);
   375→   375→          if (x0 == null || y0 == null || x1 == null || y1 == null) break;
