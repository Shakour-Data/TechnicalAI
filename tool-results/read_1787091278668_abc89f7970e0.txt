    14→/* --─ Types ---------------------------------------------- */

    16→interface InstrumentItem {
    17→  l18: string;
    18→  l30: string;
    19→  pl: number;
    20→  pcp: number;
    21→  tno: number;
    22→  tvol: number;
    23→  tval: number;
    24→  cs: string;
    25→  category: 'stock' | 'etf' | 'bond' | 'future' | 'salaf' | 'mortgage' | 'index'
    26→    | 'currency' | 'gold' | 'silver' | 'gold_etf'
    27→    | 'crypto' | 'world_index' | 'forex' | 'energy' | 'metal' | 'commodity';
    28→  insCode?: string;
    29→  tgjuKey?: string;
    30→  finpySector?: string;  // finpy-tse sector name for industry indices
    31→  finpyIndex?: string;   // finpy-tse index function key (CWI, EWI, etc.)
    32→  isMainIndex?: boolean; // True for main market indices
    33→  index?: number;
    34→  indexChange?: number;
    35→  indexChangePercent?: number;
    36→  indexMin?: number;
    37→  indexMax?: number;
    38→}

    47→  mortgage: InstrumentItem[];

    49→}

    52→  currencies: InstrumentItem[];
    53→  gold: InstrumentItem[];

    55→  goldEtfs: InstrumentItem[];
    56→  crypto: InstrumentItem[];

    58→  forex: InstrumentItem[];
    59→  energy: InstrumentItem[];

    62→  items: InstrumentItem[];
    63→}

   140→/* --─ Helpers -------------------------------------------- */
   141→
   142→function formatNum(num: number): string {
   143→  if (num == null) return '—';
   144→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 0 }).format(num);
   145→}
   146→
   147→function formatIdx(num: number): string {
   148→  if (num == null) return '—';
   149→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 2 }).format(num);
   150→}

   157→    </svg>
   158→  );
   159→}

   161→/* --─ Component ------------------------------------------ */
   162→
   163→export default function SymbolSearch({
   164→  onSelect,
   165→  placeholder = 'جستجوی ابزار مالی ...',
   166→  className,
   167→}: SymbolSearchProps) {
   168→  const [query, setQuery] = React.useState('');
   169→  const [open, setOpen] = React.useState(false);
   170→  const [results, setResults] = React.useState<InstrumentItem[]>([]);
   171→  const [loading, setLoading] = React.useState(false);
   172→  const [activeIndex, setActiveIndex] = React.useState(-1);
   173→  const [totalMatched, setTotalMatched] = React.useState(0);
   174→  const [activeCategory, setActiveCategory] = React.useState<CategoryKey>('all');
   175→  const [activeIndustry, setActiveIndustry] = React.useState<string | null>(null);
   176→  const [showIndustryPicker, setShowIndustryPicker] = React.useState(false);
   177→
   178→  const cacheRef = React.useRef<InstrumentsData | null>(null);
   179→  const tgjuCacheRef = React.useRef<TgjuData | null>(null);
   180→  const fetchRef = React.useRef<Promise<InstrumentsData> | null>(null);
   181→  const tgjuFetchRef = React.useRef<Promise<TgjuData | null> | null>(null);
   182→  const inputRef = React.useRef<HTMLInputElement>(null);
   183→  const dropdownRef = React.useRef<HTMLDivElement>(null);
   184→  const itemRefs = React.useRef<(HTMLDivElement | null)[]>([]);
   185→  const industryScrollRef = React.useRef<HTMLDivElement>(null);
   186→
   187→  /* -- Data fetching -- */
   188→  const fetchData = React.useCallback(async (): Promise<InstrumentsData> => {
   189→    if (cacheRef.current) return cacheRef.current;
   190→    if (fetchRef.current) return fetchRef.current;
   191→    const p = async () => {
   192→      try {
   193→        const r = await fetch('/api/instruments');
   194→        if (!r.ok) throw new Error(`${r.status}`);
   195→        const d: InstrumentsData = await r.json();
   196→        cacheRef.current = d;
   197→        return d;
   198→      } finally {
   199→        fetchRef.current = null;
   200→      }
   201→    };
   202→    fetchRef.current = p();
   203→    return fetchRef.current;
   204→  }, []);
   205→
   206→  const fetchTgjuData = React.useCallback(async (): Promise<TgjuData | null> => {
   207→    if (tgjuCacheRef.current) return tgjuCacheRef.current;
   208→    if (tgjuFetchRef.current) return tgjuFetchRef.current;
   209→    const p = async () => {
   210→      try {
   211→        const r = await fetch('/api/tgju-instruments');
   212→        if (!r.ok) return null;
   213→        const d: TgjuData = await r.json();
   214→        tgjuCacheRef.current = d;
   215→        return d;
   216→      } catch {
   217→        return null;
   218→      } finally {
   219→        tgjuFetchRef.current = null;
   220→      }
   221→    };
   222→    tgjuFetchRef.current = p();
   223→    return p();
   224→  }, []);
   225→
   226→  /* -- Check if category is TGJU-based -- */
   227→  const isTgjuCategory = (cat: CategoryKey) => TGJU_CATEGORIES.has(cat);
   228→
   229→  /* -- Get all items for current category -- */
   230→  const getItemsForCategory = React.useCallback(async (cat: CategoryKey): Promise<InstrumentItem[]> => {
   231→    if (isTgjuCategory(cat)) {
   232→      const tgju = tgjuCacheRef.current || await fetchTgjuData();
   233→      if (!tgju) return [];
   234→      switch (cat) {
   235→        case 'currency': return tgju.currencies;
   236→        case 'gold': return [...tgju.gold, ...tgju.silver];
   237→        case 'gold_etf': return tgju.goldEtfs;
   238→        case 'silver': return tgju.silver;
   239→        case 'crypto': return tgju.crypto;
   240→        case 'world_index': return tgju.worldIndices;
   241→        case 'forex': return tgju.forex;
   242→        case 'energy': return tgju.energy;
   243→        case 'metal': return tgju.metals;
   244→        case 'commodity': return tgju.commodities;
   245→        default: return [];
   246→      }
   247→    }
   248→
   249→    // TSE categories
   250→    const data = cacheRef.current || await fetchData();
   251→    switch (cat) {
   252→      case 'indices': return data.indices;
   253→      case 'stocks':
   254→        return activeIndustry
   255→          ? data.stocks.filter((s) => s.cs === activeIndustry)
   256→          : data.stocks;
   257→      case 'etf': return data.etfs;
   258→      case 'bond': return [...data.bonds, ...data.mortgage];
   259→      case 'derivative': return [...data.futures, ...data.salaf];
   260→      case 'all':
   261→      default: {
   262→        const tgju = tgjuCacheRef.current;
   263→        return [
   264→          ...data.indices,
   265→          ...data.stocks,
   266→          ...data.etfs,
   267→          ...(tgju?.currencies || []),
   268→          ...(tgju?.forex || []),
   269→          ...(tgju?.crypto || []),
   270→          ...(tgju?.gold || []),
   271→          ...(tgju?.silver || []),
   272→          ...(tgju?.goldEtfs || []),
   273→          ...(tgju?.worldIndices || []),
   274→          ...(tgju?.energy || []),
   275→          ...(tgju?.metals || []),
   276→          ...(tgju?.commodities || []),
   277→          ...data.bonds,
   278→          ...data.futures,
   279→          ...data.salaf,
   280→          ...data.mortgage,
   281→        ];
   282→      }
   283→    }
   284→  }, [activeIndustry, fetchData, fetchTgjuData]);
   285→
   286→  /* -- Popular items -- */
   287→  const getPopular = React.useCallback(async () => {
   288→    const data = cacheRef.current || await fetchData();
   289→    const tgju = tgjuCacheRef.current;
   290→    const all = [
   291→      ...(data.indices.map((i) => ({ ...i, tval: i.index || 0 }))),
   292→      ...data.stocks,
   293→      ...data.etfs,
   294→      ...(tgju?.currencies || []),
   295→      ...(tgju?.crypto?.slice(0, 5) || []),
   296→      ...(tgju?.gold?.slice(0, 3) || []),
   297→    ];
   298→    const sorted = [...all].sort((a, b) => (b.tval || 0) - (a.tval || 0)).slice(0, 15);
   299→    setResults(sorted);
   300→    setTotalMatched(all.length);
   301→  }, [fetchData]);
   302→
   303→  /* -- Filter by search query -- */
   304→  const doFilter = React.useCallback((items: InstrumentItem[], q: string) => {
   305→    const n = q.trim().toLowerCase();
   306→    const f = items.filter(
   307→      (s) =>
   308→        s.l18?.toLowerCase().includes(n) ||
   309→        s.l30?.toLowerCase().includes(n) ||
   310→        s.cs?.toLowerCase().includes(n),
   311→    );
   312→    setTotalMatched(f.length);
   313→    setResults(f.slice(0, MAX_RESULTS));
   314→    setActiveIndex(-1);
   315→  }, []);
   316→
   317→  /* -- Show dropdown with data -- */
   318→  const showDropdown = React.useCallback(async (q: string) => {
   319→    setOpen(true);
   320→    await Promise.all([fetchData(), fetchTgjuData()]);
   321→    const items = await getItemsForCategory(activeCategory);
   322→    if (q.trim().length === 0) {
   323→      if (activeCategory === 'all') {
   324→        await getPopular();
   325→      } else {
   326→        setResults(items.slice(0, MAX_RESULTS));
   327→        setTotalMatched(items.length);
   328→      }
   329→    } else {
   330→      doFilter(items, q);
   331→    }
   332→    setLoading(false);
   333→  }, [fetchData, fetchTgjuData, getItemsForCategory, activeCategory, getPopular, doFilter]);
   334→
   335→  /* -- Category change -- */
   336→  const handleCategoryChange = React.useCallback(async (cat: CategoryKey) => {
   337→    setActiveCategory(cat);
   338→    if (cat !== 'stocks') setActiveIndustry(null);
   339→    setShowIndustryPicker(false);
   340→
   341→    const items = await getItemsForCategory(cat);
   342→    if (query.trim().length > 0) {
   343→      doFilter(items, query);
   344→    } else if (cat === 'all') {
   345→      await getPopular();
   346→    } else {
   347→      setResults(items.slice(0, MAX_RESULTS));
   348→      setTotalMatched(items.length);
   349→    }
   350→    setActiveIndex(-1);
   351→  }, [getItemsForCategory, query, doFilter, getPopular]);
   352→
   353→  /* -- Industry change -- */
   354→  const handleIndustryChange = React.useCallback((industry: string | null) => {
   355→    setActiveIndustry(industry);
   356→    setShowIndustryPicker(false);
   357→    if (cacheRef.current) {
   358→      const items = industry
   359→        ? cacheRef.current.stocks.filter((s) => s.cs === industry)
   360→        : cacheRef.current.stocks;
   361→      if (query.trim().length > 0) {
   362→        doFilter(items, query);
   363→      } else {
   364→        setResults(items.slice(0, MAX_RESULTS));
   365→        setTotalMatched(items.length);
   366→      }
   367→      setActiveIndex(-1);
   368→    }
   369→  }, [query, doFilter]);
   370→
   371→  const handleChange = React.useCallback((value: string) => {
   372→    setQuery(value);
   373→    getItemsForCategory(activeCategory).then((items) => {
   374→      if (value.trim().length === 0) {
   375→        if (activeCategory === 'all') getPopular();
   376→        else {
   377→          setResults(items.slice(0, MAX_RESULTS));
   378→          setTotalMatched(items.length);
   379→        }
   380→      } else {
   381→        doFilter(items, value);
   382→      }
   383→    });
   384→  }, [activeCategory, getItemsForCategory, getPopular, doFilter]);
   385→
   386→  const handleFocus = React.useCallback(() => {
   387→    setLoading(true);
   388→    showDropdown(query);
   389→  }, [query, showDropdown]);
   390→
   391→  const selectSymbol = React.useCallback((s: InstrumentItem) => {
   392→    setQuery(s.l18);
   393→    setOpen(false);
   394→    onSelect?.(s.l18, s.category, s.insCode, s.tgjuKey, s.finpySector, s.finpyIndex);
   395→    inputRef.current?.blur();
   396→  }, [onSelect]);
   397→
   398→  const handleKeyDown = React.useCallback((e: React.KeyboardEvent) => {
   399→    if (!open || results.length === 0) {
   400→      if (e.key === 'ArrowDown' || e.key === 'Enter') { e.preventDefault(); handleFocus(); }
   401→      return;
   402→    }
   403→    if (e.key === 'ArrowDown') { e.preventDefault(); setActiveIndex((p) => p < results.length - 1 ? p + 1 : 0); }
   404→    else if (e.key === 'ArrowUp') { e.preventDefault(); setActiveIndex((p) => p > 0 ? p - 1 : results.length - 1); }
   405→    else if (e.key === 'Enter') { e.preventDefault(); if (activeIndex >= 0) selectSymbol(results[activeIndex]); }
   406→    else if (e.key === 'Escape') { e.preventDefault(); setOpen(false); inputRef.current?.blur(); }
   407→  }, [open, results, activeIndex, selectSymbol, handleFocus]);
   408→
   409→  React.useEffect(() => {
   410→    if (activeIndex >= 0) itemRefs.current[activeIndex]?.scrollIntoView({ block: 'nearest' });
   411→  }, [activeIndex]);
   412→
   413→  React.useEffect(() => {
   414→    if (!open) return;
   415→    const handler = (e: MouseEvent) => {
   416→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node) &&
   417→          inputRef.current && !inputRef.current.contains(e.target as Node)) {
   418→        setOpen(false);
   419→        setShowIndustryPicker(false);
   420→      }
   421→    };
   422→    document.addEventListener('mousedown', handler);
   423→    return () => document.removeEventListener('mousedown', handler);
   424→  }, [open]);
   425→
   426→  const clear = () => { setQuery(''); setActiveIndustry(null); setActiveCategory('all'); setResults([]); inputRef.current?.focus(); };
   427→  const isPopular = query.trim().length === 0 && activeCategory === 'all';
   428→  const isSearch = query.trim().length > 0;
   429→
   430→  /* -- Render instrument price -- */
   431→  const renderPrice = (item: InstrumentItem) => {
   432→    const isTgju = isTgjuCategory(item.category as CategoryKey);
   433→    if (item.category === 'index') {
   434→      return (
   435→        <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   436→          <span className='text-xs font-semibold text-[#111827]'>{formatIdx(item.pl)}</span>
   437→          <span className={cn(
   438→            'text-[11px] font-bold px-1.5 py-0.5 rounded',
   439→            item.pcp > 0 ? 'bg-emerald-50 text-emerald-700' :
   440→            item.pcp < 0 ? 'bg-red-50 text-red-700' : 'text-[#6b7280]',
   441→          )}>
   442→            {item.pcp > 0 ? '+' : ''}{item.pcp?.toFixed(2)}%
   443→          </span>
   444→        </div>
   445→      );
   446→    }
   447→    // For static TGJU items with no live price, show dash
   448→    if (isTgju && item.pl === 0) {
   449→      return (
   450→        <div className='flex shrink-0 flex-col items-end gap-0.5'>
   451→          <span className='text-xs text-[#6b7280]'>—</span>
   452→        </div>
   453→      );
   454→    }
   455→    const isUp = item.pcp > 0;
   456→    const isDown = item.pcp < 0;
   457→    return (
   458→      <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   459→        <span className='text-xs font-semibold text-[#111827]'>{formatNum(item.pl)}</span>
   460→        <span className={cn(
   461→          'text-[11px] font-bold px-1.5 py-0.5 rounded',
   462→          isUp ? 'bg-emerald-50 text-emerald-700' :
   463→          isDown ? 'bg-red-50 text-red-700' : 'text-[#6b7280]',
   464→        )}>
   465→          {isUp ? '+' : ''}{item.pcp?.toFixed(2)}%
   466→        </span>
   467→      </div>
   468→    );
   469→  };
   470→
   471→  const industries = cacheRef.current?.industries || [];
   472→
   473→  return (
   474→    <div dir='rtl' className={cn('relative', className)}>
   475→
   476→      <div className='relative'>
   477→        <Search className='absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-[#6b7280] pointer-events-none' />
   478→        <Input
   479→          ref={inputRef}
   480→          type='text'
   481→          value={query}
   482→          onChange={(e) => handleChange(e.target.value)}
   483→          onFocus={handleFocus}
   484→          onKeyDown={handleKeyDown}
   485→          placeholder={placeholder}
   486→          suppressHydrationWarning
   487→          className={cn(
   488→            'h-10 w-full rounded-lg border border-[#e5e7eb] bg-white',
   489→            'text-sm text-[#111827] placeholder:text-[#6b7280]',
   490→            'transition-all',
   491→            'focus-visible:border-amber-500/60 focus-visible:ring-amber-500/20 focus-visible:ring-[3px]',
   492→            'pr-9',
   493→          )}
   494→          autoComplete='off' spellCheck={false}
   495→        />
   496→        {query && (
   497→          <button onClick={clear} className='absolute left-3 top-1/2 -translate-y-1/2 text-[#6b7280] hover:text-[#111827] transition-colors' type='button'>
   498→            <X className='w-4 h-4' />
   499→          </button>
   500→        )}
   501→      </div>
   502→
   503→
   504→      {open && (
