    15→// === Helper Functions =========================================================
    16→
    17→function clamp(v: number, min: number, max: number): number {
    18→  return Math.max(min, Math.min(max, v));
    19→}
    20→
    21→function normalize(arr: number[]): number[] {
    22→  const sum = arr.reduce((a, b) => a + b, 0);
    23→  if (sum === 0) return arr.map(() => 1 / arr.length);
    24→  return arr.map((v) => v / sum);
    25→}
    26→
    27→function sigmoid(x: number): number {
    28→  return 1 / (1 + Math.exp(-x));
    29→}

   996→// === Probability Context Builder ==============================================
   997→
   998→function buildContext(input: GraphInput): ProbContext {
   999→  const adxNorm = clamp(input.adx / 60, 0, 1);
  1000→  const srProximity = clamp(
  1001→    1 - Math.min(input.distToR1, input.distToS1) * 10,
  1002→    0,
  1003→    1
  1004→  );
  1005→  const oscillatorExtreme = clamp(
  1006→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
  1007→    0,
  1008→    1
  1009→  );
  1010→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1011→
  1012→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1013→  const overboughtRisk =
  1014→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1015→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1016→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1017→
  1018→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1019→  const oversoldBounce =
  1020→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1021→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1022→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1023→
  1024→  return {
  1025→    input,
  1026→    adxNorm,
  1027→    srProximity,
  1028→    oscillatorExtreme,
  1029→    divergenceProxy,
  1030→    overboughtRisk,
  1031→    oversoldBounce,
  1032→  };
  1033→}

  1035→// === Node Probability Computation =============================================
  1036→// For each non-terminal node, returns an array of probabilities for its
  1037→// outgoing edges. The order MUST match the order of edges in createEdges().
  1038→// =============================================================================
  1039→
  1040→function computeNodeProbabilities(
  1041→  nodeId: string,
  1042→  ctx: ProbContext,
  1043→  _numChildren: number
  1044→): number[] {
  1045→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1046→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1047→
  1048→  switch (nodeId) {
  1049→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1050→    case 'ROOT': {
  1051→      const trendW = adxNorm * 0.45 + input.mlTrend * 0.15 + 0.12;
  1052→      const breakW = srProximity * 0.35 + (1 - adxNorm) * 0.1 + 0.1;
  1053→      const revW = ctx.divergenceProxy * 0.3 + (1 - adxNorm) * 0.15 + 0.08;
  1054→      return normalize([trendW, breakW, revW]);
  1055→    }
  1056→
  1057→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1058→    case 'N_TREND': {
  1059→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1060→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1061→      const flat =
  1062→        (1 - Math.abs(input.bullConsensus - 0.5) * 2) * (1 - adxNorm * 0.7);
  1063→      return normalize([bull, bear, flat]);
  1064→    }
  1065→
  1066→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1067→    case 'N_T_BULL': {
  1068→      const momHigh =
  1069→        clamp(
  1070→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1071→            (input.macdHist > 0
  1072→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1073→              : 0),
  1074→          0,
  1075→          1
  1076→        ) * input.mlMomentum;
  1077→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1078→      return normalize([momHigh, momMod, overboughtRisk]);
  1079→    }
  1080→
  1081→    // ── N_T_B_MOM_HIGH → [R8, R9, R7] ────────────────────────────────────
  1082→    // Strong bullish momentum — all three outcomes are bullish (no range)
  1083→    case 'N_T_B_MOM_HIGH': {
  1084→      const pR8 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1085→      const pR9 = 0.15 + input.strengthIndex * 0.2;
  1086→      const pR7 = 0.35 * input.confidenceIndex + 0.15;
  1087→      return normalize([pR8, pR9, pR7]);
  1088→    }
  1089→
  1090→    // ── N_T_B_MOM_MOD → [R7, R6, R5] ─────────────────────────────────────
  1091→    // Moderate bullish — small explicit R5 based on uncertainty
  1092→    case 'N_T_B_MOM_MOD': {
  1093→      const pR7 = 0.5 * input.confidenceIndex + 0.2;
  1094→      const pR6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1095→      const pR5 = clamp((1 - adxNorm) * 0.35, 0.05, 0.25);
  1096→      return normalize([pR7, pR6, pR5]);
  1097→    }
  1098→
  1099→    // ── N_T_B_OVERBOUGHT → [R4, R1, R9] ──────────────────────────────────
  1100→    // Overbought in uptrend — correction or last-gasp surge (no range)
  1101→    case 'N_T_B_OVERBOUGHT': {
  1102→      const pR4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1103→      const pR1 = 0.15 + input.strengthIndex * 0.15;
  1104→      const pR9 = 0.1 + input.mlMomentum * 0.2;
  1105→      return normalize([pR4, pR1, pR9]);
  1106→    }
  1107→
  1108→    // ── N_T_BEAR → [N_T_BE_MOM_LOW, N_T_BE_MOM_MOD, N_T_BE_OVERSOLD] ────
  1109→    case 'N_T_BEAR': {
  1110→      const momLow =
  1111→        clamp(
  1112→          (input.momentum < 0 ? Math.abs(input.momentum) / 50 : 0) +
  1113→            (input.macdHist < 0
  1114→              ? (Math.abs(input.macdHist) / (atrSafe * 0.5)) * 0.3
  1115→              : 0),
  1116→          0,
  1117→          1
  1118→        ) * input.mlMomentum;
  1119→      const momMod = (1 - momLow) * 0.6 * input.mlMomentum;
  1120→      return normalize([momLow, momMod, oversoldBounce]);
  1121→    }
  1122→
  1123→    // ── N_T_BE_MOM_LOW → [R2, R1, R3] ────────────────────────────────────
  1124→    // Strong bearish momentum — all three outcomes are bearish (no range)
  1125→    case 'N_T_BE_MOM_LOW': {
  1126→      const pR2 = 0.45 * input.mlTrend + input.confidenceIndex * 0.2;
  1127→      const pR1 = 0.15 + input.strengthIndex * 0.2;
  1128→      const pR3 = 0.35 * input.confidenceIndex + 0.15;
  1129→      return normalize([pR2, pR1, pR3]);
  1130→    }
  1131→
  1132→    // ── N_T_BE_MOM_MOD → [R3, R4, R5] ────────────────────────────────────
  1133→    // Moderate bearish — small explicit R5 based on uncertainty
  1134→    case 'N_T_BE_MOM_MOD': {
  1135→      const pR3 = 0.5 * input.confidenceIndex + 0.2;
  1136→      const pR4 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1137→      const pR5 = clamp((1 - adxNorm) * 0.35, 0.05, 0.25);
  1138→      return normalize([pR3, pR4, pR5]);
  1139→    }
  1140→
  1141→    // ── N_T_BE_OVERSOLD → [R6, R9, R4] ───────────────────────────────────
  1142→    // Oversold in downtrend — bounce or mild continuation (no range)
  1143→    case 'N_T_BE_OVERSOLD': {
  1144→      const pR6 = 0.55 + (1 - input.mlMomentum) * 0.2;
  1145→      const pR9 = 0.15 + input.strengthIndex * 0.15;
  1146→      const pR4 = 0.1 + input.mlMomentum * 0.2;
  1147→      return normalize([pR6, pR9, pR4]);
  1148→    }
  1149→
  1150→    // ── N_T_FLAT → [N_T_F_VOL_LOW, N_T_F_VOL_HIGH, N_T_F_VOL_MOD] ────────
  1151→    case 'N_T_FLAT': {
  1152→      const volLow =
  1153→        clamp(1 - (input.atr / input.price) * 20, 0, 1) *
  1154→        (1 - input.mlVolatility) *
  1155→        0.7;
  1156→      const volHigh =
  1157→        clamp((input.atr / input.price) * 20, 0, 1) * input.mlVolatility * 0.7;
  1158→      const volMod = (1 - volLow - volHigh) * 0.5 + 0.1;
  1159→      return normalize([volLow, volHigh, volMod]);
  1160→    }
  1161→
  1162→    // ── N_T_F_VOL_LOW → [R5] ─────────────────────────────────────────────
  1163→    case 'N_T_F_VOL_LOW': {
  1164→      return [1.0];
  1165→    }
  1166→
  1167→    // ── N_T_F_VOL_HIGH → [R9, R1, R5] ────────────────────────────────────
  1168→    // High volatility shock — directional shock more likely than range
  1169→    case 'N_T_F_VOL_HIGH': {
  1170→      const pR9 = 0.35 * input.bullConsensus + 0.15;
  1171→      const pR1 = 0.35 * (1 - input.bullConsensus) + 0.15;
  1172→      const pR5 = clamp((1 - input.mlVolatility) * 0.2, 0.05, 0.2);
  1173→      return normalize([pR9, pR1, pR5]);
  1174→    }
  1175→
  1176→    // ── N_T_F_VOL_MOD → [R6, R4, R5] ─────────────────────────────────────
  1177→    // Moderate volatility — range is plausible but capped
  1178→    case 'N_T_F_VOL_MOD': {
  1179→      const pR6 = 0.35 * input.bullConsensus + 0.12;
  1180→      const pR4 = 0.35 * (1 - input.bullConsensus) + 0.12;
  1181→      const pR5 = clamp((1 - adxNorm) * 0.3, 0.08, 0.25);
  1182→      return normalize([pR6, pR4, pR5]);
  1183→    }
  1184→
  1185→    // ── N_BREAK → [N_B_UP, N_B_DOWN, N_B_NONE] ────────────────────────────
  1186→    case 'N_BREAK': {
  1187→      const upProb = input.bullConsensus * srProximity * 0.7 + 0.05;
  1188→      const downProb = (1 - input.bullConsensus) * srProximity * 0.7 + 0.05;
  1189→      const noneProb = (1 - srProximity) * 0.5 + 0.1;
  1190→      return normalize([upProb, downProb, noneProb]);
  1191→    }
  1192→
  1193→    // ── N_B_UP → [N_B_U_VOL_C, N_B_U_VOL_W] ─────────────────────────────
  1194→    case 'N_B_UP': {
  1195→      let volConfirm: number;
  1196→      let volWeak: number;
  1197→      if (input.hasVolume) {
  1198→        volConfirm = clamp(input.mfi / 60, 0, 1) * 0.6 + 0.2;
  1199→        volWeak = 1 - volConfirm;
  1200→      } else {
  1201→        volConfirm = 0.4;
  1202→        volWeak = 0.6;
  1203→      }
  1204→      return normalize([volConfirm, volWeak]);
  1205→    }
  1206→
  1207→    // ── N_B_U_VOL_C → [R7, R8, R9] ──────────────────────────────────────
  1208→    // Confirmed bullish breakout — all bullish, no range
  1209→    case 'N_B_U_VOL_C': {
  1210→      const pR7 = 0.45 + input.confidenceIndex * 0.15;
  1211→      const pR8 = 0.2 + input.strengthIndex * 0.15;
  1212→      const pR9 = 0.15 + input.strengthIndex * 0.1;
  1213→      return normalize([pR7, pR8, pR9]);
  1214→    }
  1215→
  1216→    // ── N_B_U_VOL_W → [R6, R4, R5] ──────────────────────────────────────
  1217→    // Weak volume breakout — fake breakout risk, capped R5
  1218→    case 'N_B_U_VOL_W': {
  1219→      const pR6 = 0.4 + input.bullConsensus * 0.15;
  1220→      const pR4 = 0.3 + (1 - input.bullConsensus) * 0.15;
  1221→      const pR5 = clamp((1 - input.confidenceIndex) * 0.25, 0.05, 0.2);
  1222→      return normalize([pR6, pR4, pR5]);
  1223→    }
  1224→
  1225→    // ── N_B_DOWN → [N_B_D_VOL_C, N_B_D_VOL_W] ────────────────────────────
  1226→    case 'N_B_DOWN': {
  1227→      let volConfirm: number;
  1228→      if (input.hasVolume) {
  1229→        volConfirm = clamp((100 - input.mfi) / 60, 0, 1) * 0.6 + 0.2;
  1230→      } else {
  1231→        volConfirm = 0.4;
  1232→      }
  1233→      const volWeak = 1 - volConfirm;
  1234→      return normalize([volConfirm, volWeak]);
  1235→    }
  1236→
  1237→    // ── N_B_D_VOL_C → [R3, R2, R1] ──────────────────────────────────────
  1238→    // Confirmed bearish breakout — all bearish, no range
  1239→    case 'N_B_D_VOL_C': {
  1240→      const pR3 = 0.45 + input.confidenceIndex * 0.15;
  1241→      const pR2 = 0.2 + input.strengthIndex * 0.15;
  1242→      const pR1 = 0.15 + input.strengthIndex * 0.1;
  1243→      return normalize([pR3, pR2, pR1]);
  1244→    }
  1245→
  1246→    // ── N_B_D_VOL_W → [R4, R6, R5] ──────────────────────────────────────
  1247→    // Weak volume bearish breakout — fake breakout risk, capped R5
  1248→    case 'N_B_D_VOL_W': {
  1249→      const pR4 = 0.4 + (1 - input.bullConsensus) * 0.15;
  1250→      const pR6 = 0.3 + input.bullConsensus * 0.15;
  1251→      const pR5 = clamp((1 - input.confidenceIndex) * 0.25, 0.05, 0.2);
  1252→      return normalize([pR4, pR6, pR5]);
  1253→    }
  1254→
  1255→    // ── N_B_NONE → [R5] ──────────────────────────────────────────────────
  1256→    case 'N_B_NONE': {
  1257→      return [1.0];
  1258→    }
  1259→
  1260→    // ── N_REVERSAL → [N_R_BULL, N_R_BEAR, N_R_NONE] ──────────────────────
  1261→    case 'N_REVERSAL': {
  1262→      const bullRev = overboughtRisk;
  1263→      const bearRev = oversoldBounce;
  1264→      const noneRev =
  1265→        1 - Math.max(overboughtRisk, oversoldBounce) * 0.7;
  1266→      return normalize([bullRev, bearRev, noneRev]);
  1267→    }
  1268→
  1269→    // ── N_R_BULL → [N_R_B_DIV, N_R_B_CANDLE, N_R_B_SR] ──────────────────
  1270→    case 'N_R_BULL': {
  1271→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1272→      const candleProb =
  1273→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1274→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1275→      return normalize([divProb, candleProb, srProb]);
  1276→    }
  1277→
  1278→    // ── N_R_B_DIV → [R6, R8, R5] ────────────────────────────────────────
  1279→    // Bullish divergence — explicit small R5
  1280→    case 'N_R_B_DIV': {
  1281→      const pR6 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1282→      const pR8 = 0.2 + input.mlMomentum * 0.15;
  1283→      const pR5 = clamp((1 - adxNorm) * 0.25, 0.05, 0.2);
  1284→      return normalize([pR6, pR8, pR5]);
  1285→    }
  1286→
  1287→    // ── N_R_B_CANDLE → [R7, R6, R5] ─────────────────────────────────────
  1288→    // Bullish candle pattern — explicit small R5
  1289→    case 'N_R_B_CANDLE': {
  1290→      const pR7 = 0.5 * input.confidenceIndex + 0.2;
  1291→      const pR6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1292→      const pR5 = clamp((1 - input.confidenceIndex) * 0.2, 0.05, 0.18);
  1293→      return normalize([pR7, pR6, pR5]);
  1294→    }
  1295→
  1296→    // ── N_R_B_SR → [R6, R5, R4] ─────────────────────────────────────────
  1297→    // Support bounce — explicit small R5
  1298→    // Edge order: R6, R5, R4 — probabilities computed then reordered
  1299→    case 'N_R_B_SR': {
  1300→      const pR6 = 0.5 + input.srAvgStrength * 0.2;
  1301→      const pR4 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1302→      const pR5 = clamp((1 - input.srAvgStrength) * 0.2, 0.05, 0.18);
  1303→      // Return in edge order: [R6, R5, R4]
  1304→      return normalize([pR6, pR5, pR4]);
  1305→    }
  1306→
  1307→    // ── N_R_BEAR → [N_R_BE_DIV, N_R_BE_CANDLE, N_R_BE_SR] ───────────────
  1308→    // Mirror of N_R_BULL
  1309→    case 'N_R_BEAR': {
  1310→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1311→      const candleProb =
  1312→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1313→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1314→      return normalize([divProb, candleProb, srProb]);
  1315→    }
  1316→
  1317→    // ── N_R_BE_DIV → [R4, R2, R5] ───────────────────────────────────────
  1318→    // Bearish divergence — explicit small R5
  1319→    case 'N_R_BE_DIV': {
  1320→      const pR4 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1321→      const pR2 = 0.2 + input.mlMomentum * 0.15;
  1322→      const pR5 = clamp((1 - adxNorm) * 0.25, 0.05, 0.2);
  1323→      return normalize([pR4, pR2, pR5]);
  1324→    }
  1325→
  1326→    // ── N_R_BE_CANDLE → [R3, R4, R5] ────────────────────────────────────
  1327→    // Bearish candle pattern — explicit small R5
  1328→    case 'N_R_BE_CANDLE': {
  1329→      const pR3 = 0.5 * input.confidenceIndex + 0.2;
  1330→      const pR4 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1331→      const pR5 = clamp((1 - input.confidenceIndex) * 0.2, 0.05, 0.18);
  1332→      return normalize([pR3, pR4, pR5]);
  1333→    }
  1334→
  1335→    // ── N_R_BE_SR → [R4, R5, R6] ─────────────────────────────────────────
  1336→    // Resistance rejection — explicit small R5
  1337→    // Edge order: R4, R5, R6 — probabilities computed then reordered
  1338→    case 'N_R_BE_SR': {
  1339→      const pR4 = 0.5 + input.srAvgStrength * 0.2;
  1340→      const pR6 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1341→      const pR5 = clamp((1 - input.srAvgStrength) * 0.2, 0.05, 0.18);
  1342→      // Return in edge order: [R4, R5, R6]
  1343→      return normalize([pR4, pR5, pR6]);
  1344→    }
  1345→
  1346→    // ── N_R_NONE → [R5] ──────────────────────────────────────────────────
  1347→    case 'N_R_NONE': {
  1348→      return [1.0];
  1349→    }
  1350→
  1351→    // ── Terminal nodes (R1–R9) have no outgoing edges ─────────────────────
  1352→    default:
  1353→      return [];
  1354→  }
  1355→}

  1357→// === Graph Traversal ==========================================================
  1358→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1359→// =============================================================================
  1360→
  1361→function traverseGraph(
  1362→  nodeId: string,
  1363→  pathProb: number,
  1364→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1365→  ctx: ProbContext,
  1366→  adj: Map<string, number[]>,
  1367→  edges: GraphEdge[],
  1368→  edgeProbabilities: Record<number, number>,
  1369→  nodeAccum: Record<string, number>,
  1370→  scenarios: Record<string, number>,
  1371→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1372→): void {
  1373→  // Accumulate path probability at this node
  1374→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1375→
  1376→  const outEdgeIndices = adj.get(nodeId);
  1377→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1378→
  1379→  // Compute outgoing edge probabilities for this node
  1380→  const probs = computeNodeProbabilities(
  1381→    nodeId,
  1382→    ctx,
  1383→    outEdgeIndices.length
  1384→  );
  1385→
  1386→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1387→    const edgeIdx = outEdgeIndices[i];
  1388→    const edge = edges[edgeIdx];
  1389→    const prob = probs[i] || 0;
  1390→
  1391→    // Store edge probability
  1392→    edgeProbabilities[edgeIdx] = prob;
  1393→
  1394→    // Determine branch for child nodes
  1395→    let childBranch = currentBranch;
  1396→    if (nodeId === 'ROOT') {
  1397→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1398→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1399→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1400→    }
  1401→
  1402→    const childPathProb = pathProb * prob;
  1403→
  1404→    // Check if target is a terminal node
  1405→    const isTerminal =
  1406→      edge.to.length === 2 &&
  1407→      edge.to[0] === 'R' &&
  1408→      edge.to[1] >= '1' &&
  1409→      edge.to[1] <= '9';
  1410→
  1411→    if (isTerminal) {
  1412→      // Accumulate scenario probability
  1413→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1414→      // Accumulate branch contribution
  1415→      if (childBranch) {
  1416→        contributions[edge.to][childBranch] += childPathProb;
  1417→      }
  1418→    }
  1419→
  1420→    // Recurse into child
  1421→    traverseGraph(
  1422→      edge.to,
  1423→      childPathProb,
  1424→      childBranch,
  1425→      ctx,
  1426→      adj,
  1427→      edges,
  1428→      edgeProbabilities,
  1429→      nodeAccum,
  1430→      scenarios,
  1431→      contributions
  1432→    );
  1433→  }
  1434→}

  1436→// === Main Export: buildDecisionGraph ===========================================
  1437→
  1438→export function buildDecisionGraph(input: GraphInput): GraphData {
  1439→  // 1. Build static graph structure
  1440→  const nodes = createNodes();
  1441→  const edges = createEdges();
  1442→  const nodePositions = createPositions();
  1443→
  1444→  // 2. Build adjacency map: nodeId → array of edge indices
  1445→  const adj = new Map<string, number[]>();
  1446→  for (let i = 0; i < edges.length; i++) {
  1447→    const from = edges[i].from;
  1448→    if (!adj.has(from)) adj.set(from, []);
  1449→    adj.get(from)!.push(i);
  1450→  }
  1451→
  1452→  // 3. Build probability context
  1453→  const ctx = buildContext(input);
  1454→
  1455→  // 4. Initialize accumulation structures
  1456→  const edgeProbabilities: Record<number, number> = {};
  1457→  const nodeAccum: Record<string, number> = {};
  1458→  const scenarios: Record<string, number> = {};
  1459→  const contributions: Record<
  1460→    string,
  1461→    { trend: number; breakout: number; reversal: number }
  1462→  > = {};
  1463→
  1464→  for (let i = 1; i <= 9; i++) {
  1465→    const key = `R${i}`;
  1466→    scenarios[key] = 0;
  1467→    contributions[key] = { trend: 0, breakout: 0, reversal: 0 };
  1468→  }
  1469→
  1470→  // 5. Traverse graph from ROOT
  1471→  traverseGraph(
  1472→    'ROOT',
  1473→    1.0,
  1474→    null,
  1475→    ctx,
  1476→    adj,
  1477→    edges,
  1478→    edgeProbabilities,
  1479→    nodeAccum,
  1480→    scenarios,
  1481→    contributions
  1482→  );
  1483→
  1484→  // 6. Post-process scenario probabilities
  1485→  // Convert to percentage, clamp [2, 35], re-normalize to sum=100
  1486→  const rawPcts: Record<string, number> = {};
  1487→  for (let i = 1; i <= 9; i++) {
  1488→    const key = `R${i}`;
  1489→    rawPcts[key] = Math.round((scenarios[key] || 0) * 100);
  1490→  }
  1491→
  1492→  // Clamp each to [2, 35]
  1493→  const clamped = Object.entries(rawPcts).map(([k, v]) => ({
  1494→    key: k,
  1495→    value: clamp(v, 2, 35),
  1496→  }));
  1497→
  1498→  // Re-normalize to sum=100
  1499→  const clampedSum = clamped.reduce((s, c) => s + c.value, 0);
  1500→  const finalPcts: Record<string, number> = {};
  1501→  if (clampedSum > 0) {
  1502→    // Distribute proportionally and round, then fix rounding errors
  1503→    let runningTotal = 0;
  1504→    const entries = clamped
  1505→      .map((c) => {
  1506→        const raw = (c.value / clampedSum) * 100;
  1507→        return { key: c.key, raw };
  1508→      })
  1509→      .sort((a, b) => b.raw - a.raw);
  1510→
  1511→    for (let i = 0; i < entries.length; i++) {
  1512→      const isLast = i === entries.length - 1;
  1513→      if (isLast) {
  1514→        finalPcts[entries[i].key] = Math.round(100 - runningTotal);
  1515→      } else {
  1516→        const rounded = Math.round(entries[i].raw);
  1517→        finalPcts[entries[i].key] = rounded;
  1518→        runningTotal += rounded;
  1519→      }
  1520→    }
  1521→  }
  1522→
  1523→  // Ensure no negative values due to rounding
  1524→  for (const key of Object.keys(finalPcts)) {
  1525→    if (finalPcts[key] < 0) finalPcts[key] = 0;
  1526→  }
  1527→
  1528→  // 7. Compute branch probabilities (from ROOT outgoing edges)
  1529→  const rootEdgeIndices = adj.get('ROOT') || [];
  1530→  const rootProbs = computeNodeProbabilities('ROOT', ctx, rootEdgeIndices.length);
  1531→  const branchProbabilities = {
  1532→    trend: rootProbs[0] || 0,
  1533→    breakout: rootProbs[1] || 0,
  1534→    reversal: rootProbs[2] || 0,
  1535→  };
  1536→
  1537→  // 8. Build node display values (path probability as percentage)
  1538→  const nodeValues: Record<string, string> = {};
  1539→  for (const [nodeId, accum] of Object.entries(nodeAccum)) {
  1540→    nodeValues[nodeId] = `${Math.round(accum * 100)}%`;
  1541→  }
  1542→
  1543→  // Override terminal node values with final post-processed percentages
  1544→  for (const [key, pct] of Object.entries(finalPcts)) {
  1545→    nodeValues[key] = `${pct}%`;
  1546→  }
  1547→
  1548→  // 9. Build path contributions (as percentages)
  1549→  const pathContributions: Record<
  1550→    string,
  1551→    { trend: number; breakout: number; reversal: number }
  1552→  > = {};
  1553→  for (let i = 1; i <= 9; i++) {
  1554→    const key = `R${i}`;
  1555→    const c = contributions[key];
  1556→    const total = c.trend + c.breakout + c.reversal;
  1557→    if (total > 0) {
  1558→      pathContributions[key] = {
  1559→        trend: Math.round((c.trend / total) * 100),
  1560→        breakout: Math.round((c.breakout / total) * 100),
  1561→        reversal: Math.round((c.reversal / total) * 100),
  1562→      };
  1563→    } else {
  1564→      pathContributions[key] = { trend: 33, breakout: 34, reversal: 33 };
  1565→    }
  1566→  }
  1567→
  1568→  // 10. Compute 30-day probability trend
  1569→  const probFractions: Record<string, number> = {};
  1570→  for (const [k, v] of Object.entries(finalPcts)) {
  1571→    probFractions[k] = v / 100;
  1572→  }
  1573→  const probabilityTrend = calcTrendFromProbs(probFractions, 30);
  1574→
  1575→  // 11. Return complete GraphData
  1576→  return {
  1577→    nodes,
  1578→    edges,
  1579→    nodePositions,
  1580→    edgeProbabilities,
  1581→    nodeValues,
  1582→    branchProbabilities,
  1583→    scenarioProbabilities: finalPcts,
  1584→    pathContributions,
  1585→    probabilityTrend,
  1586→  };
  1587→}
