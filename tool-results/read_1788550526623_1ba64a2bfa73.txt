   273→/* Advanced query parser — supports operators like country:US, exchange:NASD, sector:Tech */
   274→interface ParsedQuery {
   275→  country?: string;
   276→  exchange?: string;
   277→  sector?: string;
   278→  terms: string[];
   279→}
   280→
   281→function parseQuery(q: string): ParsedQuery {
   282→  const country: string[] = [];
   283→  const exchange: string[] = [];
   284→  const sector: string[] = [];
   285→  const terms: string[] = [];
   286→
   287→  for (const part of q.split(/\s+/)) {
   288→    if (!part) continue;
   289→    const lower = part.toLowerCase();
   290→    if (lower.startsWith('country:') || lower.startsWith('کشور:')) {
   291→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   292→      if (val) country.push(val.toLowerCase());
   293→    } else if (lower.startsWith('exchange:') || lower.startsWith('صرافی:')) {
   294→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   295→      if (val) exchange.push(val.toLowerCase());
   296→    } else if (lower.startsWith('sector:') || lower.startsWith('صنعت:')) {
   297→      const val = part.includes(':') ? part.split(':').slice(1).join(':') : '';
   298→      if (val) sector.push(val.toLowerCase());
   299→    } else {
   300→      terms.push(lower);
   301→    }
   302→  }
   303→
   304→  return {
   305→    country: country.length > 0 ? country.join(' ') : undefined,
   306→    exchange: exchange.length > 0 ? exchange.join(' ') : undefined,
   307→    sector: sector.length > 0 ? sector.join(' ') : undefined,
   308→    terms,
   309→  };
   310→}

   316→const RECENT_KEY = 'fin-search-recent';
   317→const MAX_RECENT = 8;
   318→
   319→function getRecentSearches(): string[] {
   320→  if (typeof window === 'undefined') return [];
   321→  try { const raw = localStorage.getItem(RECENT_KEY); return raw ? JSON.parse(raw) : []; }
   322→  catch { return []; }
   323→}
   324→
   325→function addRecentSearch(name: string) {
   326→  if (typeof window === 'undefined' || !name) return;
   327→  try {
   328→    let list = getRecentSearches();
   329→    list = list.filter((r) => r !== name);
   330→    list.unshift(name);
   331→    if (list.length > MAX_RECENT) list = list.slice(0, MAX_RECENT);
   332→    localStorage.setItem(RECENT_KEY, JSON.stringify(list));
   333→  } catch { /* ignore */ }
   334→}

   532→    const parsed = parseQuery(q);
   533→    const { country: opCountry, exchange: opExchange, sector: opSector, terms } = parsed;
   534→
   535→    const scoreAndFilter = (item: InstrumentItem) => {
   536→      /* Apply query operators */
   537→      if (opCountry) {
   538→        const match = (item.country?.toLowerCase().includes(opCountry)) ||
   539→          (item.countryEn?.toLowerCase().includes(opCountry));
   540→        if (!match) return null;
   541→      }
   542→      if (opExchange) {
   543→        const match = item.exchange?.toLowerCase().includes(opExchange);
   544→        if (!match) return null;
   545→      }
   546→      if (opSector) {
   547→        const match = item.sector?.toLowerCase().includes(opSector) ||
   548→          item.cs?.toLowerCase().includes(opSector);
   549→        if (!match) return null;
   550→      }
   551→
   552→      const fields = [item.l18, item.l30, item.cs, item.yahooSymbol || '', item.nameEn || '', item.exchange || '', item.country || '', item.countryEn || '', item.sector || ''];
   553→      const fullText = fields.join(' ').toLowerCase();
   554→      let score = 0;
   555→
   556→      for (const term of terms) {
   557→        if (!term) continue;
   558→        if (item.l18.toLowerCase().includes(term)) score += 10;
   559→        if (item.nameEn?.toLowerCase().includes(term)) score += 8;
   560→        if (item.yahooSymbol?.toLowerCase() === term) score += 20;
   561→        else if (item.yahooSymbol?.toLowerCase().includes(term)) score += 15;
   562→        if (item.l30.toLowerCase().includes(term)) score += 7;
   563→        if (item.exchange?.toLowerCase().includes(term)) score += 5;
   564→        if (item.country?.toLowerCase().includes(term)) score += 6;
   565→        if (item.countryEn?.toLowerCase() === term) score += 15;
   566→        else if (item.countryEn?.toLowerCase().includes(term)) score += 4;
   567→        if (item.sector?.toLowerCase().includes(term)) score += 5;
   568→        if (item.cs?.toLowerCase().includes(term)) score += 6;
   569→
   570→        if (!item.l18.toLowerCase().includes(term) && !item.l30.toLowerCase().includes(term) &&
   571→            !item.cs?.toLowerCase().includes(term) && !(item.yahooSymbol?.toLowerCase().includes(term)) &&
   572→            !(item.nameEn?.toLowerCase().includes(term)) && !(item.exchange?.toLowerCase().includes(term)) &&
   573→            !(item.country?.toLowerCase().includes(term)) && !(item.countryEn?.toLowerCase().includes(term)) &&
   574→            !(item.sector?.toLowerCase().includes(term))) {
   575→          return null;
   576→        }
   577→      }
   578→      /* If only operators (no terms), accept all filtered items */
   579→      if (terms.length === 0) score = 1;
   580→      return score;
   581→    };
   582→
   583→    const scored = items.map((item) => ({ item, score: scoreAndFilter(item) }))
   584→      .filter((r): r is { item: InstrumentItem; score: number } => r.score !== null && r.score > 0)
   585→      .sort((a, b) => b.score - a.score);
   586→
   587→    setTotalMatched(scored.length);
   588→    setResults(scored.slice(0, MAX_RESULTS).map((s) => s.item));

   590→  }, []);
