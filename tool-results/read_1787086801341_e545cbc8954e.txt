   110→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
   111→function sma(closes: number[], period: number): number {
   112→  if (closes.length < period) return 0;
   113→  const slice = closes.slice(-period);
   114→  return slice.reduce((a, b) => a + b, 0) / period;
   115→}
   116→
   117→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
   118→function emaCalc(closes: number[], period: number): number {
   119→  if (closes.length < period) return 0;
   120→  const k = 2 / (period + 1);
   121→  let emaVal = sma(closes.slice(0, period), period);
   122→  for (let i = period; i < closes.length; i++) {
   123→    emaVal = closes[i] * k + emaVal * (1 - k);
   124→  }
   125→  return emaVal;
   126→}

   128→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   129→function calcRSI(closes: number[], period: number = 14): number {
   130→  if (closes.length < period + 1) return 50;
   131→  let avgGain = 0;
   132→  let avgLoss = 0;
   133→  for (let i = 1; i <= period; i++) {
   134→    const diff = closes[i] - closes[i - 1];
   135→    if (diff > 0) avgGain += diff;
   136→    else avgLoss += Math.abs(diff);
   137→  }
   138→  avgGain /= period;
   139→  avgLoss /= period;
   140→  for (let i = period + 1; i < closes.length; i++) {
   141→    const diff = closes[i] - closes[i - 1];
   142→    const gain = diff > 0 ? diff : 0;
   143→    const loss = diff < 0 ? Math.abs(diff) : 0;
   144→    avgGain = (avgGain * (period - 1) + gain) / period;
   145→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   146→  }
   147→  if (avgLoss === 0) return 100;
   148→  const rs = avgGain / avgLoss;
   149→  return 100 - 100 / (1 + rs);
   150→}

   152→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   153→function calcMFI(data: OHLCV[], period: number = 14): number {
   154→  if (data.length < period + 1) return 50;
   155→  let posFlow = 0;
   156→  let negFlow = 0;
   157→  for (let i = data.length - period; i < data.length; i++) {
   158→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   159→    const mf = tp * data[i].volume;
   160→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   161→    if (tp > prevTp) posFlow += mf;
   162→    else negFlow += mf;
   163→  }
   164→  if (negFlow === 0) return 100;
   165→  const mfr = posFlow / negFlow;
   166→  return 100 - 100 / (1 + mfr);
   167→}

   169→// ─── Helper: CCI(20) ──────────────────────────────────────────────────────────
   170→function calcCCI(data: OHLCV[], period: number = 20): number {
   171→  if (data.length < period) return 0;
   172→  const tps: number[] = [];
   173→  for (let i = data.length - period; i < data.length; i++) {
   174→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   175→  }
   176→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   177→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   178→  if (meanDev === 0) return 0;
   179→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   180→}

   182→// ─── Helper: Stochastic(14,3,3) ───────────────────────────────────────────────
   183→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   184→  if (data.length < kPeriod) return { k: 50, d: 50 };
   185→  const rawKs: number[] = [];
   186→  for (let i = kPeriod - 1; i < data.length; i++) {
   187→    let lowest = Infinity;
   188→    let highest = -Infinity;
   189→    for (let j = i - kPeriod + 1; j <= i; j++) {
   190→      if (data[j].low < lowest) lowest = data[j].low;
   191→      if (data[j].high > highest) highest = data[j].high;
   192→    }
   193→    const range = highest - lowest;
   194→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   195→  }
   196→  const smoothedK: number[] = [];
   197→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   198→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   199→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   200→  }
   201→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   202→  const dSlice = smoothedK.slice(-smoothD);
   203→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   204→  return { k, d };
   205→}

   207→// ─── Helper: Williams %R(14) ──────────────────────────────────────────────────
   208→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   209→  if (data.length < period) return -50;
   210→  const slice = data.slice(-period);
   211→  let highest = -Infinity;
   212→  let lowest = Infinity;
   213→  for (const c of slice) {
   214→    if (c.high > highest) highest = c.high;
   215→    if (c.low < lowest) lowest = c.low;
   216→  }
   217→  const range = highest - lowest;
   218→  if (range === 0) return -50;
   219→  return ((highest - data[data.length - 1].close) / range) * -100;
   220→}

   222→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   223→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   224→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   225→  const emaFast = emaCalc(closes, fast);
   226→  const emaSlow = emaCalc(closes, slow);
   227→  const macdLine = emaFast - emaSlow;
   228→  // Calculate signal as EMA of MACD values
   229→  const macdValues: number[] = [];
   230→  const kFast = 2 / (fast + 1);
   231→  const kSlow = 2 / (slow + 1);
   232→  let eFast = sma(closes.slice(0, fast), fast);
   233→  let eSlow = sma(closes.slice(0, slow), slow);
   234→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   235→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   236→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   237→    macdValues.push(eFast - eSlow);
   238→  }
   239→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   240→  const kSig = 2 / (sig + 1);
   241→  let signalVal = sma(macdValues.slice(0, sig), sig);
   242→  for (let i = sig; i < macdValues.length; i++) {
   243→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   244→  }
   245→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   246→}

   248→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   249→function calcATR(data: OHLCV[], period: number = 14): number {
   250→  if (data.length < 2) return 0;
   251→  const trs: number[] = [];
   252→  for (let i = 1; i < data.length; i++) {
   253→    const tr = Math.max(
   254→      data[i].high - data[i].low,
   255→      Math.abs(data[i].high - data[i - 1].close),
   256→      Math.abs(data[i].low - data[i - 1].close)
   257→    );
   258→    trs.push(tr);
   259→  }
   260→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   261→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   262→  for (let i = period; i < trs.length; i++) {
   263→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   264→  }
   265→  return atrVal;
   266→}

   268→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   269→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   270→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   271→  const plusDMs: number[] = [];
   272→  const minusDMs: number[] = [];
   273→  const trs: number[] = [];
   274→  for (let i = 1; i < data.length; i++) {
   275→    const upMove = data[i].high - data[i - 1].high;
   276→    const downMove = data[i - 1].low - data[i].low;
   277→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   278→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   279→    trs.push(Math.max(
   280→      data[i].high - data[i].low,
   281→      Math.abs(data[i].high - data[i - 1].close),
   282→      Math.abs(data[i].low - data[i - 1].close)
   283→    ));
   284→  }
   285→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   286→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   287→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   288→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   289→  const dxValues: number[] = [];
   290→  for (let i = period; i < trs.length; i++) {
   291→    smoothTR = smoothTR - smoothTR / period + trs[i];
   292→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   293→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   294→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   295→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   296→    const diSum = diPlus + diMinus;
   297→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   298→    dxValues.push(dx);
   299→  }
   300→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   301→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   302→  for (let i = period; i < dxValues.length; i++) {
   303→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   304→  }
   305→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   306→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   307→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   308→}

   310→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   311→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   312→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   313→  let isLong = true;
   314→  let af = step;
   315→  let ep = data[0].high;
   316→  let sar = data[0].low;
   317→  for (let i = 1; i < data.length; i++) {
   318→    const prevSar = sar;
   319→    sar = prevSar + af * (ep - prevSar);
   320→    // Ensure SAR is within bounds
   321→    if (isLong) {
   322→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   323→      if (data[i].low < sar) {
   324→        isLong = false;
   325→        sar = ep;
   326→        ep = data[i].low;
   327→        af = step;
   328→      } else {
   329→        if (data[i].high > ep) {
   330→          ep = data[i].high;
   331→          af = Math.min(af + step, max);
   332→        }
   333→      }
   334→    } else {
   335→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   336→      if (data[i].high > sar) {
   337→        isLong = true;
   338→        sar = ep;
   339→        ep = data[i].high;
   340→        af = step;
   341→      } else {
   342→        if (data[i].low < ep) {
   343→          ep = data[i].low;
   344→          af = Math.min(af + step, max);
   345→        }
   346→      }
   347→    }
   348→  }
   349→  return sar;
   350→}

   352→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   353→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   354→  if (closes.length < period) {
   355→    const c = closes[closes.length - 1] ?? 0;
   356→    return { upper: c, middle: c, lower: c };
   357→  }
   358→  const slice = closes.slice(-period);
   359→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   360→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   361→  const sd = Math.sqrt(variance);
   362→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   363→}

   365→// ─── Helper: OBV ──────────────────────────────────────────────────────────────
   366→function calcOBV(data: OHLCV[]): number {
   367→  let obv = 0;
   368→  for (let i = 1; i < data.length; i++) {
   369→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   370→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   371→  }
   372→  return obv;
   373→}

   375→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   376→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   377→  const n = values.length;
   378→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   379→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   380→  for (let i = 0; i < n; i++) {
   381→    sumX += i;
   382→    sumY += values[i];
   383→    sumXY += i * values[i];
   384→    sumX2 += i * i;
   385→    sumY2 += values[i] * values[i];
   386→  }
   387→  const denom = n * sumX2 - sumX * sumX;
   388→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   389→  const slope = (n * sumXY - sumX * sumY) / denom;
   390→  const intercept = (sumY - slope * sumX) / n;
   391→  const meanY = sumY / n;
   392→  let ssRes = 0, ssTot = 0;
   393→  for (let i = 0; i < n; i++) {
   394→    const predicted = slope * i + intercept;
   395→    ssRes += (values[i] - predicted) ** 2;
   396→    ssTot += (values[i] - meanY) ** 2;
   397→  }
   398→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   399→  return { slope, intercept, r2 };
   400→}

   402→// ─── Helper: Round to nice number ─────────────────────────────────────────────
   403→function roundToNice(price: number): number {
   404→  const abs = Math.abs(price);
   405→  let step: number;
   406→  if (abs >= 100000) step = 100;
   407→  else if (abs >= 10000) step = 50;
   408→  else if (abs >= 1000) step = 10;
   409→  else if (abs >= 100) step = 5;
   410→  else step = 1;
   411→  return Math.round(price / step) * step;
   412→}

   414→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
   415→function findSwingLevels(data: OHLCV[], lookback: number = 3): { highs: { price: number; idx: number }[]; lows: { price: number; idx: number }[] } {
   416→  const highs: { price: number; idx: number }[] = [];
   417→  const lows: { price: number; idx: number }[] = [];
   418→  for (let i = lookback; i < data.length - lookback; i++) {
   419→    let isHigh = true;
   420→    let isLow = true;
   421→    for (let j = i - lookback; j <= i + lookback; j++) {
   422→      if (j === i) continue;
   423→      if (data[j].high >= data[i].high) isHigh = false;
   424→      if (data[j].low <= data[i].low) isLow = false;
   425→    }
   426→    if (isHigh) highs.push({ price: data[i].high, idx: i });
   427→    if (isLow) lows.push({ price: data[i].low, idx: i });
   428→  }
   429→  return { highs, lows };
   430→}

   475→  volumeRatio: number;
   476→  freshness: number;
   477→  distancePercent: number;
   478→}

   480→// ─── Helper: Psychological Rounding ────────────────────────────────────────
   481→function psychStep(price: number): number {
   482→  if (price >= 100000) return 5000;
   483→  if (price >= 10000) return 1000;
   484→  if (price >= 1000) return 100;
   485→  if (price >= 100) return 10;
   486→  if (price >= 10) return 1;
   487→  return 0.01;
   488→}
   489→
   490→function psychRound(price: number): number {
   491→  const step = psychStep(price);
   492→  return Math.round(price / step) * step;
   493→}

   495→// ─── Helper: Linear System Solver (Gaussian Elimination) ─────────────────
   496→function solveLinearSystem(A: number[][], b: number[]): number[] | null {
   497→  const n = b.length;
   498→  const aug = A.map((row, i) => [...row, b[i]]);
   499→
   500→  for (let col = 0; col < n; col++) {
   501→    let maxVal = Math.abs(aug[col][col]);
   502→    let maxRow = col;
   503→    for (let row = col + 1; row < n; row++) {
   504→      if (Math.abs(aug[row][col]) > maxVal) {
   505→        maxVal = Math.abs(aug[row][col]);
   506→        maxRow = row;
   507→      }
   508→    }
   509→    if (maxVal < 1e-10) return null;
   510→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   511→
   512→    for (let row = col + 1; row < n; row++) {
   513→      const factor = aug[row][col] / aug[col][col];
   514→      for (let j = col; j <= n; j++) {
   515→        aug[row][j] -= factor * aug[col][j];
   516→      }
   517→    }
   518→  }
   519→
   520→  const x = new Array(n).fill(0);
   521→  for (let i = n - 1; i >= 0; i--) {
   522→    x[i] = aug[i][n];
   523→    for (let j = i + 1; j < n; j++) {
   524→      x[i] -= aug[i][j] * x[j];
   525→    }
   526→    if (Math.abs(aug[i][i]) < 1e-10) return null;
   527→    x[i] /= aug[i][i];
   528→  }
   529→
   530→  return x;
   531→}

   533→// ─── Helper: Generate All 7-Method S/R Levels ─────────────────────────────
   534→function generateAllLevels(data: OHLCV[], endIdx: number): RawSRLevel[] {
   535→  const levels: RawSRLevel[] = [];
   536→  if (endIdx < 10) return levels;
   537→
   538→  const slice = data.slice(0, endIdx + 1);
   539→  const currentPrice = slice[slice.length - 1].close;
   540→  const closes = slice.map(d => d.close);
   541→  const noFib: [string, string] = ['—', ''];
   542→
   543→  // ── Method 1: Swing High/Low (Price Action) ──────────────────────────
   544→  for (const lookback of [3, 5, 7]) {
   545→    const swings = findSwingLevels(slice, lookback);
   546→    const tol = currentPrice * 0.005; // 0.5% tolerance
   547→
   548→    // Connect swing highs → resistance
   549→    const usedHigh = new Set<number>();
   550→    for (const h of swings.highs) {
   551→      if (usedHigh.has(h.idx)) continue;
   552→      let sum = h.price;
   553→      let count = 1;
   554→      for (const h2 of swings.highs) {
   555→        if (h2 === h || usedHigh.has(h2.idx)) continue;
   556→        if (Math.abs(h2.price - h.price) < tol) {
   557→          sum += h2.price;
   558→          count++;
   559→          usedHigh.add(h2.idx);
   560→        }
   561→      }
   562→      usedHigh.add(h.idx);
   563→      if (sum / count > currentPrice * 1.005) {
   564→        levels.push({ price: sum / count, method: 'Swing_High', type: 'resistance', ...noFib });
   565→      }
   566→    }
   567→
   568→    // Connect swing lows → support
   569→    const usedLow = new Set<number>();
   570→    for (const l of swings.lows) {
   571→      if (usedLow.has(l.idx)) continue;
   572→      let sum = l.price;
   573→      let count = 1;
   574→      for (const l2 of swings.lows) {
   575→        if (l2 === l || usedLow.has(l2.idx)) continue;
   576→        if (Math.abs(l2.price - l.price) < tol) {
   577→          sum += l2.price;
   578→          count++;
   579→          usedLow.add(l2.idx);
   580→        }
   581→      }
   582→      usedLow.add(l.idx);
   583→      if (sum / count < currentPrice * 0.995) {
   584→        levels.push({ price: sum / count, method: 'Swing_Low', type: 'support', ...noFib });
   585→      }
   586→    }
   587→  }
   588→
   589→  // ── Method 2: SMA Dynamic S/R ──────────────────────────────────────
   590→  for (const period of [20, 50, 100, 200]) {
   591→    const smaVal = sma(closes, period);
   592→    if (smaVal <= 0) continue;
   593→    const type: 'support' | 'resistance' = currentPrice > smaVal ? 'support' : 'resistance';
   594→    levels.push({ price: smaVal, method: `SMA_${period}`, type, ...noFib });
   595→  }
   596→
   597→  // ── Method 3: Bollinger Bands ───────────────────────────────────────
   598→  const bb = calcBollingerBands(closes);
   599→  if (bb.upper > 0 && bb.lower > 0) {
   600→    levels.push({ price: bb.upper, method: 'BB_Upper', type: 'resistance', ...noFib });
   601→    levels.push({ price: bb.lower, method: 'BB_Lower', type: 'support', ...noFib });
   602→  }
   603→
   604→  // ── Method 4: Fibonacci Retracement + Extension ─────────────────────
   605→  interface SwingPair { high: number; highIdx: number; low: number; lowIdx: number; range: number }
   606→  let bestSwing: SwingPair | null = null;
   607→
   608→  for (const lb of [3, 5, 7, 10, 15, 20]) {
   609→    const sw = findSwingLevels(slice, lb);
   610→    if (sw.highs.length === 0 || sw.lows.length === 0) continue;
   611→    for (const h of sw.highs) {
   612→      for (const l of sw.lows) {
   613→        if (Math.abs(h.idx - l.idx) < 3) continue;
   614→        const range = Math.abs(h.price - l.price);
   615→        if (range / currentPrice < 0.03) continue;
   616→        if (!bestSwing || range > bestSwing.range) {
   617→          bestSwing = {
   618→            high: Math.max(h.price, l.price),
   619→            highIdx: h.price > l.price ? h.idx : l.idx,
   620→            low: Math.min(h.price, l.price),
   621→            lowIdx: h.price < l.price ? l.idx : h.idx,
   622→            range,
   623→          };
   624→        }
   625→      }
   626→    }
   627→  }
   628→
   629→  if (bestSwing) {
   630→    const H = bestSwing.high;
   631→    const L = bestSwing.low;
   632→    const range = H - L;
   633→
   634→    for (const ratio of FIB_RETRACEMENTS) {
   635→      const price = H - ratio * range;
   636→      if (price <= 0) continue;
   637→      const rounded = roundToNice(price);
   638→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   639→      levels.push({
   640→        price: rounded, method: `Fibonacci_${ratio}`, type,
   641→        fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   642→      });
   643→    }
   644→
   645→    for (const extRatio of FIB_EXTENSIONS) {
   646→      const price = H + (extRatio - 1) * range;
   647→      if (price <= 0) continue;
   648→      const rounded = roundToNice(price);
   649→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   650→      levels.push({
   651→        price: rounded, method: `Fibonacci_${extRatio}`, type,
   652→        fibRatio: String(extRatio), fibLabel: FIB_LABELS[String(extRatio)] || '',
   653→      });
   654→    }
   655→
   656→    // Inverse extensions (below L)
   657→    for (const extRatio of FIB_EXTENSIONS) {
   658→      const price = L - (extRatio - 1) * range;
   659→      if (price <= 0) continue;
   660→      const rounded = roundToNice(price);
   661→      const type: 'support' | 'resistance' = rounded >= currentPrice ? 'resistance' : 'support';
   662→      levels.push({
   663→        price: rounded, method: `Fibonacci_${-extRatio}`, type,
   664→        fibRatio: String(-extRatio), fibLabel: '',
   665→      });
   666→    }
   667→
   668→    // Secondary swing confluence (check overlap with secondary swings)
   669→    const secondarySwings: SwingPair[] = [];
   670→    for (const lb of [3, 5, 7]) {
   671→      const sw = findSwingLevels(slice, lb);
   672→      for (const h of sw.highs) {
   673→        for (const l of sw.lows) {
   674→          if (Math.abs(h.idx - l.idx) < 3) continue;
   675→          const r = Math.abs(h.price - l.price);
   676→          if (r / currentPrice < 0.05) continue;
   677→          if (bestSwing && Math.abs(r - bestSwing.range) / bestSwing.range < 0.1) continue;
   678→          secondarySwings.push({
   679→            high: Math.max(h.price, l.price), highIdx: h.price > l.price ? h.idx : l.idx,
   680→            low: Math.min(h.price, l.price), lowIdx: h.price < l.price ? l.idx : h.idx,
   681→            range: r,
   682→          });
   683→        }
   684→      }
   685→    }
   686→    // Secondary swings' fib levels are merged via confluence (1% rule in mergeLevels)
   687→    const confluenceTol = currentPrice * 0.005;
   688→    for (const sec of secondarySwings.slice(0, 3)) {
   689→      const sH = sec.high, sL = sec.low, sR = sH - sL;
   690→      for (const ratio of FIB_RETRACEMENTS) {
   691→        const price = roundToNice(sH - ratio * sR);
   692→        if (price <= 0) continue;
   693→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   694→        levels.push({
   695→          price, method: `Fibonacci_${ratio}`, type,
   696→          fibRatio: String(ratio), fibLabel: FIB_LABELS[String(ratio)] || '',
   697→        });
   698→      }
   699→      for (const extR of FIB_EXTENSIONS) {
   700→        const price = roundToNice(sH + (extR - 1) * sR);
   701→        if (price <= 0) continue;
   702→        const type: 'support' | 'resistance' = price >= currentPrice ? 'resistance' : 'support';
   703→        levels.push({
   704→          price, method: `Fibonacci_${extR}`, type,
   705→          fibRatio: String(extR), fibLabel: FIB_LABELS[String(extR)] || '',
   706→        });
   707→      }
   708→    }
   709→  }
   710→
   711→  // ── Method 5: Volume Profile (VAP) ──────────────────────────────────
   712→  const dataHasVolume = slice.some(d => d.volume > 0);
   713→  if (dataHasVolume) {
   714→    let minP = Infinity, maxP = -Infinity;
   715→    for (const d of slice) {
   716→      if (d.low < minP) minP = d.low;
   717→      if (d.high > maxP) maxP = d.high;
   718→    }
   719→    const numBins = 40;
   720→    const binSize = Math.max((maxP - minP) / numBins, currentPrice * 0.001);
   721→    const vap = new Array(numBins).fill(0);
   722→    const binCenters: number[] = [];
   723→
   724→    for (let b = 0; b < numBins; b++) {
   725→      binCenters.push(minP + (b + 0.5) * binSize);
   726→    }
   727→
   728→    for (const d of slice) {
   729→      for (let b = 0; b < numBins; b++) {
   730→        if (d.low <= binCenters[b] && binCenters[b] <= d.high) {
   731→          vap[b] += d.volume;
   732→        }
   733→      }
   734→    }
   735→
   736→    const maxVAP = Math.max(...vap);
   737→    if (maxVAP > 0) {
   738→      // POC (Point of Control)
   739→      let pocIdx = 0;
   740→      for (let b = 1; b < numBins; b++) {
   741→        if (vap[b] > vap[pocIdx]) pocIdx = b;
   742→      }
   743→      const pocPrice = binCenters[pocIdx];
   744→      const pocType: 'support' | 'resistance' = pocPrice > currentPrice ? 'resistance' : 'support';
   745→      levels.push({ price: pocPrice, method: 'VAP_POC', type: pocType, ...noFib });
   746→
   747→      // HVN (High Volume Nodes)
   748→      for (let b = 0; b < numBins; b++) {
   749→        if (b === pocIdx) continue;
   750→        if (vap[b] > 0.7 * maxVAP) {
   751→          const hvnPrice = binCenters[b];
   752→          const hvnType: 'support' | 'resistance' = hvnPrice > currentPrice ? 'resistance' : 'support';
   753→          levels.push({ price: hvnPrice, method: 'VAP_HVN', type: hvnType, ...noFib });
   754→        }
   755→      }
   756→    }
   757→  }
   758→
   759→  // ── Method 6: Pivot Points (4 algorithms) ───────────────────────────
   760→  if (endIdx >= 1) {
   761→    const prev = slice[endIdx - 1]; // Previous completed bar
   762→    const pH = prev.high, pL = prev.low, pC = prev.close, pO = prev.open;
   763→
   764→    // Standard Pivot
   765→    {
   766→      const PP = (pH + pL + pC) / 3;
   767→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   768→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   769→      const R3 = pH + 2 * (PP - pL), S3 = pL - 2 * (pH - PP);
   770→ if (R1 > 0) levels.push({ price: R1, method: 'Pivot_R1', type: 'resistance', ...noFib });
   771→      if (R2 > 0) levels.push({ price: R2, method: 'Pivot_R2', type: 'resistance', ...noFib });
   772→      if (R3 > 0) levels.push({ price: R3, method: 'Pivot_R3', type: 'resistance', ...noFib });
   773→      if (S1 > 0) levels.push({ price: S1, method: 'Pivot_S1', type: 'support', ...noFib });
   774→      if (S2 > 0) levels.push({ price: S2, method: 'Pivot_S2', type: 'support', ...noFib });
   775→      if (S3 > 0) levels.push({ price: S3, method: 'Pivot_S3', type: 'support', ...noFib });
   776→    }
   777→
   778→    // Fibonacci Pivot
   779→    {
   780→      const PP = (pH + pL + pC) / 3;
   781→      const range = pH - pL;
   782→      for (const [mult, suffix] of [[0.382, '1'], [0.618, '2'], [1.0, '3']] as [number, string][]) {
   783→        const R = PP + mult * range;
   784→        const S = PP - mult * range;
   785→        if (R > 0) levels.push({ price: R, method: `Fib_Pivot_R${suffix}`, type: 'resistance', ...noFib });
   786→        if (S > 0) levels.push({ price: S, method: `Fib_Pivot_S${suffix}`, type: 'support', ...noFib });
   787→      }
   788→    }
   789→
   790→    // Woodie's Pivot
   791→    {
   792→      const PP = (pH + pL + 2 * pC) / 4;
   793→      const R1 = 2 * PP - pL, S1 = 2 * PP - pH;
   794→      const R2 = PP + (pH - pL), S2 = PP - (pH - pL);
   795→      if (R1 > 0) levels.push({ price: R1, method: 'Woodie_R1', type: 'resistance', ...noFib });
   796→      if (R2 > 0) levels.push({ price: R2, method: 'Woodie_R2', type: 'resistance', ...noFib });
   797→      if (S1 > 0) levels.push({ price: S1, method: 'Woodie_S1', type: 'support', ...noFib });
   798→      if (S2 > 0) levels.push({ price: S2, method: 'Woodie_S2', type: 'support', ...noFib });
   799→    }
   800→
   801→    // DeMark Pivot
   802→    {
   803→      const X = pC < pO ? pH + 2 * pL + pC : pC > pO ? 2 * pH + pL + pC : pH + pL + 2 * pC;
   804→      const PP = X / 4;
   805→      const R1 = X / 2 - pL;
   806→      const S1 = X / 2 - pH;
   807→      if (R1 > 0) levels.push({ price: R1, method: 'DeMark_R1', type: 'resistance', ...noFib });
   808→      if (S1 > 0) levels.push({ price: S1, method: 'DeMark_S1', type: 'support', ...noFib });
   809→    }
   810→  }
   811→
   812→  // ── Method 7: Psychological Levels ──────────────────────────────────
   813→  {
   814→    const step = psychStep(currentPrice);
   815→    const base = psychRound(currentPrice);
   816→    for (const mult of [-2, -1, 1, 2]) {
   817→      const price = base + mult * step;
   818→      if (price <= 0) continue;
   819→      const type: 'support' | 'resistance' = price > currentPrice ? 'resistance' : 'support';
   820→      levels.push({ price, method: 'Psychological', type, ...noFib });
   821→    }
   822→  }
   823→
   824→  return levels;
   825→}

   827→// ─── Helper: Merge Nearby Levels (1% Confluence) ──────────────────────────
   828→function mergeLevels(rawLevels: RawSRLevel[], currentPrice: number): MergedSRLevel[] {
   829→  if (rawLevels.length === 0) return [];
   830→
   831→  const tol = currentPrice * 0.01; // 1% tolerance
   832→  const merged: MergedSRLevel[] = [];
   833→  const used = new Set<number>();
   834→
   835→  for (let i = 0; i < rawLevels.length; i++) {
   836→    if (used.has(i)) continue;
   837→
   838→    let sumPrice = rawLevels[i].price;
   839→    let count = 1;
   840→    const methods = new Set<string>();
   841→    methods.add(rawLevels[i].method);
   842→    let bestFibRatio = rawLevels[i].fibRatio;
   843→    let bestFibLabel = rawLevels[i].fibLabel;
   844→
   845→    for (let j = i + 1; j < rawLevels.length; j++) {
   846→      if (used.has(j)) continue;
   847→      if (Math.abs(rawLevels[j].price - rawLevels[i].price) < tol) {
   848→        sumPrice += rawLevels[j].price;
   849→        count++;
   850→        methods.add(rawLevels[j].method);
   851→        used.add(j);
   852→        // Prefer Fibonacci label if available
   853→        if (rawLevels[j].fibRatio !== '—' && bestFibRatio === '—') {
   854→          bestFibRatio = rawLevels[j].fibRatio;
   855→          bestFibLabel = rawLevels[j].fibLabel;
   856→        }
   857→      }
   858→    }
   859→    used.add(i);
   860→
   861→    const avgPrice = sumPrice / count;
   862→    const type: 'support' | 'resistance' = avgPrice >= currentPrice ? 'resistance' : 'support';
   863→
   864→    merged.push({
   865→      price: avgPrice,
   866→      type,
   867→      methods: [...methods],
   868→      overlapCount: methods.size,
   869→      fibRatio: bestFibRatio,
   870→      fibLabel: bestFibLabel,
   871→      score: 0,
   872→      strength: 1,
   873→      touchCount: 0,
   874→      volumeRatio: 0,
   875→      freshness: 0,
   876→      distancePercent: 0,
   877→    });
   878→  }
   879→
   880→  return merged;
   881→}

   883→// ─── Helper: Compute Level Features (5 ML features) ──────────────────────
   884→function computeLevelFeatures(
   885→  levelPrice: number,
   886→  data: OHLCV[],
   887→  endIdx: number,
   888→  currentPrice: number,
   889→  overlapCount: number,
   890→): number[] {
   891→  const lookback = Math.min(30, endIdx);
   892→  const startIdx = endIdx - lookback;
   893→  const touchTol = currentPrice * 0.002; // 0.2%
   894→
   895→  let touchCount = 0;
   896→  let totalVolAtTouches = 0;
   897→  let totalVol = 0;
   898→  let barsSinceLastTouch = lookback;
   899→
   900→  for (let i = startIdx; i <= endIdx; i++) {
   901→    const bar = data[i];
   902→    totalVol += bar.volume;
   903→
   904→    if (bar.low <= levelPrice + touchTol && bar.high >= levelPrice - touchTol) {
   905→      touchCount++;
   906→      totalVolAtTouches += bar.volume;
   907→      barsSinceLastTouch = endIdx - i;
   908→    }
   909→  }
   910→
   911→  const avgVol = lookback > 0 ? totalVol / lookback : 0;
   912→  const volumeRatio = (touchCount > 0 && avgVol > 0) ? (totalVolAtTouches / touchCount) / avgVol : 1;
   913→  const freshness = Math.max(0, Math.min(1, 1 - barsSinceLastTouch / 30));
   914→  const distancePercent = Math.min(1, Math.abs(levelPrice - currentPrice) / currentPrice);
   915→
   916→  // Features: [touch_count, volume_ratio, overlap_count, freshness, distance_percent]
   917→  return [touchCount, volumeRatio, overlapCount - 1, freshness, distancePercent];
   918→}

   920→// ─── Helper: Train S/R Scoring Weights (Ridge OLS) ───────────────────────
   921→function trainSRWeights(data: OHLCV[]): number[] {
   922→  const defaultWeights = [0.25, 0.20, 0.25, 0.15, 0.15];
   923→
   924→  if (data.length < 50) return defaultWeights;
   925→
   926→  const X: number[][] = [];
   927→  const y: number[] = [];
   928→
   929→  for (let i = 40; i <= data.length - 11; i += 5) {
   930→    const barPrice = data[i].close;
   931→    const rawLevels = generateAllLevels(data, i);
   932→    if (rawLevels.length === 0) continue;
   933→    const merged = mergeLevels(rawLevels, barPrice);
   934→
   935→    for (const level of merged) {
   936→      const features = computeLevelFeatures(level.price, data, i, barPrice, level.overlapCount);
   937→      X.push(features);
   938→
   939→      // Label: did price return within 10 bars?
   940→      let returned = false;
   941→      const retTol = barPrice * 0.005;
   942→      for (let j = i + 1; j <= Math.min(i + 10, data.length - 1); j++) {
   943→        if (data[j].low <= level.price + retTol && data[j].high >= level.price - retTol) {
   944→          returned = true;
   945→          break;
   946→        }
   947→      }
   948→      y.push(returned ? 1 : 0);
   949→    }
   950→  }
   951→
   952→  if (X.length < 20) return defaultWeights;
   953→
   954→  // Ridge OLS: w = (X^T X + λI)^(-1) X^T y
   955→  const n = X.length;
   956→  const p = 5;
   957→  const lambda = 0.01;
   958→
   959→  const XtX: number[][] = Array.from({ length: p }, () => new Array(p).fill(0));
   960→  for (let i = 0; i < n; i++) {
   961→    for (let a = 0; a < p; a++) {
   962→      for (let b = 0; b < p; b++) {
   963→        XtX[a][b] += X[i][a] * X[i][b];
   964→      }
   965→    }
   966→  }
   967→  for (let a = 0; a < p; a++) XtX[a][a] += lambda;
   968→
   969→  const Xty: number[] = new Array(p).fill(0);
   970→  for (let i = 0; i < n; i++) {
   971→    for (let a = 0; a < p; a++) {
   972→      Xty[a] += X[i][a] * y[i];
   973→    }
   974→  }
   975→
   976→  const weights = solveLinearSystem(XtX, Xty);
   977→  if (!weights) return defaultWeights;
   978→  if (weights.some(w => isNaN(w) || !isFinite(w))) return defaultWeights;
   979→
   980→  for (let i = 0; i < p; i++) {
   981→    if (weights[i] < 0) weights[i] = 0;
   982→  }
   983→
   984→  const sum = weights.reduce((a, b) => a + b, 0);
   985→  if (sum <= 0) return defaultWeights;
   986→
   987→  return weights.map(w => w / sum);
   988→}

   990→// ─── Support / Resistance (7-Method ML-Based System) ─────────────────────
   991→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   992→  resistances: number[];
   993→  supports: number[];
   994→  supportStrengths: LevelStrength[];
   995→  resistanceStrengths: LevelStrength[];
   996→  priceTargets: LevelStrength[];
   997→} {

   999→    return {
  1000→      resistances: [], supports: [],
  1001→      supportStrengths: [], resistanceStrengths: [],
  1002→      priceTargets: [],

  1004→  }

  1007→  const rawLevels = generateAllLevels(data, data.length - 1);

  1013→  const weights = trainSRWeights(data);
  1014→
  1015→  // ── Step 4: Compute features & score each merged level ────────────────
  1016→  for (const level of merged) {
  1017→    const features = computeLevelFeatures(level.price, data, data.length - 1, currentPrice, level.overlapCount);
  1018→
  1019→    const touchScore = Math.min(features[0] / 5, 1);
  1020→    const volumeScore = Math.min(features[1] / 2, 1);
  1021→    const overlapScore = Math.min(features[2] * 2, 10) / 10;
  1022→    const freshnessScore = features[3];
  1023→    const distanceScore = 1 - features[4];
  1024→
  1025→    const powerScore = (
  1026→      weights[0] * touchScore +
  1027→      weights[1] * volumeScore +
  1028→      weights[2] * overlapScore +
  1029→      weights[3] * freshnessScore +
  1030→      weights[4] * distanceScore
  1031→    ) * 10;
  1032→
  1033→    level.score = Math.max(0, Math.min(10, powerScore));
  1034→    level.strength = Math.max(1, Math.min(10, Math.round(level.score)));
  1035→    level.touchCount = features[0];
  1036→    level.volumeRatio = features[1];
  1037→    level.freshness = features[3];
  1038→    level.distancePercent = features[4];
  1039→  }

  1042→  for (const level of merged) {
  1043→    level.price = psychRound(level.price);
  1044→  }

  1047→  for (const level of merged) {
  1048→    level.type = level.price >= currentPrice ? 'resistance' : 'support';
  1049→  }

  1052→  {
  1053→    const deduped: MergedSRLevel[] = [];
  1054→    for (const level of merged) {
  1055→      const existing = deduped.find(d => Math.abs(d.price - level.price) < currentPrice * 0.001);
  1056→      if (existing) {
  1057→        if (level.score > existing.score) {
  1058→          Object.assign(existing, level);
  1059→        }
  1060→      } else {
  1061→        deduped.push(level);
  1062→      }
  1063→    }
  1064→    merged = deduped;
  1065→  }
  1066→
  1067→  // ── Step 6: Separate into supports and resistances ─────────────────────
  1068→  let supports = merged.filter(l => l.type === 'support');
  1069→  let resistances = merged.filter(l => l.type === 'resistance');
  1070→
  1071→  // ── Step 7: Distance filtering (strict 5-10% gap rules) ─────────────
  1072→  // Rules:
  1073→  //   1. Consecutive lines of same type: 5% ≤ gap ≤ 10%
  1074→  //   2. Nearest support ≤ 10% below price (can be < 5%)
  1075→  //   3. Nearest resistance ≤ 10% above price (can be < 5%)

  1078→    levels: MergedSRLevel[],
  1079→    type: 'support' | 'resistance'

  1084→    const sorted = [...levels].sort((a, b) =>
  1085→      type === 'support' ? b.price - a.price : a.price - b.price
  1086→    );
  1087→
  1088→    // ── Rule 2/3: Nearest level must be within 10% of price ──
  1089→    const nearest = sorted[0];
  1090→    const distFromPrice = Math.abs(nearest.price - currentPrice) / currentPrice;
  1091→    if (distFromPrice > 0.10) {
  1092→      // Nearest level is too far — generate a synthetic level at ~5% from price
  1093→      const syntheticPrice = type === 'support'
  1094→        ? Math.round(currentPrice * 0.95 / psychStep(currentPrice)) * psychStep(currentPrice)
  1095→        : Math.round(currentPrice * 1.05 / psychStep(currentPrice)) * psychStep(currentPrice);

  1097→        price: syntheticPrice,
  1098→        type,
  1099→        score: 3,
  1100→        strength: 3,
  1101→        overlapCount: 1,
  1102→        methods: ['Psychological'],
  1103→        fibRatio: '—',
  1104→        fibLabel: '',
  1105→      };
  1106→      sorted.unshift(synthetic);
  1107→    }
  1108→
  1109→    // ── Rule 1: Select levels with 5-10% gaps between consecutive ──
  1110→    const result: MergedSRLevel[] = [sorted[0]];
  1111→
  1112→    for (let i = 1; i < sorted.length && result.length < 6; i++) {
  1113→      const prevPrice = result[result.length - 1].price;
  1114→      const currPrice = sorted[i].price;
  1115→      const gap = Math.abs(currPrice - prevPrice) / prevPrice;
  1116→
  1117→      if (gap < 0.05) {
  1118→        // Too close — keep the stronger one
  1119→        if (sorted[i].score > result[result.length - 1].score) {
  1120→          result[result.length - 1] = sorted[i];
  1121→        }
  1122→      } else if (gap <= 0.10) {
  1123→        // Perfect gap (5-10%) — accept
  1124→        result.push(sorted[i]);
  1125→      } else {
  1126→        // Gap > 10% — try to find an intermediate level
  1127→        let found = false;
  1128→        for (let j = i + 1; j < sorted.length; j++) {
  1129→          const midPrice = sorted[j].price;
  1130→          const gapFromPrev = Math.abs(midPrice - prevPrice) / prevPrice;
  1131→          const gapToNext = Math.abs(currPrice - midPrice) / midPrice;
  1132→          if (gapFromPrev >= 0.05 && gapFromPrev <= 0.10) {
  1133→            result.push(sorted[j]);
  1134→            found = true;
  1135→            // Skip levels that are too close to the inserted one
  1136→            i = j;
  1137→            break;
  1138→          }
  1139→        }
  1140→        if (!found) {
  1141→          // No intermediate found — generate a synthetic level at ~7.5% from prev
  1142→          const synthGap = 0.075;
  1143→          const syntheticPrice = type === 'support'
  1144→            ? Math.round((prevPrice * (1 - synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice)
  1145→            : Math.round((prevPrice * (1 + synthGap)) / psychStep(prevPrice)) * psychStep(prevPrice);
  1146→          // Make sure it's not too close to next level
  1147→          const gapToNext = Math.abs(currPrice - syntheticPrice) / syntheticPrice;
  1148→          if (gapToNext >= 0.05) {
  1149→            const synthetic: MergedSRLevel = {
  1150→              price: syntheticPrice,
  1151→              type,
  1152→              score: 2,
  1153→              strength: 2,
  1154→              overlapCount: 1,
  1155→              methods: ['Psychological'],
  1156→              fibRatio: '—',
  1157→              fibLabel: '',
  1158→            };
  1159→            result.push(synthetic);
  1160→          }
  1161→        }
  1162→      }
  1163→    }
  1164→
  1165→    return result;
  1166→  };
  1167→
  1168→  const finalSupports = filterAndSelectLevels(supports, 'support');
  1169→  const finalResistances = filterAndSelectLevels(resistances, 'resistance');
  1170→
  1171→  // ── Step 9: Build LevelStrength objects ────────────────────────────────
  1172→  const toLevelStrength = (m: MergedSRLevel): LevelStrength => {
  1173→    let grade: string;
  1174→    if (m.score >= 8.5) grade = 'Very Strong';
  1175→    else if (m.score >= 7) grade = 'Strong';
  1176→    else if (m.score >= 5) grade = 'Moderate';
  1177→    else grade = 'Weak';
  1178→
  1179→    return {
  1180→      price: m.price,
  1181→      strength: m.strength,
  1182→      score: m.score,
  1183→      grade,
  1184→      isTarget: m.score >= 7,
  1185→      overlapCount: m.overlapCount,
  1186→      methods: m.methods,
  1187→      fibRatio: m.fibRatio,
  1188→      fibLabel: m.fibLabel,
  1189→    };
  1190→  };
  1191→
  1192→  const supportStrengths = finalSupports.map(toLevelStrength);
  1193→  const resistanceStrengths = finalResistances.map(toLevelStrength);
  1194→
  1195→  // Price targets: all levels with isTarget=true
  1196→  const priceTargets = [...supportStrengths, ...resistanceStrengths].filter(l => l.isTarget);
  1197→
  1198→  return {
  1199→    resistances: resistanceStrengths.map(l => l.price),
  1200→    supports: supportStrengths.map(l => l.price),
  1201→    supportStrengths,
  1202→    resistanceStrengths,
  1203→    priceTargets,
  1204→  };
  1205→}

  1244→  // Find nearest support below price

  1246→  if (belowPrice.length > 0) nearestS = belowPrice[0];

  1251→// ─── Helper: divergence detection on last 20 bars (SIMPLIFIED for speed) ───────
  1252→// Returns true if bullish divergence detected, false if bearish, null if none
  1253→function detectDivergenceSimple(data: OHLCV[], barIdx: number): boolean | null {
  1254→  const lookback = 20;
  1255→  const start = Math.max(0, barIdx - lookback);
  1256→  const slice = data.slice(start, barIdx + 1);
  1257→  if (slice.length < 10) return null;
  1258→
  1259→  const mid = Math.floor(slice.length / 2);
  1260→  const firstHalf = slice.slice(0, mid);
  1261→  const secondHalf = slice.slice(mid);
  1262→
  1263→  const priceHigh1 = Math.max(...firstHalf.map(d => d.high));
  1264→  const priceHigh2 = Math.max(...secondHalf.map(d => d.high));
  1265→  const priceLow1 = Math.min(...firstHalf.map(d => d.low));
  1266→  const priceLow2 = Math.min(...secondHalf.map(d => d.low));
  1267→
  1268→  // Bearish divergence: price higher high but RSI proxy (MFI-like) lower
  1269→  // Use close/avg ratio as simple momentum proxy instead of computing full RSI
  1270→  const avgClose1 = firstHalf.reduce((s, d) => s + d.close, 0) / firstHalf.length;
  1271→  const avgClose2 = secondHalf.reduce((s, d) => s + d.close, 0) / secondHalf.length;
  1272→  const momentum1 = priceHigh1 > 0 ? (priceHigh1 - avgClose1) / priceHigh1 : 0;
  1273→  const momentum2 = priceHigh2 > 0 ? (priceHigh2 - avgClose2) / priceHigh2 : 0;
  1274→
  1275→  if (priceHigh2 > priceHigh1 && momentum2 < momentum1 * 0.8) {
  1276→    return false; // bearish divergence
  1277→  }
  1278→
  1279→  const negMomentum1 = avgClose1 > 0 ? (avgClose1 - priceLow1) / avgClose1 : 0;
  1280→  const negMomentum2 = avgClose2 > 0 ? (avgClose2 - priceLow2) / avgClose2 : 0;
  1281→
  1282→  if (priceLow2 < priceLow1 && negMomentum2 < negMomentum1 * 0.8) {
  1283→    return true; // bullish divergence
  1284→  }
  1285→
  1286→  return null;
  1287→}

  1438→    const prevStoch = calcStochastic(prevSlice);
  1439→    if (prevStoch.k < prevStoch.d && stoch.k > stoch.d) {
  1440→      f_stochCross = 0.9; // K crossed above D (bullish)
  1441→    }
  1442→  }
  1443→
  1444→  // f_macdCross — MACD/Signal crossover detection
  1445→  let f_macdCross = 0.1;
  1446→  if (barIdx >= 1) {
  1447→    const prevCloses = closes.slice(0, -1);
  1448→    if (prevCloses.length >= 35) {
  1449→      const prevMacd = calcMACD(prevCloses);
  1450→      if (prevMacd.line <= prevMacd.signal && macd.line > macd.signal) {
  1451→        f_macdCross = 0.9; // MACD crossed above signal (bullish)
  1452→      }
  1453→    }
  1454→  }
  1455→
  1456→  // f_div — Divergence detection (bullish div = 0.85, else 0.15)
  1457→  const divResult = detectDivergenceSimple(data, barIdx);
  1458→  const f_div = divResult === true ? 0.85 : 0.15;
  1459→
  1460→  // ── Assemble feature vector matching FEATURES_ORDER ──────────────────────
  1461→  return {
  1462→    rsi: f_rsi,
  1463→    mfi: f_mfi,
  1464→    cci: f_cci,
  1465→    adx: s_adx,
  1466→    macd: f_macd,
  1467→    stoch: f_stoch,
  1468→    bb: s_bb,
  1469→    ma21: s_ma21,
  1470→    ma100: s_ma100,
  1471→    ema: s_ema,
  1472→    atr: s_atr,
  1473→    trend: s_trend,
  1474→    sr: s_sr,
  1475→    stochCross: f_stochCross,

  1479→}
  1480→
  1481→// ═══════════════════════════════════════════════════════════════════════════════
  1482→// TRAINING DATA CREATION (Layer 0 — Data Preparation)
  1483→// ═══════════════════════════════════════════════════════════════════════════════
  1484→
  1485→/**
  1486→ * Creates supervised training data from historical OHLCV data.
  1487→ * For each bar, computes features and labels based on future price movement.
  1488→ * Label = 1 if price rises >1% within `horizon` bars, else 0.
  1489→ * Samples every `sampleStep` bars to reduce computation.
  1490→ */
  1491→function createTrainingData(
  1492→  data: OHLCV[],
  1493→  horizon: number = 5,
  1494→  sampleStep: number = 5
  1495→): { X: number[][]; y: number[] } {
  1496→  const X: number[][] = [];
  1497→  const y: number[] = [];
  1498→
  1499→  // Start from horizon+30 to ensure enough data for indicators
  1500→  // End at data.length - horizon - 1 to have future bars for labeling
  1501→  // Sample every sampleStep bars for performance
  1502→  for (let i = horizon + 30; i < data.length - horizon - 1; i += sampleStep) {
  1503→    const features = computeFeaturesAtBar(data, i);
  1504→
  1505→    // Convert to array following FEATURES_ORDER
  1506→    const featureVec: number[] = [];
  1507→    let valid = true;
  1508→    for (const key of FEATURES_ORDER) {
  1509→      const val = features[key];
  1510→      if (val === undefined || val === null || Number.isNaN(val)) {
  1511→        valid = false;
  1512→        break;
  1513→      }
  1514→      featureVec.push(val);
  1515→    }
  1516→    if (!valid) continue;
  1517→
  1518→    // Label: 1 if future close > 1% above current close
  1519→    const currentClose = data[i].close;
  1520→    const futureClose = data[i + horizon].close;
  1521→    const label = futureClose > currentClose * 1.01 ? 1 : 0;
  1522→
  1523→    X.push(featureVec);
  1524→    y.push(label);
  1525→  }
  1526→
  1527→  return { X, y };
  1528→}
  1529→
  1530→// ═══════════════════════════════════════════════════════════════════════════════
  1531→// VDss GRAPH STRUCTURE (Layer 5 — State Machine)
  1532→// ═══════════════════════════════════════════════════════════════════════════════
  1533→
  1534→/**
  1535→ * VDss Graph: Each node represents a market state.
  1536→ * Edges represent transitions with types: 'up', 'down', 'pullback', 'risk', 'terminal'.
  1537→ * Terminal nodes map to scenario results (R1=strong bullish, R5=deep correction).
  1538→ */
  1539→const VDSS_GRAPH = {
  1540→  // outgoing edges from each node: [target, type]
  1541→  outgoing: {
  1542→    A: [['B', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1543→    B: [['C', 'up'], ['G', 'pullback'], ['A', 'risk']] as [string, string][],
  1544→    C: [['D', 'up'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1545→    D: [['E', 'up'], ['B', 'pullback'], ['G', 'pullback'], ['L', 'risk']] as [string, string][],
  1546→    E: [['F', 'up'], ['D', 'pullback'], ['G', 'pullback'], ['H', 'down'], ['R1', 'terminal']] as [string, string][],
  1547→    F: [['R1', 'terminal'], ['E', 'pullback'], ['D', 'down'], ['G', 'risk']] as [string, string][],
  1548→    G: [['B', 'up'], ['A', 'pullback'], ['H', 'down'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1549→    H: [['G', 'pullback'], ['I', 'down'], ['B', 'up'], ['L', 'risk'], ['R2', 'terminal']] as [string, string][],
  1550→    I: [['H', 'pullback'], ['J', 'down'], ['G', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1551→    J: [['I', 'pullback'], ['K', 'down'], ['H', 'up'], ['L', 'risk'], ['R3', 'terminal']] as [string, string][],
  1552→    K: [['J', 'pullback'], ['I', 'up'], ['L', 'risk'], ['R4', 'terminal'], ['R5', 'terminal']] as [string, string][],
  1553→    L: [['C', 'up'], ['H', 'pullback'], ['J', 'down'], ['K', 'risk'], ['R5', 'terminal']] as [string, string][],
  1554→  },
  1555→  // which result scenario each terminal node maps to
  1556→  terminalMap: { R1: 'R1', R2: 'R2', R3: 'R3', R4: 'R4', R5: 'R5' } as Record<string, string>,
  1557→  startNode: 'A' as string,
  1558→};
  1559→
  1560→// ═══════════════════════════════════════════════════════════════════════════════
  1561→// LAYER 5 — Edge Weight Calculation
  1562→// ═══════════════════════════════════════════════════════════════════════════════
  1563→
  1564→/**
  1565→ * Calculates transition edge weights for the VDss graph.
  1566→ * Uses bull/bear consensus, ADX trend strength, and ML-derived coefficients
  1567→ * to determine the relative probability of up/down/pullback/risk transitions.
  1568→ */
  1569→function calculateEdgeWeights(
  1570→  bullConsensus: number,
  1571→  adx: number,
  1572→  mlModel: AdaptiveWeightModel
  1573→): { up: number; down: number; pullback: number; risk: number } {
  1574→  // Extract ML coefficient insights if model is trained
  1575→  let trend_coef = 1 / 3;
  1576→  let momentum_coef = 1 / 3;
  1577→  let volatility_coef = 1 / 3;
  1578→
  1579→  if (mlModel.isTrained && mlModel.weights) {
  1580→    const coefs = mlModel.getCoefficients();
  1581→    // Map feature indices to coefficient categories
  1582→    // FEATURES_ORDER: ['rsi','mfi','cci','adx','macd','stoch','bb','ma21','ma100','ema','atr','trend','sr','stochCross','macdCross','div']
  1583→    const trendIdx = 11;  // 'trend' in FEATURES_ORDER
  1584→    const rsiIdx = 0;     // 'rsi' in FEATURES_ORDER
  1585→    const atrIdx = 10;    // 'atr' in FEATURES_ORDER
  1586→
  1587→    const absTrend = Math.abs(coefs[trendIdx] ?? 0);
  1588→    const absRsi = Math.abs(coefs[rsiIdx] ?? 0);
  1589→    const absAtr = Math.abs(coefs[atrIdx] ?? 0);
  1590→
  1591→    const coefSum = absTrend + absRsi + absAtr;
  1592→    if (coefSum > 1e-10) {
  1593→      trend_coef = absTrend / coefSum;
  1594→      momentum_coef = absRsi / coefSum;
  1595→      volatility_coef = absAtr / coefSum;
  1596→    }
  1597→  }
  1598→
  1599→  // Base weights driven by bull consensus
  1600→  const up_base = bullConsensus * 0.7 + 0.15;
  1601→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
  1602→  const pullback_base = 0.25;
  1603→  const risk_base = 0.12 * (1.2 - adx / 100);
  1604→
  1605→  // Apply ML coefficient adjustments
  1606→  const up = clamp(
  1607→    up_base * (1 + (momentum_coef - 1 / 3) * 0.3),
  1608→    0.05, 0.95
  1609→  );
  1610→  const down = clamp(
  1611→    down_base * (1 + (trend_coef - 1 / 3) * 0.3),
  1612→    0.05, 0.95
  1613→  );
  1614→  const pullback = clamp(
  1615→    pullback_base * (1 + (volatility_coef - 1 / 3) * 0.2),
  1616→    0.05, 0.50
  1617→  );
  1618→  const risk = clamp(
  1619→    risk_base * (1 - (trend_coef - 1 / 3) * 0.5),
  1620→    0.02, 0.30
  1621→  );
  1622→
  1623→  return { up, down, pullback, risk };
  1624→}
  1625→
  1626→// ═══════════════════════════════════════════════════════════════════════════════
  1627→// LAYER 6 — Path Probability Calculation (Graph DFS + Calibration)
  1628→// ═══════════════════════════════════════════════════════════════════════════════
  1629→
  1630→/**
  1631→ * Performs iterative DFS from the start node through the VDss graph,
  1632→ * computing raw path probabilities, then calibrating them against
  1633→ * Layer 4 scenario probabilities.
  1634→ *
  1635→ * Returns:
  1636→ * - calibratedPaths: Map<pathString, probability>
  1637→ * - calibrationFactors: per-scenario multiplier used
  1638→ * - scenarioSums: raw probability sums per scenario before calibration
  1639→ */
  1640→function calculatePathProbabilities(
  1641→  edgeWeights: { up: number; down: number; pullback: number; risk: number },
  1642→  scenarioProbs: Record<string, number>
  1643→): {
  1644→  calibratedPaths: Map<string, number>;
  1645→  calibrationFactors: Record<string, number>;
  1646→  scenarioSums: Record<string, number>;
  1647→} {
  1648→  const { outgoing, terminalMap, startNode } = VDSS_GRAPH;
  1649→  const MAX_PATHS = 200;
  1650→
  1651→  // Type for DFS stack entries: [currentNode, pathString, pathProbability]
  1652→  type StackEntry = [string, string, number];
  1653→
  1654→  // Collect all terminal paths with raw probabilities
  1655→  const rawPaths: { path: string; scenario: string; prob: number }[] = [];
  1656→
  1657→  // Iterative DFS
  1658→  const stack: StackEntry[] = [[startNode, startNode, 1.0]];
  1659→
  1660→  while (stack.length > 0 && rawPaths.length < MAX_PATHS) {
  1661→    const [node, pathStr, pathProb] = stack.pop()!;
  1662→
  1663→    const edges = outgoing[node as keyof typeof outgoing];
  1664→    if (!edges) continue;
  1665→
  1666→    // Normalize outgoing edge weights to probabilities for this node
  1667→    // Map edge types to their weights
  1668→    let totalWeight = 0;
  1669→    const edgeWeightsList: number[] = [];
  1670→    for (const [, edgeType] of edges) {
  1671→      const w = edgeWeights[edgeType as keyof typeof edgeWeights] ?? 0.1;
  1672→      edgeWeightsList.push(w);
  1673→      totalWeight += w;
  1674→    }
  1675→
  1676→    if (totalWeight < 1e-10) continue;
  1677→
  1678→    // Expand edges
  1679→    for (let i = 0; i < edges.length; i++) {
  1680→      const [target, edgeType] = edges[i];
  1681→      const edgeProb = edgeWeightsList[i] / totalWeight;
  1682→      const newProb = pathProb * edgeProb;
  1683→      const newPath = pathStr + '→' + target;
  1684→
  1685→      if (edgeType === 'terminal') {
  1686→        // Reached a terminal node
  1687→        const scenario = terminalMap[target] ?? 'R3';
  1688→        rawPaths.push({ path: newPath, scenario, prob: newProb });
  1689→      } else {
  1690→        // Continue DFS
  1691→        stack.push([target, newPath, newProb]);
  1692→      }
  1693→    }
  1694→  }
  1695→
  1696→  // Group raw probabilities by scenario
  1697→  const rawSums: Record<string, number> = { R1: 0, R2: 0, R3: 0, R4: 0, R5: 0 };
  1698→  for (const rp of rawPaths) {
  1699→    rawSums[rp.scenario] = (rawSums[rp.scenario] ?? 0) + rp.prob;
  1700→  }
  1701→
  1702→  // Compute calibration factors: scenarioProbs[scenario] / rawSum[scenario]
  1703→  const calibrationFactors: Record<string, number> = {};
  1704→  for (const scenario of ['R1', 'R2', 'R3', 'R4', 'R5']) {
  1705→    const rawSum = rawSums[scenario] ?? 0;
  1706→    const targetProb = (scenarioProbs[scenario] ?? 20) / 100; // convert from percentage
  1707→    calibrationFactors[scenario] = rawSum > 1e-10 ? targetProb / rawSum : 1.0;
  1708→  }
  1709→
  1710→  // Apply calibration to each path
  1711→  const calibratedPaths = new Map<string, number>();
  1712→  for (const rp of rawPaths) {
  1713→    const factor = calibrationFactors[rp.scenario] ?? 1.0;
  1714→    calibratedPaths.set(rp.path, rp.prob * factor);
  1715→  }
  1716→
  1717→  return { calibratedPaths, calibrationFactors, scenarioSums: rawSums };
  1718→}
  1719→
  1720→// ═══════════════════════════════════════════════════════════════════════════════
  1721→// LAYER 4 — Scenario Probability Calculation (Adaptive with ML)
  1722→// ═══════════════════════════════════════════════════════════════════════════════
  1723→
  1724→/**
  1725→ * Calculates the 5 VDss scenario probabilities (R1-R5) that sum to 100.
  1726→ * Uses bull/bear consensus, key indicator values, S/R levels, and
  1727→ * ML-derived adaptive factors for fine-tuning.
  1728→ */
  1729→function calculateScenarioProbabilities(
  1730→  bullConsensus: number,
  1731→  price: number,
  1732→  R1: number,
  1733→  S1: number,
  1734→  MA100: number,
  1735→  rsi: number,
  1736→  mfi: number,
  1737→  stochK: number,
  1738→  mlModel: AdaptiveWeightModel
  1739→): { R1: number; R2: number; R3: number; R4: number; R5: number; factors: { momentum: number; volatility: number; trend: number } } {
  1740→  // Distance metrics to key levels (spec uses exp(-3 * ...))
  1741→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0;
  1742→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0;
  1743→
  1744→  // Binary risk flags (per spec) — MFI only used when volume is available
  1745→  const overboughtRisk = (rsi > 70 || (mfi > 80 && mfi !== 50) || stochK > 80) ? 1 : 0;
  1746→  const oversoldBounce = (rsi < 30 || (mfi < 20 && mfi !== 50) || stochK < 20) ? 1 : 0;
  1747→
  1748→  // Extract adaptive factors from ML model if trained
  1749→  let momentum = 0.7;
  1750→  let volatility = 0.5;
  1751→  let trend = 0.6;
  1752→
  1753→  if (mlModel.isTrained && mlModel.weights) {
  1754→    const coefs = mlModel.getCoefficients();
  1755→    momentum = sigmoid(coefs[0] ?? 0);   // rsi coefficient → momentum factor
  1756→    volatility = sigmoid(coefs[10] ?? 0); // atr coefficient → volatility factor
  1757→    trend = sigmoid(coefs[11] ?? 0);     // trend coefficient → trend factor
  1758→  }
  1759→
  1760→  // Below MA100 flag
  1761→  const belowMA100 = price < MA100 ? 1 : 0;
  1762→
  1763→  // ── Raw scenario probabilities (matching user spec) ─────────────────────
  1764→
  1765→  // R1: Strong Bullish — aggressive breakout
  1766→  const raw_R1 =
  1767→    bullConsensus * distR1 * (1 - overboughtRisk * momentum * 0.7) +
  1768→    oversoldBounce * 0.3 * distR1 * momentum;
  1769→
  1770→  // R2: Gradual Uptrend — steady rise
  1771→  const raw_R2 =
  1772→    ((1 - Math.abs(bullConsensus - 0.65)) * 0.8 +
  1773→    (bullConsensus > 0.4 ? 0.2 : 0)) * trend;
  1774→
  1775→  // R3: Range-bound
  1776→  const raw_R3 =
  1777→    (1 - Math.abs(bullConsensus - 0.5)) * 0.6 * (1 + (1 - volatility) * 0.2);
  1778→
  1779→  // R4: Moderate Correction
  1780→  const raw_R4 =
  1781→    ((1 - bullConsensus) * distS1 * 0.7 +
  1782→    overboughtRisk * 0.5 * momentum) *
  1783→    (0.8 + 0.2 * (1 - trend));
  1784→
  1785→  // R5: Deep Correction
  1786→  const raw_R5 =
  1787→    ((1 - bullConsensus) ** 2 * 0.5 + belowMA100 * 0.2) *
  1788→    (1 + (1 - trend) * 0.3);
  1789→
  1790→  // ── Normalize to sum = 100 ───────────────────────────────────────────
  1791→  const sum = raw_R1 + raw_R2 + raw_R3 + raw_R4 + raw_R5;
  1792→  const pR1 = sum > 0 ? Math.round(raw_R1 / sum * 100) : 20;
  1793→  const pR2 = sum > 0 ? Math.round(raw_R2 / sum * 100) : 20;
  1794→  const pR3 = sum > 0 ? Math.round(raw_R3 / sum * 100) : 20;
  1795→  const pR4 = sum > 0 ? Math.round(raw_R4 / sum * 100) : 20;
  1796→  const pR5 = 100 - (pR1 + pR2 + pR3 + pR4);
  1797→
  1798→  return {
  1799→    R1: pR1, R2: pR2, R3: pR3, R4: pR4, R5: pR5,
  1800→    factors: { momentum, volatility, trend },
  1801→  };
  1802→}
