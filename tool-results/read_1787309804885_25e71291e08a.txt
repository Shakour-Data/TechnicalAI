   145→/* --─ Helpers -------------------------------------------- */
   146→
   147→function formatNum(num: number): string {
   148→  if (num == null) return '—';
   149→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 0 }).format(num);
   150→}
   151→
   152→function formatIdx(num: number): string {
   153→  if (num == null) return '—';
   154→  return new Intl.NumberFormat('fa-IR', { maximumFractionDigits: 2 }).format(num);
   155→}
   156→

   162→    </svg>
   163→  );
   164→}

   166→/* --─ Component ------------------------------------------ */
   167→
   168→export default function SymbolSearch({
   169→  onSelect,
   170→  placeholder = 'جستجوی ابزار مالی ...',
   171→  className,
   172→}: SymbolSearchProps) {
   173→  const [query, setQuery] = React.useState('');
   174→  const [open, setOpen] = React.useState(false);
   175→  const [results, setResults] = React.useState<InstrumentItem[]>([]);
   176→  const [loading, setLoading] = React.useState(false);
   177→  const [activeIndex, setActiveIndex] = React.useState(-1);
   178→  const [totalMatched, setTotalMatched] = React.useState(0);
   179→  const [activeCategory, setActiveCategory] = React.useState<CategoryKey>('all');
   180→  const [activeIndustry, setActiveIndustry] = React.useState<string | null>(null);
   181→  const [showIndustryPicker, setShowIndustryPicker] = React.useState(false);
   182→  const [indexSubFilter, setIndexSubFilter] = React.useState<'all' | 'main' | 'sector'>('all');
   183→
   184→  const cacheRef = React.useRef<InstrumentsData | null>(null);
   185→  const tgjuCacheRef = React.useRef<TgjuData | null>(null);
   186→  const fetchRef = React.useRef<Promise<InstrumentsData> | null>(null);
   187→  const tgjuFetchRef = React.useRef<Promise<TgjuData | null> | null>(null);
   188→  const inputRef = React.useRef<HTMLInputElement>(null);
   189→  const dropdownRef = React.useRef<HTMLDivElement>(null);
   190→  const itemRefs = React.useRef<(HTMLDivElement | null)[]>([]);
   191→  const industryScrollRef = React.useRef<HTMLDivElement>(null);
   192→
   193→  /* -- Data fetching -- */
   194→  const fetchData = React.useCallback(async (): Promise<InstrumentsData> => {
   195→    if (cacheRef.current) return cacheRef.current;
   196→    if (fetchRef.current) return fetchRef.current;
   197→    const p = async () => {
   198→      try {
   199→        const r = await fetch('/api/instruments');
   200→        if (!r.ok) { console.warn('[SymbolSearch] /api/instruments failed:', r.status); return { indices: [], stocks: [], etfs: [], bonds: [], futures: [], salaf: [], mortgage: [], industries: [] }; }
   201→        const d: InstrumentsData = await r.json();
   202→        cacheRef.current = d;
   203→        return d;
   204→      } catch (e) {
   205→        console.warn('[SymbolSearch] fetchData error:', e);
   206→        return { indices: [], stocks: [], etfs: [], bonds: [], futures: [], salaf: [], mortgage: [], industries: [] };
   207→      } finally {
   208→        fetchRef.current = null;
   209→      }
   210→    };
   211→    fetchRef.current = p();
   212→    return fetchRef.current;
   213→  }, []);
   214→
   215→  const fetchTgjuData = React.useCallback(async (): Promise<TgjuData | null> => {
   216→    if (tgjuCacheRef.current) return tgjuCacheRef.current;
   217→    if (tgjuFetchRef.current) return tgjuFetchRef.current;
   218→    const p = async () => {
   219→      try {
   220→        const r = await fetch('/api/tgju-instruments');
   221→        if (!r.ok) return null;
   222→        const d: TgjuData = await r.json();
   223→        tgjuCacheRef.current = d;
   224→        return d;
   225→      } catch {
   226→        return null;
   227→      } finally {
   228→        tgjuFetchRef.current = null;
   229→      }
   230→    };
   231→    tgjuFetchRef.current = p();
   232→    return p();
   233→  }, []);
   234→
   235→  /* -- Check if category is TGJU-based -- */
   236→  const isTgjuCategory = (cat: CategoryKey) => TGJU_CATEGORIES.has(cat);
   237→
   238→  /* -- Get all items for current category -- */
   239→  const getItemsForCategory = React.useCallback(async (cat: CategoryKey): Promise<InstrumentItem[]> => {
   240→    if (isTgjuCategory(cat)) {
   241→      const tgju = tgjuCacheRef.current || await fetchTgjuData();
   242→      if (!tgju) return [];
   243→      switch (cat) {
   244→        case 'currency': return tgju.currencies;
   245→        case 'gold': return [...tgju.gold, ...tgju.silver];
   246→        case 'gold_etf': return tgju.goldEtfs;
   247→        case 'silver': return tgju.silver;
   248→        case 'crypto': return tgju.crypto;
   249→        case 'world_index': return tgju.worldIndices;
   250→        case 'foreign_stock': return tgju.foreignStocks;
   251→        case 'forex': return tgju.forex;
   252→        case 'energy': return tgju.energy;
   253→        case 'metal': return tgju.metals;
   254→        case 'commodity': return tgju.commodities;
   255→        default: return [];
   256→      }
   257→    }
   258→
   259→    // TSE categories
   260→    const data = cacheRef.current || await fetchData();
   261→    switch (cat) {
   262→      case 'indices': {
   263→        let idxItems = data.indices;
   264→        if (indexSubFilter === 'main') {
   265→          idxItems = idxItems.filter((i) => i.cs === 'شاخص اصلی');
   266→        } else if (indexSubFilter === 'sector') {
   267→          idxItems = idxItems.filter((i) => i.cs === 'صنعت');
   268→        }
   269→        return idxItems;
   270→      }
   271→      case 'stocks':
   272→        return activeIndustry
   273→          ? data.stocks.filter((s) => s.cs === activeIndustry)
   274→          : data.stocks;
   275→      case 'etf': return data.etfs;
   276→      case 'bond': return [...data.bonds, ...data.mortgage];
   277→      case 'derivative': return [...data.futures, ...data.salaf];
   278→      case 'all':
   279→      default: {
   280→        const tgju = tgjuCacheRef.current;
   281→        return [
   282→          ...data.indices,
   283→          ...data.stocks,
   284→          ...data.etfs,
   285→          ...(tgju?.currencies || []),
   286→          ...(tgju?.forex || []),
   287→          ...(tgju?.crypto || []),
   288→          ...(tgju?.gold || []),
   289→          ...(tgju?.silver || []),
   290→          ...(tgju?.goldEtfs || []),
   291→          ...(tgju?.worldIndices || []),
   292→          ...(tgju?.foreignStocks || []),
   293→          ...(tgju?.energy || []),
   294→          ...(tgju?.metals || []),
   295→          ...(tgju?.commodities || []),
   296→          ...data.bonds,
   297→          ...data.futures,
   298→          ...data.salaf,
   299→          ...data.mortgage,
   300→        ];
   301→      }
   302→    }
   303→  }, [activeIndustry, indexSubFilter, fetchData, fetchTgjuData]);
   304→
   305→  /* -- Popular items -- */
   306→  const getPopular = React.useCallback(async () => {
   307→    const data = cacheRef.current || await fetchData();
   308→    const tgju = tgjuCacheRef.current;
   309→    const all = [
   310→      ...(data.indices.map((i) => ({ ...i, tval: i.index || 0 }))),
   311→      ...data.stocks,
   312→      ...data.etfs,
   313→      ...(tgju?.currencies || []),
   314→      ...(tgju?.crypto?.slice(0, 5) || []),
   315→      ...(tgju?.foreignStocks?.slice(0, 3) || []),
   316→      ...(tgju?.gold?.slice(0, 3) || []),
   317→    ];
   318→    const sorted = [...all].sort((a, b) => (b.tval || 0) - (a.tval || 0)).slice(0, 15);
   319→    setResults(sorted);
   320→    setTotalMatched(all.length);
   321→  }, [fetchData]);
   322→
   323→  /* -- Filter by search query -- */
   324→  const doFilter = React.useCallback((items: InstrumentItem[], q: string) => {
   325→    const n = q.trim().toLowerCase();
   326→    const f = items.filter(
   327→      (s) =>
   328→        s.l18?.toLowerCase().includes(n) ||
   329→        s.l30?.toLowerCase().includes(n) ||
   330→        s.cs?.toLowerCase().includes(n),
   331→    );
   332→    setTotalMatched(f.length);
   333→    setResults(f.slice(0, MAX_RESULTS));
   334→    setActiveIndex(-1);
   335→  }, []);
   336→
   337→  /* -- Show dropdown with data -- */
   338→  const showDropdown = React.useCallback(async (q: string) => {
   339→    setOpen(true);
   340→    await Promise.allSettled([fetchData(), fetchTgjuData()]);
   341→    const items = await getItemsForCategory(activeCategory);
   342→    if (q.trim().length === 0) {
   343→      if (activeCategory === 'all') {
   344→        await getPopular();
   345→      } else {
   346→        setResults(items.slice(0, MAX_RESULTS));
   347→        setTotalMatched(items.length);
   348→      }
   349→    } else {
   350→      doFilter(items, q);
   351→    }
   352→    setLoading(false);
   353→  }, [fetchData, fetchTgjuData, getItemsForCategory, activeCategory, getPopular, doFilter]);
   354→
   355→  /* -- Category change -- */
   356→  const handleCategoryChange = React.useCallback(async (cat: CategoryKey) => {
   357→    setActiveCategory(cat);
   358→    if (cat !== 'stocks') setActiveIndustry(null);
   359→    if (cat !== 'indices') setIndexSubFilter('all');
   360→    setShowIndustryPicker(false);
   361→
   362→    const items = await getItemsForCategory(cat);
   363→    if (query.trim().length > 0) {
   364→      doFilter(items, query);
   365→    } else if (cat === 'all') {
   366→      await getPopular();
   367→    } else {
   368→      setResults(items.slice(0, MAX_RESULTS));
   369→      setTotalMatched(items.length);
   370→    }
   371→    setActiveIndex(-1);
   372→  }, [getItemsForCategory, query, doFilter, getPopular]);
   373→
   374→  /* -- Industry change -- */
   375→  const handleIndustryChange = React.useCallback((industry: string | null) => {
   376→    setActiveIndustry(industry);
   377→    setShowIndustryPicker(false);
   378→    if (cacheRef.current) {
   379→      const items = industry
   380→        ? cacheRef.current.stocks.filter((s) => s.cs === industry)
   381→        : cacheRef.current.stocks;
   382→      if (query.trim().length > 0) {
   383→        doFilter(items, query);
   384→      } else {
   385→        setResults(items.slice(0, MAX_RESULTS));
   386→        setTotalMatched(items.length);
   387→      }
   388→      setActiveIndex(-1);
   389→    }
   390→  }, [query, doFilter]);
   391→
   392→  const handleChange = React.useCallback((value: string) => {
   393→    setQuery(value);
   394→    getItemsForCategory(activeCategory).then((items) => {
   395→      if (value.trim().length === 0) {
   396→        if (activeCategory === 'all') getPopular();
   397→        else {
   398→          setResults(items.slice(0, MAX_RESULTS));
   399→          setTotalMatched(items.length);
   400→        }
   401→      } else {
   402→        doFilter(items, value);
   403→      }
   404→    });
   405→  }, [activeCategory, getItemsForCategory, getPopular, doFilter]);
   406→
   407→  const handleFocus = React.useCallback(() => {
   408→    setLoading(true);
   409→    showDropdown(query);
   410→  }, [query, showDropdown]);
   411→
   412→  const selectSymbol = React.useCallback((s: InstrumentItem) => {
   413→    setQuery(s.l18);
   414→    setOpen(false);
   415→    onSelect?.(s.l18, s.category, s.insCode, s.tgjuKey);
   416→    inputRef.current?.blur();
   417→  }, [onSelect]);
   418→
   419→  const handleKeyDown = React.useCallback((e: React.KeyboardEvent) => {
   420→    if (!open || results.length === 0) {
   421→      if (e.key === 'ArrowDown' || e.key === 'Enter') { e.preventDefault(); handleFocus(); }
   422→      return;
   423→    }
   424→    if (e.key === 'ArrowDown') { e.preventDefault(); setActiveIndex((p) => p < results.length - 1 ? p + 1 : 0); }
   425→    else if (e.key === 'ArrowUp') { e.preventDefault(); setActiveIndex((p) => p > 0 ? p - 1 : results.length - 1); }
   426→    else if (e.key === 'Enter') { e.preventDefault(); if (activeIndex >= 0) selectSymbol(results[activeIndex]); }
   427→    else if (e.key === 'Escape') { e.preventDefault(); setOpen(false); inputRef.current?.blur(); }
   428→  }, [open, results, activeIndex, selectSymbol, handleFocus]);
   429→
   430→  React.useEffect(() => {
   431→    if (activeIndex >= 0) itemRefs.current[activeIndex]?.scrollIntoView({ block: 'nearest' });
   432→  }, [activeIndex]);
   433→
   434→  React.useEffect(() => {
   435→    if (!open) return;
   436→    const handler = (e: MouseEvent) => {
   437→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node) &&
   438→          inputRef.current && !inputRef.current.contains(e.target as Node)) {
   439→        setOpen(false);
   440→        setShowIndustryPicker(false);
   441→      }
   442→    };
   443→    document.addEventListener('mousedown', handler);
   444→    return () => document.removeEventListener('mousedown', handler);
   445→  }, [open]);
   446→
   447→  const clear = () => { setQuery(''); setActiveIndustry(null); setActiveCategory('all'); setResults([]); inputRef.current?.focus(); };
   448→  const isPopular = query.trim().length === 0 && activeCategory === 'all';
   449→  const isSearch = query.trim().length > 0;
   450→
   451→  /* -- Render instrument price -- */
   452→  const renderPrice = (item: InstrumentItem) => {
   453→    const isTgju = isTgjuCategory(item.category as CategoryKey);
   454→    if (item.category === 'index' || item.category === 'tse_index') {
   455→      return (
   456→        <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   457→          <span className='text-xs font-semibold text-gray-900'>{formatIdx(item.pl)}</span>
   458→          <span className={cn(
   459→            'text-[11px] font-bold px-1.5 py-0.5 rounded',
   460→            item.pcp > 0 ? 'bg-emerald-50 text-emerald-700' :
   461→            item.pcp < 0 ? 'bg-red-50 text-red-700' : 'text-gray-500',
   462→          )}>
   463→            {item.pcp > 0 ? '+' : ''}{toPersianDigits(item.pcp?.toFixed(2))}٪
   464→          </span>
   465→        </div>
   466→      );
   467→    }
   468→    // For static TGJU items with no live price, show dash
   469→    if (isTgju && item.pl === 0) {
   470→      return (
   471→        <div className='flex shrink-0 flex-col items-end gap-0.5'>
   472→          <span className='text-xs text-gray-500'>—</span>
   473→        </div>
   474→      );
   475→    }
   476→    const isUp = item.pcp > 0;
   477→    const isDown = item.pcp < 0;
   478→    return (
   479→      <div className='flex shrink-0 flex-col items-end gap-0.5 tabular-nums'>
   480→        <span className='text-xs font-semibold text-gray-900'>{formatNum(item.pl)}</span>
   481→        <span className={cn(
   482→          'text-[11px] font-bold px-1.5 py-0.5 rounded',
   483→          isUp ? 'bg-emerald-50 text-emerald-700' :
   484→          isDown ? 'bg-red-50 text-red-700' : 'text-gray-500',
   485→        )}>
   486→          {isUp ? '+' : ''}{toPersianDigits(item.pcp?.toFixed(2))}٪
   487→        </span>
   488→      </div>
   489→    );
   490→  };
   491→
   492→  const industries = cacheRef.current?.industries || [];
   493→
   494→  return (
   495→    <div dir='rtl' className={cn('relative', className)}>
   496→
   497→      <div className='relative'>
   498→        <Search className='absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-500 pointer-events-none' />
   499→        <Input
   500→          ref={inputRef}
   501→          type='text'
   502→          value={query}
   503→          onChange={(e) => handleChange(e.target.value)}
   504→          onFocus={handleFocus}
