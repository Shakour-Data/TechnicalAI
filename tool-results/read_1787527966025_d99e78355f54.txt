     1→// =============================================================================

     3→// =============================================================================
     4→// This module replaces the old 12-node graph and heuristic probability system.
     5→// The decision graph IS the primary probability engine — all 9 scenario
     6→// probabilities (R1–R9) are derived from aggregating path probabilities through
     7→// this directed acyclic graph (DAG).

    10→// === Helper Functions =========================================================
    11→
    12→function clamp(v: number, min: number, max: number): number {
    13→  return Math.max(min, Math.min(max, v));
    14→}
    15→
    16→function normalize(arr: number[]): number[] {
    17→  const sum = arr.reduce((a, b) => a + b, 0);
    18→  if (sum === 0) return arr.map(() => 1 / arr.length);
    19→  return arr.map((v) => v / sum);
    20→}
    21→
    22→function sigmoid(x: number): number {
    23→  return 1 / (1 + Math.exp(-x));
    24→}

   988→// === Probability Context Builder ==============================================
   989→
   990→function buildContext(input: GraphInput): ProbContext {
   991→  const adxNorm = clamp(input.adx / 60, 0, 1);
   992→  const srProximity = clamp(
   993→    1 - Math.min(input.distToR1, input.distToS1) * 10,
   994→    0,
   995→    1
   996→  );
   997→  const oscillatorExtreme = clamp(
   998→    (Math.abs(input.rsi - 50) / 50 + Math.abs(input.cci) / 200) / 2,
   999→    0,
  1000→    1
  1001→  );
  1002→  const divergenceProxy = oscillatorExtreme * (1 - adxNorm);
  1003→
  1004→  // Pre-compute for reuse across N_T_BULL and N_REVERSAL
  1005→  const overboughtRisk =
  1006→    clamp((input.rsi - 60) / 30, 0, 1) * 0.3 +
  1007→    clamp((input.mfi - 70) / 30, 0, 1) * 0.2 +
  1008→    clamp((input.stochK - 75) / 25, 0, 1) * 0.2;
  1009→
  1010→  // Pre-compute for reuse across N_T_BEAR and N_REVERSAL
  1011→  const oversoldBounce =
  1012→    clamp((30 - input.rsi) / 30, 0, 1) * 0.3 +
  1013→    clamp((20 - input.mfi) / 20, 0, 1) * 0.2 +
  1014→    clamp((20 - input.stochK) / 20, 0, 1) * 0.2;
  1015→
  1016→  return {
  1017→    input,
  1018→    adxNorm,
  1019→    srProximity,
  1020→    oscillatorExtreme,
  1021→    divergenceProxy,
  1022→    overboughtRisk,
  1023→    oversoldBounce,
  1024→  };
  1025→}

  1027→// === Node Probability Computation =============================================
  1028→// For each non-terminal node, returns an array of probabilities for its
  1029→// outgoing edges. The order MUST match the order of edges in createEdges().
  1030→// =============================================================================
  1031→
  1032→function computeNodeProbabilities(
  1033→  nodeId: string,
  1034→  ctx: ProbContext,
  1035→  _numChildren: number
  1036→): number[] {
  1037→  const { input, adxNorm, srProximity, overboughtRisk, oversoldBounce } = ctx;
  1038→  const atrSafe = input.atr > 0 ? input.atr : 0.0001;
  1039→
  1040→  switch (nodeId) {
  1041→    // ── ROOT → [N_TREND, N_BREAK, N_REVERSAL] ─────────────────────────────
  1042→    case 'ROOT': {
  1043→      const trendW = adxNorm * 0.45 + input.mlTrend * 0.15 + 0.12;
  1044→      const breakW = srProximity * 0.35 + (1 - adxNorm) * 0.1 + 0.1;
  1045→      const revW = ctx.divergenceProxy * 0.3 + (1 - adxNorm) * 0.15 + 0.08;
  1046→      return normalize([trendW, breakW, revW]);
  1047→    }
  1048→
  1049→    // ── N_TREND → [N_T_BULL, N_T_BEAR, N_T_FLAT] ──────────────────────────
  1050→    case 'N_TREND': {
  1051→      const bull = input.bullConsensus * (adxNorm * 0.4 + 0.6);
  1052→      const bear = (1 - input.bullConsensus) * (adxNorm * 0.4 + 0.6);
  1053→      const flat =
  1054→        (1 - Math.abs(input.bullConsensus - 0.5) * 2) * (1 - adxNorm * 0.7);
  1055→      return normalize([bull, bear, flat]);
  1056→    }
  1057→
  1058→    // ── N_T_BULL → [N_T_B_MOM_HIGH, N_T_B_MOM_MOD, N_T_B_OVERBOUGHT] ─────
  1059→    case 'N_T_BULL': {
  1060→      const momHigh =
  1061→        clamp(
  1062→          (input.momentum > 0 ? input.momentum / 50 : 0) +
  1063→            (input.macdHist > 0
  1064→              ? (input.macdHist / (atrSafe * 0.5)) * 0.3
  1065→              : 0),
  1066→          0,
  1067→          1
  1068→        ) * input.mlMomentum;
  1069→      const momMod = (1 - momHigh) * 0.6 * input.mlMomentum;
  1070→      return normalize([momHigh, momMod, overboughtRisk]);
  1071→    }
  1072→
  1073→    // ── N_T_B_MOM_HIGH → [R3, R4, R5] ────────────────────────────────────
  1074→    case 'N_T_B_MOM_HIGH': {
  1075→      const pR3 = 0.55 * input.mlTrend + input.confidenceIndex * 0.2;
  1076→      const pR4 = 0.15 + input.strengthIndex * 0.15;
  1077→      const pR5 = 1 - pR3 - pR4;
  1078→      return normalize([pR3, pR4, pR5]);
  1079→    }
  1080→
  1081→    // ── N_T_B_MOM_MOD → [R2, R1, R5] ─────────────────────────────────────
  1082→    case 'N_T_B_MOM_MOD': {
  1083→      const pR2 = 0.5 * input.confidenceIndex + 0.2;
  1084→      const pR1 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1085→      const pR5 = 1 - pR2 - pR1;
  1086→      return normalize([pR2, pR1, pR5]);
  1087→    }
  1088→
  1089→    // ── N_T_B_OVERBOUGHT → [R6, R9, R5] ──────────────────────────────────
  1090→    case 'N_T_B_OVERBOUGHT': {
  1091→      const pR6 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1092→      const pR9 = 0.15 + input.strengthIndex * 0.15;
  1093→      const pR5 = 1 - pR6 - pR9;
  1094→      return normalize([pR6, pR9, pR5]);
  1095→    }
  1096→
  1097→    // ── N_T_BEAR → [N_T_BE_MOM_LOW, N_T_BE_MOM_MOD, N_T_BE_OVERSOLD] ────
  1098→    case 'N_T_BEAR': {
  1099→      const momLow =
  1100→        clamp(
  1101→          (input.momentum < 0 ? Math.abs(input.momentum) / 50 : 0) +
  1102→            (input.macdHist < 0
  1103→              ? (Math.abs(input.macdHist) / (atrSafe * 0.5)) * 0.3
  1104→              : 0),
  1105→          0,
  1106→          1
  1107→        ) * input.mlMomentum;
  1108→      const momMod = (1 - momLow) * 0.6 * input.mlMomentum;
  1109→      return normalize([momLow, momMod, oversoldBounce]);
  1110→    }
  1111→
  1112→    // ── N_T_BE_MOM_LOW → [R8, R9, R5] ────────────────────────────────────
  1113→    case 'N_T_BE_MOM_LOW': {
  1114→      const pR8 = 0.55 * input.mlTrend + input.confidenceIndex * 0.2;
  1115→      const pR9 = 0.15 + input.strengthIndex * 0.15;
  1116→      const pR5 = 1 - pR8 - pR9;
  1117→      return normalize([pR8, pR9, pR5]);
  1118→    }
  1119→
  1120→    // ── N_T_BE_MOM_MOD → [R7, R6, R5] ────────────────────────────────────
  1121→    case 'N_T_BE_MOM_MOD': {
  1122→      const pR7 = 0.5 * input.confidenceIndex + 0.2;
  1123→      const pR6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1124→      const pR5 = 1 - pR7 - pR6;
  1125→      return normalize([pR7, pR6, pR5]);
  1126→    }
  1127→
  1128→    // ── N_T_BE_OVERSOLD → [R1, R4, R5] ───────────────────────────────────
  1129→    case 'N_T_BE_OVERSOLD': {
  1130→      const pR1 = 0.55 + (1 - input.mlMomentum) * 0.2;
  1131→      const pR4 = 0.15 + input.strengthIndex * 0.15;
  1132→      const pR5 = 1 - pR1 - pR4;
  1133→      return normalize([pR1, pR4, pR5]);
  1134→    }
  1135→
  1136→    // ── N_T_FLAT → [N_T_F_VOL_LOW, N_T_F_VOL_HIGH, N_T_F_VOL_MOD] ────────
  1137→    case 'N_T_FLAT': {
  1138→      const volLow =
  1139→        clamp(1 - (input.atr / input.price) * 20, 0, 1) *
  1140→        (1 - input.mlVolatility) *
  1141→        0.7;
  1142→      const volHigh =
  1143→        clamp((input.atr / input.price) * 20, 0, 1) * input.mlVolatility * 0.7;
  1144→      const volMod = (1 - volLow - volHigh) * 0.5 + 0.1;

  1147→
  1148→    // ── N_T_F_VOL_LOW → [R5] ─────────────────────────────────────────────
  1149→    case 'N_T_F_VOL_LOW': {
  1150→      return [1.0];
  1151→    }
  1152→
  1153→    // ── N_T_F_VOL_HIGH → [R4, R9, R5] ────────────────────────────────────
  1154→    case 'N_T_F_VOL_HIGH': {
  1155→      const pR4 = 0.3 * input.bullConsensus + 0.1;
  1156→      const pR9 = 0.3 * (1 - input.bullConsensus) + 0.1;

  1162→    case 'N_T_F_VOL_MOD': {

  1164→      const pR6 = 0.35 * (1 - input.bullConsensus) + 0.1;
  1165→      const pR5 = 1 - pR1 - pR6;
  1166→      return normalize([pR1, pR6, pR5]);

  1169→    // ── N_BREAK → [N_B_UP, N_B_DOWN, N_B_NONE] ────────────────────────────
  1170→    case 'N_BREAK': {
  1171→      const upProb = input.bullConsensus * srProximity * 0.7 + 0.05;
  1172→      const downProb = (1 - input.bullConsensus) * srProximity * 0.7 + 0.05;
  1173→      const noneProb = (1 - srProximity) * 0.5 + 0.1;
  1174→      return normalize([upProb, downProb, noneProb]);
  1175→    }
  1176→
  1177→    // ── N_B_UP → [N_B_U_VOL_C, N_B_U_VOL_W] ─────────────────────────────
  1178→    case 'N_B_UP': {
  1179→      let volConfirm: number;
  1180→      let volWeak: number;
  1181→      if (input.hasVolume) {
  1182→        volConfirm = clamp(input.mfi / 60, 0, 1) * 0.6 + 0.2;
  1183→        volWeak = 1 - volConfirm;
  1184→      } else {
  1185→        volConfirm = 0.4;
  1186→        volWeak = 0.6;
  1187→      }
  1188→      return normalize([volConfirm, volWeak]);
  1189→    }
  1190→
  1191→    // ── N_B_U_VOL_C → [R2, R3, R5] ──────────────────────────────────────
  1192→    case 'N_B_U_VOL_C': {
  1193→      const pR2 = 0.55 + input.confidenceIndex * 0.15;
  1194→      const pR3 = 0.2 + input.strengthIndex * 0.15;
  1195→      const pR5 = 1 - pR2 - pR3;
  1196→      return normalize([pR2, pR3, pR5]);
  1197→    }
  1198→
  1199→    // ── N_B_U_VOL_W → [R1, R6, R5] ──────────────────────────────────────
  1200→    case 'N_B_U_VOL_W': {
  1201→      const pR1 = 0.4 + input.bullConsensus * 0.15;
  1202→      const pR6 = 0.3 + (1 - input.bullConsensus) * 0.15;
  1203→      const pR5 = 1 - pR1 - pR6;
  1204→      return normalize([pR1, pR6, pR5]);
  1205→    }
  1206→
  1207→    // ── N_B_DOWN → [N_B_D_VOL_C, N_B_D_VOL_W] ────────────────────────────
  1208→    case 'N_B_DOWN': {
  1209→      let volConfirm: number;
  1210→      if (input.hasVolume) {
  1211→        volConfirm = clamp((100 - input.mfi) / 60, 0, 1) * 0.6 + 0.2;
  1212→      } else {
  1213→        volConfirm = 0.4;
  1214→      }
  1215→      const volWeak = 1 - volConfirm;
  1216→      return normalize([volConfirm, volWeak]);
  1217→    }
  1218→
  1219→    // ── N_B_D_VOL_C → [R7, R8, R5] ──────────────────────────────────────
  1220→    case 'N_B_D_VOL_C': {
  1221→      const pR7 = 0.55 + input.confidenceIndex * 0.15;
  1222→      const pR8 = 0.2 + input.strengthIndex * 0.15;
  1223→      const pR5 = 1 - pR7 - pR8;
  1224→      return normalize([pR7, pR8, pR5]);
  1225→    }
  1226→
  1227→    // ── N_B_D_VOL_W → [R6, R1, R5] ──────────────────────────────────────
  1228→    case 'N_B_D_VOL_W': {
  1229→      const pR6 = 0.4 + (1 - input.bullConsensus) * 0.15;
  1230→      const pR1 = 0.3 + input.bullConsensus * 0.15;
  1231→      const pR5 = 1 - pR6 - pR1;
  1232→      return normalize([pR6, pR1, pR5]);
  1233→    }
  1234→
  1235→    // ── N_B_NONE → [R5] ──────────────────────────────────────────────────
  1236→    case 'N_B_NONE': {
  1237→      return [1.0];
  1238→    }
  1239→
  1240→    // ── N_REVERSAL → [N_R_BULL, N_R_BEAR, N_R_NONE] ──────────────────────
  1241→    case 'N_REVERSAL': {
  1242→      const bullRev = overboughtRisk;
  1243→      const bearRev = oversoldBounce;
  1244→      const noneRev =
  1245→        1 - Math.max(overboughtRisk, oversoldBounce) * 0.7;
  1246→      return normalize([bullRev, bearRev, noneRev]);

  1248→

  1250→    case 'N_R_BULL': {
  1251→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1252→      const candleProb =
  1253→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1254→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1255→      return normalize([divProb, candleProb, srProb]);
  1256→    }
  1257→
  1258→    // ── N_R_B_DIV → [R1, R3, R5] ────────────────────────────────────────
  1259→    case 'N_R_B_DIV': {
  1260→      const pR1 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1261→      const pR3 = 0.2 + input.mlMomentum * 0.15;
  1262→      const pR5 = 1 - pR1 - pR3;
  1263→      return normalize([pR1, pR3, pR5]);
  1264→    }
  1265→
  1266→    // ── N_R_B_CANDLE → [R2, R1, R5] ─────────────────────────────────────
  1267→    case 'N_R_B_CANDLE': {
  1268→      const pR2 = 0.5 * input.confidenceIndex + 0.2;
  1269→      const pR1 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1270→      const pR5 = 1 - pR2 - pR1;
  1271→      return normalize([pR2, pR1, pR5]);
  1272→    }
  1273→
  1274→    // ── N_R_B_SR → [R1, R5, R6] ─────────────────────────────────────────
  1275→    // Edge order: R1, R5, R6 — probabilities computed then reordered
  1276→    case 'N_R_B_SR': {
  1277→      const pR1 = 0.5 + input.srAvgStrength * 0.2;
  1278→      const pR6 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1279→      const pR5 = 1 - pR1 - pR6;
  1280→      // Return in edge order: [R1, R5, R6]
  1281→      return normalize([pR1, pR5, pR6]);
  1282→    }
  1283→
  1284→    // ── N_R_BEAR → [N_R_BE_DIV, N_R_BE_CANDLE, N_R_BE_SR] ───────────────
  1285→    // Mirror of N_R_BULL
  1286→    case 'N_R_BEAR': {
  1287→      const divProb = clamp(Math.abs(input.cci) / 150, 0, 1) * 0.5;
  1288→      const candleProb =
  1289→        clamp(Math.abs(input.fisherTransform) / 2, 0, 1) * 0.3;
  1290→      const srProb = clamp(input.srAvgStrength, 0, 1) * 0.3;
  1291→      return normalize([divProb, candleProb, srProb]);
  1292→    }
  1293→
  1294→    // ── N_R_BE_DIV → [R6, R8, R5] ───────────────────────────────────────
  1295→    case 'N_R_BE_DIV': {
  1296→      const pR6 = 0.5 + (1 - input.mlMomentum) * 0.2;
  1297→      const pR8 = 0.2 + input.mlMomentum * 0.15;
  1298→      const pR5 = 1 - pR6 - pR8;
  1299→      return normalize([pR6, pR8, pR5]);
  1300→    }
  1301→
  1302→    // ── N_R_BE_CANDLE → [R7, R6, R5] ────────────────────────────────────
  1303→    case 'N_R_BE_CANDLE': {
  1304→      const pR7 = 0.5 * input.confidenceIndex + 0.2;
  1305→      const pR6 = 0.5 * (1 - input.confidenceIndex) + 0.15;
  1306→      const pR5 = 1 - pR7 - pR6;
  1307→      return normalize([pR7, pR6, pR5]);
  1308→    }
  1309→
  1310→    // ── N_R_BE_SR → [R6, R5, R1] ─────────────────────────────────────────
  1311→    // Edge order: R6, R5, R1 — probabilities computed then reordered
  1312→    case 'N_R_BE_SR': {
  1313→      const pR6 = 0.5 + input.srAvgStrength * 0.2;
  1314→      const pR1 = 0.15 + (1 - input.srAvgStrength) * 0.1;
  1315→      const pR5 = 1 - pR6 - pR1;
  1316→      // Return in edge order: [R6, R5, R1]
  1317→      return normalize([pR6, pR5, pR1]);
  1318→    }
  1319→
  1320→    // ── N_R_NONE → [R5] ──────────────────────────────────────────────────
  1321→    case 'N_R_NONE': {
  1322→      return [1.0];
  1323→    }
  1324→
  1325→    // ── Terminal nodes (R1–R9) have no outgoing edges ─────────────────────
  1326→    default:
  1327→      return [];
  1328→  }
  1329→}

  1331→// === Graph Traversal ==========================================================
  1332→// Recursive DFS from ROOT, accumulating path probabilities at terminal nodes.
  1333→// =============================================================================
  1334→
  1335→function traverseGraph(
  1336→  nodeId: string,
  1337→  pathProb: number,
  1338→  currentBranch: 'trend' | 'breakout' | 'reversal' | null,
  1339→  ctx: ProbContext,
  1340→  adj: Map<string, number[]>,
  1341→  edges: GraphEdge[],
  1342→  edgeProbs: Record<number, number>,
  1343→  nodeAccum: Record<string, number>,
  1344→  scenarios: Record<string, number>,
  1345→  contributions: Record<string, { trend: number; breakout: number; reversal: number }>
  1346→): void {
  1347→  // Accumulate path probability at this node
  1348→  nodeAccum[nodeId] = (nodeAccum[nodeId] || 0) + pathProb;
  1349→
  1350→  const outEdgeIndices = adj.get(nodeId);
  1351→  if (!outEdgeIndices || outEdgeIndices.length === 0) return;
  1352→
  1353→  // Compute outgoing edge probabilities for this node
  1354→  const probs = computeNodeProbabilities(
  1355→    nodeId,
  1356→    ctx,
  1357→    outEdgeIndices.length
  1358→  );
  1359→
  1360→  for (let i = 0; i < outEdgeIndices.length; i++) {
  1361→    const edgeIdx = outEdgeIndices[i];
  1362→    const edge = edges[edgeIdx];
  1363→    const prob = probs[i] || 0;
  1364→
  1365→    // Store edge probability
  1366→    edgeProbs[edgeIdx] = prob;
  1367→
  1368→    // Determine branch for child nodes
  1369→    let childBranch = currentBranch;
  1370→    if (nodeId === 'ROOT') {
  1371→      if (edge.to === 'N_TREND') childBranch = 'trend';
  1372→      else if (edge.to === 'N_BREAK') childBranch = 'breakout';
  1373→      else if (edge.to === 'N_REVERSAL') childBranch = 'reversal';
  1374→    }
  1375→
  1376→    const childPathProb = pathProb * prob;
  1377→
  1378→    // Check if target is a terminal node
  1379→    const isTerminal =
  1380→      edge.to.length === 2 &&
  1381→      edge.to[0] === 'R' &&
  1382→      edge.to[1] >= '1' &&
  1383→      edge.to[1] <= '9';
  1384→
  1385→    if (isTerminal) {
  1386→      // Accumulate scenario probability
  1387→      scenarios[edge.to] = (scenarios[edge.to] || 0) + childPathProb;
  1388→      // Accumulate branch contribution
  1389→      if (childBranch) {
  1390→        contributions[edge.to][childBranch] += childPathProb;
  1391→      }
  1392→    }
  1393→
  1394→    // Recurse into child
  1395→    traverseGraph(
  1396→      edge.to,
  1397→      childPathProb,
  1398→      childBranch,
  1399→      ctx,
  1400→      adj,
  1401→      edges,
  1402→      edgeProbs,
  1403→      nodeAccum,
  1404→      scenarios,
  1405→      contributions
  1406→    );
  1407→  }
  1408→}

  1410→// === Main Export: buildDecisionGraph ===========================================
  1411→
  1412→export function buildDecisionGraph(input: GraphInput): GraphData {
  1413→  // 1. Build static graph structure
  1414→  const nodes = createNodes();
  1415→  const edges = createEdges();
  1416→  const nodePositions = createPositions();
  1417→
  1418→  // 2. Build adjacency map: nodeId → array of edge indices
  1419→  const adj = new Map<string, number[]>();
  1420→  for (let i = 0; i < edges.length; i++) {
  1421→    const from = edges[i].from;
  1422→    if (!adj.has(from)) adj.set(from, []);
  1423→    adj.get(from)!.push(i);
  1424→  }
  1425→
  1426→  // 3. Build probability context
  1427→  const ctx = buildContext(input);
  1428→
  1429→  // 4. Initialize accumulation structures
  1430→  const edgeProbs: Record<number, number> = {};
  1431→  const nodeAccum: Record<string, number> = {};
  1432→  const scenarios: Record<string, number> = {};
  1433→  const contributions: Record<
  1434→    string,
  1435→    { trend: number; breakout: number; reversal: number }
  1436→  > = {};
  1437→
  1438→  for (let i = 1; i <= 9; i++) {
  1439→    const key = `R${i}`;
  1440→    scenarios[key] = 0;
  1441→    contributions[key] = { trend: 0, breakout: 0, reversal: 0 };
  1442→  }
  1443→
  1444→  // 5. Traverse graph from ROOT
  1445→  traverseGraph(
  1446→    'ROOT',
  1447→    1.0,
  1448→    null,
  1449→    ctx,
  1450→    adj,
  1451→    edges,
  1452→    edgeProbs,
  1453→    nodeAccum,
  1454→    scenarios,
  1455→    contributions
  1456→  );
  1457→
  1458→  // 6. Post-process scenario probabilities
  1459→  // Convert to percentage, clamp [2, 35], re-normalize to sum=100
  1460→  const rawPcts: Record<string, number> = {};
  1461→  for (let i = 1; i <= 9; i++) {
  1462→    const key = `R${i}`;
  1463→    rawPcts[key] = Math.round((scenarios[key] || 0) * 100);
  1464→  }
  1465→
  1466→  // Clamp each to [2, 35]
  1467→  const clamped = Object.entries(rawPcts).map(([k, v]) => ({
  1468→    key: k,
  1469→    value: clamp(v, 2, 35),
  1470→  }));
  1471→
  1472→  // Re-normalize to sum=100
  1473→  const clampedSum = clamped.reduce((s, c) => s + c.value, 0);
  1474→  const finalPcts: Record<string, number> = {};
  1475→  if (clampedSum > 0) {
  1476→    // Distribute proportionally and round, then fix rounding errors
  1477→    let runningTotal = 0;
  1478→    const entries = clamped
  1479→      .map((c) => {
  1480→        const raw = (c.value / clampedSum) * 100;
  1481→        return { key: c.key, raw };
  1482→      })
  1483→      .sort((a, b) => b.raw - a.raw);
  1484→
  1485→    for (let i = 0; i < entries.length; i++) {
  1486→      const isLast = i === entries.length - 1;
  1487→      if (isLast) {
  1488→        finalPcts[entries[i].key] = Math.round(100 - runningTotal);
  1489→      } else {
  1490→        const rounded = Math.round(entries[i].raw);
  1491→        finalPcts[entries[i].key] = rounded;
  1492→        runningTotal += rounded;
  1493→      }
  1494→    }
  1495→  }
  1496→
  1497→  // Ensure no negative values due to rounding
  1498→  for (const key of Object.keys(finalPcts)) {
  1499→    if (finalPcts[key] < 0) finalPcts[key] = 0;
  1500→  }
  1501→
  1502→  // 7. Compute branch probabilities (from ROOT outgoing edges)
  1503→  const rootEdgeIndices = adj.get('ROOT') || [];
  1504→  const rootProbs = computeNodeProbabilities('ROOT', ctx, rootEdgeIndices.length);
  1505→  const branchProbabilities = {
  1506→    trend: rootProbs[0] || 0,
  1507→    breakout: rootProbs[1] || 0,
  1508→    reversal: rootProbs[2] || 0,
  1509→  };
  1510→
  1511→  // 8. Build node display values (path probability as percentage)
  1512→  const nodeValues: Record<string, string> = {};
  1513→  for (const [nodeId, accum] of Object.entries(nodeAccum)) {
  1514→    nodeValues[nodeId] = `${Math.round(accum * 100)}%`;
  1515→  }
  1516→
  1517→  // Override terminal node values with final post-processed percentages
  1518→  for (const [key, pct] of Object.entries(finalPcts)) {
  1519→    nodeValues[key] = `${pct}%`;
  1520→  }
  1521→
  1522→  // 9. Build path contributions (as percentages)
  1523→  const pathContributions: Record<
  1524→    string,
  1525→    { trend: number; breakout: number; reversal: number }
  1526→  > = {};
  1527→  for (let i = 1; i <= 9; i++) {
  1528→    const key = `R${i}`;
  1529→    const c = contributions[key];
  1530→    const total = c.trend + c.breakout + c.reversal;
  1531→    if (total > 0) {
  1532→      pathContributions[key] = {
  1533→        trend: Math.round((c.trend / total) * 100),
  1534→        breakout: Math.round((c.breakout / total) * 100),
  1535→        reversal: Math.round((c.reversal / total) * 100),
  1536→      };
  1537→    } else {
  1538→      pathContributions[key] = { trend: 33, breakout: 34, reversal: 33 };
  1539→    }
  1540→  }
  1541→
  1542→  // 10. Return complete GraphData
  1543→  return {
  1544→    nodes,
  1545→    edges,
  1546→    nodePositions,
  1547→    edgeProbabilities,
  1548→    nodeValues,
  1549→    branchProbabilities,
  1550→    scenarioProbabilities: finalPcts,
  1551→    pathContributions,
  1552→  };
  1553→}
