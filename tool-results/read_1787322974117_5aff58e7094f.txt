   140→
   141→
   142→def _safe_div(a: float, b: float, fill: float = 0.0) -> float:
   143→    """Divide with zero-safety."""
   144→    return a / b if abs(b) > 1e-12 else fill
   145→
   146→
   147→def extract_features(df: pd.DataFrame) -> pd.DataFrame:
   148→    """
   149→    Build 40+ technical features from OHLCV data.
   150→
   151→    Parameters
   152→    ----------
   153→    df : pd.DataFrame
   154→        Must contain columns: open, high, low, close, volume (lowercase).
   155→        Index should be sequential integers.
   156→
   157→    Returns
   158→    -------
   159→    pd.DataFrame  —  same row count, NaNs at the head where windows can't fill.
   160→    """
   161→    out = pd.DataFrame(index=df.index)
   162→    c = df["close"].astype(float)
   163→    h = df["high"].astype(float)
   164→    l = df["low"].astype(float)
   165→    o = df["open"].astype(float)
   166→    v = df["volume"].astype(float)
   167→
   168→    # ── Price returns ──
   169→    for period in (1, 3, 5, 10, 20):
   170→        out[f"return_{period}d"] = c.pct_change(period)
   171→
   172→    # ── Rolling statistics ──
   173→    for w in (5, 10, 20, 60):
   174→        roll = c.rolling(w, min_periods=1)
   175→        out[f"close_mean_{w}"] = roll.mean()
   176→        out[f"close_std_{w}"] = roll.std()
   177→        out[f"close_min_{w}"] = roll.min()
   178→        out[f"close_max_{w}"] = roll.max()
   179→
   180→    # ── RSI(14) ──
   181→    delta = c.diff()
   182→    gain = delta.clip(lower=0).rolling(14, min_periods=1).mean()
   183→    loss = (-delta.clip(upper=0)).rolling(14, min_periods=1).mean()
   184→    rs = _safe_div(gain, loss, fill=100.0)
   185→    out["rsi_14"] = 100.0 - (100.0 / (1.0 + rs))
   186→
   187→    # ── MACD(12,26,9) ──
   188→    ema12 = c.ewm(span=12, adjust=False).mean()
   189→    ema26 = c.ewm(span=26, adjust=False).mean()
   190→    out["macd"] = ema12 - ema26
   191→    out["macd_signal"] = out["macd"].ewm(span=9, adjust=False).mean()
   192→    out["macd_hist"] = out["macd"] - out["macd_signal"]
   193→
   194→    # ── Stochastic(14,3,3) ──
   195→    low14 = l.rolling(14, min_periods=1).min()
   196→    high14 = h.rolling(14, min_periods=1).max()
   197→    out["stoch_k"] = 100.0 * _safe_div(c - low14, high14 - low14)
   198→    out["stoch_d"] = out["stoch_k"].rolling(3, min_periods=1).mean()
   199→
   200→    # ── CCI(20) ──
   201→    typical = (h + l + c) / 3.0
   202→    sma20 = typical.rolling(20, min_periods=1).mean()
   203→    mad20 = typical.rolling(20, min_periods=1).apply(lambda x: np.mean(np.abs(x - x.mean())), raw=True)
   204→    out["cci_20"] = _safe_div(typical - sma20, 0.015 * mad20)
   205→
   206→    # ── ATR(14) ──
   207→    tr = pd.concat([h - l, (h - c.shift(1)).abs(), (l - c.shift(1)).abs()], axis=1).max(axis=1)
   208→    out["atr_14"] = tr.rolling(14, min_periods=1).mean()
   209→
   210→    # ── ADX(14) ──
   211→    plus_dm = h.diff()
   212→    minus_dm = -l.diff()
   213→    plus_dm = plus_dm.where((plus_dm > minus_dm) & (plus_dm > 0), 0.0)
   214→    minus_dm = minus_dm.where((minus_dm > plus_dm) & (minus_dm > 0), 0.0)
   215→    tr_smooth = tr.ewm(span=14, adjust=False).mean()
   216→    plus_di = 100.0 * _safe_div(plus_dm.ewm(span=14, adjust=False).mean(), tr_smooth)
   217→    minus_di = 100.0 * _safe_div(minus_dm.ewm(span=14, adjust=False).mean(), tr_smooth)
   218→    dx_sum = _safe_div(abs(plus_di - minus_di), plus_di + minus_di) * 100.0
   219→    out["adx_14"] = dx_sum.ewm(span=14, adjust=False).mean()
   220→
   221→    # ── Bollinger Bands position ──
   222→    bb_mid = c.rolling(20, min_periods=1).mean()
   223→    bb_std = c.rolling(20, min_periods=1).std()
   224→    out["bb_position"] = _safe_div(c - bb_mid, 2.0 * bb_std)
   225→
   226→    # ── Volume features ──
   227→    out["volume_sma_20"] = v.rolling(20, min_periods=1).mean()
   228→    out["volume_ratio"] = _safe_div(v, out["volume_sma_20"])
   229→    out["volume_zscore"] = _safe_div(v - out["volume_sma_20"], v.rolling(20, min_periods=1).std())
   230→
   231→    # OBV (vectorized)
   232→    direction = pd.Series(np.sign(c.diff()), index=df.index).fillna(0)
   233→    obv = (direction * v).cumsum()
   234→    out["obv"] = obv
   235→
   236→    # ── Price patterns (rolling 10-bar) ──
   237→    is_hh = (h > h.shift(1)).astype(float)
   238→    is_ll = (l < l.shift(1)).astype(float)
   239→    is_inside = ((h <= h.shift(1)) & (l >= l.shift(1))).astype(float)
   240→    pw = 10
   241→    out["higher_highs"] = is_hh.rolling(pw, min_periods=1).mean()
   242→    out["lower_lows"] = is_ll.rolling(pw, min_periods=1).mean()
   243→    out["inside_bar_ratio"] = is_inside.rolling(pw, min_periods=1).mean()
   244→
   245→    # ── Momentum ──
   246→    out["roc_10"] = _safe_div(c - c.shift(10), c.shift(10)) * 100.0
   247→    momentum_score = (
   248→        out.get("rsi_14", pd.Series(50, index=df.index)) - 50.0
   249→        + out.get("macd_hist", pd.Series(0, index=df.index)) * 10.0
   250→        + out.get("stoch_k", pd.Series(50, index=df.index)) - 50.0
   251→    ) / 3.0
   252→    out["momentum_score"] = momentum_score
   253→
   254→    # ── Lag features (returns & RSI for last 5 periods) ──
   255→    for lag in range(1, 6):
   256→        out[f"return_1d_lag{lag}"] = out["return_1d"].shift(lag)
   257→        out[f"rsi_14_lag{lag}"] = out["rsi_14"].shift(lag)
   258→
   259→    return out
   260→

   267→def build_ohlcv_df(ohlcv: List[List]) -> pd.DataFrame:
   268→    """Convert raw OHLCV list to DataFrame with lowercase columns."""
   269→    df = pd.DataFrame(ohlcv, columns=["date", "open", "high", "low", "close", "volume"])
   270→    for col in ("open", "high", "low", "close", "volume"):
   271→        df[col] = pd.to_numeric(df[col], errors="coerce")
   272→    df["date"] = df["date"].astype(str)
   273→    return df

   276→# ─────────────────────────────────────────────────────────────────────────────
   277→# Target builders
   278→# ─────────────────────────────────────────────────────────────────────────────
   279→
   280→
   281→def build_direction_target(df: pd.DataFrame) -> pd.Series:
   282→    """Binary: 1 if 5-day forward return > 0.5%, else 0."""
   283→    fwd = df["close"].shift(-FORWARD_DAYS) / df["close"] - 1.0
   284→    return (fwd > DIRECTION_THRESHOLD).astype(int)
   285→
   286→
   287→def build_regression_target(df: pd.DataFrame) -> pd.Series:
   288→    """5-day forward return as percentage."""
   289→    return (df["close"].shift(-FORWARD_DAYS) / df["close"] - 1.0) * 100.0
   290→
   291→
   292→def build_volatility_target(df: pd.DataFrame) -> pd.Series:
   293→    """Next-10-day realized volatility (std of daily returns * sqrt(252) * 100)."""
   294→    ret = df["close"].pct_change()
   295→    return ret.shift(-VOLATILITY_WINDOW).rolling(VOLATILITY_WINDOW).std() * np.sqrt(252) * 100.0

   302→
   303→def _save_model_artifact(model: Any, path: str, input_dim: int) -> None:
   304→    """Save model as ONNX (preferred) or pickle fallback."""
   305→    os.makedirs(os.path.dirname(path), exist_ok=True)
   306→    if ONNX_EXPORT_AVAILABLE:
   307→        try:
   308→            if isinstance(model, (xgb.XGBModel,)):
   309→                # XGBoost native ONNX export
   310→                initial_type = [('float_input', FloatTensorType([None, input_dim]))]
   311→                onnx_model = xgb.to_onnx(model, initial_types=initial_type)
   312→                import onnx
   313→                onnx.save(onnx_model, path)
   314→                return
   315→            elif hasattr(model, 'predict_proba') or hasattr(model, 'predict'):
   316→                # sklearn model
   317→                initial_type = [('float_input', FloatTensorType([None, input_dim]))]
   318→                onnx_model = skl2onnx.convert_sklearn(model, initial_types=initial_type)
   319→                import onnx
   320→                onnx.save(onnx_model, path)
   321→                return
   322→        except Exception as e:
   323→            log("WARN", f"ONNX export failed for {path}: {e}, falling back to pickle")
   324→    # Pickle fallback
   325→    pkl_path = path.replace(".onnx", ".pkl")
   326→    with open(pkl_path, "wb") as f:
   327→        pickle.dump(model, f)

   330→def _load_model_artifact(path: str) -> Tuple[Any, bool]:
   331→    """Load model from ONNX or pickle. Returns (model_or_session, is_onnx)."""
   332→    if os.path.exists(path):
   333→        try:
   334→            session = ort.InferenceSession(path)
   335→            return session, True
   336→        except Exception:
   337→            pass
   338→    pkl_path = path.replace(".onnx", ".pkl")
   339→    if os.path.exists(pkl_path):
   340→        with open(pkl_path, "rb") as f:
   341→            return pickle.load(f), False
   342→    return None, False
   343→

   348→        json.dump(meta, f, indent=2, default=str)

   378→# ─────────────────────────────────────────────────────────────────────────────

   380→# ─────────────────────────────────────────────────────────────────────────────

   382→
   383→def train_models(
   384→    symbol: str,
   385→    ohlcv: List[List],
   386→    model_names: Optional[List[str]] = None,
   387→) -> Dict[str, Any]:
   388→    """
   389→    Train requested models for a symbol.
   390→
   391→    Returns a dict with metrics and metadata.
   392→    """
   393→    if model_names is None:
   394→        model_names = MODEL_TYPES
   395→
   396→    # Validate
   397→    for mn in model_names:
   398→        if mn not in MODEL_TYPES:
   399→            raise ValueError(f"Unknown model type: {mn}")
   400→
   401→    df = build_ohlcv_df(ohlcv)
   402→    if len(df) < MIN_CANDLES:
   403→        raise ValueError(f"Need at least {MIN_CANDLES} candles, got {len(df)}")
   404→
   405→    log("INFO", f"Training {model_names} for {symbol} with {len(df)} candles")
   406→
   407→    # Feature engineering
   408→    features_df = extract_features(df)
   409→    feature_cols = [c for c in features_df.columns]
   410→
   411→    # Drop rows with NaN
   412→    valid_mask = features_df[feature_cols].notna().all(axis=1)
   413→    X = features_df.loc[valid_mask, feature_cols].values
   414→    feature_index = features_df.index[valid_mask]
   415→
   416→    # Build targets
   417→    dir_target = build_direction_target(df).loc[feature_index].values
   418→    reg_target = build_regression_target(df).loc[feature_index].values
   419→    vol_target = build_volatility_target(df).loc[feature_index].values
   420→
   421→    # Drop rows where targets are NaN
   422→    valid2 = ~(np.isnan(dir_target) | np.isnan(reg_target) | np.isnan(vol_target))
   423→    X = X[valid2]
   424→    dir_target = dir_target[valid2]
   425→    reg_target = reg_target[valid2]
   426→    vol_target = vol_target[valid2]
   427→
   428→    if len(X) < MIN_CANDLES:
   429→        raise ValueError(f"After cleaning, only {len(X)} valid samples (need {MIN_CANDLES})")
   430→
   431→    # Scale features
   432→    scaler = StandardScaler()
   433→    X_scaled = scaler.fit_transform(X)
   434→    input_dim = X_scaled.shape[1]
   435→
   436→    # Save scaler
   437→    _save_scaler(scaler, symbol)
   438→    cache_scaler(symbol, scaler)
   439→
   440→    # TimeSeriesSplit
   441→    tscv = TimeSeriesSplit(n_splits=5)
   442→    metrics: Dict[str, Dict] = {}
   443→    all_feature_importance: Dict[str, float] = {}
   444→
   445→    sym_dir = os.path.join(MODELS_DIR, symbol)
   446→    os.makedirs(sym_dir, exist_ok=True)
   447→
   448→    for mn in model_names:
   449→        log("INFO", f"  Training model: {mn}")
   450→
   451→        if mn == "xgboost_direction":
   452→            model = xgb.XGBClassifier(
   453→                n_estimators=200,
   454→                max_depth=5,
   455→                learning_rate=0.05,
   456→                subsample=0.8,
   457→                colsample_bytree=0.8,
   458→                use_label_encoder=False,
   459→                eval_metric="logloss",
   460→                verbosity=0,
   461→                n_jobs=-1,
   462→            )
   463→            target = dir_target
   464→            fold_metrics = {"accuracy": [], "f1": [], "auc": []}
   465→
   466→            for train_idx, test_idx in tscv.split(X_scaled):
   467→                model.fit(X_scaled[train_idx], target[train_idx])
   468→                preds = model.predict(X_scaled[test_idx])
   469→                proba = model.predict_proba(X_scaled[test_idx])[:, 1]
   470→                fold_metrics["accuracy"].append(accuracy_score(target[test_idx], preds))
   471→                fold_metrics["f1"].append(f1_score(target[test_idx], preds, zero_division=0))
   472→                try:
   473→                    fold_metrics["auc"].append(roc_auc_score(target[test_idx], proba))
   474→                except Exception:
   475→                    fold_metrics["auc"].append(0.5)
   476→
   477→            # Final fit on all data
   478→            model.fit(X_scaled, target)
   479→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   480→            imp = model.feature_importances_
   481→
   482→        elif mn == "xgboost_regression":
   483→            model = xgb.XGBRegressor(
   484→                n_estimators=200,
   485→                max_depth=5,
   486→                learning_rate=0.05,
   487→                subsample=0.8,
   488→                colsample_bytree=0.8,
   489→                verbosity=0,
   490→                n_jobs=-1,
   491→            )
   492→            target = reg_target
   493→            fold_metrics = {"mae": [], "rmse": [], "r2": []}
   494→
   495→            for train_idx, test_idx in tscv.split(X_scaled):
   496→                model.fit(X_scaled[train_idx], target[train_idx])
   497→                preds = model.predict(X_scaled[test_idx])
   498→                fold_metrics["mae"].append(mean_absolute_error(target[test_idx], preds))
   499→                fold_metrics["rmse"].append(np.sqrt(mean_squared_error(target[test_idx], preds)))
   500→                fold_metrics["r2"].append(r2_score(target[test_idx], preds))
   501→
   502→            model.fit(X_scaled, target)
   503→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   504→            imp = model.feature_importances_
   505→
   506→        elif mn == "ensemble_direction":
   507→            model = GradientBoostingClassifier(
   508→                n_estimators=200,
   509→                max_depth=4,
   510→                learning_rate=0.05,
   511→                subsample=0.8,
   512→                max_features="sqrt",
   513→                random_state=42,
   514→            )
   515→            target = dir_target
   516→            fold_metrics = {"accuracy": [], "f1": [], "auc": []}
   517→
   518→            for train_idx, test_idx in tscv.split(X_scaled):
   519→                model.fit(X_scaled[train_idx], target[train_idx])
   520→                preds = model.predict(X_scaled[test_idx])
   521→                proba = model.predict_proba(X_scaled[test_idx])[:, 1]
   522→                fold_metrics["accuracy"].append(accuracy_score(target[test_idx], preds))
   523→                fold_metrics["f1"].append(f1_score(target[test_idx], preds, zero_division=0))
   524→                try:
   525→                    fold_metrics["auc"].append(roc_auc_score(target[test_idx], proba))
   526→                except Exception:
   527→                    fold_metrics["auc"].append(0.5)
   528→
   529→            model.fit(X_scaled, target)
   530→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   531→            imp = model.feature_importances_
   532→
   533→        elif mn == "ensemble_regression":
   534→            model = GradientBoostingRegressor(
   535→                n_estimators=200,
   536→                max_depth=4,
   537→                learning_rate=0.05,
   538→                subsample=0.8,
   539→                max_features="sqrt",
   540→                random_state=42,
   541→            )
   542→            target = reg_target
   543→            fold_metrics = {"mae": [], "rmse": [], "r2": []}
   544→
   545→            for train_idx, test_idx in tscv.split(X_scaled):
   546→                model.fit(X_scaled[train_idx], target[train_idx])
   547→                preds = model.predict(X_scaled[test_idx])
   548→                fold_metrics["mae"].append(mean_absolute_error(target[test_idx], preds))
   549→                fold_metrics["rmse"].append(np.sqrt(mean_squared_error(target[test_idx], preds)))
   550→                fold_metrics["r2"].append(r2_score(target[test_idx], preds))
   551→
   552→            model.fit(X_scaled, target)
   553→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   554→            imp = model.feature_importances_
   555→
   556→        elif mn == "volatility_model":
   557→            model = RandomForestRegressor(
   558→                n_estimators=200,
   559→                max_depth=6,
   560→                min_samples_leaf=5,
   561→                random_state=42,
   562→                n_jobs=-1,
   563→            )
   564→            target = vol_target
   565→            fold_metrics = {"mae": [], "rmse": [], "r2": []}
   566→
   567→            for train_idx, test_idx in tscv.split(X_scaled):
   568→                model.fit(X_scaled[train_idx], target[train_idx])
   569→                preds = model.predict(X_scaled[test_idx])
   570→                fold_metrics["mae"].append(mean_absolute_error(target[test_idx], preds))
   571→                fold_metrics["rmse"].append(np.sqrt(mean_squared_error(target[test_idx], preds)))
   572→                fold_metrics["r2"].append(r2_score(target[test_idx], preds))
   573→
   574→            model.fit(X_scaled, target)
   575→            metrics[mn] = {k: round(float(np.mean(v)), 4) for k, v in fold_metrics.items()}
   576→            imp = model.feature_importances_
   577→
   578→        else:
   579→            continue
   580→
   581→        # Save model artifact
   582→        onnx_path = os.path.join(sym_dir, f"{mn}.onnx")
   583→        _save_model_artifact(model, onnx_path, input_dim)
   584→
   585→        # Cache in memory
   586→        cache_model(symbol, mn, model, metrics.get(mn, {}))
   587→
   588→        # Save metadata
   589→        meta = {
   590→            "model_type": mn,
   591→            "symbol": symbol,
   592→            "metrics": metrics.get(mn, {}),
   593→            "feature_cols": feature_cols,
   594→            "input_dim": input_dim,
   595→            "training_samples": int(len(X)),
   596→            "date_range": [str(df["date"].iloc[0]), str(df["date"].iloc[-1])],
   597→            "trained_at": datetime.utcnow().isoformat() + "Z",
   598→        }
   599→        _save_meta(meta, os.path.join(sym_dir, f"{mn}_meta.json"))
   600→
   601→        # Accumulate feature importance (average across models)
   602→        for i, col in enumerate(feature_cols):
   603→            all_feature_importance[col] = all_feature_importance.get(col, 0.0) + float(imp[i])
   604→
   605→    # Normalize feature importance
   606→    total_imp = sum(all_feature_importance.values())
   607→    if total_imp > 0:
   608→        all_feature_importance = {k: round(v / total_imp, 4) for k, v in all_feature_importance.items()}
   609→
   610→    # Top 10
   611→    top_features = dict(sorted(all_feature_importance.items(), key=lambda x: x[1], reverse=True)[:10])
   612→
   613→    return {
   614→        "status": "ok",
   615→        "symbol": symbol,
   616→        "models_trained": model_names,
   617→        "metrics": metrics,
   618→        "feature_importance": top_features,
   619→        "training_samples": len(X),
   620→        "date_range": [str(df["date"].iloc[0]), str(df["date"].iloc[-1])],
   621→    }
   622→
   623→
   624→# ─────────────────────────────────────────────────────────────────────────────
   625→# Prediction
   626→# ─────────────────────────────────────────────────────────────────────────────
   627→
   628→
   629→def _predict_with_model(model: Any, features: np.ndarray, is_onnx: bool) -> np.ndarray:
   630→    """Run prediction on a single row of features."""
   631→    if is_onnx:
   632→        session = model  # ONNX Runtime session
   633→        input_name = session.get_inputs()[0].name
   634→        result = session.run(None, {input_name: features.astype(np.float32)})
   635→        # Classification: result[1] is probabilities; Regression: result[0] is values
   636→        if len(result) == 2 and result[1].shape[1] >= 2:
   637→            return result[1]  # probabilities
   638→        return result[0]  # regression
   639→    else:
   640→        # sklearn or xgboost native
   641→        if hasattr(model, "predict_proba"):
   642→            return model.predict_proba(features)  # type: ignore
   643→        return model.predict(features)  # type: ignore
   644→
   645→
   646→def predict_sessions(
   647→    symbol: str,
   648→    ohlcv: List[List],
   649→    horizon: int = 30,
   650→) -> Dict[str, Any]:
   651→    """
   652→    Generate multi-session prediction for a symbol.
   653→
   654→    Uses iterative prediction: predict 5 sessions, append, re-extract, repeat.
   655→    """
   656→    df = build_ohlcv_df(ohlcv)
   657→    if len(df) < MIN_CANDLES:
   658→        raise ValueError(f"Need at least {MIN_CANDLES} candles for prediction, got {len(df)}")
   659→
   660→    # Load scaler
   661→    scaler = _load_scaler(symbol)
   662→    if scaler is None:
   663→        raise ValueError(f"No trained scaler for symbol {symbol}")
   664→
   665→    # Determine which models are available
   666→    sym_dir = os.path.join(MODELS_DIR, symbol)
   667→    available_models = []
   668→    dir_model = None
   669→    dir_is_onnx = False
   670→    reg_model = None
   671→    reg_is_onnx = False
   672→
   673→    for mn in ("xgboost_direction", "ensemble_direction"):
   674→        model, is_onnx = _load_model_artifact(os.path.join(sym_dir, f"{mn}.onnx"))
   675→        if model is not None:
   676→            dir_model = model
   677→            dir_is_onnx = is_onnx
   678→            available_models.append(mn)
   679→            break
   680→
   681→    for mn in ("xgboost_regression", "ensemble_regression"):
   682→        model, is_onnx = _load_model_artifact(os.path.join(sym_dir, f"{mn}.onnx"))
   683→        if model is not None:
   684→            reg_model = model
   685→            reg_is_onnx = is_onnx
   686→            available_models.append(mn)
   687→            break
   688→
   689→    if dir_model is None and reg_model is None:
   690→        raise ValueError(f"No trained model for symbol {symbol}")
   691→
   692→    # Extract current features for the last candle
   693→    features_df = extract_features(df)
   694→    feature_cols = list(features_df.columns)
   695→    last_row = features_df.iloc[-1:].values
   696→    if np.any(np.isnan(last_row)):
   697→        # Forward-fill NaNs with column means
   698→        col_means = features_df.mean().values
   699→        nan_mask = np.isnan(last_row)
   700→        last_row[nan_mask] = np.take(col_means, np.where(nan_mask)[1])
   701→
   702→    current_features = {}
   703→    for i, col in enumerate(feature_cols):
   704→        current_features[col] = round(float(last_row[0, i]), 4)
   705→
   706→    # Iterative prediction
   707→    working_df = df.copy()
   708→    sessions: List[Dict] = []
   709→    last_close = float(df["close"].iloc[-1])
   710→    iterations = (horizon + PREDICTION_STEPS - 1) // PREDICTION_STEPS
   711→
   712→    for it in range(iterations):
   713→        remaining = horizon - len(sessions)
   714→        steps = min(PREDICTION_STEPS, remaining)
   715→
   716→        # Extract features from current working dataframe
   717→        feat_df = extract_features(working_df)
   718→        feat_cols = list(feat_df.columns)
   719→        last_features = feat_df.iloc[-1:].values
   720→
   721→        # Handle NaNs
   722→        if np.any(np.isnan(last_features)):
   723→            col_means = feat_df.mean().values
   724→            for j in range(last_features.shape[1]):
   725→                if np.isnan(last_features[0, j]):
   726→                    last_features[0, j] = col_means[j]
   727→
   728→        scaled = scaler.transform(last_features)
   729→
   730→        predicted_change = 0.0
   731→        direction = "flat"
   732→        confidence = 0.5
   733→
   734→        if reg_model is not None:
   735→            reg_result = _predict_with_model(reg_model, scaled, reg_is_onnx)
   736→            predicted_change = float(reg_result[0])
   737→
   738→        if dir_model is not None:
   739→            dir_result = _predict_with_model(dir_model, scaled, dir_is_onnx)
   740→            # dir_result is probability array [p_down, p_up]
   741→            if dir_result.ndim == 2 and dir_result.shape[1] >= 2:
   742→                p_up = float(dir_result[0, 1])
   743→            else:
   744→                p_up = float(dir_result[0]) if float(dir_result[0]) > 0.5 else 0.5
   745→            confidence = p_up if p_up > 0.5 else (1.0 - p_up)
   746→            direction = "up" if p_up > 0.5 else "down"
   747→
   748→        # If only regression model, derive direction from predicted change
   749→        if dir_model is None and reg_model is not None:
   750→            if predicted_change > DIRECTION_THRESHOLD * 100:
   751→                direction = "up"
   752→                confidence = min(0.9, 0.5 + abs(predicted_change) / 5.0)
   753→            elif predicted_change < -DIRECTION_THRESHOLD * 100:
   754→                direction = "down"
   755→                confidence = min(0.9, 0.5 + abs(predicted_change) / 5.0)
   756→
   757→        # Generate steps sessions
   758→        for s in range(steps):
   759→            session_num = len(sessions) + 1
   760→            step_change = predicted_change / steps
   761→            new_close = last_close * (1.0 + step_change / 100.0)
   762→            change_pct = round((new_close - last_close) / last_close * 100, 4)
   763→
   764→            sessions.append({
   765→                "session": session_num,
   766→                "predicted_close": round(new_close, 2),
   767→                "confidence": round(confidence, 4),
   768→                "direction": direction,
   769→                "change_pct": change_pct,
   770→            })
   771→            last_close = new_close
   772→
   773→        # Append predicted candle to working_df for next iteration
   774→        avg_change = predicted_change / steps
   775→        new_close_row = last_close * (1.0 + avg_change / 100.0)
   776→        new_row = {
   777→            "date": f"pred_{len(sessions)}",
   778→            "open": last_close,
   779→            "high": max(last_close, new_close_row) * (1.0 + abs(avg_change) / 200.0),
   780→            "low": min(last_close, new_close_row) * (1.0 - abs(avg_change) / 200.0),
   781→            "close": new_close_row,
   782→            "volume": 0.0,
   783→        }
   784→        working_df = pd.concat([working_df, pd.DataFrame([new_row])], ignore_index=True)
   785→
   786→    # Aggregate results
   787→    closes = [s["predicted_close"] for s in sessions]
   788→    min_price = min(closes)
   789→    max_price = max(closes)
   790→    final_close = closes[-1]
   791→    initial_close = float(df["close"].iloc[-1])
   792→    overall_change = (final_close - initial_close) / initial_close * 100
   793→
   794→    if overall_change > 1.0:
   795→        overall_direction = "up"
   796→    elif overall_change < -1.0:
   797→        overall_direction = "down"
   798→    else:
   799→        overall_direction = "flat"
   800→
   801→    avg_confidence = np.mean([s["confidence"] for s in sessions])
   802→    price_range = max_price - min_price
   803→    price_range_pct = price_range / initial_close * 100
   804→
   805→    if price_range_pct > 5.0:
   806→        risk_level = "high"
   807→    elif price_range_pct > 2.5:
   808→        risk_level = "medium"
   809→    else:
   810→        risk_level = "low"
   811→
   812→    return {
   813→        "status": "ok",
   814→        "symbol": symbol,
   815→        "prediction": {
   816→            "sessions": sessions,
   817→            "overall_direction": overall_direction,
   818→            "overall_confidence": round(float(avg_confidence), 4),
   819→            "target_price_min": round(min_price, 2),
   820→            "target_price_max": round(max_price, 2),
   821→            "risk_level": risk_level,
   822→        },
   823→        "models_used": available_models,
   824→        "current_features": current_features,
   825→    }
   826→
   827→
   828→# ─────────────────────────────────────────────────────────────────────────────
   829→# Model listing
   830→# ─────────────────────────────────────────────────────────────────────────────
   831→
   832→
   833→def list_all_models() -> Dict[str, Any]:
   834→    """List all trained models across all symbols."""
   835→    result: Dict[str, List[Dict]] = {}
   836→    if not os.path.exists(MODELS_DIR):
   837→        return {"status": "ok", "models": {}}
   838→
   839→    for sym in sorted(os.listdir(MODELS_DIR)):
   840→        sym_path = os.path.join(MODELS_DIR, sym)
   841→        if not os.path.isdir(sym_path):
   842→            continue
   843→        models = []
   844→        for mn in MODEL_TYPES:
   845→            meta_path = os.path.join(sym_path, f"{mn}_meta.json")
   846→            if os.path.exists(meta_path):
   847→                meta = _load_meta(meta_path)
   848→                if meta:
   849→                    models.append(meta)
   850→        if models:
   851→            result[sym] = models
   852→
   853→    return {"status": "ok", "models": result}
   854→
   855→
   856→def get_symbol_models(symbol: str) -> Dict[str, Any]:
   857→    """Get models for a specific symbol."""
   858→    sym_path = os.path.join(MODELS_DIR, symbol)
   859→    if not os.path.isdir(sym_path):
   860→        return {"status": "error", "message": f"No trained model for symbol {symbol}"}
   861→
   862→    models = []
   863→    for mn in MODEL_TYPES:
   864→        meta_path = os.path.join(sym_path, f"{mn}_meta.json")
   865→        if os.path.exists(meta_path):
   866→            meta = _load_meta(meta_path)
   867→            if meta:
   868→                models.append(meta)
   869→
   870→    if not models:
   871→        return {"status": "error", "message": f"No trained model for symbol {symbol}"}
   872→
   873→    return {"status": "ok", "symbol": symbol, "models": models}

   880→
   881→def _json_response(handler: BaseHTTPRequestHandler, data: Dict, status: int = 200) -> None:
   882→    """Send a JSON response."""
   883→    body = json.dumps(data, ensure_ascii=False, default=str).encode("utf-8")
   884→    handler.send_response(status)
   885→    handler.send_header("Content-Type", "application/json; charset=utf-8")
   886→    handler.send_header("Content-Length", str(len(body)))
   887→    handler.end_headers()
   888→    handler.wfile.write(body)
   889→
   890→
   891→def _read_body(handler: BaseHTTPRequestHandler) -> bytes:
   892→    """Read request body."""
   893→    length = int(handler.headers.get("Content-Length", 0))
   894→    return handler.rfile.read(length) if length > 0 else b""

   896→
   897→class MLHandler(BaseHTTPRequestHandler):
   898→    """HTTP request handler for ML Training Service."""
   899→
   900→    # Suppress default stderr logging
   901→    def log_message(self, format: str, *args: Any) -> None:
   902→        pass
   903→
   904→    def _route(self, method: str) -> None:
   905→        """Route request to appropriate handler."""
   906→        parsed = urlparse(self.path)
   907→        path = parsed.path.rstrip("/")
   908→
   909→        try:
   910→            # ── Health ──
   911→            if path == "/health" and method == "GET":
   912→                _json_response(self, {
   913→                    "status": "ok",
   914→                    "service": "ml-trainer",
   915→                    "port": PORT,
   916→                    "version": "1.0.0",
   917→                    "onnx_export": ONNX_EXPORT_AVAILABLE,
   918→                    "cached_symbols": list(_model_cache.keys()),
   919→                })
   920→                return
   921→
   922→            # ── Train ──
   923→            if path == "/train" and method == "POST":
   924→                body = json.loads(_read_body(self))
   925→                symbol = body.get("symbol", "").strip().upper()
   926→                ohlcv = body.get("ohlcv", [])
   927→                models = body.get("models", MODEL_TYPES)
   928→
   929→                if not symbol:
   930→                    _json_response(self, {"status": "error", "message": "Missing 'symbol' field"}, 400)
   931→                    return
   932→                if not ohlcv or len(ohlcv) < MIN_CANDLES:
   933→                    _json_response(self, {"status": "error", "message": f"Need at least {MIN_CANDLES} OHLCV candles"}, 400)
   934→                    return
   935→
   936→                result = train_models(symbol, ohlcv, models)
   937→                _json_response(self, result)
   938→                return
   939→
   940→            # ── Predict ──
   941→            if path == "/predict" and method == "POST":
   942→                body = json.loads(_read_body(self))
   943→                symbol = body.get("symbol", "").strip().upper()
   944→                ohlcv = body.get("ohlcv", [])
   945→                horizon = min(body.get("horizon", 30), 90)  # cap at 90
   946→
   947→                if not symbol:
   948→                    _json_response(self, {"status": "error", "message": "Missing 'symbol' field"}, 400)
   949→                    return
   950→                if not ohlcv or len(ohlcv) < MIN_CANDLES:
   951→                    _json_response(self, {"status": "error", "message": f"Need at least {MIN_CANDLES} OHLCV candles"}, 400)
   952→                    return
   953→
   954→                result = predict_sessions(symbol, ohlcv, horizon)
   955→                _json_response(self, result)
   956→                return
   957→
   958→            # ── List all models ──
   959→            if path == "/models" and method == "GET":
   960→                _json_response(self, list_all_models())
   961→                return
   962→
   963→            # ── Get symbol models ──
   964→            if path.startswith("/models/") and method == "GET":
   965→                symbol = path[len("/models/"):].strip().upper()
   966→                if not symbol:
   967→                    _json_response(self, {"status": "error", "message": "Missing symbol in path"}, 400)
   968→                    return
   969→                _json_response(self, get_symbol_models(symbol))
   970→                return
   971→
   972→            # ── Retrain all ──
   973→            if path == "/retrain-all" and method == "POST":
   974→                body = json.loads(_read_body(self))
   975→                ohlcv_data = body.get("ohlcv_data", {})
   976→
   977→                if not ohlcv_data:
   978→                    _json_response(self, {"status": "error", "message": "Missing 'ohlcv_data' field"}, 400)
   979→                    return
   980→
   981→                results = {}
   982→                errors = []
   983→                for sym, candles in ohlcv_data.items():
   984→                    try:
   985→                        sym_upper = sym.strip().upper()
   986→                        r = train_models(sym_upper, candles, MODEL_TYPES)
   987→                        results[sym_upper] = r
   988→                    except Exception as e:
   989→                        errors.append({"symbol": sym, "error": str(e)})
   990→
   991→                _json_response(self, {
   992→                    "status": "ok",
   993→                    "retrained": list(results.keys()),
   994→                    "results": results,
   995→                    "errors": errors,
   996→                })
   997→                return
   998→
   999→            # ── 404 ──
  1000→            _json_response(self, {"status": "error", "message": f"Not found: {method} {path}"}, 404)
  1001→
  1002→        except json.JSONDecodeError:
  1003→            _json_response(self, {"status": "error", "message": "Invalid JSON body"}, 400)
  1004→        except ValueError as e:
  1005→            _json_response(self, {"status": "error", "message": str(e)}, 400)
  1006→        except Exception as e:
  1007→            log("ERROR", f"Unhandled {method} {path}: {traceback.format_exc()}")
  1008→            _json_response(self, {"status": "error", "message": f"Internal error: {str(e)}"}, 500)
  1009→
  1010→    def do_GET(self) -> None:
  1011→        self._route("GET")

  1021→        self.send_header("Access-Control-Allow-Headers", "Content-Type")
  1022→        self.end_headers()
