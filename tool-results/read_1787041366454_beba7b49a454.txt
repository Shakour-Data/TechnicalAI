    52→   121→function calcRSI(closes: number[], period: number = 14): number {
    53→   122→  if (closes.length < period + 1) return 50;
    54→   123→  let avgGain = 0;
    55→   124→  let avgLoss = 0;
    56→   125→  for (let i = 1; i <= period; i++) {
    57→   126→    const diff = closes[i] - closes[i - 1];
    58→   127→    if (diff > 0) avgGain += diff;
    59→   128→    else avgLoss += Math.abs(diff);
    60→   129→  }
    61→   130→  avgGain /= period;
    62→   131→  avgLoss /= period;
    63→   132→  for (let i = period + 1; i < closes.length; i++) {
    64→   133→    const diff = closes[i] - closes[i - 1];
    65→   134→    const gain = diff > 0 ? diff : 0;
    66→   135→    const loss = diff < 0 ? Math.abs(diff) : 0;
    67→   136→    avgGain = (avgGain * (period - 1) + gain) / period;
    68→   137→    avgLoss = (avgLoss * (period - 1) + loss) / period;
    69→   138→  }
    70→   139→  if (avgLoss === 0) return 100;
    71→   140→  const rs = avgGain / avgLoss;
    72→   141→  return 100 - 100 / (1 + rs);
    73→   142→}

    86→   155→  }
    87→   156→  if (negFlow === 0) return 100;
    88→   157→  const mfr = posFlow / negFlow;
    89→   158→  return 100 - 100 / (1 + mfr);

   106→   175→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   107→   176→  if (data.length < kPeriod) return { k: 50, d: 50 };
   108→   177→  const rawKs: number[] = [];
   109→   178→  for (let i = kPeriod - 1; i < data.length; i++) {
   110→   179→    let lowest = Infinity;
   111→   180→    let highest = -Infinity;
   112→   181→    for (let j = i - kPeriod + 1; j <= i; j++) {
   113→   182→      if (data[j].low < lowest) lowest = data[j].low;
   114→   183→      if (data[j].high > highest) highest = data[j].high;
   115→   184→    }
   116→   185→    const range = highest - lowest;
   117→   186→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   118→   187→  }
   119→   188→  const smoothedK: number[] = [];
   120→   189→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   121→   190→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   122→   191→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   123→   192→  }
   124→   193→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   125→   194→  const dSlice = smoothedK.slice(-smoothD);
   126→   195→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   127→   196→  return { k, d };
   128→   197→}

   146→   215→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   147→   216→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   148→   217→  const emaFast = emaCalc(closes, fast);
   149→   218→  const emaSlow = emaCalc(closes, slow);
   150→   219→  const macdLine = emaFast - emaSlow;
   151→   220→  // Calculate signal as EMA of MACD values
   152→   221→  const macdValues: number[] = [];
   153→   222→  const kFast = 2 / (fast + 1);
   154→   223→  const kSlow = 2 / (slow + 1);
   155→   224→  let eFast = sma(closes.slice(0, fast), fast);
   156→   225→  let eSlow = sma(closes.slice(0, slow), slow);
   157→   226→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   158→   227→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   159→   228→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   160→   229→    macdValues.push(eFast - eSlow);
   161→   230→  }
   162→   231→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   163→   232→  const kSig = 2 / (sig + 1);
   164→   233→  let signalVal = sma(macdValues.slice(0, sig), sig);
   165→   234→  for (let i = sig; i < macdValues.length; i++) {
   166→   235→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   167→   236→  }
   168→   237→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   169→   238→}

   184→   253→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   185→   254→  for (let i = period; i < trs.length; i++) {

   192→   261→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   193→   262→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   194→   263→  const plusDMs: number[] = [];
   195→   264→  const minusDMs: number[] = [];
   196→   265→  const trs: number[] = [];
   197→   266→  for (let i = 1; i < data.length; i++) {
   198→   267→    const upMove = data[i].high - data[i - 1].high;
   199→   268→    const downMove = data[i - 1].low - data[i].low;
   200→   269→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   201→   270→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   202→   271→    trs.push(Math.max(
   203→   272→      data[i].high - data[i].low,
   204→   273→      Math.abs(data[i].high - data[i - 1].close),
   205→   274→      Math.abs(data[i].low - data[i - 1].close)
   206→   275→    ));
   207→   276→  }
   208→   277→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   209→   278→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   210→   279→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   211→   280→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   212→   281→  const dxValues: number[] = [];
   213→   282→  for (let i = period; i < trs.length; i++) {
   214→   283→    smoothTR = smoothTR - smoothTR / period + trs[i];
   215→   284→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   216→   285→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   217→   286→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   218→   287→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   219→   288→    const diSum = diPlus + diMinus;
   220→   289→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   221→   290→    dxValues.push(dx);
   222→   291→  }
   223→   292→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   224→   293→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   225→   294→  for (let i = period; i < dxValues.length; i++) {
   226→   295→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   227→   296→  }
   228→   297→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   229→   298→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   230→   299→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   231→   300→}

   233→   302→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   234→   303→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   235→   304→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   236→   305→  let isLong = true;
   237→   306→  let af = step;
   238→   307→  let ep = data[0].high;
   239→   308→  let sar = data[0].low;
   240→   309→  for (let i = 1; i < data.length; i++) {
   241→   310→    const prevSar = sar;
   242→   311→    sar = prevSar + af * (ep - prevSar);
   243→   312→    // Ensure SAR is within bounds
   244→   313→    if (isLong) {
   245→   314→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   246→   315→      if (data[i].low < sar) {
   247→   316→        isLong = false;
   248→   317→        sar = ep;
   249→   318→        ep = data[i].low;
   250→   319→        af = step;
   251→   320→      } else {
   252→   321→        if (data[i].high > ep) {
   253→   322→          ep = data[i].high;
   254→   323→          af = Math.min(af + step, max);
   255→   324→        }
   256→   325→      }
   257→   326→    } else {
   258→   327→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   259→   328→      if (data[i].high > sar) {
   260→   329→        isLong = true;
   261→   330→        sar = ep;
   262→   331→        ep = data[i].high;
   263→   332→        af = step;
   264→   333→      } else {
   265→   334→        if (data[i].low < ep) {
   266→   335→          ep = data[i].low;
   267→   336→          af = Math.min(af + step, max);
   268→   337→        }
   269→   338→      }
   270→   339→    }
   271→   340→  }
   272→   341→  return sar;
   273→   342→}

   298→   367→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   299→   368→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   300→   369→  const n = values.length;
   301→   370→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   302→   371→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   303→   372→  for (let i = 0; i < n; i++) {
   304→   373→    sumX += i;
   305→   374→    sumY += values[i];
   306→   375→    sumXY += i * values[i];
   307→   376→    sumX2 += i * i;
   308→   377→    sumY2 += values[i] * values[i];
   309→   378→  }
   310→   379→  const denom = n * sumX2 - sumX * sumX;
   311→   380→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   312→   381→  const slope = (n * sumXY - sumX * sumY) / denom;
   313→   382→  const intercept = (sumY - slope * sumX) / n;
   314→   383→  const meanY = sumY / n;
   315→   384→  let ssRes = 0, ssTot = 0;
   316→   385→  for (let i = 0; i < n; i++) {
   317→   386→    const predicted = slope * i + intercept;
   318→   387→    ssRes += (values[i] - predicted) ** 2;
   319→   388→    ssTot += (values[i] - meanY) ** 2;
   320→   389→  }
   321→   390→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   322→   391→  return { slope, intercept, r2 };
   323→   392→}

   355→   424→// ─── Support / Resistance (6 each, adaptive gap, multi-factor strength) ───
   356→   425→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   357→   426→  resistances: number[];
   358→   427→  supports: number[];
   359→   428→  supportStrengths: LevelStrength[];
   360→   429→  resistanceStrengths: LevelStrength[];
   361→   430→  priceTargets: LevelStrength[];
   362→   431→} {
   363→   432→  const closes = data.map(d => d.close);
   364→   433→  const avgVolume = data.reduce((s, d) => s + d.volume, 0) / data.length;
   365→   434→
   366→   435→  // Tag each raw level with its source type for confluence scoring
   367→   436→  interface TaggedLevel { price: number; source: string };
   368→   437→  const allRaw: TaggedLevel[] = [];
   369→   438→
   370→   439→  // 1. Pivot Points (source: 'pivot')
   371→   440→  const prev = data[data.length - 2] ?? data[data.length - 1];
   372→   441→  if (prev) {
   373→   442→    const pp = (prev.high + prev.low + prev.close) / 3;
   374→   443→    const hl = prev.high - prev.low;
   375→   444→    const pivotR = [
   376→   445→      2 * pp - prev.low, pp + hl,
   377→   446→      prev.high + 2 * (pp - prev.low),
   378→   447→      prev.high + 2 * (pp - prev.low) + hl,
   379→   448→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   380→   449→    ];
   381→   450→    const pivotS = [
   382→   451→      2 * pp - prev.high, pp - hl,
   383→   452→      prev.low - 2 * (prev.high - pp),
   384→   453→      prev.low - 2 * (prev.high - pp) - hl,
   385→   454→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   386→   455→    ];
   387→   456→    for (const p of pivotR) allRaw.push({ price: p, source: 'pivot' });
   388→   457→    for (const p of pivotS) allRaw.push({ price: p, source: 'pivot' });
   389→   458→  }
   390→   459→
   391→   460→  // 2. Swing levels (source: 'swing')
   392→   461→  for (const lb of [3, 5, 7, 10]) {
   393→   462→    const swings = findSwingLevels(data, lb);
   394→   463→    for (const h of swings.highs) allRaw.push({ price: h, source: 'swing' });
   395→   464→    for (const l of swings.lows) allRaw.push({ price: l, source: 'swing' });
   396→   465→  }
   397→   466→
   398→   467→  // 3. Moving Average levels (source: 'ma')
   399→   468→  for (const period of [5, 10, 21, 50, 100, 200]) {
   400→   469→    const v = sma(closes, period);
   401→   470→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01)
   402→   471→      allRaw.push({ price: v, source: 'ma' });
   403→   472→  }
   404→   473→  const ema12 = emaCalc(closes, 12);
   405→   474→  const ema26 = emaCalc(closes, 26);
   406→   475→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01)
   407→   476→    allRaw.push({ price: ema12, source: 'ma' });
   408→   477→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01)
   409→   478→    allRaw.push({ price: ema26, source: 'ma' });
   410→   479→
   411→   480→  // 4. Bollinger Band levels (source: 'bb')
   412→   481→  const bbLevels = calcBollingerBands(closes);
   413→   482→  if (bbLevels.upper > 0) allRaw.push({ price: bbLevels.upper, source: 'bb' });
   414→   483→  if (bbLevels.lower > 0) allRaw.push({ price: bbLevels.lower, source: 'bb' });
   415→   484→
   416→   485→  // 5. Round number / psychological levels (source: 'round')
   417→   486→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
   418→   487→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
   419→   488→    const p = Math.round(currentPrice * mult / magnitude) * magnitude;
   420→   489→    allRaw.push({ price: p, source: 'round' });
   421→   490→  }
   422→   491→
   423→   492→  // 6. Recent high/low zones (source: 'hilo')
   424→   493→  for (const w of [5, 22, 66]) {
   425→   494→    const slice = data.slice(-Math.min(w, data.length));
   426→   495→    if (slice.length > 0) {
   427→   496→      allRaw.push({ price: Math.max(...slice.map(d => d.high)), source: 'hilo' });
   428→   497→      allRaw.push({ price: Math.min(...slice.map(d => d.low)), source: 'hilo' });
   429→   498→    }
   430→   499→  }
   431→   500→
   432→   501→  // 7. Round all and filter invalid
   433→   502→  const rounded: TaggedLevel[] = allRaw
   434→   503→    .map(l => ({ price: roundToNice(l.price), source: l.source }))
   435→   504→    .filter(l => l.price > 0);
   436→   505→
   437→   506→  // 8. Cluster nearby levels (within 0.5%) — merge sources, keep most common price
   438→   507→  interface Cluster {
   439→   508→    price: number;
   440→   509→    sources: Set<string>;
   441→   510→    touchCount: number;
   442→   511→    totalVolumeAtTouches: number;
   443→   512→    mostRecentTouchBar: number;
   444→   513→  }
   445→   514→  const clusters: Cluster[] = [];
   446→   515→  for (const lvl of rounded) {
   447→   516→    let merged = false;
   448→   517→    for (const c of clusters) {
   449→   518→      if (Math.abs(c.price - lvl.price) / Math.max(c.price, 1) < 0.005) {
   450→   519→        c.sources.add(lvl.source);
   451→   520→        merged = true;
   452→   521→        break;
   453→   522→      }
   454→   523→    }
   455→   524→    if (!merged) {
   456→   525→      clusters.push({ price: lvl.price, sources: new Set([lvl.source]), touchCount: 0, totalVolumeAtTouches: 0, mostRecentTouchBar: -1 });
   457→   526→    }
   458→   527→  }
   459→   528→
   460→   529→  // 9. Count actual price TOUCHES (rejections) and volume at each cluster level
   461→   530→  // A real touch = bar wick reached the level AND bar body closed away from it
   462→   531→  const touchThreshold = currentPrice * 0.004; // 0.4% tight proximity
   463→   532→  for (let i = 0; i < data.length; i++) {
   464→   533→    const bar = data[i];
   465→   534→    for (const c of clusters) {
   466→   535→      const isResistance = c.price > currentPrice;
   467→   536→      if (isResistance) {
   468→   537→        // Resistance: bar high reached near the level AND closed below it (rejection)
   469→   538→        if (bar.high >= c.price - touchThreshold && bar.close < c.price) {
   470→   539→          c.touchCount++;
   471→   540→          c.totalVolumeAtTouches += bar.volume;
   472→   541→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   473→   542→        }
   474→   543→      } else {
   475→   544→        // Support: bar low reached near the level AND closed above it (rejection)
   476→   545→        if (bar.low <= c.price + touchThreshold && bar.close > c.price) {
   477→   546→          c.touchCount++;
   478→   547→          c.totalVolumeAtTouches += bar.volume;
   479→   548→          c.mostRecentTouchBar = Math.max(c.mostRecentTouchBar, i);
   480→   549→        }
   481→   550→      }
   482→   551→    }
   483→   552→  }
   484→   553→
   485→   554→  // 10. Calculate multi-factor strength (0-10)
   486→   555→  const levelStrengths: LevelStrength[] = clusters.map(c => {
   487→   556→    // Factor 1: Source Confluence (0-2.5 pts)
   488→   557→    const sourceScore = Math.min(2.5, c.sources.size * 0.5);
   489→   558→    const hasSwingAndPivot = c.sources.has('swing') && c.sources.has('pivot') ? 0.5 : 0;
   490→   559→    const maConfluence = c.sources.has('ma') ? 0.5 : 0;
   491→   560→    const confluenceScore = Math.min(2.5, sourceScore + hasSwingAndPivot + maConfluence);
   492→   561→
   493→   562→    // Factor 2: Touch Count — CAPPED by recency (0-2 pts)
   494→   563→    // Only count touches in the most recent 50% of data to avoid ancient history inflating
   495→   564→    const recentBarStart = Math.floor(data.length * 0.5);
   496→   565→    const recentTouches = c.mostRecentTouchBar >= recentBarStart ? c.touchCount : Math.max(0, c.touchCount - 1);
   497→   566→    let touchScore: number;
   498→   567→    if (recentTouches === 0) touchScore = 0;
   499→   568→    else if (recentTouches === 1) touchScore = 0.4;
   500→   569→    else if (recentTouches <= 2) touchScore = 0.8;
   501→   570→    else if (recentTouches <= 4) touchScore = 1.2;
   502→   571→    else if (recentTouches <= 7) touchScore = 1.6;
   503→   572→    else touchScore = 2.0;
   504→   573→
   505→   574→    // Factor 3: Volume Confirmation (0-1.5 pts)
   506→   575→    let volumeScore = 0;
   507→   576→    if (c.touchCount > 0) {
   508→   577→      const avgVolAtTouch = c.totalVolumeAtTouches / c.touchCount;
   509→   578→      const volRatio = avgVolume > 0 ? avgVolAtTouch / avgVolume : 1;
   510→   579→      if (volRatio > 2.0) volumeScore = 1.5;
   511→   580→      else if (volRatio > 1.5) volumeScore = 1.2;
   512→   581→      else if (volRatio > 1.0) volumeScore = 0.8;
   513→   582→      else if (volRatio > 0.7) volumeScore = 0.4;
   514→   583→      else volumeScore = 0.2;
   515→   584→    }
   516→   585→
   517→   586→    // Factor 4: Freshness / Recency (0-1.5 pts)
   518→   587→    let freshnessScore = 0;
   519→   588→    if (c.touchCount > 0 && data.length > 0) {
   520→   589→      const barsAgo = data.length - 1 - c.mostRecentTouchBar;
   521→   590→      if (barsAgo <= 3) freshnessScore = 1.5;
   522→   591→      else if (barsAgo <= 10) freshnessScore = 1.2;
   523→   592→      else if (barsAgo <= 25) freshnessScore = 0.8;
   524→   593→      else if (barsAgo <= 50) freshnessScore = 0.4;
   525→   594→      else freshnessScore = 0.1;
   526→   595→    }
   527→   596→
   528→   597→    // Factor 5: Proximity to current price (0-1 pt)
   529→   598→    const distRatio = Math.abs(c.price - currentPrice) / currentPrice;
   530→   599→    let proximityScore = 0;
   531→   600→    if (distRatio < 0.015) proximityScore = 1.0;
   532→   601→    else if (distRatio < 0.03) proximityScore = 0.7;
   533→   602→    else if (distRatio < 0.06) proximityScore = 0.4;
   534→   603→    else if (distRatio < 0.10) proximityScore = 0.2;
   535→   604→
   536→   605→    const total = confluenceScore + touchScore + volumeScore + freshnessScore + proximityScore;
   537→   606→    const strength = Math.min(10, Math.max(1, Math.round(total)));
   538→   607→
   539→   608→    return { price: c.price, strength, isTarget: false };
   540→   609→  });
   541→   610→
   542→   611→  // 11. Separate above/below current price
   543→   612→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   544→   613→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   545→   614→
   546→   615→  // 12. Adaptive gap enforcement — keep highest strength in cluster
   547→   616→  function enforceGap(levels: LevelStrength[], targetCount: number): LevelStrength[] {
   548→   617→    const gapSteps = [0.05, 0.04, 0.03, 0.025, 0.02, 0.015, 0.01, 0.008, 0.005];
   549→   618→    for (const gapPct of gapSteps) {
   550→   619→      if (levels.length === 0) return [];
   551→   620→      const result: LevelStrength[] = [levels[0]];
   552→   621→      for (let i = 1; i < levels.length; i++) {
   553→   622→        if (result.length >= targetCount) break;
   554→   623→        const prevPrice = result[result.length - 1].price;
   555→   624→        const currPrice = levels[i].price;
   556→   625→        const gap = Math.abs(currPrice - prevPrice) / prevPrice;
   557→   626→        if (gap >= gapPct) {
   558→   627→          result.push(levels[i]);
   559→   628→        } else {
   560→   629→          // Merge: keep stronger level's price, add 0.5 for confluence
   561→   630→          const keep = levels[i].strength >= result[result.length - 1].strength ? levels[i] : result[result.length - 1];
   562→   631→          const mergedStrength = Math.min(10, Math.max(keep.strength, result[result.length - 1].strength) + 0.5);
   563→   632→          result[result.length - 1] = { ...keep, strength: Math.round(mergedStrength) };
   564→   633→        }
   565→   634→      }
   566→   635→      if (result.length >= targetCount) return result;
   567→   636→    }
   568→   637→    return levels.slice(0, targetCount);
   569→   638→  }
   570→   639→
   571→   640→  const finalResistances = enforceGap(above, 6);
   572→   641→  const finalSupports = enforceGap(below, 6);
   573→   642→
   574→   643→  // 13. Determine price targets: top 2 by strength, or any above 7
   575→   644→  const allLevels = [...finalResistances, ...finalSupports];
   576→   645→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   577→   646→  let targets: LevelStrength[];
   578→   647→  if (sortedByStrength.some(l => l.strength >= 7)) {
   579→   648→    targets = sortedByStrength.filter(l => l.strength >= 7).slice(0, 3);
   580→   649→  } else {
   581→   650→    targets = sortedByStrength.slice(0, 2);
   582→   651→  }
   583→   652→
   584→   653→  // Mark targets
   585→   654→  targets = targets.map(t => ({ ...t, isTarget: true }));
   586→   655→
   587→   656→  // Also mark in final arrays
   588→   657→  finalResistances.forEach(r => {
   589→   658→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   590→   659→  });
   591→   660→  finalSupports.forEach(s => {
   592→   661→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   593→   662→  });
   594→   663→
   595→   664→  return {
   596→   665→    resistances: finalResistances.map(l => l.price),
   597→   666→    supports: finalSupports.map(l => l.price),
   598→   667→    supportStrengths: finalSupports,
   599→   668→    resistanceStrengths: finalResistances,
   600→   669→    priceTargets: targets,
   601→   670→  };
   602→   671→}

   605→   674→function calcTrend(closes: number[], period: number): TrendResult {
   606→   675→  const slice = closes.slice(-Math.min(period, closes.length));
   607→   676→  const { slope, r2 } = linearRegression(slice);
   608→   677→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   609→   678→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   610→   679→  const priceThreshold = avgPrice * 0.001; // 0.1%
   611→   680→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   612→   681→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   613→   682→}
