   145→// ═══════════════════════════════════════════════════════════════════════════════
   146→
   147→interface EdgeInfo {
   148→  from: string;
   149→  to: string;
   150→  label: string;
   151→  type: string;
   152→  prob: number;
   153→}
   154→
   155→interface PathInfo {
   156→  nodes: string[];
   157→  edges: number[];
   158→  prob: number;
   159→  target: string;
   160→}
   161→
   162→// Calculate edge probabilities based on market conditions
   163→function calcEdgeProbabilities(
   164→  bullScore: number, adxVal: number
   165→): Record<string, number> {
   166→  const adxW = Math.min(adxVal / 100, 1);
   167→  const typeWeights: Record<string, number> = {
   168→    up: bullScore * 0.6 + adxW * 0.4,
   169→    pullback: 0.3,
   170→    down: (1 - bullScore) * 0.6 + adxW * 0.4,
   171→    risk: 0.15 * (1.2 - adxW),
   172→  };
   173→
   174→  // Build adjacency: for each node, group outgoing edges by type
   175→  const adj: Record<string, { idx: number; type: string }[]> = {};
   176→  EDGES.forEach((e, i) => {
   177→    if (!adj[e[0]]) adj[e[0]] = [];
   178→    adj[e[0]].push({ idx: i, type: e[3] });
   179→  });
   180→
   181→  const edgeProbs: Record<string, number> = {};
   182→
   183→  for (const [node, outEdges] of Object.entries(adj)) {
   184→    // Sum weights of outgoing edges
   185→    const totalWeight = outEdges.reduce((sum, e) => sum + (typeWeights[e.type] ?? 0.1), 0);
   186→    if (totalWeight === 0) {
   187→      outEdges.forEach(e => { edgeProbs[String(e.idx)] = 1 / outEdges.length; });
   188→      continue;
   189→    }
   190→    outEdges.forEach(e => {
   191→      edgeProbs[String(e.idx)] = (typeWeights[e.type] ?? 0.1) / totalWeight;
   192→    });
   193→  }
   194→
   195→  return edgeProbs;
   196→}
   197→
   198→// Find all paths from 'A' to terminal nodes using DFS
   199→function findAllPaths(edgeProbs: Record<string, number>): PathInfo[] {
   200→  const adj: Record<string, { to: string; edgeIdx: number }[]> = {};
   201→  EDGES.forEach((e, i) => {
   202→    if (!adj[e[0]]) adj[e[0]] = [];
   203→    adj[e[0]].push({ to: e[1], edgeIdx: i });
   204→  });
   205→
   206→  const paths: PathInfo[] = [];
   207→  const MAX_PATHS = 200;
   208→
   209→  function dfs(node: string, visited: Set<string>, currentPath: string[], edgeIndices: number[], currentProb: number) {
   210→    if (SCENARIO_KEYS.includes(node as any)) {
   211→      paths.push({ nodes: [...currentPath, node], edges: [...edgeIndices], prob: currentProb, target: node });
   212→      return;
   213→    }
   214→    if (paths.length >= MAX_PATHS) return;
   215→    if (visited.has(node)) return;
   216→    visited.add(node);
   217→
   218→    const outs = adj[node] ?? [];
   219→    for (const { to, edgeIdx } of outs) {
   220→      const ep = edgeProbs[String(edgeIdx)] ?? 0.1;
   221→      dfs(to, visited, [...currentPath, node], [...edgeIndices, edgeIdx], currentProb * ep);
   222→    }
   223→    visited.delete(node);
   224→  }
   225→
   226→  dfs('A', new Set(), [], [], 1);
   227→  return paths.sort((a, b) => b.prob - a.prob);
   228→}
