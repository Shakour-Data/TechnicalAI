    50→  }
    51→
    52→  const item = pendingQueue.shift()!;
    53→  activeCall = true;
    54→
    55→  callZaiSDK(item.messages)
    56→    .then(text => {
    57→      lastAITime = Date.now();
    58→      cooldownUntil = 0;
    59→      item.resolve(text);
    60→    })
    61→    .catch(err => {
    62→      const msg = err.message || '';
    63→      if (msg.includes('429')) {
    64→        const cd = Math.max(cooldownUntil - Date.now(), 0) + 90_000;
    65→        cooldownUntil = Date.now() + Math.min(cd, 300_000);
    66→        console.warn(`[AI v5.1] 429, cooldown ${Math.min(cd, 300_000) / 1000}s`);
    67→      }
    68→      item.reject(err);
    69→    })
    70→    .finally(() => {
    71→      activeCall = false;
    72→      setTimeout(processAIQueue, 2000);
    73→    });
    74→}

    76→function queueAI(messages: { role: string; content: string }[], timeoutMs = 90_000): Promise<string> {
    77→  return new Promise((resolve, reject) => {
    78→    const timer = setTimeout(() => {
    79→      const idx = pendingQueue.findIndex(q => q.messages === messages);
    80→      if (idx >= 0) pendingQueue.splice(idx, 1);
    81→      reject(new Error('AI request timeout'));
    82→    }, timeoutMs);
    83→
    84→    pendingQueue.push({
    85→      messages,
    86→      resolve: (text) => { clearTimeout(timer); resolve(text); },
    87→      reject: (err) => { clearTimeout(timer); reject(err); },
    88→    });
    89→
    90→    processAIQueue();
    91→  });
    92→}

    98→// Common emoji ranges to strip from AI output

   100→  /\p{Emoji_Presentation}/gu,
   101→  /\p{Extended_Pictographic}/gu,

   104→function stripEmojis(text: string): string {
   105→  let result = text;
   106→  for (const pattern of EMOJI_PATTERNS) {
   107→    result = result.replace(pattern, '').trim();
   108→  }
   109→  return result;

   112→async function callZaiSDK(messages: { role: string; content: string }[], maxRetries = 4): Promise<string> {
   113→  for (let attempt = 1; attempt <= maxRetries; attempt++) {
   114→    try {
   115→      const zai = await getZai();
   116→      const completion = await zai.chat.completions.create({
   117→        messages,
   118→        thinking: { type: 'disabled' },
   119→      });
   120→      const raw = completion.choices[0]?.message?.content;
   121→      if (!raw || raw.trim().length === 0) {
   122→        throw new Error('Empty AI response');
   123→      }
   124→      const text = stripEmojis(raw).trim();
   125→      if (text.length < 10) {
   126→        throw new Error('AI response too short: ' + text.slice(0, 100));
   127→      }
   128→      return text;
   129→    } catch (err: unknown) {
   130→      const msg = err instanceof Error ? err.message : String(err);
   131→      if (msg.includes('429') && attempt < maxRetries) {
   132→        const waitMs = Math.round(jitter(20_000 * Math.pow(2, attempt - 1))); // 20s, 40s, 80s, 160s + jitter
   133→        console.warn(`[AI v5.1] 429 retry ${attempt}/${maxRetries}, waiting ${Math.round(waitMs / 1000)}s...`);
   134→        await sleep(waitMs);
   135→        continue;
   136→      }
   137→      throw err;
   138→    }
   139→  }
   140→  throw new Error('Max retries exceeded');
   141→}
