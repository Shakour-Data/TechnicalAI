   255→  const handleSelect = useCallback(async (symbol: string, category?: string, insCode?: string, tgjuKey?: string, finpySector?: string, finpyIndex?: string, webId?: number, yahooSymbol?: string) => {
   256→    // Abort any previous in-flight fetch
   257→    if (fetchControllerRef.current) {
   258→      fetchControllerRef.current.abort();
   259→    }
   260→    const controller = new AbortController();
   261→    fetchControllerRef.current = controller;
   262→
   263→    // Yahoo Finance stock: fetch historical data via Yahoo Finance API
   264→    if (category === 'yahoo_stock' && yahooSymbol) {
   265→      setLoading(true);
   266→      setLoadingMessage('در حال دریافت داده‌های تاریخی از یاهو فایننس ...');

   269→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId, yahooSymbol };
   270→      try {
   271→        const res = await fetch('/api/yahoo-analysis?symbol=' + encodeURIComponent(yahooSymbol), { signal: controller.signal });
   272→        if (controller.signal.aborted) return;
   273→        if (!res.ok) {
   274→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   275→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   276→          throw new Error(errMsg);
   277→        }
   278→        const json = await res.json();
   279→        if (controller.signal.aborted) return;
   280→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   281→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   282→        }
   283→        setData({ ...json, isYahoo: true });
   284→        setActivePanel('visual');
   285→      } catch (err) {
   286→        if (controller.signal.aborted) return;
   287→        setError(String(err));
   288→      } finally {
   289→        if (!controller.signal.aborted) {
   290→          setLoading(false);
   291→          setLoadingMessage(null);
   292→        }
   293→      }
   294→      return;
   295→    }
   296→
   297→    // TGJU instrument: fetch historical data via tgju.org chart API
   298→    if (category && TGJU_CATEGORIES.has(category) && tgjuKey) {
   299→      setLoading(true);
   300→      setLoadingMessage('در حال دریافت داده‌های تاریخی ... (حدود ۱۵ ثانیه)');
   301→      setError(null);
   302→      setData(null);
   303→      lastFetchRef.current = { symbol, category, insCode, tgjuKey, finpySector, finpyIndex, webId, yahooSymbol };
   304→      try {
   305→        const res = await fetch('/api/tgju-analysis?key=' + encodeURIComponent(tgjuKey), { signal: controller.signal });
   306→        if (controller.signal.aborted) return;
   307→        if (!res.ok) {
   308→          let errMsg = 'خطا در دریافت داده‌های تاریخی';
   309→          try { const err = await res.json(); if (err.error) errMsg = err.error; } catch (_) { /* non-JSON error */ }
   310→          throw new Error(errMsg);
   311→        }
   312→        const json = await res.json();
   313→        if (controller.signal.aborted) return;
   314→        if (!json.candles || json.candles.length === 0 || !json.ta) {
   315→          throw new Error(json.error || 'داده‌های تاریخی کافی برای تحلیل وجود ندارد');
   316→        }
   317→        setData({ ...json, isTgju: true });
   318→        setActivePanel('visual');
   319→      } catch (err) {
   320→        if (controller.signal.aborted) return;
   321→        setError(String(err));
