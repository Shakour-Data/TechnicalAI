    39→def gregorian_to_jalali(gy, gm, gd):
    40→    """Convert Gregorian to Jalali date string using jdatetime."""
    41→    try:
    42→        jdate = jdatetime.date.fromgregorian(year=gy, month=gm, day=gd)
    43→        return f"{jdate.year}/{str(jdate.month).zfill(2)}/{str(jdate.day).zfill(2)}"
    44→    except:
    45→        return f"{gy}/{str(gm).zfill(2)}/{str(gd).zfill(2)}"

    48→# ── z-ai CLI page_reader wrapper ─────────────────────────────────
    49→def zai_page_reader(url: str, timeout: int = 120) -> str:
    50→    """Fetch URL content using z-ai CLI page_reader. Retries up to 3 times."""
    51→    tmp = f'/tmp/zai_{hash(url) % 100000}_{int(time.time())}.json'
    52→    last_error = None
    53→    for attempt in range(3):
    54→        try:
    55→            result = subprocess.run(
    56→                ['z-ai', 'function', '-n', 'page_reader',
    57→                 '-a', json.dumps({'url': url}),
    58→                 '-o', tmp],
    59→                capture_output=True, text=True, timeout=timeout
    60→            )
    61→            if result.returncode != 0:
    62→                last_error = f"z-ai CLI failed (exit {result.returncode}): {result.stderr[:300]}"
    63→                print(f'[z-ai] Attempt {attempt+1} failed: {last_error[:100]}', flush=True)
    64→                time.sleep(3 * (attempt + 1))  # Backoff: 3s, 6s, 9s
    65→                continue
    66→            with open(tmp) as f:
    67→                data = json.load(f)
    68→            html = data.get('data', {}).get('html', '')
    69→            if not html:
    70→                last_error = 'Empty response from page_reader'
    71→                print(f'[z-ai] Attempt {attempt+1}: empty response', flush=True)
    72→                time.sleep(3 * (attempt + 1))
    73→                continue
    74→            return html
    75→        except subprocess.TimeoutExpired:
    76→            last_error = f'z-ai CLI timed out after {timeout}s'
    77→            print(f'[z-ai] Attempt {attempt+1}: {last_error}', flush=True)
    78→            time.sleep(3)
    79→        except Exception as e:
    80→            last_error = str(e)
    81→            print(f'[z-ai] Attempt {attempt+1} error: {last_error[:100]}', flush=True)
    82→            time.sleep(3)
    83→        finally:
    84→            if os.path.exists(tmp):
    85→                try:
    86→                    os.unlink(tmp)
    87→                except:
    88→                    pass
    89→            # New temp file for next attempt
    90→            tmp = f'/tmp/zai_{hash(url) % 100000}_{int(time.time())}.json'
    91→    raise RuntimeError(f'z-ai page_reader failed after 3 attempts: {last_error}')

    94→# ── TSETMC B2 API fetcher ────────────────────────────────────────
    95→def fetch_b2_data(web_id: str, timeout: int = 120) -> list:
    96→    """Fetch index B2 history from TSETMC CDN via z-ai CLI."""
    97→    url = f'http://cdn.tsetmc.com/api/Index/GetIndexB2History/{web_id}'
    98→    html = zai_page_reader(url, timeout)
    99→
   100→    # Extract JSON from <pre> tag
   101→    pre = re.search(r'<pre[^>]*>([\s\S]*?)</pre>', html, re.IGNORECASE)
   102→    json_str = pre.group(1) if pre else re.sub(r'<[^>]+>', '', html).strip()
   103→
   104→    # Unescape HTML entities
   105→    json_str = json_str.replace('&amp;', '&').replace('&lt;', '<').replace('&gt;', '>').replace('&quot;', '"')
   106→
   107→    # Try direct parse first (full JSON: {"indexB2":[...]}
   108→    try:
   109→        data = json.loads(json_str)
   110→        return data.get('indexB2', [])
   111→    except json.JSONDecodeError:
   112→        pass
   113→
   114→    # Fallback: handle truncated JSON — find last complete object
   115→    last_brace = json_str.rfind('}')
   116→    if last_brace > 0:
   117→        # Try to find the complete outer object: {"indexB2":[...]}
   118→        for end_pos in range(last_brace, max(last_brace - 10, 0), -1):
   119→            try:
   120→                candidate = json_str[:end_pos + 1]
   121→                data = json.loads(candidate)
   122→                items = data.get('indexB2', [])
   123→                if items:
   124→                    return items
   125→            except json.JSONDecodeError:
   126→                continue
   127→
   128→    raise RuntimeError(f'Failed to parse B2 JSON ({len(json_str)} chars)')

   131→def b2_to_candles(b2_items: list) -> list:
   132→    """Convert B2 items to candle list."""
   133→    candles = []
   134→    for item in b2_items:
   135→        close = item.get('xNivInuClMresIbs', 0)
   136→        if close <= 0:
   137→            continue
   138→        deven = str(item.get('dEven', 0))
   139→        if len(deven) < 8:
   140→            continue
   141→        date_str = gregorian_to_jalali(int(deven[:4]), int(deven[4:6]), int(deven[6:8]))
   142→        first = item.get('xNivInuPhMresIbs', 0)
   143→        prev = item.get('xNivInuPbMresIbs', 0)
   144→        open_price = first if first > 0 else prev
   145→        candles.append({
   146→            'date': date_str,
   147→            'open': round(open_price, 1),
   148→            'high': round(max(open_price, close), 1),
   149→            'low': round(min(open_price, close), 1),
   150→            'close': round(close, 1),
   151→            'volume': 0,
   152→        })
   153→    candles.sort(key=lambda x: x['date'])
   154→    return candles

   157→# ── File cache ───────────────────────────────────────────────────
   158→def file_cache_path(key: str) -> str:
   159→    """Generate cache file path from key. Uses key prefix for filename prefix."""
   160→    safe = re.sub(r'[^a-zA-Z0-9\u0600-\u06FF_-]', '_', key)
   161→    # Determine file prefix from key: stock_xxx → stock-, idx_xxx → index-, etc.
   162→    prefix = 'index'
   163→    for p in ('stock', 'symbols', 'sec', 'idx'):
   164→        if key.startswith(p + '_'):
   165→            prefix = p
   166→            break
   167→    return os.path.join(FILE_CACHE_DIR, f'{prefix}-{safe}.json')
   168→
   169→
   170→def load_file_cache(key: str) -> list | None:
   171→    path = file_cache_path(key)
   172→    if not os.path.exists(path):
   173→        return None
   174→    try:
   175→        with open(path) as f:
   176→            entry = json.load(f)
   177→        if time.time() * 1000 - entry.get('time', 0) > CACHE_TTL * 1000:
   178→            return None
   179→        return entry.get('data', [])
   180→    except:
   181→        return None
   182→
   183→
   184→def save_file_cache(key: str, data):
   185→    try:
   186→        os.makedirs(FILE_CACHE_DIR, exist_ok=True)
   187→        with open(file_cache_path(key), 'w') as f:
   188→            json.dump({'data': data, 'time': int(time.time() * 1000)}, f, ensure_ascii=False)
   189→    except Exception:
   190→        pass

   193→# ── finpy-tse stock data helpers ─────────────────────────────────
   194→def _fetch_price_history(symbol: str, adjust: bool) -> list:
   195→    """Run finpy_tse.Get_Price_History in a thread-safe function.
   196→    Returns list of candle dicts on success, or raises on failure."""
   197→    import finpy_tse
   198→
   199→    df = finpy_tse.Get_Price_History(
   200→        stock=symbol,
   201→        ignore_date=True,
   202→        adjust_price=adjust,
   203→        show_weekday=False,
   204→        double_date=False,
   205→    )
   206→    if df is None or (hasattr(df, 'empty') and df.empty):
   207→        raise RuntimeError(f'No data returned for symbol "{symbol}"')
   208→
   209→    # Reset index to get J-Date as a column
   210→    df = df.reset_index()
   211→    # J-Date is the former index, named 'J-Date' or the first column
   212→    date_col = 'J-Date' if 'J-Date' in df.columns else df.columns[0]
   213→
   214→    # When adjust_price=True, use Adj columns; otherwise use raw columns
   215→    if adjust and 'Adj Close' in df.columns:
   216→        open_col, high_col, low_col, close_col = 'Adj Open', 'Adj High', 'Adj Low', 'Adj Close'
   217→    else:
   218→        open_col, high_col, low_col, close_col = 'Open', 'High', 'Low', 'Close'
   219→
   220→    candles = []
   221→    for _, row in df.iterrows():
   222→        # Convert date from 1404-01-15 to 1404/01/15
   223→        raw_date = str(row[date_col])
   224→        date_str = raw_date.replace('-', '/')
   225→        try:
   226→            candles.append({
   227→                'date': date_str,
   228→                'open': int(row[open_col]),
   229→                'high': int(row[high_col]),
   230→                'low': int(row[low_col]),
   231→                'close': int(row[close_col]),
   232→                'volume': int(row['Volume']),
   233→            })
   234→        except (ValueError, TypeError, KeyError):
   235→            # Skip rows with bad data
   236→            continue
   237→
   238→    # Sort by date ascending
   239→    candles.sort(key=lambda x: x['date'])
   240→    return candles

   355→def resolve_sector_name(name: str) -> str | None:
   356→    """Resolve sector name to finpy-tse sector name."""
   357→    if name in SECTOR_WEB_IDS:
   358→        return name
   359→    if name in SECTOR_ALIASES:
   360→        return SECTOR_ALIASES[name]
   361→    # Try partial match
   362→    for key in SECTOR_WEB_IDS:
   363→        if name in key or key in name:
   364→            return key
   365→    return None

   368→class Handler(BaseHTTPRequestHandler):
   369→    def log_message(self, fmt, *args):
   370→        # Suppress default HTTP logging
   371→        pass
   372→
   373→    def _send_json(self, data, status=200):
   374→        body = json.dumps(data, ensure_ascii=False).encode('utf-8')
   375→        self.send_response(status)
   376→        self.send_header('Content-Type', 'application/json; charset=utf-8')
   377→        self.send_header('Content-Length', str(len(body)))
   378→        self.end_headers()
   379→        self.wfile.write(body)
   380→
   381→    def do_GET(self):
   382→        parsed = urlparse(self.path)
   383→        params = parse_qs(parsed.query)
   384→        path = parsed.path
   385→
   386→        try:
   387→            if path == '/health':
   388→                return self._send_json({
   389→                    'status': 'ok',
   390→                    'service': 'finpy-tse-index-service',
   391→                    'main_indices': len(MAIN_INDEX_IDS),
   392→                    'sectors': len(SECTOR_WEB_IDS),
   393→                    'stock_history': True,
   394→                    'symbols_search': True,
   395→                })
   396→
   397→            elif path == '/api/index-history':
   398→                return self._handle_index_history(params)
   399→
   400→            elif path == '/api/sector-history':
   401→                return self._handle_sector_history(params)
   402→
   403→            elif path == '/api/sector-list':
   404→                return self._send_json({
   405→                    'sectors': list(SECTOR_WEB_IDS.keys()),
   406→                    'main_indices': list(MAIN_INDEX_IDS.keys()),
   407→                })
   408→
   409→            elif path == '/api/stock-history':
   410→                return self._handle_stock_history(params)
   411→
   412→            elif path == '/api/symbols-search':
   413→                return self._handle_symbols_search(params)
   414→
   415→            else:
   416→                self._send_json({'error': 'Not found'}, 404)
   417→        except Exception as e:
   418→            traceback.print_exc()
   419→            self._send_json({'error': str(e)}, 500)

   421→    # ── Stock price history ───────────────────────────────────────
   422→    def _handle_stock_history(self, params):
   423→        """Handle /api/stock-history?symbol=خودرو&adjust=1
   424→
   425→        Tries finpy_tse.Get_Price_History via ThreadPoolExecutor with 30s timeout.
   426→        If it fails/times out, returns JSON error (caller falls back to file cache).
   427→        """
   428→        symbol = (params.get('symbol') or [''])[0].strip()
   429→        if not symbol:
   430→            return self._send_json({'error': 'پارامتر symbol الزامی است', 'candles': []}, 400)
   431→
   432→        adjust_str = (params.get('adjust') or ['1'])[0].strip()
   433→        adjust = adjust_str != '0'
   434→
   435→        cache_key = f'stock_{symbol}_{"adj" if adjust else "raw"}'
   436→
   437→        # Check file cache first
   438→        cached = load_file_cache(cache_key)
   439→        if cached:
   440→            print(f'[stock] Cache hit: {symbol} (adjust={adjust}, {len(cached)} candles)', flush=True)
   441→            return self._send_json({
   442→                'symbol': symbol,
   443→                'adjust': adjust,
   444→                'count': len(cached),
   445→                'source': 'cache',
   446→                'candles': cached,
   447→            })
   448→
   449→        # Run finpy-tse in a thread with 30s timeout
   450→        print(f'[stock] Fetching price history for "{symbol}" (adjust={adjust}) via finpy-tse...', flush=True)
   451→        try:
   452→            with ThreadPoolExecutor(max_workers=1) as executor:
   453→                future = executor.submit(_fetch_price_history, symbol, adjust)
   454→                try:
   455→                    candles = future.result(timeout=30)
   456→                except FuturesTimeoutError:
   457→                    future.cancel()
   458→                    print(f'[stock] Timeout after 30s for "{symbol}"', flush=True)
   459→                    return self._send_json({
   460→                        'error': 'finpy-tse timed out (30s) — likely CDN unreachable from sandbox',
   461→                        'symbol': symbol,
   462→                        'candles': [],
   463→                    }, 504)
   464→        except Exception as e:
   465→            err_msg = str(e)
   466→            print(f'[stock] Error for "{symbol}": {err_msg[:200]}', flush=True)
   467→            return self._send_json({
   468→                'error': err_msg,
   469→                'symbol': symbol,
   470→                'candles': [],
   471→            }, 500)
   472→
   473→        if not candles:
   474→            return self._send_json({
   475→                'error': f'داده‌ای برای نماد "{symbol}" یافت نشد',
   476→                'symbol': symbol,

   492→    def _handle_symbols_search(self, params):
   493→        """Handle /api/symbols-search?query=خودرو
   494→
   495→        Tries finpy_tse.Build_Market_StockList via ThreadPoolExecutor with 60s timeout.
   496→        Filters results by query against ticker and name. Caches the full list.
   497→        """
   498→        query = (params.get('query') or [''])[0].strip()
   499→        if not query:
   500→            return self._send_json({'error': 'پارامتر query الزامی است', 'results': []}, 400)
   501→
   502→        # Normalize query for matching (remove zero-width non-joiners, extra spaces)
   503→        query_normalized = re.sub(r'[\u200c\u200d]', '', query).strip()
   504→
   505→        cache_key = 'symbols_all'
   506→        cached_list = load_file_cache(cache_key)
   507→
   508→        if cached_list is not None:
   509→            print(f'[symbols] Cache hit for stock list ({len(cached_list)} stocks)', flush=True)
   510→            matches = self._filter_symbols(cached_list, query, query_normalized)
   511→            return self._send_json({
   512→                'query': query,
   513→                'count': len(matches),
   514→                'results': matches,
   515→                'source': 'cache',
   516→            })
   517→
   518→        # Fetch full stock list via finpy-tse with 60s timeout
   519→        print(f'[symbols] Fetching stock list via finpy-tse...', flush=True)
   520→        try:
   521→            with ThreadPoolExecutor(max_workers=1) as executor:
   522→                future = executor.submit(_fetch_stock_list)
   523→                try:
   524→                    stock_list = future.result(timeout=60)
   525→                except FuturesTimeoutError:
   526→                    future.cancel()
   527→                    print(f'[symbols] Timeout after 60s for stock list', flush=True)
   528→                    return self._send_json({
   529→                        'error': 'finpy-tse stock list fetch timed out (60s) — likely CDN unreachable',
   530→                        'query': query,
   531→                        'results': [],
   532→                    }, 504)
   533→        except Exception as e:
   534→            err_msg = str(e)
   535→            print(f'[symbols] Error: {err_msg[:200]}', flush=True)
   536→            return self._send_json({
   537→                'error': err_msg,
   538→                'query': query,
   539→                'results': [],
   540→            }, 500)
   541→
   542→        if not stock_list:
   543→            return self._send_json({
   544→                'error': 'فهرست نمادها خالی است',
   545→                'query': query,
   546→                'results': [],
   547→            }, 404)
   548→
   549→        # Cache the full list
   550→        save_file_cache(cache_key, stock_list)
   551→        print(f'[symbols] Cached {len(stock_list)} stocks', flush=True)
   552→
   553→        matches = self._filter_symbols(stock_list, query, query_normalized)
   554→        return self._send_json({
   555→            'query': query,
   556→            'count': len(matches),
   557→            'results': matches,
   558→            'source': 'finpy-tse',
   559→        })

   561→    def _filter_symbols(self, stock_list: list, query: str, query_normalized: str) -> list:
   562→        """Filter stock list by query, matching against ticker and name."""
   563→        matches = []
   564→        for stock in stock_list:
   565→            ticker = stock.get('ticker', '')
   566→            name = stock.get('name', '')
   567→            # Normalize for comparison
   568→            ticker_n = re.sub(r'[\u200c\u200d]', '', ticker).strip()
   569→            name_n = re.sub(r'[\u200c\u200d]', '', name).strip()
   570→
   571→            if (query in ticker or query in name or
   572→                query_normalized in ticker_n or query_normalized in name_n):
   573→                matches.append(stock)
   574→
   575→        # Sort: exact ticker prefix matches first, then name matches
   576→        def sort_key(s):
   577→            t = s.get('ticker', '')
   578→            n = s.get('name', '')
   579→            exact_ticker = 0 if t == query else 1
   580→            starts_ticker = 0 if t.startswith(query) else 1
   581→            starts_name = 0 if n.startswith(query) else 1
   582→            return (exact_ticker, starts_ticker, starts_name, t)
   583→
   584→        matches.sort(key=sort_key)
   585→        return matches[:50]  # Limit to 50 results
