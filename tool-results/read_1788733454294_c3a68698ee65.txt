     3→// ═══════════════════════════════════════════════════════════════════════════════

     5→//   - ML-optimized weights (linear regression on bounce detection)
     6→//   - Overlap scoring between levels
     7→//   - Psychological number rounding
     8→//   - 5-10% distance constraints (relaxed for nearest level)
     9→//   - 6 supports + 6 resistances output

    18→/**
    19→ * ML-optimized feature weights used to compute each S/R level's composite score.
    20→ *
    21→ * Trained via OLS regression on historical bounce detection (see `trainMLWeights`).
    22→ * Each weight represents the relative importance of one feature in predicting
    23→ * whether a price level will act as support/resistance.
    24→ *
    25→ * Weights are always non-negative, clamped to a minimum of 0.05 per feature,
    26→ * and normalized so that `touch + volume + overlap + freshness + distance === 1`.
    27→ *
    28→ * @example Default weights (domain-knowledge fallback):
    29→ * ```
    30→ * { touch: 0.28, volume: 0.22, overlap: 0.25, freshness: 0.15, distance: 0.10 }
    31→ * ```
    32→ */
    33→export interface MLWeights {
    34→  /** Weight for `touchCount` — how many times price has touched this level (within 1%). */
    35→  touch: number;
    36→  /** Weight for `volumeRatio` — average volume at touches vs. overall average volume. */
    37→  volume: number;
    38→  /** Weight for `overlapCount` — number of other levels within 1% (confluence signal). */
    39→  overlap: number;
    40→  /** Weight for `freshness` — recency of last touch (0 = oldest, 1 = most recent bar). */
    41→  freshness: number;
    42→  /** Weight for `distancePercent` — how far the level is from the current price. */
    43→  distance: number;
    44→}
    45→
    46→/**
    47→ * A single support or resistance level with its ML-computed score and metadata.
    48→ *
    49→ * Produced by {@link analyzeSupportResistance} after the full pipeline:
    50→ * discovery → feature extraction → ML scoring → distance filtering → score capping.
    51→ *
    52→ * Levels are classified as **targets** when their score ≥ 7.0.
    53→ */
    54→export interface SRLevel {
    55→  /** Price value of the support/resistance level (psychologically rounded). */
    56→  price: number;
    57→  /** Composite ML score from 0 to 10 (0.5-step granularity). See {@link calculatePowerScore}. */
    58→  score: number;
    59→  /** Human-readable strength grade (Persian): بسیار قوی / قوی / متوسط / ضعیف / بسیار ضعیف. */
    60→  grade: string;
    61→  /** Number of other discovered levels within 1% of this level — indicates confluence. */
    62→  overlapCount: number;
    63→  /** Whether this level is a price target (score ≥ 7.0). */
    64→  isTarget: boolean;
    65→  /** How many bars touched this level (within 1% tolerance). */
    66→  touchCount: number;
    67→  /** Average volume at touches / overall average volume (capped at 5×). Enhanced by volume profile. */
    68→  volumeRatio: number;
    69→  /** Days (bars) since the last touch of this level. 999 for synthetic levels. */
    70→  daysSinceLastTouch: number;
    71→  /** Absolute distance from current price as a percentage: `|price − currentPrice| / currentPrice`. */
    72→  distancePercent: number;
    73→}

    75→/**
    76→ * Complete result of support/resistance analysis.
    77→ *
    78→ * Returned by {@link analyzeSupportResistance}. Contains up to 6 supports and 6 resistances,
    79→ * the trained ML weights, and price targets (levels with score ≥ 7.0).

    81→export interface SRAnalysisResult {

    88→  /** Resistance levels with score ≥ 7.0 — potential upside price targets. */
    89→  upwardTargets: SRLevel[];

   103→  /** Number of independent sources that identified this level (convergence score). */

   118→  /** Number of other discovered levels within 1% of this level. */

   136→// ═══════════════════════════════════════════════════════════════════════════════
   137→// 1. Psychological Number Rounding
   138→// ═══════════════════════════════════════════════════════════════════════════════
   139→
   140→/**
   141→ * Round a price to the nearest "psychological" number — the round numbers that
   142→ * traders and algorithms naturally watch as support/resistance.
   143→ *
   144→ * The rounding step scales with price magnitude:
   145→ * | Price Range        | Rounding Step |
   146→ * |--------------------|---------------|
   147→ * | ≥ 1,000,000        | 50,000        |
   148→ * | ≥ 500,000          | 10,000        |
   149→ * | ≥ 100,000          | 5,000         |
   150→ * | ≥ 50,000           | 2,000         |
   151→ * | ≥ 10,000           | 1,000         |
   152→ * | ≥ 5,000            | 500           |
   153→ * | ≥ 1,000            | 100           |
   154→ * | ≥ 500              | 50            |
   155→ * | ≥ 100              | 10            |
   156→ * | ≥ 10               | 5             |
   157→ * | ≥ 1                | 1             |
   158→ * | < 1                | 0.01          |
   159→ *
   160→ * @param price - The raw price to round.
   161→ * @returns The price rounded to the nearest psychological step.
   162→ *
   163→ * @example
   164→ * ```ts
   165→ * roundToPsychological(15372)   // → 15400  (step=100)
   166→ * roundToPsychological(482.5)   // → 500    (step=50)
   167→ * roundToPsychological(0.0342)  // → 0.03   (step=0.01)
   168→ * ```
   169→ */
   170→export function roundToPsychological(price: number): number {
   171→  const abs = Math.abs(price);
   172→  let step: number;
   173→  if (abs >= 1_000_000) step = 50_000;
   174→  else if (abs >= 500_000) step = 10_000;
   175→  else if (abs >= 100_000) step = 5_000;
   176→  else if (abs >= 50_000) step = 2_000;
   177→  else if (abs >= 10_000) step = 1_000;
   178→  else if (abs >= 5_000) step = 500;
   179→  else if (abs >= 1_000) step = 100;
   180→  else if (abs >= 500) step = 50;
   181→  else if (abs >= 100) step = 10;
   182→  else if (abs >= 10) step = 5;
   183→  else if (abs >= 1) step = 1;
   184→  else step = 0.01;
   185→  return Math.round(price / step) * step;
   186→}

   188→// ═══════════════════════════════════════════════════════════════════════════════
   189→// 2. Level Discovery (multi-source)
   190→// ═══════════════════════════════════════════════════════════════════════════════
   191→
   192→/**
   193→ * Identify swing highs and swing lows in the OHLCV data.
   194→ *
   195→ * A bar is a swing high if its `high` is strictly greater than all neighbouring bars'
   196→ * highs within `lookback` bars on each side. Similarly for swing lows.
   197→ *
   198→ * @param data     - Historical OHLCV bars.
   199→ * @param lookback - Number of bars to check on each side of a candidate.
   200→ * @returns Arrays of swing-high and swing-low prices.
   201→ */
   202→function findSwingLevels(data: OHLCV[], lookback: number): { highs: number[]; lows: number[] } {
   203→  const highs: number[] = [];
   204→  const lows: number[] = [];
   205→  for (let i = lookback; i < data.length - lookback; i++) {
   206→    let isHigh = true;
   207→    let isLow = true;
   208→    for (let j = i - lookback; j <= i + lookback; j++) {
   209→      if (j === i) continue;
   210→      if (data[j].high >= data[i].high) isHigh = false;
   211→      if (data[j].low <= data[i].low) isLow = false;
   212→    }
   213→    if (isHigh) highs.push(data[i].high);
   214→    if (isLow) lows.push(data[i].low);
   215→  }
   216→  return { highs, lows };
   217→}
   218→
   219→/**
   220→ * Simple Moving Average of the last `period` values.
   221→ *
   222→ * @param values - Array of numeric values (typically closing prices).
   223→ * @param period - Window size; only the last `period` values are used.
   224→ * @returns The SMA value, or 0 if there are fewer than `period` values.
   225→ */
   226→function sma(values: number[], period: number): number {
   227→  if (values.length < period) return 0;
   228→  const slice = values.slice(-period);
   229→  return slice.reduce((a, b) => a + b, 0) / period;
   230→}
   231→
   232→/**
   233→ * Exponential Moving Average over the full array.
   234→ *
   235→ * Uses the standard EMA formula: `EMA_i = value_i × k + EMA_{i-1} × (1 − k)`
   236→ * where `k = 2 / (period + 1)`.
   237→ *
   238→ * @param values - Array of numeric values (typically closing prices).
   239→ * @param period - EMA period (smoothing factor).
   240→ * @returns The final EMA value, or 0 if there are fewer than `period` values.
   241→ */
   242→function emaCalc(values: number[], period: number): number {
   243→  if (values.length < period) return 0;
   244→  const k = 2 / (period + 1);
   245→  let result = values[0];
   246→  for (let i = 1; i < values.length; i++) {
   247→    result = values[i] * k + result * (1 - k);
   248→  }
   249→  return result;
   250→}
   251→
   252→/**
   253→ * Calculate Bollinger Bands (20-period, 2σ) from closing prices.
   254→ *
   255→ * @param closes - Array of closing prices.
   256→ * @returns Object with `upper`, `middle` (SMA-20), and `lower` band values.
   257→ *          Returns `{ upper: 0, middle: 0, lower: 0 }` if fewer than 20 closes.
   258→ */
   259→function calcBB(closes: number[]): { upper: number; middle: number; lower: number } {
   260→  const period = 20;
   261→  if (closes.length < period) return { upper: 0, middle: 0, lower: 0 };
   262→  const slice = closes.slice(-period);
   263→  const mean = slice.reduce((a, b) => a + b, 0) / period;
   264→  const std = Math.sqrt(slice.reduce((s, v) => s + (v - mean) ** 2, 0) / period);
   265→  return { upper: mean + 2 * std, middle: mean, lower: mean - 2 * std };
   266→}
   267→
   268→/**
   269→ * Discover all candidate support/resistance levels from 7 independent sources.
   270→ *
   271→ * **Sources:**
   272→ * 1. **Pivot Points** — Classic pivot math (PP, R1-R3, S1-S3) from the prior bar.
   273→ * 2. **Fibonacci Retracement** — 23.6%, 38.2%, 50%, 61.8%, 78.6% of the recent 66-bar range.
   274→ * 3. **Swing Highs/Lows** — Local extrema at lookback periods 3, 5, 7, and 10.
   275→ * 4. **Moving Averages** — SMA(5,10,21,50,100,200) and EMA(12,26), filtered to >1% from price.
   276→ * 5. **Bollinger Bands** — Upper and lower bands (20-period, 2σ).
   277→ * 6. **Recent High/Low Zones** — Rolling high and low over 5, 22, and 66 bars.
   278→ * 7. **Psychological Round Numbers** — Round numbers at 5% intervals from 80%–125% of price.
   279→ *
   280→ * All discovered prices are rounded to psychological numbers (see {@link roundToPsychological}).
   281→ * Nearby levels (within 1%) are merged and their `sourceCount` is incremented (confluence tracking).
   282→ *
   283→ * @param data         - Historical OHLCV bars (at least 10 required for meaningful results).
   284→ * @param currentPrice - The current market price, used for MA/psychological filtering.
   285→ * @returns Array of {@link RawLevel} objects with psychologically-rounded prices and source counts.
   286→ */
   287→function discoverAllLevels(data: OHLCV[], currentPrice: number): RawLevel[] {
   288→  const rawMap = new Map<number, number>(); // level → sourceCount
   289→  const closes = data.map(d => d.close);
   290→
   291→  const addLevel = (price: number) => {
   292→    const rounded = roundToPsychological(price);
   293→    if (rounded <= 0) return;
   294→    const prev = rawMap.get(rounded) ?? 0;
   295→    // Only count as new source if it's not a 1% duplicate of existing
   296→    let isDup = false;
   297→    for (const key of rawMap.keys()) {
   298→      if (Math.abs(key - rounded) / Math.max(key, 1) < 0.01) {
   299→        rawMap.set(key, prev + 1);
   300→        isDup = true;
   301→        break;
   302→      }
   303→    }
   304→    if (!isDup) rawMap.set(rounded, 1);
   305→  };
   306→
   307→  // 1. Pivot Points (5R + 5S)
   308→  const prev = data[data.length - 2] ?? data[data.length - 1];
   309→  if (prev) {
   310→    const pp = (prev.high + prev.low + prev.close) / 3;
   311→    const hl = prev.high - prev.low;
   312→    addLevel(2 * pp - prev.low);
   313→    addLevel(pp + hl);
   314→    addLevel(prev.high + 2 * (pp - prev.low));
   315→    addLevel(prev.low - 2 * (prev.high - pp));
   316→    addLevel(pp - hl);
   317→    addLevel(prev.low - 2 * (prev.high - pp) - hl);
   318→  }
   319→
   320→  // 2. Fibonacci retracement levels from recent range
   321→  const recentHigh = Math.max(...data.slice(-66).map(d => d.high));
   322→  const recentLow = Math.min(...data.slice(-66).map(d => d.low));
   323→  const fibRange = recentHigh - recentLow;
   324→  for (const mult of [0.236, 0.382, 0.5, 0.618, 0.786]) {
   325→    addLevel(recentLow + fibRange * mult);
   326→  }
   327→
   328→  // 3. Swing levels (multiple lookback periods)
   329→  for (const lb of [3, 5, 7, 10]) {
   330→    const swings = findSwingLevels(data, lb);
   331→ swings.highs.forEach(addLevel);
   332→    swings.lows.forEach(addLevel);
   333→  }
   334→
   335→  // 4. Moving Average levels
   336→  for (const period of [5, 10, 21, 50, 100, 200]) {
   337→    const v = sma(closes, period);
   338→    if (v > 0 && Math.abs(v - currentPrice) / currentPrice > 0.01) addLevel(v);
   339→  }
   340→  const ema12 = emaCalc(closes, 12);
   341→  const ema26 = emaCalc(closes, 26);
   342→  if (ema12 > 0 && Math.abs(ema12 - currentPrice) / currentPrice > 0.01) addLevel(ema12);
   343→  if (ema26 > 0 && Math.abs(ema26 - currentPrice) / currentPrice > 0.01) addLevel(ema26);
   344→
   345→  // 5. Bollinger Bands
   346→  const bb = calcBB(closes);
   347→  if (bb.upper > 0) addLevel(bb.upper);
   348→  if (bb.lower > 0) addLevel(bb.lower);
   349→
   350→  // 6. Recent high/low zones
   351→  for (const window of [5, 22, 66]) {
   352→    const slice = data.slice(-Math.min(window, data.length));
   353→    if (slice.length > 0) {
   354→      addLevel(Math.max(...slice.map(d => d.high)));
   355→      addLevel(Math.min(...slice.map(d => d.low)));
   356→    }
   357→  }
   358→
   359→  // 7. Psychological round numbers around current price
   360→  const magnitude = Math.pow(10, Math.floor(Math.log10(Math.max(1, currentPrice))));
   361→  for (let mult = 0.80; mult <= 1.25; mult += 0.05) {
   362→    addLevel(Math.round(currentPrice * mult / magnitude) * magnitude);
   363→  }
   364→
   365→  return Array.from(rawMap.entries()).map(([price, sourceCount]) => ({ price, sourceCount }));
   366→}

   368→// ═══════════════════════════════════════════════════════════════════════════════
   369→// 3. Feature Extraction for Each Level
   370→// ═══════════════════════════════════════════════════════════════════════════════
   371→
   372→/**
   373→ * Extract 5 ML features for a single candidate level.
   374→ *
   375→ * **Features:**
   376→ * | Feature          | Description                                                      | Range    |
   377→ * |------------------|------------------------------------------------------------------|----------|
   378→ * | `touchCount`     | Bars where price came within 1% of the level                     | 0–N      |
   379→ * | `volumeRatio`    | Avg volume at touches ÷ overall avg volume (capped at 5×)        | 0–5      |
   380→ * | `overlapCount`   | Other levels within 1% of this level (confluence)                | 0–N      |
   381→ * | `freshness`      | Recency of last touch: (lastTouchIndex+1) / totalBars             | 0–1      |
   382→ * | `distancePercent`| `|level − currentPrice| / currentPrice`                            | 0–1+     |
   383→ *
   384→ * @param level        - The candidate level price.
   385→ * @param allLevels    - All discovered levels (for overlap computation).
   386→ * @param data         - Historical OHLCV bars.
   387→ * @param currentPrice - Current market price.
   388→ * @returns The 5 extracted features as a {@link LevelFeatures} object.
   389→ */
   390→function extractFeatures(
   391→  level: number,
   392→  allLevels: RawLevel[],
   393→  data: OHLCV[],
   394→  currentPrice: number,
   395→): LevelFeatures {
   396→  const hasVol = data.some(d => d.volume > 0);
   397→  const avgVolume = hasVol
   398→    ? data.slice(-50).reduce((s, d) => s + d.volume, 0) / Math.min(50, data.length)
   399→    : 0;
   400→
   401→  // ── Touch Count: how many times price touched within 1% of this level ──
   402→  const touchThreshold = level * 0.01;
   403→  let touchCount = 0;
   404→  let lastTouchIndex = -1;
   405→  let touchVolumeSum = 0;
   406→
   407→  for (let i = 0; i < data.length; i++) {
   408→    const d = data[i];
   409→    if (Math.abs(d.high - level) <= touchThreshold || Math.abs(d.low - level) <= touchThreshold) {
   410→      touchCount++;
   411→      lastTouchIndex = i;
   412→      if (hasVol && d.volume > 0) touchVolumeSum += d.volume;
   413→    }
   414→  }
   415→
   416→  // ── Volume Ratio: avg volume at touches / overall avg volume ──
   417→  const volumeRatio = (hasVol && touchCount > 0 && avgVolume > 0)
   418→    ? Math.min(5, (touchVolumeSum / touchCount) / avgVolume)
   419→    : 1.0;
   420→
   421→  // ── Overlap Count: how many OTHER levels are within 1% of this level ──
   422→  const overlapCount = allLevels.filter(
   423→    other => other.price !== level && Math.abs(other.price - level) / Math.max(level, 1) < 0.01
   424→  ).length;
   425→
   426→  // ── Freshness: how recent was the last touch (0 = oldest, 1 = most recent) ──
   427→  const freshness = lastTouchIndex >= 0
   428→    ? (lastTouchIndex + 1) / data.length
   429→    : 0.1;
   430→
   431→  // ── Distance from current price ──
   432→  const distancePercent = Math.abs(level - currentPrice) / currentPrice;
   433→
   434→  return { touchCount, volumeRatio, overlapCount, freshness, distancePercent };
   435→}

   437→// ═══════════════════════════════════════════════════════════════════════════════
   438→// 4. Simple Linear Regression (for ML weight optimization)
   439→// ═══════════════════════════════════════════════════════════════════════════════
   440→
   441→/**
   442→ * Train weights using OLS linear regression on bounce detection.
   443→ * Bounce = price touched level and reversed direction within 5 candles.
   444→ * This serves as a proxy for "success_rate" since we don't have future data.
   445→ */
   446→function trainMLWeights(data: OHLCV[], allLevels: RawLevel[]): MLWeights {
   447→  const MIN_SAMPLES = 20;
   448→  const features: number[][] = [];
   449→  const targets: number[] = [];
   450→
   451→  const closes = data.map(d => d.close);
   452→  const currentPrice = closes[closes.length - 1];
   453→
   454→  for (const level of allLevels) {
   455→    // Only consider levels within 30% of current price
   456→    if (Math.abs(level.price - currentPrice) / currentPrice > 0.30) continue;
   457→
   458→    const feats = extractFeatures(level.price, allLevels, data, currentPrice);
   459→
   460→    // Normalize features to 0-1 range for regression stability
   461→    const normalized = [
   462→      Math.min(1, feats.touchCount / 10),           // touch score 0-1
   463→      Math.min(1, feats.volumeRatio / 3),            // volume score 0-1
   464→      Math.min(1, feats.overlapCount / 5),           // overlap score 0-1
   465→      feats.freshness,                                // already 0-1
   466→      Math.min(1, feats.distancePercent / 0.20),     // distance score 0-1
   467→    ];
   468→
   469→    // Detect bounces: price touched level and reversed within 5 candles
   470→    let bounceCount = 0;
   471→    let touchOpportunities = 0;
   472→    for (let i = 1; i < data.length - 5; i++) {
   473→      const d = data[i];
   474→      const threshold = level.price * 0.01;
   475→      const touched = Math.abs(d.high - level.price) <= threshold || Math.abs(d.low - level.price) <= threshold;
   476→      if (!touched) continue;
   477→
   478→      touchOpportunities++;
   479→      const isResistance = d.high >= level.price;
   480→      if (isResistance) {
   481→        // For resistance: check if next 5 candles closed lower
   482→        const futureCloses = closes.slice(i + 1, i + 6);
   483→        if (futureCloses.length >= 3 && futureCloses.every(c => c < d.close)) bounceCount++;
   484→      } else {
   485→        // For support: check if next 5 candles closed higher
   486→        const futureCloses = closes.slice(i + 1, i + 6);
   487→        if (futureCloses.length >= 3 && futureCloses.every(c => c > d.close)) bounceCount++;
   488→      }
   489→    }
   490→
   491→    // Success rate: proportion of touches that resulted in bounces
   492→    const successRate = touchOpportunities > 0 ? bounceCount / touchOpportunities : 0.5;
   493→
   494→    features.push(normalized);
   495→    targets.push(successRate);
   496→  }
   497→
   498→  // Need at least MIN_SAMPLES for meaningful regression
   499→  if (features.length < MIN_SAMPLES) {
   500→    return { ...DEFAULT_WEIGHTS };
   501→  }
   502→
   503→  // OLS: w = (X^T X)^-1 X^T y
   504→  const n = features.length;
   505→  const p = 5; // number of features
   506→
   507→  // X^T X (p x p)
   508→  const XtX: number[][] = Array.from({ length: p }, () => Array(p).fill(0));
   509→  // X^T y (p x 1)
   510→  const Xty: number[] = Array(p).fill(0);
   511→
   512→  for (let i = 0; i < n; i++) {
   513→    for (let j = 0; j < p; j++) {
   514→      for (let k = 0; k < p; k++) {
   515→        XtX[j][k] += features[i][j] * features[i][k];
   516→      }
   517→      Xty[j] += features[i][j] * targets[i];
   518→    }
   519→  }
   520→
   521→  // Add small ridge regularization for numerical stability
   522→  for (let j = 0; j < p; j++) {
   523→    XtX[j][j] += 0.01;
   524→  }
   525→
   526→  // Solve using Gaussian elimination
   527→  const weights = solveLinearSystem(XtX, Xty);
   528→  if (!weights) return { ...DEFAULT_WEIGHTS };
   529→
   530→  // Take absolute values and normalize to sum = 1
   531→  const absWeights = weights.map(Math.abs);
   532→  const sum = absWeights.reduce((a, b) => a + b, 0);
   533→  if (sum === 0) return { ...DEFAULT_WEIGHTS };
   534→
   535→  const normalized = absWeights.map(w => w / sum);
   536→
   537→  // Ensure minimum weight of 0.05 per feature
   538→  const clamped = normalized.map(w => Math.max(0.05, w));
   539→  const clampSum = clamped.reduce((a, b) => a + b, 0);
   540→
   541→  return {
   542→    touch: clamped[0] / clampSum,
   543→    volume: clamped[1] / clampSum,
   544→    overlap: clamped[2] / clampSum,
   545→    freshness: clamped[3] / clampSum,
   546→    distance: clamped[4] / clampSum,
   547→  };
   548→}
   549→
   550→/** Solve Ax = b using Gaussian elimination with partial pivoting */
   551→function solveLinearSystem(A: number[][], b: number[]): number[] | null {
   552→  const n = b.length;
   553→  // Augmented matrix
   554→  const aug = A.map((row, i) => [...row, b[i]]);
   555→
   556→  for (let col = 0; col < n; col++) {
   557→    // Find pivot
   558→    let maxRow = col;
   559→    let maxVal = Math.abs(aug[col][col]);
   560→    for (let row = col + 1; row < n; row++) {
   561→      if (Math.abs(aug[row][col]) > maxVal) {
   562→        maxVal = Math.abs(aug[row][col]);
   563→        maxRow = row;
   564→      }
   565→    }
   566→    if (maxVal < 1e-12) return null; // Singular
   567→
   568→    // Swap rows
   569→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   570→
   571→    // Eliminate below
   572→    for (let row = col + 1; row < n; row++) {
   573→      const factor = aug[row][col] / aug[col][col];
   574→      for (let j = col; j <= n; j++) {
   575→        aug[row][j] -= factor * aug[col][j];
   576→      }
   577→    }
   578→  }
   579→
   580→  // Back substitution
   581→  const x = Array(n).fill(0);
   582→  for (let i = n - 1; i >= 0; i--) {
   583→    x[i] = aug[i][n];
   584→    for (let j = i + 1; j < n; j++) {
   585→      x[i] -= aug[i][j] * x[j];
   586→    }
   587→    x[i] /= aug[i][i];
   588→  }
   589→
   590→  return x;
   591→}

   593→// ═══════════════════════════════════════════════════════════════════════════════
   594→// 5. Score Calculation
   595→// ═══════════════════════════════════════════════════════════════════════════════
   596→
   597→/**
   598→ * Compute the composite 0–10 power score for a level using ML weights.
   599→ *
   600→ * Each feature is first mapped to a 0–10 sub-score:
   601→ * - `touchScore`    = min(10, 1 + touchCount × 0.8)
   602→ * - `volumeScore`   = min(10, max(1, volumeRatio × 3.3))
   603→ * - `overlapScore`  = min(10, overlapCount × 2)
   604→ * - `freshnessScore`= min(10, 1 + freshness × 9)
   605→ * - `distanceScore` = min(10, max(1, 10 − distancePercent × 30))  — closer = higher
   606→ *
   607→ * The raw score is the weighted sum × 10, then clamped to [0.5, 10] with 0.1 rounding.
   608→ *
   609→ * @param feats   - The 5 extracted features for this level.
   610→ * @param weights - The ML-optimized (or default) feature weights.
   611→ * @returns Composite score in [0.5, 10] with one decimal place.
   612→ */
   613→function calculatePowerScore(
   614→  feats: LevelFeatures,
   615→  weights: MLWeights,
   616→): number {
   617→  // Normalize individual scores to 0-10 scale
   618→  const touchScore = Math.min(10, 1 + feats.touchCount * 0.8);
   619→  const volumeScore = Math.min(10, Math.max(1, feats.volumeRatio * 3.3));
   620→  const overlapScore = Math.min(10, feats.overlapCount * 2);
   621→  const freshnessScore = Math.min(10, 1 + feats.freshness * 9);
   622→  // Distance: closer = higher score (but not too close)
   623→  const distanceScore = Math.min(10, Math.max(1, 10 - feats.distancePercent * 30));
   624→
   625→  const raw = (
   626→    weights.touch * touchScore +
   627→    weights.volume * volumeScore +
   628→    weights.overlap * overlapScore +
   629→    weights.freshness * freshnessScore +
   630→    weights.distance * distanceScore
   631→  ) * 10;
   632→
   633→  return Math.min(10, Math.max(0.5, Math.round(raw * 10) / 10));
   634→}
   635→
   636→/**
   637→ * Map a numeric score to a Persian-language strength grade.
   638→ *
   639→ * | Score Range | Grade (Persian) | English Equivalent |
   640→ * |-------------|-----------------|-------------------|
   641→ * | ≥ 8.5       | بسیار قوی       | Very Strong       |
   642→ * | ≥ 7.0       | قوی             | Strong            |
   643→ * | ≥ 5.0       | متوسط           | Moderate          |
   644→ * | ≥ 3.0       | ضعیف            | Weak              |
   645→ * | < 3.0       | بسیار ضعیف      | Very Weak         |
   646→ *
   647→ * @param score - The 0–10 composite score.
   648→ * @returns Persian grade string.
   649→ */
   650→function getGrade(score: number): string {
   651→  if (score >= 8.5) return 'بسیار قوی';
   652→  if (score >= 7.0) return 'قوی';
   653→  if (score >= 5.0) return 'متوسط';
   654→  if (score >= 3.0) return 'ضعیف';
   655→  return 'بسیار ضعیف';
   656→}
   657→
   658→// ═══════════════════════════════════════════════════════════════════════════════
   659→// 6. Distance Filtering (5-10% gap, relaxed for nearest level)
   660→// ═══════════════════════════════════════════════════════════════════════════════
   661→
   662→/**
   663→ * Filter and space levels according to the 5–10% distance-gap rules.
   664→ *
   665→ * **Rules:**
   666→ * - **Nearest level:** can be any distance up to 10% from the current price.
   667→ * - **Consecutive levels:** must be 5–10% apart from the previous selected level.
   668→ *   - If a candidate is < 5% away, it is merged into the previous level
   669→ *     (keeping the higher-score one and boosting its score by 0.5).
   670→ *   - If a candidate is > 10% away, it is skipped entirely.
   671→ * - **Synthetic extension:** If fewer than `targetCount` levels are found, synthetic
   672→ *   levels are generated at 7.5% intervals (midpoint of the 5–10% range)
   673→ *   with score 2.0 and grade "ضعیف".
   674→ *
   675→ * Supports are sorted descending (nearest first); resistances ascending (nearest first).
   676→ *
   677→ * @param levels       - Pre-scored S/R levels (all on one side of the current price).
   678→ * @param currentPrice - The current market price.
   679→ * @param isSupport    - `true` for supports (below price), `false` for resistances (above).
   680→ * @param targetCount  - Desired number of output levels (typically 6).
   681→ * @returns Filtered array of at most `targetCount` levels, properly spaced.
   682→ */
   683→function filterByDistance(
   684→  levels: SRLevel[],
   685→  currentPrice: number,
   686→  isSupport: boolean,
   687→  targetCount: number,
   688→): SRLevel[] {
   689→  if (levels.length === 0) return [];
   690→
   691→  const MIN_GAP = 0.05;
   692→  const MAX_GAP = 0.10;
   693→  const FIRST_MAX_GAP = 0.10; // nearest level: up to 10% from price
   694→  const EXTEND_GAP = 0.075;   // synthetic extension: 7.5% (middle of 5-10%)
   695→
   696→  // Sort: supports descending (nearest first), resistances ascending (nearest first)
   697→  const sorted = isSupport
   698→    ? [...levels].sort((a, b) => b.price - a.price)
   699→    : [...levels].sort((a, b) => a.price - b.price);
   700→
   701→  const result: SRLevel[] = [];
   702→
   703→  for (const level of sorted) {
   704→    if (result.length === 0) {
   705→      // First (nearest) level: must be within 10% of price
   706→      const gap = isSupport
   707→        ? (currentPrice - level.price) / currentPrice  // positive for support
   708→        : (level.price - currentPrice) / currentPrice;  // positive for resistance
   709→
   710→      if (gap <= 0) continue; // level is on wrong side of price
   711→      if (gap > FIRST_MAX_GAP) continue; // too far from price
   712→
   713→      result.push(level);
   714→    } else {
   715→      // Subsequent levels: must be 5-10% from previous level
   716→      const prev = result[result.length - 1].price;
   717→      const gap = Math.abs(level.price - prev) / prev;
   718→
   719→      if (gap < MIN_GAP) {
   720→        // Too close: merge by keeping the higher-score one and boosting its score
   721→        const lastIdx = result.length - 1;
   722→        if (level.score > result[lastIdx].score) {
   723→          result[lastIdx] = {
   724→            ...level,
   725→            score: Math.min(10, level.score + 0.5),
   726→            overlapCount: Math.max(result[lastIdx].overlapCount, level.overlapCount) + 1,
   727→          };
   728→        } else {
   729→          result[lastIdx] = {
   730→            ...result[lastIdx],
   731→            score: Math.min(10, result[lastIdx].score + 0.5),
   732→            overlapCount: result[lastIdx].overlapCount + 1,
   733→          };
   734→        }
   735→      } else if (gap <= MAX_GAP) {
   736→        result.push(level);
   737→      }
   738→      // gap > MAX_GAP: skip this level (too far from previous)
   739→    }
   740→
   741→    if (result.length >= targetCount) break;
   742→  }
   743→
   744→  // If not enough levels, extend with synthetic ones
   745→  while (result.length < targetCount && result.length > 0) {
   746→    const last = result[result.length - 1];
   747→    const nextPrice = isSupport
   748→      ? last.price * (1 - EXTEND_GAP)
   749→      : last.price * (1 + EXTEND_GAP);
   750→
   751→    if (nextPrice <= 0) break;
   752→
   753→    const rounded = roundToPsychological(nextPrice);
   754→    if (rounded === last.price) break; // rounding made no change
   755→
   756→    result.push({
   757→      price: rounded,
   758→      score: 2.0,
   759→      grade: 'ضعیف',
   760→      overlapCount: 0,
   761→      isTarget: false,
   762→      touchCount: 0,
   763→      volumeRatio: 1.0,
   764→      daysSinceLastTouch: 999,
   765→      distancePercent: Math.abs(rounded - currentPrice) / currentPrice,
   766→    });
   767→  }
   768→
   769→  return result.slice(0, targetCount);
   770→}
   771→
   772→// ═══════════════════════════════════════════════════════════════════════════════
   773→// 7. Price Target Determination (score ≥ 7)
   774→// ═══════════════════════════════════════════════════════════════════════════════
   775→
   776→/**
   777→ * Determine price targets from supports and resistances.
   778→ *
   779→ * A level becomes a target when its score ≥ 7.0 ("Strong" or better).
   780→ * - **Upward targets:** resistance levels with score ≥ 7.0 (price targets on the upside).
   781→ * - **Downward targets:** support levels with score ≥ 7.0 (price targets on the downside).
   782→ *
   783→ * @param supports     - Final support levels.
   784→ * @param resistances  - Final resistance levels.
   785→ * @returns Upward and downward target arrays with `isTarget` set to `true`.
   786→ */
   787→function determineTargets(
   788→  supports: SRLevel[],
   789→  resistances: SRLevel[],
   790→): { upwardTargets: SRLevel[]; downwardTargets: SRLevel[] } {
   791→  const upwardTargets = resistances
   792→    .filter(r => r.score >= 7.0)
   793→    .map(r => ({ ...r, isTarget: true }));
   794→
   795→  const downwardTargets = supports
   796→    .filter(s => s.score >= 7.0)
   797→    .map(s => ({ ...s, isTarget: true }));
   798→
   799→  return { upwardTargets, downwardTargets };
   800→}
   801→
   802→// ═══════════════════════════════════════════════════════════════════════════════
   803→// 7.5. Score Capping (max 2 levels can have score 10)
   804→// ═══════════════════════════════════════════════════════════════════════════════
   805→
   806→/**
   807→ * Cap the number of "perfect" scores (≥ 9.5, which rounds to 10) to at most 2 per side.
   808→ *
   809→ * This prevents the output from having many levels with identical maximum scores,
   810→ * which would make it impossible for users to prioritize. The 3rd+ levels that would
   811→ * round to 10 are capped at 9.4 and their grade is recalculated.
   812→ *
   813→ * @param levels - Array of scored S/R levels (mutated in place for efficiency).
   814→ * @returns The same array with excess perfect scores capped.
   815→ */
   816→function capMaxPerfectScores(levels: SRLevel[]): SRLevel[] {
   817→  const MAX_PERFECT = 2;
   818→  const CAP_SCORE = 9.4;
   819→
   820→  // Count how many levels already have score >= 9.5 (which rounds to 10)
   821→  let perfectCount = 0;
   822→  for (const level of levels) {
   823→    if (level.score >= 9.5) perfectCount++;
   824→  }
   825→
   826→  // If already at most 2, nothing to do
   827→  if (perfectCount <= MAX_PERFECT) return levels;
   828→
   829→  // Sort by score descending to find which ones to cap
   830→  const indexed = levels.map((l, i) => ({ level: l, idx: i }));
   831→  indexed.sort((a, b) => b.level.score - a.level.score);
   832→
   833→  let allowed = 0;
   834→  for (const item of indexed) {
   835→    if (item.level.score >= 9.5) {
   836→      if (allowed < MAX_PERFECT) {
   837→        allowed++;
   838→      } else {
   839→        // Cap this level
   840→        item.level.score = CAP_SCORE;
   841→        item.level.grade = getGrade(CAP_SCORE);
   842→      }
   843→    }
   844→  }
   845→
   846→  return levels;
   847→}
   848→
   849→// ═══════════════════════════════════════════════════════════════════════════════
   850→// 8. MAIN ANALYSIS FUNCTION
   851→// ═══════════════════════════════════════════════════════════════════════════════
   852→
   853→/**
   854→ * Perform ML-based support/resistance analysis on historical OHLCV data.
   855→ *
   856→ * This is the main entry point for the S/R analyzer. It combines 7 independent
   857→ * level-discovery sources, ML-optimized scoring, distance-gap filtering, and score
   858→ * capping to produce up to 6 supports and 6 resistances with price targets.
   859→ *
   860→ * ## Algorithm Flow
   861→ *
   862→ * 1. **Discover levels** from 7 sources
   863→ *    (Pivot Points, Fibonacci, Swing H/L, Moving Averages, Bollinger Bands,
   864→ *    Recent H/L Zones, Psychological Round Numbers).
   865→ *    Nearby levels (< 0.5% apart) are merged with confluence tracking.
   866→ *
   867→ * 2. **Compute volume profile** over the last 60 bars (80 bins) for enhanced
   868→ *    volume scoring — no Level-2 data required.
   869→ *
   870→ * 3. **Train ML weights** via OLS regression on historical bounce detection.
   871→ *    Falls back to domain-knowledge defaults if < 20 samples are available.
   872→ *
   873→ * 4. **Extract 5 features** per level and compute scores:
   874→ *    - `touchCount` — bars touching within 1%
   875→ *    - `volumeRatio` — enhanced by volume-profile concentration & value-area boost
   876→ *    - `overlapCount` — confluence with other levels within 1%
   877→ *    - `freshness` — recency of last touch
   878→ *    - `distancePercent` — distance from current price
   879→ *
   880→ * 5. **Distance filter** — enforce 5–10% gap rules (relaxed for the nearest level):
   881→ *    - Nearest level: up to 10% from price
   882→ *    - Subsequent levels: 5–10% from previous level (too-close → merge, too-far → skip)
   883→ *    - Synthetic extension at 7.5% intervals if fewer than 6 levels found
   884→ *
   885→ * 6. **Score cap** — at most 2 levels per side may have score ≥ 9.5 (rounds to 10);
   886→ *    excess are capped at 9.4.
   887→ *
   888→ * 7. **Target selection** — levels with score ≥ 7.0 are marked as price targets
   889→ *    (upward = resistances, downward = supports).
   890→ *
   891→ * @param data         - Historical OHLCV bars. Must contain at least 10 bars.
   892→ * @param currentPrice - The current market price. Must be > 0.
   893→ * @returns {@link SRAnalysisResult} with up to 6 supports, 6 resistances, ML weights, and targets.
   894→ *          Returns empty arrays and default weights if input is invalid.
   895→ *
   896→ * @example
   897→ * ```ts
   898→ * const result = analyzeSupportResistance(ohlcvData, 15372.50);
   899→ * console.log(result.supports);      // up to 6 support levels, nearest→farthest
   900→ * console.log(result.resistances);   // up to 6 resistance levels, nearest→farthest
   901→ * console.log(result.upwardTargets);  // resistances with score ≥ 7
   902→ * console.log(result.mlWeights);      // trained or default weights
   903→ * ```
   904→ */
   905→export function analyzeSupportResistance(
   906→  data: OHLCV[],
   907→  currentPrice: number,
   908→): SRAnalysisResult {
   909→  if (!data || data.length < 10 || currentPrice <= 0) {
   910→    return {
   911→      supports: [],
   912→      resistances: [],
   913→      mlWeights: { ...DEFAULT_WEIGHTS },
   914→      upwardTargets: [],
   915→      downwardTargets: [],
   916→    };
   917→  }
   918→
   919→  // Step 1: Discover all potential levels from multiple sources
   920→  const allLevels = discoverAllLevels(data, currentPrice);
   921→
   922→  // Step 1.5: Compute volume profile for enhanced volume scoring (no Level-2 data needed)
   923→  const volumeProfile = approximateVolumeProfile(data.slice(-60), 80);
   924→
   925→  // Step 2: Train ML weights using bounce detection
   926→  const mlWeights = trainMLWeights(data, allLevels);
   927→
   928→  // Step 3: Extract features and calculate scores for each level
   929→  // Enhanced with volume profile concentration and precise touch count
   930→  const scoredLevels: SRLevel[] = allLevels.map(level => {
   931→    const feats = extractFeatures(level.price, allLevels, data, currentPrice);
   932→
   933→    // Volume profile enhancement: boost volume score using VP concentration
   934→    const vpBin = volumeProfile.bins.find(b => level.price >= b.priceLow && level.price < b.priceHigh);
   935→    const maxVolPct = Math.max(...volumeProfile.bins.map(b => b.volumePercent), 0.001);
   936→    const vpConcentration = vpBin ? vpBin.volumePercent / maxVolPct : 0;
   937→    const inValueArea = level.price >= volumeProfile.valueAreaLow && level.price <= volumeProfile.valueAreaHigh;
   938→
   939→    // Enhanced volume ratio: blend original with VP concentration
   940→    const enhancedVolumeRatio = feats.volumeRatio * 0.6 + (1 + vpConcentration * 2) * 0.4;
   941→    const enhancedFeats = { ...feats, volumeRatio: inValueArea ? enhancedVolumeRatio * 1.2 : enhancedVolumeRatio };
   942→
   943→    // Precise touch count using 0.2% tolerance (from volume-profile.ts)
   944→    const touchResult = countTouch(level.price, data.slice(-50), 0.2);
   945→    const enhancedTouchCount = Math.max(feats.touchCount, touchResult.touchCount);
   946→
   947→    const score = calculatePowerScore({ ...enhancedFeats, touchCount: enhancedTouchCount }, mlWeights);
   948→    return {
   949→      price: level.price,
   950→      score,
   951→      grade: getGrade(score),
   952→      overlapCount: feats.overlapCount,
   953→      isTarget: false,
   954→      touchCount: enhancedTouchCount,
   955→      volumeRatio: enhancedFeats.volumeRatio,
   956→      daysSinceLastTouch: Math.round((1 - feats.freshness) * data.length),
   957→      distancePercent: feats.distancePercent,
   958→    };
   959→  });
   960→
   961→  // Step 4: Separate supports and resistances
   962→  const supports = scoredLevels.filter(l => l.price < currentPrice);
   963→  const resistances = scoredLevels.filter(l => l.price > currentPrice);
   964→
   965→  // Step 5: Filter by distance rules (5-10% gap, relaxed for nearest)
   966→  const finalSupports = filterByDistance(supports, currentPrice, true, 6);
   967→  const finalResistances = filterByDistance(resistances, currentPrice, false, 6);
   968→
   969→  // Step 5.5: Cap max score=10 to at most 2 levels per side
   970→  const cappedSupports = capMaxPerfectScores(finalSupports);
   971→  const cappedResistances = capMaxPerfectScores(finalResistances);
   972→
   973→  // Step 6: Determine price targets
   974→  const { upwardTargets, downwardTargets } = determineTargets(cappedSupports, cappedResistances);
   975→
   976→  // Mark targets in the final arrays
   977→  for (const s of cappedSupports) {
   978→    if (s.score >= 7.0) s.isTarget = true;
   979→  }
   980→  for (const r of cappedResistances) {
   981→    if (r.score >= 7.0) r.isTarget = true;
   982→  }
   983→
   984→  return {
   985→    supports: cappedSupports,
   986→    resistances: cappedResistances,
   987→    mlWeights,
   988→    upwardTargets,
   989→    downwardTargets,
   990→  };
   991→}
   992→
