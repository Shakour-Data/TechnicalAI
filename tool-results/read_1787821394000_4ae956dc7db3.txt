   211→/**
   212→ * Parse a number that might contain Persian digits, commas, or other formatting.
   213→ * Handles null/undefined/empty strings gracefully.
   214→ */
   215→function parsePersianNum(str: string | null | undefined | number): number {
   216→  if (str === null || str === undefined) return 0;
   217→  const s = String(str);
   218→  if (!s || s.trim() === '-' || s.trim() === '') return 0;
   219→  return Number(s.replace(/[,۰-۹]/g, (c: string) => {
   220→    const persianDigits = '۰۱۲۳۴۵۶۷۸۹';
   221→    const idx = persianDigits.indexOf(c);
   222→    return idx >= 0 ? String(idx) : c === ',' ? '' : c;
   223→  }));
   224→}
   225→
   226→/**
   227→ * Check if a value looks like a date string (contains / or - and is not a number).
   228→ */
   229→function looksLikeDate(val: string | null | undefined | number): boolean {
   230→  if (!val) return false;
   231→  const s = String(val).trim();
   232→  return /[\/\-]/.test(s) && isNaN(Number(s.replace(/[,۰-۹\/\-]/g, '')));
   233→}
   234→
   235→/**
   236→ * Check if a value looks like a percentage (ends with % or is small like 0.5 to 50).
   237→ */
   238→function looksLikePercentage(val: string | null | undefined | number): boolean {
   239→  if (!val) return false;
   240→  const s = String(val).trim();
   241→  if (s.endsWith('%')) return true;
   242→  const n = parsePersianNum(val);
   243→  return !isNaN(n) && Math.abs(n) < 100 && n !== 0;
   244→}

   246→/**
   247→ * Auto-detect column mapping for TGJU API response.
   248→ * Expected format: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   249→ * But the API may change column order.
   250→ */
   251→function detectColumnMap(row: (string | number)[]): {
   252→  openIdx: number; highIdx: number; lowIdx: number; closeIdx: number;
   253→  jalaliDateIdx: number; gregorianDateIdx: number;
   254→} | null {
   255→  if (!row || row.length < 6) return null;
   256→
   257→  let jalaliDateIdx = -1;
   258→  let gregorianDateIdx = -1;
   259→  const numericIndices: number[] = [];
   260→  const percentageIndices: number[] = [];
   261→
   262→  for (let i = 0; i < row.length; i++) {
   263→    const val = row[i];
   264→    if (looksLikeDate(val)) {
   265→      // Jalali dates have Persian digits or format like 1404/05/25
   266→      const s = String(val);
   267→      if (/[۰-۹]/.test(s) || /^14\d{2}/.test(s.replace(/[۰-۹]/g, (d: string) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d))))) {
   268→        if (jalaliDateIdx === -1) jalaliDateIdx = i;
   269→      } else {
   270→        if (gregorianDateIdx === -1) gregorianDateIdx = i;
   271→      }
   272→    } else if (looksLikePercentage(val)) {
   273→      percentageIndices.push(i);
   274→    } else {
   275→      const n = parsePersianNum(val);
   276→      if (isFinite(n) && n !== 0) {
   277→        numericIndices.push(i);
   278→      }
   279→    }
   280→  }
   281→
   282→  // If we couldn't auto-detect dates, fall back to assumed positions
   283→  if (jalaliDateIdx === -1 && row.length >= 8) {
   284→    // Default assumption: row[6] = gregorian, row[7] = jalali
   285→    if (looksLikeDate(row[7])) jalaliDateIdx = 7;
   286→    else if (looksLikeDate(row[6])) jalaliDateIdx = 6;
   287→    if (gregorianDateIdx === -1 && looksLikeDate(row[6]) && jalaliDateIdx !== 6) gregorianDateIdx = 6;
   288→    else if (gregorianDateIdx === -1 && looksLikeDate(row[7]) && jalaliDateIdx !== 7) gregorianDateIdx = 7;
   289→  }
   290→
   291→  // Numeric columns (excluding percentages) should be OHLC + change
   292→  // OHLC: typically 4 largest absolute values among non-percentage numerics
   293→  // change: the remaining one
   294→  const ohlcCandidates = numericIndices.filter(i => !percentageIndices.includes(i));
   295→
   296→  if (ohlcCandidates.length < 4) {
   297→    // Fallback: assume first 4 numeric columns are OHLC
   298→    if (numericIndices.length >= 4) {
   299→      return {
   300→        openIdx: numericIndices[0],
   301→        highIdx: numericIndices[2],
   302→        lowIdx: numericIndices[1],
   303→        closeIdx: numericIndices[3],
   304→        jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : 7,
   305→        gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : 6,
   306→      };
   307→    }
   308→    return null;
   309→  }
   310→
   311→  // Sort by absolute value descending to identify OHLC (largest) vs change (smallest)
   312→  const sorted = ohlcCandidates
   313→    .map(i => ({ idx: i, val: Math.abs(parsePersianNum(row[i])) }))
   314→    .sort((a, b) => b.val - a.val);
   315→
   316→  // The 4 largest should be OHLC. Among them, identify high (max) and low (min).
   317→  // open and close are the other two.
   318→  const top4 = sorted.slice(0, 4);
   319→  const maxIdx = top4.reduce((a, b) => a.val > b.val ? a : b).idx;
   320→  const minIdx = top4.reduce((a, b) => a.val < b.val ? a : b).idx;
   321→  const others = top4.filter(x => x.idx !== maxIdx && x.idx !== minIdx);
   322→
   323→  // Among the two remaining, the one closer to high is open, the other is close
   324→  // (heuristic: in an uptrend open < close, but we can't assume direction)
   325→  // Default: first = open, second = close
   326→  const openIdx = others[0]?.idx ?? ohlcCandidates[0];
   327→  const closeIdx = others[1]?.idx ?? ohlcCandidates[3];
   328→
   329→  return {
   330→    openIdx,
   331→    highIdx: maxIdx,
   332→    lowIdx: minIdx,
   333→    closeIdx,
   334→    jalaliDateIdx: jalaliDateIdx >= 0 ? jalaliDateIdx : row.length - 1,
   335→    gregorianDateIdx: gregorianDateIdx >= 0 ? gregorianDateIdx : row.length - 2,
   336→  };
   337→}

   339→/**
   340→ * Validate a candle's OHLC values for sanity.
   341→ * Returns true if the candle is valid.
   342→ */
   343→function isValidCandle(o: number, h: number, l: number, c: number): boolean {
   344→  if (!isFinite(o) || !isFinite(h) || !isFinite(l) || !isFinite(c)) return false;
   345→  if (o <= 0 || h <= 0 || l <= 0 || c <= 0) return false;
   346→  // High must be >= Low, Open, Close
   347→  if (h < l || h < o || h < c) return false;
   348→  // Low must be <= High, Open, Close
   349→  if (l > o || l > c) return false;
   350→  // No infinite or extreme values
   351→  if (o > 1e15 || h > 1e15 || l > 1e15 || c > 1e15) return false;
   352→  return true;
   353→}

   358→const INSTRUMENTS_TTL = 5 * 60 * 1000; // 5 minutes
   359→
   360→const historyCache = new Map<string, { data: TgjuOHLC[]; ts: number }>();
   361→const HISTORY_TTL = 30 * 60 * 1000; // 30 minutes
   362→
   363→// Cache for static instrument prices (from chart API last 3 candles)
   364→const staticPriceCache = new Map<string, { close: number; high: number; low: number; prevClose: number; ts: number }>();
   365→const STATIC_PRICE_TTL = 10 * 60 * 1000; // 10 minutes

   367→/* ─── Fetch live instruments (currency + gold from tgju-api) + static list ── */
   368→
   369→export async function fetchTgjuInstruments(): Promise<TgjuInstrument[]> {
   370→  const now = Date.now();
   371→  if (liveInstrumentsCache.data && now - liveInstrumentsCache.ts < INSTRUMENTS_TTL) {
   372→    return liveInstrumentsCache.data;
   373→  }
   374→
   375→  const controller = new AbortController();
   376→  const timeout = setTimeout(() => controller.abort(), 10000);
   377→
   378→  try {
   379→    const [currencyRes, goldRes] = await Promise.all([
   380→      fetch(`${TGJU_API_BASE}/api/price/currency`, { signal: controller.signal }),
   381→      fetch(`${TGJU_API_BASE}/api/price/gold`, { signal: controller.signal }),
   382→    ]);
   383→
   384→    clearTimeout(timeout);
   385→
   386→    const instruments: TgjuInstrument[] = [];
   387→
   388→    if (currencyRes.ok) {
   389→      const currencies: TgjuPriceItem[] = await currencyRes.json();
   390→      for (const c of currencies) {
   391→        const price = parsePersianNum(c.price);
   392→        const low = parsePersianNum(c.low_price);
   393→        const high = parsePersianNum(c.high_price);
   394→        instruments.push({
   395→          title: c.title,
   396→          key: c.key,
   397→          price,
   398→          highPrice: high,
   399→          lowPrice: low,
   400→          change: 0,
   401→          changePercent: 0,
   402→          category: 'currency',
   403→        });
   404→      }
   405→    }
   406→
   407→    if (goldRes.ok) {
   408→      const goldCategories: TgjuGoldCategory[] = await goldRes.json();
   409→      for (const cat of goldCategories) {
   410→        for (const item of cat.prices) {
   411→          const price = parsePersianNum(item.price);
   412→          const low = parsePersianNum(item.low_price);
   413→          const high = parsePersianNum(item.high_price);
   414→
   415→          let category: TgjuCategoryType = 'gold';
   416→          if (item.key.startsWith('ime_fund_')) category = 'gold_etf';
   417→          else if (item.key.startsWith('silver_')) category = 'silver';
   418→
   419→          instruments.push({
   420→            title: item.title,
   421→            key: item.key,
   422→            price,
   423→            highPrice: high,
   424→            lowPrice: low,
   425→            change: 0,
   426→            changePercent: 0,
   427→            category,
   428→            groupTitle: cat.title,
   429→          });
   430→        }
   431→      }
   432→    }
   433→
   434→    // Add static instruments (crypto, world indices, forex, energy, metals, commodities)
   435→    for (const s of STATIC_INSTRUMENTS) {
   436→      const cached = staticPriceCache.get(s.key);
   437→      const price = cached?.close ?? 0;
   438→      instruments.push({
   439→        title: s.title,
   440→        key: s.key,
   441→        price,
   442→        highPrice: cached?.high ?? 0,
   443→        lowPrice: cached?.low ?? 0,
   444→        change: cached ? (cached.close - cached.prevClose) : 0,
   445→        changePercent: cached && cached.prevClose > 0 ? ((cached.close - cached.prevClose) / cached.prevClose) * 100 : 0,
   446→        category: s.category,
   447→        groupTitle: s.groupTitle,
   448→      });
   449→    }
   450→
   451→    liveInstrumentsCache.data = instruments;
   452→    liveInstrumentsCache.ts = now;
   453→
   454→    // ── Background: fetch chart data for currency/gold to get accurate daily change ──
   455→    backgroundFetchCurrencyGoldChange(instruments).catch(() => {});
   456→
   457→    return instruments;
   458→  } catch {
   459→    clearTimeout(timeout);
   460→    return liveInstrumentsCache.data || [];
   461→  }
   462→}
   463→
   464→/**
   465→ * Background task: fetch last 2 candles for currency/gold items to compute accurate daily change.
   466→ * Uses direct fetch to avoid queue contention.
   467→ * Only updates change/changePercent — keeps the live price from tgju-api.
   468→ */
   469→async function backgroundFetchCurrencyGoldChange(instruments: TgjuInstrument[]) {
   470→  const currencyGoldItems = instruments.filter(
   471→    (i) => i.category === 'currency' || i.category === 'gold' || i.category === 'silver' || i.category === 'gold_etf'
   472→  );
   473→
   474→  // Limit to first 10 to avoid rate-limiting (most important: USD, EUR, gold coin, etc.)
   475→  const priorityKeys = currencyGoldItems.slice(0, 10).map((i) => i.key);
   476→
   477→  for (const key of priorityKeys) {
   478→    try {
   479→      const url = `${TGJU_CHART_API}/${key}?lang=fa&order_dir=desc&start=0&length=2`;
   480→      const res = await fetch(url, { signal: AbortSignal.timeout(10_000) });
   481→      if (!res.ok) continue;
   482→      const jsonStr = await res.text();
   483→      if (!jsonStr.startsWith('{')) continue;
   484→
   485→      const data = JSON.parse(jsonStr);
   486→      const rows: string[][] = data.data;
   487→      if (!rows || rows.length < 2) continue;
   488→
   489→      const newest = rows[0];
   490→      const prev = rows[1];
   491→      const latestClose = parsePersianNum(newest[3]);
   492→      const prevClose = parsePersianNum(prev[3]);
   493→
   494→      if (latestClose > 0 && prevClose > 0) {
   495→        const inst = liveInstrumentsCache.data?.find((i) => i.key === key);
   496→        if (inst) {
   497→          inst.change = latestClose - prevClose;
   498→          inst.changePercent = ((latestClose - prevClose) / prevClose) * 100;
   499→        }
   500→      }
   501→    } catch {
   502→      // skip individual errors
   503→    }
   504→  }
   505→}

   507→/* ─── Fetch historical OHLC data ── */
   508→
   509→export async function fetchTgjuHistory(tgjuKey: string): Promise<TgjuOHLC[]> {
   510→  const cached = historyCache.get(tgjuKey);
   511→  const now = Date.now();
   512→  if (cached && now - cached.ts < HISTORY_TTL) {
   513→    return cached.data;
   514→  }
   515→
   516→  try {
   517→    const url = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=5000`;
   518→
   519→    // Server-side fetch — no CORS issues, no ZAI queue needed
   520→    const controller = new AbortController();
   521→    const timer = setTimeout(() => controller.abort(), 25_000);
   522→    let jsonStr: string;
   523→    try {
   524→      const res = await fetch(url, { signal: controller.signal, headers: { 'Accept': 'application/json' } });
   525→      clearTimeout(timer);
   526→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   527→      jsonStr = await res.text();
   528→    } catch (fetchErr) {
   529→      clearTimeout(timer);
   530→      // This is a server-side route — no CORS. If direct fetch fails, the API is down.
   531→      console.warn(`[TGJU] Direct fetch failed for ${tgjuKey}:`, fetchErr instanceof Error ? fetchErr.message : fetchErr);
   532→      return cached?.data || [];
   533→    }
   534→
   535→    if (!jsonStr.startsWith('{')) {
   536→      console.warn(`[TGJU] Non-JSON response for ${tgjuKey}: ${jsonStr.slice(0, 100)}`);
   537→      return cached?.data || [];
   538→    }
   539→
   540→    const parsed = JSON.parse(jsonStr);
   541→    const rows: (string | number)[][] = parsed.data;
   542→
   543→    if (!rows || rows.length === 0) {
   544→      console.warn(`[TGJU] Empty data array for ${tgjuKey}`);
   545→      return cached?.data || [];
   546→    }
   547→
   548→    // ── Auto-detect column mapping from first valid row ──
   549→    let colMap = detectColumnMap(rows[0]);
   550→
   551→    // If auto-detect failed, try a few more rows
   552→    if (!colMap) {
   553→      for (let r = 1; r < Math.min(5, rows.length); r++) {
   554→        colMap = detectColumnMap(rows[r]);
   555→        if (colMap) break;
   556→      }
   557→    }
   558→
   559→    // If still no auto-detect, fall back to assumed column order
   560→    if (!colMap) {
   561→      console.warn(`[TGJU] Could not auto-detect columns for ${tgjuKey}, using default order`);
   562→      colMap = { openIdx: 0, highIdx: 2, lowIdx: 1, closeIdx: 3, jalaliDateIdx: 7, gregorianDateIdx: 6 };
   563→    } else {
   564→      console.log(`[TGJU] Column map for ${tgjuKey}: O=${colMap.openIdx} H=${colMap.highIdx} L=${colMap.lowIdx} C=${colMap.closeIdx} Date=${colMap.jalaliDateIdx}`);
   565→    }
   566→
   567→    const { openIdx, highIdx, lowIdx, closeIdx, jalaliDateIdx, gregorianDateIdx } = colMap;
   568→
   569→    // ── Parse rows ──
   570→    const candles: TgjuOHLC[] = [];
   571→    let invalidCount = 0;
   572→
   573→    for (const row of rows) {
   574→      const rawDate = String(row[jalaliDateIdx] || row[gregorianDateIdx] || '');
   575→      const date = rawDate.replace(/[\/]/g, '-');
   576→      const o = parsePersianNum(row[openIdx]);
   577→      const h = parsePersianNum(row[highIdx]);
   578→      const l = parsePersianNum(row[lowIdx]);
   579→      const c = parsePersianNum(row[closeIdx]);
   580→
   581→      if (!isValidCandle(o, h, l, c) || date.length === 0) {
   582→        invalidCount++;
   583→        continue;
   584→      }
   585→
   586→      candles.push({ date, open: o, high: h, low: l, close: c });
   587→    }
   588→
   589→    if (invalidCount > 0) {
   590→      console.warn(`[TGJU] ${tgjuKey}: ${invalidCount}/${rows.length} invalid candles skipped`);
   591→    }
   592→
   593→    if (candles.length === 0) {
   594→      return cached?.data || [];
   595→    }
   596→
   597→    // ── Detect and fix data drift ──
   598→    // Check if prices suddenly jump > 50% between consecutive candles
   599→    // (could indicate unit change, decimal shift, or API error)
   600→    let driftCount = 0;
   601→    for (let i = 1; i < candles.length; i++) {
   602→      const prevClose = candles[i - 1].close;
   603→      const currClose = candles[i].close;
   604→      if (prevClose > 0) {
   605→        const pctChange = Math.abs(currClose - prevClose) / prevClose;
   606→        if (pctChange > 0.5) { // > 50% single-day change is suspicious for most instruments
   607→          driftCount++;
   608→        }
   609→      }
   610→    }
   611→    if (driftCount > candles.length * 0.1) { // > 10% of candles have suspicious jumps
   612→      console.warn(`[TGJU] ${tgjuKey}: Possible data drift detected (${driftCount} suspicious jumps out of ${candles.length} candles)`);
   613→    }
   614→
   615→    // Reverse: API returns oldest-first (asc), we want newest-last for charting
   616→    // But check: if the first date is newer than the last, data is already desc
   617→    const firstDate = candles[0].date;
   618→    const lastDate = candles[candles.length - 1].date;
   619→    if (firstDate > lastDate) {
   620→      candles.reverse();
   621→    }
   622→
   623→    console.log(`[TGJU] ${tgjuKey}: Parsed ${candles.length} candles (${candles[0]?.date} to ${candles[candles.length - 1]?.date})`);
   624→
   625→    // Cache the result
   626→    historyCache.set(tgjuKey, { data: candles, ts: now });
   627→
   628→    // Also update static price cache for this instrument
   629→    if (candles.length >= 2) {
   630→      const last = candles[candles.length - 1];
   631→      const prev = candles[candles.length - 2];
   632→      staticPriceCache.set(tgjuKey, {
   633→        close: last.close,
   634→        high: last.high,
   635→        low: last.low,
   636→        prevClose: prev.close,
   637→        ts: now,
   638→      });
   639→
   640→      // Update live instruments cache if this is a static instrument
   641→      if (liveInstrumentsCache.data) {
   642→        const inst = liveInstrumentsCache.data.find((i) => i.key === tgjuKey);
   643→        if (inst && inst.price === 0) {
   644→          inst.price = last.close;
   645→          inst.highPrice = last.high;
   646→          inst.lowPrice = last.low;
   647→          inst.change = last.close - prev.close;
   648→          inst.changePercent = prev.close > 0 ? ((last.close - prev.close) / prev.close) * 100 : 0;
   649→        }
   650→      }
   651→    }
   652→
   653→    return candles;
   654→  } catch (err) {
   655→    console.error(`[TGJU] Failed to fetch history for ${tgjuKey}:`, err);
   656→    return cached?.data || [];
   657→  }
   658→}
