    67→

    69→

    71→  R1: { label: 'تداوم صعود هیجانی', type: 'up', border: '#59e39b', badgeBg: 'rgba(89,227,155,0.2)', badgeColor: '#59e39b' },
    72→  R2: { label: 'پولبک سالم', type: 'pullback', border: '#ffb25f', badgeBg: 'rgba(255,178,95,0.2)', badgeColor: '#ffb25f' },
    73→  R3: { label: 'اصلاح کنترل‌شده', type: 'down', border: '#ff758a', badgeBg: 'rgba(255,117,138,0.2)', badgeColor: '#ff758a' },
    74→  R4: { label: 'اصلاح عمیق', type: 'down', border: '#ffb11b', badgeBg: 'rgba(255,177,27,0.2)', badgeColor: '#ffb11b' },
    75→  R5: { label: 'تضعیف ساختار', type: 'down', border: '#ef4d62', badgeBg: 'rgba(239,77,98,0.2)', badgeColor: '#ef4d62' },
    76→};
    77→
    78→const STRATEGY_MAP: Record<string, { text: string; tagCls: string }> = {
    79→  R1: { text: 'صعودی قوی — احتمال بالای عبور از مقاومت‌ها', tagCls: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30' },
    80→  R2: { text: 'صعود تدریجی — ورود در اصلاح توصیه می‌شود', tagCls: 'bg-cyan-500/15 text-cyan-400 border-cyan-500/30' },
    81→  R3: { text: 'بازار رنج — منتظر خروج از محدوده بمانید', tagCls: 'bg-amber-500/15 text-amber-400 border-amber-500/30' },
    82→  R4: { text: 'اصلاحی — احتیاط و کاهش حجم معاملات', tagCls: 'bg-orange-500/15 text-orange-400 border-orange-500/30' },
    83→  R5: { text: 'نزولی قوی — خروج فوری توصیه می‌شود', tagCls: 'bg-red-500/15 text-red-400 border-red-500/30' },
    84→};

    87→// Component
    88→// ═══════════════════════════════════════════════════════════════════════════════
    89→
    90→export default function VdesAnalysis(props: VdesAnalysisProps) {
    91→  const {
    92→    symbolName, candles, currentPrice, resistances, supports, ma21, ma100,
    93→    rsi, mfi, cci, adx, stochK, stochD, macdLine, macdSignal, macdHist,
    94→    diPlus, diMinus, sar, atr, obv, hasVolume, bollingerUpper, bollingerMiddle, bollingerLower,
    95→    trendDirection, trendAngle, trendR2, overallSignal, scenarios,
    96→    supportStrengths, resistanceStrengths, priceTargets,
    97→  } = props;
    98→
    99→  const R1_level = resistances[0] ?? currentPrice * 1.05;
   100→  const R2_level = resistances[1] ?? currentPrice * 1.10;
   101→  const S1_level = supports[0] ?? currentPrice * 0.95;
   102→  const S2_level = supports[1] ?? currentPrice * 0.90;
   103→  const S3_level = supports[2] ?? currentPrice * 0.85;
   104→
   105→  const targetMin = Math.round(R1_level + (R2_level - R1_level) * 0.5);
   106→  const targetMax = Math.round(R2_level + (R2_level - R1_level) * 0.8);
   107→
   108→  // ── Dominant scenario ──────────────────────────────────────────
   109→  let highestKey = 'R3';
   110→  let highestProb = 0;
   111→  for (const key of SCENARIO_KEYS) {
   112→    if (scenarios[key].probability > highestProb) {
   113→      highestProb = scenarios[key].probability;
   114→      highestKey = key;
   115→    }
   116→  }
   117→  const strategy = STRATEGY_MAP[highestKey];
   118→  const totalProb = SCENARIO_KEYS.reduce((sum, k) => sum + scenarios[k].probability, 0);
   119→
   120→  // ── RSI signal ─────────────────────────────────────────────────
   121→  const rsiSignal = rsi > 70 ? 'اشباع خرید شدید' : rsi > 60 ? 'اشباع خرید' : rsi > 40 ? 'خنثی' : rsi > 30 ? 'اشباع فروش' : 'اشباع فروش شدید';
   122→  const rsiColor = rsi > 70 ? 'text-red-400' : rsi > 60 ? 'text-amber-400' : rsi > 40 ? 'text-gray-300' : rsi > 30 ? 'text-amber-400' : 'text-emerald-400';
   123→
   124→  // ── Stochastic signal ──────────────────────────────────────────
   125→  const stochSignal = stochK > 80 ? 'اشباع خرید' : stochK < 20 ? 'اشباع فروش' : stochK > stochD ? 'صعودی' : 'نزولی';
   126→
   127→  // ── MACD signal ────────────────────────────────────────────────
   128→  const macdSignalText = macdHist > 0 ? 'مومنتوم مثبت (هیستوگرام بالای صفر)' : 'مومنتوم منفی (هیستوگرام زیر صفر)';
   129→  const macdBullish = macdLine > macdSignal;
   130→
   131→  // ── Trend text ─────────────────────────────────────────────────
   132→  const trendText = trendDirection === 'up'
   133→    ? `صعودی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${(trendR2 * 100).toFixed(1)}%`
   134→    : trendDirection === 'down'
   135→    ? `نزولی با زاویه ${toFa(Math.abs(trendAngle))} درجه و ضریب تعیین R²=${(trendR2 * 100).toFixed(1)}%`
   136→    : 'خنثی و بدون جهت مشخص';
   137→
   138→  // ── DI signal ──────────────────────────────────────────────────
   139→  const diSignal = diPlus > diMinus
   140→    ? `DI+ (${toFa(diPlus)}) بالاتر از DI- (${toFa(diMinus)}) — فشار خرید غالب`
   141→    : `DI- (${toFa(diMinus)}) بالاتر از DI+ (${toFa(diPlus)}) — فشار فروش غالب`;
   142→
   143→  // ── SAR signal ─────────────────────────────────────────────────
   144→  const sarSignal = sar < currentPrice
   145→    ? `SAR (${toFa(sar)}) زیر قیمت — تأیید روند صعودی`
   146→    : `SAR (${toFa(sar)}) بالای قیمت — تأیید روند نزولی`;
   147→
   148→  // ── Bollinger Band position ────────────────────────────────────
   149→  const bbRange = bollingerUpper - bollingerLower;
   150→  const bbPos = bbRange > 0 ? ((currentPrice - bollingerLower) / bbRange * 100).toFixed(0) : '50';
   151→  const bbSignal = currentPrice > bollingerUpper ? 'بالای باند بالایی (اشباع خرید)'
   152→    : currentPrice < bollingerLower ? 'زیر باند پایینی (اشباع فروش)'
   153→    : `داخل باندها (${bbPos}٪ از بازه)`;
   154→
   155→  // ── TradingView scenario colors ────────────────────────────────
   156→  // Limit candles for chart rendering to prevent freeze on large datasets (e.g. derivatives with 10k+ candles)
   157→  const chartCandles = useMemo(() => {
   158→    if (candles.length <= 2000) return candles;
   159→    return candles.slice(-2000);
   160→  }, [candles]);
   161→
   162→  const tvScenarios = useMemo(() => {
   163→    const colors: Record<string, string> = {
   164→      R1: '#34c98b', R2: '#3ad5db', R3: '#ff7b32', R4: '#ffb11b', R5: '#ef4d62',
   165→    };
   166→    return Object.fromEntries(
   167→      SCENARIO_KEYS.map(k => [k, {
   168→        targetMin: scenarios[k].targetMin,
   169→        targetMax: scenarios[k].targetMax,
   170→        name: scenarios[k].name,
   171→        probability: scenarios[k].probability,
   172→        color: colors[k],
   173→      }])
   174→    );
   175→  }, [scenarios]);
   176→
   177→  // ── Strategy recommendation text ────────────────────────────────
   178→  const mfiWarning = hasVolume ? ` (MFI: ${toFa(mfi)})` : '';
   179→  const strategyText = rsi > 70 || (hasVolume && mfi > 80)
   180→    ? `با توجه به هشدار اشباع خرید (RSI: ${toFa(rsi)}${mfiWarning}) و فاصله قیمت تا مقاومت ${toFa(R1_level)}، استراتژی محتاطانه، انتظار برای اصلاح قیمت و ورود در محدوده حمایت ${toFa(S1_level)} تا ${toFa(S2_level)} ریال می‌باشد. در این محدوده می‌توان با حد ضرر ${toFa(S2_level)} ریال وارد موقعیت خرید شد.`
   181→    : rsi < 30 || (hasVolume && mfi < 20)
   182→    ? `با توجه به اشباع فروش (RSI: ${toFa(rsi)}${mfiWarning}) و نزدیکی به حمایت ${toFa(S1_level)}، فرصت خرید در محدوده فعلی با حد ضرر ${toFa(S2_level)} ریال قابل بررسی است. هدف اولیه ${toFa(R1_level)} و هدف ثانویه ${toFa(R2_level)} ریال تعیین می‌شود.`
   183→    : `با توجه به وضعیت خنثی اندیکاتورها (RSI: ${toFa(rsi)}, ADX: ${toFa(adx)}، قدرت روند: ${adx > 25 ? 'قوی' : 'ضعیف'})، انتظار برای خروج قیمت از محدوده ${toFa(S1_level)} تا ${toFa(R1_level)} ریال و سپس تصمیم‌گیری توصیه می‌شود.${hasVolume ? ' حجم معاملات و مومنتوم MACD را برای تأیید سیگنال پایش کنید.' : ' مومنتوم MACD و حرکات قیمتی را برای تأیید سیگنال پایش کنید.'}`;
   184→
   185→  return (
   186→    <div className="space-y-5" dir="rtl">
   187→      {/* ═══ HEADER ═══ */}
   188→      <div className="rounded-[32px] px-6 py-5 border border-amber-500/20"
   189→        style={{
   190→          background: 'rgba(18,28,46,0.7)',
   191→          backdropFilter: 'blur(14px)',
   192→          boxShadow: '0 25px 50px -12px rgba(0,0,0,0.6)',
   193→        }}>
   194→        <h1 className="text-xl font-bold mb-1" style={{ color: '#f8e365' }}>
   195→          📈 تحلیل تکنیکال {symbolName}
   196→        </h1>
   197→        <div className="flex flex-wrap gap-3 mt-2">
   198→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   199→            📍 قیمت مرجع: <b className="text-gray-200">{toFa(currentPrice)} ریال</b>
   200→          </span>
   201→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   202→            🎯 هدف کوتاه‌مدت: <b className="text-gray-200">{toFa(targetMin)} — {toFa(targetMax)} ریال</b>
   203→          </span>
   204→          <span className="px-3.5 py-1 rounded-full border border-white/5 bg-white/4 text-xs text-gray-400">
   205→            📊 روند: <b className="text-gray-200">{trendText}</b>
   206→          </span>
   207→          <span className={`px-3.5 py-1 rounded-full border text-xs font-medium ${
   208→            rsi > 70 ? 'bg-red-500/15 text-red-400 border-red-500/30'
   209→            : rsi < 30 ? 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30'
   210→            : 'bg-amber-500/15 text-amber-400 border-amber-500/30'
   211→          }`}>
   212→            RSI: {toFa(rsi)} ({rsiSignal})
   213→          </span>
   214→        </div>
   215→      </div>
   216→
   217→      {/* ═══ TRADINGVIEW CHART ═══ */}
   218→      <div className="rounded-[28px] p-4 border border-white/6"
   219→        style={{
   220→          background: 'rgba(14,22,40,0.7)',
   221→          backdropFilter: 'blur(8px)',
   222→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   223→        }}>
   224→        <h2 className="text-sm font-semibold mb-3 text-gray-200 flex items-center gap-3">
   225→          <span style={{ color: '#f8e365' }}>📊</span>
   226→          نمودار قیمتی سهم (۱ روزه) — به همراه MA21، MA100، سطوح حمایت/مقاومت و اهداف قیمتی
   227→        </h2>
   228→        <TradingViewChart
   229→          symbolName={symbolName}
   230→          candles={chartCandles}
   231→          supports={supports}
   232→          resistances={resistances}
   233→          supportStrengths={supportStrengths}
   234→          resistanceStrengths={resistanceStrengths}
   235→          ma21={ma21}
   236→          ma100={ma100}
   237→          scenarios={tvScenarios}
   238→        />
   239→        <div className="mt-3 flex flex-wrap gap-4 text-[10px] text-gray-500">
   240→          <span>🔴 خطوط قرمز: مقاومت‌ها ({resistances.map(toFa).join(' ، ')})</span>
   241→          <span>🟢 خطوط سبز: حمایت‌ها ({supports.map(toFa).join(' ، ')})</span>
   242→          <span>🟣 خط بنفش: MA100 ({toFa(ma100)})</span>
   243→          <span>🔵 خط آبی: MA21 ({toFa(ma21)})</span>
   244→          <span>🟡 خطوط زرد: اهداف قیمتی سناریوها (رند شده)</span>
   245→        </div>
   246→      </div>
   247→
   248→      {/* ═══ DUAL ANALYSIS LAYOUT ═══ */}
   249→      <div className="rounded-[28px] p-5 border border-white/6"
   250→        style={{
   251→          background: 'rgba(14,22,40,0.7)',
   252→          backdropFilter: 'blur(8px)',
   253→          boxShadow: '0 20px 40px -12px rgba(0,0,0,0.5)',
   254→        }}>
   255→        <h2 className="text-sm font-semibold mb-4 text-gray-200 flex items-center gap-3">
   256→          <span style={{ color: '#f8e365' }}>🧠</span>
   257→          تحلیل جامع روند و سطوح کلیدی
   258→        </h2>
   259→
   260→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-5">
   261→          {/* ── Text Analysis (Right in RTL) ── */}
   262→          <div className="rounded-2xl p-4 border border-white/4 space-y-4 flex flex-col"
   263→            style={{ background: 'rgba(8,20,35,0.5)' }}>
   264→
   265→            {/* ── PARAGRAPH 1: General Trend & Price Position ── */}
   266→            <p className="text-sm text-gray-300 leading-[1.85]">
   267→              <strong style={{ color: '#f8e365' }}>روند کلی و موقعیت قیمت:</strong>{' '}
   268→              سهم {symbolName} در حال حاضر در محدوده{' '}
   269→              <b className="text-gray-100">{toFa(currentPrice)} ریال</b> معامله می‌شود و روند میان‌مدت{' '}
   270→              <b className={trendDirection === 'up' ? 'text-emerald-400' : trendDirection === 'down' ? 'text-red-400' : 'text-amber-400'}>{trendDirection === 'up' ? 'صعودی' : trendDirection === 'down' ? 'نزولی' : 'خنثی'}</b>
   271→              {' '}است (زاویه {toFa(Math.abs(trendAngle))}°، R²={+(trendR2 * 100).toFixed(1)}%).
   272→              قیمت نسبت به MA21 ({toFa(ma21)} ریال){' '}
   273→              {currentPrice > ma21 ? <span className="text-emerald-400">بالاتر</span> : <span className="text-red-400">پایین‌تر</span>}
   274→              {' '}و نسبت به MA100 ({toFa(ma100)} ریال){' '}
   275→              {currentPrice > ma100 ? <span className="text-emerald-400">بالاتر</span> : <span className="text-red-400">پایین‌تر</span>}
   276→              {' '}قرار دارد. اندیکاتور Parabolic SAR ({toFa(sar)}) نیز{' '}
   277→              {sar < currentPrice ? <span>زیر قیمت قرار دارد که <b className="text-emerald-400">تأیید روند صعودی</b> است.</span> : <span>بالای قیمت قرار دارد که <b className="text-red-400">تأیید روند نزولی</b> است.</span>}
   278→              {' '}شاخص ADX ({toFa(adx)}) نشان‌دهنده {' '}
   279→              {adx > 40 ? <b className="text-emerald-400">روند قدرتمند</b> : adx > 25 ? <b className="text-amber-400">روند متوسط</b> : <b className="text-gray-500">روند ضعیف یا رنج</b>}
   280→              {' '}می‌باشد. {' '}{diSignal}.
   281→            </p>
   282→
   283→            {/* ── PARAGRAPH 2: Oscillator & Momentum Analysis ── */}
   284→            <p className="text-sm text-gray-300 leading-[1.85]">
   285→              <strong style={{ color: '#f8e365' }}>تحلیل اسیلاتورها و مومنتوم:</strong>{' '}
   286→              اندیکاتور RSI ({toFa(rsi)}) در ناحیه{' '}
   287→              <b className={rsiColor}>{rsiSignal}</b> قرار دارد.
   288→              {hasVolume && <>
   289→              اندیکاتور MFI ({toFa(mfi)}) نیز{' '}
   290→              {mfi > 80 ? <span className="text-red-400">اشباع خرید را تأیید می‌کند</span> : mfi < 20 ? <span className="text-emerald-400">اشباع فروش را نشان می‌دهد</span> : <span>در محدوده عادی است</span>}
   291→              {' '}</>}
   292→              اندیکاتور CCI ({toFa(cci)}){' '}
   293→              {cci > 100 ? <span className="text-red-400">بالاتر از +100 (قدرت خریداری قوی)</span> : cci < -100 ? <span className="text-emerald-400">پایین‌تر از -100 (قدرت فروشندگان)</span> : <span>در محدوده عادی (-100 تا +100)</span>}
   294→              . استوکاستیک (%K={toFa(stochK)}، %D={toFa(stochD)}) وضعیت{' '}
   295→              <b className={stochK > 80 ? 'text-red-400' : stochK < 20 ? 'text-emerald-400' : 'text-gray-300'}>{stochSignal}</b> را نشان می‌دهد.
   296→              {' '}MACD (خط={toFa(macdLine)}، سیگنال={toFa(macdSignal)}) با {macdBullish ? <span className="text-emerald-400">عبور خط اصلی بالای خط سیگنال — سیگنال صعودی</span> : <span className="text-red-400">خط اصلی زیر خط سیگنال — سیگنال نزولی</span>}
   297→              . هیستوگرام MACD ({toFa(macdHist)}) {' '}
   298→              {macdHist > 0 ? <span className="text-emerald-400">مثبت</span> : <span className="text-red-400">منفی</span>}
   299→              {' '}و {' '}{macdSignalText}.
   300→            </p>
   301→
   302→            {/* ── PARAGRAPH 3: Bollinger Bands & Volatility ── */}
   303→            <p className="text-sm text-gray-300 leading-[1.85]">
   304→              <strong style={{ color: '#f8e365' }}>تحلیل نوسانات و باند بولینگر:</strong>{' '}
   305→              قیمت در باند بولینگر{' '}
   306→              <b className="text-gray-200">{bbSignal}</b>
   307→              {' '}قرار دارد. باند بالایی: {toFa(bollingerUpper)}، باند میانی (MA20): {toFa(bollingerMiddle)}، باند پایینی: {toFa(bollingerLower)} ریال.
   308→              {currentPrice > bollingerUpper

   495→      قدرت: {strength}/10
   496→    </span>
   497→  );
