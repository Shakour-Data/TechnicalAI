   100→  function computeVisibility(filter) {
   101→    var edges = DATA.decisionGraph.edges;
   102→    var nodes = DATA.decisionGraph.nodes;
   103→    if (filter === 'all') {
   104→      var allE = []; for (var i=0;i<edges.length;i++) allE.push(i);
   105→      var allN = {}; for (var j=0;j<nodes.length;j++) allN[nodes[j].id]=true;

   109→      var eSet = {}, nSet = { ROOT: true };
   110→      nSet[filter] = true;
   111→      for (var i=0;i<edges.length;i++) {
   112→        if (edges[i].to === filter) { eSet[i]=true; nSet[edges[i].from]=true; }
   113→      }
   114→      for (var i=0;i<edges.length;i++) {
   115→        if (nSet[edges[i].to] && edges[i].to.indexOf('R')!==0) { eSet[i]=true; nSet[edges[i].from]=true; }
   116→      }
   117→      var arr=[]; for (var k in eSet) arr.push(parseInt(k));
   118→      return { visEdges: arr, visNodes: nSet };

   120→    if (filter==='trend'||filter==='breakout'||filter==='reversal') {
   121→      var bt = 'branch-'+filter;

   160→  // ── Draw edges ──
   161→  function drawEdges() {
   162→    var w = container.clientWidth, h = container.clientHeight;
   163→    ctx.clearRect(0, 0, w, h);
   164→    var sx = getScale();
   165→    var edges = DATA.decisionGraph.edges;
   166→    var pos = DATA.decisionGraph.nodePositions;
   167→    var probs = DATA.decisionGraph.edgeProbabilities;
   168→    var vis = computeVisibility(state.filter);
   169→    var visSet = {}; for (var i=0;i<vis.visEdges.length;i++) visSet[vis.visEdges[i]]=true;
   170→
   171→    for (var ei=0; ei<edges.length; ei++) {
   172→      var e = edges[ei];
   173→      var fp = pos[e.from], tp = pos[e.to];
   174→      if (!fp || !tp) continue;
   175→      var fn = getNode(e.from), tn = getNode(e.to);
   176→      if (!fn || !tn) continue;
   177→      var fd = nodeDims(fn), td = nodeDims(tn);
   178→      var ax = (fp.right + fd.w/2)*sx, ay = (fp.top + fd.h/2)*sx;
   179→      var bx = (tp.right + td.w/2)*sx, by = (tp.top + td.h/2)*sx;
   180→      var dx = bx-ax, dy = by-ay, dist = Math.sqrt(dx*dx+dy*dy);
   181→      if (dist === 0) continue;
   182→      var bend = Math.min(52, Math.max(18, dist*0.11));
   183→      var mx = (ax+bx)/2, my = (ay+by)/2;
   184→      var nx = -dy/dist, ny = dx/dist;
   185→      var cx = mx + nx*bend, cy = my + ny*bend;
   186→      var isVis = visSet[ei];
   187→      var ec = EDGE_COLORS[e.type] || '#6b7280';
   188→
   189→      // Draw curve
   190→      ctx.save();
   191→      ctx.globalAlpha = isVis ? 0.72 : 0.06;
   192→      ctx.strokeStyle = ec;
   193→      ctx.lineWidth = isVis ? 2 : 0.8;
   194→      ctx.beginPath();
   195→      ctx.moveTo(ax, ay);
   196→      ctx.quadraticCurveTo(cx, cy, bx, by);
   197→      ctx.stroke();
   198→
   199→      // Arrow
   200→      if (isVis) {
   201→        var t = 0.92;
   202→        var arx = (1-t)*(1-t)*ax + 2*(1-t)*t*cx + t*t*bx;
   203→        var ary = (1-t)*(1-t)*ay + 2*(1-t)*t*cy + t*t*by;
   204→        var atx = 2*(1-t)*(cx-ax) + 2*t*(bx-cx);
   205→        var aty = 2*(1-t)*(cy-ay) + 2*t*(by-cy);
   206→        var ang = Math.atan2(aty, atx);
   207→        var al = 8;
   208→        ctx.fillStyle = ec;
   209→        ctx.beginPath();
   210→        ctx.moveTo(bx, by);
   211→        ctx.lineTo(bx - al*Math.cos(ang-0.4), by - al*Math.sin(ang-0.4));
   212→        ctx.lineTo(bx - al*Math.cos(ang+0.4), by - al*Math.sin(ang+0.4));
   213→        ctx.closePath();
   214→        ctx.fill();
   215→      }
   216→      ctx.restore();
   217→
   218→      // Labels on visible edges (not branch edges)
   219→      if (isVis && ei >= 6) {
   220→        var prob = probs[ei] || 0;
   221→        var probLabel = toFa(prob*100) + '٪';
   222→        ctx.save();
   223→        ctx.font = '10px Vazirmatn, Tahoma, sans-serif';
   224→        ctx.textAlign = 'center';
   225→        ctx.fillStyle = '#ffffff';
   226→        ctx.globalAlpha = 0.95;
   227→        ctx.strokeStyle = '#07111b';
   228→        ctx.lineWidth = 4; ctx.lineJoin = 'round';
   229→        ctx.strokeText(e.label, cx, cy-5);
   230→        ctx.fillText(e.label, cx, cy-5);
   231→        ctx.font = 'bold 9px Vazirmatn, Tahoma, sans-serif';
   232→        ctx.fillStyle = ec;
   233→        ctx.globalAlpha = 0.9;
   234→        ctx.strokeText(probLabel, cx, cy+8);
   235→        ctx.fillText(probLabel, cx, cy+8);
   236→        ctx.restore();
   237→      }
   238→    }
   239→  }
