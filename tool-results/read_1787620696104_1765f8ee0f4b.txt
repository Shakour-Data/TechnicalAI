    52→

    54→  up: '#34c98b',
    55→  pullback: '#4186ff',
    56→  down: '#ff7b32',
    57→  risk: '#ef4d62',
    58→  cyan: '#3ad5db',
    59→  purple: '#a04ac5',
    60→  gold: '#ffb11b',
    61→};
    62→
    63→const EDGE_COLORS: Record<string, string> = {
    64→  'branch-trend': COLORS.cyan,
    65→  'branch-breakout': COLORS.gold,
    66→  'branch-reversal': COLORS.purple,
    67→  up: COLORS.up,
    68→  pullback: COLORS.pullback,
    69→  down: COLORS.down,
    70→  risk: COLORS.risk,
    71→};
    72→
    73→const SCENARIO_KEYS = ['R1', 'R2', 'R3', 'R4', 'R5', 'R6', 'R7', 'R8', 'R9'] as const;
    74→
    75→const SCENARIO_META: Record<string, { label: string; color: string }> = {
    76→  R1: { label: 'صعودی با احتیاط', color: '#047857' },
    77→  R2: { label: 'صعودی قوی', color: '#059669' },
    78→  R3: { label: 'صعودی شتاب‌دار', color: '#0e7490' },
    79→  R4: { label: 'شوک صعودی', color: '#0891b2' },
    80→  R5: { label: 'رنج کم‌نوسان', color: '#b45309' },
    81→  R6: { label: 'نزولی با احتیاط', color: '#c2410c' },
    82→  R7: { label: 'نزولی قوی', color: '#ea580c' },
    83→  R8: { label: 'نزولی شتاب‌دار', color: '#dc2626' },
    84→  R9: { label: 'شوک نزولی', color: '#b91c1c' },
    85→};
    86→
    87→const SCENARIO_DISPLAY: Record<string, string> = {
    88→  ROOT: 'ریشه',
    89→  BR1: 'پیروی از روند', BR2: 'شکست', BR3: 'بازگشت',
    90→  EA: 'وضعیت روند', EB: 'وضعیت شکست', EC: 'وضعیت واگرایی',
    91→  R1: 'سناریوی ۱', R2: 'سناریوی ۲', R3: 'سناریوی ۳',
    92→  R4: 'سناریوی ۴', R5: 'سناریوی ۵', R6: 'سناریوی ۶',
    93→  R7: 'سناریوی ۷', R8: 'سناریوی ۸', R9: 'سناریوی ۹',
    94→};
    95→
    96→const BRANCH_META: Record<string, { label: string; color: string }> = {
    97→  trend: { label: 'پیروی از روند', color: COLORS.cyan },
    98→  breakout: { label: 'شکست', color: COLORS.gold },
    99→  reversal: { label: 'بازگشت', color: COLORS.purple },
   100→};
   101→
   102→const TYPE_FILTERS = [
   103→  { key: 'all', label: 'همه مسیرها' },
   104→  { key: 'up', label: 'صعودی' },
   105→  { key: 'pullback', label: 'رنج و خنثی' },
   106→  { key: 'down', label: 'نزولی' },
   107→  { key: 'risk', label: 'شوک و ریسک' },
   108→];

   110→// ═══════════════════════════════════════════════════════════════════════════════
   111→// Component
   112→// ═══════════════════════════════════════════════════════════════════════════════
   113→
   114→export default function VdssGraph(props: VdssGraphProps) {
   115→  const {
   116→    symbolName,
   117→    currentPrice,
   118→    scenarios,
   119→    decisionGraph,
   120→  } = props;
   121→
   122→  const [activeFilter, setActiveFilter] = useState('all');
   123→  const [selectedNode, setSelectedNode] = useState<string | null>(null);
   124→  const graphRef = useRef<HTMLDivElement>(null);
   125→  const svgRef = useRef<SVGSVGElement>(null);
   126→  const nodeRefs = useRef<Record<string, HTMLDivElement | null>>({});
   127→  const shellRef = useRef<HTMLDivElement>(null);
   128→
   129→  // ── Derived data from decisionGraph ──────────────────────────────
   130→  const edges = decisionGraph?.edges ?? [];
   131→  const nodes = decisionGraph?.nodes ?? [];
   132→  const nodePositions = decisionGraph?.nodePositions ?? {};
   133→  const edgeProbabilities = decisionGraph?.edgeProbabilities ?? {};
   134→  const branchProbs = decisionGraph?.branchProbabilities ?? { trend: 0.33, breakout: 0.33, reversal: 0.34 };
   135→  const scenarioProbabilities = decisionGraph?.scenarioProbabilities ?? {};
   136→  const pathContributions = decisionGraph?.pathContributions ?? {};
   137→  const nodeValues = decisionGraph?.nodeValues ?? {};
   138→
   139→  // Build node lookup map
   140→  const nodeMap = useMemo(() => {
   141→    const map: Record<string, (typeof nodes)[number]> = {};
   142→    for (const n of nodes) map[n.id] = n;
   143→    return map;
   144→  }, [nodes]);
   145→
   146→  // ── Visible edges/nodes based on filter ──────────────────────────
   147→  const { visibleEdgeIndices, visibleNodes } = useMemo(() => {
   148→    if (activeFilter === 'all') {
   149→      const allEdgeIdx = edges.map((_, i) => i);
   150→      const allNodes = new Set(nodes.map(n => n.id));
   151→      return { visibleEdgeIndices: allEdgeIdx, visibleNodes: allNodes };
   152→    }
   153→    if (SCENARIO_KEYS.includes(activeFilter as typeof SCENARIO_KEYS[number])) {
   154→      // Show edges going TO this scenario node, plus their source nodes
   155→      const edgeSet = new Set<number>();
   156→      const nodeSet = new Set<string>(['ROOT', activeFilter]);
   157→      edges.forEach((e, i) => {
   158→        if (e.to === activeFilter) {
   159→          edgeSet.add(i);
   160→          nodeSet.add(e.from);
   161→        }
   162→      });
   163→      // Also show intermediate nodes connected to those sources
   164→      edges.forEach((e, i) => {
   165→        if (nodeSet.has(e.to) && !e.to.startsWith('R')) {
   166→          edgeSet.add(i);
   167→          nodeSet.add(e.from);
   168→        }
   169→      });
   170→      return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   171→    }
   172→    if (activeFilter === 'trend' || activeFilter === 'breakout' || activeFilter === 'reversal') {
   173→      const branchType = `branch-${activeFilter}` as const;
   174→      const edgeSet = new Set<number>();
   175→      const nodeSet = new Set<string>(['ROOT']);
   176→      edges.forEach((e, i) => {
   177→        if (e.type === branchType || (e.type !== `branch-trend` && e.type !== `branch-breakout` && e.type !== `branch-reversal` && nodeSet.has(e.from))) {
   178→          // Include branch-type edges
   179→          if (e.type === branchType) {
   180→            edgeSet.add(i);
   181→            nodeSet.add(e.from);
   182→            nodeSet.add(e.to);
   183→          }
   184→        }
   185→      });
   186→      // Now add all edges whose source is already in nodeSet
   187→      const expanded = new Set<number>();
   188→      const expandedNodes = new Set<string>(nodeSet);
   189→      edges.forEach((e, i) => {
   190→        if (expandedNodes.has(e.from)) {
   191→          expanded.add(i);
   192→          expandedNodes.add(e.to);
   193→        }
   194→      });
   195→      return { visibleEdgeIndices: [...expanded], visibleNodes: expandedNodes };
   196→    }
   197→    // Type filter (up/pullback/down/risk)
   198→    const edgeSet = new Set<number>();
   199→    const nodeSet = new Set<string>(['ROOT']);
   200→    edges.forEach((e, i) => {
   201→      if (e.type === activeFilter) {
   202→        edgeSet.add(i);
   203→        nodeSet.add(e.from);
   204→        nodeSet.add(e.to);
   205→      }
   206→    });
   207→    return { visibleEdgeIndices: [...edgeSet], visibleNodes: nodeSet };
   208→  }, [activeFilter, edges, nodes]);
   209→
   210→  // ── Draw SVG edges ───────────────────────────────────────────────
   211→  const drawEdges = useCallback(() => {
   212→    const svg = svgRef.current;
   213→    const graph = graphRef.current;
   214→    if (!svg || !graph || edges.length === 0) return;
   215→
   216→    let markersSvg = '';
   217→    const seenColors = new Set<string>();
   218→    for (const e of edges) {
   219→      const c = EDGE_COLORS[e.type];
   220→      if (c && !seenColors.has(e.type)) {
   221→        seenColors.add(e.type);
   222→        markersSvg += `<marker id="arrow-${e.type}" markerWidth="8" markerHeight="8" refX="7" refY="4" orient="auto"><path d="M0,0 L8,4 L0,8 z" fill="${c}"/></marker>`;
   223→      }
   224→    }
   225→
   226→    let pathsSvg = '';
   227→    const graphRect = graph.getBoundingClientRect();
   228→
   229→    for (let ei = 0; ei < edges.length; ei++) {
   230→      const { from: fromId, to: toId, label, type } = edges[ei];
   231→      const isVisible = visibleEdgeIndices.includes(ei);
   232→      const fromEl = nodeRefs.current[fromId];
   233→      const toEl = nodeRefs.current[toId];
   234→      if (!fromEl || !toEl) continue;
   235→
   236→      const fRect = fromEl.getBoundingClientRect();
   237→      const tRect = toEl.getBoundingClientRect();
   238→      const ax = fRect.left - graphRect.left + fRect.width / 2;
   239→      const ay = fRect.top - graphRect.top + fRect.height / 2;
   240→      const bx = tRect.left - graphRect.left + tRect.width / 2;
   241→      const by = tRect.top - graphRect.top + tRect.height / 2;
   242→      const dx = bx - ax;
   243→      const dy = by - ay;
   244→      const dist = Math.sqrt(dx * dx + dy * dy);
   245→      if (dist === 0) continue;
   246→
   247→      // Curved edges with bend proportional to distance
   248→      const bend = Math.min(40, Math.max(12, dist * 0.08));
   249→      const mx = (ax + bx) / 2;
   250→      const my = (ay + by) / 2;
   251→      const nx = -dy / dist;
   252→      const ny = dx / dist;
   253→      const cx = mx + nx * bend;
   254→      const cy = my + ny * bend;
   255→
   256→      const d = `M ${ax} ${ay} Q ${cx} ${cy} ${bx} ${by}`;
   257→      const prob = edgeProbabilities[ei] ?? 0;
   258→      const probLabel = toPersianDigits((prob * 100).toFixed(0)) + '٪';
   259→      const edgeColor = EDGE_COLORS[type] ?? '#6b7280';
   260→
   261→      pathsSvg += `<path d="${d}" stroke="${edgeColor}" stroke-width="${isVisible ? 2 : 0.8}" opacity="${isVisible ? 0.7 : 0.05}" fill="none" marker-end="url(#arrow-${type})" data-type="${type}" data-from="${fromId}" data-to="${toId}" class="edge-path" style="transition: opacity .25s, stroke-width .25s;"/>`;
   262→
   263→      // Edge labels — skip deterministic root→branch and branch→event edges (index < 6)
   264→      if (isVisible && ei >= 6) {
   265→        pathsSvg += `<text x="${cx}" y="${cy - 5}" fill="#374151" font-size="8" text-anchor="middle" paint-order="stroke" stroke="#ffffff" stroke-width="3" stroke-linejoin="round" opacity="0.8" data-type="${type}" class="edge-label">${label}</text>`;
   266→        pathsSvg += `<text x="${cx}" y="${cy + 7}" fill="${edgeColor}" font-size="9" font-weight="bold" text-anchor="middle" paint-order="stroke" stroke="#ffffff" stroke-width="2.5" stroke-linejoin="round" opacity="0.85" class="edge-prob">${probLabel}</text>`;
   267→      }
   268→    }
   269→
   270→    svg.innerHTML = `<defs>${markersSvg}</defs>${pathsSvg}`;
   271→  }, [visibleEdgeIndices, edges, edgeProbabilities]);
   272→
   273→  useEffect(() => { drawEdges(); }, [drawEdges]);
   274→
   275→  useEffect(() => {
   276→    const el = shellRef.current;
   277→    if (!el) return;
   278→    const observer = new ResizeObserver(() => drawEdges());
   279→    observer.observe(el);
   280→    return () => observer.disconnect();
   281→  }, [drawEdges]);
   282→
   283→
   284→  // ── Detail panel content ─────────────────────────────────────────
   285→  const detailContent = useMemo(() => {
   286→    if (!selectedNode) return null;
   287→    const node = nodeMap[selectedNode];
   288→    if (!node) return null;
   289→
   290→    // For terminal scenario nodes, show branch breakdown
   291→    if (node.isTerminal && SCENARIO_KEYS.includes(selectedNode as typeof SCENARIO_KEYS[number])) {
   292→      const contrib = pathContributions[selectedNode] ?? { trend: 0, breakout: 0, reversal: 0 };
   293→      const meta = SCENARIO_META[selectedNode];
   294→      const s = scenarios[selectedNode as keyof typeof scenarios];
   295→      return (
   296→        <div className="space-y-3">
   297→          <h3 className="text-sm font-bold" style={{ color: meta.color }}>{node.title}</h3>
   298→          <p className="text-xs text-[#374151]">{node.desc}</p>
   299→          <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border border-[#e5e7eb] bg-[#f3f4f6] text-[#374151]">{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   300→          <div className="text-xs text-[#6b7280]" dir="ltr">{nodeValues[selectedNode] ?? '--'}</div>
   301→
   302→          <div className="mt-3 pt-3 border-t border-[#e5e7eb]">
   303→            <p className="text-xs font-bold text-[#374151] mb-2">سهم هر استراتژی:</p>
   304→            {Object.entries(BRANCH_META).map(([bKey, bMeta]) => {
   305→              const val = contrib[bKey as 'trend' | 'breakout' | 'reversal'];
   306→              const pct = (val * 100).toFixed(1);
   307→              return (
   308→                <div key={bKey} className="flex items-center justify-between py-1.5 border-b border-dashed border-[#e5e7eb]">
   309→                  <div className="flex items-center gap-1.5">
   310→                    <span className="inline-block w-2 h-2 rounded-full" style={{ background: bMeta.color }} />
   311→                    <span className="text-[11px] text-[#374151]">{bMeta.label}</span>
   312→                  </div>
   313→                  <span className="text-[11px] font-bold" style={{ color: bMeta.color }}>{toPersianDigits(pct)}٪</span>
   314→                </div>
   315→              );
   316→            })}
   317→            <div className="flex items-center justify-between py-1.5">
   318→              <span className="text-[11px] font-bold text-[#111827]">مجموع</span>
   319→              <span className="text-[11px] font-black" style={{ color: meta.color }}>{toFa(s?.probability ?? 0)}٪</span>
   320→            </div>
   321→          </div>
   322→        </div>
   323→      );
   324→    }
   325→
   326→    // For non-terminal nodes, show incoming/outgoing edges
   327→    const inputs = edges.filter(e => e.to === selectedNode);
   328→    const outputs = edges.filter(e => e.from === selectedNode);
   329→
   330→    return (
   331→      <div className="space-y-3">
   332→        <h3 className="text-sm font-bold text-[#111827]">{node.title}</h3>
   333→        <p className="text-xs text-[#374151]"><b>مقدار / وضعیت:</b> {nodeValues[selectedNode] ?? '--'}</p>
   334→        <span className="inline-block px-2 py-0.5 rounded-md text-[10px] border border-[#e5e7eb] bg-[#f3f4f6] text-[#374151]">{node.type === 'decision' ? 'گره تصمیم‌گیری' : node.type === 'event' ? 'گره رویداد شانسی' : 'گره نتیجه'}</span>
   335→        <p className="text-xs text-[#6b7280] leading-relaxed">{node.desc}</p>
   336→        {inputs.length > 0 && (
   337→          <div>
   338→            <p className="text-xs font-medium text-[#374151] mb-1">مسیرهای ورودی ({toFa(inputs.length)}):</p>
   339→            <ul className="space-y-1">
   340→              {inputs.map((e, i) => {
   341→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   342→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   343→                return (
   344→                  <li key={i} className="text-[11px] text-[#6b7280] leading-relaxed border-t border-dashed border-[#e5e7eb] pt-1.5">
   345→                    <b className="text-[#374151]">{SCENARIO_DISPLAY[e.from] || e.from} ← {SCENARIO_DISPLAY[e.to] || e.to}</b>
   346→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${edgeColor}20`, color: edgeColor }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   347→                    <br />{e.label}
   348→                  </li>
   349→                );
   350→              })}
   351→            </ul>
   352→          </div>
   353→        )}
   354→        {outputs.length > 0 && (
   355→          <div>
   356→            <p className="text-xs font-medium text-[#374151] mb-1">مسیرهای خروجی ({toFa(outputs.length)}):</p>
   357→            <ul className="space-y-1">
   358→              {outputs.map((e, i) => {
   359→                const ep = edgeProbabilities[edges.indexOf(e)] ?? 0;
   360→                const edgeColor = EDGE_COLORS[e.type] ?? '#6b7280';
   361→                return (
   362→                  <li key={i} className="text-[11px] text-[#6b7280] leading-relaxed border-t border-dashed border-[#e5e7eb] pt-1.5">
   363→                    <b className="text-[#374151]">{SCENARIO_DISPLAY[e.from] || e.from} → {SCENARIO_DISPLAY[e.to] || e.to}</b>
   364→                    <span className="mr-2 px-1.5 py-0.5 rounded text-[9px] font-bold" style={{ background: `${edgeColor}20`, color: edgeColor }}>{toPersianDigits((ep * 100).toFixed(1))}٪</span>
   365→                    <br />{e.label}
   366→                  </li>
   367→                );
   368→              })}
   369→            </ul>
   370→          </div>
   371→        )}
   372→      </div>
   373→    );
   374→  }, [selectedNode, nodeMap, nodeValues, edges, edgeProbabilities, pathContributions, scenarios]);
   375→
   376→  // ── All filter buttons ────────────────────────────────────────────
   377→  const allFilters = [
   378→    ...TYPE_FILTERS,
   379→    { key: 'sep1', label: '│', isSep: true as const },
   380→    ...Object.entries(BRANCH_META).map(([k, v]) => ({ key: k, label: v.label, branchKey: k, branchColor: v.color })),
   381→    { key: 'sep2', label: '│', isSep: true as const },
   382→    ...SCENARIO_KEYS.map(k => ({ key: k, label: SCENARIO_META[k].label, scenarioKey: k })),
   383→  ];
   384→
   385→  // ═══ Design dimensions ═══
   386→  const DESIGN_W = 1500;
   387→  const DESIGN_H = 820;
   388→  const DISPLAY_W = 1200;
   389→  const DISPLAY_H = DESIGN_H;
   390→  const scaleX = DISPLAY_W / DESIGN_W;
   391→  const scaleY = DISPLAY_H / DESIGN_H;
   392→
   393→  // ── Loading state ────────────────────────────────────────────────
   394→  if (!decisionGraph) {
   395→    return <VdssGraphSkeleton />;
   396→  }
   397→
   398→  return (
   399→    <div className="space-y-3" dir="rtl">
   400→      {/* ═══ HEADER ═══ */}
   401→      <div className="flex items-center justify-between gap-4 px-4 py-3 rounded-2xl border border-[#e5e7eb]"
   402→        style={{ background: 'linear-gradient(105deg, #ffffff, #f3f4f6)', boxShadow: '0 4px 16px rgba(0,0,0,.06)' }}>
   403→        <div className="flex items-center gap-3">
   404→          <div className="w-10 h-10 rounded-xl border border-amber-200 flex items-center justify-center text-amber-800 text-xl font-bold"
   405→            style={{ boxShadow: 'inset 0 0 22px rgba(146,64,14,.06), 0 0 22px rgba(146,64,14,.04)' }}>◈</div>
   406→          <div>
   407→            <h2 className="text-base font-bold text-[#111827]">گراف تصمیم {symbolName}</h2>
   408→            <p className="text-[11px] text-[#6b7280]">مدل ۳‌شاخه‌ای | پیروی از روند، شکست، بازگشت | ۲۷ مسیر به ۹ سناریو</p>
   409→          </div>
   410→        </div>
   411→        <div className="text-left text-xs text-[#6b7280] leading-relaxed pr-4 border-r border-[#e5e7eb]">
   412→          نقطه مرجع: <b className="text-cyan-700">{toFa(currentPrice)}</b><br />
   413→          افق برآورد: ۱۰ تا ۲۵ جلسه معاملاتی
   414→        </div>
   415→      </div>
   416→
   417→      {/* ═══ Metric Cards ═══ */}
   418→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   419→        <MetricCard label="مقدار مرجع" value={toFa(currentPrice) + ' ریال'} color="text-cyan-700" />
   420→        <MetricCard label="احتمال روند" value={toPersianDigits((branchProbs.trend * 100).toFixed(0)) + '٪'} color="text-cyan-700" />
   421→        <MetricCard label="احتمال شکست" value={toPersianDigits((branchProbs.breakout * 100).toFixed(0)) + '٪'} color="text-amber-700" />

   559→                    borderLeftWidth,
   560→                    borderLeftStyle: borderStyle,

   564→                    boxShadow: selectedShadow,

   715→                <span className="text-xs font-bold text-[#374151]">{meta.label}</span>

   730→                  {toFa(s.targetMin)} — {toFa(s.targetMax)}

   732→              </div>
   733→            );
   734→          })}

   738→        </div>
   739→      </div>
   740→    </div>
   741→  );
   742→}
