   106→   106→// — S/R strength-based line styling (solid only, thickness = strength) ————
   107→   107→function srLineStyle(strength: number, isTarget: boolean): { lineWidth: number; lineStyle: 0; color: string } {
   108→   108→  if (isTarget) {
   109→   109→    return { lineWidth: 4, lineStyle: 0, color: '' };
   110→   110→  }
   111→   111→  if (strength >= 9) return { lineWidth: 5, lineStyle: 0, color: '' };
   112→   112→  if (strength >= 7) return { lineWidth: 4, lineStyle: 0, color: '' };
   113→   113→  if (strength >= 5) return { lineWidth: 3, lineStyle: 0, color: '' };
   114→   114→  if (strength >= 3) return { lineWidth: 2, lineStyle: 0, color: '' };
   115→   115→  return { lineWidth: 1, lineStyle: 0, color: '' };
   116→   116→}
