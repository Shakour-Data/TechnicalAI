   262→   262→  if (keys.length === 0) return new Map();
   263→   263→
   264→   264→  const now = Date.now();
   265→   265→  if (staticPriceCache.data.size > 0 && now - staticPriceCache.ts < STATIC_PRICE_TTL) {
   266→   266→    // Return cached data for requested keys

   326→   326→/* ─── Fetch live instruments (currency + gold from tgju-api) + static list ── */
   327→   327→
   328→   328→export async function fetchTgjuInstruments(): Promise<TgjuInstrument[]> {
   329→   329→  const now = Date.now();
   330→   330→  if (liveInstrumentsCache.data && now - liveInstrumentsCache.ts < INSTRUMENTS_TTL) {
   331→   331→    return liveInstrumentsCache.data;
   332→   332→  }
   333→   333→
   334→   334→  const controller = new AbortController();
   335→   335→  const timeout = setTimeout(() => controller.abort(), 10000);
   336→   336→
   337→   337→  try {
   338→   338→    const [currencyRes, goldRes] = await Promise.all([
   339→   339→      fetch(`${TGJU_API_BASE}/api/price/currency`, { signal: controller.signal }),
   340→   340→      fetch(`${TGJU_API_BASE}/api/price/gold`, { signal: controller.signal }),
   341→   341→    ]);
   342→   342→
   343→   343→    clearTimeout(timeout);
   344→   344→
   345→   345→    const instruments: TgjuInstrument[] = [];
   346→   346→    const currencyGoldKeys: string[] = []; // keys to fetch chart data for accurate change
   347→   347→
   348→   348→    if (currencyRes.ok) {
   349→   349→      const currencies: TgjuPriceItem[] = await currencyRes.json();
   350→   350→      for (const c of currencies) {
   351→   351→        const price = parsePersianNum(c.price);
   352→   352→        const low = parsePersianNum(c.low_price);
   353→   353→        const high = parsePersianNum(c.high_price);
   354→   354→        instruments.push({
   355→   355→          title: c.title,
   356→   356→          key: c.key,
   357→   357→          price,
   358→   358→          highPrice: high,
   359→   359→          lowPrice: low,
   360→   360→          change: 0, // Will be updated from chart data
   361→   361→          changePercent: 0, // Will be updated from chart data
   362→   362→          category: 'currency',
   363→   363→        });
   364→   364→        currencyGoldKeys.push(c.key);
   365→   365→      }
   366→   366→    }
   367→   367→
   368→   368→    if (goldRes.ok) {
   369→   369→      const goldCategories: TgjuGoldCategory[] = await goldRes.json();
   370→   370→      for (const cat of goldCategories) {
   371→   371→        for (const item of cat.prices) {
   372→   372→          const price = parsePersianNum(item.price);
   373→   373→          const low = parsePersianNum(item.low_price);
   374→   374→          const high = parsePersianNum(item.high_price);
   375→   375→
   376→   376→          let category: TgjuCategoryType = 'gold';
   377→   377→          if (item.key.startsWith('ime_fund_')) category = 'gold_etf';
   378→   378→          else if (item.key.startsWith('silver_')) category = 'silver';
   379→   379→
   380→   380→          instruments.push({
   381→   381→            title: item.title,
   382→   382→            key: item.key,
   383→   383→            price,
   384→   384→            highPrice: high,
   385→   385→            lowPrice: low,
   386→   386→            change: 0, // Will be updated from chart data
   387→   387→            changePercent: 0, // Will be updated from chart data
   388→   388→            category,
   389→   389→            groupTitle: cat.title,
   390→   390→          });
   391→   391→          currencyGoldKeys.push(item.key);
   392→   392→        }
   393→   393→      }
   394→   394→    }
   395→   395→
   396→   396→    // Add static instruments (prices will be populated by background batch fetch)
   397→   397→    for (const s of STATIC_INSTRUMENTS) {
   398→   398→      const cached = staticPriceCache.data.get(s.key);
   399→   399→      instruments.push({
   400→   400→        title: s.title,
   401→   401→        key: s.key,
   402→   402→        price: cached?.close ?? 0,
   403→   403→        highPrice: cached?.high ?? 0,
   404→   404→        lowPrice: cached?.low ?? 0,
   405→   405→        change: cached ? (cached.close - cached.prevClose) : 0,
   406→   406→        changePercent: cached && cached.prevClose > 0 ? ((cached.close - cached.prevClose) / cached.prevClose) * 100 : 0,
   407→   407→        category: s.category,
   408→   408→        groupTitle: s.groupTitle,
   409→   409→      });
   410→   410→    }
   411→   411→
   412→   412→    liveInstrumentsCache.data = instruments;
   413→   413→    liveInstrumentsCache.ts = now;
   414→   414→
   415→   415→    // ── Background: fetch chart data for ALL TGJU keys ──
   416→   416→    // This gives us accurate prevClose for currency/gold (instead of using intraday low)
   417→   417→    // and populates prices for static instruments
   418→   418→    const allKeys = [
   419→   419→      ...currencyGoldKeys,
   420→   420→      ...STATIC_INSTRUMENTS.map(s => s.key),
   421→   421→    ];
   422→   422→    batchFetchChartPrices(allKeys).then((priceMap) => {
   423→   423→      if (priceMap.size > 0 && liveInstrumentsCache.data) {
   424→   424→        for (const inst of liveInstrumentsCache.data) {
   425→   425→          const p = priceMap.get(inst.key);
   426→   426→          if (!p) continue;
   427→   427→
   428→   428→          if (inst.category === 'currency' || inst.category === 'gold' ||
   429→   429→              inst.category === 'silver' || inst.category === 'gold_etf') {
   430→   430→            // Currency/gold: keep live price, but use chart prevClose for accurate change
   431→   431→            if (p.prevClose > 0) {
   432→   432→              inst.change = inst.price - p.prevClose;
   433→   433→              inst.changePercent = (inst.change / p.prevClose) * 100;
   434→   434→            }
   435→   435→          } else if (inst.price === 0) {
   436→   436→            // Static instruments: update all fields from chart data
   437→   437→            inst.price = p.close;
   438→   438→            inst.highPrice = p.high;
   439→   439→            inst.lowPrice = p.low;
   440→   440→            inst.change = p.close - p.prevClose;
   441→   441→            inst.changePercent = p.prevClose > 0 ? ((p.close - p.prevClose) / p.prevClose) * 100 : 0;
   442→   442→          }
   443→   443→        }
   444→   444→      }
   445→   445→    }).catch(() => {});
   446→   446→
   447→   447→    return instruments;
   448→   448→  } catch {
   449→   449→    clearTimeout(timeout);
   450→   450→    return liveInstrumentsCache.data || [];
   451→   451→  }
   452→   452→}

   454→   454→/* ─── Fetch historical OHLC data via z-ai page_reader ── */
   455→   455→
   456→   456→export async function fetchTgjuHistory(tgjuKey: string): Promise<TgjuOHLC[]> {
   457→   457→  const cached = historyCache.get(tgjuKey);
   458→   458→  const now = Date.now();
   459→   459→  if (cached && now - cached.ts < HISTORY_TTL) {
   460→   460→    return cached.data;
   461→   461→  }
   462→   462→
   463→   463→  // Historical data requires page_reader (rate-limited) — on-demand fetch per user click
   464→   464→  try {
   465→   465→    const { rateLimitedPageReader } = await import('@/lib/zai-shared');
   466→   466→
   467→   467→    const url = `${TGJU_CHART_API}/${tgjuKey}?lang=fa&order_dir=asc&start=0&length=5000`;
   468→   468→
   469→   469→    const html = await rateLimitedPageReader(url, 30_000);
   470→   470→    const jsonStr = html.replace(/<[^>]+>/g, '').trim();
   471→   471→
   472→   472→    if (!jsonStr.startsWith('{')) {
   473→   473→      return cached?.data || [];
   474→   474→    }
   475→   475→
   476→   476→    const data = JSON.parse(jsonStr);
   477→   477→    const rows: string[][] = data.data;
   478→   478→
   479→   479→    if (!rows || rows.length === 0) {
   480→   480→      return cached?.data || [];
   481→   481→    }
   482→   482→
   483→   483→    // Parse: [open, low, high, close, change, changePct, gregorianDate, jalaliDate]
   484→   484→    const candles: TgjuOHLC[] = rows
   485→   485→      .map((row) => ({
   486→   486→        date: (row[6] || '').replace(/\//g, '-'),
   487→   487→        open: parsePersianNum(row[0]),
   488→   488→        high: parsePersianNum(row[2]),
   489→   489→        low: parsePersianNum(row[1]),
   490→   490→        close: parsePersianNum(row[3]),
   491→   491→      }))
   492→   492→      .filter((c) => c.open > 0 && c.high > 0 && c.low > 0 && c.close > 0)
   493→   493→      .reverse(); // newest-last for charting
   494→   494→
   495→   495→    historyCache.set(tgjuKey, { data: candles, ts: now });
   496→   496→    return candles;
   497→   497→  } catch (err) {
   498→   498→    console.error(`[TGJU] Failed to fetch history for ${tgjuKey}:`, err);
   499→   499→    return cached?.data || [];
   500→   500→  }
   501→   501→}
