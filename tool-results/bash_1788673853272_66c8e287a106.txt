    17→function sma(closes: number[], period: number): number {
    18→  if (closes.length < period) return 0;
    19→  const slice = closes.slice(-period);
    20→  return slice.reduce((a, b) => a + b, 0) / period;
    21→}
    22→
    23→function emaCalc(closes: number[], period: number): number {
    24→  if (closes.length < period) return 0;
    25→  const k = 2 / (period + 1);
    26→  let v = sma(closes.slice(0, period), period);
    27→  for (let i = period; i < closes.length; i++) {
    28→    v = closes[i] * k + v * (1 - k);
    29→  }
    30→  return v;
    31→}

    33→function calcRSI(closes: number[], period = 14): number {
    34→  if (closes.length < period + 1) return 50;
    35→  let avgGain = 0, avgLoss = 0;
    36→  for (let i = 1; i <= period; i++) {
    37→    const diff = closes[i] - closes[i - 1];
    38→    if (diff > 0) avgGain += diff; else avgLoss += Math.abs(diff);
    39→  }
    40→  avgGain /= period; avgLoss /= period;
    41→  for (let i = period + 1; i < closes.length; i++) {
    42→    const diff = closes[i] - closes[i - 1];
    43→    avgGain = (avgGain * (period - 1) + (diff > 0 ? diff : 0)) / period;
    44→    avgLoss = (avgLoss * (period - 1) + (diff < 0 ? Math.abs(diff) : 0)) / period;
    45→  }
    46→  if (avgLoss === 0) return 100;
    47→  return 100 - 100 / (1 + avgGain / avgLoss);
    48→}

    50→function calcCCI(data: OHLCV[], period = 20): number {
    51→  if (data.length < period) return 0;
    52→  const tps: number[] = [];
    53→  for (let i = data.length - period; i < data.length; i++) {
    54→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
    55→  }
    56→  const mean = tps.reduce((a, b) => a + b, 0) / tps.length;
    57→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - mean), 0) / tps.length;
    58→  return meanDev === 0 ? 0 : (tps[tps.length - 1] - mean) / (0.015 * meanDev);
    59→}

    61→function calcMFI(data: OHLCV[], period = 14): number {
    62→  if (data.length < period + 1) return 50;
    63→  let posFlow = 0, negFlow = 0;
    64→  for (let i = data.length - period; i < data.length; i++) {
    65→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
    66→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
    67→    const mf = tp * data[i].volume;
    68→    if (tp > prevTp) posFlow += mf; else negFlow += mf;
    69→  }
    70→  return negFlow === 0 ? 100 : 100 - 100 / (1 + posFlow / negFlow);
    71→}

    73→function calcStochastic(data: OHLCV[], kPeriod = 14, smoothK = 3, smoothD = 3): { k: number; d: number } {
    74→  const rawKs: number[] = [];
    75→  for (let i = kPeriod - 1; i < data.length; i++) {
    76→    let lowest = Infinity, highest = -Infinity;
    77→    for (let j = i - kPeriod + 1; j <= i; j++) {
    78→      if (data[j].low < lowest) lowest = data[j].low;
    79→      if (data[j].high > highest) highest = data[j].high;
    80→    }
    81→    const range = highest - lowest;
    82→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
    83→  }
    84→  const smoothedK: number[] = [];
    85→  for (let i = smoothK - 1; i < rawKs.length; i++) {
    86→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
    87→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
    88→  }
    89→  const k = smoothedK[smoothedK.length - 1] ?? 50;
    90→  const dSlice = smoothedK.slice(-smoothD);
    91→  const d = dSlice.length > 0 ? dSlice.reduce((a, b) => a + b, 0) / dSlice.length : 50;
    92→  return { k, d };
    93→}

    95→function calcMACD(closes: number[], fast = 12, slow = 26, sig = 9): { line: number; signal: number; histogram: number } {
    96→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
    97→  const kFast = 2 / (fast + 1), kSlow = 2 / (slow + 1), kSig = 2 / (sig + 1);
    98→  let eFast = sma(closes.slice(0, fast), fast);
    99→  let eSlow = sma(closes.slice(0, slow), slow);
   100→  const macdVals: number[] = [];
   101→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   102→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   103→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   104→    macdVals.push(eFast - eSlow);
   105→  }
   106→  if (macdVals.length < sig) return { line: 0, signal: 0, histogram: 0 };
   107→  let s = sma(macdVals.slice(0, sig), sig);
   108→  for (let i = sig; i < macdVals.length; i++) {
   109→    s = macdVals[i] * kSig + s * (1 - kSig);
   110→  }
   111→  const line = macdValsacdVals.length - 1];
   112→  return { line, signal: s, histogram: line - s };
   113→}

   115→function calcBollingerBands(closes: number[], period = 20, mult = 2): { upper: number; middle: number; lower: number } {
   116→  if (closes.length < period) return { upper: 0, middle: 0, lower: 0 };
   117→  const slice = closes.slice(-period);
   118→  const mid = slice.reduce((a, b) => a + b, 0) / period;
   119→  const variance = slice.reduce((a, b) => a + (b - mid) ** 2, 0) / period;
   120→  const sd = Math.sqrt(variance);
   121→  return { upper: mid + mult * sd, middle: mid, lower: mid - mult * sd };
   122→}

   124→function calcADX(data: OHLCV[], period = 14): { adx: number; diPlus: number; diMinus: number } {
   125→  if (data.length < period * 2) return { adx: 0, diPlus: 0, diMinus: 0 };
   126→  let sumPDM = 0, sumMDM = 0, sumTR = 0;
   127→  for (let i = data.length - period; i < data.length; i++) {
   128→    const upMove = data[i].high - data[i - 1].high;
   129→    const downMove = data[i - 1].low - data[i].low;
   130→    sumPDM += upMove > downMove && upMove > 0 ? upMove : 0;
   131→    sumMDM += downMove > upMove && downMove > 0 ? downMove : 0;
   132→    sumTR += Math.max(
   133→      data[i].high - data[i].low,
   134→      Math.abs(data[i].high - data[i - 1].close),
   135→      Math.abs(data[i].low - data[i - 1].close)
   136→    );
   137→  }
   138→  const diPlus = sumTR > 0 ? (sumPDM / sumTR) * 100 : 0;
   139→  const diMinus = sumTR > 0 ? (sumMDM / sumTR) * 100 : 0;
   140→  const dx = (diPlus + diMinus) > 0 ? Math.abs(diPlus - diMinus) / (diPlus + diMinus) * 100 : 0;
   141→  return { adx: dx, diPlus, diMinus };
   142→}

   144→function calcATR(data: OHLCV[], period = 14): number {
   145→  if (data.length < period + 1) return 0;
   146→  let sum = 0;
   147→  for (let i = data.length - period; i < data.length; i++) {
   148→    sum += Math.max(
   149→      data[i].high - data[i].low,
   150→      Math.abs(data[i].high - data[i - 1].close),
   151→      Math.abs(data[i].low - data[i - 1].close)
   152→    );
   153→  }
   154→  return sum / period;
   155→}

   163→// ─── 16-Feature Extraction for VDss ─────────────────────────────────────────

   165→const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));
   166→
   167→/**
   168→ * Extract the 16 VDSS features at a given point in the data.
   169→ * Feature order must match VDSS_FEATURE_NAMES exactly.
   170→ *
   171→ * @param data - Full OHLCV history (chronological)
   172→ * @param endIdx - Index of the candle to compute features for
   173→ * @param hasVolume - Whether data has valid volume
   174→ * @returns Array of 16 feature values in [0,1] range
   175→ */
   176→export function extractVDSSFeatures(data: OHLCV[], endIdx: number, hasVolume: boolean): number[] {
   177→  const slice = data.slice(0, endIdx + 1);
   178→  const closes = slice.map(d => d.close);
   179→  const price = closes[closes.length - 1];
   180→  if (price <= 0) return new Array(16).fill(0.5);
   181→
   182→  // ── Compute indicators ──
   183→  const rsi = calcRSI(closes);
   184→  const mfi = hasVolume ? calcMFI(slice) : 50;
   185→  const cci = calcCCI(slice);
   186→  const adxResult = calcADX(slice);
   187→  const macd = calcMACD(closes);
   188→  const stoch = calcStochastic(slice);
   189→  const bb = calcBollingerBands(closes);
   190→  const atr = calcATR(slice);
   191→  const ma21 = sma(closes, 21);
   192→  const ma100 = sma(closes, 100);
   193→  const ema12 = emaCalc(closes, 12);
   194→  const ema26 = emaCalc(closes, 26);
   195→
   196→  // ── LAYER 1: Raw scores (s_*) ──
   197→  const s_rsi = rsi > 80 ? 0.95 : rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   198→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   199→  const s_mfi = mfi > 80 ? 0.95 : mfi > 70 ? 0.85 : mfi > 60 ? 0.7 : mfi > 50 ? 0.55
   200→    : mfi > 40 ? 0.45 : mfi > 30 ? 0.3 : mfi > 20 ? 0.15 : 0.05;
   201→  const s_cci = cci > 200 ? 0.9 : cci > 100 ? 0.75 : cci > 0 ? 0.6
   202→    : cci > -100 ? 0.4 : cci > -200 ? 0.25 : 0.1;
   203→  const s_adx = adxResult.diPlus > adxResult.diMinus
   204→    ? clamp(0.5 + (adxResult.diPlus - adxResult.diMinus) / 100, 0.5, 1)
   205→    : clamp(0.5 - (adxResult.diMinus - adxResult.diPlus) / 100, 0, 0.5);
   206→  const s_macd = macd.histogram > 0 && macd.line > macd.signal ? 0.9
   207→    : macd.histogram > 0 ? 0.7 : macd.line > macd.signal ? 0.55
   208→    : macd.histogram < 0 && macd.line < macd.signal ? 0.1 : 0.3;
   209→  const s_stoch = clamp(
   210→    (stoch.k > 80 ? 0.9 : stoch.k > 70 ? 0.8 : stoch.k > 50 ? 0.6
   211→      : stoch.k > 30 ? 0.4 : stoch.k > 20 ? 0.2 : 0.1)
   212→    + (stoch.k > stoch.d ? 0.1 : -0.1), 0, 1);
   213→  const bbRange = bb.upper - bb.lower;
   214→  const s_bb = bbRange === 0 ? 0.5 : clamp((price - bb.lower) / bbRange, 0, 1);
   215→  const s_ma21 = clamp(price > ma21 ? 0.7 + ((price - ma21) / ma21) * 0.5 : 0.3 + ((price - ma21) / ma21) * 0.5, 0, 1);
   216→  const s_ma100 = clamp(price > ma100 ? 0.65 + ((price - ma100) / ma100) * 0.3 : 0.35 + ((price - ma100) / ma100) * 0.3, 0, 1);
   217→  const emaGap = ema12 > 0 && ema26 > 0 ? (ema12 - ema26) / ema26 : 0;
   218→  const s_ema = clamp(0.5 + emaGap * 5, 0, 1);
   219→  const s_atr = clamp(1 - atr / price * 10, 0.2, 1);
   220→
   221→  // Trend (R² + angle)
   222→  const trendPeriod = Math.min(21, closes.length);
   223→  const trendSlice = closes.slice(-trendPeriod);
   224→  const n = trendSlice.length;
   225→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   226→  for (let i = 0; i < n; i++) { sumX += i; sumY += trendSlice[i]; sumXY += i * trendSlice[i]; sumX2 += i * i; sumY2 += trendSlice[i] ** 2; }
   227→  const denom = n * sumX2 - sumX * sumX;
   228→  let r2 = 0, angle = 0;
   229→  if (denom !== 0) {
   230→    const slope = (n * sumXY - sumX * sumY) / denom;
   231→    const meanY = sumY / n;
   232→    let ssRes = 0, ssTot = 0;
   233→    for (let i = 0; i < n; i++) {
   234→      ssRes += (trendSlice[i] - (slope * i + meanY - slope * (n - 1) / 2)) ** 2;
   235→      ssTot += (trendSlice[i] - meanY) ** 2;
   236→    }
   237→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
   238→    angle = meanY > 0 ? Math.atan(slope / meanY) * (180 / Math.PI) : 0;
   239→  }
   240→  const tStr = r2 * Math.abs(angle) / 45;
   241→  const direction = denom !== 0 ? (n * sumXY - sumX * sumY) / denom : 0;
   242→  const s_trend = direction > 0 ? clamp(0.5 + tStr / 2, 0.5, 1)
   243→    : direction < 0 ? clamp(0.5 - tStr / 2, 0, 0.5) : 0.5;
   244→
   245→  // Simplified S/R: distance to nearest local high/low
   246→  let s_sr = 0.5;
   247→  if (slice.length > 20) {
   248→    const lookback = slice.slice(-30);
   249→    let nearestHigh = Infinity, nearestLow = 0;
   250→    for (const d of lookback) {
   251→      if (d.high > price && d.high < nearestHigh) nearestHigh = d.high;
   252→      if (d.low < price && d.low > nearestLow) nearestLow = d.low;
   253→    }
   254→    const dR = nearestHigh < Infinity ? Math.abs(price - nearestHigh) / nearestHigh : 1;
   255→    const dS = nearestLow > 0 ? Math.abs(price - nearestLow) / nearestLow : 1;
   256→    s_sr = clamp(0.5 - dR * 1.5 + dS * 1.0, 0, 1);
   257→  }
   258→
   259→  // ── LAYER 2: Momentum correction ──
   260→  const TB = 5;
   261→  let f_rsi = s_rsi, f_mfi = s_mfi, f_cci = s_cci, f_macd = s_macd, f_stoch = s_stoch;
   262→  if (slice.length > TB + 15) {
   263→    const pastSlice = slice.slice(0, -TB);
   264→    const pastCloses = pastSlice.map(d => d.close);
   265→    const rsiPast = calcRSI(pastCloses);
   266→    const mfiPast = hasVolume ? calcMFI(pastSlice) : 50;
   267→    const cciPast = calcCCI(pastSlice);
   268→    const macdPast = calcMACD(pastCloses);
   269→    const stochPast = calcStochastic(pastSlice);
   270→
   271→    f_rsi = clamp(s_rsi + (rsi - rsiPast) / 100 * 0.08, 0, 1);
   272→    f_mfi = clamp(s_mfi + (mfi - mfiPast) / 100 * 0.08, 0, 1);
   273→    f_cci = clamp(s_cci + (cci - cciPast) / 300 * 0.06, 0, 1);
   274→    f_macd = clamp(s_macd + (macd.histogram - macdPast.histogram) / (Math.abs(macd.histogram) + Math.abs(macdPast.histogram) + 1) * 3 * 0.08, 0, 1);
   275→    f_stoch = clamp(s_stoch + (stoch.k - stochPast.k) / 50 * 0.06, 0, 1);
   276→  }
   277→
   278→  // ── LAYER 2: Cross signals (binary per spec) ──
   279→  let f_stochCross = 0.1;
   280→  let f_macdCross = 0.1;
   281→  let f_div = 0.15;
   282→
   283→  if (slice.length > 20) {
   284→    const prevSlice = slice.slice(0, -1);
   285→    const prevStoch = calcStochastic(prevSlice);
   286→    const prevMacd = calcMACD(prevSlice.map(d => d.close));
   287→
   288→    // Stochastic bullish cross: K_prev < D_prev AND K_curr > D_curr
   289→    f_stochCross = (prevStoch.k < prevStoch.d && stoch.k > stoch.d) ? 0.9 : 0.1;
   290→
   291→    // MACD bullish cross: line_prev < signal_prev AND line_curr > signal_curr
   292→    f_macdCross = (prevMacd.line < prevMacd.signal && macd.line > macd.signal) ? 0.9 : 0.1;
   293→
   294→    // Divergence detection
   295→    if (closes.length > 40) {
   296→      const p20 = closes[closes.length - 21];
   297→      const rsi20 = calcRSI(closes.slice(0, -20));
   298→      const pChange = price - p20;
   299→      const rsiChange = rsi - rsi20;
   300→      if (pChange !== 0 && rsiChange !== 0) {
   301→        f_div = (pChange > 0 && rsiChange < 0) ? 0.85 : (pChange < 0 && rsiChange > 0) ? 0.85 : 0.15;
   302→      }
   303→    }
   304→  }
   305→
   306→  // ── Return 16 features in exact order ──
   307→  return [
   308→    f_rsi, f_mfi, f_cci, s_adx,
   309→    f_macd, f_stoch, s_bb, s_ma21, s_ma100, s_ema,
   310→    s_atr, s_trend, s_sr,
   311→    f_stochCross, f_macdCross, f_div,
   312→  ];
   313→}

   317→interface CachedModel {
   318→  result: AdaptiveModelResult;
   319→  time: number;
   320→}
   321→
   322→const modelCache = new Map<string, CachedModel>();
   323→const CACHE_TTL = 60 * 60 * 1000; // 1 hour
   324→
   325→export function getCachedAdaptiveModel(symbol: string): AdaptiveModelResult | null {
   326→  const cached = modelCache.get(symbol);
   327→  if (cached && Date.now() - cached.time < CACHE_TTL) return cached.result;
   328→  return null;
   329→}

   331→export function setCachedAdaptiveModel(symbol: string, result: AdaptiveModelResult): void {
   332→  modelCache.set(symbol, { result, time: Date.now() });
   333→}

   335→// ─── Main Training Function ─────────────────────────────────────────────────
   336→
   337→/**
   338→ * Train the Adaptive Weight Model for a given symbol.
   339→ * Extracts 16 VDSS features for each historical candle, creates binary labels,
   340→ * trains Logistic Regression with TimeSeriesSplit validation.
   341→ *
   342→ * @param data - Full OHLCV history (chronological, oldest first)
   343→ * @param symbol - Symbol identifier for caching
   344→ * @param minSamples - Minimum samples needed (default 70)
   345→ * @returns AdaptiveModelResult or null
   346→ */
   347→export function trainAdaptiveModel(data: OHLCV[], symbol: string, minSamples = 70): AdaptiveModelResult | null {
   348→  const FORWARD_DAYS = 5;
   349→  const START_IDX = 40; // Need enough history for all indicators to stabilize
   350→
   351→  if (data.length < START_IDX + FORWARD_DAYS + minSamples) return null;
   352→
   353→  const hasVolume = data.some(d => d.volume > 0);
   354→
   355→  const X: number[][] = [];
   356→  const y: number[] = [];
   357→
   358→  for (let i = START_IDX; i < data.length - FORWARD_DAYS; i++) {
   359→    const features = extractVDSSFeatures(data, i, hasVolume);
   360→    const currentPrice = data[i].close;
   361→    const futurePrice = data[i + FORWARD_DAYS].close;
   362→
   363→    if (currentPrice <= 0) continue;
   364→    if (features.some(f => !isFinite(f))) continue;
   365→
   366→    // Label: 1 if 5-day forward return > 1%, else 0
   367→    const forwardReturn = (futurePrice - currentPrice) / currentPrice;
   368→    const label = forwardReturn > 0.01 ? 1 : 0;
   369→
   370→    X.push(features);
   371→    y.push(label);
   372→  }
   373→
   374→  if (X.length < minSamples) return null;
   375→
   376→  const model = new AdaptiveWeightModel(minSamples);
   377→  const success = model.train(X, y);
   378→
   379→  if (!success) return null;
   380→
   381→  // Predict current features
   382→  const currentFeatures = extractVDSSFeatures(data, data.length - 1, hasVolume);
   383→  model.predictScore(currentFeatures);
   384→
   385→  const result = model.getResult();
   386→  setCachedAdaptiveModel(symbol, result);
   387→
   388→  return result;
   389→}

   391→// ─── Calculate bullConsensus with ML Weights (Layer 3) ──────────────────────
   392→
   393→/**
   394→ * Calculate bullConsensus using ML weights.
   395→ * Falls back to default weights if ML model is not trained.
   396→ *
   397→ * @param features - 16 VDSS features for the current candle
   398→ * @param mlResult - Adaptive model result (or null for fallback)
   399→ * @param hasVolume - Whether MFI feature is valid
   400→ */
   401→export function calculateBullConsensus(
   402→  features: number[],
   403→  mlResult: AdaptiveModelResult | null,
   404→  hasVolume: boolean,
   405→): { bullConsensus: number; usedML: boolean } {
   406→  if (mlResult && mlResult.isTrained && mlResult.weights.length === 16) {
   407→    const weights = mlResult.weights;
   408→    let bullConsensus = 0;
   409→    let totalWeight = 0;
   410→
   411→    for (let j = 0; j < 16; j++) {
   412→      // Skip MFI (index 1) if no volume
   413→      if (j === 1 && !hasVolume) continue;
   414→      bullConsensus += weights[j] * features[j];
   415→      totalWeight += weights[j];
   416→    }
   417→
   418→    if (totalWeight > 0) bullConsensus /= totalWeight;
   419→
   420→    // Blend with ML direct prediction (dynamic weight based on accuracy)
   421→    if (mlResult.predictionProb !== null) {
   422→      const mlWeight = Math.max(0, Math.min(mlResult.recentAccuracy * 0.30, 0.30));
   423→      bullConsensus = bullConsensus * (1 - mlWeight) + mlResult.predictionProb * mlWeight;
   424→    }
   425→
   426→    return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: true };
   427→  }
   428→
   429→  // Fallback: default weights (per spec, only when ML is unavailable)
   430→  const DEFAULT_WEIGHTS = [
   431→    0.15, 0.12, 0.10, 0.08,  // rsi, mfi, cci, adx
   432→    0.14, 0.10, 0.08,         // macd, stoch, bb
   433→    0.10, 0.08, 0.10,         // ma21, ma100, ema
   434→    0.05, 0.10, 0.05,         // atr, trend, sr
   435→    0.06, 0.06, 0.08,         // stochCross, macdCross, div
   436→  ];
   437→
   438→  let bullConsensus = 0;
   439→  let totalWeight = 0;
   440→  for (let j = 0; j < 16; j++) {
   441→    if (j === 1 && !hasVolume) continue;
   442→    bullConsensus += DEFAULT_WEIGHTS[j] * features[j];
   443→    totalWeight += DEFAULT_WEIGHTS[j];
   444→  }
   445→  if (totalWeight > 0) bullConsensus /= totalWeight;
   446→
   447→  return { bullConsensus: Math.max(0, Math.min(1, bullConsensus)), usedML: false };
   448→}

   503→// ─── Decision Graph Scenario Probabilities (3-Branch DAG) ──────────────────
   504→// Implements the exact structure from TechnicalAnalysisDssGraph.txt:
   505→//   Root → 3 branches (Trend Following A, Breakout B, Reversal C)
   506→//   Each branch → 9 edges → 9 leaf scenarios
   507→//   P(Scenario_S) = Σ_branch P(Branch) × P(Edge_S | Branch)
   508→//   Where P(Branch) is derived from EMV maximization,
   509→//   and P(Edge) is the conditional probability per edge formula.
   510→// ═══════════════════════════════════════════════════════════════════════════
   511→
   512→export interface DecisionGraphContext {
   513→  bullConsensus: number;
   514→  price: number;
   515→  R1_level: number;
   516→  S1_level: number;

   525→  adx: number;
   526→  diPlus: number;

   529→  bbUpper: number;
   530→  bbLower: number;
   531→  bbMiddle: number;
   532→  hasVolume: boolean;
   533→  volume: number;
   534→  avgVolume20: number;
   535→  mlResult: AdaptiveModelResult | null;
   536→  mlPrediction: MLPredictionResult | null;
   537→  mlRegime: MLRegimeResult | null;
   538→}

   540→/**
   541→ * Calculate 9-scenario probabilities using the 3-branch decision graph.
   542→ * Each branch (Trend Following, Breakout, Reversal) emits conditional probabilities
   543→ * for each scenario. Final probability is the EMV-weighted combination.
   544→ *
   545→ * Per TechnicalAnalysisDssGraph.txt:
   546→ *   P_Bull_Final = α × P_Bull_Current + (1-α) × P_Bull_Forecast
   547→ *   P_Bear_Final = α × P_Bear_Current + (1-α) × P_Bear_Forecast
   548→ *   P_Neutral_Final = 1 - (P_Bull_Final + P_Bear_Final)
   549→ *
   550→ *   P(Scenario_S) = P_Bull_Final × CP_S|Bull + P_Bear_Final × CP_S|Bear + P_Neutral_Final × CP_S|Neutral
   551→ */
   552→export function calculateScenarioProbabilities(
   553→  bullConsensus: number,
   554→  price: number,
   555→  SC1: number,
   556→  S1: number,
   557→  MA100: number,
   558→  rsi: number,
   559→  mfi: number,
   560→  stochK: number,
   561→  hasVolume: boolean,
   562→  mlResult: AdaptiveModelResult | null,
   563→  adx: number = 25,
   564→  atr: number = 0,
   565→): { pSC1: number; pSC2: number; pSC3: number; pSC4: number; pSC5: number; pSC6: number; pSC7: number; pSC8: number; pSC9: number; factors: { momentum: number; volatility: number; trend: number } } {
   566→  // ── Adaptive ML factors ──
   567→  const mlParams = mlResult?.adaptiveParams;
   568→  const factors = {
   569→    momentum: mlParams?.momentumFactor ?? 0.7,
   570→    volatility: mlParams?.volatilityFactor ?? 0.5,
   571→    trend: mlParams?.trendFactor ?? 0.6,
   572→  };
   573→
   574→  const bull = bullConsensus;
   575→  const bear = 1 - bullConsensus;
   576→
   577→  // ═══════════════════════════════════════════════════════════════════════
   578→  // STEP 1: Compute P_Bull_Current, P_Bear_Current from current indicators
   579→  // ═══════════════════════════════════════════════════════════════════════
   580→  // Use Bayesian weighting: W_i = P(Success|Tool_i) × Prior_i / Σ
   581→  // For current signals, each indicator provides a directional score.
   582→
   583→  // RSI signal (0=bearish, 1=bullish)
   584→  const rsiBull = rsi > 70 ? 0.85 : rsi > 60 ? 0.7 : rsi > 50 ? 0.55
   585→    : rsi > 40 ? 0.45 : rsi > 30 ? 0.3 : rsi > 20 ? 0.15 : 0.05;
   586→
   587→  // MFI signal (volume-dependent)
   588→  const mfiBull = hasVolume
   589→    ? (mfi > 80 ? 0.85 : mfi > 60 ? 0.65 : mfi > 40 ? 0.45 : mfi > 20 ? 0.3 : 0.15)
   590→    : 0.5; // neutral if no volume
   591→
   592→  // Stochastic signal
   593→  const stochBull = stochK > 80 ? 0.85 : stochK > 60 ? 0.65 : stochK > 40 ? 0.45
   594→    : stochK > 20 ? 0.3 : 0.15;
   595→
   596→  // ADX trend strength normalized (0-1)
   597→  const adxNorm = Math.min(adx / 45, 1);
   598→
   599→  // Price vs MA100 position
   600→  const maPosBull = MA100 > 0 ? clamp((price - MA100) / (MA100 * 0.05), 0, 1) : 0.5;
   601→
   602→  // Distance to S/R (proximity = more likely to react)
   603→  const distR1 = R1 > 0 ? Math.exp(-3 * Math.abs(price - R1) / R1) : 0.5;
   604→  const distS1 = S1 > 0 ? Math.exp(-3 * Math.abs(price - S1) / S1) : 0.5;
   605→
   606→  // Priors per category (from instructions): trend=0.28, oscillator=0.25, volume=0.15, volatility=0.10, leading=0.12, patterns=0.10
   607→  // Adjusted: if no volume, redistribute 0.15 proportionally
   608→  let priorTrend = 0.28, priorOsc = 0.25, priorVol = hasVolume ? 0.15 : 0;
   609→  let priorVolatility = 0.10, priorLeading = 0.12, priorPattern = 0.10;
   610→  if (!hasVolume) {
   611→    const totalRemaining = priorTrend + priorOsc + priorVolatility + priorLeading + priorPattern;
   612→    const redistribute = 0.15;
   613→    priorTrend += redistribute * (priorTrend / totalRemaining);
   614→    priorOsc += redistribute * (priorOsc / totalRemaining);
   615→    priorVolatility += redistribute * (priorVolatility / totalRemaining);
   616→    priorLeading += redistribute * (priorLeading / totalRemaining);
   617→    priorPattern += redistribute * (priorPattern / totalRemaining);
   618→  }
   619→
   620→  // P_Bull_Current = Σ(W_i × Signal_Bull_i) / Σ(W_i)
   621→  const currentBullSignals = [
   622→    { signal: rsiBull, weight: priorOsc * 0.4 },           // RSI (oscillator)
   623→    { signal: mfiBull, weight: priorVol },                 // MFI (volume)
   624→    { signal: stochBull, weight: priorOsc * 0.35 },        // Stochastic (oscillator)
   625→    { signal: maPosBull, weight: priorTrend * 0.5 },       // MA position (trend)
   626→    { signal: adxNorm > 0.5 && bull > 0.5 ? bull : 0.5, weight: priorTrend * 0.3 }, // ADX trend (trend)
   627→    { signal: distS1, weight: priorPattern * 0.5 },        // Near support = bullish bounce
   628→  ];
   629→
   630→  let sumWeight = 0;
   631→  let sumWeightedBull = 0;
   632→  for (const { signal, weight } of currentBullSignals) {
   633→    if (weight <= 0) continue;
   634→    sumWeightedBull += weight * signal;
   635→    sumWeight += weight;
   636→  }
   637→  const P_Bull_Current = sumWeight > 0 ? clamp(sumWeightedBull / sumWeight, 0, 1) : bull;
   638→  const P_Bear_Current = 1 - P_Bull_Current;
   639→
   640→  // ═══════════════════════════════════════════════════════════════════════
   641→  // STEP 2: P_Bull_Forecast (from ML prediction, or decayed current)
   642→  // ═══════════════════════════════════════════════════════════════════════
   643→  // If ML prediction available, use predicted return to estimate future bull prob
   644→  // Otherwise use exponential decay of current state
   645→  const alpha = 0.5; // weight of current state (per instructions)
   646→
   647→  // Use ML predicted return if available, otherwise use consensus as proxy
   648→  const mlReturn = mlResult?.predictionProb;
   649→  let P_Bull_Forecast: number;
   650→  if (mlReturn !== null && mlReturn !== undefined) {
   651→    // ML gives direct P(bull) prediction — use it as forecast
   652→    P_Bull_Forecast = clamp(mlReturn, 0, 1);
   653→  } else {
   654→    // Decay: forecast = current × decay + 0.5 × (1-decay)  (mean-revert to 0.5)
   655→    const decayFactor = 0.85;
   656→    P_Bull_Forecast = P_Bull_Current * decayFactor + 0.5 * (1 - decayFactor);
   657→  }
   658→  const P_Bear_Forecast = 1 - P_Bull_Forecast;
   659→
   660→  // ═══════════════════════════════════════════════════════════════════════
   661→  // STEP 3: Final directional probabilities (combining current + forecast)
   662→  // ═══════════════════════════════════════════════════════════════════════
   663→  const P_Bull_Final = clamp(alpha * P_Bull_Current + (1 - alpha) * P_Bull_Forecast, 0.02, 0.98);
   664→  const P_Bear_Final = clamp(alpha * P_Bear_Current + (1 - alpha) * P_Bear_Forecast, 0.02, 0.98);
   665→  const P_Neutral_Final = Math.max(0.04, 1 - P_Bull_Final - P_Bear_Final);
   666→
   667→  // Normalize so they sum to 1
   668→  const dirSum = P_Bull_Final + P_Bear_Final + P_Neutral_Final;
   669→  const pBull = P_Bull_Final / dirSum;
   670→  const pBear = P_Bear_Final / dirSum;
   671→  const pNeutral = P_Neutral_Final / dirSum;
   672→
   673→  // ═══════════════════════════════════════════════════════════════════════
   674→
   675→  // Helper: Overbought/Oversold
   676→  const overbought = rsi > 70 ? (rsi - 70) / 30 : 0;
   677→  const oversold = rsi < 30 ? (30 - rsi) / 30 : 0;
   678→
   679→  // Helper: Volatility state
   680→  const atrPct = price > 0 ? (atr || price * 0.02) / price : 0.02;
   681→  const volHigh = clamp((atrPct - 0.02) / 0.03, 0, 1);
   682→  const volLow = clamp(1 - (atrPct - 0.01) / 0.02, 0, 1);
   683→
   684→  // Helper: Divergence detection
   685→  let divBull = 0;
   686→  let divBear = 0;
   687→  if (mlResult?.predictionProb !== undefined) {
   688→    const mlBull = mlResult.predictionProb ?? 0.5;
   689→    if (bull > 0.6 && mlBull < 0.4) divBear = (bull - mlBull) * 1.5;
   690→    if (bull < 0.4 && mlBull > 0.6) divBull = (mlBull - bull) * 1.5;
   691→  }
   692→  divBull = clamp(divBull, 0, 0.3);
   693→  divBear = clamp(divBear, 0, 0.3);
   694→
   695→  // Helper: Volume confirmation
   696→  const volConfirm = hasVolume ? 0.6 : 0.4;
   697→
   698→  // Helper: ADX > 40 = strong trend
   699→  const adxStrong = adx > 40 ? 1 : adx > 25 ? (adx - 25) / 15 : 0;
   700→
   701→  // ── Conditional probability distributions CP(S_i | State) ──
   702→  // Per attachment instructions: 3-branch decision graph
   703→  //   Branch A: Trend Following → scenarios A1-A9
   704→  //   Branch B: Breakout Strategy → scenarios B1-B9
   705→  //   Branch C: Reversal Strategy → scenarios C1-C9
   706→  // Final: P(S_i) = P(Bull)×CP_i|Bull + P(Bear)×CP_i|Bear + P(Neutral)×CP_i|Neutral
   707→  // Each CP distribution sums to 1 after normalization.
   708→  //
   709→  // ADX governs range/trend split (per attachment: ADX < 20 = weak trend)
   710→  // FIX: Range scenarios capped so they never dominate (>40% of any CP dist)
   711→
   712→  // ADX-based trend strength (per attachment: ADX_Normalized = min(ADX/60, 1))
   713→  const adxNormVal = Math.min(adx / 60, 1);
   714→  // Range weight: inversely proportional to ADX, but capped at 0.35 (was 0.6!)
   715→  const adxRange = adx < 20 ? 0.35 : adx < 30 ? 0.20 : 0.10;
   716→  const adxTrend = 1 - adxRange;
   717→
   718→  // ── Per attachment edge formulas (lines 657-716): ──
   719→  // A1: Strong Bull  = P_Bull × P_ADX>40 × P_Forecast_Bull_Strong
   720→  // A2: Accel Bull   = P_Bull × P_Volatility_High × P_Volume_High_Future
   721→  // A3: Cautious Bull= P_Bull × P_Divergence_Bearish_Future
   722→  // A4: Strong Bear  = P_Bear × P_ADX>40 × P_Forecast_Bear_Strong
   723→  // A5: Accel Bear   = P_Bear × P_Volatility_High × P_Volume_High_Future
   724→  // A6: Cautious Bear= P_Bear × P_Divergence_Bullish_Future
   725→  // A7: LowVol Range = P_Neutral × P_Volatility_Low_Future
   726→  // A8: HighVol Range= P_Neutral × P_Volatility_High_Future
   727→  // A9: Shock        = P_Shock × P_Forecast_Anomaly (0.02-0.05)
   728→
   729→  // CP_i|Bull: Bull scenarios dominate, range scenarios limited
   730→  const cpBull = [
   731→    // SC1: Strong Bull — ADX strength and trend quality
   732→    0.28 * adxTrend * (0.5 + 0.5 * adxNormVal),
   733→    // SC2: Accelerating Bull — momentum + volatility
   734→    0.15 * adxTrend * (0.3 + 0.7 * factors.momentum),
   735→    // SC3: Cautious Bull — reduced by overbought risk
   736→    0.12 * (1 - overbought * 0.6) * adxTrend,
   737→    // SC4: Strong Bear — small probability in bull market, driven by divergence
   738→    0.03 * (1 + divBear),
   739→    // SC5: Accelerating Bear — very small in bull market
   740→    0.02,
   741→    // SC6: Cautious Bear — driven by divergence warning
   742→    0.03 * (1 + divBear),
   743→    // SC7: Low-Vol Range — bounded, not dominant
   744→    adxRange * 0.45 * volLow,
   745→    // SC8: High-Vol Range — bounded
   746→    adxRange * 0.30 * volHigh,
   747→    // SC9: Shock — fixed small probability
   748→    0.04 * (1 + overbought * 0.5 + volHigh * 0.5),
   749→  ];
   750→
   751→  // CP_i|Bear: Mirror of bull (bear scenarios dominate)
   752→  const cpBear = [
   753→    // SC1: Strong Bull — small in bear market
   754→    0.03 * (1 + divBull),
   755→    // SC2: Accelerating Bull — very small
   756→    0.02,
   757→    // SC3: Cautious Bull — divergence-driven
   758→    0.03 * (1 + divBull),
   759→    // SC4: Strong Bear — ADX strength
   760→    0.28 * adxTrend * (0.5 + 0.5 * adxNormVal),
   761→    // SC5: Accelerating Bear — momentum
   762→    0.15 * adxTrend * (0.3 + 0.7 * factors.momentum),
   763→    // SC6: Cautious Bear — reduced by oversold risk
   764→    0.12 * (1 - oversold * 0.6) * adxTrend,
   765→    // SC7: Low-Vol Range — bounded
   766→    adxRange * 0.45 * volLow,
   767→    // SC8: High-Vol Range — bounded
   768→    adxRange * 0.30 * volHigh,
   769→    // SC9: Shock
   770→    0.04 * (1 + oversold * 0.5 + volHigh * 0.5),
   771→  ];
   772→
   773→  // CP_i|Neutral: More balanced distribution per attachment
   774→  // Per attachment: neutral market has trend signals too, just weaker
   775→  // Range gets adxRange but trend scenarios get reasonable weight
   776→  const cpNeutral = [
   777→    // SC1: Mild Bull — smaller in neutral
   778→    0.08 * adxTrend * (0.5 + 0.5 * adxNormVal),
   779→    // SC2: Bull Acceleration
   780→    0.04 * adxTrend * (0.3 + 0.7 * factors.momentum),
   781→    // SC3: Cautious Bull
   782→    0.07 * adxTrend,
   783→    // SC4: Mild Bear
   784→    0.08 * adxTrend * (0.5 + 0.5 * adxNormVal),
   785→    // SC5: Bear Acceleration
   786→    0.04 * adxTrend * (0.3 + 0.7 * factors.momentum),
   787→    // SC6: Cautious Bear
   788→    0.07 * adxTrend,
   789→    // SC7: Low-Vol Range — adxRange weighted, split between vol states
   790→    adxRange * 0.50 * volLow,
   791→    // SC8: High-Vol Range
   792→    adxRange * 0.35 * volHigh,
   793→    // SC9: Shock — small fixed
   794→    0.04,
   795→  ];
   796→
   797→  const normCP = (cp: number[]): number[] => {
   798→    const s = cp.reduce((a, b) => a + b, 0);
   799→    return s > 0 ? cp.map(v => v / s) : cp.map(() => 1 / 9);
   800→  };
   801→  const ncpBull = normCP(cpBull);
   802→  const ncpBear = normCP(cpBear);
   803→  const ncpNeutral = normCP(cpNeutral);
   804→
   805→  // ── Final probabilities per instructions formula ──
   806→  // P(S_i) = P(Bull)*CP_i|Bull + P(Bear)*CP_i|Bear + P(Neutral)*CP_i|Neutral
   807→  const rawS = [0, 0, 0, 0, 0, 0, 0, 0, 0];
   808→  for (let i = 0; i < 9; i++) {
   809→    rawS[i] = pBull * ncpBull[i] + pBear * ncpBear[i] + pNeutral * ncpNeutral[i];
   810→  }
   811→
   812→  // Normalize to sum = 100
   813→  const totalRaw = rawS.reduce((a, b) => a + b, 0);
   814→  if (totalRaw <= 0) {
   815→    return {
   816→      pSC1: 12, pSC2: 10, pSC3: 8, pSC4: 12, pSC5: 10, pSC6: 8,
   817→      pSC7: 15, pSC8: 15, pSC9: 10, factors,
   818→    };
   819→  }
   820→
   821→  const pSC1 = Math.round(rawS[0] / totalRaw * 100);
   822→  const pSC2 = Math.round(rawS[1] / totalRaw * 100);
   823→  const pSC3 = Math.round(rawS[2] / totalRaw * 100);
   824→  const pSC4 = Math.round(rawS[3] / totalRaw * 100);
   825→  const pSC5 = Math.round(rawS[4] / totalRaw * 100);
   826→  const pSC6 = Math.round(rawS[5] / totalRaw * 100);
   827→  const pSC7 = Math.round(rawS[6] / totalRaw * 100);
   828→  const pSC8 = Math.round(rawS[7] / totalRaw * 100);
   829→  const pSC9 = Math.max(1, 100 - (pSC1 + pSC2 + pSC3 + pSC4 + pSC5 + pSC6 + pSC7 + pSC8));
   830→  return { pSC1, pSC2, pSC3, pSC4, pSC5, pSC6, pSC7, pSC8, pSC9, factors };
   831→}
   832→
   833→export function calculateEdgeWeights(
   834→  bullConsensus: number,
   835→  adx: number,
   836→  mlResult: AdaptiveModelResult | null,
   837→): { up: number; down: number; pullback: number; risk: number } {
   838→  // Default coefficients
   839→  let trendCoef = 1 / 3;
   840→  let momentumCoef = 1 / 3;
   841→  let volatilityCoef = 1 / 3;
   842→
   843→  if (mlResult?.isTrained && mlResult.coefficients.length === 16) {
   844→    const ec = mlResult.coefficients;
   845→    const absTrend = Math.abs(ec[11] ?? 0);
   846→    const absMomentum = Math.abs(ec[0] ?? 0);
   847→    const absVolatility = Math.abs(ec[10] ?? 0);
   848→    const total = absTrend + absMomentum + absVolatility;
   849→    if (total > 0) {
   850→      trendCoef = absTrend / total;
   851→      momentumCoef = absMomentum / total;
   852→      volatilityCoef = absVolatility / total;
   853→    }
   854→  }
   855→
   856→  // Dynamic base calculations
   857→  const up_base = bullConsensus * 0.7 + 0.15;
   858→  const down_base = (1 - bullConsensus) * 0.7 + 0.15;
   859→  const pullback_base = 0.25;
   860→  const risk_base = 0.12 * (1.2 - Math.min(adx / 100, 1));
   861→
   862→  // Apply ML-driven adjustments
   863→  const up = clamp(up_base * (1 + (momentumCoef - 1 / 3) * 0.3), 0.05, 0.95);
   864→  const down = clamp(down_base * (1 + (trendCoef - 1 / 3) * 0.3), 0.05, 0.95);
   865→  const pullback = clamp(pullback_base * (1 + (volatilityCoef - 1 / 3) * 0.2), 0.05, 0.50);
   866→  const risk = clamp(risk_base * (1 - (trendCoef - 1 / 3) * 0.5), 0.02, 0.30);
   867→
   868→  return { up, down, pullback, risk };
   869→}

   935→export interface PricePrediction {
   936→  predictedPrices: number[];
   937→  predictedReturn5d: number;
   938→  confidence: number;
   939→  trendDirection: 'up' | 'down' | 'flat';
   940→}
   941→
   942→export function predictPrices(data: OHLCV[]): PricePrediction | null {
   943→  if (data.length < 40) return null;
   944→  const PREDICT_DAYS = 5;
   945→  const closes = data.map(d => d.close);
   946→  const price = closes[closes.length - 1];
   947→  if (price <= 0) return null;
   948→
   949→  const LOOKBACK = 20;
   950→  const returns: number[] = [];
   951→  for (let i = closes.length - LOOKBACK; i < closes.length; i++) {
   952→    returns.push(i > 0 && closes[i - 1] > 0 ? (closes[i] - closes[i - 1]) / closes[i - 1] : 0);
   953→  }
   954→
   955→  const mom3 = returns.slice(-3).reduce((a, b) => a + b, 0) / 3;
   956→  const mom10 = returns.slice(-10).reduce((a, b) => a + b, 0) / 10;
   957→  const ma20 = sma(closes, 20);
   958→  const deviation = price > 0 && ma20 > 0 ? (price - ma20) / ma20 : 0;
   959→
   960→  const trendSlice = closes.slice(-20);
   961→  const tn = trendSlice.length;
   962→  let sx = 0, sy = 0, sxy = 0, sx2 = 0;
   963→  for (let i = 0; i < tn; i++) { sx += i; sy += trendSlice[i]; sxy += i * trendSlice[i]; sx2 += i * i; }
   964→  const d = tn * sx2 - sx * sx;
   965→  let slope = 0, r2 = 0;
   966→  if (d !== 0) {
   967→    slope = (tn * sxy - sx * sy) / d;
   968→    const meanY = sy / tn;
   969→    let ssRes = 0, ssTot = 0;
   970→    for (let i = 0; i < tn; i++) { ssRes += (trendSlice[i] - (slope * i + meanY - slope * (tn - 1) / 2)) ** 2; ssTot += (trendSlice[i] - meanY) ** 2; }
   971→    r2 = ssTot > 0 ? 1 - ssRes / ssTot : 0;
   972→  }
   973→
   974→  let atr = 0;
   975→  for (let i = Math.max(1, data.length - 14); i < data.length; i++) {
   976→    atr += Math.max(data[i].high - data[i].low, Math.abs(data[i].high - data[i - 1].close), Math.abs(data[i].low - data[i - 1].close));
   977→  }
   978→  atr /= Math.min(14, data.length - 1);
   979→  const dailyVol = price > 0 ? atr / price : 0.02;
   980→
   981→  const trendDaily = price > 0 ? slope / price : 0;
   982→  const meanRevDaily = -deviation * 0.05;
   983→  const momentumDaily = mom3 * 0.4 + mom10 * 0.1;
   984→  const trendW = r2 * 0.6, mrW = Math.min(Math.abs(deviation) * 5, 0.4), momW = 0.2;
   985→  const predDaily = trendDaily * trendW + meanRevDaily * mrW + momentumDaily * momW;
   986→
   987→  const predictedPrices: number[] = [];
   988→  let cp = price;
   989→  for (let dd = 1; dd <= PREDICT_DAYS; dd++) {
   990→    const decay = Math.exp(-0.1 * (dd - 1));
   991→    cp = cp * (1 + predDaily * decay);
   992→    predictedPrices.push(Math.round(cp));
   993→  }
   994→
   995→  const predictedReturn5d = (predictedPrices[PREDICT_DAYS - 1] - price) / price;
   996→  return {
   997→    predictedPrices,
   998→    predictedReturn5d,
   999→    confidence: Math.min(0.9, Math.max(0.1, r2 * 0.7 + 0.2)),
  1000→    trendDirection: predictedReturn5d > 0.005 ? 'up' : predictedReturn5d < -0.005 ? 'down' : 'flat',
  1001→  };
  1002→}

  1004→export function evaluatePredictedIndicators(data: OHLCV[], prediction: PricePrediction): { mlScore: number; predictedBullConsensus: number } {
  1005→  const price = data[data.length - 1].close;
  1006→  if (price <= 0 || prediction.predictedPrices.length === 0) return { mlScore: 0.5, predictedBullConsensus: 0.5 };
  1007→  const avgP = prediction.predictedPrices.reduce((a, b) => a + b, 0) / prediction.predictedPrices.length;
  1008→  const closes = data.map(d => d.close);
  1009→  const ma21 = sma(closes, 21), ma100 = sma(closes, 100), rsi = calcRSI(closes);
  1010→  const rsiP = Math.max(0, Math.min(1, (rsi + (avgP > price ? 5 : -5)) / 100));
  1011→  const ma21S = ma21 > 0 ? (avgP > ma21 ? 0.7 : 0.3) : 0.5;
  1012→  const ma100S = ma100 > 0 ? (avgP > ma100 ? 0.65 : 0.35) : 0.5;
  1013→  const dirS = avgP > price ? 0.65 : 0.35;
  1014→  return {
  1015→    mlScore: Math.max(0, Math.min(1, 0.5 + prediction.predictedReturn5d * 10 * prediction.confidence)),
  1016→    predictedBullConsensus: Math.max(0, Math.min(1, rsiP * 0.3 + ma21S * 0.25 + ma100S * 0.2 + dirS * 0.25)),
  1017→  };
  1018→}
