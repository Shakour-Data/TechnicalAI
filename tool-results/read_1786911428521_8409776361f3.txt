    84→    84→// ─── Helper: SMA ──────────────────────────────────────────────────────────────
    85→    85→function sma(closes: number[], period: number): number {
    86→    86→  if (closes.length < period) return 0;
    87→    87→  const slice = closes.slice(-period);
    88→    88→  return slice.reduce((a, b) => a + b, 0) / period;
    89→    89→}

    91→    91→// ─── Helper: EMA ──────────────────────────────────────────────────────────────
    92→    92→function emaCalc(closes: number[], period: number): number {
    93→    93→  if (closes.length < period) return 0;
    94→    94→  const k = 2 / (period + 1);
    95→    95→  let emaVal = sma(closes.slice(0, period), period);
    96→    96→  for (let i = period; i < closes.length; i++) {
    97→    97→    emaVal = closes[i] * k + emaVal * (1 - k);
    98→    98→  }
    99→    99→  return emaVal;
   100→   100→}

   102→   102→// ─── Helper: RSI(14) ──────────────────────────────────────────────────────────
   103→   103→function calcRSI(closes: number[], period: number = 14): number {
   104→   104→  if (closes.length < period + 1) return 50;
   105→   105→  let avgGain = 0;
   106→   106→  let avgLoss = 0;
   107→   107→  for (let i = 1; i <= period; i++) {
   108→   108→    const diff = closes[i] - closes[i - 1];
   109→   109→    if (diff > 0) avgGain += diff;
   110→   110→    else avgLoss += Math.abs(diff);
   111→   111→  }
   112→   112→  avgGain /= period;
   113→   113→  avgLoss /= period;
   114→   114→  for (let i = period + 1; i < closes.length; i++) {
   115→   115→    const diff = closes[i] - closes[i - 1];
   116→   116→    const gain = diff > 0 ? diff : 0;
   117→   117→    const loss = diff < 0 ? Math.abs(diff) : 0;
   118→   118→    avgGain = (avgGain * (period - 1) + gain) / period;
   119→   119→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   120→   120→  }
   121→   121→  if (avgLoss === 0) return 100;
   122→   122→  const rs = avgGain / avgLoss;
   123→   123→  return 100 - 100 / (1 + rs);
   124→   124→}

   126→   126→// ─── Helper: MFI(14) ──────────────────────────────────────────────────────────
   127→   127→function calcMFI(data: OHLCV[], period: number = 14): number {
   128→   128→  if (data.length < period + 1) return 50;
   129→   129→  let posFlow = 0;
   130→   130→  let negFlow = 0;
   131→   131→  for (let i = data.length - period; i < data.length; i++) {
   132→   132→    const tp = (data[i].high + data[i].low + data[i].close) / 3;
   133→   133→    const mf = tp * data[i].volume;
   134→   134→    const prevTp = (data[i - 1].high + data[i - 1].low + data[i - 1].close) / 3;
   135→   135→    if (tp > prevTp) posFlow += mf;
   136→   136→    else negFlow += mf;
   137→   137→  }
   138→   138→  if (negFlow === 0) return 100;
   139→   139→  const mfr = posFlow / negFlow;
   140→   140→  return 100 - 100 / (1 + mfr);
   141→   141→}

   144→   144→function calcCCI(data: OHLCV[], period: number = 20): number {
   145→   145→  if (data.length < period) return 0;
   146→   146→  const tps: number[] = [];
   147→   147→  for (let i = data.length - period; i < data.length; i++) {
   148→   148→    tps.push((data[i].high + data[i].low + data[i].close) / 3);
   149→   149→  }
   150→   150→  const meanTP = tps.reduce((a, b) => a + b, 0) / period;
   151→   151→  const meanDev = tps.reduce((a, b) => a + Math.abs(b - meanTP), 0) / period;
   152→   152→  if (meanDev === 0) return 0;
   153→   153→  return (tps[tps.length - 1] - meanTP) / (0.015 * meanDev);
   154→   154→}

   157→   157→function calcStochastic(data: OHLCV[], kPeriod: number = 14, smoothK: number = 3, smoothD: number = 3): { k: number; d: number } {
   158→   158→  if (data.length < kPeriod) return { k: 50, d: 50 };
   159→   159→  const rawKs: number[] = [];
   160→   160→  for (let i = kPeriod - 1; i < data.length; i++) {
   161→   161→    let lowest = Infinity;
   162→   162→    let highest = -Infinity;
   163→   163→    for (let j = i - kPeriod + 1; j <= i; j++) {
   164→   164→      if (data[j].low < lowest) lowest = data[j].low;
   165→   165→      if (data[j].high > highest) highest = data[j].high;
   166→   166→    }
   167→   167→    const range = highest - lowest;
   168→   168→    rawKs.push(range === 0 ? 50 : ((data[i].close - lowest) / range) * 100);
   169→   169→  }
   170→   170→  const smoothedK: number[] = [];
   171→   171→  for (let i = smoothK - 1; i < rawKs.length; i++) {
   172→   172→    const slice = rawKs.slice(i - smoothK + 1, i + 1);
   173→   173→    smoothedK.push(slice.reduce((a, b) => a + b, 0) / smoothK);
   174→   174→  }
   175→   175→  const k = smoothedK[smoothedK.length - 1] ?? 50;
   176→   176→  const dSlice = smoothedK.slice(-smoothD);
   177→   177→  const d = dSlice.reduce((a, b) => a + b, 0) / dSlice.length;
   178→   178→  return { k, d };
   179→   179→}

   182→   182→function calcWilliamsR(data: OHLCV[], period: number = 14): number {
   183→   183→  if (data.length < period) return -50;
   184→   184→  const slice = data.slice(-period);
   185→   185→  let highest = -Infinity;
   186→   186→  let lowest = Infinity;
   187→   187→  for (const c of slice) {
   188→   188→    if (c.high > highest) highest = c.high;
   189→   189→    if (c.low < lowest) lowest = c.low;
   190→   190→  }
   191→   191→  const range = highest - lowest;
   192→   192→  if (range === 0) return -50;
   193→   193→  return ((highest - data[data.length - 1].close) / range) * -100;
   194→   194→}

   196→   196→// ─── Helper: MACD(12,26,9) ────────────────────────────────────────────────────
   197→   197→function calcMACD(closes: number[], fast: number = 12, slow: number = 26, sig: number = 9): { line: number; signal: number; histogram: number } {
   198→   198→  if (closes.length < slow + sig) return { line: 0, signal: 0, histogram: 0 };
   199→   199→  const emaFast = emaCalc(closes, fast);
   200→   200→  const emaSlow = emaCalc(closes, slow);
   201→   201→  const macdLine = emaFast - emaSlow;
   202→   202→  // Calculate signal as EMA of MACD values
   203→   203→  const macdValues: number[] = [];
   204→   204→  const kFast = 2 / (fast + 1);
   205→   205→  const kSlow = 2 / (slow + 1);
   206→   206→  let eFast = sma(closes.slice(0, fast), fast);
   207→   207→  let eSlow = sma(closes.slice(0, slow), slow);
   208→   208→  for (let i = Math.max(fast, slow); i < closes.length; i++) {
   209→   209→    eFast = closes[i] * kFast + eFast * (1 - kFast);
   210→   210→    eSlow = closes[i] * kSlow + eSlow * (1 - kSlow);
   211→   211→    macdValues.push(eFast - eSlow);
   212→   212→  }
   213→   213→  if (macdValues.length < sig) return { line: macdLine, signal: 0, histogram: macdLine };
   214→   214→  const kSig = 2 / (sig + 1);
   215→   215→  let signalVal = sma(macdValues.slice(0, sig), sig);
   216→   216→  for (let i = sig; i < macdValues.length; i++) {
   217→   217→    signalVal = macdValues[i] * kSig + signalVal * (1 - kSig);
   218→   218→  }
   219→   219→  return { line: macdLine, signal: signalVal, histogram: macdLine - signalVal };
   220→   220→}

   222→   222→// ─── Helper: ATR(14) ──────────────────────────────────────────────────────────
   223→   223→function calcATR(data: OHLCV[], period: number = 14): number {
   224→   224→  if (data.length < 2) return 0;
   225→   225→  const trs: number[] = [];
   226→   226→  for (let i = 1; i < data.length; i++) {
   227→   227→    const tr = Math.max(
   228→   228→      data[i].high - data[i].low,
   229→   229→      Math.abs(data[i].high - data[i - 1].close),
   230→   230→      Math.abs(data[i].low - data[i - 1].close)
   231→   231→    );
   232→   232→    trs.push(tr);
   233→   233→  }
   234→   234→  if (trs.length < period) return trs.reduce((a, b) => a + b, 0) / trs.length;
   235→   235→  let atrVal = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
   236→   236→  for (let i = period; i < trs.length; i++) {
   237→   237→    atrVal = (atrVal * (period - 1) + trs[i]) / period;
   238→   238→  }
   239→   239→  return atrVal;
   240→   240→}

   242→   242→// ─── Helper: ADX(14) + DI+/DI- ────────────────────────────────────────────────
   243→   243→function calcADX(data: OHLCV[], period: number = 14): { adx: number; diPlus: number; diMinus: number } {
   244→   244→  if (data.length < period + 1) return { adx: 0, diPlus: 0, diMinus: 0 };
   245→   245→  const plusDMs: number[] = [];
   246→   246→  const minusDMs: number[] = [];
   247→   247→  const trs: number[] = [];
   248→   248→  for (let i = 1; i < data.length; i++) {
   249→   249→    const upMove = data[i].high - data[i - 1].high;
   250→   250→    const downMove = data[i - 1].low - data[i].low;
   251→   251→    plusDMs.push(upMove > downMove && upMove > 0 ? upMove : 0);
   252→   252→    minusDMs.push(downMove > upMove && downMove > 0 ? downMove : 0);
   253→   253→    trs.push(Math.max(
   254→   254→      data[i].high - data[i].low,
   255→   255→      Math.abs(data[i].high - data[i - 1].close),
   256→   256→      Math.abs(data[i].low - data[i - 1].close)
   257→   257→    ));
   258→   258→  }
   259→   259→  if (trs.length < period) return { adx: 0, diPlus: 0, diMinus: 0 };
   260→   260→  let smoothTR = trs.slice(0, period).reduce((a, b) => a + b, 0);
   261→   261→  let smoothPlusDM = plusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   262→   262→  let smoothMinusDM = minusDMs.slice(0, period).reduce((a, b) => a + b, 0);
   263→   263→  const dxValues: number[] = [];
   264→   264→  for (let i = period; i < trs.length; i++) {
   265→   265→    smoothTR = smoothTR - smoothTR / period + trs[i];
   266→   266→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMs[i];
   267→   267→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMs[i];
   268→   268→    const diPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   269→   269→    const diMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   270→   270→    const diSum = diPlus + diMinus;
   271→   271→    const dx = diSum === 0 ? 0 : (Math.abs(diPlus - diMinus) / diSum) * 100;
   272→   272→    dxValues.push(dx);
   273→   273→  }
   274→   274→  if (dxValues.length === 0) return { adx: 0, diPlus: 0, diMinus: 0 };
   275→   275→  let adxVal = dxValues.slice(0, period).reduce((a, b) => a + b, 0) / Math.min(period, dxValues.length);
   276→   276→  for (let i = period; i < dxValues.length; i++) {
   277→   277→    adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   278→   278→  }
   279→   279→  const lastDIPlus = smoothTR === 0 ? 0 : (smoothPlusDM / smoothTR) * 100;
   280→   280→  const lastDIMinus = smoothTR === 0 ? 0 : (smoothMinusDM / smoothTR) * 100;
   281→   281→  return { adx: adxVal, diPlus: lastDIPlus, diMinus: lastDIMinus };
   282→   282→}

   284→   284→// ─── Helper: Parabolic SAR ────────────────────────────────────────────────────
   285→   285→function calcSAR(data: OHLCV[], step: number = 0.02, max: number = 0.2): number {
   286→   286→  if (data.length < 5) return data[data.length - 1]?.low ?? 0;
   287→   287→  let isLong = true;
   288→   288→  let af = step;
   289→   289→  let ep = data[0].high;
   290→   290→  let sar = data[0].low;
   291→   291→  for (let i = 1; i < data.length; i++) {
   292→   292→    const prevSar = sar;
   293→   293→    sar = prevSar + af * (ep - prevSar);
   294→   294→    // Ensure SAR is within bounds
   295→   295→    if (isLong) {
   296→   296→      if (i >= 2) sar = Math.min(sar, data[i - 1].low, data[i - 2]?.low ?? Infinity);
   297→   297→      if (data[i].low < sar) {
   298→   298→        isLong = false;
   299→   299→        sar = ep;
   300→   300→        ep = data[i].low;
   301→   301→        af = step;
   302→   302→      } else {
   303→   303→        if (data[i].high > ep) {
   304→   304→          ep = data[i].high;
   305→   305→          af = Math.min(af + step, max);
   306→   306→        }
   307→   307→      }
   308→   308→    } else {
   309→   309→      if (i >= 2) sar = Math.max(sar, data[i - 1].high, data[i - 2]?.high ?? -Infinity);
   310→   310→      if (data[i].high > sar) {
   311→   311→        isLong = true;
   312→   312→        sar = ep;
   313→   313→        ep = data[i].high;
   314→   314→        af = step;
   315→   315→      } else {
   316→   316→        if (data[i].low < ep) {
   317→   317→          ep = data[i].low;
   318→   318→          af = Math.min(af + step, max);
   319→   319→        }
   320→   320→      }
   321→   321→    }
   322→   322→  }
   323→   323→  return sar;
   324→   324→}

   326→   326→// ─── Helper: Bollinger Bands(20,2) ────────────────────────────────────────────
   327→   327→function calcBollingerBands(closes: number[], period: number = 20, stdDev: number = 2): { upper: number; middle: number; lower: number } {
   328→   328→  if (closes.length < period) {
   329→   329→    const c = closes[closes.length - 1] ?? 0;
   330→   330→    return { upper: c, middle: c, lower: c };
   331→   331→  }
   332→   332→  const slice = closes.slice(-period);
   333→   333→  const middle = slice.reduce((a, b) => a + b, 0) / period;
   334→   334→  const variance = slice.reduce((a, b) => a + (b - middle) ** 2, 0) / period;
   335→   335→  const sd = Math.sqrt(variance);
   336→   336→  return { upper: middle + stdDev * sd, lower: middle - stdDev * sd, middle };
   337→   337→}

   340→   340→function calcOBV(data: OHLCV[]): number {
   341→   341→  let obv = 0;
   342→   342→  for (let i = 1; i < data.length; i++) {
   343→   343→    if (data[i].close > data[i - 1].close) obv += data[i].volume;
   344→   344→    else if (data[i].close < data[i - 1].close) obv -= data[i].volume;
   345→   345→  }
   346→   346→  return obv;
   347→   347→}

   349→   349→// ─── Helper: Linear Regression ────────────────────────────────────────────────
   350→   350→function linearRegression(values: number[]): { slope: number; intercept: number; r2: number } {
   351→   351→  const n = values.length;
   352→   352→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0, r2: 0 };
   353→   353→  let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
   354→   354→  for (let i = 0; i < n; i++) {
   355→   355→    sumX += i;
   356→   356→    sumY += values[i];
   357→   357→    sumXY += i * values[i];
   358→   358→    sumX2 += i * i;
   359→   359→    sumY2 += values[i] * values[i];
   360→   360→  }
   361→   361→  const denom = n * sumX2 - sumX * sumX;
   362→   362→  if (denom === 0) return { slope: 0, intercept: sumY / n, r2: 0 };
   363→   363→  const slope = (n * sumXY - sumX * sumY) / denom;
   364→   364→  const intercept = (sumY - slope * sumX) / n;
   365→   365→  const meanY = sumY / n;
   366→   366→  let ssRes = 0, ssTot = 0;
   367→   367→  for (let i = 0; i < n; i++) {
   368→   368→    const predicted = slope * i + intercept;
   369→   369→    ssRes += (values[i] - predicted) ** 2;
   370→   370→    ssTot += (values[i] - meanY) ** 2;
   371→   371→  }
   372→   372→  const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;
   373→   373→  return { slope, intercept, r2 };
   374→   374→}

   377→   377→function roundToNice(price: number): number {
   378→   378→  const abs = Math.abs(price);
   379→   379→  let step: number;
   380→   380→  if (abs >= 100000) step = 100;
   381→   381→  else if (abs >= 10000) step = 50;
   382→   382→  else if (abs >= 1000) step = 10;
   383→   383→  else if (abs >= 100) step = 5;
   384→   384→  else step = 1;
   385→   385→  return Math.round(price / step) * step;
   386→   386→}

   388→   388→// ─── Helper: Swing Highs/Lows ─────────────────────────────────────────────────
   389→   389→function findSwingLevels(data: OHLCV[]): { highs: number[]; lows: number[] } {
   390→   390→  const highs: number[] = [];
   391→   391→  const lows: number[] = [];
   392→   392→  const lookback = 3;
   393→   393→  for (let i = lookback; i < data.length - lookback; i++) {
   394→   394→    let isHigh = true;
   395→   395→    let isLow = true;
   396→   396→    for (let j = i - lookback; j <= i + lookback; j++) {
   397→   397→      if (j === i) continue;
   398→   398→      if (data[j].high >= data[i].high) isHigh = false;
   399→   399→      if (data[j].low <= data[i].low) isLow = false;
   400→   400→    }
   401→   401→    if (isHigh) highs.push(data[i].high);
   402→   402→    if (isLow) lows.push(data[i].low);
   403→   403→  }
   404→   404→  return { highs, lows };
   405→   405→}

   407→   407→// ─── Support / Resistance (5 each, rounded, with strength) ──────────────────
   408→   408→function calcSupportResistance(data: OHLCV[], currentPrice: number): {
   409→   409→  resistances: number[];
   410→   410→  supports: number[];
   411→   411→  supportStrengths: LevelStrength[];
   412→   412→  resistanceStrengths: LevelStrength[];
   413→   413→  priceTargets: LevelStrength[];
   414→   414→} {
   415→   415→  const allRawLevels: number[] = [];
   416→   416→
   417→   417→  // 1. Pivot Points from previous day
   418→   418→  const prev = data[data.length - 2] ?? data[data.length - 1];
   419→   419→  if (prev) {
   420→   420→    const pp = (prev.high + prev.low + prev.close) / 3;
   421→   421→    const hl = prev.high - prev.low;
   422→   422→    const pivotR = [
   423→   423→      2 * pp - prev.low,
   424→   424→      pp + hl,
   425→   425→      prev.high + 2 * (pp - prev.low),
   426→   426→      prev.high + 2 * (pp - prev.low) + hl,
   427→   427→      prev.high + 2 * (pp - prev.low) + hl + (pp - prev.low),
   428→   428→    ];
   429→   429→    const pivotS = [
   430→   430→      2 * pp - prev.high,
   431→   431→      pp - hl,
   432→   432→      prev.low - 2 * (prev.high - pp),
   433→   433→      prev.low - 2 * (prev.high - pp) - hl,
   434→   434→      prev.low - 2 * (prev.high - pp) - hl - (prev.high - pp),
   435→   435→    ];
   436→   436→    allRawLevels.push(...pivotR, ...pivotS);
   437→   437→  }
   438→   438→
   439→   439→  // 2. Swing levels
   440→   440→  const swings = findSwingLevels(data);
   441→   441→  allRawLevels.push(...swings.highs, ...swings.lows);
   442→   442→
   443→   443→  // 3. Round all and filter invalid
   444→   444→  const rounded = allRawLevels.map(roundToNice).filter(l => l > 0);
   445→   445→
   446→   446→  // 4. Remove duplicates (within 0.5%)
   447→   447→  const unique: number[] = [];
   448→   448→  for (const level of rounded) {
   449→   449→    const isDup = unique.some(u => Math.abs(u - level) / u < 0.005);
   450→   450→    if (!isDup) unique.push(level);
   451→   451→  }
   452→   452→
   453→   453→  // 5. Calculate strength for each unique level
   454→   454→  // Strength = count of other levels within ±5% range
   455→   455→  // More levels nearby = stronger
   456→   456→  const levelStrengths: LevelStrength[] = unique.map(level => {
   457→   457→    const range5pct = level * 0.05;
   458→   458→    const nearbyCount = unique.filter(other => other !== level && Math.abs(other - level) <= range5pct).length;
   459→   459→    // Base strength: 1 for level itself, +1 for each nearby level (max 10)
   460→   460→    const strength = Math.min(10, Math.max(1, 1 + nearbyCount * 2));
   461→   461→    return { price: level, strength, isTarget: false };
   462→   462→  });
   463→   463→
   464→   464→  // 6. Separate above/below and enforce 5% min gap
   465→   465→  const above = levelStrengths.filter(l => l.price > currentPrice).sort((a, b) => a.price - b.price);
   466→   466→  const below = levelStrengths.filter(l => l.price < currentPrice).sort((a, b) => b.price - a.price);
   467→   467→
   468→   468→  // Enforce 5% minimum gap between adjacent levels
   469→   469→  function enforceGap(levels: LevelStrength[], minGapPct: number = 0.05): LevelStrength[] {
   470→   470→    if (levels.length === 0) return [];
   471→   471→    const result: LevelStrength[] = [levels[0]];
   472→   472→    for (let i = 1; i < levels.length; i++) {
   473→   473→      const prev = result[result.length - 1].price;
   474→   474→      const curr = levels[i].price;
   475→   475→      const gap = Math.abs(curr - prev) / prev;
   476→   476→      if (gap >= minGapPct) {
   477→   477→        result.push(levels[i]);
   478→   478→      } else {
   479→   479→        // Merge: keep the one with higher strength
   480→   480→        if (levels[i].strength > result[result.length - 1].strength) {
   481→   481→          result[result.length - 1] = { ...levels[i], strength: Math.min(10, result[result.length - 1].strength + 1) };
   482→   482→        } else {
   483→   483→          result[result.length - 1] = { ...result[result.length - 1], strength: Math.min(10, result[result.length - 1].strength + 1) };
   484→   484→        }
   485→   485→      }
   486→   486→    }
   487→   487→    return result;
   488→   488→  }
   489→   489→
   490→   490→  const finalResistances = enforceGap(above, 0.05).slice(0, 5);
   491→   491→  const finalSupports = enforceGap(below, 0.05).slice(0, 5);
   492→   492→
   493→   493→  // 7. Determine price targets: strength > 7 is a target
   494→   494→  // If none > 7, pick top 2 strongest
   495→   495→  const allLevels = [...finalResistances, ...finalSupports];
   496→   496→  const sortedByStrength = [...allLevels].sort((a, b) => b.strength - a.strength);
   497→   497→  let targets: LevelStrength[];
   498→   498→  if (sortedByStrength.some(l => l.strength > 7)) {
   499→   499→    targets = sortedByStrength.filter(l => l.strength > 7);
   500→   500→  } else {
   501→   501→    targets = sortedByStrength.slice(0, 2);
   502→   502→  }
   503→   503→
   504→   504→  // Mark targets
   505→   505→  targets = targets.map(t => ({ ...t, isTarget: true }));
   506→   506→
   507→   507→  // Also mark in final arrays
   508→   508→  finalResistances.forEach(r => {
   509→   509→    if (targets.some(t => t.price === r.price)) r.isTarget = true;
   510→   510→  });
   511→   511→  finalSupports.forEach(s => {
   512→   512→    if (targets.some(t => t.price === s.price)) s.isTarget = true;
   513→   513→  });
   514→   514→
   515→   515→  return {
   516→   516→    resistances: finalResistances.map(l => l.price),
   517→   517→    supports: finalSupports.map(l => l.price),
   518→   518→    supportStrengths: finalSupports,
   519→   519→    resistanceStrengths: finalResistances,
   520→   520→    priceTargets: targets,
   521→   521→  };
   522→   522→}

   525→   525→function calcTrend(closes: number[], period: number): TrendResult {
   526→   526→  const slice = closes.slice(-Math.min(period, closes.length));
   527→   527→  const { slope, r2 } = linearRegression(slice);
   528→   528→  const avgPrice = slice.reduce((a, b) => a + b, 0) / slice.length;
   529→   529→  const angle = avgPrice === 0 ? 0 : Math.atan(slope / avgPrice) * (180 / Math.PI);
   530→   530→  const priceThreshold = avgPrice * 0.001; // 0.1%
   531→   531→  const direction = slope > priceThreshold ? 'up' : slope < -priceThreshold ? 'down' : 'flat';
   532→   532→  return { direction, slope, angle, r2: Math.max(0, Math.min(1, r2)) };
   533→   533→}
